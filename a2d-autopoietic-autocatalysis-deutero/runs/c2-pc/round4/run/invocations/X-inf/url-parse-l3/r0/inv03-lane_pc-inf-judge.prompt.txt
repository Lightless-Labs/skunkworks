fn is_ascii_letter(b: u8) -> bool {
    b.is_ascii_alphabetic()
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: offsets.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn scan_global(input: &str) -> Result<(), ParseError> {
    for (i, &b) in input.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn map_codec_parse(err: CodecError, base: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

fn validate_component(text: &str, base: usize, component: Component) -> Result<(), ParseError> {
    decode_component(component, text)
        .map(|_| ())
        .map_err(|e| map_codec_parse(e, base))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_global(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !is_ascii_letter(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let mut auth_end = input.len();
    for (i, &b) in bytes.iter().enumerate().skip(auth_start) {
        if matches!(b, b'/' | b'?' | b'#') {
            auth_end = i;
            break;
        }
    }
    let authority = &input[auth_start..auth_end];
    let auth_bytes = authority.as_bytes();

    let ats: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + ats[1],
        });
    }

    let (userinfo, host_start_rel) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };
    let host_abs_start = auth_start + host_start_rel;
    let hostport = &authority[host_start_rel..];

    let (host, port, ipv6) = if hostport.as_bytes().first() == Some(&b'[') {
        let close = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: host_abs_start }),
        };
        let interior = &hostport[1..close];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_abs_start });
        }
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(&suffix[1..], host_abs_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_abs_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport.as_bytes().iter().position(|&b| b == b':');
        let (host_text, port_text) = match split {
            Some(i) => (&hostport[..i], Some((&hostport[i + 1..], i))),
            None => (hostport, None),
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_abs_start });
        }
        for (i, &b) in host_text.as_bytes().iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_abs_start + i,
                });
            }
        }
        let port = match port_text {
            Some((p, split_at)) => Some(parse_port(p, host_abs_start + split_at + 1)?),
            None => None,
        };
        (host_text.to_ascii_lowercase(), port, false)
    };

    let fragment_mark = bytes[auth_end..].iter().position(|&b| b == b'#')
        .map(|i| auth_end + i);
    let before_fragment_end = fragment_mark.unwrap_or(input.len());
    let query_mark = bytes[auth_end..before_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|i| auth_end + i);

    let path_end = query_mark.unwrap_or(before_fragment_end);
    let path = if auth_end == path_end {
        "/".to_owned()
    } else {
        input[auth_end..path_end].to_owned()
    };
    let query = query_mark.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_mark.map(|f| input[f + 1..].to_owned());

    if let Some(ui) = &userinfo {
        validate_component(ui, auth_start, Component::UserInfo)?;
    }
    if auth_end != path_end {
        validate_component(&path, auth_end, Component::Path)?;
    }
    if let Some(q) = query_mark {
        validate_component(&input[q + 1..before_fragment_end], q + 1, Component::Query)?;
    }
    if let Some(f) = fragment_mark {
        validate_component(&input[f + 1..], f + 1, Component::Fragment)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(text: &str, at: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    for (i, b) in text.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + i });
        }
    }
    if text.len() >= 2 && text.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }
    let mut value: u32 = 0;
    for b in text.bytes() {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u32);
        if value > 65535 {
            return Err(ParseError::InvalidPort { at });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(value as u16)
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<String> = Vec::new();
    let mut terminal_navigation = false;

    for (i, segment) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).expect("path was validated");
        if decoded == "." {
            terminal_navigation = i + 1 == raw_segments.len();
        } else if decoded == ".." {
            if !stack.is_empty() {
                stack.pop();
            }
            terminal_navigation = i + 1 == raw_segments.len();
        } else {
            stack.push((*segment).to_owned());
            terminal_navigation = false;
        }
    }

    if terminal_navigation {
        stack.push(String::new());
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn map_path_error(err: PathError) -> ResolveError {
    ResolveError::Path(err)
}

fn normalize_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(map_path_error)?;
    Ok(url)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalize_url_path(base_url)?;

    scan_global(reference).map_err(ResolveError::Reference)?;

    let bytes = reference.as_bytes();
    let first_delim = bytes
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());
    if bytes[..first_delim].contains(&b':') {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    let fragment_mark = reference.find('#');
    let before_fragment_end = fragment_mark.unwrap_or(reference.len());
    let query_mark = reference[..before_fragment_end].find('?');
    let path_end = query_mark.unwrap_or(before_fragment_end);
    let ref_path = &reference[..path_end];
    let ref_query = query_mark.map(|q| reference[q + 1..before_fragment_end].to_owned());
    let ref_fragment = fragment_mark.map(|f| reference[f + 1..].to_owned());

    let (selected_path, selected_query) = if ref_path.is_empty() {
        let query = if query_mark.is_some() {
            ref_query
        } else {
            base_url.query.clone()
        };
        (base_url.path.clone(), query)
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_url.path[..slash], ref_path)
        };
        (merged, ref_query)
    };

    let authority = match &base_url.userinfo {
        Some(ui) => format!("{}@", ui),
        None => String::new(),
    };
    let host = if base_url.ipv6 {
        format!("[{}]", base_url.host)
    } else {
        base_url.host.clone()
    };
    let port = base_url.port.map(|p| format!(":{}", p)).unwrap_or_default();
    let query = selected_query
        .map(|q| format!("?{}", q))
        .unwrap_or_default();
    let fragment = ref_fragment
        .map(|f| format!("#{}", f))
        .unwrap_or_default();
    let constructed = format!(
        "{}://{}{}{}{}{}{}",
        base_url.scheme, authority, host, port, selected_path, query, fragment
    );

    let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalize_url_path(url)
}

fn is_ascii_letter(b: u8) -> bool {

fn scan_input(input: &str) -> Result<(), ParseError> {
    for (at, &b) in input.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_bytes(input: &str) -> Result<(Vec<u8>, Vec<usize>), CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let Some(hi) = hex_value(bytes[i + 1]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            let Some(lo) = hex_value(bytes[i + 2]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            decoded.push((hi << 4) | lo);
            offsets.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    Ok((decoded, offsets))
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let (decoded, offsets) = decode_bytes(input)?;
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let decoded_at = e.valid_up_to();
            let original_at = offsets.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };

        if safe {
            out.push(char::from(b));
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(char::from(HEX[(b >> 4) as usize]));
            out.push(char::from(HEX[(b & 0x0f) as usize]));
        }
    }
    out
}

fn parse_port(port: &str, at: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }

    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + i });
        }
    }

    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }

    let mut value: u32 = 0;
    for b in port.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at })?;
    }

    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at });
    }

    Ok(value as u16)
}

fn validate_component(
    component: Component,
    text: &str,
    input_offset: usize,
) -> Result<(), ParseError> {
    match decode_component(component, text) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => {
            Err(ParseError::BadEscape { at: input_offset + at })
        }
        Err(CodecError::InvalidUtf8 { at }) => {
            Err(ParseError::InvalidUtf8 { at: input_offset + at })
        }
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_input(input)?;

    let bytes = input.as_bytes();
    let colon = bytes.iter().position(|&b| b == b':');
    let Some(colon) = colon else {
        return Err(ParseError::InvalidScheme { at: input.len() });
    };

    if colon == 0 || !is_ascii_letter(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }

    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let rest = &input[authority_start..];
    let authority_len = rest
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(rest.len());
    let authority_end = authority_start + authority_len;
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();

    if at_positions.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport, hostport_start) = if let Some(&at_pos) = at_positions.first() {
        if at_pos == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at_pos,
            });
        }
        (
            Some(authority[..at_pos].to_owned()),
            &authority[at_pos + 1..],
            authority_start + at_pos + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    let (host, port, ipv6) = if hostport.starts_with('[') {
        let close = hostport.as_bytes()[1..]
            .iter()
            .position(|&b| b == b']')
            .map(|i| i + 1)
            .ok_or(ParseError::InvalidIpv6 { at: hostport_start })?;
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        let suffix = &hostport[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), parsed_port, true)
    } else {
        let split = hostport.find(':');
        let (host_text, port_text) = match split {
            Some(i) => (&hostport[..i], Some((&hostport[i + 1..], i))),
            None => (hostport, None),
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in host_text.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + i,
                });
            }
        }

        let parsed_port = match port_text {
            Some((p, colon_pos)) => Some(parse_port(p, hostport_start + colon_pos + 1)?),
            None => None,
        };
        (host_text.to_ascii_lowercase(), parsed_port, false)
    };

    let after_authority = &input[authority_end..];
    let fragment_rel = after_authority.find('#');
    let before_fragment = fragment_rel
        .map(|i| &after_authority[..i])
        .unwrap_or(after_authority);
    let fragment = fragment_rel.map(|i| after_authority[i + 1..].to_owned());

    let query_rel = before_fragment.find('?');
    let path_raw = query_rel
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_rel.map(|i| before_fragment[i + 1..].to_owned());
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    if let Some(ref u) = userinfo {
        validate_component(Component::UserInfo, u, authority_start)?;
    }
    if !path_raw.is_empty() {
        validate_component(Component::Path, path_raw, authority_end)?;
    }
    if let Some(q) = query_rel {
        validate_component(Component::Query, &before_fragment[q + 1..], authority_end + q + 1)?;
    }
    if let Some(f) = fragment_rel {
        validate_component(
            Component::Fragment,
            &after_authority[f + 1..],
            authority_end + f + 1,
        )?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }

    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).expect("path already validated");
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        terminal_navigation = index + 1 == segments.len() && (is_dot || is_dotdot);

        if is_dot {
            continue;
        } else if is_dotdot {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    Ok(result)
}

fn add_path_error(err: PathError) -> ResolveError {
    ResolveError::Path(err)
}

fn append_optional(out: &mut String, marker: char, value: Option<&str>) {
    if let Some(value) = value {
        out.push(marker);
        out.push_str(value);
    }
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(add_path_error)?;
    scan_input(reference).map_err(ResolveError::Reference)?;

    let first_delim = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delim].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(add_path_error)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(add_path_error)?;
        return Ok(url);
    }

    let after_fragment = reference.find('#');
    let before_fragment = after_fragment
        .map(|i| &reference[..i])
        .unwrap_or(reference);
    let fragment = after_fragment.map(|i| &reference[i + 1..]);

    let query_pos = before_fragment.find('?');
    let reference_path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let reference_query = query_pos.map(|i| &before_fragment[i + 1..]);

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_path.clone(),
            if query_pos.is_none() {
                base_url.query.as_deref()
            } else {
                reference_query
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
            format!("{slash}{reference_path}")
        };
        (merged, reference_query)
    };

    let mut constructed = String::new();
    constructed.push_str(&base_url.scheme);
    constructed.push_str("://");
    if let Some(userinfo) = base_url.userinfo.as_deref() {
        constructed.push_str(userinfo);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&selected_path);
    append_optional(&mut constructed, '?', selected_query);
    append_optional(&mut constructed, '#', fragment);

    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(add_path_error)?;
    Ok(url)
}

fn parse_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {

fn parse_url_inner(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Some(at) = global_byte_error(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !is_scheme_first(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    if let Some(i) = bytes[1..colon]
        .iter()
        .position(|&b| !is_scheme_rest(b))
    {
        return Err(ParseError::InvalidScheme { at: i + 1 });
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];

    let mut at_positions = authority.match_indices('@').map(|(i, _)| i);
    let first_at = at_positions.next();
    if let Some(second_at) = at_positions.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second_at,
        });
    }

    let (userinfo, host_start_in_authority) = match first_at {
        Some(i) if i == 0 => {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + i,
            });
        }
        Some(i) => (Some(authority[..i].to_string()), i + 1),
        None => (None, 0),
    };

    let host_port = &authority[host_start_in_authority..];
    let absolute_host_start = authority_start + host_start_in_authority;

    let (host, port, ipv6) = if host_port.as_bytes().first() == Some(&b'[') {
        let close_rel = match host_port.as_bytes()[1..]
            .iter()
            .position(|&b| b == b']')
        {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: absolute_host_start }),
        };

        let interior = &host_port[1..close_rel];
        if interior.is_empty() || interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: absolute_host_start });
        }

        let suffix = &host_port[close_rel + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(
                &suffix[1..],
                absolute_host_start + close_rel + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: absolute_host_start + close_rel + 1,
            });
        };

        (interior.to_ascii_lowercase(), port, true)
    } else {
        let host_port_bytes = host_port.as_bytes();
        let host_colon = host_port_bytes.iter().position(|&b| b == b':');
        let (host_text, port) = match host_colon {
            Some(i) => (
                &host_port[..i],
                Some(parse_port(
                    &host_port[i + 1..],
                    absolute_host_start + i + 1,
                )?),
            ),
            None => (host_port, None),
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost {
                at: absolute_host_start,
            });
        }
        if let Some(i) = host_text.bytes().position(|b| {
            !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_'))
        }) {
            return Err(ParseError::InvalidHost {
                at: absolute_host_start + i,
            });
        }

        (host_text.to_ascii_lowercase(), port, false)
    };

    let after_authority = &input[authority_end..];
    let fragment_pos = after_authority.find('#');
    let before_fragment = fragment_pos
        .map(|i| &after_authority[..i])
        .unwrap_or(after_authority);
    let fragment = fragment_pos.map(|i| after_authority[i + 1..].to_string());

    let query_pos = before_fragment.find('?');
    let path_text = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_pos.map(|i| before_fragment[i + 1..].to_string());

    let path = if path_text.is_empty() {
        "/".to_string()
    } else {
        path_text.to_string()
    };

    let mut components: Vec<(Component, &str, usize)> = Vec::new();
    if let Some(ref u) = userinfo {
        components.push((Component::UserInfo, u.as_str(), authority_start));
    }
    if !path_text.is_empty() {
        components.push((
            Component::Path,
            path_text,
            authority_end,
        ));
    }
    if let Some(qpos) = query_pos {
        components.push((
            Component::Query,
            &before_fragment[qpos + 1..],
            authority_end + qpos + 1,
        ));
    }
    if let Some(fpos) = fragment_pos {
        components.push((
            Component::Fragment,
            &after_authority[fpos + 1..],
            authority_end + fpos + 1,
        ));
    }

    for (component, text, offset) in components {
        match decode_component(component, text) {
            Ok(_) => {}
            Err(CodecError::BadEscape { at }) => {
                return Err(ParseError::BadEscape { at: offset + at });
            }
            Err(CodecError::InvalidUtf8 { at }) => {
                return Err(ParseError::InvalidUtf8 { at: offset + at });
            }
        }
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    parse_url_inner(input)
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len()
                || !bytes[i + 1].is_ascii_hexdigit()
                || !bytes[i + 2].is_ascii_hexdigit()
            {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = (bytes[i + 1] as char).to_digit(16).unwrap() as u8;
            let lo = (bytes[i + 2] as char).to_digit(16).unwrap() as u8;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_string()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };

        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }

    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let mut segments: Vec<&str> = Vec::new();
    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut terminal_navigation = false;

    for (i, raw) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 },
        })?;
        let navigation = decoded == "." || decoded == "..";
        terminal_navigation = i + 1 == raw_segments.len() && navigation;

        if decoded == "." {
            continue;
        } else if decoded == ".." {
            segments.pop();
        } else {
            segments.push(raw);
        }
    }

    if terminal_navigation {
        segments.push("");
    }

    if segments.is_empty() {
        Ok("/".to_string())
    } else {
        Ok(format!("/{}", segments.join("/")))
    }
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_pos = reference.find('#');
    let before_fragment = fragment_pos
        .map(|i| &reference[..i])
        .unwrap_or(reference);
    let fragment = fragment_pos.map(|i| &reference[i + 1..]);

    let query_pos = before_fragment.find('?');
    let path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_pos.map(|i| &before_fragment[i + 1..]);

    (path, query, fragment)
}

fn path_error(error: PathError) -> ResolveError {
    ResolveError::Path(error)
}

fn normalize_url_path(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(path_error)?;
    Ok(url)
}

fn render_authority(url: &Url) -> String {
    let mut result = String::new();
    if let Some(userinfo) = &url.userinfo {
        result.push_str(userinfo);
        result.push('@');
    }
    if url.ipv6 {
        result.push('[');
        result.push_str(&url.host);
        result.push(']');
    } else {
        result.push_str(&url.host);
    }
    if let Some(port) = url.port {
        result.push(':');
        result.push_str(&port.to_string());
    }
    result
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalize_url_path(base_url)?;

    if let Some(at) = global_byte_error(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let first_delimiter = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        return normalize_url_path(url);
    }

    let (ref_path, ref_query, ref_fragment) = split_reference(reference);
    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_url.path.clone(),
            if ref_query.is_some() {
                ref_query.map(str::to_string)
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_string()
        } else {
            let slash = base_url.path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_url.path[..=slash], ref_path)
        };
        (merged, ref_query.map(str::to_string))
    };

    let mut constructed = format!(
        "{}://{}{}",
        base_url.scheme,
        render_authority(&base_url),
        selected_path
    );
    if let Some(query) = selected_query {
        constructed.push('?');
        constructed.push_str(&query);
    }
    if let Some(fragment) = ref_fragment {
        constructed.push('#');
        constructed.push_str(fragment);
    }

    let url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalize_url_path(url)
}
