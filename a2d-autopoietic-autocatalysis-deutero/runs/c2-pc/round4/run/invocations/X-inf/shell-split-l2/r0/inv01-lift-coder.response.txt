#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { start: usize },
    Double { start: usize },
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }

    result
}

fn expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<Option<(usize, String)>, SplitError> {
    let after_dollar = dollar + 1;
    let Some(next) = input[after_dollar..].chars().next() else {
        return Ok(None);
    };

    if next == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    cursor += input[cursor..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let Some(close_pos) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        };

        let contents = &input[content_start..close_pos];
        let name_end = contents
            .char_indices()
            .take_while(|(_, c)| is_name_char(*c))
            .map(|(i, c)| i + c.len_utf8())
            .last()
            .unwrap_or(0);

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let remainder = &contents[name_end..];

        let selected = if remainder.is_empty() {
            lookup(env, name).unwrap_or("")
        } else if let Some(default_text) = remainder.strip_prefix(":-") {
            match lookup(env, name) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default_text),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok(Some((close_pos + 1, selected.to_string())));
    }

    if !is_name_start(next) {
        return Ok(None);
    }

    let mut end = after_dollar;
    while end < input.len() {
        let c = input[end..].chars().next().unwrap();
        if !is_name_char(c) {
            break;
        }
        end += c.len_utf8();
    }

    let value = lookup(env, &input[after_dollar..end])
        .unwrap_or("")
        .to_string();
    Ok(Some((end, value)))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let next_pos = cursor + c.len_utf8();

        match quote {
            Quote::Single { .. } => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                cursor = next_pos;
            }
            Quote::Double { .. } => {
                if c == '"' {
                    quote = Quote::None;
                    cursor = next_pos;
                } else if c == '\\' {
                    let slash_pos = cursor;
                    if next_pos == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash_pos });
                    }
                    let escaped = input[next_pos..].chars().next().unwrap();
                    let after_escaped = next_pos + escaped.len_utf8();
                    if escaped == '"' || escaped == '\\' || escaped == '$' {
                        word.push(escaped);
                        cursor = after_escaped;
                    } else if escaped == '\n' {
                        cursor = after_escaped;
                    } else {
                        word.push('\\');
                        cursor = next_pos;
                    }
                } else if c == '$' {
                    active = true;
                    if let Some((end, value)) = expansion(input, cursor, env)? {
                        word.push_str(&value);
                        cursor = end;
                    } else {
                        word.push('$');
                        cursor = next_pos;
                    }
                } else {
                    word.push(c);
                    cursor = next_pos;
                }
            }
            Quote::None => {
                if c == ' ' || c == '\t' || c == '\r' || c == '\n' {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next_pos;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single { start: cursor };
                    cursor = next_pos;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double { start: cursor };
                    cursor = next_pos;
                } else if c == '\\' {
                    if next_pos == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_pos..].chars().next().unwrap();
                    let after_escaped = next_pos + escaped.len_utf8();
                    if escaped == '\n' {
                        cursor = after_escaped;
                    } else {
                        active = true;
                        word.push(escaped);
                        cursor = after_escaped;
                    }
                } else if c == '$' {
                    active = true;
                    if let Some((end, value)) = expansion(input, cursor, env)? {
                        word.push_str(&value);
                        cursor = end;
                    } else {
                        word.push('$');
                        cursor = next_pos;
                    }
                } else {
                    active = true;
                    word.push(c);
                    cursor = next_pos;
                }
            }
        }
    }

    match quote {
        Quote::Single { start } => {
            return Err(SplitError::UnterminatedSingle { pos: start });
        }
        Quote::Double { start } => {
            return Err(SplitError::UnterminatedDouble { pos: start });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
