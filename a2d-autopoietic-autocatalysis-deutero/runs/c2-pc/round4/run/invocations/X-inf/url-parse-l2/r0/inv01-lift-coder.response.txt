use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let Some(high) = hex_value(bytes[i + 1]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            let Some(low) = hex_value(bytes[i + 2]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            decoded.push((high << 4) | low);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => Err(CodecError::InvalidUtf8 {
            at: origins[error.valid_up_to()],
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for byte in text.bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn validate_port(port: &str, port_start: usize) -> Result<Option<u16>, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    if let Some((offset, _)) = port
        .bytes()
        .enumerate()
        .find(|(_, byte)| !byte.is_ascii_digit())
    {
        return Err(ParseError::InvalidPort {
            at: port_start + offset,
        });
    }

    let digits = port.as_bytes();
    if digits.len() >= 2 && digits[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for &digit in digits {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((digit - b'0') as u32))
        {
            Some(n) => n,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
        if value > u16::MAX as u32 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(Some(value as u16))
}

fn validate_component(
    component: Component,
    text: &str,
    original_start: usize,
) -> Result<(), ParseError> {
    match decode_component(component, text) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => {
            Err(ParseError::BadEscape { at: original_start + at })
        }
        Err(CodecError::InvalidUtf8 { at }) => {
            Err(ParseError::InvalidUtf8 { at: original_start + at })
        }
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    if let Some((at, _)) = bytes
        .iter()
        .enumerate()
        .find(|(_, byte)| !(0x21..=0x7e).contains(byte))
    {
        return Err(ParseError::InvalidByte { at });
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(position) => position,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &byte)| (byte == b'@').then_some(i))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[hostport_start..];
    let absolute_hostport_start = authority_start + hostport_start;
    let (host, port, ipv6) = if hostport.as_bytes().first() == Some(&b'[') {
        let opening = absolute_hostport_start;
        let close = hostport.as_bytes()[1..]
            .iter()
            .position(|&byte| byte == b']')
            .map(|offset| offset + 1)
            .ok_or(ParseError::InvalidIpv6 { at: opening })?;
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: opening });
        }

        let suffix = &hostport[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            validate_port(
                &suffix[1..],
                absolute_hostport_start + close + 2,
            )?
        } else {
            return Err(ParseError::InvalidHost {
                at: absolute_hostport_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), parsed_port, true)
    } else {
        let split = hostport.as_bytes().iter().position(|&byte| byte == b':');
        let (host_text, parsed_port) = if let Some(index) = split {
            (
                &hostport[..index],
                validate_port(
                    &hostport[index + 1..],
                    absolute_hostport_start + index + 1,
                )?,
            )
        } else {
            (hostport, None)
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost {
                at: absolute_hostport_start,
            });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: absolute_hostport_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), parsed_port, false)
    };

    let fragment_delimiter = bytes[authority_end..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|offset| authority_end + offset);
    let before_fragment_end = fragment_delimiter.unwrap_or(bytes.len());
    let query_delimiter = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|offset| authority_end + offset);

    let path_end = query_delimiter.unwrap_or(before_fragment_end);
    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };
    let query = query_delimiter.map(|start| input[start + 1..before_fragment_end].to_owned());
    let fragment = fragment_delimiter.map(|start| input[start + 1..].to_owned());

    if let Some(ref text) = userinfo {
        validate_component(Component::UserInfo, text, authority_start)?;
    }
    if path_end != authority_end {
        validate_component(Component::Path, &path, authority_end)?;
    }
    if let Some(start) = query_delimiter {
        validate_component(
            Component::Query,
            query.as_deref().unwrap_or(""),
            start + 1,
        )?;
    }
    if let Some(start) = fragment_delimiter {
        validate_component(
            Component::Fragment,
            fragment.as_deref().unwrap_or(""),
            start + 1,
        )?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}
