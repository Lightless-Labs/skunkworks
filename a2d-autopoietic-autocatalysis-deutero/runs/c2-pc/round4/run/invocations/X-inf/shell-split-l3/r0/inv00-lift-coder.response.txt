#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn append_text(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn decode_default(text: &str) -> String {
    let mut output = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                output.push(next);
            }
        } else {
            output.push(c);
        }
    }
    output
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.text.push(c);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        word.text.push(next);
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        word.text.push('\\');
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let after_dollar = next_i;
                    if input[after_dollar..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[after_dollar..].starts_with('{') {
                        let (value, end) = expand_braced(input, i, env)?;
                        append_text(&mut word, &value, false);
                        active = true;
                        i = end;
                    } else if let Some((name, end)) = expand_name(input, after_dollar) {
                        let value = lookup(name, env).unwrap_or("");
                        append_text(&mut word, value, false);
                        active = true;
                        i = end;
                    } else {
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    word.text.push(c);
                    i = next_i;
                }

                let _ = open_pos;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        word.text.push(next);
                        active = true;
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let after_dollar = next_i;
                    if input[after_dollar..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[after_dollar..].starts_with('{') {
                        let (value, end) = expand_braced(input, i, env)?;
                        append_text(&mut word, &value, true);
                        active = true;
                        i = end;
                    } else if let Some((name, end)) = expand_name(input, after_dollar) {
                        let value = lookup(name, env).unwrap_or("");
                        append_text(&mut word, value, true);
                        active = true;
                        i = end;
                    } else {
                        word.text.push('$');
                        active = true;
                        i = next_i;
                    }
                } else {
                    if matches!(c, '*' | '?' | '[') {
                        word.glob.push(word.text.len());
                    }
                    word.text.push(c);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand_name(input: &str, start: usize) -> Option<(&str, usize)> {
    let first = input[start..].chars().next()?;
    if !is_name_start(first) {
        return None;
    }
    let mut end = start + first.len_utf8();
    while end < input.len() {
        let c = input[end..].chars().next().unwrap();
        if !is_name_continue(c) {
            break;
        }
        end += c.len_utf8();
    }
    Some((&input[start..end], end))
}

fn expand_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let content_start = dollar + 2;
    let mut j = content_start;
    let mut closing = None;
    while j < input.len() {
        let c = input[j..].chars().next().unwrap();
        let next = j + c.len_utf8();
        if c == '\\' {
            if next < input.len() {
                let escaped = input[next..].chars().next().unwrap();
                j = next + escaped.len_utf8();
            } else {
                j = next;
            }
        } else if c == '}' {
            closing = Some(j);
            break;
        } else {
            j = next;
        }
    }

    let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let mut name_end = 0;
    let mut chars = contents.char_indices();
    let Some((_, first)) = chars.next() else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    if !is_name_start(first) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    name_end += first.len_utf8();
    while name_end < contents.len() {
        let c = contents[name_end..].chars().next().unwrap();
        if !is_name_continue(c) {
            break;
        }
        name_end += c.len_utf8();
    }

    let name = &contents[..name_end];
    let value = if name_end == contents.len() {
        lookup(name, env).unwrap_or("").to_string()
    } else if contents[name_end..].starts_with(":-") {
        let default = &contents[name_end + 2..];
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}
