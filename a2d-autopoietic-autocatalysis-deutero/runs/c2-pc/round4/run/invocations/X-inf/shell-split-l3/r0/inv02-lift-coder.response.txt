#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append_text(word: &mut Word, text: &str, active_globs: bool) {
    for c in text.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if active_globs && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn find_braced_end(chars: &[char], start: usize) -> Option<usize> {
    let mut i = start;
    while i < chars.len() {
        if chars[i] == '\\' {
            i += 1;
            if i < chars.len() {
                i += 1;
            }
        } else if chars[i] == '}' {
            return Some(i);
        } else {
            i += 1;
        }
    }
    None
}

fn decode_default(chars: &[char]) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < chars.len() {
        if chars[i] == '\\' && i + 1 < chars.len() {
            i += 1;
            out.push(chars[i]);
        } else {
            out.push(chars[i]);
        }
        i += 1;
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some((pos, _)) = input.char_indices().find(|(_, c)| *c == '\0') {
        return Err(SplitError::Nul { pos });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut i = 0usize;

    while i < chars.len() {
        let (pos, c) = chars[i];
        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_text(&mut current, &c.to_string(), false);
                }
                i += 1;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i += 1;
                } else if c == '\\' {
                    if i + 1 >= chars.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = chars[i + 1].1;
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_text(&mut current, &next.to_string(), false);
                        }
                        i += 2;
                    } else {
                        append_text(&mut current, "\\", false);
                        i += 1;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '(' {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if i + 1 < chars.len()
                        && (is_name_start(chars[i + 1].1) || chars[i + 1].1 == '{')
                    {
                        let (value, consumed) = parse_expansion(&chars, i, pos, env)?;
                        append_text(&mut current, &value, false);
                        active = true;
                        i += consumed;
                    } else {
                        append_text(&mut current, "$", false);
                        i += 1;
                    }
                } else {
                    append_text(&mut current, &c.to_string(), false);
                    i += 1;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += 1;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_start = pos;
                    active = true;
                    i += 1;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_start = pos;
                    active = true;
                    i += 1;
                } else if c == '\\' {
                    if i + 1 >= chars.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = chars[i + 1].1;
                    if next == '\n' {
                        i += 2;
                    } else {
                        append_text(&mut current, &next.to_string(), false);
                        active = true;
                        i += 2;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if c == '$' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '(' {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if i + 1 < chars.len()
                        && (is_name_start(chars[i + 1].1) || chars[i + 1].1 == '{')
                    {
                        let (value, consumed) = parse_expansion(&chars, i, pos, env)?;
                        append_text(&mut current, &value, true);
                        active = true;
                        i += consumed;
                    } else {
                        append_text(&mut current, "$", true);
                        active = true;
                        i += 1;
                    }
                } else {
                    append_text(&mut current, &c.to_string(), true);
                    active = true;
                    i += 1;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

fn parse_expansion(
    chars: &[(usize, char)],
    dollar_index: usize,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let next = chars[dollar_index + 1].1;
    if next != '{' {
        let mut end = dollar_index + 1;
        while end < chars.len() && is_name_continue(chars[end].1) {
            end += 1;
        }
        let name: String = chars[dollar_index + 1..end]
            .iter()
            .map(|(_, c)| *c)
            .collect();
        return Ok((lookup(env, &name).unwrap_or("").to_string(), end - dollar_index));
    }

    let content_start = dollar_index + 2;
    let close = find_braced_end(
        &chars.iter().map(|(_, c)| *c).collect::<Vec<_>>(),
        content_start,
    )
    .ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;

    let contents: Vec<char> = chars[content_start..close].iter().map(|(_, c)| *c).collect();
    let mut name_end = 0;
    if contents.first().copied().is_some_and(is_name_start) {
        name_end = 1;
        while name_end < contents.len() && is_name_continue(contents[name_end]) {
            name_end += 1;
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let name: String = contents[..name_end].iter().collect();
    let selected = if name_end == contents.len() {
        lookup(env, &name).unwrap_or("").to_string()
    } else if name_end + 2 <= contents.len()
        && contents[name_end] == ':'
        && contents[name_end + 1] == '-'
    {
        let value = lookup(env, &name).unwrap_or("");
        if !value.is_empty() {
            value.to_string()
        } else {
            decode_default(&contents[name_end + 2..])
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((selected, close + 1 - dollar_index))
}
