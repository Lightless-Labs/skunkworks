#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    if pos == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: unquoted, 1: quoted, 2: after closing quote
    let mut quoted_open = 0usize;
    let mut record_index = 0usize;

    while pos < bytes.len() {
        let b = bytes[pos];
        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if state != 1 && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        if state == 1 {
            if b == quote {
                if bytes.get(pos + 1) == Some(&quote) {
                    field.push(quote as char);
                    pos += 2;
                } else {
                    state = 2;
                    pos += 1;
                }
            } else {
                let ch = input[pos..].chars().next().unwrap();
                field.push(ch);
                pos += ch.len_utf8();
            }
            continue;
        }

        if state == 2 {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                state = 0;
                pos += 1;
            } else if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                state = 0;
                pos += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte {
                    record: record_index,
                    byte: pos,
                });
            }
            continue;
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            pos += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            pos += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field.is_empty() {
                quoted_open = pos;
                state = 1;
                pos += 1;
            } else {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: pos,
                });
            }
        } else {
            let ch = input[pos..].chars().next().unwrap();
            field.push(ch);
            pos += ch.len_utf8();
        }
    }

    if state == 1 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quoted_open,
        });
    }
    if !record.is_empty() || !field.is_empty() || state == 2 {
        record.push(field);
        records.push(record);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        let mut found = None;
        for (i, value) in header.iter().enumerate() {
            if value == name {
                found = Some(i);
                if self.duplicates != DuplicatePolicy::KeepLast {
                    return found;
                }
            }
        }
        found
    }
}

fn table_from_records(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in h.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(n, _)| *n == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name, second));
            }
        }
        for (i, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }
    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote).map_err(|component| {
            CsvError::InvalidConfig { component }
        })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len() && start == 0;
    }
    b[start].is_ascii_digit()
        && b[start] != b'0'
        && b[start..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.starts_with(b"-"));
    let Some(dot) = b.iter().position(|&x| x == b'.') else {
        return false;
    };
    if dot + 3 != b.len() || !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    let integral = &b[start..dot];
    if integral.is_empty() || !integral.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if integral.len() > 1 && integral[0] == b'0' {
        return false;
    }
    if integral[0] == b'0' && start > 0 && b[dot + 1..].iter().all(|&x| x == b'0') {
        return false;
    }
    true
}

fn parse_integer_value(s: &str) -> Option<i64> {
    s.parse::<i64>().ok()
}

fn parse_decimal_value(s: &str) -> Option<i64> {
    let neg = s.starts_with('-');
    let body = if neg { &s[1..] } else { s };
    let dot = body.find('.')?;
    let whole: i128 = body[..dot].parse().ok()?;
    let frac: i128 = body[dot + 1..].parse().ok()?;
    let signed = (whole * 100 + frac) * if neg { -1 } else { 1 };
    i64::try_from(signed).ok()
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    let mut stopped = false;
    for (row_index, row) in table.rows.iter().enumerate() {
        if stopped { break; }
        let mut out = Vec::with_capacity(columns.len());
        let mut failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if failed {
                out.push(None);
                continue;
            }
            let column = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let error = |kind| CellError { row: row_index, spec: spec_index, column, kind };
            let cell = match column {
                None => { result.errors.push(error(CellErrorKind::UnknownColumn)); failed = true; out.push(None); continue; }
                Some(i) => match row.get(i) {
                    None => { result.errors.push(error(CellErrorKind::MissingCell)); failed = true; out.push(None); continue; }
                    Some(v) => v,
                }
            };
            if cell.is_empty() && spec.nullable {
                out.push(Some(Value::Null));
                continue;
            }
            if cell.is_empty() && spec.kind != ColumnType::Text {
                result.errors.push(error(CellErrorKind::Empty)); failed = true; out.push(None); continue;
            }
            let value = match spec.kind {
                ColumnType::Text => Ok(Value::Text(cell.clone())),
                ColumnType::Integer => {
                    if !integer_syntax(cell) { Err(CellErrorKind::InvalidInteger) }
                    else { parse_integer_value(cell).map(Value::Integer).ok_or(CellErrorKind::IntegerOverflow) }
                }
                ColumnType::Decimal => {
                    if !decimal_syntax(cell) { Err(CellErrorKind::InvalidDecimal) }
                    else { parse_decimal_value(cell).map(Value::Decimal).ok_or(CellErrorKind::DecimalOverflow) }
                }
                ColumnType::Boolean => match cell.as_str() {
                    "yes" | "1" => Ok(Value::Boolean(true)),
                    "no" | "0" => Ok(Value::Boolean(false)),
                    _ => Err(CellErrorKind::InvalidBoolean),
                }
            };
            match value {
                Ok(v) => out.push(Some(v)),
                Err(kind) => { result.errors.push(error(kind)); failed = true; out.push(None); }
            }
            if failed && policy != ErrorPolicy::Collect {
                if policy == ErrorPolicy::Stop { stopped = true; }
                while out.len() < columns.len() { out.push(None); }
                break;
            }
        }
        result.rows.push(out);
        if policy == ErrorPolicy::FirstPerRow {
            failed = false;
        }
    }
    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn scan_nuls(table: &Table) -> Option<WriteError> {
    let mut records: Vec<&Vec<String>> = Vec::new();
    if let Some(h) = &table.header { records.push(h); }
    records.extend(table.rows.iter());
    for (record, row) in records.into_iter().enumerate() {
        for (column, value) in row.iter().enumerate() {
            if let Some(byte) = value.as_bytes().iter().position(|&b| b == 0) {
                return Some(WriteError::Nul { record, column, byte });
            }
        }
    }
    None
}

fn duplicate_header_error(header: &[String]) -> Option<WriteError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(n, _)| *n == name) {
            return Some(WriteError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn encode_field(value: &str, delimiter: u8, quote: u8, first_field_first_record: bool, sole: bool) -> String {
    let bytes = value.as_bytes();
    let special = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_first_record && value.starts_with('\u{feff}'))
        || (sole && value.is_empty());
    if !special { return value.to_string(); }
    let mut out = String::new();
    out.push(quote as char);
    for ch in value.chars() {
        if ch as u32 == quote as u32 { out.push(quote as char); }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;
    if let Some(err) = scan_nuls(table) { return Err(err); }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width { record: i + 1, expected: header.len(), actual: row.len() });
            }
        }
    }

    let mut original: Vec<Vec<String>> = Vec::new();
    if let Some(h) = &table.header { original.push(h.clone()); }
    original.extend(table.rows.clone());
    for (record, row) in original.iter().enumerate() {
        if row.is_empty() { return Err(WriteError::ZeroColumns { record }); }
    }

    let mut records = original;
    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(err) = duplicate_header_error(header) { return Err(err); }
            }
            let mut retained: Vec<(String, usize)> = Vec::new();
            for (i, name) in header.iter().enumerate() {
                if let Some(existing) = retained.iter_mut().find(|(n, _)| n == name) {
                    if table.duplicates == DuplicatePolicy::KeepLast { existing.1 = i; }
                } else {
                    retained.push((name.clone(), i));
                }
            }
            retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
            let indexes: Vec<usize> = retained.iter().map(|x| x.1).collect();
            let sorted_header: Vec<String> = retained.into_iter().map(|x| x.0).collect();
            let mut projected: Vec<Vec<String>> = table.rows.iter().map(|row| indexes.iter().map(|&i| row[i].clone()).collect()).collect();
            projected.sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
            records = vec![sorted_header];
            records.extend(projected);
        } else {
            records = table.rows.clone();
            records.sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
        }
    }

    if records.is_empty() { return Ok(String::new()); }
    let mut output = String::new();
    let terminator = if config.mode == WriteMode::Canonical || config.line_ending == LineEnding::Lf {
        "\n"
    } else {
        "\r\n"
    };
    for (ri, row) in records.iter().enumerate() {
        if ri > 0 { output.push_str(terminator); }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 { output.push(config.delimiter as char); }
            output.push_str(&encode_field(field, config.delimiter, config.quote, ri == 0 && ci == 0, row.len() == 1));
        }
    }
    if config.mode == WriteMode::Canonical || config.final_terminator {
        output.push_str(terminator);
    }
    Ok(output)
}
