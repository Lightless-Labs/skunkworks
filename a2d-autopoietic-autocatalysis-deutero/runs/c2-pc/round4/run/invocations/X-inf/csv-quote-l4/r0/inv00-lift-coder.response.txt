use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut quote_start = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record: records.len(), byte: i });
        }

        if state != 2 && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record: records.len(), byte: i });
            }
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            state = 0;
            i += 2;
            continue;
        }

        if state != 2 && b == b'\n' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            state = 0;
            i += 1;
            continue;
        }

        match state {
            0 => {
                if b == delimiter {
                    record.push(String::new());
                } else if b == quote {
                    state = 2;
                    quote_start = i;
                } else {
                    field.push(b as char);
                    state = 1;
                }
            }
            1 => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote { record: records.len(), byte: i });
                } else {
                    field.push(b as char);
                }
            }
            2 => {
                if b == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 1;
                    } else {
                        state = 3;
                    }
                } else {
                    field.push(b as char);
                }
            }
            _ => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                } else {
                    return Err(CsvError::TrailingByte { record: records.len(), byte: i });
                }
            }
        }
        i += 1;
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote { record: records.len(), byte: quote_start });
    }

    // A terminator consumes its record immediately. Track whether the current
    // record has content or syntax so EOF after a terminator adds nothing.
    if !record.is_empty() || !field.is_empty() || state != 0 {
        record.push(field);
        records.push(record);
    } else if !bytes.is_empty() && !bytes.ends_with(b"\n") {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut firsts: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = firsts.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        firsts.insert(name, second);
    }
    None
}

fn make_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;

    let mut records = parse_dialect(input, config.delimiter, config.quote)?;
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(ref header) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(header) {
                return Err(error);
            }
        }
        for (index, row) in records.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows: records, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), finished: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = make_table(&self.input, self.config);
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start]) && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.starts_with(b"-"));
    if b.len() < start + 4 || b.get(b.len() - 3) != Some(&b'.') {
        return false;
    }
    let integer = &s[start..s.len() - 3];
    let frac = &b[b.len() - 2..];
    let integer_ok = integer == "0"
        || (!integer.is_empty()
            && (b'1'..=b'9').contains(&integer.as_bytes()[0])
            && integer.as_bytes()[1..].iter().all(u8::is_ascii_digit));
    integer_ok && frac.iter().all(u8::is_ascii_digit)
        && !(s.starts_with("-0.") && s[3..].bytes().all(|x| x == b'0'))
}

fn evaluate(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if spec.nullable && cell.is_empty() {
        return Ok(Value::Null);
    }
    if cell.is_empty() && spec.kind != ColumnType::Text {
        return Err(CellErrorKind::Empty);
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_owned())),
        ColumnType::Integer => {
            if !integer_syntax(cell) { return Err(CellErrorKind::InvalidInteger); }
            cell.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(cell) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = cell.starts_with('-');
            let body = if neg { &cell[1..] } else { cell };
            let digits = body.replace('.', "");
            let magnitude = digits.parse::<i128>().unwrap_or(i128::MAX);
            let signed = if neg { -magnitude } else { magnitude };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut stop_row = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if stop_row {
                output.push(None);
                continue;
            }
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, value, error) = match resolved {
                None => (None, None, Some(CellErrorKind::UnknownColumn)),
                Some(index) => match row.get(index) {
                    None => (Some(index), None, Some(CellErrorKind::MissingCell)),
                    Some(cell) => match evaluate(cell, spec) {
                        Ok(value) => (Some(index), Some(value), None),
                        Err(error) => (Some(index), None, Some(error)),
                    },
                },
            };
            output.push(value);
            if let Some(kind) = error {
                result.errors.push(CellError { row: row_index, spec: spec_index, column, kind });
                if policy != ErrorPolicy::Collect {
                    stop_row = true;
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(output);
                        return result;
                    }
                }
            }
        }
        result.rows.push(output);
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode {
    Preserve,
    Canonical,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut originals: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        originals.push(header);
    }
    originals.extend(table.rows.iter());

    for (record, fields) in originals.iter().enumerate() {
        for (column, field) in fields.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul { record, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, fields) in originals.iter().enumerate() {
        if fields.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(CsvError::DuplicateHeader { name, first, second }) = duplicate_header(header) {
                    return Err(WriteError::DuplicateHeader { name, first, second });
                }
            }
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                records.push(header.clone());
            }
            records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut keep: HashMap<&str, usize> = HashMap::new();
                for (i, name) in header.iter().enumerate() {
                    match table.duplicates {
                        DuplicatePolicy::KeepLast => { keep.insert(name, i); }
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            keep.entry(name).or_insert(i);
                        }
                    }
                }
                let mut cols: Vec<usize> = keep.values().copied().collect();
                cols.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));
                records.push(cols.iter().map(|&i| header[i].clone()).collect());
                let mut rows: Vec<Vec<String>> = table.rows.iter().map(|row| {
                    cols.iter().map(|&i| row[i].clone()).collect()
                }).collect();
                rows.sort_by(|a, b| compare_fields(a, b));
                records.extend(rows);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_fields(a, b));
                records = rows;
            }
        }
    }

    let mut output = String::new();
    let terminator = match (config.mode, config.line_ending) {
        (WriteMode::Canonical, _) | (_, LineEnding::Lf) => "\n",
        (_, LineEnding::CrLf) => "\r\n",
    };

    for (record_index, fields) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(terminator);
        }
        for (column_index, field) in fields.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            let bytes = field.as_bytes();
            let quote = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
                || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
                || (record_index == 0 && column_index == 0 && field.starts_with('\u{feff}'))
                || (fields.len() == 1 && field.is_empty());
            if quote {
                output.push(config.quote as char);
                for &b in bytes {
                    output.push(b as char);
                    if b == config.quote {
                        output.push(config.quote as char);
                    }
                }
                output.push(config.quote as char);
            } else {
                output.push_str(field);
            }
        }
    }

    if config.mode == WriteMode::Canonical {
        if !records.is_empty() {
            output.push('\n');
        }
    } else if config.final_terminator && !records.is_empty() {
        output.push_str(terminator);
    }
    Ok(output)
}

fn compare_fields(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let ordering = left.as_bytes().cmp(right.as_bytes());
        if ordering != std::cmp::Ordering::Equal {
            return ordering;
        }
    }
    a.len().cmp(&b.len())
}
