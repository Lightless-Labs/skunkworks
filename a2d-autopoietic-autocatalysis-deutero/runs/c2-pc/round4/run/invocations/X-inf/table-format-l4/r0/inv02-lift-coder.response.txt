#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

type LogicalLine = Vec<Token>;
type CellLines = Vec<LogicalLine>;

fn fixed_overhead(n: usize, mode: &Mode) -> usize {
    match mode {
        Mode::Grid | Mode::Markdown => 3 * n + 1,
        Mode::Compact => 2 * n.saturating_sub(1),
    }
}

fn minimum_column_width(mode: &Mode) -> usize {
    match mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    }
}

fn validate_cell(
    cell: &str,
    row: usize,
    column: usize,
) -> Result<Vec<String>, TableError> {
    let mut normalized = String::with_capacity(cell.len());
    let mut chars = cell.char_indices().peekable();

    while let Some((byte, ch)) = chars.next() {
        if ch == '\r' {
            if matches!(chars.peek(), Some((_, '\n'))) {
                chars.next();
                normalized.push('\n');
                continue;
            }
            return Err(TableError::InvalidChar { row, column, byte });
        }

        if (ch.is_control() && ch != '\n') || ch == '\u{2028}' || ch == '\u{2029}' {
            return Err(TableError::InvalidChar { row, column, byte });
        }

        normalized.push(ch);
    }

    Ok(normalized
        .split('\n')
        .map(|line| line.trim_end_matches(' ').to_owned())
        .collect())
}

fn tokenize(line: &str, markdown: bool) -> LogicalLine {
    let mut tokens = Vec::new();
    for ch in line.chars() {
        if markdown && ch == '|' {
            tokens.push(Token {
                text: "\\|".to_owned(),
                width: 2,
            });
        } else if markdown && ch == '\\' {
            tokens.push(Token {
                text: "\\\\".to_owned(),
                width: 2,
            });
        } else {
            tokens.push(Token {
                text: ch.to_string(),
                width: 1,
            });
        }
    }
    tokens
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }

            let mut out = String::new();
            let mut used = 0;
            for token in tokens {
                if used + token.width > width.saturating_sub(1) {
                    break;
                }
                used += token.width;
                out.push_str(&token.text);
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // Every token fits in all legal widths.
                if end == start {
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn align_fragment(fragment: &str, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(scalar_len(fragment));
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), fragment, " ".repeat(right))
}

fn emit_data_line(
    fragments: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    let mut out = String::new();
    match mode {
        Mode::Grid | Mode::Markdown => {
            out.push('|');
            for (i, fragment) in fragments.iter().enumerate() {
                out.push(' ');
                out.push_str(&align_fragment(fragment, widths[i], &alignments[i]));
                out.push(' ');
                out.push('|');
            }
        }
        Mode::Compact => {
            for i in 0..fragments.len() {
                if i > 0 {
                    out.push_str("  ");
                }
                let aligned = align_fragment(&fragments[i], widths[i], &alignments[i]);
                if i + 1 == fragments.len() {
                    let right_padding = match alignments[i] {
                        Alignment::Left => widths[i].saturating_sub(scalar_len(&fragments[i])),
                        Alignment::Right | Alignment::Center => 0,
                    };
                    let keep = aligned.len().saturating_sub(right_padding);
                    out.push_str(&aligned[..keep]);
                } else {
                    out.push_str(&aligned);
                }
            }
        }
    }
    out.push('\n');
    out
}

fn border(widths: &[usize]) -> String {
    let mut out = String::from("+");
    for width in widths {
        out.push_str(&"-".repeat(width + 2));
        out.push('+');
    }
    out.push('\n');
    out
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_col = minimum_column_width(&options.mode);
    let minimum = fixed_overhead(n, &options.mode) + n * min_col;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    let mut logical: Vec<Vec<CellLines>> = Vec::with_capacity(rows.len());
    for (r, row) in rows.iter().enumerate() {
        let mut parsed_row = Vec::with_capacity(n);
        for (c, cell) in row.iter().enumerate() {
            let lines = validate_cell(cell, r, c)?;
            parsed_row.push(
                lines
                    .iter()
                    .map(|line| tokenize(line, matches!(options.mode, Mode::Markdown)))
                    .collect(),
            );
        }
        logical.push(parsed_row);
    }

    let mut widths = vec![min_col; n];
    for row in &logical {
        for (column, cell_lines) in row.iter().enumerate() {
            for line in cell_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let overhead = fixed_overhead(n, &options.mode);
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let candidate = (0..n)
                .rev()
                .find(|&i| widths[i] > min_col);
            match candidate {
                Some(i) => widths[i] -= 1,
                None => break,
            }
        }
    }

    let alignments = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in logical.iter().enumerate() {
        let mut cell_fragments: Vec<Vec<String>> = Vec::with_capacity(n);
        let mut height = 1;
        for (column, cell_lines) in row.iter().enumerate() {
            let mut all = Vec::new();
            for line in cell_lines {
                all.extend(fragments(line, widths[column], &overflows[column]));
            }
            height = height.max(all.len());
            cell_fragments.push(all);
        }

        for fragment_index in 0..height {
            let current: Vec<String> = (0..n)
                .map(|column| {
                    cell_fragments[column]
                        .get(fragment_index)
                        .cloned()
                        .unwrap_or_default()
                })
                .collect();
            output.push_str(&emit_data_line(
                &current,
                &widths,
                &alignments,
                &options.mode,
            ));
        }

        if matches!(options.mode, Mode::Grid) && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
        if matches!(options.mode, Mode::Grid) && row_index + 1 == rows.len() {
            output.push_str(&border(&widths));
        }
        if matches!(options.mode, Mode::Markdown) && row_index == 0 {
            output.push('|');
            for (i, width) in widths.iter().enumerate() {
                output.push(' ');
                match alignments[i] {
                    Alignment::Left => {
                        output.push(':');
                        output.push_str(&"-".repeat(width - 1));
                    }
                    Alignment::Right => {
                        output.push_str(&"-".repeat(width - 1));
                        output.push(':');
                    }
                    Alignment::Center => {
                        output.push(':');
                        output.push_str(&"-".repeat(width - 2));
                        output.push(':');
                    }
                }
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }
    }

    Ok(output)
}
