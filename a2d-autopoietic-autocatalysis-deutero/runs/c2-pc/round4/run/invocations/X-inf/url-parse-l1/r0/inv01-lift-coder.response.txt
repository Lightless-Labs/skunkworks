#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: bytes.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }

    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    let authority_marker = colon + 1;
    if bytes.get(authority_marker) != Some(&b'/')
        || bytes.get(authority_marker + 1) != Some(&b'/')
    {
        return Err(ParseError::MissingAuthority { at: authority_marker });
    }

    let authority_start = authority_marker + 2;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map_or(bytes.len(), |offset| authority_start + offset);
    let authority = &bytes[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .iter()
        .enumerate()
        .filter_map(|(offset, &byte)| (byte == b'@').then_some(offset))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport, hostport_start) = if let Some(&offset) = at_positions.first() {
        if offset == 0 {
            return Err(ParseError::InvalidUserInfo { at: authority_start });
        }
        (
            Some(input[authority_start..authority_start + offset].to_owned()),
            &authority[offset + 1..],
            authority_start + offset + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    let host_colon = hostport.iter().position(|&byte| byte == b':');
    let (host_bytes, port_bytes, port_start) = if let Some(offset) = host_colon {
        (
            &hostport[..offset],
            Some(&hostport[offset + 1..]),
            hostport_start + offset + 1,
        )
    } else {
        (hostport, None, 0)
    };

    if host_bytes.is_empty() {
        return Err(ParseError::InvalidHost { at: hostport_start });
    }

    for (offset, &byte) in host_bytes.iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: hostport_start + offset,
            });
        }
    }

    let port = if let Some(port_bytes) = port_bytes {
        if port_bytes.is_empty() {
            return Err(ParseError::InvalidPort {
                at: port_start,
            });
        }

        for (offset, &byte) in port_bytes.iter().enumerate() {
            if !byte.is_ascii_digit() {
                return Err(ParseError::InvalidPort {
                    at: port_start + offset,
                });
            }
        }

        if port_bytes.len() >= 2 && port_bytes[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        let mut value: u32 = 0;
        for &byte in port_bytes {
            value = value
                .saturating_mul(10)
                .saturating_add(u32::from(byte - b'0'));
            if value > u16::MAX as u32 {
                return Err(ParseError::InvalidPort { at: port_start });
            }
        }

        if value == 0 {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        Some(value as u16)
    } else {
        None
    };

    let component_start = authority_end;
    let fragment_pos = bytes[component_start..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|offset| component_start + offset);

    let before_fragment_end = fragment_pos.unwrap_or(bytes.len());
    let query_pos = bytes[component_start..before_fragment_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|offset| component_start + offset);

    let path_end = query_pos.unwrap_or(before_fragment_end);
    let path = if path_end == component_start {
        "/".to_owned()
    } else {
        input[component_start..path_end].to_owned()
    };

    let query = query_pos.map(|start| {
        let end = fragment_pos.unwrap_or(bytes.len());
        input[start + 1..end].to_owned()
    });

    let fragment = fragment_pos.map(|start| input[start + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: String::from_utf8(host_bytes.to_ascii_lowercase()).unwrap(),
        port,
        path,
        query,
        fragment,
    })
}
