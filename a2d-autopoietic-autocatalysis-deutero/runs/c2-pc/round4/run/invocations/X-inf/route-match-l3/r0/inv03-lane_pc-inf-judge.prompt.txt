fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    let parsed = split_head(pattern).map_err(|at| Error::InvalidPattern { at })?;
    let mut result = Vec::with_capacity(parsed.segments.len());
    let mut names = std::collections::HashSet::new();

    for (start, raw) in parsed.raw_segments {
        if raw.starts_with(':') || raw.starts_with('*') {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if marker == b'*' && !parsed_is_last(&parsed, start) {
                return Err(Error::InvalidPattern { at: start });
            }
            result.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start).map_err(|at| Error::InvalidPattern { at })?;
            result.push(Segment::Static(decoded));
        }
    }
    Ok((result, parsed.trailing))
}

struct SplitResult<'a> {
    raw_segments: Vec<(usize, &'a str)>,
    segments: Vec<String>,
    trailing: bool,
}

fn split_head(input: &str) -> Result<SplitResult<'_>, usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok(SplitResult {
            raw_segments: Vec::new(),
            segments: Vec::new(),
            trailing: false,
        });
    }

    let trailing = input.ends_with('/');
    let effective_end = if trailing { input.len() - 1 } else { input.len() };
    let mut raw_segments = Vec::new();
    let mut segments = Vec::new();
    let mut start = 1;

    for (offset, byte) in input.as_bytes()[1..effective_end].iter().enumerate() {
        if *byte == b'/' {
            let absolute = offset + 1;
            if absolute == start {
                return Err(start);
            }
            raw_segments.push((start, &input[start..absolute]));
            start = absolute + 1;
        }
    }
    if start == effective_end {
        return Err(start);
    }
    raw_segments.push((start, &input[start..effective_end]));
    for (at, raw) in &raw_segments {
        segments.push(decode_segment(raw, *at)?);
    }

    Ok(SplitResult {
        raw_segments,
        segments,
        trailing,
    })
}

fn parsed_is_last(parsed: &SplitResult<'_>, start: usize) -> bool {
    parsed
        .raw_segments
        .last()
        .map(|(last_start, _)| *last_start == start)
        .unwrap_or(false)
}

        segments: split.segments,
        trailing: split.trailing,
    })

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(out).unwrap())
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let wildcard_index = route
        .segments
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));

    let fixed_len = wildcard_index.unwrap_or(route.segments.len());
    if wildcard_index.is_none() && path.len() != fixed_len {
        return None;
    }
    if wildcard_index.is_some() && path.len() < fixed_len {
        return None;
    }

    for (i, segment) in route.segments.iter().take(fixed_len).enumerate() {
        match segment {
            Segment::Static(value) if value != &path[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_index {
        if let Segment::Wildcard(name) = &route.segments[index] {
            params.push((name.clone(), path[index..].join("/")));
        }
    }
    Some(params)

            trailing_slash: false,
        });
    }

    let end = if trailing_slash {

    } else {
        input.len()

    let mut names = std::collections::HashSet::new();

    while start < end {
        let rel_end = input[start..end].find('/').map(|n| start + n).unwrap_or(end);
        if rel_end == start {
            return Err(path_error(pattern, start));
        }

        let segment = &input[start..rel_end];
        if pattern && (segment.starts_with(':') || segment.starts_with('*')) {
            let marker = segment.as_bytes()[0];
            let name = &segment[1..];
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }

            if marker == b'*' {
                if rel_end != end {
                    return Err(Error::InvalidPattern { at: start });
                }
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            let decoded = decode_static_segment(segment, start, pattern)?;
            segments.push(Segment::Static(decoded));
        }

        if rel_end == end {
            break;
        }
        start = rel_end + 1;
        if start == end {
            return Err(path_error(pattern, start));
        }

        segments: segments
            .into_iter()
            .map(|segment| match segment {

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, bool, bool, Vec<u8>), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok((Vec::new(), false, false, vec![3]));
    }

    let bytes = input.as_bytes();
    let trailing_slash = bytes.last() == Some(&b'/');
    let end = if trailing_slash {
        input.len() - 1
    } else {
        input.len()
    };
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut rank = Vec::new();
    let mut wildcard = false;
    let mut start = 1;

    while start < end {
        let rel_end = input[start..end].find('/').map(|n| start + n).unwrap_or(end);
        if rel_end == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let segment = &input[start..rel_end];
        if segment.starts_with(':') || segment.starts_with('*') {
            let marker = segment.as_bytes()[0];
            let name = &segment[1..];
            if !valid_capture_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if marker == b'*' {
                if rel_end != end {
                    return Err(Error::InvalidPattern { at: start });
                }
                wildcard = true;
                rank.push(0);
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                rank.push(1);
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            rank.push(2);
            segments.push(Segment::Static(decode_static_segment(
                segment, start, true,
            )?));
        }

        if rel_end == end {
            break;
        }
        start = rel_end + 1;
        if start == end {
            return Err(Error::InvalidPattern { at: start });
        }
    }

    if !wildcard {
        rank.push(3);
    }

    Ok((segments, trailing_slash, wildcard, rank))
}

    pub fn new(policy: SlashPolicy) -> Self {

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {

            trailing_slash,

            rank,

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let parsed = parse_request_path(path)?;
        let mut candidates = Vec::new();

        for (id, route) in self.routes.iter().enumerate() {
            if let Some(params) = match_route(route, &parsed.segments) {
                let slash_matches = self.policy != SlashPolicy::Strict
                    || route.trailing_slash == parsed.trailing_slash;
                if slash_matches {
                    candidates.push((id, route, params));
                }
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<_> = if method == "HEAD" {
            if candidates.iter().any(|(_, route, _)| route.method == "HEAD") {
                candidates
                    .iter()
                    .filter(|(_, route, _)| route.method == "HEAD")
                    .collect()
            } else {
                candidates
                    .iter()
                    .filter(|(_, route, _)| route.method == "GET")
                    .collect()
            }
        } else {
            candidates
                .iter()
                .filter(|(_, route, _)| route.method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(_, route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let &(id, route, params) = eligible
            .into_iter()
            .max_by(|(id_a, a, _), (id_b, b, _)| {
                a.rank.cmp(&b.rank).then_with(|| id_b.cmp(id_a))
            })
            .expect("eligible routes are nonempty");

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing_slash != parsed.trailing_slash
        {
            let head_end = path
                .char_indices()
                .find_map(|(i, c)| matches!(c, '?' | '#').then_some(i))
                .unwrap_or(path.len());
            let mut target = path[..head_end].to_owned();
            if route.trailing_slash {
                target.push('/');
            } else if target.ends_with('/') && target != "/" {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: id,
            params,
            redirect,
        })
    }
}

fn decode_request_segment(input: &str, base: usize) -> Result<String, Error> {
    decode_static_segment(input, base, false)
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut request_index = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(expected) => {
                if request.get(request_index)? != expected {
                    return None;
                }
                request_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(request_index)?.clone()));
                request_index += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), request[request_index..].join("/")));
                request_index = request.len();
                break;
            }
        }
    }

    if request_index != request.len() {
        return None;
    }
    Some(params)
}
