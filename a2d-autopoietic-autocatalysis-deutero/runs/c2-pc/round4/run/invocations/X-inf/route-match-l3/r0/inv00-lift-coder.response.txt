#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        let (segments, trailing) = parse_pattern(pattern)?;
        let has_wildcard = matches!(segments.last(), Some(Segment::Wildcard(_)));
        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing,
            rank,
            has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let (head, _) = path.split_once(['?', '#']).unwrap_or((path, ""));
        let parsed = parse_path(head)?;
        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();

        for (index, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((index, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<usize> = if method == "HEAD" {
            if candidates
                .iter()
                .any(|(i, _)| self.routes[*i].method == "HEAD")
            {
                candidates
                    .iter()
                    .enumerate()
                    .filter_map(|(ci, (ri, _))| {
                        (self.routes[*ri].method == "HEAD").then_some(ci)
                    })
                    .collect()
            } else {
                candidates
                    .iter()
                    .enumerate()
                    .filter_map(|(ci, (ri, _))| {
                        (self.routes[*ri].method == "GET").then_some(ci)
                    })
                    .collect()
            }
        } else {
            candidates
                .iter()
                .enumerate()
                .filter_map(|(ci, (ri, _))| {
                    (self.routes[*ri].method == method).then_some(ci)
                })
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(ri, _)| self.routes[*ri].method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let winner = *eligible
            .iter()
            .max_by(|&&a, &&b| {
                self.routes[candidates[a].0]
                    .rank
                    .cmp(&self.routes[candidates[b].0].rank)
                    .then_with(|| candidates[b].0.cmp(&candidates[a].0))
            })
            .unwrap();
        let (route_index, params) = &candidates[winner];
        let route = &self.routes[*route_index];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != parsed.trailing
            && !parsed.segments.is_empty()
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: *route_index,
            params: params.clone(),
            redirect,
        })
    }
}

struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)
    }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    let parsed = split_head(pattern).map_err(|at| Error::InvalidPattern { at })?;
    let mut result = Vec::with_capacity(parsed.segments.len());
    let mut names = std::collections::HashSet::new();

    for (start, raw) in parsed.raw_segments {
        if raw.starts_with(':') || raw.starts_with('*') {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if marker == b'*' && !parsed_is_last(&parsed, start) {
                return Err(Error::InvalidPattern { at: start });
            }
            result.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, start).map_err(|at| Error::InvalidPattern { at })?;
            result.push(Segment::Static(decoded));
        }
    }
    Ok((result, parsed.trailing))
}

struct SplitResult<'a> {
    raw_segments: Vec<(usize, &'a str)>,
    segments: Vec<String>,
    trailing: bool,
}

fn split_head(input: &str) -> Result<SplitResult<'_>, usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok(SplitResult {
            raw_segments: Vec::new(),
            segments: Vec::new(),
            trailing: false,
        });
    }

    let trailing = input.ends_with('/');
    let effective_end = if trailing { input.len() - 1 } else { input.len() };
    let mut raw_segments = Vec::new();
    let mut segments = Vec::new();
    let mut start = 1;

    for (offset, byte) in input.as_bytes()[1..effective_end].iter().enumerate() {
        if *byte == b'/' {
            let absolute = offset + 1;
            if absolute == start {
                return Err(start);
            }
            raw_segments.push((start, &input[start..absolute]));
            start = absolute + 1;
        }
    }
    if start == effective_end {
        return Err(start);
    }
    raw_segments.push((start, &input[start..effective_end]));
    for (at, raw) in &raw_segments {
        segments.push(decode_segment(raw, *at)?);
    }

    Ok(SplitResult {
        raw_segments,
        segments,
        trailing,
    })
}

fn parsed_is_last(parsed: &SplitResult<'_>, start: usize) -> bool {
    parsed
        .raw_segments
        .last()
        .map(|(last_start, _)| *last_start == start)
        .unwrap_or(false)
}

fn parse_path(path: &str) -> Result<ParsedPath, Error> {
    let split = split_head(path).map_err(|at| Error::InvalidPath { at })?;
    Ok(ParsedPath {
        segments: split.segments,
        trailing: split.trailing,
    })
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(base + i);
            }
            out.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(out).unwrap())
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let wildcard_index = route
        .segments
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));

    let fixed_len = wildcard_index.unwrap_or(route.segments.len());
    if wildcard_index.is_none() && path.len() != fixed_len {
        return None;
    }
    if wildcard_index.is_some() && path.len() < fixed_len {
        return None;
    }

    for (i, segment) in route.segments.iter().take(fixed_len).enumerate() {
        match segment {
            Segment::Static(value) if value != &path[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_index {
        if let Segment::Wildcard(name) = &route.segments[index] {
            params.push((name.clone(), path[index..].join("/")));
        }
    }
    Some(params)
}
