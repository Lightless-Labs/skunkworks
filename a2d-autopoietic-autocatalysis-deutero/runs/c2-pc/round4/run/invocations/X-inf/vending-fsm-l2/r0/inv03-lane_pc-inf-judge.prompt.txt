#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

pub struct State {

    pub selected: Option<String>,
    pub escrow: Vec<u16>,

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashSet::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.clone()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let denomination_index = DENOMINATIONS.iter().position(|&d| d == coin);
        if denomination_index.is_none() {
            return Err(Error::BadCoin { coin });
        }

        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }

        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match self.selected.as_deref() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };

        let product = self
            .products
            .iter()
            .find(|product| product.id == selected_id)
            .expect("selected product is always in the catalog");
        let price = product.price;
        let id = product.id.clone();

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for (index, &denomination) in DENOMINATIONS.iter().enumerate() {
            let count = change.iter().filter(|&&coin| coin == denomination).count() as u16;
            proposed_tubes[index] -= count;
        }

        for &coin in &self.escrow {
            let index = DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow contains only accepted denominations");
            proposed_tubes[index] += 1;
        }

        for (index, &count) in proposed_tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let descending_indices = [3usize, 2, 1, 0];
    let mut remaining = amount;
    let mut greedy_counts = [0u16; 4];

    for &index in &descending_indices {
        let denomination = DENOMINATIONS[index];
        let count = tubes[index].min(remaining / denomination);
        greedy_counts[index] = count;
        remaining -= count * denomination;
    }

    if remaining == 0 {
        return Some(counts_to_descending_coins(&greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    search_combinations(
        amount,
        tubes,
        &descending_indices,
        0,
        [0; 4],
        0,
        &mut best,
    );

    best.map(|counts| counts_to_descending_coins(&counts))
}

fn search_combinations(
    remaining: u16,
    tubes: &[u16; 4],
    order: &[usize; 4],
    depth: usize,
    counts: [u16; 4],
    total_coins: u16,
    best: &mut Option<[u16; 4]>,
) {
    if depth == order.len() {
        if remaining == 0 {
            let is_better = match best {
                None => true,
                Some(current) => {
                    let current_total: u16 = current.iter().sum();
                    total_coins < current_total
                        || (total_coins == current_total
                            && lexicographically_greater_desc(&counts, current, order))
                }
            };
            if is_better {
                *best = Some(counts);
            }
        }
        return;
    }

    let index = order[depth];
    let denomination = DENOMINATIONS[index];
    let max_count = tubes[index].min(remaining / denomination);

    for count in 0..=max_count {
        let mut next_counts = counts;
        next_counts[index] = count;
        search_combinations(
            remaining - count * denomination,
            tubes,
            order,
            depth + 1,
            next_counts,
            total_coins + count,
            best,
        );
    }
}

fn lexicographically_greater_desc(
    candidate: &[u16; 4],
    current: &[u16; 4],
    order: &[usize; 4],
) -> bool {
    for &index in order {
        if candidate[index] != current[index] {
            return candidate[index] > current[index];
        }
    }
    false
}

fn counts_to_descending_coins(counts: &[u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for &index in &[3usize, 2, 1, 0] {
        for _ in 0..counts[index] {
            coins.push(DENOMINATIONS[index]);
        }
    }
    coins
}

                    coin: DENOMINATIONS[index],

        Ok(())

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;

    for i in (0..4).rev() {
        let denomination = DENOMINATIONS[i];
        let count = tubes[i].min(remaining / denomination);
        greedy[i] = count;
        remaining -= count * denomination;
    }

    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;

    fn search(
        index: usize,
        remaining: u16,
        tubes: &[u16; 4],
        current: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
        best_count: &mut u32,
    ) {
        if index == 0 {
            if remaining <= tubes[0] {
                current[0] = remaining;
                let total: u32 = current.iter().map(|&count| u32::from(count)).sum();
                let is_better = total < *best_count
                    || (total == *best_count
                        && best
                            .as_ref()
                            .map_or(true, |old| (0..4).rev().find_map(|i| {
                                if current[i] != old[i] {
                                    Some(current[i] > old[i])
                                } else {
                                    None
                                }
                            }).unwrap_or(false)));
                if is_better {
                    *best = Some(*current);
                    *best_count = total;
                }
            }
            return;
        }

        let denomination = DENOMINATIONS[index];
        let max_count = tubes[index].min(remaining / denomination);
        for count in 0..=max_count {
            current[index] = count;
            search(
                index - 1,
                remaining - count * denomination,
                tubes,
                current,
                best,
                best_count,
            );
        }
        current[index] = 0;
    }

    search(
        3,
        amount,
        tubes,
        &mut [0; 4],
        &mut best,
        &mut best_count,
    );

    best
}

            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !Self::DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.as_ref().ok_or(Error::NoSelection)?.clone();
        let product = self
            .products
            .iter()
            .find(|product| product.id == id)
            .expect("selected IDs always belong to the catalog");
        let price = product.price;

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = self.choose_change(amount)?;
        let mut proposed_tubes = self.tubes;

        for &coin in &change {
            let index = Self::DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("change contains only supported denominations");
            proposed_tubes[index] -= 1;
        }

        for &coin in &self.escrow {
            let index = Self::DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow contains only supported denominations");
            proposed_tubes[index] += 1;
        }

        for (index, &count) in proposed_tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::DENOMINATIONS[index],
                });
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let mut returned = self.escrow.clone();
        returned.reverse();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }

    fn choose_change(&self, amount: u16) -> Result<Vec<u16>, Error> {
        if amount == 0 {
            return Ok(Vec::new());
        }

        let mut remaining = amount;
        let mut greedy = Vec::new();
        for index in (0..4).rev() {
            let coin = Self::DENOMINATIONS[index];
            let take = self.tubes[index].min(remaining / coin);
            for _ in 0..take {
                greedy.push(coin);
            }
            remaining -= take * coin;
        }

        if remaining == 0 {
            return Ok(greedy);
        }

        let mut best: Option<[u16; 4]> = None;
        let mut chosen = [0u16; 4];
        Self::search_change(
            3,
            amount,
            &self.tubes,
            &mut chosen,
            &mut best,
        );

        let counts = best.ok_or(Error::NoChange { amount })?;
        let mut result = Vec::new();
        for index in (0..4).rev() {
            for _ in 0..counts[index] {
                result.push(Self::DENOMINATIONS[index]);
            }
        }
        Ok(result)
    }

    fn search_change(
        index: isize,
        remaining: u16,
        tubes: &[u16; 4],
        chosen: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
    ) {
        if remaining == 0 {
            let candidate_coins: u32 = chosen.iter().map(|&count| u32::from(count)).sum();
            let is_better = match best {
                None => true,
                Some(current) => {
                    let current_coins: u32 =
                        current.iter().map(|&count| u32::from(count)).sum();
                    candidate_coins < current_coins
                        || (candidate_coins == current_coins
                            && (0..4).rev().find_map(|i| {
                                if chosen[i] != current[i] {
                                    Some(chosen[i] > current[i])
                                } else {
                                    None
                                }
                            }) == Some(true))
                }
            };
            if is_better {
                *best = Some(*chosen);
            }
            return;
        }

        if index < 0 {
            return;
        }

        let i = index as usize;
        let coin = Self::DENOMINATIONS[i];
        let max_take = tubes[i].min(remaining / coin);
        for take in 0..=max_take {
            chosen[i] = take;
            Self::search_change(
                index - 1,
                remaining - take * coin,
                tubes,
                chosen,
                best,
            );
        }
        chosen[i] = 0;
    }
}
