#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let route = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(route);
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (segments, trailing_slash) = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>, &[u8])> = None;
        for (id, route) in self.routes.iter().enumerate() {
            if route.trailing_slash != trailing_slash {
                continue;
            }
            let Some(params) = match_route(&route.segments, &segments) else {
                continue;
            };
            match best {
                None => best = Some((id, params, &route.rank)),
                Some((_, _, rank)) if route.rank.as_slice() > rank => {
                    best = Some((id, params, &route.rank));
                }
                _ => {}
            }
        }

        best.map(|(route, params, _)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

fn parse_pattern(pattern: &str) -> Result<Route, Error> {
    let (raw_segments, trailing_slash) = split_head(pattern, true)?;
    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = std::collections::HashSet::new();
    let mut rank = Vec::new();

    for (index, (raw, start)) in raw_segments.iter().enumerate() {
        if raw.starts_with(':') || raw.starts_with('*') {
            let wildcard = raw.as_bytes()[0] == b'*';
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if wildcard && index + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            segments.push(if wildcard {
                rank.push(0);
                Segment::Wildcard(name.to_owned())
            } else {
                rank.push(1);
                Segment::Param(name.to_owned())
            });
        } else {
            let decoded = decode_segment(raw, *start).map_err(|at| Error::InvalidPattern { at })?;
            rank.push(2);
            segments.push(Segment::Static(decoded));
        }
    }

    if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
        rank.push(3);
    }

    Ok(Route {
        segments,
        trailing_slash,
        rank,
    })
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    let (raw_segments, trailing_slash) = split_head(path, false)?;
    let mut segments = Vec::with_capacity(raw_segments.len());
    for (raw, start) in raw_segments {
        segments.push(decode_segment(raw, start).map_err(|at| Error::InvalidPath { at })?);
    }
    Ok((segments, trailing_slash))
}

fn split_head(input: &str, pattern: bool) -> Result<(Vec<(&str, usize)>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let mut result = Vec::new();
    let mut start = 1;
    let mut i = 1;

    while i < bytes.len() {
        if bytes[i] == b'/' {
            if i == start {
                return Err(invalid(start));
            }
            result.push((&input[start..i], start));
            start = i + 1;
        }
        i += 1;
    }

    let trailing_slash = bytes[bytes.len() - 1] == b'/';
    if trailing_slash {
        if start == bytes.len() {
            return Err(invalid(bytes.len() - 1));
        }
        result.push((&input[start..bytes.len() - 1], start));
    } else {
        result.push((&input[start..], start));
    }

    if trailing_slash {
        return Err(invalid(bytes.len() - 1));
    }

    Ok((result, false))
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, start: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(start + i);
            }
            let Some(high) = upper_hex(bytes[i + 1]) else {
                return Err(start + i);
            };
            let Some(low) = upper_hex(bytes[i + 2]) else {
                return Err(start + i);
            };
            let value = (high << 4) | low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(start + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(start + i);
            }
            decoded.push(b);
            i += 1;
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in pattern {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(path_index)?.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), path[path_index..].join("/")));
                return Some(params);
            }
        }
    }

    (path_index == path.len()).then_some(params)
}
