#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(open) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
                let _ = open;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after = next_i + n.len_utf8();
                    if n == '"' || n == '\\' || n == '$' {
                        word.push(n);
                        i = after;
                    } else if n == '\n' {
                        i = after;
                    } else {
                        word.push('\\');
                        i = next_i;
                    }
                } else if c == '$' {
                    active = true;
                    i = expand(input, i, &mut word, env)?;
                } else {
                    word.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if c == ' ' || c == '\t' || c == '\r' || c == '\n' {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after = next_i + n.len_utf8();
                    if n != '\n' {
                        active = true;
                        word.push(n);
                    }
                    i = after;
                } else if c == '$' {
                    let new_i = expand(input, i, &mut word, env)?;
                    active = true;
                    i = new_i;
                } else {
                    active = true;
                    word.push(c);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(
    input: &str,
    dollar: usize,
    word: &mut String,
    env: &[(&str, &str)],
) -> Result<usize, SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        word.push('$');
        return Ok(after_dollar);
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut scan = content_start;
        let mut close = None;

        while scan < input.len() {
            let c = input[scan..].chars().next().unwrap();
            let end = scan + c.len_utf8();
            if c == '\\' {
                if end < input.len() {
                    let skipped = input[end..].chars().next().unwrap();
                    scan = end + skipped.len_utf8();
                } else {
                    scan = end;
                }
            } else if c == '}' {
                close = Some(scan);
                break;
            } else {
                scan = end;
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let name_end_rel = contents
            .char_indices()
            .find_map(|(idx, ch)| (!is_name_continue(ch)).then_some(idx))
            .unwrap_or(contents.len());

        if name_end_rel == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        let name = &contents[..name_end_rel];
        let rest = &contents[name_end_rel..];

        if !name.chars().next().map(is_name_start).unwrap_or(false) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let selected = if rest.is_empty() {
            lookup(name, env).unwrap_or("")
        } else if let Some(default) = rest.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
            .as_str()
            .to_owned()
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        word.push_str(&selected);
        Ok(close + 1)
    } else if is_name_start(first) {
        let mut end = after_dollar + first.len_utf8();
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if is_name_continue(c) {
                end += c.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        word.push_str(lookup(name, env).unwrap_or(""));
        Ok(end)
    } else {
        word.push('$');
        Ok(after_dollar)
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        let next = i + c.len_utf8();
        if c == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            out.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            out.push(c);
            i = next;
        }
    }
    out
}
