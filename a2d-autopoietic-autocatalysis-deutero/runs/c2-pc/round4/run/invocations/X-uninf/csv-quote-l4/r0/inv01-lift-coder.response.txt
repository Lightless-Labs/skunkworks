#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|v| v == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut row = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: unquoted, 1: quoted, 2: after closing quote
    let mut quote_start = 0usize;
    let mut record = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if state != 1 && b == b'\r' && (i + 1 == bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if state == 1 {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                    continue;
                }
                state = 2;
                i += 1;
                continue;
            }
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
            continue;
        }

        if state == 2 {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                state = 0;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record += 1;
                state = 0;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record, byte: i });
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if !field.is_empty() {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
            quote_start = i;
            state = 1;
            i += 1;
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
        }
    }

    if state == 1 {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if !row.is_empty() || !field.is_empty() || state == 2 || bytes.last() == Some(&delimiter) {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(n, _)| *n == name) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    if records.is_empty() {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    }

    let mut iter = records.into_iter();
    let header = iter.next().unwrap();
    let rows: Vec<Vec<String>> = iter.collect();

    if config.duplicates == DuplicatePolicy::Reject {
        if let Some(err) = duplicate_header(&header) {
            return Err(err);
        }
    }

    for (idx, row) in rows.iter().enumerate() {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: idx + 1,
                expected: header.len(),
                actual: row.len(),
            });
        }
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn valid_integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start == 0 && b.len() == 1;
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn valid_decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    let dot = match b.iter().position(|&c| c == b'.') {
        Some(pos) => pos,
        None => return false,
    };
    if dot < start || b.len() != dot + 3 {
        return false;
    }
    let whole = &b[start..dot];
    if whole.is_empty()
        || !whole.iter().all(u8::is_ascii_digit)
        || (whole.len() > 1 && whole[0] == b'0')
    {
        return false;
    }
    if b[dot + 1..].iter().any(|c| !c.is_ascii_digit()) {
        return false;
    }
    !(start == 1 && whole == b"0" && b[dot + 1..] == *b"00")
}

fn parse_decimal_value(s: &str) -> Option<i64> {
    let negative = s.starts_with('-');
    let unsigned = if negative { &s[1..] } else { s };
    let dot = unsigned.find('.')?;
    let whole = unsigned[..dot].parse::<i64>().ok()?;
    let frac = unsigned[dot + 1..].parse::<i64>().ok()?;
    let abs = whole.checked_mul(100)?.checked_add(frac)?;
    if negative {
        abs.checked_neg()
    } else {
        Some(abs)
    }
}

fn evaluate_cell(
    table: &Table,
    row_index: usize,
    spec_index: usize,
    spec: &ColumnSpec,
) -> Result<Value, CellError> {
    let column = match &spec.column {
        ColumnRef::Index(index) => Some(*index),
        ColumnRef::Name(name) => table.column_index(name),
    };
    let Some(column) = column else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: None,
            kind: CellErrorKind::UnknownColumn,
        });
    };

    let Some(cell) = table.rows[row_index].get(column) else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: Some(column),
            kind: CellErrorKind::MissingCell,
        });
    };

    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind: CellErrorKind::Empty,
            });
        }
    }

    let invalid = |kind| CellError {
        row: row_index,
        spec: spec_index,
        column: Some(column),
        kind,
    };

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.clone())),
        ColumnType::Integer => {
            if !valid_integer_syntax(cell) {
                return Err(invalid(CellErrorKind::InvalidInteger));
            }
            cell.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| invalid(CellErrorKind::IntegerOverflow))
        }
        ColumnType::Decimal => {
            if !valid_decimal_syntax(cell) {
                return Err(invalid(CellErrorKind::InvalidDecimal));
            }
            parse_decimal_value(cell)
                .map(Value::Decimal)
                .ok_or_else(|| invalid(CellErrorKind::DecimalOverflow))
        }
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(invalid(CellErrorKind::InvalidBoolean)),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for row_index in 0..table.rows.len() {
        let mut output = vec![None; columns.len()];
        let mut failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if policy == ErrorPolicy::Stop && !result.errors.is_empty() {
                break;
            }
            if failed && policy == ErrorPolicy::FirstPerRow {
                break;
            }

            match evaluate_cell(table, row_index, spec_index, spec) {
                Ok(value) => output[spec_index] = Some(value),
                Err(error) => {
                    result.errors.push(error);
                    if policy == ErrorPolicy::FirstPerRow {
                        failed = true;
                    }
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(output);
                        return result;
                    }
                }
            }
        }

        result.rows.push(output);
    }

    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns {
        record: usize,
    },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn records_of(table: &Table) -> Vec<&[String]> {
    let mut records = Vec::new();
    if let Some(header) = &table.header {
        records.push(header.as_slice());
    }
    records.extend(table.rows.iter().map(Vec::as_slice));
    records
}

fn preflight_write(table: &Table) -> Result<(), WriteError> {
    for (record, row) in records_of(table).iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (idx, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: idx + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in records_of(table).iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    Ok(())
}

fn quote_field(field: &str, delimiter: u8, quote: u8, first_field_of_first_record: bool, sole: bool) -> String {
    let bytes = field.as_bytes();
    let starts_or_ends_space_tab = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let starts_bom = first_field_of_first_record && field.starts_with('\u{feff}');
    let must_quote = bytes.iter().any(|&b| b == delimiter || b == quote || b == b'\r' || b == b'\n')
        || starts_or_ends_space_tab
        || starts_bom
        || (sole && field.is_empty());

    if !must_quote {
        return field.to_string();
    }

    let mut out = String::with_capacity(field.len() + 2);
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
        }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let order = left.as_bytes().cmp(right.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;
    preflight_write(table)?;

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some(CsvError::DuplicateHeader { name, first, second }) =
            duplicate_header(table.header.as_ref().unwrap())
        {
            return Err(WriteError::DuplicateHeader {
                name,
                first,
                second,
            });
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();

    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                output_records.push(header.clone());
            }
            output_records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained: Vec<usize> = Vec::new();
                for (idx, name) in header.iter().enumerate() {
                    let chosen = match table.duplicates {
                        DuplicatePolicy::KeepLast => {
                            header.iter().rposition(|n| n == name) == Some(idx)
                        }
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            header.iter().position(|n| n == name) == Some(idx)
                        }
                    };
                    if chosen {
                        retained.push(idx);
                    }
                }
                retained.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));
                output_records.push(retained.iter().map(|&idx| header[idx].clone()).collect());
                let mut rows: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| retained.iter().map(|&idx| row[idx].clone()).collect())
                    .collect();
                rows.sort_by(|a, b| compare_rows(a, b));
                output_records.extend(rows);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_rows(a, b));
                output_records = rows;
            }
        }
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let terminator = match config.mode {
        WriteMode::Canonical => "\n",
        WriteMode::Preserve => match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        },
    };

    let mut out = String::new();
    for (record_index, row) in output_records.iter().enumerate() {
        if record_index > 0 {
            out.push_str(terminator);
        }
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 {
                out.push(config.delimiter as char);
            }
            out.push_str(&quote_field(
                field,
                config.delimiter,
                config.quote,
                record_index == 0 && column_index == 0,
                row.len() == 1,
            ));
        }
    }

    match config.mode {
        WriteMode::Canonical => out.push('\n'),
        WriteMode::Preserve if config.final_terminator => out.push_str(terminator),
        WriteMode::Preserve => {}
    }

    Ok(out)
}
