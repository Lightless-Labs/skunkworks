fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_' || b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(offset + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(offset + i)?;
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(offset + i);
            }
            decoded.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !input.starts_with('/') {
        return Err(invalid(0));
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();

    let mut start = 1;
    while start < end {
        let mut finish = start;
        while finish < end && bytes[finish] != b'/' {
            finish += 1;
        }
        if finish == start {
            return Err(invalid(start));
        }
        let raw = &input[start..finish];

        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(invalid(start));
            }
            if !names.insert(name.to_owned()) {
                return Err(invalid(start));
            }
            let segment = if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                if finish < end {
                    return Err(invalid(start));
                }
                Segment::Wildcard(name.to_owned())
            };
            segments.push(segment);
        } else {
            let decoded = decode_segment(raw, start).map_err(invalid)?;
            segments.push(Segment::Static(decoded));
        }
        start = finish + 1;
    }

    if segments.is_empty() {
        return Err(invalid(1));
    }
    Ok((segments, trailing))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_path(pattern, true)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method != method || route.segments.len() != segments.len() {
                continue;
            }
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            let same_shape = route.segments.iter().zip(&segments).all(|(a, b)| match (a, b) {
                (Segment::Static(x), Segment::Static(y)) => x == y,
                (Segment::Param(_), Segment::Param(_))
                | (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
                _ => false,
            });
            if same_shape {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, request_trailing) = parse_path(head, false)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>, Vec<u8>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != request_trailing {
                continue;
            }
            if let Some((params, rank)) = match_route(route, &request_segments) {
                candidates.push((route, params, rank));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };
        let eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(route, _, _)| route.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for (route, _, _) in candidates_for_allowed(self, head, request_trailing) {
                allowed.insert(route.method.clone());
            }
            if allowed.contains("GET") {
                allowed.insert("HEAD".to_owned());
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let mut best = 0;
        for i in 1..eligible.len() {
            if eligible[i].2 > eligible[best].2 {
                best = i;
            }
        }
        let (route, params, _) = &eligible[best];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !request_segments.is_empty()
        {
            if route.trailing {
                Some(format!("{head}/"))
            } else {
                Some(head.strip_suffix('/').unwrap_or(head).to_owned())
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|route| route.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.insert(*key, *value).is_some() {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let required: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|segment| match segment {
                Segment::Param(name) | Segment::Wildcard(name) => Some(name.as_str()),
                Segment::Static(_) => None,
            })
            .collect();
        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }
        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !required.contains(key))
            .collect();
        extras.sort_unstable();
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Param(name) => {
                    let value = supplied[name.as_str()];
                    if value.is_empty()
                        || value.bytes().any(|b| !is_printable_ascii(b) || b == b'/')
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if value.bytes().any(|b| !is_printable_ascii(b))
                        || (!value.is_empty()
                            && value.split('/').any(|part| part.is_empty()))
                    {
                        return Err(Error::InvalidValue { name: name.clone() });
                    }
                }
                Segment::Static(_) => {}
            }
        }

        if route.segments.is_empty() {
            return Ok("/".to_owned());
        }
        let mut output = String::new();
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    output.push('/');
                    output.push_str(&encode_component(value));
                }
                Segment::Param(name) => {
                    output.push('/');
                    output.push_str(&encode_component(supplied[name.as_str()]));
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        output.push('/');
                        for (i, part) in value.split('/').enumerate() {
                            if i > 0 {
                                output.push('/');
                            }
                            output.push_str(&encode_component(part));
                        }
                    }
                }
            }
        }
        if route.trailing && !output.is_empty() {
            output.push('/');
        }
        if output.is_empty() {
            output.push('/');
        }
        Ok(output)
    }
}

fn is_printable_ascii(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn encode_component(value: &str) -> String {
    let mut output = String::new();
    for b in value.bytes() {
        if is_unreserved(b) {
            output.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(b >> 4) as usize] as char);
            output.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    output
}

fn match_route(route: &Route, request: &[Segment]) -> Option<(Vec<(String, String)>, Vec<u8>)> {
    let wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
    let fixed = if wildcard {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };
    if (!wildcard && request.len() != fixed) || (wildcard && request.len() < fixed) {
        return None;
    }

    let mut params = Vec::new();
    let mut rank = Vec::new();
    for i in 0..fixed {
        match (&route.segments[i], &request[i]) {
            (Segment::Static(a), Segment::Static(b)) if a == b => rank.push(2),
            (Segment::Param(name), Segment::Static(value)) => {
                rank.push(1);
                params.push((name.clone(), value.clone()));
            }
            _ => return None,
        }
    }
    if wildcard {
        let Segment::Wildcard(name) = &route.segments[fixed] else {
            unreachable!()
        };
        rank.push(0);
        let value = request[fixed..]
            .iter()
            .map(|segment| match segment {
                Segment::Static(value) => value.as_str(),
                _ => "",
            })
            .collect::<Vec<_>>()
            .join("/");
        params.push((name.clone(), value));
    } else {
        rank.push(3);
    }
    Some((params, rank))
}

fn candidates_for_allowed<'a>(
    router: &'a Router,
    head: &str,
    request_trailing: bool,
) -> Vec<(&'a Route, Vec<(String, String)>, Vec<u8>)> {
    let Ok((request, _)) = parse_path(head, false) else {
        return Vec::new();
    };
    router
        .routes
        .iter()
        .filter(|route| {
            router.policy != SlashPolicy::Strict || route.trailing == request_trailing
        })
        .filter_map(|route| {
            match_route(route, &request).map(|(params, rank)| (route, params, rank))
        })
        .collect()
}

fn hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,

fn decode_segment(s: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let err = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(err(base + i));
            }
            let hi = hex(bytes[i + 1]).ok_or_else(|| err(base + i))?;
            let lo = hex(bytes[i + 2]).ok_or_else(|| err(base + i))?;
            let d = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&d) || d == b'/' || is_unreserved(d) {
                return Err(err(base + i));
            }
            out.push(d);
            i += 3;
        } else {
            if !b.is_ascii() || !is_unreserved(b) {
                return Err(err(base + i));
            }
            out.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn split_path<'a>(path: &'a str, pattern: bool) -> Result<(Vec<(usize, &'a str)>, bool), Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    if !path.starts_with('/') {
        return Err(invalid(0));
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = path.ends_with('/');
    let end = if trailing { path.len() - 1 } else { path.len() };
    let bytes = path.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;
    let mut i = 1;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(invalid(start));
            }
            parts.push((start, &path[start..i]));
            start = i + 1;
        }
        i += 1;
    }
    Ok((parts, trailing))
}

fn parse_pattern(path: &str) -> Result<(Vec<Segment>, bool), Error> {
    let (parts, trailing) = split_path(path, true)?;
    let mut segments = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();
    for (idx, (at, part)) in parts.iter().enumerate() {
        if let Some(name) = part.strip_prefix(':') {
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: *at });
            }
            if !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: *at });
            }
            segments.push(Segment::Param(name.to_string()));
        } else if let Some(name) = part.strip_prefix('*') {
            if !valid_capture_name(name) || idx + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *at });
            }
            if !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: *at });
            }
            segments.push(Segment::Wildcard(name.to_string()));
        } else {
            segments.push(Segment::Static(decode_segment(part, *at, true)?));
        }
    }
    Ok((segments, trailing))
}

fn parse_request(path: &str) -> Result<(Vec<String>, bool), Error> {
    let head_end = path
        .char_indices()
        .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
        .unwrap_or(path.len());
    let head = &path[..head_end];
    let (parts, trailing) = split_path(head, false)?;
    let mut segments = Vec::with_capacity(parts.len());
    for (at, part) in parts {
        segments.push(decode_segment(part, at, false)?);
    }
    Ok((segments, trailing))
}

fn shape_conflicts(a: &Route, b: &Route, policy: SlashPolicy) -> bool {
    if a.method != b.method || a.segments.len() != b.segments.len() {
        return false;
    }
    if policy == SlashPolicy::Strict && a.trailing != b.trailing {
        return false;
    }
    a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
        (Segment::Static(a), Segment::Static(b)) => a == b,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn rank(route: &Route) -> Vec<u8> {
    let mut v = Vec::new();
    for seg in &route.segments {
        match seg {
            Segment::Static(_) => v.push(2),
            Segment::Param(_) => v.push(1),
            Segment::Wildcard(_) => {
                v.push(0);
                return v;
            }
        }
    }
    v.push(3);
    v
}

fn encode_component(s: &str) -> String {
    let mut out = String::new();
    for &b in s.as_bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }
        let id = self.routes.len();
        let candidate = Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
        };
        if let Some(route) = self
            .routes
            .iter()
            .filter(|route| shape_conflicts(route, &candidate, self.policy))
            .min_by_key(|route| route.id)
        {
            return Err(Error::Conflict { with: route.id });
        }
        self.routes.push(candidate);
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing) = parse_request(path)?;
        let path_matches = |route: &&Route| {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                return false;
            }
            let wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
            if (!wildcard && route.segments.len() != segments.len())
                || (wildcard && segments.len() + 1 < route.segments.len())
            {
                return false;
            }
            for (i, seg) in route.segments.iter().enumerate() {
                match seg {
                    Segment::Static(s) => {
                        if segments.get(i) != Some(s) {
                            return false;
                        }
                    }
                    Segment::Param(_) => {
                        if segments.get(i).map_or(true, |s| s.is_empty()) {
                            return false;
                        }
                    }
                    Segment::Wildcard(_) => {}
                }
            }
            true
        };

        let candidates: Vec<&Route> = self.routes.iter().filter(path_matches).collect();
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<&Route> = if method == "HEAD" {
            let heads: Vec<&Route> = candidates
                .iter()
                .copied()
                .filter(|r| r.method == "HEAD")
                .collect();
            if heads.is_empty() {
                candidates
                    .iter()
                    .copied()
                    .filter(|r| r.method == "GET")
                    .collect()
            } else {
                heads
            }
        } else {
            candidates
                .iter()
                .copied()
                .filter(|r| r.method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for route in &candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_string());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let winner = eligible
            .into_iter()
            .max_by(|a, b| {
                rank(a)
                    .cmp(&rank(b))
                    .then_with(|| b.id.cmp(&a.id))
            })
            .unwrap();

        let mut params = Vec::new();
        for (i, seg) in winner.segments.iter().enumerate() {
            match seg {
                Segment::Param(name) => params.push((name.clone(), segments[i].clone())),
                Segment::Wildcard(name) => {
                    let tail = segments[i..].join("/");
                    params.push((name.clone(), tail));
                }
                Segment::Static(_) => {}
            }
        }

        let redirect = if self.policy == SlashPolicy::Redirect
            && !segments.is_empty()
            && winner.trailing != trailing
        {
            let mut head = path
                .char_indices()
                .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
                .map_or(path, |i| &path[..i])
                .to_string();
            if winner.trailing {
                if !head.ends_with('/') {
                    head.push('/');
                }
            } else if head.ends_with('/') {
                head.pop();
            }
            Some(head)
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.contains_key(*key) {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
            supplied.insert(*key, *value);
        }

        let mut capture_names = Vec::new();
        for seg in &route.segments {
            match seg {
                Segment::Param(name) | Segment::Wildcard(name) => {
                    capture_names.push(name.as_str())
                }
                Segment::Static(_) => {}
            }
        }
        for key in &capture_names {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extras: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !capture_names.contains(key))
            .collect();
        extras.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extras.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for seg in &route.segments {
            let (name, wildcard) = match seg {
                Segment::Param(name) => (name, false),
                Segment::Wildcard(name) => (name, true),
                Segment::Static(_) => continue,
            };
            let value = supplied[name.as_str()];
            let bytes = value.as_bytes();
            let valid_ascii = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let valid_shape = if wildcard {
                value.is_empty()
                    || value
                        .split('/')
                        .all(|part| !part.is_empty())
            } else {
                !value.is_empty() && !value.contains('/')
            };
            if !valid_ascii || !valid_shape {
                return Err(Error::InvalidValue { name: name.clone() });
            }
        }

        if route.segments.is_empty() {
            return Ok("/".to_string());
        }
        let mut out = String::from("/");
        let mut emitted = 0usize;
        for seg in &route.segments {
            match seg {
                Segment::Static(value) => {
                    if emitted > 0 {
                        out.push('/');
                    }
                    out.push_str(&encode_component(value));
                    emitted += 1;
                }
                Segment::Param(name) => {
                    if emitted > 0 {
                        out.push('/');
                    }
                    out.push_str(&encode_component(supplied[name.as_str()]));
                    emitted += 1;
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        if emitted > 0 {
                            out.push('/');
                        }
                        let mut first = true;
                        for part in value.split('/') {
                            if !first {
                                out.push('/');
                            }
                            out.push_str(&encode_component(part));
                            first = false;
                        }
                        emitted += value.split('/').count();
                    }
                }
            }
        }
        if route.trailing && !out.ends_with('/') {
            out.push('/');
        }
        Ok(out)
    }
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let mut end = bytes.len();
    let trailing = bytes.last() == Some(&b'/');
    if trailing {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut names = HashSet::new();
    let mut start = 1usize;
    let mut wildcard_seen = false;

    while start < end {
        let mut stop = start;
        while stop < end && bytes[stop] != b'/' {
            stop += 1;
        }
        if stop == start {
            return Err(start);
        }

        let part = &bytes[start..stop];
        if pattern && (part[0] == b':' || part[0] == b'*') {
            let is_wild = part[0] == b'*';
            let name_bytes = &part[1..];
            let name = std::str::from_utf8(name_bytes).unwrap_or("");
            if !valid_capture_name(name) {
                return Err(start);
            }
            if !names.insert(name.to_string()) {
                return Err(start);
            }
            if is_wild {
                if stop < end {
                    return Err(start);
                }
                wildcard_seen = true;
                segments.push(Segment::Wildcard(name.to_string()));
            } else {
                segments.push(Segment::Param(name.to_string()));
            }
        } else {
            if wildcard_seen {
                return Err(start);
            }
            let decoded = decode_static(part, start)?;
            segments.push(Segment::Static(decoded));
        }

        if stop == end {
            break;
        }
        start = stop + 1;
    }

    Ok((segments, trailing))
}

            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];

        for route in &self.routes {

                continue;

            .iter()
            .filter(|(r, _, _)| r.method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = BTreeSet::new();
            for (route, _, _) in &path_candidates {
                allowed.insert(route.method.clone());
                if route.method == "GET" {
                    allowed.insert("HEAD".to_string());
                }
            }
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        eligible.sort_by(|a, b| b.2.cmp(&a.2).then_with(|| a.0.id.cmp(&b.0.id)));
        let (route, params, _) = eligible[0];

        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != request_trailing
            && !request_segments.is_empty()
        {
            let mut value = head.to_string();
            if route.trailing {
                value.push('/');
            } else if value.ends_with('/') {
                value.pop();
            }
            Some(value)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for (key, value) in params {
            if supplied.insert(*key, *value).is_some() {
                return Err(Error::DuplicateParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut required = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(n) | Segment::Wildcard(n) => required.push(n.as_str()),
                Segment::Static(_) => {}
            }
        }

        for key in &required {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let required_set: HashSet<&str> = required.iter().copied().collect();
        let extra = supplied
            .keys()
            .filter(|key| !required_set.contains(**key))
            .min_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extra {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for key in &required {
            let value = supplied[key];
            let valid_ascii = value.bytes().all(|b| (0x20..=0x7e).contains(&b));
            let wildcard = route.segments.iter().any(
                |s| matches!(s, Segment::Wildcard(n) if n == key),
            );
            let valid_shape = if wildcard {
                value.is_empty()
                    || value
                        .split('/')
                        .all(|part| !part.is_empty() && part.bytes().all(|b| b != b'/'))
            } else {
                !value.is_empty() && !value.contains('/')
            };
            if !valid_ascii || !valid_shape {
                return Err(Error::InvalidValue {
                    name: (*key).to_string(),
                });
            }
        }

        let mut output = String::from("/");
        let mut emitted = false;
        for segment in &route.segments {
            match segment {
                Segment::Static(s) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(s));
                    emitted = true;
                }
                Segment::Param(n) => {
                    if emitted {
                        output.push('/');
                    }
                    output.push_str(&encode_segment(supplied[n.as_str()]));
                    emitted = true;
                }
                Segment::Wildcard(n) => {
                    let value = supplied[n.as_str()];
                    if !value.is_empty() {
                        if emitted {
                            output.push('/');
                        }
                        let mut first = true;
                        for part in value.split('/') {
                            if !first {
                                output.push('/');
                            }
                            output.push_str(&encode_segment(part));
                            first = false;
                        }
                        emitted = true;
                    }
                }
            }
        }

        if !emitted {
            return Ok("/".to_string());
        }
        if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn match_route(route: &Route, request: &[Segment]) -> Option<Vec<(String, String)>> {
    let wildcard = route.has_wildcard;
    let fixed_len = if wildcard {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };
    if (!wildcard && request.len() != fixed_len) || (wildcard && request.len() < fixed_len) {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        let (Segment::Static(expected), Segment::Static(actual)) =
            (&route.segments[i], &request[i])
        else {
            match (&route.segments[i], &request[i]) {
                (Segment::Param(name), Segment::Static(value)) => {
                    params.push((name.clone(), value.clone()));
                }
                _ => return None,
            }
            continue;
        };
        if expected != actual {
            return None;
        }
    }

    if wildcard {
        if let Segment::Wildcard(name) = &route.segments[fixed_len] {
            let tail = request[fixed_len..]
                .iter()
                .map(|s| match s {
                    Segment::Static(v) => v.as_str(),
                    _ => "",
                })
                .collect::<Vec<_>>()
                .join("/");
            params.push((name.clone(), tail));
        }
    }
    Some(params)
}

fn rank(route: &Route) -> Vec<u8> {
    let mut result = Vec::new();
    for segment in &route.segments {
        result.push(match segment {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        });
        if matches!(segment, Segment::Wildcard(_)) {
            return result;
        }
    }
    result.push(3);
    result
}
