#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_literal(word: &mut Word, c: char, active: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_expansion(word: &mut Word, value: &str, active: bool) {
    for c in value.chars() {
        append_literal(word, c, active);
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    after_dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    if input[after_dollar..].starts_with('{') {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    cursor += input[cursor..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let mut name_end = 0;
        for (offset, c) in contents.char_indices() {
            if (offset == 0 && is_name_start(c)) || (offset > 0 && is_name_continue(c)) {
                name_end = offset + c.len_utf8();
            } else {
                break;
            }
        }

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let remainder = &contents[name_end..];
        let value = if remainder.is_empty() {
            lookup(name, env).unwrap_or("").to_string()
        } else if let Some(default) = remainder.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok((close + 1, value))
    } else {
        let mut cursor = after_dollar;
        let first = input[cursor..].chars().next().unwrap();
        if !is_name_start(first) {
            return Ok((after_dollar, "$".to_string()));
        }

        cursor += first.len_utf8();
        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if !is_name_continue(c) {
                break;
            }
            cursor += c.len_utf8();
        }

        let name = &input[after_dollar..cursor];
        Ok((cursor, lookup(name, env).unwrap_or("").to_string()))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active_word = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let next_cursor = cursor + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut word, c, false);
                }
                cursor = next_cursor;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    cursor = next_cursor;
                } else if c == '\\' {
                    let slash = cursor;
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let following = input[next_cursor..].chars().next().unwrap();
                    let after_following = next_cursor + following.len_utf8();
                    if matches!(following, '"' | '\\' | '$' | '\n') {
                        if following != '\n' {
                            append_literal(&mut word, following, false);
                        }
                        cursor = after_following;
                    } else {
                        append_literal(&mut word, '\\', false);
                        cursor = next_cursor;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    if input[next_cursor..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_cursor..].starts_with('{')
                        || input[next_cursor..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let (end, value) = parse_expansion(input, cursor, next_cursor, env)?;
                        append_expansion(&mut word, &value, false);
                        active_word = true;
                        cursor = end;
                    } else {
                        append_literal(&mut word, '$', false);
                        cursor = next_cursor;
                    }
                } else {
                    append_literal(&mut word, c, false);
                    cursor = next_cursor;
                }
                let _ = open;
            }
            Quote::None => {
                if is_separator(c) {
                    if active_word {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active_word = false;
                    }
                    cursor = next_cursor;
                } else if c == '\'' {
                    active_word = true;
                    quote = Quote::Single(cursor);
                    cursor = next_cursor;
                } else if c == '"' {
                    active_word = true;
                    quote = Quote::Double(cursor);
                    cursor = next_cursor;
                } else if c == '\\' {
                    let slash = cursor;
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let following = input[next_cursor..].chars().next().unwrap();
                    let after_following = next_cursor + following.len_utf8();
                    if following == '\n' {
                        cursor = after_following;
                    } else {
                        active_word = true;
                        append_literal(&mut word, following, false);
                        cursor = after_following;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    if input[next_cursor..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_cursor..].starts_with('{')
                        || input[next_cursor..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let (end, value) = parse_expansion(input, cursor, next_cursor, env)?;
                        append_expansion(&mut word, &value, true);
                        active_word = true;
                        cursor = end;
                    } else {
                        active_word = true;
                        append_literal(&mut word, '$', true);
                        cursor = next_cursor;
                    }
                } else {
                    active_word = true;
                    append_literal(&mut word, c, true);
                    cursor = next_cursor;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active_word {
        words.push(word);
    }
    Ok(words)
}
