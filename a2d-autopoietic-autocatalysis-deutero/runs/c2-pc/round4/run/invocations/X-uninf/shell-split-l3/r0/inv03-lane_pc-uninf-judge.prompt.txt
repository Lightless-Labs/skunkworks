#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append_scalar(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_string(word: &mut Word, value: &str, active_glob: bool) {
    for c in value.chars() {
        append_scalar(word, c, active_glob);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn next_char(input: &str, pos: usize) -> Option<(char, usize)> {
    input[pos..].chars().next().map(|c| (c, pos + c.len_utf8()))
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut pos = 0;
    while let Some((c, next)) = next_char(text, pos) {
        if c == '\\' {
            if let Some((escaped, after)) = next_char(text, next) {
                out.push(escaped);
                pos = after;
            } else {
                out.push('\\');
                pos = next;
            }
        } else {
            out.push(c);
            pos = next;
        }
    }
    out
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_open = dollar + 2;
    let mut pos = after_open;
    let mut close = None;

    while let Some((c, next)) = next_char(input, pos) {
        if c == '\\' {
            if let Some((_, after_escaped)) = next_char(input, next) {
                pos = after_escaped;
            } else {
                pos = next;
            }
        } else if c == '}' {
            close = Some(pos);
            break;
        } else {
            pos = next;
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_open..close];

    let name_end = contents
        .char_indices()
        .find_map(|(i, c)| (!is_name_continue(c)).then_some(i))
        .unwrap_or(contents.len());

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    if !name.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let rest = &contents[name_end..];
    let value = if rest.is_empty() {
        lookup(env, name).unwrap_or("").to_string()
    } else if let Some(default_text) = rest.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default_text),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_open = 0usize;
    let mut pos = 0usize;

    while let Some((c, next)) = next_char(input, pos) {
        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_scalar(&mut word, c, false);
                }
                pos = next;
                continue;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    pos = next;
                    continue;
                }
                if c == '\\' {
                    if let Some((escaped, after)) = next_char(input, next) {
                        match escaped {
                            '"' | '\\' | '$' => {
                                append_scalar(&mut word, escaped, false);
                                pos = after;
                            }
                            '\n' => pos = after,
                            _ => {
                                append_scalar(&mut word, '\\', false);
                                pos = next;
                            }
                        }
                    } else {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    continue;
                }
                if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                }
                if c == '$' {
                    if let Some((following, _)) = next_char(input, next) {
                        if following == '(' {
                            return Err(SplitError::CommandSubstitution { pos });
                        }
                        if following == '{' {
                            let (value, after) = parse_expansion(input, pos, env)?;
                            active = true;
                            append_string(&mut word, &value, false);
                            pos = after;
                            continue;
                        }
                        if is_name_start(following) {
                            let mut end = next;
                            while let Some((name_char, after)) = next_char(input, end) {
                                if is_name_continue(name_char) {
                                    end = after;
                                } else {
                                    break;
                                }
                            }
                            let value = lookup(env, &input[next..end]).unwrap_or("");
                            active = true;
                            append_string(&mut word, value, false);
                            pos = end;
                            continue;
                        }
                    }
                    append_scalar(&mut word, '$', false);
                    pos = next;
                    continue;
                }
                append_scalar(&mut word, c, false);
                pos = next;
                continue;
            }
            Quote::None => {}
        }

        match c {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    words.push(word);
                    word = Word {
                        text: String::new(),
                        glob: Vec::new(),
                    };
                    active = false;
                }
                pos = next;
            }
            '\'' => {
                active = true;
                quote = Quote::Single;
                quote_open = pos;
                pos = next;
            }
            '"' => {
                active = true;
                quote = Quote::Double;
                quote_open = pos;
                pos = next;
            }
            '\\' => {
                if let Some((escaped, after)) = next_char(input, next) {
                    if escaped == '\n' {
                        pos = after;
                    } else {
                        active = true;
                        append_scalar(&mut word, escaped, false);
                        pos = after;
                    }
                } else {
                    return Err(SplitError::TrailingEscape { pos });
                }
            }
            '`' => return Err(SplitError::CommandSubstitution { pos }),
            '$' => {
                if let Some((following, _)) = next_char(input, next) {
                    if following == '(' {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if following == '{' {
                        let (value, after) = parse_expansion(input, pos, env)?;
                        active = true;
                        append_string(&mut word, &value, true);
                        pos = after;
                        continue;
                    }
                    if is_name_start(following) {
                        let mut end = next;
                        while let Some((name_char, after)) = next_char(input, end) {
                            if is_name_continue(name_char) {
                                end = after;
                            } else {
                                break;
                            }
                        }
                        let value = lookup(env, &input[next..end]).unwrap_or("");
                        active = true;
                        append_string(&mut word, value, true);
                        pos = end;
                        continue;
                    }
                }
                active = true;
                append_scalar(&mut word, '$', true);
                pos = next;
            }
            _ => {
                active = true;
                append_scalar(&mut word, c, true);
                pos = next;
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_open }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_open }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_literal(word: &mut Word, c: char, active: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if active && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_expansion(word: &mut Word, value: &str, active: bool) {
    for c in value.chars() {
        append_literal(word, c, active);
    }
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    after_dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    if input[after_dollar..].starts_with('{') {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut close = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    cursor += input[cursor..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let mut name_end = 0;
        for (offset, c) in contents.char_indices() {
            if (offset == 0 && is_name_start(c)) || (offset > 0 && is_name_continue(c)) {
                name_end = offset + c.len_utf8();
            } else {
                break;
            }
        }

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let remainder = &contents[name_end..];
        let value = if remainder.is_empty() {
            lookup(name, env).unwrap_or("").to_string()
        } else if let Some(default) = remainder.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok((close + 1, value))
    } else {
        let mut cursor = after_dollar;
        let first = input[cursor..].chars().next().unwrap();
        if !is_name_start(first) {
            return Ok((after_dollar, "$".to_string()));
        }

        cursor += first.len_utf8();
        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if !is_name_continue(c) {
                break;
            }
            cursor += c.len_utf8();
        }

        let name = &input[after_dollar..cursor];
        Ok((cursor, lookup(name, env).unwrap_or("").to_string()))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active_word = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let next_cursor = cursor + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut word, c, false);
                }
                cursor = next_cursor;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    cursor = next_cursor;
                } else if c == '\\' {
                    let slash = cursor;
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let following = input[next_cursor..].chars().next().unwrap();
                    let after_following = next_cursor + following.len_utf8();
                    if matches!(following, '"' | '\\' | '$' | '\n') {
                        if following != '\n' {
                            append_literal(&mut word, following, false);
                        }
                        cursor = after_following;
                    } else {
                        append_literal(&mut word, '\\', false);
                        cursor = next_cursor;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    if input[next_cursor..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_cursor..].starts_with('{')
                        || input[next_cursor..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let (end, value) = parse_expansion(input, cursor, next_cursor, env)?;
                        append_expansion(&mut word, &value, false);
                        active_word = true;
                        cursor = end;
                    } else {
                        append_literal(&mut word, '$', false);
                        cursor = next_cursor;
                    }
                } else {
                    append_literal(&mut word, c, false);
                    cursor = next_cursor;
                }
                let _ = open;
            }
            Quote::None => {
                if is_separator(c) {
                    if active_word {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active_word = false;
                    }
                    cursor = next_cursor;
                } else if c == '\'' {
                    active_word = true;
                    quote = Quote::Single(cursor);
                    cursor = next_cursor;
                } else if c == '"' {
                    active_word = true;
                    quote = Quote::Double(cursor);
                    cursor = next_cursor;
                } else if c == '\\' {
                    let slash = cursor;
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let following = input[next_cursor..].chars().next().unwrap();
                    let after_following = next_cursor + following.len_utf8();
                    if following == '\n' {
                        cursor = after_following;
                    } else {
                        active_word = true;
                        append_literal(&mut word, following, false);
                        cursor = after_following;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if c == '$' {
                    if input[next_cursor..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[next_cursor..].starts_with('{')
                        || input[next_cursor..]
                            .chars()
                            .next()
                            .is_some_and(is_name_start)
                    {
                        let (end, value) = parse_expansion(input, cursor, next_cursor, env)?;
                        append_expansion(&mut word, &value, true);
                        active_word = true;
                        cursor = end;
                    } else {
                        active_word = true;
                        append_literal(&mut word, '$', true);
                        cursor = next_cursor;
                    }
                } else {
                    active_word = true;
                    append_literal(&mut word, c, true);
                    cursor = next_cursor;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active_word {
        words.push(word);
    }
    Ok(words)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_literal(word: &mut Word, c: char, wildcard_active: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_atomic(word: &mut Word, text: &str, wildcard_active: bool) {
    for c in text.chars() {
        append_literal(word, c, wildcard_active);
    }
}

fn first_env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(open_pos) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_literal(&mut word, c, false);
                }
                i = next_i;
                let _ = open_pos;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$') {
                        append_literal(&mut word, escaped, false);
                        i = after;
                    } else if escaped == '\n' {
                        i = after;
                    } else {
                        append_literal(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let after_dollar = next_i;
                    if input[after_dollar..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[after_dollar..].starts_with('{') {
                        let (end, value) = parse_braced(input, i, env)?;
                        append_atomic(&mut word, &value, false);
                        active = true;
                        i = end;
                    } else if let Some((name, end)) = parse_dollar_name(input, after_dollar) {
                        let value = first_env_value(env, name).unwrap_or("");
                        append_atomic(&mut word, value, false);
                        active = true;
                        i = end;
                    } else {
                        append_literal(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    append_literal(&mut word, c, false);
                    i = next_i;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if escaped == '\n' {
                        i = after;
                    } else {
                        append_literal(&mut word, escaped, false);
                        active = true;
                        i = after;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let after_dollar = next_i;
                    if input[after_dollar..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[after_dollar..].starts_with('{') {
                        let (end, value) = parse_braced(input, i, env)?;
                        append_atomic(&mut word, &value, true);
                        active = true;
                        i = end;
                    } else if let Some((name, end)) = parse_dollar_name(input, after_dollar) {
                        let value = first_env_value(env, name).unwrap_or("");
                        append_atomic(&mut word, value, true);
                        active = true;
                        i = end;
                    } else {
                        append_literal(&mut word, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_literal(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_dollar_name(input: &str, start: usize) -> Option<(&str, usize)> {
    let first = input.get(start..)?.chars().next()?;
    if !is_name_start(first) {
        return None;
    }
    let mut end = start + first.len_utf8();
    while end < input.len() {
        let c = input[end..].chars().next().unwrap();
        if !is_name_continue(c) {
            break;
        }
        end += c.len_utf8();
    }
    Some((&input[start..end], end))
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let content_start = dollar + 2;
    let mut cursor = content_start;
    let mut close = None;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let next = cursor + c.len_utf8();
        if c == '\\' {
            if next < input.len() {
                let escaped = input[next..].chars().next().unwrap();
                cursor = next + escaped.len_utf8();
            } else {
                cursor = next;
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor = next;
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let mut name_end = 0;
    let mut chars = contents.char_indices();
    let Some((_, first)) = chars.next() else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    if !is_name_start(first) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    name_end = first.len_utf8();
    while name_end < contents.len() {
        let c = contents[name_end..].chars().next().unwrap();
        if !is_name_continue(c) {
            break;
        }
        name_end += c.len_utf8();
    }

    let name = &contents[..name_end];
    let default = if name_end == contents.len() {
        None
    } else if contents[name_end..].starts_with(":-") {
        Some(&contents[name_end + 2..])
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    let value = match default {
        None => first_env_value(env, name).unwrap_or("").to_string(),
        Some(text) => match first_env_value(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(text),
        },
    };

    Ok((close + 1, value))
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        let next = i + c.len_utf8();
        if c == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            result.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            result.push(c);
            i = next;
        }
    }
    result
}
