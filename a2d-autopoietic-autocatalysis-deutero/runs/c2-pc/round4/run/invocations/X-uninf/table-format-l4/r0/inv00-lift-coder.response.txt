#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

type LogicalLine = Vec<Token>;
type CellLines = Vec<LogicalLine>;

fn checked_add(a: usize, b: usize) -> usize {
    a.saturating_add(b)
}

fn checked_mul(a: usize, b: usize) -> usize {
    a.saturating_mul(b)
}

fn invalid_char(ch: char) -> bool {
    if ch == '\r' || ch == '\u{2028}' || ch == '\u{2029}' {
        return true;
    }
    ch.is_control() && ch != '\n'
}

fn validate(rows: &[Vec<String>]) -> Result<(), TableError> {
    for (row_index, row) in rows.iter().enumerate() {
        for (column_index, cell) in row.iter().enumerate() {
            let mut iter = cell.char_indices().peekable();
            while let Some((byte, ch)) = iter.next() {
                if ch == '\r' {
                    if matches!(iter.peek(), Some((_, '\n'))) {
                        iter.next();
                        continue;
                    }
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
                if invalid_char(ch) {
                    return Err(TableError::InvalidChar {
                        row: row_index,
                        column: column_index,
                        byte,
                    });
                }
            }
        }
    }
    Ok(())
}

fn tokenize_line(line: &str, markdown: bool) -> LogicalLine {
    let trimmed = line.trim_end_matches(' ');
    let mut tokens = Vec::new();

    for ch in trimmed.chars() {
        if markdown && (ch == '|' || ch == '\\') {
            let mut text = String::new();
            text.push('\\');
            text.push(ch);
            tokens.push(Token { text, width: 2 });
        } else {
            tokens.push(Token {
                text: ch.to_string(),
                width: 1,
            });
        }
    }

    tokens
}

fn normalize_cell(cell: &str, markdown: bool) -> CellLines {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| tokenize_line(line, markdown))
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().fold(0usize, |sum, token| checked_add(sum, token.width))
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            let total = token_width(tokens);
            if total <= width {
                return vec![tokens.to_vec()];
            }

            let mut result = Vec::new();
            let mut used = 0usize;
            let limit = width.saturating_sub(1);
            for token in tokens {
                let next = checked_add(used, token.width);
                if next > limit {
                    break;
                }
                used = next;
                result.push(token.clone());
            }
            result.push(Token {
                text: "…".to_string(),
                width: 1,
            });
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0usize;

            while start < tokens.len() {
                let mut end = start;
                let mut used = 0usize;
                while end < tokens.len() {
                    let next = checked_add(used, tokens[end].width);
                    if next > width {
                        break;
                    }
                    used = next;
                    end += 1;
                }
                // The minimum width guarantees every token fits.
                if end == start {
                    end += 1;
                }
                result.push(tokens[start..end].to_vec());
                start = end;
            }

            result
        }
    }
}

fn tokens_text(tokens: &[Token]) -> String {
    let mut output = String::new();
    for token in tokens {
        output.push_str(&token.text);
    }
    output
}

fn aligned(text: &str, content_width: usize, width: usize, alignment: &Alignment) -> String {
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };

    let mut result = String::new();
    result.extend(std::iter::repeat_n(' ', left));
    result.push_str(text);
    result.extend(std::iter::repeat_n(' ', right));
    result
}

fn line_content(
    fragments_for_cells: &[Vec<Vec<Token>>],
    fragment_index: usize,
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) -> String {
    let mut output = String::new();

    if !matches!(mode, Mode::Compact) {
        output.push('|');
    }

    for column in 0..widths.len() {
        let fragment = fragments_for_cells[column]
            .get(fragment_index)
            .map(Vec::as_slice)
            .unwrap_or(&[]);
        let text = tokens_text(fragment);
        let content_width = token_width(fragment);
        let padded = aligned(&text, content_width, widths[column], &alignments[column]);

        if matches!(mode, Mode::Compact) {
            output.push_str(&padded[..0]); // Keep construction in scalar space below.
            let (left, right) = match alignments[column] {
                Alignment::Left => (0, widths[column].saturating_sub(content_width)),
                Alignment::Right => (widths[column].saturating_sub(content_width), 0),
                Alignment::Center => {
                    let extra = widths[column].saturating_sub(content_width);
                    (extra / 2, extra - extra / 2)
                }
            };
            output.extend(std::iter::repeat_n(' ', left));
            output.push_str(&text);
            if column + 1 < widths.len() {
                output.extend(std::iter::repeat_n(' ', right));
                output.push_str("  ");
            }
        } else {
            output.push(' ');
            output.push_str(&padded);
            output.push(' ');
            output.push('|');
        }
    }

    output
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row_index, row) in rows.iter().enumerate() {
        if row.is_empty() {
            return Err(TableError::EmptyRow { row: row_index });
        }
    }

    let column_count = rows[0].len();
    for (row_index, row) in rows.iter().enumerate() {
        if row.len() != column_count {
            return Err(TableError::Ragged {
                row: row_index,
                expected: column_count,
                actual: row.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != column_count {
        return Err(TableError::AlignmentCount {
            expected: column_count,
            actual: options.alignments.len(),
        });
    }

    if !options.overflow.is_empty() && options.overflow.len() != column_count {
        return Err(TableError::OverflowCount {
            expected: column_count,
            actual: options.overflow.len(),
        });
    }

    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_column_width = if matches!(options.mode, Mode::Markdown) {
        3
    } else {
        1
    };
    let overhead = if matches!(options.mode, Mode::Compact) {
        checked_mul(2, column_count.saturating_sub(1))
    } else {
        checked_add(checked_mul(3, column_count), 1)
    };
    let minimum = checked_add(overhead, checked_mul(column_count, minimum_column_width));

    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    validate(rows)?;

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; column_count]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; column_count]
    } else {
        options.overflow.clone()
    };

    let markdown = matches!(options.mode, Mode::Markdown);
    let normalized: Vec<Vec<CellLines>> = rows
        .iter()
        .map(|row| row.iter().map(|cell| normalize_cell(cell, markdown)).collect())
        .collect();

    let mut widths = vec![minimum_column_width; column_count];
    for row in &normalized {
        for (column, cell_lines) in row.iter().enumerate() {
            for line in cell_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total = checked_add(
            overhead,
            widths.iter().fold(0usize, |sum, width| checked_add(sum, *width)),
        );
        while total > options.max_width {
            let mut selected: Option<usize> = None;
            for column in 0..column_count {
                if widths[column] > minimum_column_width
                    && selected.map_or(true, |current| widths[column] >= widths[current])
                {
                    selected = Some(column);
                }
            }
            if let Some(column) = selected {
                widths[column] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();

    let emit_border = |out: &mut String, widths: &[usize]| {
        out.push('+');
        for width in widths {
            out.extend(std::iter::repeat_n('-', width + 2));
            out.push('+');
        }
        out.push('\n');
    };

    if matches!(options.mode, Mode::Grid) {
        emit_border(&mut output, &widths);
    }

    for (row_index, row) in normalized.iter().enumerate() {
        let cell_fragments: Vec<Vec<Vec<Token>>> = row
            .iter()
            .enumerate()
            .map(|(column, cell_lines)| {
                let mut all = Vec::new();
                for line in cell_lines {
                    all.extend(fragments(
                        line,
                        widths[column],
                        &overflows[column],
                    ));
                }
                all
            })
            .collect();

        let row_height = cell_fragments
            .iter()
            .map(Vec::len)
            .max()
            .unwrap_or(1);

        for fragment_index in 0..row_height {
            output.push_str(&line_content(
                &cell_fragments,
                fragment_index,
                &widths,
                &alignments,
                &options.mode,
            ));
            output.push('\n');
        }

        if matches!(options.mode, Mode::Grid)
            && options.header
            && row_index == 0
            && rows.len() > 1
        {
            emit_border(&mut output, &widths);
        }
    }

    if matches!(options.mode, Mode::Grid) {
        emit_border(&mut output, &widths);
    } else if matches!(options.mode, Mode::Markdown) {
        output.push('|');
        for column in 0..column_count {
            output.push(' ');
            match alignments[column] {
                Alignment::Left => {
                    output.push(':');
                    output.extend(std::iter::repeat_n('-', widths[column] - 1));
                }
                Alignment::Right => {
                    output.extend(std::iter::repeat_n('-', widths[column] - 1));
                    output.push(':');
                }
                Alignment::Center => {
                    output.push(':');
                    output.extend(std::iter::repeat_n('-', widths[column] - 2));
                    output.push(':');
                }
            }
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }

    Ok(output)
}
