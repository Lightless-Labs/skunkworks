#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        let outside_quotes = !matches!(state, State::Quoted { .. });
        if outside_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::Quoted { opening } => {
                if b == b'"' {
                    if bytes.get(i + 1) == Some(&b'"') {
                        field.push('"');
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().expect("valid UTF-8");
                    field.push(ch);
                    i += ch.len_utf8();
                }
                let _ = opening;
            }
            State::FieldStart | State::Unquoted | State::AfterQuote => {
                match b {
                    b',' => {
                        record.push(std::mem::take(&mut field));
                        state = State::FieldStart;
                        i += 1;
                    }
                    b'\n' => {
                        record.push(std::mem::take(&mut field));
                        records.push(std::mem::take(&mut record));
                        record_index += 1;
                        state = State::FieldStart;
                        i += 1;
                    }
                    b'\r' => {
                        record.push(std::mem::take(&mut field));
                        records.push(std::mem::take(&mut record));
                        record_index += 1;
                        state = State::FieldStart;
                        i += 2;
                    }
                    b'"' => match state {
                        State::FieldStart => {
                            state = State::Quoted { opening: i };
                            i += 1;
                        }
                        State::Unquoted => {
                            return Err(CsvError::UnexpectedQuote {
                                record: record_index,
                                byte: i,
                            });
                        }
                        State::AfterQuote => {
                            return Err(CsvError::TrailingByte {
                                record: record_index,
                                byte: i,
                            });
                        }
                        State::Quoted { .. } => unreachable!(),
                    },
                    _ if matches!(state, State::AfterQuote) => {
                        return Err(CsvError::TrailingByte {
                            record: record_index,
                            byte: i,
                        });
                    }
                    _ => {
                        let ch = input[i..].chars().next().expect("valid UTF-8");
                        field.push(ch);
                        state = State::Unquoted;
                        i += ch.len_utf8();
                    }
                }
            }
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    record.push(field);
    records.push(record);
    Ok(records)
}
