#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

#[derive(Debug, Clone)]
pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let rank = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
            })
            .collect();
        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let path_head = &path[..head_end];
        let parts = parse_path(path_head)?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;
        for route in &self.routes {
            if route.segments.len() != parts.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matched = true;
            for (pattern_segment, value) in route.segments.iter().zip(parts.iter()) {
                match pattern_segment {
                    Segment::Static(expected) if expected != value => {
                        matched = false;
                        break;
                    }
                    Segment::Param(name) => params.push((name.clone(), value.clone())),
                    Segment::Static(_) => {}
                }
            }

            if matched
                && best
                    .as_ref()
                    .map_or(true, |(current, _)| route.rank > current.rank)
            {
                best = Some((route, params));
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(input: &str) -> Result<Vec<Segment>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let trailing_slash = input.ends_with('/');
    let content_end = if trailing_slash {
        input.len() - 1
    } else {
        input.len()
    };
    let body = &input[1..content_end];
    let mut result = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut offset = 1;

    for raw_segment in body.split('/') {
        if raw_segment.is_empty() {
            return Err(Error::InvalidPattern { at: offset });
        }

        if raw_segment.starts_with(':') {
            let name = &raw_segment[1..];
            let valid = {
                let bytes = name.as_bytes();
                !bytes.is_empty()
                    && bytes[0].is_ascii_lowercase()
                    && bytes[1..]
                        .iter()
                        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
            };
            if !valid || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: offset });
            }
            result.push(Segment::Param(name.to_owned()));
        } else if raw_segment.starts_with('*') {
            return Err(Error::InvalidPattern { at: offset });
        } else {
            result.push(Segment::Static(decode_segment(raw_segment, offset, true)?));
        }

        offset += raw_segment.len() + 1;
    }

    if trailing_slash {
        return Err(Error::InvalidPattern {
            at: input.len() - 1,
        });
    }

    Ok(result)
}

fn parse_path(input: &str) -> Result<Vec<String>, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(Vec::new());
    }

    let trailing_slash = input.ends_with('/');
    let content_end = if trailing_slash {
        input.len() - 1
    } else {
        input.len()
    };
    let body = &input[1..content_end];
    let mut result = Vec::new();
    let mut offset = 1;

    for raw_segment in body.split('/') {
        if raw_segment.is_empty() {
            return Err(Error::InvalidPath { at: offset });
        }
        result.push(decode_segment(raw_segment, offset, false).map_err(|e| match e {
            Error::InvalidPattern { at } => Error::InvalidPath { at },
            other => other,
        })?);
        offset += raw_segment.len() + 1;
    }

    if trailing_slash {
        return Err(Error::InvalidPath {
            at: input.len() - 1,
        });
    }

    Ok(result)
}

fn decode_segment(segment: &str, base: usize, pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !is_upper_hex(bytes[i + 1])
                || !is_upper_hex(bytes[i + 2])
            {
                return Err(invalid(pattern, base + i));
            }
            let value = (hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]);
            let unreserved = is_unreserved(value);
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved {
                return Err(invalid(pattern, base + i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(invalid(pattern, base + i));
        }
    }

    // Every decoded byte is ASCII by construction.
    Ok(String::from_utf8(decoded).unwrap())
}

fn invalid(pattern: bool, at: usize) -> Error {
    if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn is_upper_hex(b: u8) -> bool {
    b.is_ascii_digit() || (b'A'..=b'F').contains(&b)
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}
