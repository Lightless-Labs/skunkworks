fn scan_ascii_printable(input: &str) -> Result<(), usize> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(at);
        }
    }
    Ok(())
}

fn codec_error_to_parse(error: CodecError, offset: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Err(at) = scan_ascii_printable(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(index) => index,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| authority_start + n)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];
    let auth_bytes = authority.as_bytes();

    let ats: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();

    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + ats[1],
        });
    }
    if ats.len() == 1 && ats[0] == 0 {
        return Err(ParseError::InvalidUserInfo { at: authority_start });
    }

    let (userinfo, host_start_rel) = if let Some(&at) = ats.first() {
        (Some(authority[..at].to_string()), at + 1)
    } else {
        (None, 0)
    };

    let host_start = authority_start + host_start_rel;
    let host_port = &authority[host_start_rel..];
    let host_port_bytes = host_port.as_bytes();

    let (host_raw, port_raw, ipv6, port_colon_abs) = if host_port.starts_with('[') {
        let close = match host_port_bytes.iter().position(|&b| b == b']') {
            Some(index) => index,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &host_port[1..close];
        if Ipv6Addr::from_str(interior).is_err() || interior.contains('%') {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let suffix = &host_port[close + 1..];
        if suffix.is_empty() {
            (&host_port[1..close], None, true, None)
        } else if suffix.as_bytes()[0] == b':' {
            let port = &suffix[1..];
            (
                &host_port[1..close],
                Some(port),
                true,
                Some(host_start + close + 1),
            )
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        }
    } else {
        let colon_rel = host_port_bytes.iter().position(|&b| b == b':');
        if let Some(index) = colon_rel {
            (
                &host_port[..index],
                Some(&host_port[index + 1..]),
                false,
                Some(host_start + index),
            )
        } else {
            (host_port, None, false, None)
        }
    };

    if host_raw.is_empty() {
        return Err(ParseError::InvalidHost { at: host_start });
    }
    if !ipv6 {
        let host_offset = host_start;
        for (i, byte) in host_raw.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_offset + i,
                });
            }
        }
    }

    let port = if let Some(port_text) = port_raw {
        let port_start = port_colon_abs.unwrap() + 1;
        if port_text.is_empty() {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        for (i, byte) in port_text.bytes().enumerate() {
            if !byte.is_ascii_digit() {
                return Err(ParseError::InvalidPort {
                    at: port_start + i,
                });
            }
        }
        if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        let mut value: u32 = 0;
        for byte in port_text.bytes() {
            value = match value.checked_mul(10).and_then(|v| v.checked_add((byte - b'0') as u32)) {
                Some(v) => v,
                None => return Err(ParseError::InvalidPort { at: port_start }),
            };
            if value > 65535 {
                return Err(ParseError::InvalidPort { at: port_start });
            }
        }
        if value == 0 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        Some(value as u16)
    } else {
        None
    };

    let remainder = &input[authority_end..];
    let fragment_rel = remainder.find('#');
    let before_fragment = fragment_rel.map(|i| &remainder[..i]).unwrap_or(remainder);
    let fragment = fragment_rel.map(|i| remainder[i + 1..].to_string());

    let query_rel = before_fragment.find('?');
    let path_end_rel = query_rel.unwrap_or(before_fragment.len());
    let raw_path = &before_fragment[..path_end_rel];
    let query = query_rel.map(|i| before_fragment[i + 1..].to_string());
    let path = if raw_path.is_empty() {
        "/".to_string()
    } else {
        raw_path.to_string()
    };

    if let Some(ref raw_userinfo) = userinfo {
        decode_component(Component::UserInfo, raw_userinfo)
            .map_err(|error| codec_error_to_parse(error, authority_start))?;
    }
    if !raw_path.is_empty() {
        decode_component(Component::Path, raw_path)
            .map_err(|error| codec_error_to_parse(error, authority_end))?;
    }
    if let Some(query_value) = &query {
        let query_offset = authority_end + query_rel.unwrap() + 1;
        decode_component(Component::Query, query_value)
            .map_err(|error| codec_error_to_parse(error, query_offset))?;
    }
    if let Some(fragment_value) = &fragment {
        let fragment_offset = authority_end + fragment_rel.unwrap() + 1;
        decode_component(Component::Fragment, fragment_value)
            .map_err(|error| codec_error_to_parse(error, fragment_offset))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: host_raw.to_ascii_lowercase(),
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(value) => Ok(value.to_string()),
        Err(error) => {
            let decoded_at = error.valid_up_to();
            let original_at = origins.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut encoded = String::with_capacity(text.len());
    for byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(*byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => *byte == b':',
                Component::Path => *byte == b'/',
                Component::Query => matches!(*byte, b'/' | b'?'),
                Component::Fragment => matches!(*byte, b'/' | b'?' | b'#'),
            };
        if safe {
            encoded.push(*byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    encoded
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|error| match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let mut stack: Vec<&str> = Vec::new();
    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut terminal_navigation = false;

    for (index, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(|error| match error {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 },
        })?;
        match decoded.as_str() {
            "." => {
                terminal_navigation = index + 1 == segments.len();
            }
            ".." => {
                if !stack.is_empty() {
                    stack.pop();
                }
                terminal_navigation = index + 1 == segments.len();
            }
            _ => {
                stack.push(segment);
                terminal_navigation = false;
            }
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_string())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn map_path_result(url: &mut Url) -> Result<(), ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(())
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_pos = reference.find('#');
    let before_fragment = fragment_pos.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = fragment_pos.map(|i| &reference[i + 1..]);
    let query_pos = before_fragment.find('?');
    let path_end = query_pos.unwrap_or(before_fragment.len());
    let query = query_pos.map(|i| &before_fragment[i + 1..]);
    (&before_fragment[..path_end], query, fragment)
}

fn raw_url_from_parts(url: &Url) -> String {
    let mut output = String::new();
    output.push_str(&url.scheme);
    output.push_str("://");
    if let Some(userinfo) = &url.userinfo {
        output.push_str(userinfo);
        output.push('@');
    }
    if url.ipv6 {
        output.push('[');
        output.push_str(&url.host);
        output.push(']');
    } else {
        output.push_str(&url.host);
    }
    if let Some(port) = url.port {
        output.push(':');
        output.push_str(&port.to_string());
    }
    output
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let mut base_url = parse_url(base).map_err(ResolveError::Base)?;
    map_path_result(&mut base_url)?;

    if let Err(at) = scan_ascii_printable(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let first_delimiter = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        map_path_result(&mut url)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        map_path_result(&mut url)?;
        return Ok(url);
    }

    let (reference_path, reference_query, reference_fragment) = split_reference(reference);
    let selected_path;
    let selected_query;

    if reference_path.is_empty() {
        selected_path = base_url.path.clone();
        selected_query = if reference_query.is_none() {
            base_url.query.clone()
        } else {
            reference_query.map(str::to_string)
        };
    } else {
        selected_query = reference_query.map(str::to_string);
        if reference_path.starts_with('/') {
            selected_path = reference_path.to_string();
        } else {
            let base_dir_end = base_url.path.rfind('/').unwrap_or(0) + 1;
            selected_path = format!("{}{}", &base_url.path[..base_dir_end], reference_path);
        }
    }

    let mut constructed = raw_url_from_parts(&base_url);
    constructed.push_str(&selected_path);
    if let Some(query) = selected_query {
        constructed.push('?');
        constructed.push_str(&query);
    }
    if let Some(fragment) = reference_fragment {
        constructed.push('#');
        constructed.push_str(fragment);
    }

    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    map_path_result(&mut url)?;
    Ok(url)
}

fn scan_bytes(input: &str) -> Result<(), usize> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(at);
        }
    }
    Ok(())
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let high = hex_value(bytes[i + 1]);
            let low = hex_value(bytes[i + 2]);
            match (high, low) {
                (Some(high), Some(low)) => {
                    decoded.push((high << 4) | low);
                    offsets.push(i);
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
            i += 3;
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_offset = error.valid_up_to();
            let at = offsets.get(decoded_offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for byte in text.bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };
        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn parse_port(input: &str, start: usize) -> Result<u16, ParseError> {
    if input.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }

    for (offset, byte) in input.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + offset });
        }
    }

    if input.len() >= 2 && input.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for byte in input.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|number| number.checked_add((byte - b'0') as u32))
        {
            Some(number) if number <= 65535 => number,
            _ => return Err(ParseError::InvalidPort { at: start }),
        };
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn validate_component(
    component: Component,
    value: &str,
    start: usize,
) -> Result<(), ParseError> {
    decode_component(component, value).map(|_| ()).map_err(|error| match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: start + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: start + at },
    })
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Err(at) = scan_bytes(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, byte) in bytes[..colon].iter().copied().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];

    let mut at_signs = authority.match_indices('@');
    let first_at = at_signs.next().map(|(at, _)| at);
    if let Some((second_at, _)) = at_signs.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second_at,
        });
    }

    let (userinfo, hostport, hostport_start) = if let Some(at) = first_at {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            authority_start + at + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let closing = hostport.find(']');
        let closing = match closing {
            Some(index) => index,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &hostport[1..closing];
        if interior.contains('%') || Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }

        let suffix = &hostport[closing + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_start + closing + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + closing + 1,
            });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), parsed_port)
    } else {
        let host_colon = hostport.find(':');
        let (host_text, parsed_port) = match host_colon {
            Some(index) => (
                &hostport[..index],
                Some(parse_port(
                    &hostport[index + 1..],
                    hostport_start + index + 1,
                )?),
            ),
            None => (hostport, None),
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), parsed_port)
    };

    let fragment_mark = input[authority_end..].find('#').map(|offset| authority_end + offset);
    let before_fragment_end = fragment_mark.unwrap_or(input.len());
    let query_mark = input[authority_end..before_fragment_end]
        .find('?')
        .map(|offset| authority_end + offset);

    let path_end = query_mark.unwrap_or(before_fragment_end);
    let path_raw = &input[authority_end..path_end];
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    let query = query_mark.map(|mark| input[mark + 1..before_fragment_end].to_owned());
    let fragment = fragment_mark.map(|mark| input[mark + 1..].to_owned());

    if let Some(value) = userinfo.as_deref() {
        validate_component(Component::UserInfo, value, authority_start)?;
    }
    if !path_raw.is_empty() {
        validate_component(Component::Path, path_raw, authority_end)?;
    }
    if let Some(mark) = query_mark {
        validate_component(
            Component::Query,
            query.as_deref().unwrap_or(""),
            mark + 1,
        )?;
    }
    if let Some(mark) = fragment_mark {
        validate_component(
            Component::Fragment,
            fragment.as_deref().unwrap_or(""),
            mark + 1,
        )?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn map_codec_to_path(error: CodecError) -> PathError {
    match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_to_path)?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(map_codec_to_path)?;
        let navigation = decoded == "." || decoded == "..";
        terminal_navigation = navigation && index + 1 == raw_segments.len();

        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn split_reference_components(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_mark = reference.find('#');
    let before_fragment = fragment_mark.map(|at| &reference[..at]).unwrap_or(reference);
    let query_mark = before_fragment.find('?');
    let path = query_mark
        .map(|at| &before_fragment[..at])
        .unwrap_or(before_fragment);
    let query = query_mark.map(|at| &before_fragment[at + 1..]);
    let fragment = fragment_mark.map(|at| &reference[at + 1..]);
    (path, query, fragment)
}

fn normalize_url_path(url: &mut Url) -> Result<(), ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(())
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let normalized_base_path =
        normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    if let Err(at) = scan_bytes(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let (reference_path, _, _) = split_reference_components(reference);
    let first_delimiter = reference
        .bytes()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        normalize_url_path(&mut url)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        normalize_url_path(&mut url)?;
        return Ok(url);
    }

    let (reference_path, reference_query, reference_fragment) =
        split_reference_components(reference);
    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            normalized_base_path,
            if reference_query.is_some() {
                reference_query
            } else {
                base_url.query.as_deref()
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = normalized_base_path
                .rfind('/')
                .map(|at| &normalized_base_path[..=at])
                .unwrap_or("/");
            format!("{slash}{reference_path}")
        };
        (merged, reference_query)
    };

    let mut absolute = String::new();
    absolute.push_str(&base_url.scheme);
    absolute.push_str("://");
    if let Some(userinfo) = base_url.userinfo.as_deref() {
        absolute.push_str(userinfo);
        absolute.push('@');
    }
    if base_url.ipv6 {
        absolute.push('[');
        absolute.push_str(&base_url.host);
        absolute.push(']');
    } else {
        absolute.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        absolute.push(':');
        absolute.push_str(&port.to_string());
    }
    absolute.push_str(&selected_path);
    if let Some(query) = selected_query {
        absolute.push('?');
        absolute.push_str(query);
    }
    if let Some(fragment) = reference_fragment {
        absolute.push('#');
        absolute.push_str(fragment);
    }

    let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    normalize_url_path(&mut url)?;
    Ok(url)
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let lo = hex_value(bytes[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(err) => {
            let decoded_at = err.valid_up_to();
            let input_at = origins.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: input_at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let auth_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_rel) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_start = authority_start + host_start_rel;
    let host_port = &authority[host_start_rel..];
    let mut ipv6 = false;
    let (host, port) = if host_port.starts_with('[') {
        let open = host_start;
        let closing_rel = match host_port.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &host_port[1..closing_rel];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        ipv6 = true;
        let suffix_start = closing_rel + 1;
        let suffix = &host_port[suffix_start..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                host_start + suffix_start + 1,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + suffix_start,
            });
        };
        (interior.to_ascii_lowercase(), parsed_port)
    } else {
        let split = host_port.as_bytes().iter().position(|&b| b == b':');
        let (raw_host, raw_port) = match split {
            Some(i) => (&host_port[..i], Some((&host_port[i + 1..], host_start + i + 1))),
            None => (host_port, None),
        };
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, &b) in raw_host.as_bytes().iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + i,
                });
            }
        }
        let parsed_port = match raw_port {
            Some((p, p_at)) => Some(parse_port(p, p_at)?),
            None => None,
        };
        (raw_host.to_ascii_lowercase(), parsed_port)
    };

    let remainder = &input[authority_end..];
    let fragment_rel = remainder.find('#');
    let before_fragment = fragment_rel.map(|i| &remainder[..i]).unwrap_or(remainder);
    let fragment = fragment_rel.map(|i| remainder[i + 1..].to_owned());

    let query_rel = before_fragment.find('?');
    let raw_path = query_rel
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_rel.map(|i| before_fragment[i + 1..].to_owned());

    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };

    let component_checks = [
        userinfo.as_deref().map(|s| (Component::UserInfo, s, authority_start)),
        Some((
            Component::Path,
            raw_path,
            authority_end,
        )),
        query
            .as_deref()
            .map(|s| (Component::Query, s, authority_end + query_rel.unwrap_or(0) + 1)),
        fragment.as_deref().map(|s| {
            (
                Component::Fragment,
                s,
                authority_end + fragment_rel.unwrap_or(0) + 1,
            )
        }),
    ];

    for check in component_checks.into_iter().flatten() {
        let (component, text, offset) = check;
        decode_component(component, text).map_err(|e| map_codec_parse(e, offset))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(port: &str, at: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + i });
        }
    }
    let bytes = port.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }
    let mut value: u32 = 0;
    for b in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
        {
            Some(v) if v <= 65535 => v,
            _ => return Err(ParseError::InvalidPort { at }),
        };
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(value as u16)
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let trailing_navigation = path
        .split('/')
        .last()
        .map(|segment| {
            matches!(
                decode_component(Component::Path, segment).as_deref(),
                Ok(".") | Ok("..")
            )
        })
        .unwrap_or(false);

    let mut stack: Vec<&str> = Vec::new();
    for segment in path[1..].split('/') {
        let decoded = decode_component(Component::Path, segment).expect("path was validated");
        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }

    if trailing_navigation {
        stack.push("");
    }
    Ok(format!("/{}", stack.join("/")))
}

fn validate_reference_bytes(reference: &str) -> Result<(), ParseError> {
    for (i, b) in reference.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn normalized_url_path(url: &mut Url) -> Result<(), PathError> {
    url.path = normalise_path(&url.path)?;
    Ok(())
}

fn absolute_reference(reference: &str) -> bool {
    let end = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    reference[..end].contains(':')
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let hash = reference.find('#');
    let before_fragment = hash.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = hash.map(|i| &reference[i + 1..]);
    let query_pos = before_fragment.find('?');
    let path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_pos.map(|i| &before_fragment[i + 1..]);
    (path, query, fragment)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let mut base_url = parse_url(base).map_err(ResolveError::Base)?;
    base_url.path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    validate_reference_bytes(reference).map_err(ResolveError::Reference)?;

    if absolute_reference(reference) {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        normalized_url_path(&mut url).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        normalized_url_path(&mut url).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (reference_path, reference_query, reference_fragment) = split_reference(reference);
    let (path, query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if reference_query.is_some() {
                reference_query.map(str::to_owned)
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_url.path[..slash], reference_path)
        };
        (merged, reference_query.map(str::to_owned))
    };

    let mut constructed = String::new();
    constructed.push_str(&base_url.scheme);
    constructed.push_str("://");
    if let Some(userinfo) = &base_url.userinfo {
        constructed.push_str(userinfo);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&path);
    if let Some(query) = &query {
        constructed.push('?');
        constructed.push_str(query);
    }
    if let Some(fragment) = reference_fragment {
        constructed.push('#');
        constructed.push_str(fragment);
    }

    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalized_url_path(&mut url).map_err(ResolveError::Path)?;
    Ok(url)
}
