#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map_or(input.len(), |offset| authority_start + offset);

    let authority = &input[authority_start..authority_end];
    let authority_bytes = &bytes[authority_start..authority_end];

    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(offset, &byte)| (byte == b'@').then_some(offset))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_offset) = if let Some(&at_offset) = at_positions.first() {
        if at_offset == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at_offset,
            });
        }
        (Some(authority[..at_offset].to_owned()), at_offset + 1)
    } else {
        (None, 0)
    };

    let host_port = &authority[host_start_offset..];
    let host_port_bytes = &authority_bytes[host_start_offset..];
    let host_port_start = authority_start + host_start_offset;

    let host_colon = host_port_bytes.iter().position(|&byte| byte == b':');
    let (host_part, port_part) = match host_colon {
        Some(offset) => (&host_port[..offset], Some(&host_port[offset + 1..])),
        None => (host_port, None),
    };

    if host_part.is_empty() {
        return Err(ParseError::InvalidHost { at: host_port_start });
    }

    for (offset, &byte) in host_part.as_bytes().iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_port_start + offset,
            });
        }
    }

    let port = if let Some(port_text) = port_part {
        let port_start = host_port_start + host_colon.unwrap() + 1;
        if port_text.is_empty() {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        for (offset, &byte) in port_text.as_bytes().iter().enumerate() {
            if !byte.is_ascii_digit() {
                return Err(ParseError::InvalidPort {
                    at: port_start + offset,
                });
            }
        }

        if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        let mut value: u32 = 0;
        for &byte in port_text.as_bytes() {
            value = match value
                .checked_mul(10)
                .and_then(|current| current.checked_add((byte - b'0') as u32))
            {
                Some(value) => value,
                None => return Err(ParseError::InvalidPort { at: port_start }),
            };
        }
        if !(1..=65535).contains(&value) {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        Some(value as u16)
    } else {
        None
    };

    let after_authority = &input[authority_end..];
    let fragment_offset = after_authority.find('#');
    let before_fragment = fragment_offset.map_or(after_authority, |offset| &after_authority[..offset]);
    let fragment = fragment_offset.map(|offset| after_authority[offset + 1..].to_owned());

    let query_offset = before_fragment.find('?');
    let path_text = query_offset.map_or(before_fragment, |offset| &before_fragment[..offset]);
    let query = query_offset.map(|offset| before_fragment[offset + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: host_part.to_ascii_lowercase(),
        port,
        path: if path_text.is_empty() {
            "/".to_owned()
        } else {
            path_text.to_owned()
        },
        query,
        fragment,
    })
}
