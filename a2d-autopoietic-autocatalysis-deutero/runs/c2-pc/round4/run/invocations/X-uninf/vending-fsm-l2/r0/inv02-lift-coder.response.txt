```rust
use std::collections::HashSet;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const ASCENDING: [u16; 4] = [1, 3, 4, 10];
const DESCENDING: [u16; 4] = [10, 4, 3, 1];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashSet::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.clone()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: ASCENDING[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !ASCENDING.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = self.selected.as_ref().ok_or(Error::NoSelection)?;
        let product = self
            .products
            .iter()
            .find(|product| &product.id == selected_id)
            .expect("selected product must be in the catalog");

        let price = product.price;
        let id = product.id.clone();
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change_counts = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for (index, &count) in change_counts.iter().enumerate() {
            proposed_tubes[index] -= count;
        }
        for &coin in &self.escrow {
            let index = denomination_index(coin).expect("escrow contains accepted coins");
            let updated = u32::from(proposed_tubes[index]) + 1;
            if updated > 100 {
                return Err(Error::TubeLimit {
                    coin: ASCENDING[index],
                });
            }
            proposed_tubes[index] = updated as u16;
        }

        let mut change = Vec::new();
        for &coin in DESCENDING.iter() {
            let count = change_counts[denomination_index(coin).unwrap()];
            for _ in 0..count {
                change.push(coin);
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let refunded = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(refunded)
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|byte| byte.is_ascii_lowercase())
}

fn denomination_index(coin: u16) -> Option<usize> {
    ASCENDING.iter().position(|&denomination| denomination == coin)
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;

    for &coin in DESCENDING.iter() {
        let index = denomination_index(coin).unwrap();
        let count = tubes[index].min(remaining / coin);
        greedy[index] = count;
        remaining -= count * coin;
    }

    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    search_change(0, amount, tubes, &mut [0; 4], &mut best);
    best
}

fn search_change(
    index: usize,
    remaining: u16,
    tubes: &[u16; 4],
    current: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index == DESCENDING.len() {
        if remaining == 0
            && best
                .as_ref()
                .is_none_or(|candidate| better_change(current, candidate))
        {
            *best = Some(*current);
        }
        return;
    }

    let coin = DESCENDING[index];
    let tube_index = denomination_index(coin).unwrap();
    let max_count = tubes[tube_index].min(remaining / coin);

    for count in 0..=max_count {
        current[tube_index] = count;
        search_change(
            index + 1,
            remaining - count * coin,
            tubes,
            current,
            best,
        );
    }
    current[tube_index] = 0;
}

fn better_change(candidate: &[u16; 4], incumbent: &[u16; 4]) -> bool {
    let candidate_total: u32 = candidate.iter().map(|&count| u32::from(count)).sum();
    let incumbent_total: u32 = incumbent.iter().map(|&count| u32::from(count)).sum();

    if candidate_total != incumbent_total {
        return candidate_total < incumbent_total;
    }

    for &coin in DESCENDING.iter() {
        let index = denomination_index(coin).unwrap();
        if candidate[index] != incumbent[index] {
            return candidate[index] > incumbent[index];
        }
    }
    false
}
```
