pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,

pub struct Match {

}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {

            routes: Vec::new(),

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })

                continue;

                        Some(i)
                    } else {
                        None
                    }
                })
                .collect()
        } else {

                .enumerate()
                .filter_map(|(i, (r, _))| if r.method == method { Some(i) } else { None })

            return Err(Error::MethodNotAllowed { allowed });

fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)
    }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    let raw = split_head(pattern).map_err(|at| Error::InvalidPattern { at })?;
    let (parts, trailing) = raw;
    let mut result = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (part, start) in parts {
        let first = part.as_bytes()[0];
        if first == b':' || first == b'*' {
            let wildcard = first == b'*';
            let name = &part[1..];
            if !valid_name(name) || !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard && result.len() + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard {
                result.push(Segment::Wildcard(name.to_string()));
            } else {
                result.push(Segment::Param(name.to_string()));
            }
        } else {
            let decoded = decode_segment(part, start)
                .map_err(|at| Error::InvalidPattern { at })?;
            result.push(Segment::Static(decoded));
        }
    }

    Ok((result, trailing))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    let (parts, trailing) = split_head(path).map_err(|at| Error::InvalidPath { at })?;
    let mut result = Vec::with_capacity(parts.len());
    for (part, start) in parts {
        result.push(decode_segment(part, start).map_err(|at| Error::InvalidPath { at })?);
    }
    Ok((result, trailing))
}

fn split_head(input: &str) -> Result<(Vec<(&str, usize)>, bool), usize> {
    if input.is_empty() || !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut parts = Vec::new();
    let mut start = 1;

    while start < end {
        let next = input[start..end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(end);
        if next == start {
            return Err(start);
        }
        parts.push((&input[start..next], start));
        start = next + 1;
    }

    if parts.is_empty() {
        return Err(1);
    }
    Ok((parts, trailing))
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(segment: &str, start: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(start + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(start + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(start + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(start + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(start + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn hex_upper(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_segments(
    pattern: &[Segment],
    request: &[String],
) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                if request.get(i)? != expected {
                    return None;
                }
                i += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(i)?.clone()));
                i += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), request[i..].join("/")));
                i = request.len();
            }
        }
    }

    if i == request.len() {
        Some(params)
    } else {
        None
    }
}
