#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
}

#[derive(Debug)]
pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_capture_name(name: &str) -> bool {
    let mut bytes = name.bytes();
    matches!(bytes.next(), Some(b'a'..=b'z'))
        && bytes.all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(s: &str, offset: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(offset + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(offset + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(offset + i);
            }
            out.push(value);
            i += 3;
        } else {
            if !unreserved(b) {
                return Err(offset + i);
            }
            out.push(b);
            i += 1;
        }
    }
    // All decoded bytes are printable ASCII.
    Ok(String::from_utf8(out).expect("validated ASCII"))
}

struct ParsedPath {
    segments: Vec<(String, usize)>,
    trailing_slash: bool,
}

fn parse_path_head(input: &str) -> Result<ParsedPath, usize> {
    if !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing_slash: false,
        });
    }

    let bytes = input.as_bytes();
    let trailing_slash = bytes.last() == Some(&b'/');
    let end = if trailing_slash {
        bytes.len() - 1
    } else {
        bytes.len()
    };

    let mut segments = Vec::new();
    let mut start = 1;
    let mut i = 1;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(start);
            }
            let raw = &input[start..i];
            segments.push((decode_segment(raw, start)?, start));
            start = i + 1;
        }
        i += 1;
    }
    Ok(ParsedPath {
        segments,
        trailing_slash,
    })
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), usize> {
    let parsed = parse_path_head(pattern)?;
    let mut segments = Vec::with_capacity(parsed.segments.len());
    let mut names = std::collections::HashSet::new();
    let count = parsed.segments.len();

    for (index, (raw, offset)) in parsed.segments.iter().enumerate() {
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_capture_name(name) || !names.insert(name.to_string()) {
                return Err(*offset);
            }
            segments.push(Segment::Param(name.to_string()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if index + 1 != count {
                return Err(*offset);
            }
            if !valid_capture_name(name) || !names.insert(name.to_string()) {
                return Err(*offset);
            }
            segments.push(Segment::Wildcard(name.to_string()));
        } else {
            segments.push(Segment::Static(decode_segment(raw, *offset)?));
        }
    }

    let mut rank: Vec<u8> = segments
        .iter()
        .map(|s| match s {
            Segment::Static(_) => 2,
            Segment::Param(_) => 1,
            Segment::Wildcard(_) => 0,
        })
        .collect();
    if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
        rank.push(3);
    }
    Ok((segments, parsed.trailing_slash, rank))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (segments, trailing_slash, rank) =
            parse_pattern(pattern).map_err(|at| Error::InvalidPattern { at })?;
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_string(),
            segments,
            trailing_slash,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_path_head(head).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();
        for (route_id, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != parsed.trailing_slash
            {
                continue;
            }

            let wildcard_index = route
                .segments
                .iter()
                .position(|s| matches!(s, Segment::Wildcard(_)));
            let fixed_len = wildcard_index.unwrap_or(route.segments.len());
            if parsed.segments.len() < fixed_len {
                continue;
            }
            if wildcard_index.is_none() && parsed.segments.len() != route.segments.len() {
                continue;
            }

            let mut params = Vec::new();
            let mut matched = true;
            for i in 0..fixed_len {
                let request = &parsed.segments[i].0;
                match &route.segments[i] {
                    Segment::Static(value) if value == request => {}
                    Segment::Param(name) => params.push((name.clone(), request.clone())),
                    _ => {
                        matched = false;
                        break;
                    }
                }
            }
            if !matched {
                continue;
            }
            if let Some(wi) = wildcard_index {
                let Segment::Wildcard(name) = &route.segments[wi] else {
                    unreachable!()
                };
                let value = parsed.segments[wi..]
                    .iter()
                    .map(|(s, _)| s.as_str())
                    .collect::<Vec<_>>()
                    .join("/");
                params.push((name.clone(), value));
            }
            candidates.push((route_id, params));
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates
                .iter()
                .any(|(id, _)| self.routes[*id].method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(id, _)| self.routes[*id].method == eligible_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(id, _)| self.routes[*id].method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let (route_id, params) = eligible
            .into_iter()
            .max_by(|(id_a, _), (id_b, _)| {
                self.routes[**id_a]
                    .rank
                    .cmp(&self.routes[**id_b].rank)
                    // Earlier registration wins ties.
                    .then_with(|| id_b.cmp(id_a))
            })
            .expect("eligible routes are nonempty");

        let route = &self.routes[*route_id];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing_slash != parsed.trailing_slash
            && !head.eq("/")
        {
            if route.trailing_slash {
                Some(format!("{head}/"))
            } else {
                Some(head[..head.len() - 1].to_string())
            }
        } else {
            None
        };

        Ok(Match {
            route: *route_id,
            params: params.clone(),
            redirect,
        })
    }
}
