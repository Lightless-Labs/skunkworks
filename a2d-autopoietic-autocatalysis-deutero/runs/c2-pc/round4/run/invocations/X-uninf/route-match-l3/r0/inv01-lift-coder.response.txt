#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        let (segments, trailing) = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let mut has_wildcard = false;

        for segment in &segments {
            match segment {
                Segment::Static(_) => rank.push(2),
                Segment::Param(_) => rank.push(1),
                Segment::Wildcard(_) => {
                    rank.push(0);
                    has_wildcard = true;
                }
            }
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_string(),
            segments,
            trailing,
            rank,
            has_wildcard,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (request_segments, trailing) = parse_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if let Some(params) = match_segments(&route.segments, &request_segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<usize> = if method == "HEAD" {
            let has_head = candidates.iter().any(|(r, _)| r.method == "HEAD");
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (r, _))| {
                    if (has_head && r.method == "HEAD") || (!has_head && r.method == "GET") {
                        Some(i)
                    } else {
                        None
                    }
                })
                .collect()
        } else {
            candidates
                .iter()
                .enumerate()
                .filter_map(|(i, (r, _))| if r.method == method { Some(i) } else { None })
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> =
                candidates.iter().map(|(r, _)| r.method.clone()).collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut winner = eligible[0];
        for &index in eligible.iter().skip(1) {
            let candidate = candidates[index].0;
            let current = candidates[winner].0;
            if candidate.rank > current.rank
                || (candidate.rank == current.rank && candidate.id < current.id)
            {
                winner = index;
            }
        }

        let (route, params) = &candidates[winner];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != trailing
            && !head.is_empty()
            && head != "/"
        {
            if route.trailing {
                Some(format!("{head}/"))
            } else if head.ends_with('/') {
                Some(head[..head.len() - 1].to_string())
            } else {
                None
            }
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)
    }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    let raw = split_head(pattern).map_err(|at| Error::InvalidPattern { at })?;
    let (parts, trailing) = raw;
    let mut result = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (part, start) in parts {
        let first = part.as_bytes()[0];
        if first == b':' || first == b'*' {
            let wildcard = first == b'*';
            let name = &part[1..];
            if !valid_name(name) || !names.insert(name.to_string()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard && result.len() + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard {
                result.push(Segment::Wildcard(name.to_string()));
            } else {
                result.push(Segment::Param(name.to_string()));
            }
        } else {
            let decoded = decode_segment(part, start)
                .map_err(|at| Error::InvalidPattern { at })?;
            result.push(Segment::Static(decoded));
        }
    }

    Ok((result, trailing))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    let (parts, trailing) = split_head(path).map_err(|at| Error::InvalidPath { at })?;
    let mut result = Vec::with_capacity(parts.len());
    for (part, start) in parts {
        result.push(decode_segment(part, start).map_err(|at| Error::InvalidPath { at })?);
    }
    Ok((result, trailing))
}

fn split_head(input: &str) -> Result<(Vec<(&str, usize)>, bool), usize> {
    if input.is_empty() || !input.starts_with('/') {
        return Err(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = input.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { input.len() - 1 } else { input.len() };
    let mut parts = Vec::new();
    let mut start = 1;

    while start < end {
        let next = input[start..end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(end);
        if next == start {
            return Err(start);
        }
        parts.push((&input[start..next], start));
        start = next + 1;
    }

    if parts.is_empty() {
        return Err(1);
    }
    Ok((parts, trailing))
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(segment: &str, start: usize) -> Result<String, usize> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(start + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(start + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(start + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(start + i);
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(start + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn hex_upper(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_segments(
    pattern: &[Segment],
    request: &[String],
) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut i = 0;

    for segment in pattern {
        match segment {
            Segment::Static(expected) => {
                if request.get(i)? != expected {
                    return None;
                }
                i += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), request.get(i)?.clone()));
                i += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), request[i..].join("/")));
                i = request.len();
            }
        }
    }

    if i == request.len() {
        Some(params)
    } else {
        None
    }
}
