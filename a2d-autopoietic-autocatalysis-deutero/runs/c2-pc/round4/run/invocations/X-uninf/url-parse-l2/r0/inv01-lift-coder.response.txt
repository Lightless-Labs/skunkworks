#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(value) => Ok(value.to_owned()),
        Err(error) => {
            let decoded_offset = error.valid_up_to();
            let original_offset = offsets.get(decoded_offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_offset })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut encoded = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            encoded.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    encoded
}

fn is_scheme_first(byte: u8) -> bool {
    byte.is_ascii_alphabetic()
}

fn is_scheme_rest(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')
}

fn map_codec_error(error: CodecError, base: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(index) => index,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !is_scheme_first(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !is_scheme_rest(byte) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());

    let authority = &bytes[authority_start..authority_end];
    let at_positions: Vec<usize> = authority
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    if at_positions.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_local) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(input[authority_start..authority_start + at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let remaining = &authority[host_start_local..];
    let absolute_host_start = authority_start + host_start_local;
    let (host, port, ipv6) = if remaining.first() == Some(&b'[') {
        let close = match remaining.iter().position(|&byte| byte == b']') {
            Some(index) => index,
            None => {
                return Err(ParseError::InvalidIpv6 {
                    at: absolute_host_start,
                });
            }
        };

        let interior = &remaining[1..close];
        let interior_str = std::str::from_utf8(interior).expect("global ASCII check");
        if interior_str.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 {
                at: absolute_host_start,
            });
        }

        let suffix_start = close + 1;
        let port = if suffix_start == remaining.len() {
            None
        } else if remaining[suffix_start] == b':' {
            Some(parse_port(
                &remaining[suffix_start + 1..],
                absolute_host_start + suffix_start + 1,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: absolute_host_start + suffix_start,
            });
        };

        (
            interior_str.to_ascii_lowercase(),
            port,
            true,
        )
    } else {
        let host_port_colon = remaining.iter().position(|&byte| byte == b':');
        let host_end = host_port_colon.unwrap_or(remaining.len());
        let host_bytes = &remaining[..host_end];
        if host_bytes.is_empty() {
            return Err(ParseError::InvalidHost {
                at: absolute_host_start,
            });
        }
        for (offset, &byte) in host_bytes.iter().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: absolute_host_start + offset,
                });
            }
        }

        let port = if let Some(port_colon) = host_port_colon {
            Some(parse_port(
                &remaining[port_colon + 1..],
                absolute_host_start + port_colon + 1,
            )?)
        } else {
            None
        };

        (
            std::str::from_utf8(host_bytes)
                .expect("host bytes are ASCII")
                .to_ascii_lowercase(),
            port,
            false,
        )
    };

    let after_authority = &input[authority_end..];
    let fragment_rel = after_authority.find('#');
    let before_fragment = fragment_rel
        .map(|index| &after_authority[..index])
        .unwrap_or(after_authority);
    let fragment = fragment_rel.map(|index| after_authority[index + 1..].to_owned());

    let query_rel = before_fragment.find('?');
    let before_query = query_rel
        .map(|index| &before_fragment[..index])
        .unwrap_or(before_fragment);
    let query = query_rel.map(|index| before_fragment[index + 1..].to_owned());

    let path = if before_query.is_empty() {
        "/".to_owned()
    } else {
        before_query.to_owned()
    };

    let mut component_checks: Vec<(Component, &str, usize)> = Vec::new();
    if let Some(userinfo_value) = userinfo.as_deref() {
        component_checks.push((
            Component::UserInfo,
            userinfo_value,
            authority_start,
        ));
    }
    if !before_query.is_empty() {
        component_checks.push((
            Component::Path,
            before_query,
            authority_end,
        ));
    }
    if let Some(query_value) = query.as_deref() {
        let query_offset = authority_end + query_rel.unwrap() + 1;
        component_checks.push((Component::Query, query_value, query_offset));
    }
    if let Some(fragment_value) = fragment.as_deref() {
        let fragment_offset = authority_end + fragment_rel.unwrap() + 1;
        component_checks.push((Component::Fragment, fragment_value, fragment_offset));
    }

    for (component, value, base) in component_checks {
        if let Err(error) = decode_component(component, value) {
            return Err(map_codec_error(error, base));
        }
    }

    let mut scheme = input[..colon].to_owned();
    scheme.make_ascii_lowercase();

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(bytes: &[u8], port_start: usize) -> Result<u16, ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + offset,
            });
        }
    }

    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for &byte in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|number| number.checked_add((byte - b'0') as u32))
        {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    Ok(value as u16)
}
