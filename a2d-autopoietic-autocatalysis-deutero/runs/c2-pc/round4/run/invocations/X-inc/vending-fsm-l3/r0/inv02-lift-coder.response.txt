#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    events: Vec<Event>,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn event(events: &mut Vec<Event>, kind: EventKind) {
    events.push(Event {
        seq: events.len(),
        kind,
    });
}

fn greedy_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    let mut remaining = amount;
    let mut counts = [0u16; 4];
    for index in (0..4).rev() {
        let coin = DENOMINATIONS[index];
        let take = tubes[index].min(remaining / coin);
        counts[index] = take;
        remaining -= take * coin;
    }
    if remaining != 0 {
        return None;
    }
    Some(counts_to_descending(counts))
}

fn counts_to_descending(counts: [u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for index in (0..4).rev() {
        for _ in 0..counts[index] {
            result.push(DENOMINATIONS[index]);
        }
    }
    result
}

fn exact_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;

    for c10 in 0..=tubes[3].min(amount / 10) {
        let after10 = amount - c10 * 10;
        for c4 in 0..=tubes[2].min(after10 / 4) {
            let after4 = after10 - c4 * 4;
            for c3 in 0..=tubes[1].min(after4 / 3) {
                let after3 = after4 - c3 * 3;
                if after3 > tubes[0] {
                    continue;
                }

                let candidate = [after3, c3, c4, c10];
                let total = candidate.iter().map(|&n| n as u32).sum();
                let is_lex_greater = best.is_some_and(|current| {
                    (3..4)
                        .rev()
                        .find_map(|i| {
                            if candidate[i] != current[i] {
                                Some(candidate[i] > current[i])
                            } else {
                                None
                            }
                        })
                        .or_else(|| {
                            (0..3)
                                .rev()
                                .find_map(|i| {
                                    if candidate[i] != current[i] {
                                        Some(candidate[i] > current[i])
                                    } else {
                                        None
                                    }
                                })
                        })
                        .unwrap_or(false)
                });

                if total < best_count || (total == best_count && is_lex_greater) {
                    best = Some(candidate);
                    best_count = total;
                }
            }
        }
    }

    best.map(counts_to_descending)
}

fn change_counts(change: &[u16]) -> [u16; 4] {
    let mut counts = [0u16; 4];
    for &coin in change {
        if let Some(index) = DENOMINATIONS.iter().position(|&d| d == coin) {
            counts[index] += 1;
        }
    }
    counts
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = std::collections::HashSet::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.clone()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for index in 0..4 {
            if tubes[index] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        event(
            &mut self.events,
            EventKind::Inserted {
                coin,
                credit: self.credit,
            },
        );
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        event(
            &mut self.events,
            EventKind::Selected { id: id.to_owned() },
        );
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.as_ref().ok_or(Error::NoSelection)?.clone();
        let index = self
            .products
            .iter()
            .position(|product| product.id == id)
            .expect("selected product must exist");
        let product = &self.products[index];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let price = product.price;
        let amount = self.credit - price;
        let change = if let Some(greedy) = greedy_change(amount, &self.tubes) {
            greedy
        } else {
            exact_change(amount, &self.tubes).ok_or(Error::NoChange { amount })?
        };

        let deductions = change_counts(&change);
        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= deductions[i];
        }
        for &coin in &self.escrow {
            let i = DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow contains only accepted denominations");
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        self.tubes = proposed;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        event(
            &mut self.events,
            EventKind::Vended {
                id: id.clone(),
                price,
                change: change.clone(),
            },
        );
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        event(
            &mut self.events,
            EventKind::Refunded {
                coins: coins.clone(),
            },
        );
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.cmp(&b.id));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        event(
            &mut self.events,
            EventKind::PriceChanged {
                id: id.to_owned(),
                old,
                new: price,
            },
        );
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }

        let stock = u32::from(self.products[index].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = stock as u16;
        event(
            &mut self.events,
            EventKind::Restocked {
                id: id.to_owned(),
                added: count,
                stock: stock as u16,
            },
        );
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }
}
