#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,
    Conflict { with: usize },
    UnknownName,
    DuplicateParam { name: String },
    MissingParam { name: String },
    ExtraParam { name: String },
    InvalidValue { name: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
}

struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method == method
                && same_shape(&route.segments, &segments)
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .as_bytes()
            .iter()
            .position(|&b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_request_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<(&Route, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .map(|(route, params)| (*route, params.clone()))
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            rank(&b.segments)
                .cmp(&rank(&a.segments))
                .then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && !parsed.segments.is_empty()
            && route.trailing != parsed.trailing
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for &(key, value) in params {
            if supplied.contains_key(key) {
                return Err(Error::DuplicateParam {
                    name: key.to_owned(),
                });
            }
            supplied.insert(key, value);
        }

        let captures: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|segment| match segment {
                Segment::Param(name) | Segment::Wildcard(name) => Some(name.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &captures {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut extra: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !captures.contains(key))
            .collect();
        extra.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extra.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            let (key, is_wildcard) = match segment {
                Segment::Param(name) => (name, false),
                Segment::Wildcard(name) => (name, true),
                Segment::Static(_) => continue,
            };
            let value = supplied[key.as_str()];
            if !valid_build_value(value, is_wildcard) {
                return Err(Error::InvalidValue { name: key.clone() });
            }
        }

        let mut output = build_route_path(route, &supplied);
        if route.trailing && output != "/" {
            output.push('/');
        }
        Ok(output)
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &[u8]) -> bool {
    !name.is_empty()
        && name[0].is_ascii_lowercase()
        && name[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = pattern.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };

    let mut segments = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut start = 1;
    let mut i = 1;

    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }

            let raw = &pattern[start..i];
            let raw_bytes = raw.as_bytes();
            if raw_bytes[0] == b':' || raw_bytes[0] == b'*' {
                let marker = raw_bytes[0];
                let name = &raw_bytes[1..];

                if !valid_capture_name(name) {
                    return Err(Error::InvalidPattern { at: start });
                }

                if marker == b'*' && i != end {
                    return Err(Error::InvalidPattern { at: start });
                }

                let name_text = std::str::from_utf8(name).unwrap().to_owned();
                if !seen.insert(name_text.clone()) {
                    return Err(Error::InvalidPattern { at: start });
                }

                if marker == b':' {
                    segments.push(Segment::Param(name_text));
                } else {
                    segments.push(Segment::Wildcard(name_text));
                }
            } else {
                segments.push(Segment::Static(decode_static(raw, start, true)?));
            }

            start = i + 1;
        }
        i += 1;
    }

    Ok((segments, trailing))
}

fn parse_request_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };

    let mut segments = Vec::new();
    let mut start = 1;
    let mut i = 1;

    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(Error::InvalidPath { at: start });
            }
            segments.push(decode_static(&path[start..i], start, false)?);
            start = i + 1;
        }
        i += 1;
    }

    Ok(ParsedPath { segments, trailing })
}

fn decode_static(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !matches!(bytes[i + 1], b'0'..=b'9' | b'A'..=b'F')
                || !matches!(bytes[i + 2], b'0'..=b'9' | b'A'..=b'F')
            {
                return Err(parse_error(pattern, offset + i));
            }
            let value = (hex_value(bytes[i + 1]).unwrap() << 4)
                | hex_value(bytes[i + 2]).unwrap();
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(parse_error(pattern, offset + i));
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(parse_error(pattern, offset + i));
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn parse_error(pattern: bool, at: usize) -> Error {
    if pattern {
        Error::InvalidPattern { at }
    } else {
        Error::InvalidPath { at }
    }
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    if a.len() != b.len() {
        return false;
    }
    a.iter().zip(b).all(|(left, right)| match (left, right) {
        (Segment::Static(x), Segment::Static(y)) => x == y,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let has_wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
    let prefix_len = if has_wildcard {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };

    if path.len() < prefix_len || (!has_wildcard && path.len() != route.segments.len()) {
        return None;
    }

    for (pattern_segment, value) in route.segments[..prefix_len].iter().zip(path.iter()) {
        match pattern_segment {
            Segment::Static(expected) if expected != value => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if has_wildcard {
        if let Segment::Wildcard(name) = &route.segments[route.segments.len() - 1] {
            params.push((name.clone(), path[prefix_len..].join("/")));
        }
    }

    Some(params)
}

fn rank(segments: &[Segment]) -> Vec<u8> {
    let mut values = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => values.push(2),
            Segment::Param(_) => values.push(1),
            Segment::Wildcard(_) => {
                values.push(0);
                return values;
            }
        }
    }
    values.push(3);
    values
}

fn valid_build_value(value: &str, wildcard: bool) -> bool {
    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) {
        return false;
    }
    if wildcard {
        if value.is_empty() {
            return true;
        }
        value.split('/').all(|part| !part.is_empty())
    } else {
        !value.is_empty() && !value.contains('/')
    }
}

fn append_component(output: &mut String, value: &str) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for b in value.bytes() {
        if is_unreserved(b) {
            output.push(b as char);
        } else {
            output.push('%');
            output.push(HEX[(b >> 4) as usize] as char);
            output.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
}

fn build_route_path(
    route: &Route,
    supplied: &std::collections::HashMap<&str, &str>,
) -> String {
    let mut output = String::from("/");
    let mut emitted = false;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if emitted {
                    output.push('/');
                }
                append_component(&mut output, value);
                emitted = true;
            }
            Segment::Param(name) => {
                if emitted {
                    output.push('/');
                }
                append_component(&mut output, supplied[name.as_str()]);
                emitted = true;
            }
            Segment::Wildcard(name) => {
                let value = supplied[name.as_str()];
                if !value.is_empty() {
                    for component in value.split('/') {
                        if emitted {
                            output.push('/');
                        }
                        append_component(&mut output, component);
                        emitted = true;
                    }
                }
            }
        }
    }

    output
}
