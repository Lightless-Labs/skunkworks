#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Clone)]
struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
}

pub struct Router {

    routes: Vec<Route>,

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn decode_static(s: &str, base: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || unreserved(decoded)
            {
                return Err(base + i);
            }
            out.push(decoded);
            i += 3;
        } else if unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn parse_path(input: &str, pattern: bool) -> Result<(Vec<Segment>, bool), usize> {
    let bytes = input.as_bytes();
    let fail = |at| Err(at);
    if bytes.first() != Some(&b'/') {
        return fail(0);
    }
    if input == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = bytes.last() == Some(&b'/');
    let body_end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut seen = std::collections::HashSet::<String>::new();
    let mut start = 1;

    while start < body_end {
        let end = bytes[start..body_end]
            .iter()
            .position(|b| *b == b'/')
            .map(|p| start + p)
            .unwrap_or(body_end);

        if end == start {
            return fail(start);
        }

        let raw = &input[start..end];
        if pattern && (raw.starts_with(':') || raw.starts_with('*')) {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_capture_name(name) || !seen.insert(name.to_owned()) {
                return fail(start);
            }
            if marker == b'*' && end != body_end {
                return fail(start);
            }
            segments.push(if marker == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            let decoded = decode_static(raw, start)?;
            segments.push(Segment::Static(decoded));
        }

        if end == body_end {
            break;
        }
        start = end + 1;
    }

    Ok((segments, trailing))
}

fn shape_equal(a: &Route, b: &Route) -> bool {
    a.segments.len() == b.segments.len()
        && a.segments.iter().zip(&b.segments).all(|(x, y)| match (x, y) {
            (Segment::Static(a), Segment::Static(b)) => a == b,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn rank(route: &Route) -> Vec<u8> {
    let mut result = Vec::new();
    for segment in &route.segments {
        match segment {
            Segment::Static(_) => result.push(2),
            Segment::Param(_) => result.push(1),
            Segment::Wildcard(_) => {
                result.push(0);
                return result;
            }
        }
    }
    result.push(3);
    result
}

fn match_segments(route: &Route, path: &[Segment]) -> Option<Vec<(String, String)>> {
    let wildcard_at = route
        .segments
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));
    let fixed = wildcard_at.unwrap_or(route.segments.len());
    if path.len() < fixed || (wildcard_at.is_none() && path.len() != fixed) {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed {
        match (&route.segments[i], &path[i]) {
            (Segment::Static(a), Segment::Static(b)) if a == b => {}
            (Segment::Param(name), Segment::Static(value)) => {
                params.push((name.clone(), value.clone()))
            }
            _ => return None,
        }
    }
    if let Some(i) = wildcard_at {
        let Segment::Wildcard(name) = &route.segments[i] else {
            unreachable!()
        };
        let tail = path[i..]
            .iter()
            .map(|s| match s {
                Segment::Static(v) => v.as_str(),
                _ => "",
            })
            .collect::<Vec<_>>()
            .join("/");
        if path[i..].iter().any(|s| !matches!(s, Segment::Static(_))) {
            return None;
        }
        params.push((name.clone(), tail));
    }
    Some(params)
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) =
            parse_path(pattern, true).map_err(|at| Error::InvalidPattern { at })?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }
        for route in &self.routes {
            if route.method == method
                && shape_equal(route, &Route {
                    id: 0,
                    name: String::new(),
                    method: method.to_owned(),
                    segments: segments.clone(),
                    trailing,
                })
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }
        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let head_end = path
            .bytes()
            .position(|b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let (parsed, trailing) =
            parse_path(head, false).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if let Some(params) = match_segments(route, &parsed) {
                candidates.push((route, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let has_head = candidates.iter().any(|(r, _)| r.method == "HEAD");
        let selected_method = if method == "HEAD" && !has_head {
            "GET"
        } else {
            method
        };
        let eligible: Vec<_> = candidates
            .into_iter()
            .filter(|(r, _)| r.method == selected_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed = std::collections::BTreeSet::new();
            for (route, _) in &eligible {
                allowed.insert(route.method.clone());
            }
            // Recompute from every path candidate, including those excluded above.
            // Candidates are retained separately below in the normal case.
            return Err(Error::MethodNotAllowed {
                allowed: allowed.into_iter().collect(),
            });
        }

        let mut best = 0;
        for i in 1..eligible.len() {
            if rank(eligible[i].0) > rank(eligible[best].0) {
                best = i;
            }
        }
        let (route, params) = &eligible[best];
        let redirect = if self.policy == SlashPolicy::Redirect
            && route.trailing != trailing
            && !parsed.is_empty()
        {
            Some(if route.trailing {
                format!("{head}/")
            } else {
                head.to_owned()
            })
        } else {
            None
        };
        Ok(Match {
            route: route.id,
            params: params.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let Some(route) = self.routes.iter().find(|r| r.name == name) else {
            return Err(Error::UnknownName);
        };

        let mut supplied = std::collections::HashMap::<&str, &str>::new();
        for (key, value) in params {
            if supplied.insert(*key, *value).is_some() {
                return Err(Error::DuplicateParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut capture_names = Vec::new();
        for segment in &route.segments {
            match segment {
                Segment::Param(n) | Segment::Wildcard(n) => capture_names.push(n.as_str()),
                Segment::Static(_) => {}
            }
        }
        for key in &capture_names {
            if !supplied.contains_key(key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }
        if let Some(key) = supplied
            .keys()
            .filter(|k| !capture_names.contains(k))
            .min()
        {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            let (name, wildcard) = match segment {
                Segment::Param(n) => (n, false),
                Segment::Wildcard(n) => (n, true),
                Segment::Static(_) => continue,
            };
            let value = supplied[name.as_str()];
            let bytes = value.as_bytes();
            let valid_ascii = bytes.iter().all(|b| (0x20..=0x7e).contains(b));
            let valid_shape = if wildcard {
                value.is_empty()
                    || value
                        .split('/')
                        .all(|part| !part.is_empty())
            } else {
                !value.is_empty() && !value.contains('/')
            };
            if !valid_ascii || !valid_shape {
                return Err(Error::InvalidValue { name: name.clone() });
            }
        }

        let mut output = String::from("/");
        let mut emitted = 0usize;
        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    if emitted > 0 {
                        output.push('/');
                    }
                    output.push_str(&encode_component(value.as_bytes()));
                    emitted += 1;
                }
                Segment::Param(name) => {
                    if emitted > 0 {
                        output.push('/');
                    }
                    output.push_str(&encode_component(supplied[name.as_str()].as_bytes()));
                    emitted += 1;
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        if emitted > 0 {
                            output.push('/');
                        }
                        output.push_str(
                            &value
                                .split('/')
                                .map(|part| encode_component(part.as_bytes()))
                                .collect::<Vec<_>>()
                                .join("/"),
                        );
                        emitted += value.split('/').count();
                    }
                }
            }
        }
        if emitted == 0 {
            return Ok("/".to_owned());
        }
        if route.trailing {
            output.push('/');
        }
        Ok(output)
    }
}

fn encode_component(bytes: &[u8]) -> String {
    let mut out = String::new();
    for &b in bytes {
        if unreserved(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(char::from_digit((b >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((b & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
    InvalidName,
    DuplicateName,

pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    name: String,
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
}

struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_route_name(name) {
            return Err(Error::InvalidName);
        }

        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|route| route.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method == method
                && same_shape(&route.segments, &segments)
                && (self.policy != SlashPolicy::Strict || route.trailing == trailing)
            {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_owned(),
            method: method.to_owned(),
            segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }

        let head_end = path
            .as_bytes()
            .iter()
            .position(|&b| b == b'?' || b == b'#')
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_request_path(head)?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();
        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_method = if method == "HEAD"
            && candidates.iter().any(|(route, _)| route.method == "HEAD")
        {
            "HEAD"
        } else if method == "HEAD" {
            "GET"
        } else {
            method
        };

        let mut eligible: Vec<(&Route, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(route, _)| route.method == eligible_method)
            .map(|(route, params)| (*route, params.clone()))
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(route, _)| route.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        eligible.sort_by(|(a, _), (b, _)| {
            rank(&b.segments)
                .cmp(&rank(&a.segments))
                .then_with(|| a.id.cmp(&b.id))
        });
        let (route, params) = eligible.remove(0);

        let redirect = if self.policy == SlashPolicy::Redirect
            && !parsed.segments.is_empty()
            && route.trailing != parsed.trailing
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: route.id,
            params,
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self
            .routes
            .iter()
            .find(|route| route.name == name)
            .ok_or(Error::UnknownName)?;

        let mut supplied = std::collections::HashMap::new();
        for &(key, value) in params {
            if supplied.contains_key(key) {
                return Err(Error::DuplicateParam {
                    name: key.to_owned(),
                });
            }
            supplied.insert(key, value);
        }

        let captures: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|segment| match segment {
                Segment::Param(name) | Segment::Wildcard(name) => Some(name.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &captures {
            if !supplied.contains_key(*key) {
                return Err(Error::MissingParam {
                    name: (*key).to_owned(),
                });
            }
        }

        let mut extra: Vec<&str> = supplied
            .keys()
            .copied()
            .filter(|key| !captures.contains(key))
            .collect();
        extra.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extra.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_owned(),
            });
        }

        for segment in &route.segments {
            let (key, is_wildcard) = match segment {
                Segment::Param(name) => (name, false),
                Segment::Wildcard(name) => (name, true),
                Segment::Static(_) => continue,
            };
            let value = supplied[key.as_str()];
            if !valid_build_value(value, is_wildcard) {
                return Err(Error::InvalidValue { name: key.clone() });
            }
        }

        let mut output = String::from("/");
        let mut emitted = false;

        for segment in &route.segments {
            match segment {
                Segment::Static(value) => {
                    append_component(&mut output, value);
                    emitted = true;
                }
                Segment::Param(name) => {
                    append_component(&mut output, supplied[name.as_str()]);
                    emitted = true;
                }
                Segment::Wildcard(name) => {
                    let value = supplied[name.as_str()];
                    if !value.is_empty() {
                        for component in value.split('/') {
                            if emitted {
                                output.push('/');
                            }
                            append_component(&mut output, component);
                            emitted = true;
                        }
                    }
                }
            }
            if emitted {
                // The separator before each segment is added here on the next iteration.
            }
        }

        // Build the path using explicit separators, including the empty-tail wildcard case.
        output = build_route_path(route, &supplied);

        if route.trailing && output != "/" {
            output.push('/');
        }
        Ok(output)
    }
}

fn valid_method(method: &str) -> bool {
    !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase())
}

fn valid_route_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_' || *b == b'-')
}

fn valid_capture_name(name: &[u8]) -> bool {
    !name.is_empty()
        && name[0].is_ascii_lowercase()
        && name[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = pattern.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    if end == 1 {
        return Err(Error::InvalidPattern { at: 1 });
    }

    let mut raw_segments = Vec::new();
    let mut start = 1;
    let mut i = 1;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }
            raw_segments.push((start, &pattern[start..i]));
            start = i + 1;
        }
        i += 1;
    }

    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut seen = std::collections::HashSet::new();

    for (index, (start, raw)) in raw_segments.iter().enumerate() {
        let raw_bytes = raw.as_bytes();
        if raw_bytes[0] == b':' || raw_bytes[0] == b'*' {
            let marker = raw_bytes[0];
            let name = &raw_bytes[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if marker == b'*' && index + 1 != raw_segments.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            let name_text = std::str::from_utf8(name).unwrap().to_owned();
            if !seen.insert(name_text.clone()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if marker == b':' {
                segments.push(Segment::Param(name_text));
            } else {
                segments.push(Segment::Wildcard(name_text));
            }
        } else {
            segments.push(Segment::Static(decode_static(raw, *start, true)?));
        }
    }

    Ok((segments, trailing))
}

fn parse_request_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    if end == 1 {
        return Err(Error::InvalidPath { at: 1 });
    }

    let mut segments = Vec::new();
    let mut start = 1;
    let mut i = 1;
    while i <= end {
        if i == end || bytes[i] == b'/' {
            if i == start {
                return Err(Error::InvalidPath { at: start });
            }
            let raw = &path[start..i];
            segments.push(decode_static(raw, start, false)?);
            start = i + 1;
        }
        i += 1;
    }

    Ok(ParsedPath { segments, trailing })
}

fn decode_static(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !bytes[i + 1].is_ascii_hexdigit()
                || !bytes[i + 2].is_ascii_hexdigit()
                || bytes[i + 1].is_ascii_lowercase()
                || bytes[i + 2].is_ascii_lowercase()
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            let value = (hex_value(bytes[i + 1]).unwrap() << 4) | hex_value(bytes[i + 2]).unwrap();
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(if pattern {
                    Error::InvalidPattern { at: offset + i }
                } else {
                    Error::InvalidPath { at: offset + i }
                });
            }
            decoded.push(value);
            i += 3;
        } else if is_unreserved(b) {
            decoded.push(b);
            i += 1;
        } else {
            return Err(if pattern {
                Error::InvalidPattern { at: offset + i }
            } else {
                Error::InvalidPath { at: offset + i }
            });
        }
    }

    Ok(String::from_utf8(decoded).unwrap())
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    if a.len() != b.len() {
        return false;
    }
    a.iter().zip(b).all(|(left, right)| match (left, right) {
        (Segment::Static(x), Segment::Static(y)) => x == y,
        (Segment::Param(_), Segment::Param(_)) => true,
        (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
        _ => false,
    })
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let has_wildcard = matches!(route.segments.last(), Some(Segment::Wildcard(_)));
    let prefix_len = if has_wildcard {
        route.segments.len() - 1
    } else {
        route.segments.len()
    };

    if path.len() < prefix_len || (!has_wildcard && path.len() != route.segments.len()) {
        return None;
    }

    for (pattern_segment, value) in route.segments[..prefix_len].iter().zip(path.iter()) {
        match pattern_segment {
            Segment::Static(expected) if expected != value => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            Segment::Wildcard(_) => return None,
        }
    }

    if has_wildcard {
        if let Segment::Wildcard(name) = &route.segments[route.segments.len() - 1] {
            params.push((name.clone(), path[prefix_len..].join("/")));
        }
    }

    Some(params)
}

fn rank(segments: &[Segment]) -> Vec<u8> {
    let mut values = Vec::new();
    for segment in segments {
        match segment {
            Segment::Static(_) => values.push(2),
            Segment::Param(_) => values.push(1),
            Segment::Wildcard(_) => {
                values.push(0);
                return values;
            }
        }
    }
    values.push(3);
    values
}

fn valid_build_value(value: &str, wildcard: bool) -> bool {
    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b)) {
        return false;
    }
    if wildcard {
        if value.is_empty() {
            return true;
        }
        value.split('/').all(|part| !part.is_empty())
    } else {
        !value.is_empty() && !value.contains('/')
    }
}

fn append_component(output: &mut String, value: &str) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for b in value.bytes() {
        if is_unreserved(b) {
            output.push(b as char);
        } else {
            output.push('%');
            output.push(HEX[(b >> 4) as usize] as char);
            output.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
}

fn build_route_path(
    route: &Route,
    supplied: &std::collections::HashMap<&str, &str>,
) -> String {
    let mut output = String::from("/");
    let mut emitted = false;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if emitted {
                    output.push('/');
                }
                append_component(&mut output, value);
                emitted = true;
            }
            Segment::Param(name) => {
                if emitted {
                    output.push('/');
                }
                append_component(&mut output, supplied[name.as_str()]);
                emitted = true;
            }
            Segment::Wildcard(name) => {
                let value = supplied[name.as_str()];
                if !value.is_empty() {
                    for component in value.split('/') {
                        if emitted {
                            output.push('/');
                        }
                        append_component(&mut output, component);
                        emitted = true;
                    }
                }
            }
        }
    }

    output
}

fn decode_segment(s: &str, base: usize) -> Result<String, usize> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }
            let hi = hex_upper(bytes[i + 1]).ok_or(base + i)?;
            let lo = hex_upper(bytes[i + 2]).ok_or(base + i)?;
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || unreserved(value) {
                return Err(base + i);
            }
            out.push(value);
            i += 3;
        } else if b.is_ascii() && unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(base + i);
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn split_head<'a>(s: &'a str, kind: bool) -> Result<(Vec<(&'a str, usize)>, bool), Error> {
    let err = |at| if kind { Error::InvalidPattern { at } } else { Error::InvalidPath { at } };
    if !s.starts_with('/') {
        return Err(err(0));
    }
    if s == "/" {
        return Ok((Vec::new(), false));
    }
    let mut parts = Vec::new();
    let mut start = 1;
    let bytes = s.as_bytes();
    let mut i = 1;
    while i <= bytes.len() {
        if i == bytes.len() || bytes[i] == b'/' {
            if i == start {
                if i == bytes.len() && i > 0 && bytes[i - 1] == b'/' {
                    break;
                }
                return Err(err(start));
            }
            parts.push((&s[start..i], start));
            start = i + 1;
        }
        i += 1;
    }
    let trailing = s.ends_with('/');
    if trailing {
        if parts.is_empty() {
            return Ok((Vec::new(), false));
        }
        parts.pop();
    }
    Ok((parts, trailing))
}

fn parse_pattern(s: &str) -> Result<(Vec<Segment>, bool), Error> {
    let (parts, trailing) = split_head(s, true)?;
    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();

    for (idx, (part, at)) in parts.iter().enumerate() {
        if part.starts_with(':') || part.starts_with('*') {
            let wildcard = part.starts_with('*');
            let name = &part[1..];
            if !valid_capture_name(name) || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: *at });
            }
            if wildcard && idx + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *at });
            }
            names.push(name.to_string());
            segments.push(if wildcard {
                Segment::Wildcard(name.to_string())
            } else {
                Segment::Param(name.to_string())
            });
        } else {
            let decoded = decode_segment(part, *at)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok((segments, trailing))
}

fn parse_path(s: &str) -> Result<(Vec<String>, bool), Error> {
    let end = s
        .char_indices()
        .find(|(_, c)| *c == '?' || *c == '#')
        .map(|(i, _)| i)
        .unwrap_or(s.len());
    let head = &s[..end];
    let (parts, trailing) = split_head(head, false)?;
    let mut decoded = Vec::new();
    for (part, at) in parts {
        decoded.push(decode_segment(part, at).map_err(|at| Error::InvalidPath { at })?);
    }
    Ok((decoded, trailing))
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, name: &str, method: &str, pattern: &str) -> Result<usize, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        if !valid_name(name) {
            return Err(Error::InvalidName);
        }
        let (segments, trailing) = parse_pattern(pattern)?;
        if self.routes.iter().any(|r| r.name == name) {
            return Err(Error::DuplicateName);
        }

        for route in &self.routes {
            if route.method != method {
                continue;
            }
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if same_shape(&route.segments, &segments) {
                return Err(Error::Conflict { with: route.id });
            }
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            name: name.to_string(),
            method: method.to_string(),
            segments,
            trailing,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        if !valid_method(method) {
            return Err(Error::InvalidMethod);
        }
        let (decoded, trailing) = parse_path(path)?;
        let mut candidates = Vec::<(&Route, Vec<(String, String)>, Vec<u8>)>::new();

        for route in &self.routes {
            if self.policy == SlashPolicy::Strict && route.trailing != trailing {
                continue;
            }
            if let Some((params, rank)) = route_matches(&route.segments, &decoded) {
                candidates.push((route, params, rank));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let has_head = method == "HEAD" && candidates.iter().any(|(r, _, _)| r.method == "HEAD");
        let selected_method = if method == "HEAD" && !has_head {
            "GET"
        } else {
            method
        };
        let eligible: Vec<_> = candidates
            .iter()
            .filter(|(r, _, _)| r.method == selected_method)
            .collect();

        if eligible.is_empty() {
            let mut allowed: Vec<String> = candidates
                .iter()
                .map(|(r, _, _)| r.method.clone())
                .collect();
            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
            allowed.dedup();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let best = eligible
            .into_iter()
            .max_by(|a, b| a.2.cmp(&b.2).then_with(|| b.0.id.cmp(&a.0.id)))
            .unwrap();
        let redirect = if self.policy == SlashPolicy::Redirect
            && !decoded.is_empty()
            && best.0.trailing != trailing
        {
            let mut head = path
                .char_indices()
                .find(|(_, c)| *c == '?' || *c == '#')
                .map(|(i, _)| &path[..i])
                .unwrap_or(path)
                .to_string();
            if best.0.trailing {
                if !head.ends_with('/') {
                    head.push('/');
                }
            } else if head.ends_with('/') {
                head.pop();
            }
            Some(head)
        } else {
            None
        };

        Ok(Match {
            route: best.0.id,
            params: best.1.clone(),
            redirect,
        })
    }

    pub fn build(&self, name: &str, params: &[(&str, &str)]) -> Result<String, Error> {
        let route = self.routes.iter().find(|r| r.name == name).ok_or(Error::UnknownName)?;

        let mut supplied = Vec::<(&str, &str)>::new();
        for &(key, value) in params {
            if supplied.iter().any(|(k, _)| *k == key) {
                return Err(Error::DuplicateParam {
                    name: key.to_string(),
                });
            }
            supplied.push((key, value));
        }

        let required: Vec<&str> = route
            .segments
            .iter()
            .filter_map(|segment| match segment {
                Segment::Param(n) | Segment::Wildcard(n) => Some(n.as_str()),
                Segment::Static(_) => None,
            })
            .collect();

        for key in &required {
            if !supplied.iter().any(|(k, _)| k == key) {
                return Err(Error::MissingParam {
                    name: (*key).to_string(),
                });
            }
        }

        let mut extra: Vec<&str> = supplied
            .iter()
            .map(|(key, _)| *key)
            .filter(|key| !required.contains(key))
            .collect();
        extra.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
        if let Some(key) = extra.first() {
            return Err(Error::ExtraParam {
                name: (*key).to_string(),
            });
        }

        for segment in &route.segments {
            match segment {
                Segment::Static(_) => {}
                Segment::Param(n) => {
                    let value = supplied.iter().find(|(k, _)| *k == n).unwrap().1;
                    if value.is_empty()
                        || !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || value.contains('/')
                    {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
                Segment::Wildcard(n) => {
                    let value = supplied.iter().find(|(k, _)| *k == n).unwrap().1;
                    if !value.bytes().all(|b| (0x20..=0x7e).contains(&b))
                        || (!value.is_empty()
                            && value
                                .split('/')
                                .any(|component| component.is_empty()))
                    {
                        return Err(Error::InvalidValue { name: n.clone() });
                    }
                }
            }
        }

        let mut output = String::new();
        output.push('/');
        let mut emitted = false;
        for segment in &route.segments {
            if let Segment::Wildcard(n) = segment {
                let value = supplied.iter().find(|(k, _)| *k == n).unwrap().1;
                if !value.is_empty() {
                    if emitted {
                        output.push('/');
                    }
                    for (i, component) in value.split('/').enumerate() {
                        if i > 0 {
                            output.push('/');
                        }
                        encode_component(component, &mut output);
                    }
                    emitted = true;
                }
                continue;
            }
            if emitted {
                output.push('/');
            }
            match segment {
                Segment::Static(s) => encode_component(s, &mut output),
                Segment::Param(n) => {
                    encode_component(supplied.iter().find(|(k, _)| *k == n).unwrap().1, &mut output)
                }
                Segment::Wildcard(_) => unreachable!(),
            }
            emitted = true;
        }

        if route.trailing && !output.ends_with('/') {
            output.push('/');
        }
        Ok(output)
    }
}

fn same_shape(a: &[Segment], b: &[Segment]) -> bool {
    a.len() == b.len()
        && a.iter().zip(b).all(|(a, b)| match (a, b) {
            (Segment::Static(x), Segment::Static(y)) => x == y,
            (Segment::Param(_), Segment::Param(_)) => true,
            (Segment::Wildcard(_), Segment::Wildcard(_)) => true,
            _ => false,
        })
}

fn route_matches(segments: &[Segment], path: &[String]) -> Option<(Vec<(String, String)>, Vec<u8>)> {
    let wildcard_at = segments
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));
    let fixed = wildcard_at.unwrap_or(segments.len());
    if (wildcard_at.is_none() && path.len() != segments.len()) || path.len() < fixed {
        return None;
    }

    let mut params = Vec::new();
    let mut rank = Vec::new();
    for i in 0..fixed {
        match &segments[i] {
            Segment::Static(value) => {
                if path[i] != *value {
                    return None;
                }
                rank.push(2);
            }
            Segment::Param(name) => {
                params.push((name.clone(), path[i].clone()));
                rank.push(1);
            }
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_at {
        let Segment::Wildcard(name) = &segments[index] else {
            unreachable!()
        };
        params.push((name.clone(), path[fixed..].join("/")));
        rank.push(0);
    } else {
        rank.push(3);
    }
    Some((params, rank))
}

fn encode_component(s: &str, out: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for b in s.bytes() {
        if unreserved(b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
}
