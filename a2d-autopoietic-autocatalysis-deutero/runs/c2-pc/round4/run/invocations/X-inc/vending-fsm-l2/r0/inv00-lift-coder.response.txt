#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if self.credit + u32::from(coin) as u16 > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(product) = self.products.iter().find(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(product.id.as_str()) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(product.id.clone());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let product = self
            .products
            .iter()
            .find(|product| product.id == *id)
            .expect("selected product must be in catalog");

        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let amount = self.credit - product.price;
        let change = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for coin in &change {
            let index = denomination_index(*coin).expect("change uses valid denominations");
            proposed[index] -= 1;
        }
        for coin in &self.escrow {
            let index = denomination_index(*coin).expect("escrow uses valid denominations");
            proposed[index] += 1;
        }

        for (index, count) in proposed.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        let receipt = Receipt {
            id: product.id.clone(),
            price: product.price,
            change,
        };
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|byte| byte.is_ascii_lowercase())
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|denomination| *denomination == coin)
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    let descending = [10u16, 4, 3, 1];
    let mut remaining = amount;
    let mut greedy = [0u16; 4];

    for coin in descending {
        let index = denomination_index(coin).expect("denomination is valid");
        let taken = tubes[index].min(remaining / coin);
        greedy[index] = taken;
        remaining -= taken * coin;
    }

    if remaining == 0 {
        return Some(expand_coins(&greedy));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut counts = [0u16; 4];
    search_combinations(
        0,
        amount,
        tubes,
        &descending,
        &mut counts,
        &mut best,
    );

    best.map(|counts| expand_coins(&counts))
}

fn search_combinations(
    position: usize,
    remaining: u16,
    tubes: &[u16; 4],
    descending: &[u16; 4],
    counts: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if position == descending.len() {
        if remaining != 0 {
            return;
        }

        let is_better = match best {
            None => true,
            Some(current) => {
                let total = counts.iter().map(|count| u32::from(*count)).sum::<u32>();
                let current_total = current.iter().map(|count| u32::from(*count)).sum::<u32>();
                total < current_total || (total == current_total && lex_greater(counts, current))
            }
        };
        if is_better {
            *best = Some(*counts);
        }
        return;
    }

    let coin = descending[position];
    let index = denomination_index(coin).expect("denomination is valid");
    let max = tubes[index].min(remaining / coin);
    for count in 0..=max {
        counts[index] = count;
        search_combinations(
            position + 1,
            remaining - count * coin,
            tubes,
            descending,
            counts,
            best,
        );
    }
    counts[index] = 0;
}

fn lex_greater(left: &[u16; 4], right: &[u16; 4]) -> bool {
    for coin in [10u16, 4, 3, 1] {
        let index = denomination_index(coin).expect("denomination is valid");
        if left[index] != right[index] {
            return left[index] > right[index];
        }
    }
    false
}

fn expand_coins(counts: &[u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for coin in [10u16, 4, 3, 1] {
        let index = denomination_index(coin).expect("denomination is valid");
        for _ in 0..counts[index] {
            coins.push(coin);
        }
    }
    coins
}
