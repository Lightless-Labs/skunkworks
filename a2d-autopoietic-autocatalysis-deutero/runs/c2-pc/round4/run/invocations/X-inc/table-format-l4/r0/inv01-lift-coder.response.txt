#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let minimum_width = if matches!(options.mode, Mode::Markdown) { 3 } else { 1 };
    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let minimum = overhead + n * minimum_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            let mut byte = 0;
            let chars: Vec<(usize, char)> = cell.char_indices().collect();
            let mut i = 0;
            while i < chars.len() {
                let (offset, ch) = chars[i];
                if ch == '\r' {
                    if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                        i += 2;
                        continue;
                    }
                    return Err(TableError::InvalidChar { row, column, byte: offset });
                }
                if ch != '\n' && (ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}') {
                    return Err(TableError::InvalidChar { row, column, byte: offset });
                }
                byte = offset;
                let _ = byte;
                i += 1;
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut logical: Vec<Vec<Vec<Vec<Token>>>> = Vec::with_capacity(rows.len());
    let mut widths = vec![minimum_width; n];

    for row in rows {
        let mut rendered_row = Vec::with_capacity(n);
        for (column, cell) in row.iter().enumerate() {
            let normalized = cell.replace("\r\n", "\n");
            let mut lines = Vec::new();
            for line in normalized.split('\n') {
                let trimmed = line.trim_end_matches(' ');
                let tokens: Vec<Token> = if matches!(options.mode, Mode::Markdown) {
                    let mut result = Vec::new();
                    for ch in trimmed.chars() {
                        match ch {
                            '|' => result.push(Token { text: "\\|".to_owned(), width: 2 }),
                            '\\' => result.push(Token { text: "\\\\".to_owned(), width: 2 }),
                            _ => result.push(Token { text: ch.to_string(), width: 1 }),
                        }
                    }
                    result
                } else {
                    trimmed.chars().map(|ch| Token { text: ch.to_string(), width: 1 }).collect()
                };
                let line_width: usize = tokens.iter().map(|token| token.width).sum();
                widths[column] = widths[column].max(line_width);
                lines.push(tokens);
            }
            rendered_row.push(lines);
        }
        logical.push(rendered_row);
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let mut candidate: Option<usize> = None;
            for column in 0..n {
                if widths[column] > minimum_width
                    && candidate.map_or(true, |current| widths[column] >= widths[current])
                {
                    candidate = Some(column);
                }
            }
            if let Some(column) = candidate {
                widths[column] -= 1;
            } else {
                break;
            }
        }
    }

    let mut output = String::new();
    if matches!(options.mode, Mode::Grid) {
        push_border(&mut output, &widths);
    }

    for (row_index, row) in logical.iter().enumerate() {
        let mut fragments: Vec<Vec<Vec<String>>> = Vec::with_capacity(n);
        let mut height = 1;
        for column in 0..n {
            let mut cell_fragments = Vec::new();
            for line in &row[column] {
                let parts = split_line(line, widths[column], &overflows[column]);
                cell_fragments.extend(parts);
            }
            height = height.max(cell_fragments.len());
            fragments.push(cell_fragments);
        }

        for fragment_index in 0..height {
            match options.mode {
                Mode::Grid | Mode::Markdown => output.push('|'),
                Mode::Compact => {}
            }
            for column in 0..n {
                let content = fragments[column]
                    .get(fragment_index)
                    .map(|s| s.as_str())
                    .unwrap_or("");
                let content_width = content.chars().count();
                let extra = widths[column].saturating_sub(content_width);
                let (left, right) = match alignments[column] {
                    Alignment::Left => (0, extra),
                    Alignment::Right => (extra, 0),
                    Alignment::Center => (extra / 2, extra - extra / 2),
                };

                match options.mode {
                    Mode::Grid | Mode::Markdown => {
                        output.push(' ');
                        output.extend(std::iter::repeat_n(' ', left));
                        output.push_str(content);
                        output.extend(std::iter::repeat_n(' ', right));
                        output.push(' ');
                        output.push('|');
                    }
                    Mode::Compact => {
                        if column > 0 {
                            output.push_str("  ");
                        }
                        output.extend(std::iter::repeat_n(' ', left));
                        output.push_str(content);
                        if column + 1 < n {
                            output.extend(std::iter::repeat_n(' ', right));
                        }
                    }
                }
            }
            output.push('\n');
        }

        if matches!(options.mode, Mode::Markdown) && row_index == 0 {
            output.push('|');
            for column in 0..n {
                output.push(' ');
                let width = widths[column];
                match alignments[column] {
                    Alignment::Left => {
                        output.push(':');
                        output.extend(std::iter::repeat_n('-', width - 1));
                    }
                    Alignment::Right => {
                        output.extend(std::iter::repeat_n('-', width - 1));
                        output.push(':');
                    }
                    Alignment::Center => {
                        output.push(':');
                        output.extend(std::iter::repeat_n('-', width - 2));
                        output.push(':');
                    }
                }
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }

        if matches!(options.mode, Mode::Grid) && options.header && row_index == 0 && rows.len() > 1 {
            push_border(&mut output, &widths);
        }
    }

    if matches!(options.mode, Mode::Grid) {
        push_border(&mut output, &widths);
    }

    Ok(output)
}

fn split_line(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }

    let total: usize = tokens.iter().map(|token| token.width).sum();
    if total <= width {
        return vec![tokens.iter().map(|token| token.text.as_str()).collect()];
    }

    match overflow {
        Overflow::Clip => {
            let mut result = String::new();
            let limit = width.saturating_sub(1);
            let mut used = 0;
            for token in tokens {
                if used + token.width > limit {
                    break;
                }
                result.push_str(&token.text);
                used += token.width;
            }
            result.push('…');
            vec![result]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                let fragment = tokens[start..end].iter().map(|token| token.text.as_str()).collect();
                result.push(fragment);
                start = end;
            }
            result
        }
    }
}

fn push_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for width in widths {
        output.extend(std::iter::repeat_n('-', width + 2));
        output.push('+');
    }
    output.push('\n');
}
