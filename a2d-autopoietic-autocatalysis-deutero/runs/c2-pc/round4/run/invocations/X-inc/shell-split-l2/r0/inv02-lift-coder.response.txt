#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.bytes().position(|b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i += n;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += n;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next == '"' || next == '\\' || next == '$' {
                        word.push(next);
                        active = true;
                        i += next.len_utf8();
                    } else if next == '\n' {
                        i += 1;
                    } else {
                        word.push('\\');
                        active = true;
                    }
                } else if c == '$' {
                    active = true;
                    let (text, next_i) = expand(input, i, env)?;
                    word.push_str(&text);
                    i = next_i;
                } else {
                    word.push(c);
                    active = true;
                    i += n;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i += n;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += n;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next == '\n' {
                        i += 1;
                    } else {
                        word.push(next);
                        active = true;
                        i += next.len_utf8();
                    }
                } else if c == '$' {
                    active = true;
                    let (text, next_i) = expand(input, i, env)?;
                    word.push_str(&text);
                    i = next_i;
                } else {
                    word.push(c);
                    active = true;
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(String, usize), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok(("$".to_string(), after_dollar));
    }

    let c = input[after_dollar..].chars().next().unwrap();
    if c == '{' {
        let content_start = after_dollar + 1;
        let mut j = content_start;
        let close;
        loop {
            if j >= input.len() {
                return Err(SplitError::UnterminatedExpansion { pos: dollar });
            }
            let ch = input[j..].chars().next().unwrap();
            if ch == '\\' {
                j += 1;
                if j < input.len() {
                    let escaped = input[j..].chars().next().unwrap();
                    j += escaped.len_utf8();
                }
            } else if ch == '}' {
                close = j;
                break;
            } else {
                j += ch.len_utf8();
            }
        }

        let content = &input[content_start..close];
        let split_at = content.find(":-");
        let (name, default) = if let Some(k) = split_at {
            (&content[..k], Some(&content[k + 2..]))
        } else {
            (content, None)
        };

        let mut chars = name.chars();
        let valid = match chars.next() {
            Some(first) if is_name_start(first) => chars.all(is_name_char),
            _ => false,
        };
        if !valid {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let value = lookup(name, env).unwrap_or("");
        let result = if let Some(default_text) = default {
            if value.is_empty() {
                decode_default(default_text)
            } else {
                value.to_string()
            }
        } else {
            value.to_string()
        };
        Ok((result, close + 1))
    } else if is_name_start(c) {
        let mut end = after_dollar + c.len_utf8();
        while end < input.len() {
            let next = input[end..].chars().next().unwrap();
            if is_name_char(next) {
                end += next.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        Ok((lookup(name, env).unwrap_or("").to_string(), end))
    } else {
        Ok(("$".to_string(), after_dollar))
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                out.push(next);
                i += next.len_utf8();
            }
        } else {
            out.push(c);
            i += c.len_utf8();
        }
    }
    out
}
