pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let segments = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route { segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find(|(_, c)| *c == '?' || *c == '#')
            .map(|(i, _)| i)
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parts = parse_path(head, head_end)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;

        for (route_id, route) in self.routes.iter().enumerate() {
            if let Some(params) = match_route(&route.segments, &parts) {
                let replace = match best {
                    None => true,
                    Some((best_id, _)) => {
                        route.rank > self.routes[best_id].rank
                    }
                };
                if replace {
                    best = Some((route_id, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match { route, params }),
            None => Err(Error::NotFound),
        }
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut raw_segments = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if start == i {
                    return Err(Error::InvalidPattern { at: start });
                }
                let mut segments = parse_pattern_segments(pattern, &raw_segments)?;
                let _ = &mut segments;
                return Err(Error::InvalidPattern { at: i });
            }
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }
            raw_segments.push((start, i));
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(Error::InvalidPattern { at: start });
    }
    raw_segments.push((start, bytes.len()));
    parse_pattern_segments(pattern, &raw_segments)
}

fn parse_pattern_segments(
    pattern: &str,
    ranges: &[(usize, usize)],
) -> Result<Vec<Segment>, Error> {
    let mut result = Vec::with_capacity(ranges.len());
    let mut names = std::collections::HashSet::new();

    for (index, &(start, end)) in ranges.iter().enumerate() {
        let segment = &pattern[start..end];
        let bytes = segment.as_bytes();

        if bytes[0] == b':' || bytes[0] == b'*' {
            let name = &segment[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if bytes[0] == b'*' && index + 1 != ranges.len() {
                return Err(Error::InvalidPattern { at: start });
            }
            if bytes[0] == b':' {
                result.push(Segment::Param(name.to_owned()));
            } else {
                result.push(Segment::Wildcard(name.to_owned()));
            }
        } else {
            result.push(Segment::Static(decode_segment(segment, start, true)?));
        }
    }
    Ok(result)
}

fn parse_path(path: &str, base: usize) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: base });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut ranges = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == bytes.len() - 1 {
                if start == i {
                    return Err(Error::InvalidPath { at: base + start });
                }
                let mut decoded = Vec::new();
                for &(a, b) in &ranges {
                    decoded.push(decode_segment(&path[a..b], base + a, false)?);
                }
                decode_segment(&path[start..i], base + start, false)?;
                return Err(Error::InvalidPath { at: base + i });
            }
            if i == start {
                return Err(Error::InvalidPath { at: base + start });
            }
            ranges.push((start, i));
            start = i + 1;
        }
    }
    if start == bytes.len() {
        return Err(Error::InvalidPath { at: base + start });
    }
    ranges.push((start, bytes.len()));

    let mut decoded = Vec::with_capacity(ranges.len());
    for (a, b) in ranges {
        decoded.push(decode_segment(&path[a..b], base + a, false)?);
    }
    Ok(decoded)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn decode_segment(segment: &str, base: usize, _pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(Error::InvalidPattern { at: base + i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            let decoded = match (hi, lo) {
                (Some(hi), Some(lo)) => (hi << 4) | lo,
                _ => return Err(Error::InvalidPattern { at: base + i }),
            };
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(Error::InvalidPattern { at: base + i });
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(byte) {
            out.push(byte);
            i += 1;
        } else {
            return Err(Error::InvalidPattern { at: base + i });
        }
    }

    String::from_utf8(out).map_err(|_| Error::InvalidPattern { at: base })
}

fn match_route(route: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut pi = 0;

    for segment in route {
        match segment {
            Segment::Static(value) => {
                if path.get(pi)? != value {
                    return None;
                }
                pi += 1;
            }
            Segment::Param(name) => {
                params.push((name.clone(), path.get(pi)?.clone()));
                pi += 1;
            }
            Segment::Wildcard(name) => {
                params.push((name.clone(), path[pi..].join("/")));
                return Some(params);
            }
        }
    }

    if pi == path.len() {
        Some(params)
    } else {
        None
    }
}

                    .as_ref()

                if replace {
                    best = Some((route, params));
                }
            }
        }

        best.map(|(route, params)| Match {
            route: route.id,
            params,
        })
        .ok_or(Error::NotFound)
    }

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }

    let (parts, trailing_slash) = split_segments(pattern);
    let mut parsed = Vec::with_capacity(parts.len());
    let mut names = std::collections::HashSet::new();

    for (index, (part, start)) in parts.iter().enumerate() {
        if part.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }

        let first = part.as_bytes()[0];
        if first == b':' || first == b'*' {
            let name = &part[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if first == b'*' && index + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            parsed.push(if first == b':' {
                Segment::Param(name.to_owned())
            } else {
                Segment::Wildcard(name.to_owned())
            });
        } else {
            parsed.push(Segment::Static(decode_segment(part, *start, true)?));
        }
    }

    if trailing_slash {
        return Err(Error::InvalidPattern {
            at: pattern.len() - 1,
        });
    }
    Ok(parsed)
}

fn parse_path_head(path: &str, pattern: bool) -> Result<Vec<String>, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    if !path.starts_with('/') {
        return Err(invalid(0));
    }

    let (parts, trailing_slash) = split_segments(path);
    let mut parsed = Vec::with_capacity(parts.len());
    for (part, start) in &parts {
        if part.is_empty() {
            return Err(invalid(*start));
        }
        parsed.push(decode_segment(part, *start, false).map_err(|error| match error {
            Error::InvalidPattern { at } | Error::InvalidPath { at } => invalid(at),
            Error::NotFound => Error::NotFound,
        })?);
    }

    if trailing_slash {
        return Err(invalid(path.len() - 1));
    }
    Ok(parsed)
}

fn split_segments(input: &str) -> (Vec<(&str, usize)>, bool) {
    if input == "/" {
        return (Vec::new(), false);
    }

    let bytes = input.as_bytes();
    let trailing_slash = bytes.last() == Some(&b'/');
    let end = if trailing_slash {
        input.len() - 1
    } else {
        input.len()
    };

    let mut parts = Vec::new();
    let mut start = 1;
    for index in 1..end {
        if bytes[index] == b'/' {
            parts.push((&input[start..index], start));
            start = index + 1;
        }
    }
    if start <= end {
        parts.push((&input[start..end], start));
    }

    (parts, trailing_slash)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'_')
}

fn decode_segment(segment: &str, base: usize, is_pattern: bool) -> Result<String, Error> {
    let bytes = segment.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut index = 0;

    let invalid = |at| {
        if is_pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' {
            if index + 2 >= bytes.len() {
                return Err(invalid(base + index));
            }
            let Some(high) = uppercase_hex(bytes[index + 1]) else {
                return Err(invalid(base + index));
            };
            let Some(low) = uppercase_hex(bytes[index + 2]) else {
                return Err(invalid(base + index));
            };
            let value = high * 16 + low;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(base + index));
            }
            decoded.push(value);
            index += 3;
        } else if byte.is_ascii() && is_unreserved(byte) {
            decoded.push(byte);
            index += 1;
        } else {
            return Err(invalid(base + index));
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded bytes are ASCII"))
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let has_wildcard = route
        .segments
        .last()
        .is_some_and(|segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_len = route.segments.len() - usize::from(has_wildcard);

    if (!has_wildcard && path.len() != fixed_len) || (has_wildcard && path.len() < fixed_len) {
        return None;
    }

    let mut params = Vec::new();
    for (pattern_segment, value) in route.segments[..fixed_len].iter().zip(path.iter()) {
        match pattern_segment {
            Segment::Static(expected) if expected == value => {}
            Segment::Param(name) => params.push((name.clone(), value.clone())),
            _ => return None,
        }
    }

    if let Some(Segment::Wildcard(name)) = route.segments.last().filter(|_| has_wildcard) {
        params.push((name.clone(), path[fixed_len..].join("/")));
    }

    Some(params)
}

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| (c == '?' || c == '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parts = parse_path(head)?;

        let mut best: Option<(usize, Vec<(String, String)>)> = None;
        for route in &self.routes {
            if let Some(params) = match_route(&route.segments, &parts) {
                let replace = match best.as_ref() {
                    None => true,
                    Some((best_index, _)) => {
                        let current = &self.routes[*best_index];
                        route.rank > current.rank
                    }
                };
                if replace {
                    best = Some((route.id, params));
                }
            }
        }

        best.map(|(route, params)| Match { route, params })
            .ok_or(Error::NotFound)
    }
}

fn parse_pattern(pattern: &str) -> Result<Vec<Segment>, Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok(Vec::new());
    }

    let bytes = pattern.as_bytes();
    let mut end = bytes.len();
    if bytes[end - 1] == b'/' {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut start = 1;

    while start < end {
        let stop = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map_or(end, |offset| start + offset);

        if stop == start {
            return Err(Error::InvalidPattern { at: start });
        }

        let raw = &pattern[start..stop];
        if let Some(name) = raw.strip_prefix(':') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Param(name.to_owned()));
        } else if let Some(name) = raw.strip_prefix('*') {
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: start });
            }
            if stop != end {
                return Err(Error::InvalidPattern { at: start });
            }
            segments.push(Segment::Wildcard(name.to_owned()));
        } else {
            segments.push(Segment::Static(decode_segment(raw, start, true)?));
        }

        if stop == end {
            break;
        }
        start = stop + 1;
    }

    if end < bytes.len() {
        return Err(Error::InvalidPattern { at: bytes.len() - 1 });
    }
    Ok(segments)
}

fn parse_path(path: &str) -> Result<Vec<String>, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut end = bytes.len();
    if bytes[end - 1] == b'/' {
        end -= 1;
    }

    let mut segments = Vec::new();
    let mut start = 1;
    while start < end {
        let stop = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map_or(end, |offset| start + offset);

        if stop == start {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(decode_segment(&path[start..stop], start, false)?);

        if stop == end {
            break;
        }
        start = stop + 1;
    }

    if end < bytes.len() {
        return Err(Error::InvalidPath { at: bytes.len() - 1 });
    }
    Ok(segments)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn decode_segment(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };

    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(invalid(offset + i));
            }
            let hi = hex_upper(bytes[i + 1]).ok_or_else(|| invalid(offset + i))?;
            let lo = hex_upper(bytes[i + 2]).ok_or_else(|| invalid(offset + i))?;
            let value = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&value) || value == b'/' || is_unreserved(value) {
                return Err(invalid(offset + i));
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(invalid(offset + i));
            }
            decoded.push(b);
            i += 1;
        }
    }

    // Decoding is restricted to ASCII bytes.
    Ok(String::from_utf8(decoded).expect("decoded segment is ASCII"))
}

fn hex_upper(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn match_route(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_at = pattern
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_len = wildcard_at.unwrap_or(pattern.len());

    if wildcard_at.is_none() && path.len() != pattern.len() {
        return None;
    }
    if wildcard_at.is_some() && path.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        match &pattern[i] {
            Segment::Static(value) if value != &path[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_at {
        let Segment::Wildcard(name) = &pattern[index] else {
            unreachable!()
        };
        params.push((name.clone(), path[index..].join("/")));
    }
    Some(params)
}
