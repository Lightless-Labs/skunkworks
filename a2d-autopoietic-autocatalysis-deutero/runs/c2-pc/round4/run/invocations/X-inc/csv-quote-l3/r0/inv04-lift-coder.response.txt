#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|column| column == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|column| column == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with_config(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_config(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let records = parse_records(input, config.delimiter, config.quote)?;
    build_table(records, config)
}

fn parse_records(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut record_index = 0usize;
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut opening_quote = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != 2 && byte == b'\r' {
            if i + 1 < bytes.len() && bytes[i + 1] == b'\n' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record_index += 1;
                record_started = false;
                state = 0;
                i += 2;
                continue;
            }
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        if state != 2 && byte == b'\n' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record_index += 1;
            record_started = false;
            state = 0;
            i += 1;
            continue;
        }

        match state {
            0 => {
                if byte == delimiter {
                    row.push(std::mem::take(&mut field));
                    record_started = true;
                } else if byte == quote {
                    opening_quote = i;
                    record_started = true;
                    state = 2;
                } else {
                    if byte < 0x80 {
                        field.push(byte as char);
                        i += 1;
                    } else {
                        let ch = input[i..].chars().next().unwrap();
                        field.push(ch);
                        i += ch.len_utf8();
                    }
                    record_started = true;
                    state = 1;
                    continue;
                }
            }
            1 => {
                if byte == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = 0;
                } else if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else if byte < 0x80 {
                    field.push(byte as char);
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                    continue;
                }
            }
            2 => {
                if byte == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                        continue;
                    }
                    state = 3;
                } else if byte < 0x80 {
                    field.push(byte as char);
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                    continue;
                }
            }
            3 => {
                if byte == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = 0;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
            _ => unreachable!(),
        }
        i += 1;
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if record_started || !row.is_empty() || !field.is_empty() || state == 3 {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    let mut iter = records.into_iter();
    let Some(header) = iter.next() else {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    };

    if config.duplicates == DuplicatePolicy::Reject {
        for second in 0..header.len() {
            if let Some(first) = header[..second]
                .iter()
                .position(|value| value == &header[second])
            {
                return Err(CsvError::DuplicateHeader {
                    name: header[second].clone(),
                    first,
                    second,
                });
            }
        }
    }

    let rows: Vec<Vec<String>> = iter.collect();
    for (offset, row) in rows.iter().enumerate() {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: offset + 1,
                expected: header.len(),
                actual: row.len(),
            });
        }
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

    Name(String),

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {

    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output_rows = Vec::new();
    let mut errors = Vec::new();

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut stop = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if stop {
                break;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let (column, value, error) = match resolved {
                None => (None, None, Some(CellErrorKind::UnknownColumn)),
                Some(index) => match row.get(index) {
                    None => (Some(index), None, Some(CellErrorKind::MissingCell)),
                    Some(cell) => match convert_cell(cell, spec) {
                        Ok(value) => (Some(index), Some(value), None),
                        Err(kind) => (Some(index), None, Some(kind)),
                    },
                },
            };

            if let Some(value) = value {
                output[spec_index] = Some(value);
            } else if let Some(kind) = error {
                errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column,
                    kind,
                });
                if policy != ErrorPolicy::Collect {
                    stop = true;
                }
            }
        }

        output_rows.push(output);
        if policy == ErrorPolicy::Stop && stop {
            break;
        }
    }

    TypedResult {
        rows: output_rows,
        errors,
    }
}

fn convert_cell(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => parse_integer(cell),
        ColumnType::Decimal => parse_decimal(cell),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

fn parse_integer(text: &str) -> Result<Value, CellErrorKind> {
    let bytes = text.as_bytes();
    let mut i = 0;
    let negative = bytes.first() == Some(&b'-');
    if negative {
        i = 1;
    }

    let valid = if i == bytes.len() {
        false
    } else if bytes[i] == b'0' {
        i + 1 == bytes.len() && !negative
    } else if (b'1'..=b'9').contains(&bytes[i]) {
        bytes[i + 1..].iter().all(u8::is_ascii_digit)
    } else {
        false
    };

    if !valid {
        return Err(CellErrorKind::InvalidInteger);
    }

    text.parse::<i64>()
        .map(Value::Integer)
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(text: &str) -> Result<Value, CellErrorKind> {
    let bytes = text.as_bytes();
    let mut start = 0;
    if bytes.first() == Some(&b'-') {
        start = 1;
    }

    let dot = match bytes.iter().position(|&byte| byte == b'.') {
        Some(dot) => dot,
        None => return Err(CellErrorKind::InvalidDecimal),
    };

    let integer = &bytes[start..dot];
    let fraction = &bytes[dot + 1..];
    let integer_valid = if integer == b"0" {
        true
    } else {
        !integer.is_empty()
            && (b'1'..=b'9').contains(&integer[0])
            && integer[1..].iter().all(u8::is_ascii_digit)
    };
    if !integer_valid || fraction.len() != 2 || !fraction.iter().all(u8::is_ascii_digit) {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = start == 1;
    if negative && integer == b"0" && fraction == b"00" {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let magnitude_text = format!(
        "{}{}{}",
        if negative { "-" } else { "" },
        std::str::from_utf8(integer).unwrap(),
        std::str::from_utf8(fraction).unwrap()
    );
    magnitude_text
        .parse::<i64>()
        .map(Value::Decimal)
        .map_err(|_| CellErrorKind::DecimalOverflow)
}
