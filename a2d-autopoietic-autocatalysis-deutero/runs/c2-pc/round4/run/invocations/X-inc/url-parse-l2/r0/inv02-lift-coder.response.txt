#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            let (Some(hi), Some(lo)) = (hi, lo) else {
                return Err(CodecError::BadEscape { at: i });
            };
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_offset = error.valid_up_to();
            let original_offset = origins.get(decoded_offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_offset })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = matches!(
            byte,
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~'
        ) || match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        };

        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn codec_to_parse_error(error: CodecError, offset: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn validate_component(text: &str, offset: usize, component: Component) -> Result<(), ParseError> {
    decode_component(component, text)
        .map(|_| ())
        .map_err(|error| codec_to_parse_error(error, offset))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for at in 1..colon {
        let byte = bytes[at];
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|relative| authority_start + relative)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let authority_bytes = authority.as_bytes();

    let mut at_signs = authority_bytes
        .iter()
        .enumerate()
        .filter(|(_, byte)| **byte == b'@')
        .map(|(at, _)| at);

    let first_at = at_signs.next();
    if let Some(second_at) = at_signs.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second_at,
        });
    }

    let (userinfo, host_start_rel) = if let Some(userinfo_end) = first_at {
        if userinfo_end == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + userinfo_end,
            });
        }
        (Some(authority[..userinfo_end].to_owned()), userinfo_end + 1)
    } else {
        (None, 0)
    };

    let host_abs_start = authority_start + host_start_rel;
    let host_port = &authority[host_start_rel..];
    let host_port_bytes = host_port.as_bytes();

    let (host, port, ipv6, port_abs_start) = if host_port_bytes.first() == Some(&b'[') {
        let closing = host_port_bytes.iter().position(|&byte| byte == b']');
        let Some(closing) = closing else {
            return Err(ParseError::InvalidIpv6 { at: host_abs_start });
        };
        let interior = &host_port[1..closing];
        if interior.is_empty()
            || interior.contains('%')
            || interior.parse::<std::net::Ipv6Addr>().is_err()
        {
            return Err(ParseError::InvalidIpv6 { at: host_abs_start });
        }

        let suffix_start = closing + 1;
        let (port, port_abs_start) = if suffix_start == host_port.len() {
            (None, None)
        } else if host_port_bytes[suffix_start] == b':' {
            let start = suffix_start + 1;
            let parsed = parse_port(&host_port[start..], host_abs_start + start)?;
            (Some(parsed), Some(host_abs_start + start))
        } else {
            return Err(ParseError::InvalidHost {
                at: host_abs_start + suffix_start,
            });
        };

        (
            interior.to_ascii_lowercase(),
            port,
            true,
            port_abs_start,
        )
    } else {
        let host_port_colon = host_port_bytes.iter().position(|&byte| byte == b':');
        let (host_text, port, port_abs_start) = if let Some(split) = host_port_colon {
            let start = split + 1;
            let parsed = parse_port(&host_port[start..], host_abs_start + start)?;
            (
                &host_port[..split],
                Some(parsed),
                Some(host_abs_start + start),
            )
        } else {
            (host_port, None, None)
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_abs_start });
        }
        for (relative, &byte) in host_text.as_bytes().iter().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_abs_start + relative,
                });
            }
        }

        (host_text.to_ascii_lowercase(), port, false, port_abs_start)
    };

    let _ = port_abs_start;

    let fragment_rel = bytes[authority_end..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|relative| authority_end + relative);

    let before_fragment_end = fragment_rel.unwrap_or(bytes.len());
    let query_rel = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|relative| authority_end + relative);

    let path_end = query_rel.unwrap_or(before_fragment_end);
    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_rel.map(|query_at| {
        input[query_at + 1..before_fragment_end].to_owned()
    });
    let fragment = fragment_rel.map(|fragment_at| input[fragment_at + 1..].to_owned());

    if let Some(userinfo) = userinfo.as_deref() {
        validate_component(userinfo, authority_start, Component::UserInfo)?;
    }
    if path_end != authority_end {
        validate_component(&path, authority_end, Component::Path)?;
    }
    if let Some(query_at) = query_rel {
        validate_component(
            query.as_deref().unwrap_or(""),
            query_at + 1,
            Component::Query,
        )?;
    }
    if let Some(fragment_at) = fragment_rel {
        validate_component(
            fragment.as_deref().unwrap_or(""),
            fragment_at + 1,
            Component::Fragment,
        )?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(text: &str, absolute_start: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort {
            at: absolute_start,
        });
    }

    for (relative, byte) in text.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: absolute_start + relative,
            });
        }
    }

    if text.len() >= 2 && text.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort {
            at: absolute_start,
        });
    }

    let mut value: u32 = 0;
    for byte in text.bytes() {
        value = match value.checked_mul(10).and_then(|value| value.checked_add((byte - b'0') as u32)) {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: absolute_start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: absolute_start });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }
    Ok(value as u16)
}
