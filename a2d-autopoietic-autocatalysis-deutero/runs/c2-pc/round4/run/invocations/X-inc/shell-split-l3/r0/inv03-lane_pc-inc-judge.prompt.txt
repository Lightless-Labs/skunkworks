    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&'a str, &'a str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn append_text(word: &mut Word, text: &str, active_glob: bool) {
    for c in text.chars() {
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),

            Quote::Single => {

                    quote = Quote::None;

                }

            }

                            word.text.push(n);
                        }
                        i = after_n;
                    } else {
                        word.text.push('\\');
                        i = next_i;
                    }

                        append_text(&mut word, value, false);
                        i = end;
                    } else {
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    word.text.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single;
                    quote_pos = i;
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double;
                    quote_pos = i;
                    active = true;
                    i = next_i;

                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if n == '\n' {
                        i = after_n;
                    } else {
                        active = true;
                        append_text(&mut word, &n.to_string(), false);
                        i = after_n;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });

                    let (consumed, value) = parse_expansion(input, i, env)?;

                        active = true;

                        i = end;

                        active = true;
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    active = true;
                    append_text(&mut word, &c.to_string(), true);
                    i = next_i;
                }
            }
        }

        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_expansion<'a>(
    input: &'a str,
    dollar: usize,
    env: &'a [(&'a str, &'a str)],
) -> Result<(Option<usize>, Option<&'a str>), SplitError> {
    let after_dollar = dollar + 1;
    let Some(first) = input[after_dollar..].chars().next() else {
        return Ok((None, None));
    };

    if first == '{' {
        let content_start = after_dollar + 1;
        let mut j = content_start;
        let mut close = None;

        while j < input.len() {
            let c = input[j..].chars().next().unwrap();
            let end = j + c.len_utf8();
            if c == '\\' {
                if end < input.len() {
                    let skipped = input[end..].chars().next().unwrap();
                    j = end + skipped.len_utf8();
                } else {
                    j = end;
                }
            } else if c == '}' {
                close = Some(j);
                break;
            } else {
                j = end;
            }
        }

        let Some(close_pos) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        };

        let contents = &input[content_start..close_pos];
        let bytes = contents.as_bytes();
        let mut name_end = 0;
        if bytes.first().copied().map(|b| b.is_ascii_alphabetic() || b == b'_') != Some(true) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        name_end += 1;
        while name_end < bytes.len()
            && (bytes[name_end].is_ascii_alphanumeric() || bytes[name_end] == b'_')
        {
            name_end += 1;
        }
        let name = &contents[..name_end];

        let value = if name_end == contents.len() {
            lookup(name, env).unwrap_or("")
        } else if contents[name_end..].starts_with(":-") {
            let default_start = name_end + 2;
            let existing = lookup(name, env).unwrap_or("");
            if existing.is_empty() {
                return Ok((
                    Some(close_pos + 1),
                    Some(Box::leak(decode_default(&contents[default_start..]).into_boxed_str())),
                ));
            }
            existing
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok((Some(close_pos + 1), Some(value)));
    }

    if !is_name_start(first) {
        return Ok((None, None));
    }

    let mut end = after_dollar + first.len_utf8();
    while end < input.len() {
        let c = input[end..].chars().next().unwrap();
        if is_name_continue(c) {
            end += c.len_utf8();
        } else {
            break;
        }
    }

    let name = &input[after_dollar..end];
    Ok((Some(end), Some(lookup(name, env).unwrap_or(""))))
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        let next = i + c.len_utf8();
        if c == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            out.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            out.push(c);
            i = next;
        }
    }
    out
}

pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn append(text: &mut String, glob: &mut Vec<usize>, c: char, active: bool) {
    let offset = text.len();
    text.push(c);
    if active && matches!(c, '*' | '?' | '[') {
        glob.push(offset);
    }
}

fn append_string(text: &mut String, glob: &mut Vec<usize>, value: &str, active: bool) {
    for c in value.chars() {
        append(text, glob, c, active);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active_word = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, c)) = chars.next() {
        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append(&mut current.text, &mut current.glob, c, false);
                }
            }
            Quote::Double => match c {
                '"' => quote = Quote::None,
                '\\' => {
                    let Some(&(next_pos, next)) = chars.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        chars.next();
                        if next != '\n' {
                            append(&mut current.text, &mut current.glob, next, false);
                        }
                    } else {
                        append(&mut current.text, &mut current.glob, '\\', false);
                        let _ = next_pos;
                    }
                }
                '`' => return Err(SplitError::CommandSubstitution { pos }),
                '$' => {
                    if chars.peek().is_some_and(|(_, next)| *next == '(') {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if chars.peek().is_some_and(|(_, next)| *next == '{') {
                        let expansion = parse_braced(input, pos, &mut chars, env)?;
                        append_string(&mut current.text, &mut current.glob, &expansion, false);
                        active_word = true;
                    } else if chars.peek().is_some_and(|(_, next)| is_name_start(*next)) {
                        let mut name = String::new();
                        while let Some(&(_, next)) = chars.peek() {
                            if is_name_continue(next) {
                                name.push(next);
                                chars.next();
                            } else {
                                break;
                            }
                        }
                        if let Some(value) = lookup(env, &name) {
                            append_string(&mut current.text, &mut current.glob, value, false);
                        }
                        active_word = true;
                    } else {
                        append(&mut current.text, &mut current.glob, '$', false);
                    }
                }
                _ => append(&mut current.text, &mut current.glob, c, false),
            },
            Quote::None => match c {
                ' ' | '\t' | '\r' | '\n' => {
                    if active_word {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active_word = false;
                    }
                }
                '\'' | '"' => {
                    quote = if c == '\'' { Quote::Single } else { Quote::Double };
                    quote_start = pos;
                    active_word = true;
                }
                '\\' => {
                    let Some(&(_, next)) = chars.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    chars.next();
                    if next != '\n' {
                        append(&mut current.text, &mut current.glob, next, false);
                        active_word = true;
                    }
                }
                '`' => return Err(SplitError::CommandSubstitution { pos }),
                '$' => {
                    if chars.peek().is_some_and(|(_, next)| *next == '(') {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    if chars.peek().is_some_and(|(_, next)| *next == '{') {
                        let expansion = parse_braced(input, pos, &mut chars, env)?;
                        append_string(&mut current.text, &mut current.glob, &expansion, true);
                        active_word = true;
                    } else if chars.peek().is_some_and(|(_, next)| is_name_start(*next)) {
                        let mut name = String::new();
                        while let Some(&(_, next)) = chars.peek() {
                            if is_name_continue(next) {
                                name.push(next);
                                chars.next();
                            } else {
                                break;
                            }
                        }
                        if let Some(value) = lookup(env, &name) {
                            append_string(&mut current.text, &mut current.glob, value, true);
                        }
                        active_word = true;
                    } else {
                        append(&mut current.text, &mut current.glob, '$', true);
                        active_word = true;
                    }
                }
                _ => {
                    append(&mut current.text, &mut current.glob, c, true);
                    active_word = true;
                }
            },
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active_word {
        words.push(current);
    }
    Ok(words)
}

fn parse_braced(
    input: &str,
    dollar_pos: usize,
    chars: &mut std::iter::Peekable<std::str::CharIndices<'_>>,
    env: &[(&str, &str)],
) -> Result<String, SplitError> {
    chars.next(); // consume opening brace

    let content_start = chars.peek().map_or(input.len(), |(pos, _)| *pos);
    let mut search = chars.clone();
    let mut closing = None;

    while let Some((pos, c)) = search.next() {
        if c == '\\' {
            search.next();
        } else if c == '}' {
            closing = Some(pos);
            break;
        }
    }

    let Some(close_pos) = closing else {
        return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
    };

    let contents = &input[content_start..close_pos];
    let name_end = contents
        .char_indices()
        .find(|(_, c)| !is_name_continue(*c))
        .map_or(contents.len(), |(i, _)| i);

    if name_end == 0 || !contents.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let name = &contents[..name_end];
    let suffix = &contents[name_end..];

    let selected = if suffix.is_empty() {
        lookup(env, name).unwrap_or("")
    } else if let Some(default_text) = suffix.strip_prefix(":-") {
        match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default_text),
        }
        .leak()
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    while let Some((pos, _)) = chars.peek().copied() {
        if pos > close_pos {
            break;
        }
        chars.next();
    }

    Ok(selected.to_string())
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(ch: char) -> bool {
    ch.is_ascii_alphabetic() || ch == '_'
}

fn is_name_continue(ch: char) -> bool {
    ch.is_ascii_alphanumeric() || ch == '_'
}

fn append_scalar(word: &mut Word, ch: char, active_glob: bool) {
    let offset = word.text.len();
    word.text.push(ch);
    if active_glob && matches!(ch, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_text(word: &mut Word, text: &str, active_glob: bool) {
    for ch in text.chars() {
        append_scalar(word, ch, active_glob);
    }
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn decode_default(text: &str) -> String {
    let mut decoded = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, ch)) = chars.next() {
        if ch == '\\' {
            if let Some((_, next)) = chars.next() {
                decoded.push(next);
            }
        } else {
            decoded.push(ch);
        }
    }

    decoded
}

fn parse_braced(
    input: &str,
    dollar_pos: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_open = dollar_pos + 2;
    let mut cursor = after_open;
    let mut close = None;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        if ch == '\\' {
            cursor += ch.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if ch == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += ch.len_utf8();
        }
    }

    let close_pos = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar_pos })?;
    let contents = &input[after_open..close_pos];

    let name_end = contents
        .char_indices()
        .find_map(|(i, ch)| (!is_name_continue(ch)).then_some(i))
        .unwrap_or(contents.len());

    let name = &contents[..name_end];
    if name.is_empty()
        || !name.chars().next().map(is_name_start).unwrap_or(false)
        || !name.chars().all(is_name_continue)
    {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    }

    let remainder = &contents[name_end..];
    let value = if remainder.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar_pos });
    };

    Ok((value, close_pos + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let next_cursor = cursor + ch.len_utf8();

        match quote {
            Quote::Single(open_pos) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    append_scalar(&mut word, ch, false);
                }
                cursor = next_cursor;
                let _ = open_pos;
            }
            Quote::Double(open_pos) => {
                if ch == '"' {
                    quote = Quote::None;
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_cursor..].chars().next().unwrap();
                    let after_escaped = next_cursor + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$' | '\n') {
                        if escaped != '\n' {
                            append_scalar(&mut word, escaped, false);
                        }
                        cursor = after_escaped;
                    } else {
                        append_scalar(&mut word, '\\', false);
                        cursor = next_cursor;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if next_cursor < input.len() {
                        let next = input[next_cursor..].chars().next().unwrap();
                        if next == '(' {
                            return Err(SplitError::CommandSubstitution { pos: cursor });
                        }
                        if next == '{' {
                            let (value, after) = parse_braced(input, cursor, env)?;
                            append_text(&mut word, &value, false);
                            active = true;
                            cursor = after;
                            continue;
                        }
                        if is_name_start(next) {
                            let mut end = next_cursor;
                            while end < input.len() {
                                let item = input[end..].chars().next().unwrap();
                                if !is_name_continue(item) {
                                    break;
                                }
                                end += item.len_utf8();
                            }
                            let name = &input[next_cursor..end];
                            let value = lookup(name, env).unwrap_or("");
                            append_text(&mut word, value, false);
                            active = true;
                            cursor = end;
                            continue;
                        }
                    }
                    append_scalar(&mut word, '$', false);
                    active = true;
                    cursor = next_cursor;
                } else {
                    append_scalar(&mut word, ch, false);
                    cursor = next_cursor;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if is_separator(ch) {
                    if active {
                        words.push(std::mem::replace(
                            &mut word,
                            Word {
                                text: String::new(),
                                glob: Vec::new(),
                            },
                        ));
                        active = false;
                    }
                    cursor = next_cursor;
                } else if ch == '\'' {
                    quote = Quote::Single(cursor);
                    active = true;
                    cursor = next_cursor;
                } else if ch == '"' {
                    quote = Quote::Double(cursor);
                    active = true;
                    cursor = next_cursor;
                } else if ch == '\\' {
                    if next_cursor == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next_cursor..].chars().next().unwrap();
                    let after_escaped = next_cursor + escaped.len_utf8();
                    if escaped == '\n' {
                        cursor = after_escaped;
                    } else {
                        append_scalar(&mut word, escaped, false);
                        active = true;
                        cursor = after_escaped;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if next_cursor < input.len() {
                        let next = input[next_cursor..].chars().next().unwrap();
                        if next == '(' {
                            return Err(SplitError::CommandSubstitution { pos: cursor });
                        }
                        if next == '{' {
                            let (value, after) = parse_braced(input, cursor, env)?;
                            append_text(&mut word, &value, true);
                            active = true;
                            cursor = after;
                            continue;
                        }
                        if is_name_start(next) {
                            let mut end = next_cursor;
                            while end < input.len() {
                                let item = input[end..].chars().next().unwrap();
                                if !is_name_continue(item) {
                                    break;
                                }
                                end += item.len_utf8();
                            }
                            let name = &input[next_cursor..end];
                            let value = lookup(name, env).unwrap_or("");
                            append_text(&mut word, value, true);
                            active = true;
                            cursor = end;
                            continue;
                        }
                    }
                    append_scalar(&mut word, '$', false);
                    active = true;
                    cursor = next_cursor;
                } else {
                    append_scalar(&mut word, ch, true);
                    active = true;
                    cursor = next_cursor;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
