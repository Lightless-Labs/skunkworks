#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut record_index = 0usize;
    let mut field_start = true;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: i });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let start = i;
                i += 1;
                while i < bytes.len() && (bytes[i] & 0xc0) == 0x80 {
                    i += 1;
                }
                field.push_str(std::str::from_utf8(&bytes[start..i]).unwrap());
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                after_quote = false;
                field_start = true;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                after_quote = false;
                field_start = true;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record: record_index, byte: i });
        }

        if b == quote {
            if field_start {
                in_quotes = true;
                quote_open = i;
                field_start = false;
                i += 1;
                continue;
            }
            return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
        }
        if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_start = true;
            i += 1;
            continue;
        }
        if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            field_start = true;
            i += if b == b'\r' { 2 } else { 1 };
            continue;
        }

        field_start = false;
        let start = i;
        i += 1;
        while i < bytes.len() && (bytes[i] & 0xc0) == 0x80 {
            i += 1;
        }
        field.push_str(std::str::from_utf8(&bytes[start..i]).unwrap());
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_open });
    }

    if !record.is_empty() || !field.is_empty() || field_start || after_quote {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut header = None;
    let mut rows = records;
    if config.headers && !rows.is_empty() {
        let h = rows.remove(0);
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in h.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(old, _)| *old == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name, second));
            }
        }
        for (index, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
        header = Some(h);
    }
    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub struct CellError {

    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let digits = if b.first() == Some(&b'-') { &b[1..] } else { b };
    let valid = if digits == b"0" {
        b.first() != Some(&b'-')
    } else {
        !digits.is_empty()
            && (b'1'..=b'9').contains(&digits[0])
            && digits[1..].iter().all(u8::is_ascii_digit)
    };
    if !valid {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    let b = s.as_bytes();
    let neg = b.first() == Some(&b'-');
    let body = if neg { &b[1..] } else { b };
    let Some(dot) = body.iter().position(|&x| x == b'.') else {
        return Err(CellErrorKind::InvalidDecimal);
    };
    let whole = &body[..dot];
    let frac = &body[dot + 1..];
    let whole_valid = whole == b"0"
        || (!whole.is_empty()
            && (b'1'..=b'9').contains(&whole[0])
            && whole[1..].iter().all(u8::is_ascii_digit));
    if !whole_valid || frac.len() != 2 || !frac.iter().all(u8::is_ascii_digit) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    if neg && whole == b"0" && frac == b"00" {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let mut n: i128 = 0;
    for &digit in whole.iter().chain(frac) {
        n = n * 10 + (digit - b'0') as i128;
    }
    if neg {
        n = -n;
    }
    i64::try_from(n).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            let index = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, value) = match index {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(i) => match row.get(i) {
                    None => (Some(i), Err(CellErrorKind::MissingCell)),
                    Some(cell) => (Some(i), evaluate(cell, spec)),
                },
            };
            match value {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    result.errors.push(CellError { row: row_index, spec: spec_index, column, kind });
                    failed = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }
        result.rows.push(output);
        if failed && policy == ErrorPolicy::Stop {
            break;
        }
    }
    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn records_of(table: &Table) -> Vec<&Vec<String>> {
    let mut records = Vec::new();
    if let Some(header) = &table.header {
        records.push(header);
    }
    records.extend(table.rows.iter());
    records
}

fn duplicate_header(header: &[String]) -> Option<(String, usize, usize)> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(old, _)| *old == name) {
            return Some((name.clone(), *first, second));
        }
        seen.push((name, second));
    }
    None
}

fn quote_field(field: &str, first: bool, sole: bool, delimiter: u8, quote: u8) -> String {
    let bytes = field.as_bytes();
    let needs = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
        || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t')
        || (first && field.starts_with('\u{feff}'))
        || (sole && field.is_empty());
    if !needs {
        return field.to_string();
    }
    let mut out = String::new();
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let original = records_of(table);
    for (record_index, record) in original.iter().enumerate() {
        for (column, field) in record.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul { record: record_index, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, fields) in original.iter().enumerate() {
        if fields.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some((name, first, second)) = duplicate_header(table.header.as_ref().unwrap()) {
            return Err(WriteError::DuplicateHeader { name, first, second });
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                output_records.push(header.clone());
            }
            output_records.extend(table.rows.iter().cloned());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained: Vec<usize> = Vec::new();
                for (i, name) in header.iter().enumerate() {
                    let matching: Vec<usize> = header.iter().enumerate()
                        .filter_map(|(j, other)| if other == name { Some(j) } else { None })
                        .collect();
                    let chosen = match table.duplicates {
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => matching[0],
                        DuplicatePolicy::KeepLast => *matching.last().unwrap(),
                    };
                    if chosen == i && !retained.contains(&i) {
                        retained.push(i);
                    }
                }
                retained.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));
                output_records.push(retained.iter().map(|&i| header[i].clone()).collect());
                let mut projected: Vec<Vec<String>> = table.rows.iter()
                    .map(|row| retained.iter().map(|&i| row[i].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b) {
                        let ord = x.as_bytes().cmp(y.as_bytes());
                        if ord != std::cmp::Ordering::Equal {
                            return ord;
                        }
                    }
                    a.len().cmp(&b.len())
                });
                output_records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b) {
                        let ord = x.as_bytes().cmp(y.as_bytes());
                        if ord != std::cmp::Ordering::Equal {
                            return ord;
                        }
                    }
                    a.len().cmp(&b.len())
                });
                output_records = rows;
            }
        }
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let mut rendered = Vec::with_capacity(output_records.len());
    for (ri, record) in output_records.iter().enumerate() {
        let mut fields = Vec::with_capacity(record.len());
        for (ci, field) in record.iter().enumerate() {
            fields.push(quote_field(field, ri == 0 && ci == 0, record.len() == 1, config.delimiter, config.quote));
        }
        rendered.push(fields.join(&(config.delimiter as char).to_string()));
    }

    let (ending, final_terminator) = match config.mode {
        WriteMode::Preserve => (
            match config.line_ending {
                LineEnding::Lf => "\n",
                LineEnding::CrLf => "\r\n",
            },
            config.final_terminator,
        ),
        WriteMode::Canonical => ("\n", true),
    };
    let mut out = rendered.join(ending);
    if final_terminator {
        out.push_str(ending);
    }
    Ok(out)
}
