#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut quote_start = 0usize;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul {
                record: records.len(),
                byte: i,
            });
        }

        if state != 2 && b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: records.len(),
                    byte: i,
                });
            }
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            state = 0;
            i += 2;
            continue;
        }

        if state == 2 {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                } else {
                    state = 3;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if state == 3 {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                state = 0;
                i += 1;
            } else if b == b'\n' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                state = 0;
                i += 1;
            } else {
                return Err(CsvError::TrailingByte {
                    record: records.len(),
                    byte: i,
                });
            }
            continue;
        }

        if b == b'\n' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            state = 0;
            i += 1;
        } else if b == delimiter {
            record.push(std::mem::take(&mut field));
            state = 0;
            i += 1;
        } else if b == quote {
            if state == 0 {
                quote_start = i;
                state = 2;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote {
                    record: records.len(),
                    byte: i,
                });
            }
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            state = 1;
            i += ch.len_utf8();
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record: records.len(),
            byte: quote_start,
        });
    }

    if !record.is_empty() || !field.is_empty() || state != 0 {
        record.push(field);
        records.push(record);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn first_duplicate(header: &[String]) -> Option<(String, usize, usize)> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(existing, _)| *existing == name) {
            return Some((name.clone(), *first, second));
        }
        seen.push((name, second));
    }
    None
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut records = parse_with_dialect(input, config.delimiter, config.quote)?;
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(ref names) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some((name, first, second)) = first_duplicate(names) {
                return Err(CsvError::DuplicateHeader {
                    name,
                    first,
                    second,
                });
            }
        }

        for (row_index, row) in records.iter().enumerate() {
            if row.len() != names.len() {
                return Err(CsvError::HeaderWidth {
                    record: row_index + 1,
                    expected: names.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_table(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }

    let negative = bytes[0] == b'-';
    let start = usize::from(negative);
    if start == bytes.len() {
        return false;
    }

    if bytes[start] == b'0' {
        return !negative && start + 1 == bytes.len();
    }

    (b'1'..=b'9').contains(&bytes[start])
        && bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    let negative = bytes.first() == Some(&b'-');
    let start = usize::from(negative);

    if bytes.len() < start + 4 {
        return false;
    }

    let dot = bytes.len() - 3;
    if dot < start + 1 || bytes[dot] != b'.' {
        return false;
    }
    if !bytes[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }

    let integer = &bytes[start..dot];
    if integer.is_empty() {
        return false;
    }

    let valid_integer = if integer[0] == b'0' {
        integer.len() == 1
    } else {
        (b'1'..=b'9').contains(&integer[0])
            && integer[1..].iter().all(u8::is_ascii_digit)
    };
    if !valid_integer {
        return false;
    }

    !(negative && integer == b"0" && bytes[dot + 1..] == *b"00")
}

fn parse_decimal_value(s: &str) -> Option<i64> {
    let negative = s.starts_with('-');
    let unsigned = if negative { &s[1..] } else { s };
    let dot = unsigned.len() - 3;
    let whole: i128 = unsigned[..dot].parse().ok()?;
    let frac: i128 = unsigned[dot + 1..].parse().ok()?;
    let magnitude = whole.checked_mul(100)?.checked_add(frac)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).ok()
}

fn evaluate_cell(
    table: &Table,
    row_index: usize,
    row: &[String],
    spec_index: usize,
    spec: &ColumnSpec,
) -> Result<Value, CellError> {
    let column = match &spec.column {
        ColumnRef::Index(index) => Some(*index),
        ColumnRef::Name(name) => table.column_index(name),
    };
    let Some(column) = column else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: None,
            kind: CellErrorKind::UnknownColumn,
        });
    };
    let Some(value) = row.get(column) else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: Some(column),
            kind: CellErrorKind::MissingCell,
        });
    };

    if value.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind: CellErrorKind::Empty,
            });
        }
    }

    let err = |kind| CellError {
        row: row_index,
        spec: spec_index,
        column: Some(column),
        kind,
    };

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(value.clone())),
        ColumnType::Integer => {
            if !integer_syntax(value) {
                return Err(err(CellErrorKind::InvalidInteger));
            }
            value
                .parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| err(CellErrorKind::IntegerOverflow))
        }
        ColumnType::Decimal => {
            if !decimal_syntax(value) {
                return Err(err(CellErrorKind::InvalidDecimal));
            }
            parse_decimal_value(value)
                .map(Value::Decimal)
                .ok_or_else(|| err(CellErrorKind::DecimalOverflow))
        }
        ColumnType::Boolean => match value.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(err(CellErrorKind::InvalidBoolean)),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };
    let mut stopped = false;

    for (row_index, row) in table.rows.iter().enumerate() {
        if stopped {
            break;
        }
        let mut output = vec![None; columns.len()];
        for (spec_index, spec) in columns.iter().enumerate() {
            match evaluate_cell(table, row_index, row, spec_index, spec) {
                Ok(value) => output[spec_index] = Some(value),
                Err(error) => {
                    result.errors.push(error);
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow => break,
                        ErrorPolicy::Stop => {
                            stopped = true;
                            break;
                        }
                    }
                }
            }
        }
        result.rows.push(output);
    }
    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig {
        component: ConfigPart,
    },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns {
        record: usize,
    },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn write_field(
    field: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole: bool,
) -> String {
    let bytes = field.as_bytes();
    let needs_quote = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_first_record && field.starts_with('\u{feff}'))
        || (sole && field.is_empty());

    if !needs_quote {
        return field.to_string();
    }

    let mut out = String::with_capacity(field.len() + 2);
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
        }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header);
    }
    original_records.extend(table.rows.iter().map(Vec::as_slice));

    for (record_index, record) in original_records.iter().enumerate() {
        for (column_index, field) in record.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (row_index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: row_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, record) in original_records.iter().enumerate() {
        if record.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let (Some(header), DuplicatePolicy::Reject) = (&table.header, table.duplicates) {
            if let Some((name, first, second)) = first_duplicate(header) {
                return Err(WriteError::DuplicateHeader {
                    name,
                    first,
                    second,
                });
            }
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                output_records.push(header.clone());
            }
            output_records.extend(table.rows.iter().cloned());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut selected: Vec<(String, usize)> = Vec::new();
                for (index, name) in header.iter().enumerate() {
                    let chosen = match table.duplicates {
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            !header[..index].iter().any(|prior| prior == name)
                        }
                        DuplicatePolicy::KeepLast => {
                            !header[index + 1..].iter().any(|later| later == name)
                        }
                    };
                    if chosen {
                        selected.push((name.clone(), index));
                    }
                }
                selected.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
                output_records.push(selected.iter().map(|(name, _)| name.clone()).collect());

                let mut rows: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| {
                        selected
                            .iter()
                            .map(|(_, index)| row[*index].clone())
                            .collect()
                    })
                    .collect();
                rows.sort_by(|a, b| {
                    a.iter()
                        .map(String::as_bytes)
                        .cmp(b.iter().map(String::as_bytes))
                });
                output_records.extend(rows);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| {
                    a.iter()
                        .map(String::as_bytes)
                        .cmp(b.iter().map(String::as_bytes))
                });
                output_records.extend(rows);
            }
        }
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let line_ending = match config.mode {
        WriteMode::Canonical => "\n",
        WriteMode::Preserve => match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        },
    };
    let final_terminator = match config.mode {
        WriteMode::Canonical => true,
        WriteMode::Preserve => config.final_terminator,
    };

    let mut output = String::new();
    for (record_index, record) in output_records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(line_ending);
        }

        let sole = record.len() == 1;
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            output.push_str(&write_field(
                field,
                config.delimiter,
                config.quote,
                record_index == 0 && column_index == 0,
                sole,
            ));
        }
    }
    if final_terminator {
        output.push_str(line_ending);
    }
    Ok(output)
}
