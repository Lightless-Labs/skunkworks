#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn is_ascii_alpha(b: u8) -> bool {
    b.is_ascii_alphabetic()
}

fn is_scheme_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')
}

fn validate_global_bytes(input: &str) -> Result<(), usize> {
    input
        .as_bytes()
        .iter()
        .position(|&b| !(0x21..=0x7e).contains(&b))
        .map_or(Ok(()), |at| Err(at))
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let Some(hi) = hex_value(bytes[i + 1]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            let Some(lo) = hex_value(bytes[i + 2]) else {
                return Err(CodecError::BadEscape { at: i });
            };
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn parse_port(input: &str, start: usize) -> Result<u16, ParseError> {
    if input.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    if let Some(i) = input.bytes().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: start + i });
    }
    if input.len() >= 2 && input.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value: u32 = 0;
    for b in input.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: start })?;
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn validate_component_at(
    component: Component,
    value: &str,
    offset: usize,
) -> Result<(), ParseError> {
    match decode_component(component, value) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: offset + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: offset + at }),
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Err(at) = validate_global_bytes(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !is_ascii_alpha(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    if let Some(i) = bytes[..colon]
        .iter()
        .position(|&b| !is_scheme_char(b))
    {
        return Err(ParseError::InvalidScheme { at: i });
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map_or(input.len(), |i| authority_start + i);
    let authority = &input[authority_start..authority_end];

    let ats: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + ats[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };
    let hostport = &authority[hostport_start..];
    let hostport_abs = authority_start + hostport_start;

    let (host, port, ipv6) = if hostport.as_bytes().first() == Some(&b'[') {
        let open_abs = hostport_abs;
        let close = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: open_abs }),
        };
        let interior = &hostport[1..close];
        if interior.contains('%') || interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_abs });
        }
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(
                &suffix[1..],
                hostport_abs + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_abs + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport.bytes().position(|b| b == b':');
        let (host_text, port) = if let Some(i) = split {
            (
                &hostport[..i],
                Some(parse_port(&hostport[i + 1..], hostport_abs + i + 1)?),
            )
        } else {
            (hostport, None)
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_abs });
        }
        if let Some(i) = host_text.bytes().position(|b| {
            !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_'))
        }) {
            return Err(ParseError::InvalidHost {
                at: hostport_abs + i,
            });
        }
        (host_text.to_ascii_lowercase(), port, false)
    };

    let after_authority = &input[authority_end..];
    let fragment_rel = after_authority.find('#');
    let before_fragment = fragment_rel.map_or(after_authority, |i| &after_authority[..i]);
    let fragment = fragment_rel.map(|i| after_authority[i + 1..].to_owned());
    let fragment_start = fragment_rel.map(|i| authority_end + i + 1);

    let query_rel = before_fragment.find('?');
    let path_text = query_rel.map_or(before_fragment, |i| &before_fragment[..i]);
    let query = query_rel.map(|i| before_fragment[i + 1..].to_owned());
    let query_start = query_rel.map(|i| authority_end + i + 1);

    let path = if path_text.is_empty() {
        "/".to_owned()
    } else {
        path_text.to_owned()
    };

    if let Some(ref value) = userinfo {
        validate_component_at(
            Component::UserInfo,
            value,
            authority_start,
        )?;
    }
    if !path_text.is_empty() {
        validate_component_at(
            Component::Path,
            path_text,
            authority_end,
        )?;
    }
    if let (Some(ref value), Some(start)) = (&query, query_start) {
        validate_component_at(Component::Query, value, start)?;
    }
    if let (Some(ref value), Some(start)) = (&fragment, fragment_start) {
        validate_component_at(Component::Fragment, value, start)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<String> = Vec::new();
    let mut terminal_navigation = false;

    for (index, raw) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 + segments[..index].iter().map(|s| s.len() + 1).sum::<usize>() },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 + segments[..index].iter().map(|s| s.len() + 1).sum::<usize>() },
        })?;
        terminal_navigation = index + 1 == segments.len() && (decoded == "." || decoded == "..");
        match decoded.as_str() {
            "." => {}
            ".." => {
                stack.pop();
            }
            _ => stack.push((*raw).to_owned()),
        }
    }
    if terminal_navigation {
        stack.push(String::new());
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn map_path_error(error: PathError) -> ResolveError {
    ResolveError::Path(error)
}

fn url_prefix(url: &Url) -> String {
    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        out.push_str(userinfo);
        out.push('@');
    }
    if url.ipv6 {
        out.push('[');
        out.push_str(&url.host);
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        out.push(':');
        out.push_str(&port.to_string());
    }
    out
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let mut base_url = parse_url(base).map_err(ResolveError::Base)?;
    base_url.path = normalise_path(&base_url.path).map_err(map_path_error)?;

    if let Err(at) = validate_global_bytes(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let bytes = reference.as_bytes();
    let first_delim = bytes
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());
    if bytes[..first_delim].contains(&b':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(map_path_error)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(map_path_error)?;
        return Ok(url);
    }

    let fragment_at = reference.find('#');
    let before_fragment = fragment_at.map_or(reference, |i| &reference[..i]);
    let fragment = fragment_at.map(|i| reference[i + 1..].to_owned());
    let query_at = before_fragment.find('?');
    let reference_path = query_at.map_or(before_fragment, |i| &before_fragment[..i]);
    let reference_query = query_at.map(|i| before_fragment[i + 1..].to_owned());

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if query_at.is_none() {
                base_url.query.clone()
            } else {
                reference_query
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').map_or(0, |i| i + 1);
            format!("{}{}", &base_url.path[..slash], reference_path)
        };
        (merged, reference_query)
    };

    let mut constructed = url_prefix(&base_url);
    constructed.push_str(&selected_path);
    if let Some(query) = &selected_query {
        constructed.push('?');
        constructed.push_str(query);
    }
    if let Some(fragment) = &fragment {
        constructed.push('#');
        constructed.push_str(fragment);
    }

    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(map_path_error)?;
    Ok(url)
}
