#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record = 0usize;
    let mut field_started = false;
    let mut quoted = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !quoted && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if quoted {
            if b == b'"' {
                if bytes.get(i + 1) == Some(&b'"') {
                    field.push('"');
                    i += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().expect("valid UTF-8 input");
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            match b {
                b',' => {
                    fields.push(std::mem::take(&mut field));
                    field_started = false;
                    after_quote = false;
                    i += 1;
                }
                b'\n' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    field_started = false;
                    after_quote = false;
                    i += 1;
                }
                b'\r' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    field_started = false;
                    after_quote = false;
                    i += 2;
                }
                _ => return Err(CsvError::TrailingByte { record, byte: i }),
            }
            continue;
        }

        match b {
            b'"' if !field_started => {
                quoted = true;
                field_started = true;
                quote_start = i;
                i += 1;
            }
            b'"' => return Err(CsvError::UnexpectedQuote { record, byte: i }),
            b',' => {
                fields.push(std::mem::take(&mut field));
                field_started = false;
                i += 1;
            }
            b'\n' => {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record += 1;
                field_started = false;
                i += 1;
            }
            b'\r' => {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record += 1;
                field_started = false;
                i += 2;
            }
            _ => {
                let ch = input[i..].chars().next().expect("valid UTF-8 input");
                field.push(ch);
                field_started = true;
                i += ch.len_utf8();
            }
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if field_started || after_quote || !fields.is_empty() {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}
