#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0;
    let mut now = 0u64;
    let mut finish: Option<u64> = None;
    let mut active: Option<&Customer> = None;
    let mut previous_finish: Option<u64> = None;
    let mut dispatched_once = false;
    let mut completed = 0usize;

    while completed < customers.len() {
        let next_arrival_time = arrivals.get(next_arrival).map(|c| c.arrival);
        let next_time = match (finish, next_arrival_time) {
            (Some(f), Some(a)) => f.min(a),
            (Some(f), None) => f,
            (None, Some(a)) => a,
            (None, None) => break,
        };
        now = next_time;

        if finish == Some(now) {
            let customer = active.take().expect("scheduled finish has active customer");
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Finish,
            });
            finish = None;
            previous_finish = Some(now);
            completed += 1;
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        if finish.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = !dispatched_once
                    || previous_finish.map_or(false, |previous| now > previous);
                let base_finish = now.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    },
                )?;
                let finish_time = base_finish
                    .checked_add(if wake { wake_cost } else { 0 })
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    })?;

                events.push(Event {
                    time: now,
                    customer: customer.id,
                    kind: Kind::Start {
                        finish: finish_time,
                        wake,
                    },
                });
                dispatched_once = true;
                active = Some(customer);

                if finish_time == now {
                    events.push(Event {
                        time: now,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    active = None;
                    finish = None;
                    previous_finish = Some(now);
                    completed += 1;

                    while next_arrival < arrivals.len()
                        && arrivals[next_arrival].arrival == now
                    {
                        let arrived = arrivals[next_arrival];
                        events.push(Event {
                            time: now,
                            customer: arrived.id,
                            kind: Kind::Arrive,
                        });
                        waiting.push_back(arrived);
                        next_arrival += 1;
                    }

                    while let Some(next_customer) = waiting.pop_front() {
                        let base = now.checked_add(next_customer.service).ok_or(
                            Error::TimeOverflow {
                                customer: next_customer.id,
                                time: now,
                            },
                        )?;
                        let next_finish = base.checked_add(0).ok_or(Error::TimeOverflow {
                            customer: next_customer.id,
                            time: now,
                        })?;
                        events.push(Event {
                            time: now,
                            customer: next_customer.id,
                            kind: Kind::Start {
                                finish: next_finish,
                                wake: false,
                            },
                        });
                        active = Some(next_customer);
                        if next_finish == now {
                            events.push(Event {
                                time: now,
                                customer: next_customer.id,
                                kind: Kind::Finish,
                            });
                            active = None;
                            completed += 1;
                            previous_finish = Some(now);
                        } else {
                            finish = Some(next_finish);
                            break;
                        }
                    }
                } else {
                    finish = Some(finish_time);
                }
            }
        }
    }

    Ok(events)
}
