#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits. Zero is represented as [0].
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let magnitude = &bytes[start..];

        // Once the invalid-byte scan succeeds, the first malformed separator
        // takes precedence over leading-zero and digit-count errors.
        for (offset, &byte) in magnitude.iter().enumerate() {
            if byte == b'_' {
                let valid_before = offset > 0 && magnitude[offset - 1].is_ascii_digit();
                let valid_after =
                    offset + 1 < magnitude.len() && magnitude[offset + 1].is_ascii_digit();
                if !valid_before || !valid_after {
                    return Err(Error::Separator {
                        index: start + offset,
                    });
                }
            }
        }

        let digit_count = magnitude.iter().filter(|&&byte| byte != b'_').count();
        let first_digit = magnitude.iter().position(|&byte| byte != b'_').unwrap();

        if digit_count > 1 && magnitude[first_digit] == b'0' {
            return Err(Error::LeadingZero {
                index: start + first_digit,
            });
        }

        if digit_count > max_digits {
            let excess_digit_number = max_digits;
            let mut seen = 0;
            let mut excess_offset = first_digit;
            for (offset, &byte) in magnitude.iter().enumerate() {
                if byte != b'_' {
                    if seen == excess_digit_number {
                        excess_offset = offset;
                        break;
                    }
                    seen += 1;
                }
            }
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: start + excess_offset,
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in magnitude {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }
        digits.reverse();

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let natural_len = self.digits.len().max(1);
        let width = natural_len.max(min_digits);
        let padding = width - natural_len;

        let mut out = String::with_capacity(width + width / group.max(1));
        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                out.push('_');
            }

            if position < padding {
                out.push('0');
            } else {
                let digit_index_from_left = position - padding;
                let digit = self.digits[natural_len - 1 - digit_index_from_left];
                out.push(char::from(b'0' + digit));
            }
        }
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        use std::cmp::Ordering;

        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => {
                for (&left, &right) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match left.cmp(&right) {
                        Ordering::Equal => {}
                        ordering => return ordering,
                    }
                }
                Ordering::Equal
            }
            ordering => ordering,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let max_len = self.digits.len().max(rhs.digits.len());
        let mut digits = Vec::with_capacity(max_len + 1);
        let mut carry = 0u8;

        for index in 0..max_len {
            let left = self.digits.get(index).copied().unwrap_or(0);
            let right = rhs.digits.get(index).copied().unwrap_or(0);
            let sum = left + right + carry;
            digits.push(sum % 10);
            carry = sum / 10;
        }

        if carry != 0 {
            digits.push(carry);
        }

        Self { digits }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        use std::cmp::Ordering;

        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut digits = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i8;

        for index in 0..self.digits.len() {
            let left = self.digits[index] as i8 - borrow;
            let right = rhs.digits.get(index).copied().unwrap_or(0) as i8;

            if left < right {
                digits.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                digits.push((left - right) as u8);
                borrow = 0;
            }
        }

        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut first_invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                first_invalid = Some(index);
                break;
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut first_separator_error = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                first_separator_error = Some(index);
                break;
            }
        }
        if let Some(index) = first_separator_error {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_excess = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte.is_ascii_digit() {
                digit_count += 1;
                if first_digit.is_none() {
                    first_digit = Some((index, byte));
                }
                if digit_count == max_digits.saturating_add(1) {
                    first_excess = Some(index);
                }
            }
        }

        if digit_count > 1 && first_digit.is_some_and(|(_, digit)| digit == b'0') {
            return Err(Error::LeadingZero {
                index: first_digit.unwrap().0,
            });
        }

        if digit_count > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: first_excess.unwrap_or_else(|| {
                    bytes
                        .iter()
                        .enumerate()
                        .skip(start)
                        .filter(|(_, byte)| byte.is_ascii_digit())
                        .nth(max_digits)
                        .map(|(index, _)| index)
                        .unwrap_or(bytes.len())
                }),
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes.iter().skip(start) {
            if byte.is_ascii_digit() {
                digits.push(byte - b'0');
            }
        }

        Ok(Self::from_digits(digits))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let canonical = self.decimal_digits();
        let width = min_digits.max(canonical.len()).max(1);
        let padding = width - canonical.len();

        let mut output = String::with_capacity(width + width / group.max(1));
        for index in 0..width {
            if group > 0 && index > 0 && (width - index) % group == 0 {
                output.push('_');
            }
            if index < padding {
                output.push('0');
            } else {
                output.push((b'0' + canonical[index - padding]) as char);
            }
        }
        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => self.digits.cmp(&rhs.digits),
            other => other,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = Vec::with_capacity(self.digits.len().max(rhs.digits.len()) + 1);
        let mut carry = 0u8;
        let mut left = self.digits.iter().rev();
        let mut right = rhs.digits.iter().rev();

        loop {
            let a = left.next().copied();
            let b = right.next().copied();
            if a.is_none() && b.is_none() && carry == 0 {
                break;
            }

            let sum = a.unwrap_or(0) + b.unwrap_or(0) + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }

        result.reverse();
        Self::from_digits(result)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;
        let mut right = rhs.digits.iter().rev();

        for &left_digit in self.digits.iter().rev() {
            let mut difference = left_digit as i16 - borrow;
            let right_digit = right.next().copied().unwrap_or(0) as i16;
            if difference < right_digit {
                difference += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push((difference - right_digit) as u8);
        }

        result.reverse();
        Ok(Self::from_digits(result))
    }

    fn from_digits(mut digits: Vec<u8>) -> Self {
        let first_nonzero = digits.iter().position(|&digit| digit != 0);
        match first_nonzero {
            Some(index) => {
                digits.drain(..index);
            }
            None => {
                digits.clear();
                digits.push(0);
            }
        }
        Self { digits }
    }

    fn decimal_digits(&self) -> Vec<u8> {
        self.digits.clone()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits. The vector is never empty and has no
    // high-order zeroes except for the canonical representation of zero.
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over every separator error.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        let magnitude = &bytes[start..];

        // Once all non-underscore bytes are known to be digits, each
        // underscore must have a digit immediately on both sides.
        for (offset, &byte) in magnitude.iter().enumerate() {
            if byte == b'_'
                && (offset == 0
                    || offset + 1 == magnitude.len()
                    || magnitude[offset - 1] == b'_'
                    || magnitude[offset + 1] == b'_')
            {
                return Err(Error::Separator {
                    index: start + offset,
                });
            }
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut digits = Vec::new();

        for (offset, &byte) in magnitude.iter().enumerate() {
            if byte != b'_' {
                if first_digit.is_none() {
                    first_digit = Some((byte, start + offset));
                }
                digit_count += 1;
                digits.push(byte - b'0');
            }
        }

        if digit_count > 1 {
            let (first, index) = first_digit.expect("nonempty magnitude has a digit");
            if first == b'0' {
                return Err(Error::LeadingZero { index });
            }
        }

        if digit_count > max_digits {
            let excess_digit = max_digits + 1;
            let mut seen = 0usize;
            let mut excess_index = None;
            for (offset, &byte) in magnitude.iter().enumerate() {
                if byte != b'_' {
                    seen += 1;
                    if seen == excess_digit {
                        excess_index = Some(start + offset);
                        break;
                    }
                }
            }

            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: excess_index.expect("excess digit exists"),
            });
        }

        digits.reverse();
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        Ok(Self { digits })
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let raw_len = self.digits.len().max(1);
        let width = raw_len.max(min_digits);
        let padding = width - raw_len;
        let mut output = String::new();

        for position in 0..width {
            if group > 0 && position > 0 && (width - position) % group == 0 {
                output.push('_');
            }

            if position < padding {
                output.push('0');
            } else {
                let digit_index_from_left = position - padding;
                let digit = self.digits[raw_len - 1 - digit_index_from_left];
                output.push((b'0' + digit) as char);
            }
        }

        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> std::cmp::Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            std::cmp::Ordering::Equal => {
                for (left, right) in self.digits.iter().rev().zip(rhs.digits.iter().rev()) {
                    match left.cmp(right) {
                        std::cmp::Ordering::Equal => {}
                        order => return order,
                    }
                }
                std::cmp::Ordering::Equal
            }
            order => order,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let length = self.digits.len().max(rhs.digits.len());
        let mut result = Vec::with_capacity(length + 1);
        let mut carry = 0u8;

        for index in 0..length {
            let left = self.digits.get(index).copied().unwrap_or(0);
            let right = rhs.digits.get(index).copied().unwrap_or(0);
            let sum = left + right + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }

        if carry > 0 {
            result.push(carry);
        }

        Self { digits: result }
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == std::cmp::Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;

        for index in 0..self.digits.len() {
            let left = self.digits[index] as i16 - borrow;
            let right = rhs.digits.get(index).copied().unwrap_or(0) as i16;

            if left < right {
                result.push((left + 10 - right) as u8);
                borrow = 1;
            } else {
                result.push((left - right) as u8);
                borrow = 0;
            }
        }

        while result.len() > 1 && result.last() == Some(&0) {
            result.pop();
        }

        Ok(Self { digits: result })
    }
