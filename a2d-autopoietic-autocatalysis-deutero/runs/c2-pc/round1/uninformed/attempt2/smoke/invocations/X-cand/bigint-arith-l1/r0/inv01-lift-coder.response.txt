```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    digits: Vec<u8>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    Underflow,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();

        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let start = if bytes[0] == b'+' { 1 } else { 0 };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut first_invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                first_invalid = Some(index);
                break;
            }
        }
        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut first_separator_error = None;
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                first_separator_error = Some(index);
                break;
            }
        }
        if let Some(index) = first_separator_error {
            return Err(Error::Separator { index });
        }

        let mut digit_count = 0usize;
        let mut first_digit = None;
        let mut first_excess = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte.is_ascii_digit() {
                digit_count += 1;
                if first_digit.is_none() {
                    first_digit = Some((index, byte));
                }
                if digit_count == max_digits.saturating_add(1) {
                    first_excess = Some(index);
                }
            }
        }

        if digit_count > 1 && first_digit.is_some_and(|(_, digit)| digit == b'0') {
            return Err(Error::LeadingZero {
                index: first_digit.unwrap().0,
            });
        }

        if digit_count > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: first_excess.unwrap_or_else(|| {
                    bytes
                        .iter()
                        .enumerate()
                        .skip(start)
                        .filter(|(_, byte)| byte.is_ascii_digit())
                        .nth(max_digits)
                        .map(|(index, _)| index)
                        .unwrap_or(bytes.len())
                }),
            });
        }

        let mut digits = Vec::with_capacity(digit_count);
        for &byte in bytes.iter().skip(start) {
            if byte.is_ascii_digit() {
                digits.push(byte - b'0');
            }
        }

        Ok(Self::from_digits(digits))
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let canonical = self.decimal_digits();
        let width = min_digits.max(canonical.len()).max(1);
        let padding = width - canonical.len();

        let mut output = String::with_capacity(width + width / group.max(1));
        for index in 0..width {
            if group > 0 && index > 0 && (width - index) % group == 0 {
                output.push('_');
            }
            if index < padding {
                output.push('0');
            } else {
                output.push((b'0' + canonical[index - padding]) as char);
            }
        }
        output
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match self.digits.len().cmp(&rhs.digits.len()) {
            Ordering::Equal => self.digits.cmp(&rhs.digits),
            other => other,
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = Vec::with_capacity(self.digits.len().max(rhs.digits.len()) + 1);
        let mut carry = 0u8;
        let mut left = self.digits.iter().rev();
        let mut right = rhs.digits.iter().rev();

        loop {
            let a = left.next().copied();
            let b = right.next().copied();
            if a.is_none() && b.is_none() && carry == 0 {
                break;
            }

            let sum = a.unwrap_or(0) + b.unwrap_or(0) + carry;
            result.push(sum % 10);
            carry = sum / 10;
        }

        result.reverse();
        Self::from_digits(result)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, Error> {
        if self.cmp_value(rhs) == Ordering::Less {
            return Err(Error::Underflow);
        }

        let mut result = Vec::with_capacity(self.digits.len());
        let mut borrow = 0i16;
        let mut right = rhs.digits.iter().rev();

        for &left_digit in self.digits.iter().rev() {
            let mut difference = left_digit as i16 - borrow;
            let right_digit = right.next().copied().unwrap_or(0) as i16;
            if difference < right_digit {
                difference += 10;
                borrow = 1;
            } else {
                borrow = 0;
            }
            result.push((difference - right_digit) as u8);
        }

        result.reverse();
        Ok(Self::from_digits(result))
    }

    fn from_digits(mut digits: Vec<u8>) -> Self {
        let first_nonzero = digits.iter().position(|&digit| digit != 0);
        match first_nonzero {
            Some(index) => {
                digits.drain(..index);
            }
            None => {
                digits.clear();
                digits.push(0);
            }
        }
        Self { digits }
    }

    fn decimal_digits(&self) -> Vec<u8> {
        self.digits.clone()
    }
}
```
