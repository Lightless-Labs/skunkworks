#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

const ALPHABET: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

pub fn encode(input: &[u8]) -> String {
    let mut output = String::new();
    let mut accumulator: u32 = 0;
    let mut bit_count = 0;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bit_count += 8;

        while bit_count >= 5 {
            bit_count -= 5;
            let value = ((accumulator >> bit_count) & 0x1f) as usize;
            output.push(ALPHABET[value] as char);
        }
    }

    if bit_count != 0 {
        let value = ((accumulator << (5 - bit_count)) & 0x1f) as usize;
        output.push(ALPHABET[value] as char);
    }

    while output.len() % 8 != 0 {
        output.push('=');
    }

    output
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    // Lexical validation.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && !ALPHABET.contains(&byte) {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    // Padding placement.
    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);
    if let Some(offset) = bytes[p..].iter().position(|&byte| byte != b'=') {
        return Err(DecodeError::Padding { offset: p + offset });
    }

    // Data length.
    let residue = p % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    // Padding shape.
    let required_padding = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => unreachable!(),
    };
    if n - p != required_padding || n % 8 != 0 {
        return Err(DecodeError::Padding { offset: p });
    }

    // Unused low bits in the last data symbol must be zero.
    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };
    if unused_bits != 0 {
        let last_value = ALPHABET
            .iter()
            .position(|&symbol| symbol == bytes[p - 1])
            .unwrap();
        if last_value & ((1usize << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset: p - 1 });
        }
    }

    let mut output = Vec::with_capacity((p * 5) / 8);
    let mut accumulator: u32 = 0;
    let mut bit_count = 0;

    for &symbol in &bytes[..p] {
        let value = ALPHABET
            .iter()
            .position(|&candidate| candidate == symbol)
            .unwrap() as u32;
        accumulator = (accumulator << 5) | value;
        bit_count += 5;

        while bit_count >= 8 {
            bit_count -= 8;
            output.push(((accumulator >> bit_count) & 0xff) as u8);
        }
    }

    Ok(output)
}

    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

const ALPHABET: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

pub fn encode(input: &[u8]) -> String {
    let mut output = String::new();
    let mut accumulator = 0u32;
    let mut bit_count = 0usize;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bit_count += 8;

        while bit_count >= 5 {
            bit_count -= 5;
            let value = ((accumulator >> bit_count) & 0x1f) as usize;
            output.push(ALPHABET[value] as char);
        }

        accumulator &= (1u32 << bit_count).wrapping_sub(1);
    }

    if bit_count > 0 {
        let value = ((accumulator << (5 - bit_count)) & 0x1f) as usize;
        output.push(ALPHABET[value] as char);
    }

    while output.len() % 8 != 0 {
        output.push('=');
    }

    output
}

            return Err(DecodeError::InvalidByte { offset, byte });
        }

    if let Some(offset) = bytes[p..].iter().position(|&byte| byte != b'=') {
        return Err(DecodeError::Padding { offset: p + offset });
    }

    // Data length.
    let d = p;
    let residue = d % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    // Padding shape.
    let required_padding = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => unreachable!(),
    };
    if n - p != required_padding || n % 8 != 0 {
        return Err(DecodeError::Padding { offset: p });
    }

    // Trailing unused bits.
    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };

            .position(|&symbol| symbol == bytes[d - 1])
            .unwrap();

            return Err(DecodeError::TrailingBits { offset: d - 1 });
        }
    }

    // Decode complete five-bit values into bytes.
    let mut output = Vec::with_capacity(d * 5 / 8);

    let mut bit_count = 0usize;

            .unwrap() as u32;

        bit_count += 5;

            bit_count -= 8;
            output.push(((accumulator >> bit_count) & 0xff) as u8);
        }

        accumulator &= (1u32 << bit_count).wrapping_sub(1);
    }

    Ok(output)
}

fn symbol_value(byte: u8) -> Option<u8> {
    match byte {
        b'A'..=b'Z' => Some(byte - b'A'),
        b'2'..=b'7' => Some(byte - b'2' + 26),
        _ => None,
    }
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    // Lexical validation.
    for (offset, &byte) in bytes.iter().enumerate() {
        if symbol_value(byte).is_none() && byte != b'=' {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    // Padding placement.
    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);
    if let Some(offset) = (p..n).find(|&offset| bytes[offset] != b'=') {
        return Err(DecodeError::Padding { offset });
    }

    // Data length.
    let residue = p % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    // Padding shape.
    let required_padding = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => unreachable!(),
    };
    if n - p != required_padding || n % 8 != 0 {
        return Err(DecodeError::Padding { offset: p });
    }

    // Trailing unused bits.
    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };
    if unused_bits > 0 {
        let last_value = symbol_value(bytes[p - 1]).unwrap();
        if last_value & ((1 << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset: p - 1 });
        }
    }

    let mut output = Vec::with_capacity(p * 5 / 8);
    let mut accumulator: u32 = 0;
    let mut bit_count = 0usize;

    for &byte in &bytes[..p] {
        accumulator = (accumulator << 5) | u32::from(symbol_value(byte).unwrap());
        bit_count += 5;

        while bit_count >= 8 {
            bit_count -= 8;
            output.push(((accumulator >> bit_count) & 0xff) as u8);
        }
    }

    Ok(output)
}
