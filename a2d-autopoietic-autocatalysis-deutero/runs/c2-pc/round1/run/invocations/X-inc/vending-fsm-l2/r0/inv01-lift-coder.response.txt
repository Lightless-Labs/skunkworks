#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = Vec::<String>::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::denomination(index),
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let index = match coin {
            1 => 0,
            3 => 1,
            4 => 2,
            10 => 3,
            _ => return Err(Error::BadCoin { coin }),
        };

        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(coin);
        self.credit += coin;
        let _ = index;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = self.selected.as_ref().ok_or(Error::NoSelection)?;
        let product = self
            .products
            .iter()
            .find(|product| &product.id == selected_id)
            .expect("selected products are always in the catalog");

        let id = product.id.clone();
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = Self::choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.tubes;
        for &coin in &change {
            let index = Self::coin_index(coin).expect("change uses valid denominations");
            proposed_tubes[index] -= 1;
        }
        for &coin in &self.escrow {
            let index = Self::coin_index(coin).expect("escrow uses valid denominations");
            proposed_tubes[index] += 1;
        }

        for (index, count) in proposed_tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::denomination(index),
                });
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }

    fn denomination(index: usize) -> u16 {
        [1, 3, 4, 10][index]
    }

    fn coin_index(coin: u16) -> Option<usize> {
        match coin {
            1 => Some(0),
            3 => Some(1),
            4 => Some(2),
            10 => Some(3),
            _ => None,
        }
    }

    fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
        if amount == 0 {
            return Some(Vec::new());
        }

        let denominations = [10u16, 4, 3, 1];
        let mut remaining = amount;
        let mut greedy = Vec::new();

        for &coin in &denominations {
            let index = Self::coin_index(coin).expect("known denomination");
            let take = tubes[index].min(remaining / coin);
            for _ in 0..take {
                greedy.push(coin);
            }
            remaining -= take * coin;
        }

        if remaining == 0 {
            return Some(greedy);
        }

        let mut best: Option<Vec<u16>> = None;
        let mut candidate = Vec::new();
        Self::search_change(0, amount, tubes, &denominations, &mut candidate, &mut best);
        best
    }

    fn search_change(
        position: usize,
        remaining: u16,
        tubes: &[u16; 4],
        denominations: &[u16; 4],
        candidate: &mut Vec<u16>,
        best: &mut Option<Vec<u16>>,
    ) {
        if remaining == 0 {
            let replace = match best {
                None => true,
                Some(current) => {
                    candidate.len() < current.len()
                        || (candidate.len() == current.len()
                            && candidate.as_slice() > current.as_slice())
                }
            };
            if replace {
                *best = Some(candidate.clone());
            }
            return;
        }
        if position == denominations.len() {
            return;
        }
        if let Some(current) = best.as_ref() {
            if candidate.len() >= current.len() {
                return;
            }
        }

        let coin = denominations[position];
        let index = Self::coin_index(coin).expect("known denomination");
        let max_take = tubes[index].min(remaining / coin);

        for take in (0..=max_take).rev() {
            for _ in 0..take {
                candidate.push(coin);
            }
            Self::search_change(
                position + 1,
                remaining - take * coin,
                tubes,
                denominations,
                candidate,
                best,
            );
            for _ in 0..take {
                candidate.pop();
            }
        }
    }
}
