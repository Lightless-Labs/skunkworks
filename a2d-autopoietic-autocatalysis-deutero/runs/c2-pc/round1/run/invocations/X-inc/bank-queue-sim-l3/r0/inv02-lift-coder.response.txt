#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    index: usize,
    remaining: u64,
    started: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Running {
    index: usize,
    work_at: u64,
    finish: u64,
}

fn waiter_key(a: &Waiting, b: &Waiting, customers: &[Customer]) -> std::cmp::Ordering {
    let ca = &customers[a.index];
    let cb = &customers[b.index];

    ca.priority
        .cmp(&cb.priority)
        .then_with(|| b.started.cmp(&a.started))
        .then_with(|| {
            if a.started {
                a.pause_time.cmp(&b.pause_time)
                    .then_with(|| ca.id.cmp(&cb.id))
            } else {
                ca.arrival.cmp(&cb.arrival)
                    .then_with(|| cb.id.cmp(&ca.id))
            }
        })
}

fn best_waiter(waiting: &[Waiting], customers: &[Customer]) -> Option<usize> {
    waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiter_key(a, b, customers))
        .map(|(i, _)| i)
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut arrival_pos = 0usize;
    let mut finished_count = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut time = customers[arrivals[0]].arrival;

    while finished_count < customers.len() {
        if let Some(active) = running.as_ref() {
            if active.finish == time {
                let index = active.index;
                events.push(Event {
                    time,
                    customer: customers[index].id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
            }
        }

        while arrival_pos < arrivals.len()
            && customers[arrivals[arrival_pos]].arrival == time
        {
            let index = arrivals[arrival_pos];
            events.push(Event {
                time,
                customer: customers[index].id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                index,
                remaining: customers[index].service,
                started: false,
                pause_time: 0,
            });
            arrival_pos += 1;
        }

        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                if let Some(best) = best_waiter(&waiting, customers) {
                    let active_priority = customers[active.index].priority;
                    let best_priority = customers[waiting[best].index].priority;
                    if best_priority < active_priority {
                        let active = running.take().unwrap();
                        let remaining = active.finish - time;
                        let by = customers[waiting[best].index].id;
                        events.push(Event {
                            time,
                            customer: customers[active.index].id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        waiting.push(Waiting {
                            index: active.index,
                            remaining,
                            started: true,
                            pause_time: time,
                        });
                    }
                }
            }
        }

        loop {
            if running.is_some() {
                break;
            }
            let Some(wait_index) = best_waiter(&waiting, customers) else {
                break;
            };
            let queued = waiting.remove(wait_index);
            let customer = &customers[queued.index];
            let cost = if last_dispatched.is_some_and(|id| id != customer.id) {
                switch_cost
            } else {
                0
            };
            let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: customer.id,
                time,
            })?;
            let finish = work_at.checked_add(queued.remaining).ok_or(Error::TimeOverflow {
                customer: customer.id,
                time,
            })?;

            let kind = if queued.started {
                Kind::Resume {
                    remaining: queued.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: queued.remaining,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time,
                customer: customer.id,
                kind,
            });
            last_dispatched = Some(customer.id);
            running = Some(Running {
                index: queued.index,
                work_at,
                finish,
            });

            if finish == time {
                events.push(Event {
                    time,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                running = None;
                finished_count += 1;
                continue;
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if arrival_pos < arrivals.len() {
            next_time = Some(customers[arrivals[arrival_pos]].arrival);
        }
        if let Some(active) = running.as_ref() {
            for candidate in [Some(active.finish), (active.work_at > time).then_some(active.work_at)]
                .into_iter()
                .flatten()
            {
                if candidate > time && next_time.map_or(true, |n| candidate < n) {
                    next_time = Some(candidate);
                }
            }
        }

        if let Some(next) = next_time {
            time = next;
        } else if finished_count < customers.len() {
            // All unfinished customers must be waiting or running; an idle
            // server with no future event is therefore impossible.
            break;
        }
    }

    Ok(events)
}
