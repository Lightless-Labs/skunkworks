fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_key(*c));
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    constraint_text_list(cs.iter().copied().collect())
}

fn constraint_text_list(cs: Vec<Constraint>) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<&Evidence> {
    let mut out: Vec<&Evidence> = evidence.iter().collect();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn append_evidence(out: &mut String, evidence: &[Evidence]) {
    for e in evidence_sorted(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
}

fn target_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn target_text(platform: Platform) -> &'static str {
    match platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn feature_list(features: &[String]) -> String {
    let set: BTreeSet<&str> = features.iter().map(String::as_str).collect();
    set.into_iter().collect::<Vec<_>>().join(",")
}

fn root_token(root: &Requirement, index: usize) -> String {
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        root.package,
        constraint_text(root.constraint),
        feature_list(&root.features)
    )
}

fn dep_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        feature_list(&dep.features),
        dep.gate.as_deref().unwrap_or("*"),
        target_text(dep.platform)
    )
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    features: BTreeMap<String, BTreeSet<String>>,
    evidence: BTreeMap<String, Vec<Evidence>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn path_cmp(a: &Vec<String>, b: &Vec<String>) -> std::cmp::Ordering {
    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
}

fn compute_closure(
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
    target: Target,
) -> Closure {
    let mut origins: BTreeMap<String, Vec<Origin>> = BTreeMap::new();
    let mut requests: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
    for (i, root) in roots.iter().enumerate() {
        origins.entry(root.package.clone()).or_default().push(Origin {
            constraint: root.constraint,
            path: vec![root_token(root, i)],
        });
        requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
    }

    loop {
        let mut best_paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
        let mut queue: BTreeSet<String> = origins.keys().cloned().collect();
        for name in &queue {
            if let Some(os) = origins.get(name) {
                if let Some(path) = os.iter().map(|o| o.path.clone()).min_by(path_cmp) {
                    best_paths.insert(name.clone(), path);
                }
            }
        }

        let mut changed_path = true;
        while changed_path {
            changed_path = false;
            let names: Vec<String> = selected.keys().cloned().collect();
            for source in names {
                let Some(source_path) = best_paths.get(&source).cloned() else {
                    continue;
                };
                let Some(&version) = selected.get(&source) else {
                    continue;
                };
                let Some(pkg) = catalog.get(&source) else {
                    continue;
                };
                let Some(release) = pkg.releases.iter().find(|r| r.version == version) else {
                    continue;
                };
                let mut fs = requests.get(&source).cloned().unwrap_or_default();
                let mut local_changed = true;
                while local_changed {
                    local_changed = false;
                    for f in fs.clone() {
                        if let Some(decl) = release.features.iter().find(|x| x.name == f) {
                            for enabled in &decl.enables {
                                if fs.insert(enabled.clone()) {
                                    local_changed = true;
                                }
                            }
                        }
                    }
                }
                for (i, dep) in release.deps.iter().enumerate() {
                    if !target_matches(dep.platform, target)
                        || dep.gate.as_ref().is_some_and(|g| !fs.contains(g))
                    {
                        continue;
                    }
                    let mut path = source_path.clone();
                    path.push(dep_token(&source, version, i, dep));
                    match best_paths.get(&dep.package) {
                        Some(old) if path_cmp(old, &path) != std::cmp::Ordering::Greater => {}
                        _ => {
                            best_paths.insert(dep.package.clone(), path);
                            queue.insert(dep.package.clone());
                            changed_path = true;
                        }
                    }
                }
            }
        }

        let mut grew = false;
        for (name, path) in &best_paths {
            let Some(version) = selected.get(name) else {
                continue;
            };
            let Some(pkg) = catalog.get(name) else {
                continue;
            };
            let Some(release) = pkg.releases.iter().find(|r| r.version == *version) else {
                continue;
            };
            let fs = requests.entry(name.clone()).or_default();
            let mut local_changed = true;
            while local_changed {
                local_changed = false;
                let current: Vec<String> = fs.iter().cloned().collect();
                for f in current {
                    if let Some(decl) = release.features.iter().find(|x| x.name == f) {
                        for enabled in &decl.enables {
                            if fs.insert(enabled.clone()) {
                                local_changed = true;
                                grew = true;
                            }
                        }
                    }
                }
            }
            for (i, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !fs.contains(g))
                {
                    continue;
                }
                let mut path_to_source = path.clone();
                let mut dep_path = path.clone();
                dep_path.push(dep_token(name, *version, i, dep));
                let list = origins.entry(dep.package.clone()).or_default();
                if !list.iter().any(|o| o.constraint == dep.constraint && o.path == dep_path) {
                    list.push(Origin {
                        constraint: dep.constraint,
                        path: dep_path,
                    });
                    grew = true;
                }
                let dest_features = requests.entry(dep.package.clone()).or_default();
                let before = dest_features.len();
                dest_features.extend(dep.features.iter().cloned());
                if dest_features.len() != before {
                    grew = true;
                }
                path_to_source.clear();
            }
        }
        if !grew {
            let mut constraints = BTreeMap::new();
            let mut evidence = BTreeMap::new();
            for (name, os) in &origins {
                constraints.insert(
                    name.clone(),
                    canonical_constraints(os.iter().map(|o| o.constraint).collect()),
                );
                evidence.insert(
                    name.clone(),
                    os.iter()
                        .map(|o| Evidence {
                            constraint: o.constraint,
                            path: o.path.clone(),
                        })
                        .collect(),
                );
            }
            let mut paths = BTreeMap::new();
            for (name, os) in &origins {
                if let Some(path) = os.iter().map(|o| o.path.clone()).min_by(path_cmp) {
                    paths.insert(name.clone(), path);
                }
            }
            return Closure {
                constraints,
                features: requests,
                evidence,
                paths,
            };
        }
    }
}

fn validate_input(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", base),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", base, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.enable[{}]", base, e),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen = BTreeSet::new();
    for p in catalog {
        if !seen.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }
    for p in catalog {
        let mut versions = BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut features = BTreeSet::new();
            for f in &r.features {
                if !features.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    seen.clear();
    for lock in locks {
        if !seen.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, &'a Package>,
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = compute_closure(self.roots, selected, self.catalog, self.target);

        let missing = closure
            .constraints
            .keys()
            .filter(|name| !self.catalog.contains_key(*name))
            .next()
            .cloned();
        if let Some(package) = missing {
            self.record_error(ResolveError::Missing {
                evidence: closure.evidence.get(&package).cloned().unwrap_or_default(),
                package,
            });
            return Ok(());
        }

        for (package, version) in selected.iter() {
            if let Some(cs) = closure.constraints.get(package) {
                if cs.iter().any(|c| !matches(*version, *c)) {
                    self.record_error(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: cs.clone(),
                        evidence: closure.evidence.get(package).cloned().unwrap_or_default(),
                    });
                    return Ok(());
                }
            }
        }

        for (package, cs) in &closure.constraints {
            if selected.contains_key(package) {
                continue;
            }
            let pkg = self.catalog[package];
            if !pkg.releases.iter().any(|r| cs.iter().all(|c| matches(r.version, *c))) {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: cs.clone(),
                    evidence: closure.evidence.get(package).cloned().unwrap_or_default(),
                });
                return Ok(());
            }
        }

        for (package, version) in selected.iter() {
            let Some(features) = closure.features.get(package) else {
                continue;
            };
            let Some(pkg) = self.catalog.get(package) else {
                continue;
            };
            let Some(release) = pkg.releases.iter().find(|r| r.version == *version) else {
                continue;
            };
            for feature in features {
                if !release.features.iter().any(|f| &f.name == feature) {
                    self.record_error(ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: closure.paths.get(package).cloned().unwrap_or_default(),
                    });
                    return Ok(());
                }
            }
        }

        let next = closure
            .constraints
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();
        let Some(package) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| selected.get(*name) != Some(*version))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_map, _)) => {
                    cost < *best_cost
                        || (cost == *best_cost && vector_cmp(selected, best_map, self.catalog) == std::cmp::Ordering::Greater)
                }
            };
            if replace {
                self.best = Some((cost, selected.clone(), closure));
            }
            return Ok(());
        };

        let cs = closure.constraints.get(&package).cloned().unwrap_or_default();
        let mut candidates: Vec<Version> = self.catalog[&package]
            .releases
            .iter()
            .filter(|r| cs.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(&package) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        let mut first_local: Option<ResolveError> = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            let result = self.visit(selected);
            selected.remove(&package);
            if let Err(e) = result {
                return Err(e);
            }
            if first_local.is_none() {
                first_local = self.first_error.clone();
            }
        }
        if self.best.is_none() && self.first_error.is_none() {
            self.first_error = first_local;
        }
        Ok(())
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }
}

fn vector_cmp(
    a: &BTreeMap<String, Version>,
    b: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
) -> std::cmp::Ordering {
    for name in catalog.keys() {
        let ord = match (a.get(name), b.get(name)) {
            (None, None) => std::cmp::Ordering::Equal,
            (None, Some(_)) => std::cmp::Ordering::Less,
            (Some(_), None) => std::cmp::Ordering::Greater,
            (Some(x), Some(y)) => x.cmp(y),
        };
        if ord != std::cmp::Ordering::Equal {
            return ord;
        }
    }
    std::cmp::Ordering::Equal
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_input(catalog, roots, locks)?;

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        roots,
        catalog: &catalog_map,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut BTreeMap::new())?;

    let Some((_, selected_map, final_closure)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(selected_map.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_closure
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| cs.iter().any(|c| !matches(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }
    Ok(Resolution { selected, changes })
}
