#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(values: &[Constraint]) -> Vec<Constraint> {
    let mut result = values.to_vec();
    result.sort_by_key(|c| constraint_key(*c));
    result.dedup();
    result
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(values: &[Constraint]) -> String {
    canonical_constraints(values)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn satisfies(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

fn percent_encode(value: &str) -> String {
    let mut output = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            output.push(*byte as char);
        } else {
            output.push('%');
            output.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            output.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    output
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            percent_encode(site),
            percent_encode(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", percent_encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            percent_encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", percent_encode(package))
        }
        ResolveError::Missing { package } => {
            format!("missing|package={}", percent_encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            percent_encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            percent_encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct SearchResult {
    selected: std::collections::BTreeMap<String, Version>,
    constraints: std::collections::BTreeMap<String, Vec<Constraint>>,
}

struct Search<'a> {
    catalog: &'a [Package],
    catalog_by_name: std::collections::BTreeMap<&'a str, &'a Package>,
    locks: std::collections::BTreeMap<&'a str, Version>,
    budget: usize,
    attempts: usize,
    best: Option<SearchResult>,
    best_cost: usize,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn inspect(
        &mut self,
        selected: &std::collections::BTreeMap<String, Version>,
        roots: &[Requirement],
    ) -> Result<(), ResolveError> {
        let constraints = active_constraints(self.catalog, roots, selected);

        if let Some(name) = constraints
            .keys()
            .filter(|name| !self.catalog_by_name.contains_key(name.as_str()))
            .next()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in selected {
            if let Some(reqs) = constraints.get(name) {
                if !reqs.iter().all(|c| satisfies(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: canonical_constraints(reqs),
                    });
                }
            }
        }

        for (name, reqs) in &constraints {
            if selected.contains_key(name) {
                continue;
            }
            let package = self.catalog_by_name[name.as_str()];
            if !package
                .releases
                .iter()
                .any(|release| reqs.iter().all(|c| satisfies(release.version, *c)))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical_constraints(reqs),
                });
            }
        }

        let next = constraints
            .keys()
            .find(|name| !selected.contains_key(name.as_str()))
            .cloned();

        let Some(package_name) = next else {
            let result = SearchResult {
                selected: selected.clone(),
                constraints,
            };
            let cost = self.lock_cost(&result.selected);
            if self
                .best
                .as_ref()
                .map(|best| {
                    cost < self.best_cost
                        || (cost == self.best_cost
                            && self.vector_cmp(&result.selected, &best.selected)
                                == std::cmp::Ordering::Greater)
                })
                .unwrap_or(true)
            {
                self.best_cost = cost;
                self.best = Some(result);
            }
            return Ok(());
        };

        let package = self.catalog_by_name[package_name.as_str()];
        let reqs = &constraints[&package_name];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|release| reqs.iter().all(|c| satisfies(release.version, *c)))
            .map(|release| release.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(package_name.as_str()) {
            if let Some(position) = candidates.iter().position(|v| v == locked) {
                let version = candidates.remove(position);
                candidates.insert(0, version);
            }
        }

        for version in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            let mut child = selected.clone();
            child.insert(package_name.clone(), version);
            match self.inspect(&child, roots) {
                Ok(()) => {}
                Err(error @ ResolveError::Limit { .. }) => return Err(error),
                Err(error) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(error);
                    }
                }
            }
        }
        Ok(())
    }

    fn lock_cost(&self, selected: &std::collections::BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(name, version)| selected.get(**name).copied() != Some(**version))
            .count()
    }

    fn vector_cmp(
        &self,
        a: &std::collections::BTreeMap<String, Version>,
        b: &std::collections::BTreeMap<String, Version>,
    ) -> std::cmp::Ordering {
        for package in self.catalog {
            let name = &package.name;
            match (a.get(name), b.get(name)) {
                (Some(av), Some(bv)) => match av.cmp(bv) {
                    std::cmp::Ordering::Equal => {}
                    other => return other,
                },
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (None, None) => {}
            }
        }
        std::cmp::Ordering::Equal
    }
}

fn active_constraints(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut constraints = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in selected {
        if let Some(package) = catalog.iter().find(|package| package.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    constraints
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for values in constraints.values_mut() {
        *values = canonical_constraints(values);
    }
    constraints
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let catalog_by_name = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();
    let lock_map = locks
        .iter()
        .map(|lock| (lock.package.as_str(), lock.version))
        .collect();

    let mut search = Search {
        catalog,
        catalog_by_name,
        locks: lock_map,
        budget: max_attempts,
        attempts: 0,
        best: None,
        best_cost: usize::MAX,
        first_error: None,
    };

    let selected = std::collections::BTreeMap::new();
    if let Err(error) = search.inspect(&selected, roots) {
        return Err(error);
    }

    let Some(result) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected_vec: Vec<Selection> = result
        .selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let lock_map: std::collections::BTreeMap<&str, Version> = locks
        .iter()
        .map(|lock| (lock.package.as_str(), lock.version))
        .collect();
    let selected_map = &result.selected;
    let mut all_names = std::collections::BTreeSet::<String>::new();
    for lock in locks {
        all_names.insert(lock.package.clone());
    }
    for package in selected_map.keys() {
        all_names.insert(package.clone());
    }

    let mut changes = Vec::new();
    for name in all_names {
        let before = lock_map.get(name.as_str()).copied();
        let after = selected_map.get(&name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|package| package.name == name)
                    .map(|package| package.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if result
                    .constraints
                    .get(&name)
                    .map(|constraints| {
                        !constraints.iter().all(|c| satisfies(old, *c))
                    })
                    .unwrap_or(false)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
