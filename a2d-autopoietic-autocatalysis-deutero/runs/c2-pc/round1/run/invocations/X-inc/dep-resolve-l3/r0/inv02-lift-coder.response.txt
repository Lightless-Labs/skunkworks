#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn constraint_matches(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonicalize(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn format_version(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn format_constraint(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", format_version(v)),
        Constraint::AtLeast(v) => format!(">={}", format_version(v)),
        Constraint::Before(v) => format!("<{}", format_version(v)),
        Constraint::Caret(v) => format!("^{}", format_version(v)),
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for byte in s.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    out
}

fn format_constraints(constraints: &[Constraint]) -> String {
    canonicalize(constraints.to_vec())
        .iter()
        .map(|c| format_constraint(*c))
        .collect::<Vec<_>>()
        .join(",")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            format_version(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            format_constraints(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            format_version(*selected),
            format_constraints(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = std::collections::HashSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::HashSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = std::collections::HashSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

type Assignment = std::collections::BTreeMap<String, Version>;
type Constraints = std::collections::BTreeMap<String, Vec<Constraint>>;

fn active_constraints(roots: &[Requirement], assignment: &Assignment, catalog: &[Package]) -> Constraints {
    let mut active = Constraints::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assignment {
        if let Some(package) = catalog.iter().find(|p| &p.name == name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonicalize(std::mem::take(constraints));
    }
    active
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &std::collections::HashMap<String, Version>,
    assignment: &mut Assignment,
    attempts: &mut usize,
    max_attempts: usize,
    best: &mut Option<(usize, Vec<Option<Version>>, Assignment)>,
) -> Result<(), ResolveError> {
    let active = active_constraints(roots, assignment, catalog);

    let missing = active
        .keys()
        .filter(|name| !catalog.iter().any(|p| &p.name == *name))
        .min();
    if let Some(name) = missing {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, selected) in assignment.iter() {
        if let Some(constraints) = active.get(name) {
            if !constraints
                .iter()
                .all(|c| constraint_matches(*c, *selected))
            {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *selected,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in &active {
        if assignment.contains_key(name) {
            continue;
        }
        let package = catalog.iter().find(|p| &p.name == name).unwrap();
        if !package.releases.iter().any(|release| {
            constraints
                .iter()
                .all(|c| constraint_matches(*c, release.version))
        }) {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    let next = active.keys().find(|name| !assignment.contains_key(*name));
    let Some(name) = next else {
        let cost = locks
            .iter()
            .filter(|(package, version)| assignment.get(*package) != Some(*version))
            .count();
        let names: Vec<&String> = catalog.iter().map(|p| &p.name).collect();
        let mut sorted_names = names;
        sorted_names.sort();
        let vector = sorted_names
            .iter()
            .map(|package| assignment.get(*package).copied())
            .collect::<Vec<_>>();
        let replace = match best {
            None => true,
            Some((best_cost, best_vector, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if replace {
            *best = Some((cost, vector, assignment.clone()));
        }
        return Ok(());
    };

    let constraints = active.get(name).unwrap();
    let package = catalog.iter().find(|p| &p.name == name).unwrap();
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|release| {
            constraints
                .iter()
                .all(|c| constraint_matches(*c, release.version))
        })
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(locked) = locks.get(name) {
        if let Some(position) = candidates.iter().position(|v| v == locked) {
            let version = candidates.remove(position);
            candidates.insert(0, version);
        }
    }

    let mut first_error: Option<ResolveError> = None;
    for candidate in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        assignment.insert(name.clone(), candidate);
        let result = search(
            catalog,
            roots,
            locks,
            assignment,
            attempts,
            max_attempts,
            best,
        );
        assignment.remove(name);
        match result {
            Ok(()) => {}
            Err(ResolveError::Limit { .. }) => return result,
            Err(error) => {
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }
    match first_error {
        Some(error) => Err(error),
        None => unreachable!(),
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let lock_map: std::collections::HashMap<String, Version> = locks
        .iter()
        .map(|lock| (lock.package.clone(), lock.version))
        .collect();

    let mut attempts = 0;
    let mut best = None;
    let mut assignment = Assignment::new();
    let result = search(
        catalog,
        roots,
        &lock_map,
        &mut assignment,
        &mut attempts,
        max_attempts,
        &mut best,
    );

    let Some((_, _, selected_map)) = best else {
        return Err(result.err().unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    let final_constraints = active_constraints(roots, &selected_map, catalog);

    let mut before = std::collections::BTreeMap::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let names: std::collections::BTreeSet<String> = before
        .keys()
        .chain(selected_map.keys())
        .cloned()
        .collect();
    let mut changes = Vec::new();
    for package_name in names {
        let old = before.get(&package_name).copied();
        let new = selected_map.get(&package_name).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package_name)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old_version));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_constraints
                    .get(&package_name)
                    .is_some_and(|constraints| {
                        !constraints
                            .iter()
                            .all(|c| constraint_matches(*c, old_version))
                    })
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: package_name,
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
