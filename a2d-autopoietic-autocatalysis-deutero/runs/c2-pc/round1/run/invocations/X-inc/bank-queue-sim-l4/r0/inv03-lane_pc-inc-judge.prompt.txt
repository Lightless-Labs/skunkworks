pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };
    Some(quotient + u128::from(round_up))
}

#[derive(Clone)]
struct CustomerState {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    accepted: bool,
    ever_started: bool,
    total_waited: u64,
    queue_entry: u64,
    pause_time: Option<u64>,
    terminal: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch_time: u64,

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiter_cmp(a: &CustomerState, b: &CustomerState) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(pa), Some(pb)) => pa.cmp(&pb).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn best_waiter(states: &[CustomerState], waiting: &[usize]) -> Option<usize> {
    waiting
        .iter()
        .copied()
        .min_by(|&a, &b| waiter_cmp(&states[a], &states[b]))
}

fn expire_waiters(
    now: u64,
    states: &mut [CustomerState],
    waiting: &mut Vec<usize>,
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<(u64, u32, usize)> = waiting
        .iter()
        .copied()
        .filter_map(|i| {
            states[i]
                .deadline
                .filter(|&d| d <= now)
                .map(|d| (d, states[i].customer.id, i))
        })
        .collect();
    expired.sort_by_key(|&(deadline, id, _)| (deadline, id));

    for (_, _, i) in expired {
        if let Some(pos) = waiting.iter().position(|&j| j == i) {
            waiting.remove(pos);
            let segment = now - states[i].queue_entry;
            states[i].total_waited = states[i].total_waited.saturating_add(segment);
            states[i].terminal = true;
            stats.reneged += 1;
            stats.wait_ticks += segment as u128;
            emit(
                events,
                now,
                states[i].customer.id,
                Kind::Renege {
                    waited: states[i].total_waited,
                },
            );
        }
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        let deadline = match c.patience {
            Some(p) => Some(
                c.arrival
                    .checked_add(p)
                    .ok_or(Error::DeadlineOverflow { index, id: c.id })?,
            ),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<CustomerState> = customers
        .iter()
        .enumerate()
        .map(|(i, c)| CustomerState {
            customer: c.clone(),
            deadline: initial_deadlines[i],
            remaining: c.service,
            accepted: false,
            ever_started: false,
            total_waited: 0,
            queue_entry: c.arrival,
            pause_time: None,
            terminal: false,
        })
        .collect();

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    let mut waiting = Vec::<usize>::new();
    let mut active: Option<Active> = None;
    let mut previous_dispatch: Option<u32> = None;
    let mut first_dispatch = true;
    let mut arrival_pos = 0usize;
    let mut now = order
        .first()
        .map(|&i| customers[i].arrival)
        .unwrap_or(0);

    while arrival_pos < order.len()
        || active.is_some()
        || !waiting.is_empty()
    {
        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let i = a.index;
                active = None;
                states[i].terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - states[i].customer.arrival) as u128;
                stats.completion_order.push(states[i].customer.id);
                emit(events.as_mut(), now, states[i].customer.id, Kind::Finish);
            }
        }

        expire_waiters(now, &mut states, &mut waiting, &mut events, &mut stats);

        while arrival_pos < order.len() && customers[order[arrival_pos]].arrival == now {
            let i = order[arrival_pos];
            arrival_pos += 1;
            let c = &customers[i];
            emit(&mut events, now, c.id, Kind::Arrive);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                stats.balked += 1;
                emit(&mut events, now, c.id, Kind::Balk);
            } else {
                states[i].accepted = true;
                states[i].queue_entry = now;
                waiting.push(i);
                stats.accepted += 1;
            }
        }

        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&states, &waiting) {
                    if states[w].customer.priority < states[a.index].customer.priority {
                        let i = a.index;
                        let remaining = a.finish - now;
                        emit(
                            &mut events,
                            now,
                            states[i].customer.id,
                            Kind::Preempt {
                                remaining,
                                by: states[w].customer.id,
                            },
                        );
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch_time) as u128;
                        active = None;
                        states[i].remaining = remaining;
                        states[i].queue_entry = now;
                        states[i].pause_time = Some(now);
                        if let Some(budget) = states[i].customer.patience {
                            states[i].deadline = Some(now.checked_add(budget).ok_or(
                                Error::TimeOverflow {
                                    customer: states[i].customer.id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                },
                            )?);
                        }
                        waiting.push(i);
                    }
                }
            }
        }

        while active.is_none() {
            let Some(i) = best_waiter(&states, &waiting) else {
                break;
            };
            let pos = waiting.iter().position(|&j| j == i).unwrap();
            waiting.remove(pos);

            let segment = now - states[i].queue_entry;
            states[i].total_waited = states[i].total_waited.saturating_add(segment);
            stats.wait_ticks += segment as u128;

            let switch = if first_dispatch {
                0
            } else if previous_dispatch == Some(states[i].customer.id) {
                0
            } else {
                config.switch_cost
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: states[i].customer.id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(states[i].remaining).ok_or(Error::TimeOverflow {
                customer: states[i].customer.id,
                time: now,
                stage: OverflowStage::Finish,

fn expire_waiters(
    now: u64,
    waiting: &mut Vec<Waiting>,
    customers: &[Customer],
    events: &mut Vec<Event>,
    reneged: &mut usize,
    wait_ticks: &mut u128,
) {

    let mut retained: Vec<Waiting> = Vec::new();
    for w in waiting.drain(..) {
        if w.deadline.is_some_and(|d| d <= now) {
            expired.push(w);
        } else {
            retained.push(w);
        }
    }
    expired.sort_by(|a, b| {
        a.deadline
            .cmp(&b.deadline)
            .then_with(|| customers[a.idx].id.cmp(&customers[b.idx].id))
    });
    *waiting = retained;
    for w in expired {
        let segment = now - w.entry;
        *wait_ticks += segment as u128;
        let total = w.waited.saturating_add(segment);
        events.push(Event {
            time: now,
            customer: customers[w.idx].id,
            kind: Kind::Renege { waited: total },
        });
        *reneged += 1;
    }
}

        .iter()
        .find(|w| w.idx == idx)
        .map(|w| w.waited)
        .unwrap_or(0);
    budget.saturating_sub(spent)
}

    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(if round_up { q + 1 } else { q })
}

#[derive(Clone)]
struct Waiting {
    idx: usize,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    idx: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
    protected: bool,
    interval_start: u64,
}

fn waiter_better(a: &Waiting, b: &Waiting, customers: &[Customer]) -> bool {
    let ca = &customers[a.idx];
    let cb = &customers[b.idx];
    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }
    if a.paused != b.paused {
        return a.paused;
    }
    if a.paused {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        ca.id < cb.id
    } else if ca.arrival != cb.arrival {
        ca.arrival < cb.arrival
    } else {
        ca.id > cb.id
    }
}

fn best_waiter(waiting: &[Waiting], customers: &[Customer]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.map_or(true, |b| waiter_better(&waiting[i], &waiting[b], customers)) {
            best = Some(i);
        }
    }
    best
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut deadlines = Vec::with_capacity(customers.len());
    for (i, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index: i });
        }
        if customers[..i].iter().any(|p| p.id == c.id) {
            return Err(Error::DuplicateId { index: i, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority { index: i, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index: i,
                id: c.id,
            })?),
            None => None,
        };
        deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut previous_dispatch: Option<u32> = None;
    let mut ever_started = vec![false; customers.len()];

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();

    while next_arrival < order.len() || active.is_some() || !waiting.is_empty() {
        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let idx = a.idx;
                let id = customers[idx].id;
                busy_ticks += (now - a.interval_start) as u128;
                active = None;
                completed += 1;
                turnaround_ticks += (now - customers[idx].arrival) as u128;
                completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
            }
        }

        if let Some(a) = active.as_mut() {
            if a.work_at == now {
                a.protected = false;
            }
        }

        let mut expired: Vec<usize> = waiting
            .iter()
            .enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| (d, customers[w.idx].id, i)))
            .map(|(_, _, i)| i)
            .collect();
        expired.sort_by_key(|&i| (waiting[i].deadline.unwrap(), customers[waiting[i].idx].id));
        for i in expired.into_iter().rev() {
            let w = waiting.remove(i);
            let id = customers[w.idx].id;
            let total = w.waited + (now - w.entry);
            wait_ticks += (now - w.entry) as u128;
            reneged += 1;
            emit(&mut events, now, id, Kind::Renege { waited: total });
        }

        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let idx = order[next_arrival];
            next_arrival += 1;
            let c = &customers[idx];
            emit(&mut events, now, c.id, Kind::Arrive);
            let idle_extra = usize::from(active.is_none());
            if waiting.len() < config.waiting_capacity + idle_extra {
                accepted += 1;
                waiting.push(Waiting {
                    idx,
                    entry: now,
                    deadline: deadlines[idx],
                    waited: 0,
                    paused: false,
                    pause_time: 0,
                });
            } else {
                balked += 1;
                emit(&mut events, now, c.id, Kind::Balk);
            }
        }

        if let Some(a) = active.as_ref() {
            if !a.protected {
                if let Some(wi) = best_waiter(&waiting, customers) {
                    let run_priority = customers[a.idx].priority;
                    let waiter = &waiting[wi];
                    if customers[waiter.idx].priority < run_priority {
                        let chosen = waiter.idx;
                        let remaining = a.finish - now;
                        let run_idx = a.idx;
                        let run_id = customers[run_idx].id;
                        let total_active = now - a.interval_start;
                        busy_ticks += total_active as u128;
                        emit(
                            &mut events,
                            now,
                            run_id,
                            Kind::Preempt { remaining, by: customers[chosen].id },
                        );
                        preemptions += 1;
                        active = None;

                        let c = &customers[run_idx];
                        let old_deadline = match c.patience {
                            Some(_) => Some(now.checked_add(a.remaining /* replaced below */)),
                            None => None,
                        };
                        let consumed = now - a.interval_start;
                        let _ = old_deadline;
                        let _ = consumed;

                        let remaining_budget = c.patience.map(|budget| {
                            let already_waited = 0u64;
                            budget.saturating_sub(already_waited)
                        });
                        let new_deadline = match remaining_budget {
                            Some(budget) => Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                                customer: run_id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })?),
                            None => None,
                        };
                        waiting.push(Waiting {
                            idx: run_idx,
                            entry: now,
                            deadline: new_deadline,
                            waited: 0,
                            paused: true,
                            pause_time: now,
                        });
                    }
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let Some(wi) = best_waiter(&waiting, customers) else {
                break;
            };
            let w = waiting.remove(wi);
            let idx = w.idx;
            let id = customers[idx].id;
            let waited_now = now - w.entry;
            wait_ticks += waited_now as u128;
            let total_wait = w.waited + waited_now;

            let switch = if previous_dispatch.is_some_and(|p| p != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(if active.is_none() { customers[idx].service } else { 0 })
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;
            let remaining = customers[idx].service;
            let is_resume = ever_started[idx];
            if !is_resume {
                ever_started[idx] = true;
                started += 1;
                emit(&mut events, now, id, Kind::Start { remaining, work_at, finish });
            } else {
                emit(&mut events, now, id, Kind::Resume { remaining, work_at, finish });
            }
            previous_dispatch = Some(id);

            if finish == now {
                completed += 1;
                turnaround_ticks += (now - customers[idx].arrival) as u128;
                completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
                continue;
            }
            active = Some(Active {
                idx,
                remaining,
                work_at,
                finish,
                protected: work_at > now,
                interval_start: now,
            });
            let _ = total_wait;
        }

        // Phase F: remove newly queued expired customers. In this implementation
        // they are also considered by Phase B at the next loop iteration; process
        // them here to preserve the specified phase order.
        let mut expired: Vec<usize> = waiting
            .iter()
            .enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| (d, customers[w.idx].id, i)))
            .map(|(_, _, i)| i)
            .collect();
        expired.sort_by_key(|&i| (waiting[i].deadline.unwrap(), customers[waiting[i].idx].id));
        for i in expired.into_iter().rev() {
            let w = waiting.remove(i);
            let id = customers[w.idx].id;
            let seg = now - w.entry;
            wait_ticks += seg as u128;
            reneged += 1;
            emit(&mut events, now, id, Kind::Renege { waited: w.waited + seg });
        }

        if next_arrival >= order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now {
                next_time = Some(next_time.map_or(t, |n| n.min(t)));
            }
        };
        if next_arrival < order.len() {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            if a.protected {
                consider(a.work_at);
            }
        }
        for w in &waiting {
            if let Some(d) = w.deadline {
                consider(d);
            }
        }
        if let Some(t) = next_time {
            now = t;
        } else {
            // All remaining work should be handled at the current time.
            // This guard prevents accidental non-progress in degenerate states.
            if active.is_some() {
                let a = active.as_ref().unwrap();
                if a.finish == now {
                    continue;
                }
            }
            break;
        }
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}
