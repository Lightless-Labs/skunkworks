pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape {
        axis: Axis,
        size: usize,
    },
    Bounds {
        entry: usize,
        axis: Axis,
        index: usize,
        limit: usize,
    },
    NonFinite {
        entry: usize,
    },
    SumOverflow {
        entry: usize,
    },
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        if rows > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Row,
                size: rows,
            });
        }
        if cols > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Column,
                size: cols,
            });
        }

        for (entry_index, entry) in entries.iter().enumerate() {
            if entry.row >= rows {
                return Err(Error::Bounds {
                    entry: entry_index,
                    axis: Axis::Row,
                    index: entry.row,
                    limit: rows,
                });
            }
            if entry.col >= cols {
                return Err(Error::Bounds {
                    entry: entry_index,
                    axis: Axis::Column,
                    index: entry.col,
                    limit: cols,
                });
            }
            if !entry.value.is_finite() {
                return Err(Error::NonFinite { entry: entry_index });
            }
        }

        let mut sums: Vec<Entry> = Vec::new();
        let mut last_entry: Vec<usize> = Vec::new();

        for (entry_index, entry) in entries.iter().enumerate() {
            let position = sums
                .iter()
                .position(|sum| sum.row == entry.row && sum.col == entry.col);

            if let Some(position) = position {
                let new_sum = sums[position].value + entry.value;
                if !new_sum.is_finite() {
                    return Err(Error::SumOverflow { entry: entry_index });
                }
                sums[position].value = new_sum;
                last_entry[position] = entry_index;
            } else {
                sums.push(Entry {
                    row: entry.row,
                    col: entry.col,
                    value: 0.0 + entry.value,
                });
                last_entry.push(entry_index);
            }
        }

        let _ = last_entry;
        sums.retain(|entry| entry.value != 0.0);
        sums.sort_by(|a, b| a.row.cmp(&b.row).then_with(|| a.col.cmp(&b.col)));

        Ok(Self {
            rows,
            cols,
            entries: sums,
        })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }
}
