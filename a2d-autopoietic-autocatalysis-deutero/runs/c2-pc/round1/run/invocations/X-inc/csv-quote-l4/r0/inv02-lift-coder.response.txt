#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart { Delimiter, Quote }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy { Reject, KeepFirst, KeepLast }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = Vec::new();
    let mut record_started = false;
    let mut field_started = false;
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: closed quote
    let mut quote_start = 0usize;
    let mut record_index = 0usize;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: pos });
        }

        if state != 2 && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: pos });
        }

        if state == 2 {
            if b == quote {
                if bytes.get(pos + 1) == Some(&quote) {
                    field.push(quote);
                    pos += 2;
                } else {
                    state = 3;
                    pos += 1;
                }
            } else {
                field.push(b);
                pos += 1;
            }
            continue;
        }

        if state == 3 {
            if b == delimiter {
                record.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                state = 0;
                field_started = false;
                record_started = true;
                pos += 1;
            } else if b == b'\n' || b == b'\r' {
                record.push(String::from_utf8(field).unwrap());
                records.push(record);
                record = Vec::new();
                field = Vec::new();
                state = 0;
                field_started = false;
                record_started = false;
                record_index += 1;
                pos += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record: record_index, byte: pos });
            }
            continue;
        }

        if b == b'\n' || b == b'\r' {
            record.push(String::from_utf8(field).unwrap());
            records.push(record);
            record = Vec::new();
            field = Vec::new();
            state = 0;
            field_started = false;
            record_started = false;
            record_index += 1;
            pos += if b == b'\r' { 2 } else { 1 };
        } else if b == delimiter {
            record.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            state = 0;
            field_started = false;
            record_started = true;
            pos += 1;
        } else if b == quote {
            if state == 0 && !field_started {
                state = 2;
                quote_start = pos;
                field_started = true;
                record_started = true;
                pos += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: pos });
            }
        } else {
            field.push(b);
            state = 1;
            field_started = true;
            record_started = true;
            pos += 1;
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_start });
    }

    if record_started || field_started || !record.is_empty() {
        record.push(String::from_utf8(field).unwrap());
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut iter = records.into_iter();
    let header = if config.headers { iter.next() } else { None };

    if let Some(ref header_values) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: std::collections::HashMap<&str, usize> = std::collections::HashMap::new();
            for (column, name) in header_values.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second: column,
                    });
                }
                seen.insert(name.as_str(), column);
            }
        }
    }

    let mut rows = Vec::new();
    for (index, row) in iter.enumerate() {
        let record = index + usize::from(header.is_some());
        if let Some(ref h) = header {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
        rows.push(row);
    }

    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), result: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if self.result.is_none() {
            let parsed = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
                .and_then(|records| table_from_records(records, self.config));
            self.result = Some(parsed);
        }
        self.result.as_ref().unwrap().clone()
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef { Index(usize), Name(String) }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType { Text, Integer, Decimal, Boolean }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value { Null, Text(String), Integer(i64), Decimal(i64), Boolean(bool) }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn, MissingCell, Empty, InvalidInteger, IntegerOverflow,
    InvalidDecimal, DecimalOverflow, InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy { Collect, FirstPerRow, Stop }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start]) && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if b.len() < start + 4 || b[b.len() - 3] != b'.' {
        return false;
    }
    let integral = &b[start..b.len() - 3];
    if integral.is_empty() || !b[b.len() - 2..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    let integral_ok = if integral[0] == b'0' {
        integral.len() == 1
    } else {
        (b'1'..=b'9').contains(&integral[0]) && integral[1..].iter().all(u8::is_ascii_digit)
    };
    if !integral_ok { return false; }
    !(start == 1 && integral == b"0" && b[b.len() - 2..] == *b"00")
}

fn evaluate_cell(value: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if value.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(value.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(value) { return Err(CellErrorKind::InvalidInteger); }
            value.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(value) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = value.starts_with('-');
            let body = if neg { &value[1..] } else { value };
            let (whole, frac) = body.split_once('.').unwrap();
            let magnitude = whole.parse::<i128>().unwrap() * 100
                + frac.parse::<i128>().unwrap();
            let signed = if neg { -magnitude } else { magnitude };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match value {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        }
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = vec![None; columns.len()];
        let mut stopped = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if stopped { break; }
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, evaluated) = match resolved {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(index) => match row.get(index) {
                    None => (Some(index), Err(CellErrorKind::MissingCell)),
                    Some(cell) => (Some(index), evaluate_cell(cell, spec)),
                }
            };
            match evaluated {
                Ok(value) => out[spec_index] = Some(value),
                Err(kind) => {
                    result.errors.push(CellError { row: row_index, spec: spec_index, column, kind });
                    if policy != ErrorPolicy::Collect { stopped = true; }
                }
            }
        }
        result.rows.push(out);
        if policy == ErrorPolicy::Stop && !result.errors.is_empty() { break; }
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding { Lf, CrLf }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode { Preserve, Canonical }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn write_duplicate(header: &[String]) -> Option<WriteError> {
    let mut seen: std::collections::HashMap<&str, usize> = std::collections::HashMap::new();
    for (column, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(WriteError::DuplicateHeader {
                name: name.clone(),
                first,
                second: column,
            });
        }
        seen.insert(name.as_str(), column);
    }
    None
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let order = x.as_bytes().cmp(y.as_bytes());
        if order != std::cmp::Ordering::Equal { return order; }
    }
    a.len().cmp(&b.len())
}

fn encode_field(
    out: &mut String,
    value: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole_empty: bool,
) {
    let bytes = value.as_bytes();
    let quote_it = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_first_record && value.starts_with('\u{feff}'))
        || (sole_empty && value.is_empty());

    if !quote_it {
        out.push_str(value);
        return;
    }
    out.push(quote as char);
    for ch in value.chars() {
        if ch as u32 == quote as u32 { out.push(quote as char); }
        out.push(ch);
    }
    out.push(quote as char);
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header { original.push(header); }
    original.extend(table.rows.iter());

    for (record, row) in original.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul { record, column, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in original.iter().enumerate() {
        if row.is_empty() { return Err(WriteError::ZeroColumns { record }); }
    }

    if config.mode == WriteMode::Canonical
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some(header) = &table.header {
            if let Some(error) = write_duplicate(header) { return Err(error); }
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header { records.push(header.clone()); }
            records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut selected: std::collections::HashMap<&str, usize> =
                    std::collections::HashMap::new();
                for (i, name) in header.iter().enumerate() {
                    match table.duplicates {
                        DuplicatePolicy::KeepLast => { selected.insert(name, i); }
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            selected.entry(name).or_insert(i);
                        }
                    }
                }
                let mut indexes: Vec<usize> = selected.values().copied().collect();
                indexes.sort_by(|a, b| header[*a].as_bytes().cmp(header[*b].as_bytes()));
                let canonical_header: Vec<String> = indexes.iter().map(|i| header[*i].clone()).collect();
                records.push(canonical_header);
                let mut rows: Vec<Vec<String>> = table.rows.iter()
                    .map(|row| indexes.iter().map(|i| row[*i].clone()).collect())
                    .collect();
                rows.sort_by(|a, b| compare_rows(a, b));
                records.extend(rows);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_rows(a, b));
                records = rows;
            }
        }
    }

    if records.is_empty() { return Ok(String::new()); }

    let mut out = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let ending = if canonical || config.line_ending == LineEnding::Lf {
        "\n"
    } else {
        "\r\n"
    };

    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 { out.push_str(ending); }
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 { out.push(config.delimiter as char); }
            encode_field(
                &mut out,
                field,
                config.delimiter,
                config.quote,
                record_index == 0 && column_index == 0,
                row.len() == 1,
            );
        }
    }
    if canonical || config.final_terminator { out.push_str(ending); }
    Ok(out)
}
