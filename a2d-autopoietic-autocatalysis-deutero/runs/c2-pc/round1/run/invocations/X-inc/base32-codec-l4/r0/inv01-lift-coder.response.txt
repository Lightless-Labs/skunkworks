#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

const STANDARD: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
const HEX: &[u8; 32] = b"0123456789ABCDEFGHIJKLMNOPQRSTUV";

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => STANDARD,
        Alphabet::Hex => HEX,
    }
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = self.pending.clone();
        combined.extend_from_slice(input);
        let complete_len = combined.len() / 5 * 5;

        let mut output = String::new();
        let alphabet = alphabet_bytes(self.config.alphabet);
        for chunk in combined[..complete_len].chunks_exact(5) {
            let bits = u64::from_be_bytes([chunk[0], chunk[1], chunk[2], chunk[3], chunk[4], 0, 0, 0]);
            for i in 0..8 {
                let shift = 59 - i * 5;
                output.push(alphabet[((bits >> shift) & 31) as usize] as char);
            }
        }

        self.pending = combined[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if !self.pending.is_empty() {
            let alphabet = alphabet_bytes(self.config.alphabet);
            let mut bits = 0u64;
            for &byte in &self.pending {
                bits = (bits << 8) | byte as u64;
            }
            let bit_count = self.pending.len() * 8;
            let symbols = (bit_count + 4) / 5;
            bits <<= 40 - bit_count;

            for i in 0..symbols {
                let shift = 35 - i * 5;
                output.push(alphabet[((bits >> shift) & 31) as usize] as char);
            }

            if self.config.padding == Padding::Required {
                while output.len() % 8 != 0 {
                    output.push('=');
                }
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

#[derive(Clone)]
struct DecoderState {
    position: usize,
    data_count: usize,
    bit_buffer: u8,
    bit_count: u8,
    last_value: u8,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
    closed: bool,
}

pub struct Decoder {
    config: Config,
    state: DecoderState,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            state: DecoderState {
                position: 0,
                data_count: 0,
                bit_buffer: 0,
                bit_count: 0,
                last_value: 0,
                last_data_offset: None,
                first_padding_offset: None,
                padding_count: 0,
                expected_padding: None,
                closed: false,
            },
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.state.closed {
            return Err(StreamError::Closed);
        }

        let mut state = self.state.clone();
        let mut output = Vec::new();
        let alphabet = alphabet_bytes(self.config.alphabet);

        for &byte in input.as_bytes() {
            let offset = state.position;
            let recognized = if byte == b'=' {
                None
            } else {
                let upper = byte.to_ascii_uppercase();
                alphabet.iter().position(|&b| b == upper).map(|value| (upper, value))
            };

            if byte != b'=' && recognized.is_none() {
                return Err(StreamError::InvalidByte { offset, byte });
            }

            if let Some((upper, _)) = recognized {
                if self.config.canonical && byte != upper {
                    return Err(StreamError::NonCanonical {
                        offset,
                        expected: upper,
                    });
                }
            }

            if byte == b'=' {
                if self.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                if state.first_padding_offset.is_none() {
                    let residue = state.data_count % 8;
                    let expected = match residue {
                        2 => 6,
                        4 => 4,
                        5 => 3,
                        7 => 1,
                        _ => return Err(StreamError::Padding { offset }),
                    };
                    state.first_padding_offset = Some(offset);
                    state.expected_padding = Some(expected);
                }

                state.padding_count += 1;
                if state.padding_count > state.expected_padding.unwrap_or(0) {
                    return Err(StreamError::Padding { offset });
                }
            } else {
                if state.first_padding_offset.is_some() {
                    return Err(StreamError::Padding { offset });
                }

                let (_, value) = recognized.unwrap();
                state.data_count += 1;
                state.last_value = value as u8;
                state.last_data_offset = Some(offset);
                state.bit_buffer = (state.bit_buffer << 5) | value as u8;
                state.bit_count += 5;

                while state.bit_count >= 8 {
                    state.bit_count -= 8;
                    output.push((state.bit_buffer >> state.bit_count) as u8);
                    if state.bit_count == 0 {
                        state.bit_buffer = 0;
                    } else {
                        state.bit_buffer &= (1u16.checked_shl(state.bit_count as u32).unwrap_or(0) - 1) as u8;
                    }
                }
            }

            state.position += 1;
        }

        self.state = state;
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.state.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.state.data_count % 8;
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => {
                return Err(StreamError::InvalidLength {
                    offset: self.state.position,
                });
            }
        };

        if self.config.padding == Padding::Required
            && self.state.padding_count != expected_padding
        {
            return Err(StreamError::Padding {
                offset: self
                    .state
                    .first_padding_offset
                    .unwrap_or(self.state.position),
            });
        }

        let unused_bits = match residue {
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };

        if unused_bits != 0 && (self.state.last_value & ((1 << unused_bits) - 1)) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.state.last_data_offset.unwrap_or(self.state.position),
            });
        }

        self.state.closed = true;
        Ok(Vec::new())
    }
}
