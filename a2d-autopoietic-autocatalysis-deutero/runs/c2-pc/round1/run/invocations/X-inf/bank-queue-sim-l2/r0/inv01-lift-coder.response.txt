#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| b.arrival.cmp(&a.arrival).then_with(|| a.id.cmp(&b.id)));

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<Customer> = Vec::new();
    let mut arrival_index = 0usize;
    let mut completed = 0usize;
    let total = customers.len();

    loop {
        if completed == total && states.iter().all(|state| state.opened) {
            break;
        }

        let mut next_time = u64::MAX;
        let mut has_next = false;

        for state in &states {
            if let Some(finish) = state.busy_until {
                if !has_next || finish < next_time {
                    next_time = finish;
                    has_next = true;
                }
            }
            if !state.opened && (!has_next || state.server.opens < next_time) {
                next_time = state.server.opens;
                has_next = true;
            }
        }
        if arrival_index < arrivals.len() {
            let arrival = arrivals[arrival_index].arrival;
            if !has_next || arrival < next_time {
                next_time = arrival;
                has_next = true;
            }
        }

        if !has_next {
            break;
        }

        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (state.busy_until == Some(next_time)).then_some(index)
            })
            .collect();
        finishing.sort_by_key(|&index| states[index].server.id);
        for index in finishing {
            let state = &mut states[index];
            let customer = match events.last() {
                Some(Event::Start {
                    customer,
                    server,
                    finish,
                    ..
                }) if *server == state.server.id && *finish == next_time => *customer,
                _ => {
                    // A busy server's scheduled customer is tracked by its most recent Start.
                    events.iter().rev().find_map(|event| match event {
                        Event::Start {
                            customer,
                            server,
                            finish,
                            ..
                        } if *server == state.server.id && *finish == next_time => Some(*customer),
                        _ => None,
                    }).expect("scheduled finish must have a matching start")
                }
            };
            events.push(Event::Finish {
                time: next_time,
                customer,
                server: state.server.id,
            });
            state.busy_until = None;
            state.idle_since = next_time;
            completed += 1;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.server.opens == next_time).then_some(index)
            })
            .collect();
        opening.sort_by(|&a, &b| states[b].server.id.cmp(&states[a].server.id));
        for index in opening {
            states[index].opened = true;
            states[index].idle_since = next_time;
            events.push(Event::Open {
                time: next_time,
                server: states[index].server.id,
            });
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == next_time {
            let customer = arrivals.pop().expect("arrival index is in range");
            arrival_index += 1;
            events.push(Event::Arrive {
                time: next_time,
                customer: customer.id,
            });
            waiting.push(customer);
        }

        loop {
            if waiting.is_empty() {
                break;
            }
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };
            let customer = waiting.remove(0);
            let server = &states[server_index].server;

            let finish = customer
                .service
                .checked_add(next_time)
                .and_then(|time| time.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: next_time,
                })?;

            events.push(Event::Start {
                time: next_time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == next_time {
                events.push(Event::Finish {
                    time: next_time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = next_time;
                completed += 1;
            } else {
                states[server_index].busy_until = Some(finish);
            }
        }
    }

    Ok(events)
}
