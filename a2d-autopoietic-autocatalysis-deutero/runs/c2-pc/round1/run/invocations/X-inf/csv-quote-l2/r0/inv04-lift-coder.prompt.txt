enum State {
    FieldStart,
    Unquoted,
    Quoted,
    AfterQuote,
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut start = 0;
    if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        start = 3;
    }
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut state = State::FieldStart;
    let mut opening_quote = 0usize;
    let mut record_index = 0usize;
    let mut i = start;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != State::Quoted && byte == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
            row.push(String::from_utf8(field).expect("input is valid UTF-8"));
            field = Vec::new();
            records.push(row);
            row = Vec::new();
            state = State::FieldStart;
            record_index += 1;
            i += 2;
            continue;
        }

        if state != State::Quoted && byte == b'\n' {
            row.push(String::from_utf8(field).expect("input is valid UTF-8"));
            field = Vec::new();
            records.push(row);
            row = Vec::new();
            state = State::FieldStart;
            record_index += 1;
            i += 1;
            continue;
        }

        match state {
            State::FieldStart => {
                if byte == delimiter {
                    row.push(String::new());
                } else if byte == quote {
                    opening_quote = i;
                    state = State::Quoted;
                } else {
                    field.push(byte);
                    state = State::Unquoted;
                }
            }
            State::Unquoted => {
                if byte == delimiter {
                    row.push(String::from_utf8(field).expect("input is valid UTF-8"));
                    field = Vec::new();
                    state = State::FieldStart;
                } else if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else {
                    field.push(byte);
                }
            }
            State::Quoted => {
                if byte == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote);
                        i += 1;
                    } else {
                        state = State::AfterQuote;
                    }
                } else {
                    field.push(byte);
                }
            }
            State::AfterQuote => {
                if byte == delimiter {
                    row.push(String::from_utf8(field).expect("input is valid UTF-8"));
                    field = Vec::new();
                    state = State::FieldStart;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
        i += 1;
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if !row.is_empty() || !field.is_empty() || state != State::FieldStart {
        row.push(String::from_utf8(field).expect("input is valid UTF-8"));
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    let Some(header) = records.first().cloned() else {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    };

    if config.duplicates == DuplicatePolicy::Reject {
        for second in 0..header.len() {
            if let Some(first) = header[..second]
                .iter()
                .position(|value| value == &header[second])
            {
                return Err(CsvError::DuplicateHeader {
                    name: header[second].clone(),
                    first,
                    second,
                });
            }
        }
    }

    let mut rows = Vec::with_capacity(records.len().saturating_sub(1));
    for (record_index, row) in records.into_iter().enumerate().skip(1) {
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: record_index,
                expected: header.len(),
                actual: row.len(),
            });
        }
        rows.push(row);
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }

        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
