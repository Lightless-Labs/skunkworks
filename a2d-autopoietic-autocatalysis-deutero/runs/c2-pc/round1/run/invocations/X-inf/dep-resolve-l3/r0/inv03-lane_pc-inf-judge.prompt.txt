fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints.to_vec())
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(s: &str) -> String {
    let mut out = String::new();
    for byte in s.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-') {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((byte & 15) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

fn permits(c: Constraint, version: Version) -> bool {
    match c {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn constraints_satisfied(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| permits(*c, version))
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    seen.clear();
    for lock in locks {
        if !seen.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

type Assignments = std::collections::BTreeMap<String, Version>;
type Active = std::collections::BTreeMap<String, Vec<Constraint>>;

fn active_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    assigned: &Assignments,
) -> Active {
    let mut active = Active::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in assigned {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical_constraints(std::mem::take(constraints));
    }
    active
}

struct SearchResult {
    best: Option<Assignments>,
    best_cost: usize,
    first_error: Option<ResolveError>,
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    result: SearchResult,
    catalog_names: Vec<String>,
}

impl Search<'_> {
    fn record_error(&mut self, error: ResolveError) {
        if self.result.first_error.is_none() {
            self.result.first_error = Some(error);
        }
    }

    fn solution_cost(&self, assigned: &Assignments) -> usize {
        self.locks
            .iter()
            .filter(|lock| assigned.get(&lock.package) != Some(&lock.version))
            .count()
    }

    fn better_tie(&self, candidate: &Assignments, current: &Assignments) -> bool {
        for name in &self.catalog_names {
            match (candidate.get(name), current.get(name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn inspect(&mut self, assigned: &mut Assignments) -> bool {
        let active = active_constraints(self.roots, self.catalog, assigned);

        let missing = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .min();
        if let Some(name) = missing {
            self.record_error(ResolveError::Missing {
                package: name.clone(),
            });
            return false;
        }

        for (name, version) in assigned.iter() {
            if let Some(constraints) = active.get(name) {
                if !constraints_satisfied(*version, constraints) {
                    self.record_error(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                    return false;
                }
            }
        }

        for (name, constraints) in &active {
            if assigned.contains_key(name) {
                continue;
            }
            let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !package
                .releases
                .iter()
                .any(|r| constraints_satisfied(r.version, constraints))
            {
                self.record_error(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
                return false;
            }
        }

        let next = active
            .keys()
            .find(|name| !assigned.contains_key(*name))
            .cloned();
        let Some(name) = next else {
            let cost = self.solution_cost(assigned);
            let replace = match &self.result.best {
                None => true,
                Some(current) if cost < self.result.best_cost => true,
                Some(current) if cost == self.result.best_cost => {
                    self.better_tie(assigned, current)
                }
                _ => false,
            };
            if replace {
                self.result.best = Some(assigned.clone());
                self.result.best_cost = cost;
            }
            return true;
        };

        let constraints = active.get(&name).unwrap();
        let package = self.catalog.iter().find(|p| p.name == name).unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .filter(|r| constraints_satisfied(r.version, constraints))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
                let version = candidates.remove(index);
                candidates.insert(0, version);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.record_error(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
                return true;
            }
            self.attempts += 1;
            assigned.insert(name.clone(), version);
            let _ = self.inspect(assigned);
            assigned.remove(&name);

            if matches!(self.result.first_error, Some(ResolveError::Limit { .. })) {
                return true;
            }
        }
        false
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        result: SearchResult {
            best: None,
            best_cost: usize::MAX,
            first_error: None,
        },
        catalog_names,
    };

    let mut assigned = Assignments::new();
    search.inspect(&mut assigned);

    if let Some(ResolveError::Limit { .. }) = search.result.first_error {
        return Err(search.result.first_error.unwrap());
    }
    let Some(best) = search.result.best else {
        return Err(search.result.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let active = active_constraints(roots, catalog, &best);
    let mut before = std::collections::BTreeMap::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }

    let mut names = std::collections::BTreeSet::new();
    names.extend(before.keys().cloned());
    names.extend(best.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let old = before.get(&package).copied();
        let new = best.get(&package).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old_version));
                if !exists {
                    ChangeReason::Unavailable
                } else if active
                    .get(&package)
                    .is_some_and(|cs| !constraints_satisfied(old_version, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    attempts: usize,
    max_attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_error: Option<ResolveError>,
    limit: bool,
}

impl Search<'_> {
    fn visit(&mut self, selected: &mut Vec<(String, Version)>) {
        if self.limit {
            return;
        }
        let active = active_constraints(self.roots, self.catalog, selected);

        for name in active.keys() {
            if !self.catalog.iter().any(|p| p.name == *name) {
                if self.first_error.is_none() {
                    self.first_error = Some(ResolveError::Missing { package: name.clone() });
                }
                return;
            }
        }

        for (name, version) in selected.iter() {
            let cs = canonical(active.get(name).cloned().unwrap_or_default());
            if cs.iter().any(|c| !matches(*version, *c)) {
                if self.first_error.is_none() {
                    self.first_error = Some(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs,
                    });
                }
                return;
            }
        }

        for (name, cs) in &active {
            if selected.iter().any(|(n, _)| n == name) {
                continue;
            }
            let pkg = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !pkg.releases.iter().any(|r| {
                cs.iter().all(|c| matches(r.version, *c))
            }) {
                if self.first_error.is_none() {
                    self.first_error = Some(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: canonical(cs.clone()),
                    });
                }
                return;
            }
        }

        let next = active.keys().find(|name| !selected.iter().any(|(n, _)| n == *name));
        let Some(name) = next.cloned() else {
            let mut solution = selected.clone();
            solution.sort_by(|a, b| a.0.cmp(&b.0));
            let cost = self.locks.iter().filter(|lock| {
                !solution.iter().any(|(n, v)| n == &lock.package && *v == lock.version)
            }).count();
            let names: std::collections::BTreeSet<String> =
                self.catalog.iter().map(|p| p.name.clone()).collect();
            let better = match &self.best {
                None => true,
                Some((best_cost, best_solution)) if cost < *best_cost => true,
                Some((best_cost, best_solution)) if cost == *best_cost => {
                    let map = |s: &[(String, Version)]| {
                        names.iter().map(|n| {
                            s.iter().find(|(x, _)| x == n).map(|(_, v)| *v)
                        }).collect::<Vec<_>>()
                    };
                    map(&solution) > map(best_solution)
                }
                _ => false,
            };
            if better {
                self.best = Some((cost, solution));
            }
            return;
        };

        let pkg = self.catalog.iter().find(|p| p.name == name).unwrap();
        let cs = &active[&name];
        let mut candidates: Vec<Version> = pkg.releases.iter()
            .filter(|r| cs.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(i) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limit = true;
                return;
            }
            self.attempts += 1;
            selected.push((name.clone(), version));
            self.visit(selected);
            selected.pop();
            if self.limit {
                return;
            }
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: r.package.clone(),
            });
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: p.name.clone(),
            });
        }
        for (j, release) in p.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for p in catalog {
        if !seen.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }

    for p in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    seen.clear();
    for lock in locks {
        if !seen.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: lock.package.clone() });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
        limit: false,
    };
    search.visit(&mut Vec::new());
    if search.limit {
        return Err(ResolveError::Limit { attempts: max_attempts });
    }
    let Some((_, solution)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected = solution.iter().map(|(package, version)| Selection {
        package: package.clone(),
        version: *version,
    }).collect::<Vec<_>>();

    let final_active = active_constraints(roots, catalog, &solution);
    let mut before = std::collections::BTreeMap::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let after: std::collections::BTreeMap<String, Version> = solution.iter().cloned().collect();
    let names: std::collections::BTreeSet<String> = before.keys().chain(after.keys()).cloned().collect();
    let mut changes = Vec::new();
    for package in names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog.iter().find(|p| p.name == package)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active.get(&package).cloned().unwrap_or_default()
                    .iter().any(|c| !matches(old, *c))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change { package, before: b, after: a, reason });
    }

    Ok(Resolution { selected, changes })
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    locks: &'a BTreeMap<String, Version>,
    all_names: Vec<String>,
    attempts: usize,
    max_attempts: usize,
    best: Option<(usize, Vec<Option<Version>>, SearchResult)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn inspect(
        &mut self,
        assignments: &BTreeMap<String, Version>,
    ) -> Result<(), ResolveError> {
        let mut active: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();

        // Roots are included in the caller's initial constraints through this
        // field-like setup encoded as an initial assignment-independent map.
        // The root list is stored separately on the search object by the caller.
        let _ = &mut active;
        self.visit(assignments, active)
    }

    fn visit(
        &mut self,
        assignments: &BTreeMap<String, Version>,
        mut active: BTreeMap<String, Vec<Constraint>>,
    ) -> Result<(), ResolveError> {
        // Reconstruct active requirements from roots and selected releases.
        for (package, cs) in self.root_constraints() {
            active.entry(package).or_default().extend(cs);
        }
        for (package, version) in assignments {
            if let Some(pkg) = self.catalog.get(package) {
                if let Some(release) = pkg.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for cs in active.values_mut() {
            canonicalize(cs);
        }

        if let Some(package) = active
            .keys()
            .filter(|p| !self.catalog.contains_key(*p))
            .next()
        {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }

        for (package, version) in assignments {
            if let Some(cs) = active.get(package) {
                if !satisfies(*version, cs) {
                    return Err(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: cs.clone(),
                    });
                }
            }
        }

        for package in active.keys() {
            if assignments.contains_key(package) {
                continue;
            }
            let cs = active.get(package).unwrap();
            let pkg = self.catalog.get(package).unwrap();
            if !pkg.releases.iter().any(|r| satisfies(r.version, cs)) {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: cs.clone(),
                });
            }
        }

        let next = active
            .keys()
            .find(|p| !assignments.contains_key(*p))
            .cloned();

        let Some(package) = next else {
            let constraints = active;
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| assignments.get(*p) != Some(*v))
                .count();
            let vector = self
                .all_names
                .iter()
                .map(|p| assignments.get(p).copied())
                .collect::<Vec<_>>();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_vector, _)) => {
                    cost < *best_cost || (cost == *best_cost && vector > *best_vector)
                }
            };
            if replace {
                self.best = Some((
                    cost,
                    vector,
                    SearchResult {
                        assignments: assignments.clone(),
                        constraints,
                    },
                ));
            }
            return Ok(());
        };

        let cs = active.get(&package).unwrap();
        let pkg = self.catalog.get(&package).unwrap();
        let mut candidates: Vec<Version> = pkg
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| satisfies(*v, cs))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.get(&package) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            let mut child = assignments.clone();
            child.insert(package.clone(), candidate);
            match self.visit(&child, BTreeMap::new()) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(error);
                    }
                }
            }
        }
        Ok(())
    }

    fn root_constraints(&self) -> BTreeMap<String, Vec<Constraint>> {
        ROOTS.with(|roots| roots.borrow().clone())
    }
}

thread_local! {
    static ROOTS: std::cell::RefCell<BTreeMap<String, Vec<Constraint>>> =
        const { std::cell::RefCell::new(BTreeMap::new()) };
}

fn validate_name(site: String, name: &str) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        validate_name(format!("root[{}]", i), &root.package)?;
    }
    for (i, package) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{}]", i), &package.name)?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                validate_name(
                    format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                    &dep.package,
                )?;
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen = BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        validate_name(format!("lock[{}]", i), &lock.package)?;
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let mut root_map: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();
    for root in roots {
        root_map
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for constraints in root_map.values_mut() {
        canonicalize(constraints);
    }

    ROOTS.with(|r| *r.borrow_mut() = root_map);

    let mut all_names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    all_names.sort();
    let mut search = Search {
        catalog: &catalog_map,
        locks: &lock_map,
        all_names,
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
    };
    let result = search.inspect(&BTreeMap::new());

    let chosen = match result {
        Err(error) => return Err(error),
        Ok(()) => match search.best {
            Some((_, _, result)) => result,
            None => {
                return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
                    package: String::new(),
                    requirements: Vec::new(),
                }))
            }
        },
    };

    let selected = chosen
        .assignments
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut names = BTreeSet::new();
    names.extend(lock_map.keys().cloned());
    names.extend(chosen.assignments.keys().cloned());
    let mut changes = Vec::new();
    for package in names {
        let before = lock_map.get(&package).copied();
        let after = chosen.assignments.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !chosen
                    .constraints
                    .get(&package)
                    .is_some_and(|cs| satisfies(old, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
