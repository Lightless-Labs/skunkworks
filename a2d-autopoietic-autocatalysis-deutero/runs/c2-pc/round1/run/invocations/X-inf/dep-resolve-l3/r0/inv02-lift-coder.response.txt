#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName { site: String, name: String },
    DuplicatePackage { package: String },
    DuplicateVersion { package: String, version: Version },
    DuplicateLock { package: String },
    Missing { package: String },
    NoMatch { package: String, requirements: Vec<Constraint> },
    Conflict { package: String, selected: Version, requirements: Vec<Constraint> },
    Limit { attempts: usize },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes.iter().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
        && bytes.last() != Some(&b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical(cs.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } =>
            format!("invalid-name|site={}|name={}", encode(site), encode(name)),
        ResolveError::DuplicatePackage { package } =>
            format!("duplicate-package|package={}", encode(package)),
        ResolveError::DuplicateVersion { package, version } =>
            format!("duplicate-version|package={}|version={}", encode(package), version_text(*version)),
        ResolveError::DuplicateLock { package } =>
            format!("duplicate-lock|package={}", encode(package)),
        ResolveError::Missing { package } =>
            format!("missing|package={}", encode(package)),
        ResolveError::NoMatch { package, requirements } =>
            format!("no-match|package={}|constraints={}", encode(package), constraints_text(requirements)),
        ResolveError::Conflict { package, selected, requirements } =>
            format!("conflict|package={}|selected={}|constraints={}", encode(package), version_text(*selected), constraints_text(requirements)),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x &&
                if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn active_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    selected: &[(String, Version)],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for r in roots {
        active.entry(r.package.clone()).or_default().push(r.constraint);
    }
    for (name, version) in selected {
        if let Some(pkg) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = pkg.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active.entry(dep.package.clone()).or_default().push(dep.constraint);
                }
            }
        }
    }
    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    attempts: usize,
    max_attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_error: Option<ResolveError>,
    limit: bool,
}

impl Search<'_> {
    fn visit(&mut self, selected: &mut Vec<(String, Version)>) {
        if self.limit {
            return;
        }
        let active = active_constraints(self.roots, self.catalog, selected);

        for name in active.keys() {
            if !self.catalog.iter().any(|p| p.name == *name) {
                if self.first_error.is_none() {
                    self.first_error = Some(ResolveError::Missing { package: name.clone() });
                }
                return;
            }
        }

        for (name, version) in selected.iter() {
            let cs = canonical(active.get(name).cloned().unwrap_or_default());
            if cs.iter().any(|c| !matches(*version, *c)) {
                if self.first_error.is_none() {
                    self.first_error = Some(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs,
                    });
                }
                return;
            }
        }

        for (name, cs) in &active {
            if selected.iter().any(|(n, _)| n == name) {
                continue;
            }
            let pkg = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !pkg.releases.iter().any(|r| {
                cs.iter().all(|c| matches(r.version, *c))
            }) {
                if self.first_error.is_none() {
                    self.first_error = Some(ResolveError::NoMatch {
                        package: name.clone(),
                        requirements: canonical(cs.clone()),
                    });
                }
                return;
            }
        }

        let next = active.keys().find(|name| !selected.iter().any(|(n, _)| n == *name));
        let Some(name) = next.cloned() else {
            let mut solution = selected.clone();
            solution.sort_by(|a, b| a.0.cmp(&b.0));
            let cost = self.locks.iter().filter(|lock| {
                !solution.iter().any(|(n, v)| n == &lock.package && *v == lock.version)
            }).count();
            let names: std::collections::BTreeSet<String> =
                self.catalog.iter().map(|p| p.name.clone()).collect();
            let better = match &self.best {
                None => true,
                Some((best_cost, best_solution)) if cost < *best_cost => true,
                Some((best_cost, best_solution)) if cost == *best_cost => {
                    let map = |s: &[(String, Version)]| {
                        names.iter().map(|n| {
                            s.iter().find(|(x, _)| x == n).map(|(_, v)| *v)
                        }).collect::<Vec<_>>()
                    };
                    map(&solution) > map(best_solution)
                }
                _ => false,
            };
            if better {
                self.best = Some((cost, solution));
            }
            return;
        };

        let pkg = self.catalog.iter().find(|p| p.name == name).unwrap();
        let cs = &active[&name];
        let mut candidates: Vec<Version> = pkg.releases.iter()
            .filter(|r| cs.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(i) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limit = true;
                return;
            }
            self.attempts += 1;
            selected.push((name.clone(), version));
            self.visit(selected);
            selected.pop();
            if self.limit {
                return;
            }
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: r.package.clone(),
            });
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: p.name.clone(),
            });
        }
        for (j, release) in p.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for p in catalog {
        if !seen.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }

    for p in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    seen.clear();
    for lock in locks {
        if !seen.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: lock.package.clone() });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        attempts: 0,
        max_attempts,
        best: None,
        first_error: None,
        limit: false,
    };
    search.visit(&mut Vec::new());
    if search.limit {
        return Err(ResolveError::Limit { attempts: max_attempts });
    }
    let Some((_, solution)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected = solution.iter().map(|(package, version)| Selection {
        package: package.clone(),
        version: *version,
    }).collect::<Vec<_>>();

    let final_active = active_constraints(roots, catalog, &solution);
    let mut before = std::collections::BTreeMap::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let after: std::collections::BTreeMap<String, Version> = solution.iter().cloned().collect();
    let names: std::collections::BTreeSet<String> = before.keys().chain(after.keys()).cloned().collect();
    let mut changes = Vec::new();
    for package in names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog.iter().find(|p| p.name == package)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active.get(&package).cloned().unwrap_or_default()
                    .iter().any(|c| !matches(old, *c))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change { package, before: b, after: a, reason });
    }

    Ok(Resolution { selected, changes })
}
