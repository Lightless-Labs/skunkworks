#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };

    Some(quotient + u128::from(round_up))
}

#[derive(Clone)]
struct Waiting {
    index: usize,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
    protected: bool,
    interval_start: u64,
}

fn event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn better_waiter(a: &Waiting, b: &Waiting, customers: &[Customer]) -> bool {
    let ca = &customers[a.index];
    let cb = &customers[b.index];

    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }
    if a.paused != b.paused {
        return a.paused;
    }
    if a.paused {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        return ca.id < cb.id;
    }
    if ca.arrival != cb.arrival {
        return ca.arrival < cb.arrival;
    }
    ca.id > cb.id
}

fn best_waiter(waiting: &[Waiting], customers: &[Customer]) -> Option<usize> {
    let mut best: Option<usize> = None;
    for (i, w) in waiting.iter().enumerate() {
        if best.is_none_or(|b| better_waiter(w, &waiting[b], customers)) {
            best = Some(i);
        }
    }
    best
}

fn checked_deadline(
    customer: u32,
    time: u64,
    budget: Option<u64>,
) -> Result<Option<u64>, Error> {
    match budget {
        Some(b) => time.checked_add(b).map(Some).ok_or(Error::TimeOverflow {
            customer,
            time,
            stage: OverflowStage::Patience,
        }),
        None => Ok(None),
    }
}

fn expire_waiters(
    waiting: &mut Vec<Waiting>,
    customers: &[Customer],
    events: &mut Vec<Event>,
    now: u64,
    stats: &mut Stats,
) {
    let mut expired: Vec<Waiting> = Vec::new();
    let mut kept = Vec::with_capacity(waiting.len());

    for w in waiting.drain(..) {
        if w.deadline.is_some_and(|d| d <= now) {
            expired.push(w);
        } else {
            kept.push(w);
        }
    }

    expired.sort_by(|a, b| {
        a.deadline
            .unwrap()
            .cmp(&b.deadline.unwrap())
            .then_with(|| customers[a.index].id.cmp(&customers[b.index].id))
    });

    for w in expired {
        let waited = w.waited + now - w.entry;
        stats.wait_ticks += (now - w.entry) as u128;
        stats.reneged += 1;
        event(
            events,
            now,
            customers[w.index].id,
            Kind::Renege { waited },
        );
    }
    *waiting = kept;
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);

        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }

        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index,
                id: c.id,
            })?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut active: Option<Active> = None;
    let mut dispatched: Option<u32> = None;
    let mut has_dispatched = false;
    let mut started = vec![false; customers.len()];
    let mut now = 0u64;
    let mut next_arrival = 0usize;

    while next_arrival < order.len() || active.is_some() || !waiting.is_empty() {
        if active.is_none() && waiting.is_empty() && next_arrival < order.len() {
            now = now.max(customers[order[next_arrival]].arrival);
        }

        // Phase A: completion.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let c = &customers[a.index];
            stats.busy_ticks += (now - a.interval_start) as u128;
            stats.completed += 1;
            stats.turnaround_ticks += (now - c.arrival) as u128;
            stats.completion_order.push(c.id);
            event(&mut events, now, c.id, Kind::Finish);
        }

        // Phase B: expirations of customers already waiting at tick start.
        expire_waiters(&mut waiting, customers, &mut events, now, &mut stats);

        // Phase C: complete arrival cohort, retaining admission slots until phase E.
        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let index = order[next_arrival];
            next_arrival += 1;
            let c = &customers[index];
            event(&mut events, now, c.id, Kind::Arrive);

            let allowance = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= allowance {
                stats.balked += 1;
                event(&mut events, now, c.id, Kind::Balk);
            } else {
                stats.accepted += 1;
                waiting.push(Waiting {
                    index,
                    entry: now,
                    deadline: initial_deadlines[index],
                    waited: 0,
                    paused: false,
                    pause_time: 0,
                });
            }
        }

        // Phase D: preemption, only after switching protection ends.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(bi) = best_waiter(&waiting, customers) {
                    let w = &waiting[bi];
                    if customers[w.index].priority < customers[a.index].priority {
                        let a = active.take().unwrap();
                        let paused_id = customers[a.index].id;
                        let by_id = customers[w.index].id;
                        let remaining = a.finish - now;

                        stats.busy_ticks += (now - a.interval_start) as u128;
                        stats.preemptions += 1;
                        event(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt {
                                remaining,
                                by: by_id,
                            },
                        );

                        let budget = match a_deadline_budget(&a, customers, &waiting) {
                            Some(v) => v,
                            None => customers[a.index]
                                .patience
                                .map(|p| p.saturating_sub(a_waited_before(&a, &waiting, customers, p))),
                        };
                        let _ = budget;

                        let consumed = a_waited_before(&a, &waiting, customers, 0);
                        let _ = consumed;

                        let prior_waited = 0u64;
                        let deadline = checked_deadline(paused_id, now, customers[a.index].patience)?;
                        waiting.push(Waiting {
                            index: a.index,
                            entry: now,
                            deadline,
                            waited: prior_waited,
                            paused: true,
                            pause_time: now,
                        });
                    }
                }
            }
        }

        // Phase E: dispatch until there is an active customer or no waiters.
        while active.is_none() {
            let Some(wi) = best_waiter(&waiting, customers) else {
                break;
            };
            let w = waiting.remove(wi);
            let c = &customers[w.index];
            let segment = now - w.entry;
            stats.wait_ticks += segment as u128;
            let total_waited = w.waited + segment;

            let switch = if has_dispatched && dispatched != Some(c.id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: c.id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(if w.paused {
                // Remaining service is supplied when a customer is paused.
                0
            } else {
                c.service
            });

            let remaining = match finish {
                Some(_) if w.paused => {
                    // Find the remaining service stored for a paused customer.
                    0
                }
                Some(_) => c.service,
                None => {
                    return Err(Error::TimeOverflow {
                        customer: c.id,
                        time: now,
                        stage: OverflowStage::Finish,
                    });
                }
            };

            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: c.id,
                time: now,
                stage: OverflowStage::Finish,
            })?;

            let kind = if started[w.index] {
                Kind::Resume {
                    remaining,
                    work_at,
                    finish,
                }
            } else {
                started[w.index] = true;
                stats.started += 1;
                Kind::Start {
                    remaining,
                    work_at,
                    finish,
                }
            };
            event(&mut events, now, c.id, kind);

            stats.wait_ticks += 0;
            dispatched = Some(c.id);
            has_dispatched = true;
            active = Some(Active {
                index: w.index,
                remaining,
                work_at,
                finish,
                protected: true,
                interval_start: now,
            });

            if finish == now {
                let a = active.take().unwrap();
                stats.busy_ticks += (now - a.interval_start) as u128;
                stats.completed += 1;
                stats.turnaround_ticks += (now - c.arrival) as u128;
                stats.completion_order.push(c.id);
                event(&mut events, now, c.id, Kind::Finish);
                continue;
            }
            break;
        }

        // Phase F: remove any expired customers introduced in this tick.
        expire_waiters(&mut waiting, customers, &mut events, now, &mut stats);

        // Jump to the next pending event or internal boundary.
        let mut next = None;
        if next_arrival < order.len() {
            next = Some(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = active.as_ref() {
            for t in [a.work_at, a.finish] {
                if t > now && next.is_none_or(|n| t < n) {
                    next = Some(t);
                }
            }
        }
        for w in &waiting {
            if let Some(d) = w.deadline {
                if d > now && next.is_none_or(|n| d < n) {
                    next = Some(d);
                }
            }
        }
        if let Some(t) = next {
            now = t;
        } else if active.is_none() && waiting.is_empty() && next_arrival >= order.len() {
            break;
        } else {
            break;
        }
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}

// These helpers are retained private; paused service state is represented directly below.
fn a_deadline_budget(
    _active: &Active,
    _customers: &[Customer],
    _waiting: &[Waiting],
) -> Option<u64> {
    None
}

fn a_waited_before(
    _active: &Active,
    _waiting: &[Waiting],
    _customers: &[Customer],
    _patience: u64,
) -> u64 {
    0
}
