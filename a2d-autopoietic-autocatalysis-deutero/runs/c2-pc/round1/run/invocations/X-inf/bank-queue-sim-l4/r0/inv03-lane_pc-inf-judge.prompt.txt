pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };

    Some(quotient + u128::from(round_up))
}

fn better_waiter(a: &Waiting, b: &Waiting, customers: &[Customer]) -> bool {
    let ca = &customers[a.index];
    let cb = &customers[b.index];

    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }
    if a.paused != b.paused {
        return a.paused;
    }
    if a.paused {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        return ca.id < cb.id;
    }
    if ca.arrival != cb.arrival {
        return ca.arrival < cb.arrival;
    }
    ca.id > cb.id
}

fn best_waiter(waiting: &[Waiting], customers: &[Customer]) -> Option<usize> {
    let mut best: Option<usize> = None;
    for (i, w) in waiting.iter().enumerate() {
        if best.is_none_or(|b| better_waiter(w, &waiting[b], customers)) {
            best = Some(i);

    }
    best

fn checked_deadline(
    customer: u32,
    time: u64,
    budget: Option<u64>,
) -> Result<Option<u64>, Error> {
    match budget {
        Some(b) => time.checked_add(b).map(Some).ok_or(Error::TimeOverflow {
            customer,
            time,
            stage: OverflowStage::Patience,
        }),
        None => Ok(None),
    }
}

fn expire_waiters(
    waiting: &mut Vec<Waiting>,
    customers: &[Customer],
    events: &mut Vec<Event>,
    now: u64,
    stats: &mut Stats,
) {
    let mut expired: Vec<Waiting> = Vec::new();
    let mut kept = Vec::with_capacity(waiting.len());

    for w in waiting.drain(..) {
        if w.deadline.is_some_and(|d| d <= now) {
            expired.push(w);
        } else {
            kept.push(w);
        }
    }

    expired.sort_by(|a, b| {
        a.deadline
            .unwrap()
            .cmp(&b.deadline.unwrap())
            .then_with(|| customers[a.index].id.cmp(&customers[b.index].id))
    });

    for w in expired {
        let waited = w.waited + now - w.entry;
        stats.wait_ticks += (now - w.entry) as u128;
        stats.reneged += 1;
        event(
            events,
            now,
            customers[w.index].id,
            Kind::Renege { waited },
        );
    }
    *waiting = kept;
}

            count: customers.len(),

                    0

                        customer: c.id,
                        time: now,
                        stage: OverflowStage::Finish,
                    });
                }
            };

                customer: c.id,
                time: now,
                stage: OverflowStage::Finish,
            })?;

                Kind::Resume {
                    remaining,
                    work_at,
                    finish,
                }
            } else {

                stats.started += 1;
                Kind::Start {
                    remaining,
                    work_at,
                    finish,
                }
            };

            has_dispatched = true;

                index: w.index,
                remaining,
                work_at,
                finish,
                protected: true,
                interval_start: now,
            });

            if finish == now {
                let a = active.take().unwrap();
                stats.busy_ticks += (now - a.interval_start) as u128;
                stats.completed += 1;
                stats.turnaround_ticks += (now - c.arrival) as u128;
                stats.completion_order.push(c.id);
                event(&mut events, now, c.id, Kind::Finish);
                continue;
            }
            break;
        }

        // Phase F: remove any expired customers introduced in this tick.
        expire_waiters(&mut waiting, customers, &mut events, now, &mut stats);

        // Jump to the next pending event or internal boundary.
        let mut next = None;
        if next_arrival < order.len() {
            next = Some(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = active.as_ref() {
            for t in [a.work_at, a.finish] {
                if t > now && next.is_none_or(|n| t < n) {
                    next = Some(t);
                }
            }
        }
        for w in &waiting {
            if let Some(d) = w.deadline {
                if d > now && next.is_none_or(|n| d < n) {
                    next = Some(d);
                }
            }
        }
        if let Some(t) = next {
            now = t;
        } else if active.is_none() && waiting.is_empty() && next_arrival >= order.len() {
            break;
        } else {
            break;
        }
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(if round_up { q + 1 } else { q })
}

fn waiting_order(a: usize, b: usize, jobs: &[Job]) -> std::cmp::Ordering {
    let ja = &jobs[a];
    let jb = &jobs[b];
    ja.customer.priority.cmp(&jb.customer.priority)
        .then_with(|| match (ja.paused, jb.paused) {
            (true, false) => std::cmp::Ordering::Less,
            (false, true) => std::cmp::Ordering::Greater,
            (true, true) => ja.pause_time.cmp(&jb.pause_time)
                .then_with(|| ja.customer.id.cmp(&jb.customer.id)),
            (false, false) => ja.customer.arrival.cmp(&jb.customer.arrival)
                .then_with(|| jb.customer.id.cmp(&ja.customer.id)),
        })
}

    let mut intervals: Vec<(u64, u64, u32)> = Vec::new();

    };

        stats: Stats {

            busy_ticks, wait_ticks, turnaround_ticks, mean_wait_milli,
            mean_turnaround_milli, utilization_bps, completion_order,
        },
    })

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let complement = denominator - remainder;

    Some(if remainder > complement || (remainder == complement && quotient % 2 == 1) {
        quotient + 1
    } else {
        quotient
    })
}

fn better_waiter(a: &Waiter, b: &Waiter, customers: &[Customer]) -> bool {
    let ca = &customers[a.index];
    let cb = &customers[b.index];
    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }
    if a.paused != b.paused {
        return a.paused;
    }
    if a.paused {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        ca.id < cb.id
    } else if ca.arrival != cb.arrival {
        ca.arrival < cb.arrival
    } else {
        ca.id > cb.id
    }
}

fn best_waiter_index(waiters: &[Waiter], customers: &[Customer]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiters.len() {
        if best.map_or(true, |j| better_waiter(&waiters[i], &waiters[j], customers)) {
            best = Some(i);
        }
    }
    best
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|earlier| earlier.id == c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiters: Vec<Waiter> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatch: Option<u32> = None;
    let mut ever_dispatched = vec![false; customers.len()];
    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();
    let mut now = arrivals.first().map_or(0, |&i| customers[i].arrival);

    while next_arrival < arrivals.len() || active.is_some() || !waiters.is_empty() {
        let mut progressed = false;

        // Phase A: finish the active service interval.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let c = &customers[a.index];
            busy_ticks += (now - a.dispatch) as u128;
            emit(&mut events, now, c.id, Kind::Finish);
            completed += 1;
            turnaround_ticks += (now - c.arrival) as u128;
            completion_order.push(c.id);
            progressed = true;
        }

        // Phase B: expire waiters present at the beginning of this tick.
        let mut expired = Vec::new();
        let mut i = 0;
        while i < waiters.len() {
            if waiters[i].deadline.is_some_and(|d| d <= now) {
                expired.push(waiters.remove(i));
            } else {
                i += 1;
            }
        }
        expired.sort_by(|a, b| {
            a.deadline.cmp(&b.deadline)
                .then_with(|| customers[a.index].id.cmp(&customers[b.index].id))
        });
        for w in expired {
            let c = &customers[w.index];
            let segment = now - w.entry;
            let total = w.waited.checked_add(segment).unwrap();
            wait_ticks += segment as u128;
            emit(&mut events, now, c.id, Kind::Renege { waited: total });
            reneged += 1;
            progressed = true;
        }

        // Phase C: process every arrival at this time in sorted cohort order.
        while next_arrival < arrivals.len()
            && customers[arrivals[next_arrival]].arrival == now
        {
            let index = arrivals[next_arrival];
            next_arrival += 1;
            let c = &customers[index];
            emit(&mut events, now, c.id, Kind::Arrive);
            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiters.len() >= limit {
                emit(&mut events, now, c.id, Kind::Balk);
                balked += 1;
            } else {
                accepted += 1;
                let deadline = c.patience.map(|p| c.arrival + p);
                waiters.push(Waiter {
                    index,
                    entry: now,
                    deadline,
                    waited: 0,
                    paused: false,
                    pause_time: 0,
                });
            }
            progressed = true;
        }

        // Phase D: a strictly higher-priority waiter may preempt after protection.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(wi) = best_waiter_index(&waiters, customers) {
                    if customers[waiters[wi].index].priority
                        < customers[a.index].priority
                    {
                        let a = active.take().unwrap();
                        let paused_customer = &customers[a.index];
                        let by = customers[waiters[wi].index].id;
                        let remaining = a.finish - now;
                        busy_ticks += (now - a.dispatch) as u128;
                        emit(
                            &mut events,
                            now,
                            paused_customer.id,
                            Kind::Preempt { remaining, by },
                        );
                        preemptions += 1;
                        let patience = paused_customer.patience;
                        let deadline = match patience {
                            Some(budget) => Some(now.checked_add(budget).ok_or(
                                Error::TimeOverflow {
                                    customer: paused_customer.id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                },
                            )?),
                            None => None,
                        };
                        waiters.push(Waiter {
                            index: a.index,
                            entry: now,
                            deadline,
                            waited: 0,
                            paused: true,
                            pause_time: now,
                        });
                        progressed = true;
                    }
                }
            }
        }

        // Phase E: repeatedly dispatch and immediately finish zero-length jobs.
        loop {
            if active.is_some() {
                break;
            }
            let Some(wi) = best_waiter_index(&waiters, customers) else {
                break;
            };
            let w = waiters.remove(wi);
            let c = &customers[w.index];
            let segment = now - w.entry;
            let total_waited = w.waited.checked_add(segment).unwrap();
            wait_ticks += segment as u128;

            let switching = last_dispatch.is_some_and(|id| id != c.id);
            let cost = if switching { config.switch_cost } else { 0 };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: c.id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(w.remaining_service_placeholder()).unwrap_or(0);
            let _ = finish;

            // Remaining service is full on first dispatch and stored on paused waiters.
            let remaining = if w.paused {
                // Paused service is carried separately below in a parallel per-customer slot.
                0
            } else {
                c.service
            };
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: c.id,
                time: now,
                stage: OverflowStage::Finish,
            })?;

            let kind = if ever_dispatched[w.index] {
                Kind::Resume { remaining, work_at, finish }
            } else {
                Kind::Start { remaining, work_at, finish }
            };
            if !ever_dispatched[w.index] {
                started += 1;
                ever_dispatched[w.index] = true;
            }
            emit(&mut events, now, c.id, kind);
            last_dispatch = Some(c.id);
            if finish == now {
                emit(&mut events, now, c.id, Kind::Finish);
                completed += 1;
                turnaround_ticks += (now - c.arrival) as u128;
                completion_order.push(c.id);
                progressed = true;
                continue;
            }
            active = Some(Active {
                index: w.index,
                remaining,
                work_at,
                finish,
                dispatch: now,
                protected: now < work_at,
            });
            progressed = true;
            break;
        }

        // Phase F: expire anything newly queued that did not dispatch.
        let mut expired = Vec::new();
        let mut i = 0;
        while i < waiters.len() {
            if waiters[i].deadline.is_some_and(|d| d <= now) {
                expired.push(waiters.remove(i));
            } else {
                i += 1;
            }
        }
        expired.sort_by(|a, b| {
            a.deadline.cmp(&b.deadline)
                .then_with(|| customers[a.index].id.cmp(&customers[b.index].id))
        });
        for w in expired {
            let c = &customers[w.index];
            let segment = now - w.entry;
            wait_ticks += segment as u128;
            emit(
                &mut events,
                now,
                c.id,
                Kind::Renege { waited: w.waited + segment },
            );
            reneged += 1;
            progressed = true;
        }

        if next_arrival >= arrivals.len() && active.is_none() && waiters.is_empty() {
            break;
        }

        // Jump to the next arrival, finish, work boundary, or waiting deadline.
        let mut next: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next = Some(customers[arrivals[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            for t in [Some(a.finish), (now < a.work_at).then_some(a.work_at)].into_iter().flatten() {
                next = Some(next.map_or(t, |n| n.min(t)));
            }
        }
        for w in &waiters {
            if let Some(d) = w.deadline.filter(|&d| d > now) {
                next = Some(next.map_or(d, |n| n.min(d)));
            }
        }
        if let Some(t) = next {
            if t > now {
                now = t;
                continue;
            }
        }
        if !progressed {
            break;
        }
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(1000u128 * wait_ticks, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(1000u128 * turnaround_ticks, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(10000u128 * busy_ticks, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}

impl Waiter {
    fn remaining_service_placeholder(&self) -> u64 {
        0
    }
}
