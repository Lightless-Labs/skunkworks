#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

fn validate_config(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut i = start;
    let mut record = 0usize;
    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut field_started = false;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if in_quotes {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                    continue;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                    continue;
                }
            }
            field.push(b as char);
            i += 1;
            continue;
        }

        if b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr { record, byte: i });
            }
        }

        if after_quote {
            if b == delimiter {
                fields.push(std::mem::take(&mut field));
                field_started = false;
                after_quote = false;
                record_started = true;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                field_started = false;
                after_quote = false;
                record_started = false;
                record += 1;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record, byte: i });
        }

        if b == delimiter {
            fields.push(std::mem::take(&mut field));
            field_started = false;
            record_started = true;
            i += 1;
            continue;
        }
        if b == b'\n' || b == b'\r' {
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            field_started = false;
            record_started = false;
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
            continue;
        }
        if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
            in_quotes = true;
            quote_start = i;
            field_started = true;
            record_started = true;
            i += 1;
            continue;
        }

        field.push(b as char);
        field_started = true;
        record_started = true;
        i += 1;
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if record_started || field_started || after_quote || !fields.is_empty() {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config.delimiter, config.quote).map_err(|component| {
        CsvError::InvalidConfig { component }
    })?;

    let mut records = parse_with_dialect(input, config.delimiter, config.quote)?;
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(ref header) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in header.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(s, _)| *s == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name, second));
            }
        }

        for (i, row) in records.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = build_table(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len() && start == 0;
    }
    (b[start].is_ascii_digit() && b[start] != b'0')
        && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start >= b.len() {
        return false;
    }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    if dot + 3 != b.len() {
        return false;
    }
    if !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    let integral = &b[start..dot];
    if integral.is_empty() {
        return false;
    }
    if integral == b"0" {
        start == 0
    } else {
        integral[0].is_ascii_digit()
            && integral[0] != b'0'
            && integral[1..].iter().all(u8::is_ascii_digit)
    }
}

fn evaluate_cell(s: &str, kind: ColumnType, nullable: bool) -> Result<Value, CellErrorKind> {
    if s.is_empty() {
        if nullable {
            return Ok(Value::Null);
        }
        if kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(s) {
                return Err(CellErrorKind::InvalidInteger);
            }
            s.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) {
                return Err(CellErrorKind::InvalidDecimal);
            }
            let negative = s.starts_with('-');
            let body = if negative { &s[1..] } else { s };
            let (whole, fraction) = body.split_once('.').unwrap();
            let magnitude = whole
                .parse::<i128>()
                .unwrap()
                .checked_mul(100)
                .and_then(|n| n.checked_add(fraction.parse::<i128>().unwrap()));
            let Some(magnitude) = magnitude else {
                return Err(CellErrorKind::DecimalOverflow);
            };
            let signed = if negative { -magnitude } else { magnitude };
            i64::try_from(signed)
                .map(Value::Decimal)
                .map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let Some(column) = resolved else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                if policy == ErrorPolicy::Stop {
                    result.rows.push(output);
                    break 'rows;
                }
                if policy == ErrorPolicy::FirstPerRow {
                    break;
                }
                continue;
            };

            let Some(cell) = row.get(column) else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column),
                    kind: CellErrorKind::MissingCell,
                });
                if policy == ErrorPolicy::Stop {
                    result.rows.push(output);
                    break 'rows;
                }
                if policy == ErrorPolicy::FirstPerRow {
                    break;
                }
                continue;
            };

            match evaluate_cell(cell, spec.kind, spec.nullable) {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column),
                        kind,
                    });
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(output);
                        break 'rows;
                    }
                    if policy == ErrorPolicy::FirstPerRow {
                        break;
                    }
                }
            }
        }
        result.rows.push(output);
    }

    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {

    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn duplicate_header(header: &[String]) -> Option<WriteError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(s, _)| *s == name) {
            return Some(WriteError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn write_record(
    output: &mut String,
    row: &[String],
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
) {
    for (i, field) in row.iter().enumerate() {
        if i > 0 {
            output.push(delimiter as char);
        }
        let bytes = field.as_bytes();
        let starts_or_ends_space_tab = bytes
            .first()
            .is_some_and(|b| *b == b' ' || *b == b'\t')
            || bytes
                .last()
                .is_some_and(|b| *b == b' ' || *b == b'\t');
        let begins_bom = first_field_first_record
            && i == 0
            && field.starts_with('\u{feff}');
        let sole_empty = row.len() == 1 && field.is_empty();
        let must_quote = field.as_bytes().contains(&delimiter)
            || field.as_bytes().contains(&quote)
            || field.contains('\r')
            || field.contains('\n')
            || starts_or_ends_space_tab
            || begins_bom
            || sole_empty;

        if must_quote {
            output.push(quote as char);
            for &b in bytes {
                if b == quote {
                    output.push(quote as char);
                }
                output.push(b as char);
            }
            output.push(quote as char);
        } else {
            output.push_str(field);
        }
    }
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_config(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header);
    }
    for row in &table.rows {
        original_records.push(row);
    }

    for (record, row) in original_records.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in original_records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let (Some(header), DuplicatePolicy::Reject) = (&table.header, table.duplicates) {
            if let Some(error) = duplicate_header(header) {
                return Err(error);
            }
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                output_records.push(header.clone());
            }
            output_records.extend(table.rows.iter().cloned());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut chosen: Vec<usize> = Vec::new();
                for (i, name) in header.iter().enumerate() {
                    match table.duplicates {
                        DuplicatePolicy::KeepLast => {
                            if !header[i + 1..].iter().any(|other| other == name) {
                                chosen.push(i);
                            }
                        }
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            if !header[..i].iter().any(|other| other == name) {
                                chosen.push(i);
                            }
                        }
                    }
                }
                chosen.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));
                output_records.push(chosen.iter().map(|&i| header[i].clone()).collect());
                let mut rows: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| chosen.iter().map(|&i| row[i].clone()).collect())
                    .collect();
                rows.sort_by(|a, b| {
                    a.iter()
                        .map(String::as_bytes)
                        .cmp(b.iter().map(String::as_bytes))
                });
                output_records.extend(rows);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| {
                    a.iter()
                        .map(String::as_bytes)
                        .cmp(b.iter().map(String::as_bytes))
                });
                output_records = rows;
            }
        }
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let ending = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };

    for (record_index, row) in output_records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(ending);
        }
        write_record(
            &mut output,
            row,
            config.delimiter,
            config.quote,
            record_index == 0,
        );
    }

    if canonical || config.final_terminator {
        output.push_str(ending);
    }

    Ok(output)
}
