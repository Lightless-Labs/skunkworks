#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    paused_at: Option<u64>,
    started: bool,
}

#[derive(Clone)]
struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
    started: bool,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (Some(at_a), Some(at_b)) => at_a
                .cmp(&at_b)
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;

    while finished_count < customers.len() {
        let next_tick = {
            let mut tick: Option<u64> = None;
            if let Some(customer) = arrivals.get(next_arrival) {
                tick = Some(customer.arrival);
            }
            if let Some(active) = &running {
                tick = Some(tick.map_or(active.finish, |t| t.min(active.finish)));
                if active.work_at > now {
                    tick = Some(tick.map_or(active.work_at, |t| t.min(active.work_at)));
                }
            }
            tick
        };

        let Some(tick) = next_tick else {
            break;
        };
        now = tick;

        if running.as_ref().is_some_and(|active| active.finish == now) {
            let active = running.take().unwrap();
            events.push(Event {
                time: now,
                customer: active.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                paused_at: None,
                started: false,
            });
            next_arrival += 1;
        }

        if let Some(active) = &running {
            if now >= active.work_at {
                let best_index = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
                    .map(|(index, _)| index);

                if let Some(index) = best_index {
                    if waiting[index].customer.priority < active.customer.priority {
                        let by = waiting[index].customer.id;
                        let active = running.take().unwrap();
                        let remaining = active.finish - now;
                        events.push(Event {
                            time: now,
                            customer: active.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        waiting.push(Waiting {
                            customer: active.customer,
                            remaining,
                            paused_at: Some(now),
                            started: true,
                        });
                    }
                }
            }
        }

        while running.is_none() && !waiting.is_empty() {
            let best_index = waiting
                .iter()
                .enumerate()
                .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
                .map(|(index, _)| index)
                .unwrap();
            let selected = waiting.remove(best_index);
            let cost = if last_dispatched == Some(selected.customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };

            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: selected.customer.id,
                time: now,
            })?;
            let finish = work_at
                .checked_add(selected.remaining)
                .ok_or(Error::TimeOverflow {
                    customer: selected.customer.id,
                    time: now,
                })?;

            let kind = if selected.started {
                Kind::Resume {
                    remaining: selected.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: selected.remaining,
                    work_at,
                    finish,
                }
            };
            events.push(Event {
                time: now,
                customer: selected.customer.id,
                kind,
            });
            last_dispatched = Some(selected.customer.id);
            running = Some(Running {
                customer: selected.customer,
                remaining: selected.remaining,
                work_at,
                finish,
                started: true,
            });

            if finish == now {
                let active = running.take().unwrap();
                events.push(Event {
                    time: now,
                    customer: active.customer.id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            }
        }
    }

    Ok(events)
}
