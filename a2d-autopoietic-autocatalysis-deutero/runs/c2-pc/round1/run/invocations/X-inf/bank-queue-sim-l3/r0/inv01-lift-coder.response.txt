#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    paused_at: Option<u64>,
}

#[derive(Clone)]
struct Active {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
    started: bool,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| b.started.cmp(&a.started))
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(x), Some(y)) => x.cmp(&y),
            _ => std::cmp::Ordering::Equal,
        })
        .then_with(|| {
            if a.started {
                a.customer.id.cmp(&b.customer.id)
            } else {
                a.customer
                    .arrival
                    .cmp(&b.customer.arrival)
                    .then_with(|| b.customer.id.cmp(&a.customer.id))
            }
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;
    let mut now = arrivals[0].arrival;

    while finished_count < customers.len() {
        if let Some(current) = active.as_ref() {
            if current.finish == now {
                let done = active.take().unwrap();
                trace.push(Event {
                    time: now,
                    customer: done.customer.id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            trace.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                paused_at: None,
            });
            next_arrival += 1;
        }

        if let Some(current) = active.as_ref() {
            if now >= current.work_at {
                if let Some(best_index) = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
                    .map(|(i, _)| i)
                {
                    if waiting[best_index].customer.priority < current.customer.priority {
                        let best_id = waiting[best_index].customer.id;
                        let paused = active.take().unwrap();
                        trace.push(Event {
                            time: now,
                            customer: paused.customer.id,
                            kind: Kind::Preempt {
                                remaining: paused.finish - now,
                                by: best_id,
                            },
                        });
                        waiting.push(Waiting {
                            customer: paused.customer,
                            remaining: paused.finish - now,
                            started: true,
                            paused_at: Some(now),
                        });
                    }
                }
            }
        }

        if active.is_none() {
            while !waiting.is_empty() {
                let best_index = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, a), (_, b)| waiting_cmp(a, b))
                    .map(|(i, _)| i)
                    .unwrap();
                let selected = waiting.remove(best_index);
                let cost = if last_dispatched == Some(selected.customer.id) {
                    0
                } else if last_dispatched.is_none() {
                    0
                } else {
                    switch_cost
                };
                let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: selected.customer.id,
                    time: now,
                })?;
                let finish = work_at.checked_add(selected.remaining).ok_or(
                    Error::TimeOverflow {
                        customer: selected.customer.id,
                        time: now,
                    },
                )?;

                let kind = if selected.started {
                    Kind::Resume {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                };
                trace.push(Event {
                    time: now,
                    customer: selected.customer.id,
                    kind,
                });
                last_dispatched = Some(selected.customer.id);

                if finish == now {
                    trace.push(Event {
                        time: now,
                        customer: selected.customer.id,
                        kind: Kind::Finish,
                    });
                    finished_count += 1;
                    continue;
                }

                active = Some(Active {
                    customer: selected.customer,
                    remaining: selected.remaining,
                    work_at,
                    finish,
                    started: true,
                });
                break;
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_time = u64::MAX;
        let mut has_next = false;
        if next_arrival < arrivals.len() {
            next_time = arrivals[next_arrival].arrival;
            has_next = true;
        }
        if let Some(current) = active.as_ref() {
            for boundary in [current.work_at, current.finish] {
                if boundary > now && (!has_next || boundary < next_time) {
                    next_time = boundary;
                    has_next = true;
                }
            }
        }
        if !has_next {
            break;
        }
        now = next_time;
    }

    Ok(trace)
}
