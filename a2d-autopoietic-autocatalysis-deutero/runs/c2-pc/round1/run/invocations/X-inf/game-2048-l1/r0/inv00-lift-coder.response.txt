#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Vec<usize> = match side {
        Side::Left => (0..row.len()).collect(),
        Side::Right => (0..row.len()).rev().collect(),
    };
    let tiles: Vec<usize> = indices.into_iter().filter(|&i| row[i] != 0).collect();

    let mut outputs: Vec<(u32, Option<([usize; 2], u32)>)> = Vec::new();
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut pos = 0;

    while pos < tiles.len() {
        let first = tiles[pos];
        if pos + 1 < tiles.len() {
            let second = tiles[pos + 1];
            if row[first] == row[second] {
                let value = row[first]
                    .checked_add(row[second])
                    .ok_or(Error::MergeOverflow { first, second })?;
                if value > 2_147_483_648 {
                    return Err(Error::MergeOverflow { first, second });
                }
                outputs.push((value, Some(([first, second], value))));
                score += u64::from(value);
                pos += 2;
                continue;
            }
        }
        outputs.push((row[first], None));
        pos += 1;
    }

    let mut cells = vec![0; row.len()];
    for (output_pos, (value, merge)) in outputs.into_iter().enumerate() {
        let to = match side {
            Side::Left => output_pos,
            Side::Right => row.len() - 1 - output_pos,
        };
        cells[to] = value;
        if let Some((from, value)) = merge {
            merges.push(Merge { from, to, value });
        }
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
