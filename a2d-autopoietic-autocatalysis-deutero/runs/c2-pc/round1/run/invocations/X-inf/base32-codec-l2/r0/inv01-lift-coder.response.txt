#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

pub fn encode(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let symbols = alphabet_bytes(alphabet);
    let mut output = String::new();
    let mut accumulator: u32 = 0;
    let mut bits = 0usize;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            let index = ((accumulator >> bits) & 0x1f) as usize;
            output.push(symbols[index] as char);
        }
    }

    if bits > 0 {
        let index = ((accumulator << (5 - bits)) & 0x1f) as usize;
        output.push(symbols[index] as char);
    }

    if padding == Padding::Required && !output.is_empty() {
        let pad_count = (8 - output.len() % 8) % 8;
        output.extend(std::iter::repeat_n('=', pad_count));
    }

    output
}

pub fn decode(
    input: &str,
    alphabet: Alphabet,
    padding: Padding,
) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let symbols = alphabet_bytes(alphabet);

    // Validate every byte before checking padding placement or data length.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && !symbols.contains(&byte) {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let first_padding = bytes.iter().position(|&byte| byte == b'=');
    let data_len = first_padding.unwrap_or(bytes.len());

    match padding {
        Padding::Omitted => {
            if let Some(offset) = first_padding {
                return Err(DecodeError::Padding { offset });
            }
        }
        Padding::Required => {
            if let Some(p) = first_padding {
                if let Some(offset) = bytes[p + 1..].iter().position(|&byte| byte != b'=') {
                    return Err(DecodeError::Padding { offset: p + 1 + offset });
                }
            }
        }
    }

    let residue = data_len % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: bytes.len() });
    }

    if padding == Padding::Required {
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => unreachable!(),
        };
        let actual_padding = bytes.len() - data_len;
        if actual_padding != expected_padding || bytes.len() % 8 != 0 {
            return Err(DecodeError::Padding { offset: data_len });
        }
    }

    let mut values = Vec::with_capacity(data_len);
    for (offset, &byte) in bytes[..data_len].iter().enumerate() {
        let value = symbols
            .iter()
            .position(|&symbol| symbol == byte)
            .expect("lexical validation accepted only alphabet symbols");
        values.push((value as u8, offset));
    }

    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };

    if unused_bits > 0 {
        let (last_value, last_offset) = values[data_len - 1];
        let mask = (1u8 << unused_bits) - 1;
        if last_value & mask != 0 {
            return Err(DecodeError::TrailingBits { offset: last_offset });
        }
    }

    let mut output = Vec::with_capacity(data_len * 5 / 8);
    let mut accumulator: u32 = 0;
    let mut bit_count = 0usize;

    for &(value, _) in &values {
        accumulator = (accumulator << 5) | u32::from(value);
        bit_count += 5;

        while bit_count >= 8 {
            bit_count -= 8;
            output.push(((accumulator >> bit_count) & 0xff) as u8);
        }
    }

    Ok(output)
}
