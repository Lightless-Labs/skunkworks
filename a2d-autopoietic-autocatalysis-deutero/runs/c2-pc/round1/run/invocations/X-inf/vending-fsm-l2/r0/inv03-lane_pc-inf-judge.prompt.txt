    pub selected: Option<String>,

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.as_ref().ok_or(Error::NoSelection)?;
        let product = self
            .products
            .iter()
            .find(|product| product.id == *id)
            .expect("selected products are always in the catalog");

        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let amount = self.credit - product.price;
        let change = choose_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for (index, denomination) in DENOMINATIONS.iter().enumerate() {
            let used = change.iter().filter(|coin| *coin == denomination).count() as u16;
            proposed[index] -= used;
        }

        for coin in &self.escrow {
            let index = denomination_index(*coin).expect("escrow contains accepted coins");
            proposed[index] += 1;
        }

        for (index, count) in proposed.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        let receipt = Receipt {
            id: product.id.clone(),
            price: product.price,
            change,
        };

        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|byte| byte.is_ascii_lowercase())
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|denomination| *denomination == coin)
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    let descending_indices = [3usize, 2, 1, 0];

    let mut remaining = amount;
    let mut greedy_counts = [0u16; 4];
    for index in descending_indices {
        let denomination = DENOMINATIONS[index];
        let count = tubes[index].min(remaining / denomination);
        greedy_counts[index] = count;
        remaining -= count * denomination;
    }

    if remaining == 0 {
        return Some(counts_to_descending_coins(&greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut counts = [0u16; 4];
    search_change(
        3,
        amount,
        tubes,
        &mut counts,
        &mut best,
    );

    best.map(|chosen| counts_to_descending_coins(&chosen))
}

fn search_change(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    counts: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index < 0 {
        if remaining == 0
            && best
                .as_ref()
                .map_or(true, |current| better_counts(counts, current))
        {
            *best = Some(*counts);
        }
        return;
    }

    let i = index as usize;
    let denomination = DENOMINATIONS[i];
    let max_count = tubes[i].min(remaining / denomination);

    for count in (0..=max_count).rev() {
        counts[i] = count;
        let next_remaining = remaining - count * denomination;

        if let Some(current) = best.as_ref() {
            let used: u16 = counts.iter().sum();
            if used > current.iter().sum() {
                continue;
            }
        }

        search_change(index - 1, next_remaining, tubes, counts, best);
    }

    counts[i] = 0;
}

fn better_counts(candidate: &[u16; 4], current: &[u16; 4]) -> bool {
    let candidate_total: u16 = candidate.iter().sum();
    let current_total: u16 = current.iter().sum();

    if candidate_total != current_total {
        return candidate_total < current_total;
    }

    for index in (0..4).rev() {
        if candidate[index] != current[index] {
            return candidate[index] > current[index];
        }
    }
    false
}

fn counts_to_descending_coins(counts: &[u16; 4]) -> Vec<u16> {
    let total: usize = counts.iter().map(|count| usize::from(*count)).sum();
    let mut coins = Vec::with_capacity(total);

    for index in (0..4).rev() {
        coins.extend(std::iter::repeat_n(
            DENOMINATIONS[index],
            usize::from(counts[index]),
        ));
    }

    coins
}

    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },

        Ok(())

            Some(id) => id,
            None => return Err(Error::NoSelection),

            .iter()

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for index in (0..4).rev() {
        let coin = DENOMINATIONS[index];
        let take = tubes[index].min(remaining / coin);
        greedy[index] = take;
        remaining -= take * coin;
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut candidate = [0u16; 4];
    search_change(0, amount, tubes, &mut candidate, &mut best);
    best
}

fn search_change(
    index: usize,
    remaining: u16,
    tubes: &[u16; 4],
    candidate: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index == 4 {
        if remaining == 0 {
            let replace = match best {
                None => true,
                Some(current) => {
                    let candidate_count: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
                    let current_count: u32 = current.iter().map(|&n| u32::from(n)).sum();
                    candidate_count < current_count
                        || (candidate_count == current_count
                            && (0..4).rev().find_map(|i| {
                                if candidate[i] != current[i] {
                                    Some(candidate[i] > current[i])
                                } else {
                                    None
                                }
                            }) == Some(true))
                }
            };
            if replace {
                *best = Some(*candidate);
            }
        }
        return;
    }

    let coin = DENOMINATIONS[index];
    let max_take = tubes[index].min(remaining / coin);
    for take in 0..=max_take {
        candidate[index] = take;
        search_change(
            index + 1,
            remaining - take * coin,
            tubes,
            candidate,
            best,
        );
    }
    candidate[index] = 0;
}

            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let id = product.id.as_str();
            if id.is_empty()
                || id.len() > 8
                || !id.bytes().all(|b| b.is_ascii_lowercase())
            {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains(&id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(id);
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (i, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match self.selected.as_deref() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };
        let product = self
            .products
            .iter()
            .find(|product| product.id == selected_id)
            .expect("selected product must be in catalog");

        let price = product.price;
        let id = product.id.clone();
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let mut chosen = [0u16; 4];
        let mut remainder = amount;

        for i in (0..4).rev() {
            let coin = DENOMINATIONS[i];
            let count = self.tubes[i].min(remainder / coin);
            chosen[i] = count;
            remainder -= count * coin;
        }

        if remainder != 0 {
            if let Some(solution) = best_change(amount, self.tubes) {
                chosen = solution;
            } else {
                return Err(Error::NoChange { amount });
            }
        }

        let mut proposed_tubes = self.tubes;
        for i in 0..4 {
            proposed_tubes[i] -= chosen[i];
        }
        for &coin in &self.escrow {
            let index = DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow contains only accepted denominations");
            let proposed = u32::from(proposed_tubes[index]) + 1;
            if proposed > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
            proposed_tubes[index] = proposed as u16;
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..chosen[i] {
                change.push(DENOMINATIONS[i]);
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}

fn best_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    fn search(
        index: usize,
        remainder: u16,
        tubes: [u16; 4],
        current: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
        best_count: &mut u16,
    ) {
        if index == 4 {
            if remainder == 0 {
                let count: u16 = current.iter().sum();
                let is_better = match best {
                    None => true,
                    Some(previous) => {
                        count < *best_count
                            || (count == *best_count
                                && descending_lex_greater(current, previous))
                    }
                };
                if is_better {
                    *best = Some(*current);
                    *best_count = count;
                }
            }
            return;
        }

        let coin = DENOMINATIONS[3 - index];
        let tube_index = 3 - index;
        let max_count = tubes[tube_index].min(remainder / coin);
        for count in (0..=max_count).rev() {
            current[tube_index] = count;
            search(
                index + 1,
                remainder - count * coin,
                tubes,
                current,
                best,
                best_count,
            );
        }
        current[tube_index] = 0;
    }

    let mut best = None;
    let mut best_count = u16::MAX;
    let mut current = [0u16; 4];
    search(
        0,
        amount,
        tubes,
        &mut current,
        &mut best,
        &mut best_count,
    );
    best
}

fn descending_lex_greater(left: &[u16; 4], right: &[u16; 4]) -> bool {
    for i in (0..4).rev() {
        if left[i] != right[i] {
            return left[i] > right[i];
        }
    }
    false
}
