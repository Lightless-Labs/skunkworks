#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut position = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if position == bytes.len() {
        return Ok(Vec::new());
    }

    #[derive(Clone, Copy, PartialEq, Eq)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted,
        AfterQuote,
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = State::FieldStart;
    let mut record_index = 0usize;
    let mut quote_open = 0usize;

    while position < bytes.len() {
        let byte = bytes[position];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: position,
            });
        }

        if state != State::Quoted && byte == b'\r' {
            if bytes.get(position + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: position,
                });
            }

            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            position += 2;
            state = State::FieldStart;
            continue;
        }

        match state {
            State::FieldStart => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    position += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    position += 1;
                }
                b'"' => {
                    quote_open = position;
                    state = State::Quoted;
                    position += 1;
                }
                _ => {
                    let ch = input[position..].chars().next().unwrap();
                    field.push(ch);
                    state = State::Unquoted;
                    position += ch.len_utf8();
                }
            },
            State::Unquoted => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    position += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    position += 1;
                }
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: position,
                    });
                }
                _ => {
                    let ch = input[position..].chars().next().unwrap();
                    field.push(ch);
                    position += ch.len_utf8();
                }
            },
            State::Quoted => {
                if byte == b'"' {
                    if bytes.get(position + 1) == Some(&b'"') {
                        field.push('"');
                        position += 2;
                    } else {
                        state = State::AfterQuote;
                        position += 1;
                    }
                } else {
                    let ch = input[position..].chars().next().unwrap();
                    field.push(ch);
                    position += ch.len_utf8();
                }
            }
            State::AfterQuote => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    position += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    position += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: position,
                    });
                }
            },
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open,
        });
    }

    if state == State::FieldStart || state == State::Unquoted || state == State::AfterQuote {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}
