pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape {
        axis: Axis,
        size: usize,
    },
    Bounds {
        entry: usize,
        axis: Axis,
        index: usize,
        limit: usize,
    },
    NonFinite {
        entry: usize,
    },
    SumOverflow {
        entry: usize,
    },
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        if rows > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Row,
                size: rows,
            });
        }
        if cols > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Column,
                size: cols,
            });
        }

        for (entry_index, entry) in entries.iter().enumerate() {
            if entry.row >= rows {
                return Err(Error::Bounds {
                    entry: entry_index,
                    axis: Axis::Row,
                    index: entry.row,
                    limit: rows,
                });
            }
            if entry.col >= cols {
                return Err(Error::Bounds {
                    entry: entry_index,
                    axis: Axis::Column,
                    index: entry.col,
                    limit: cols,
                });
            }
            if !entry.value.is_finite() {
                return Err(Error::NonFinite { entry: entry_index });
            }
        }

        let mut sums: Vec<(usize, usize, f64)> = Vec::new();
        for (entry_index, entry) in entries.iter().enumerate() {
            if let Some((_, _, sum)) = sums
                .iter_mut()
                .find(|(row, col, _)| *row == entry.row && *col == entry.col)
            {
                *sum += entry.value;
                if !sum.is_finite() {
                    return Err(Error::SumOverflow { entry: entry_index });
                }
            } else {
                sums.push((entry.row, entry.col, 0.0 + entry.value));
            }
        }

        sums.sort_by_key(|&(row, col, _)| (row, col));
        let canonical = sums
            .into_iter()
            .filter_map(|(row, col, value)| {
                (value != 0.0).then_some(Entry { row, col, value })
            })
            .collect();

        Ok(Self {
            rows,
            cols,
            entries: canonical,
        })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }
}
