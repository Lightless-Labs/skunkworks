#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }

    let Some(expected_len) = board.width.checked_mul(board.height) else {
        return Err(Error::Shape);
    };
    if board.cells.len() != expected_len {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    Ok(())
}

fn move_validated(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    let width = board.width;
    let height = board.height;
    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let line_count = match direction {
        Direction::Left | Direction::Right => height,
        Direction::Up | Direction::Down => width,
    };
    let line_len = match direction {
        Direction::Left | Direction::Right => width,
        Direction::Up | Direction::Down => height,
    };

    for line in 0..line_count {
        let index_at = |position: usize| -> usize {
            match direction {
                Direction::Left => line * width + position,
                Direction::Right => line * width + (width - 1 - position),
                Direction::Up => position * width + line,
                Direction::Down => (height - 1 - position) * width + line,
            }
        };

        let mut tiles = Vec::new();
        for position in 0..line_len {
            let index = index_at(position);
            let value = board.cells[index];
            if value != 0 {
                tiles.push((index, value));
            }
        }

        let mut outputs: Vec<(u32, Option<[usize; 2]>)> = Vec::new();
        let mut i = 0;
        while i < tiles.len() {
            let (first_index, value) = tiles[i];
            if i + 1 < tiles.len() && tiles[i + 1].1 == value {
                let second_index = tiles[i + 1].0;
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: first_index,
                        second: second_index,
                    });
                }
                let merged_value = value * 2;
                outputs.push((merged_value, Some([first_index, second_index])));
                score += u64::from(merged_value);
                i += 2;
            } else {
                outputs.push((value, None));
                i += 1;
            }
        }

        for (position, (value, source)) in outputs.into_iter().enumerate() {
            let destination = index_at(position);
            cells[destination] = value;
            if let Some(from) = source {
                merges.push(Merge {
                    from,
                    to: destination,
                    value,
                });
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width,
            height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    move_validated(board, direction)
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;

    let mut legal = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        if let Ok(result) = move_validated(board, direction) {
            if result.changed {
                legal.push(direction);
            }
        }
    }
    Ok(legal)
}
