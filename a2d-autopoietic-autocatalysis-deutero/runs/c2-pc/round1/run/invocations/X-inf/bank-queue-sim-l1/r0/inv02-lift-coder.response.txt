#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0;
    let mut current_time = 0u64;
    let mut finish_time: Option<u64> = None;
    let mut last_finish: Option<u64> = None;

    while next_arrival < arrivals.len() || !waiting.is_empty() || finish_time.is_some() {
        let next_arrival_time = arrivals.get(next_arrival).map(|c| c.arrival);
        let next_time = match (finish_time, next_arrival_time) {
            (Some(f), Some(a)) => f.min(a),
            (Some(f), None) => f,
            (None, Some(a)) => a.max(current_time),
            (None, None) => break,
        };
        current_time = next_time;

        if finish_time == Some(current_time) {
            let finished_customer = match events.last() {
                Some(Event {
                    customer,
                    kind: Kind::Start { finish, .. },
                    ..
                }) if *finish == current_time => *customer,
                _ => {
                    // For positive occupations, the active customer's ID is
                    // tracked by the most recent unmatched Start below.
                    let mut active = None;
                    for event in events.iter().rev() {
                        match event.kind {
                            Kind::Finish => break,
                            Kind::Start { .. } => {
                                active = Some(event.customer);
                                break;
                            }
                            Kind::Arrive => {}
                        }
                    }
                    active.expect("a scheduled finish has an active customer")
                }
            };
            events.push(Event {
                time: current_time,
                customer: finished_customer,
                kind: Kind::Finish,
            });
            finish_time = None;
            last_finish = Some(current_time);
        }

        while next_arrival < arrivals.len()
            && arrivals[next_arrival].arrival == current_time
        {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: current_time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        if finish_time.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = last_finish.map_or(true, |t| current_time > t);
                let service_finish = current_time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: current_time,
                    },
                )?;
                let finish = service_finish.checked_add(if wake { wake_cost } else { 0 }).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: current_time,
                    },
                )?;

                events.push(Event {
                    time: current_time,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });

                if finish == current_time {
                    events.push(Event {
                        time: current_time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    last_finish = Some(current_time);
                } else {
                    finish_time = Some(finish);
                }

                // Zero-duration occupations may permit further dispatches
                // at this same tick, after the full arrival cohort.
                while finish_time.is_none() {
                    let Some(customer) = waiting.pop_front() else {
                        break;
                    };
                    let wake = last_finish.map_or(true, |t| current_time > t);
                    let service_finish = current_time.checked_add(customer.service).ok_or(
                        Error::TimeOverflow {
                            customer: customer.id,
                            time: current_time,
                        },
                    )?;
                    let finish = service_finish
                        .checked_add(if wake { wake_cost } else { 0 })
                        .ok_or(Error::TimeOverflow {
                            customer: customer.id,
                            time: current_time,
                        })?;

                    events.push(Event {
                        time: current_time,
                        customer: customer.id,
                        kind: Kind::Start { finish, wake },
                    });

                    if finish == current_time {
                        events.push(Event {
                            time: current_time,
                            customer: customer.id,
                            kind: Kind::Finish,
                        });
                        last_finish = Some(current_time);
                    } else {
                        finish_time = Some(finish);
                    }
                }
            }
        }
    }

    Ok(events)
}
