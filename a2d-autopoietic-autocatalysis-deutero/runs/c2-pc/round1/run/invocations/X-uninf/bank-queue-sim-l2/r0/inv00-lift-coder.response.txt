#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    open: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|previous| previous.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut customer_order: Vec<usize> = (0..customers.len()).collect();
    customer_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            open: false,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0usize;
    let mut now = 0u64;

    loop {
        let mut finished: Vec<(usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (state.busy_until == Some(now)).then_some((
                    index,
                    state
                        .server
                        .id,
                ))
            })
            .collect();
        finished.sort_by_key(|&(_, id)| id);
        for (index, id) in finished {
            states[index].busy_until = None;
            states[index].idle_since = now;
            // The customer id is recorded in the scheduled event below.
            let customer_id = match events.last() {
                _ => 0,
            };
            let _ = customer_id;
            let completing_customer = events
                .iter()
                .rev()
                .find_map(|event| match event {
                    Event::Start {
                        time: _,
                        customer,
                        server,
                        finish,
                    } if *server == id && *finish == now => Some(*customer),
                    _ => None,
                })
                .unwrap_or(0);
            events.push(Event::Finish {
                time: now,
                customer: completing_customer,
                server: id,
            });
        }

        let mut opening: Vec<(usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.open && state.server.opens == now).then_some((index, state.server.id))
            })
            .collect();
        opening.sort_by(|a, b| b.1.cmp(&a.1));
        for (index, id) in opening {
            states[index].open = true;
            events.push(Event::Open {
                time: now,
                server: id,
            });
        }

        while next_customer < customer_order.len()
            && customers[customer_order[next_customer]].arrival == now
        {
            let index = customer_order[next_customer];
            events.push(Event::Arrive {
                time: now,
                customer: customers[index].id,
            });
            waiting.push(index);
            next_customer += 1;
        }

        loop {
            if waiting.is_empty() {
                break;
            }

            let selected_server = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.open && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected_server else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &states[server_index].server;
            let finish = customer
                .service
                .checked_add(now)
                .and_then(|time| time.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = now;
            } else {
                states[server_index].busy_until = Some(finish);
            }
        }

        if next_customer == customer_order.len()
            && states.iter().all(|state| state.open && state.busy_until.is_none())
        {
            break;
        }

        let next_arrival = customer_order
            .get(next_customer)
            .map(|&index| customers[index].arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.open)
            .map(|state| state.server.opens)
            .min();
        let next_finish = states.iter().filter_map(|state| state.busy_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&time| time > now)
            .min();

        match next_time {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}
