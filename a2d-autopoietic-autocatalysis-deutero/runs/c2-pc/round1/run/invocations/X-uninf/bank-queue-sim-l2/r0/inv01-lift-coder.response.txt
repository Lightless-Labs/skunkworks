#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct RuntimeServer {
    id: u32,
    opens: u64,
    overhead: u64,
    is_open: bool,
    busy_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<Customer> = customers.to_vec();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut runtime_servers: Vec<RuntimeServer> = servers
        .iter()
        .map(|server| RuntimeServer {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            is_open: false,
            busy_until: None,
            idle_since: server.opens,
        })
        .collect();

    let mut events = Vec::new();
    let mut customer_index = 0usize;
    let mut completed_customers = 0usize;
    let mut time = 0u64;
    let mut waiting: Vec<usize> = Vec::new();

    loop {
        let mut finishing: Vec<(usize, u32)> = runtime_servers
            .iter()
            .enumerate()
            .filter_map(|(index, server)| {
                if server.busy_until == Some(time) {
                    Some((index, server.id))
                } else {
                    None
                }
            })
            .collect();
        finishing.sort_by_key(|(_, id)| *id);

        for (index, _) in finishing {
            let customer_id = match runtime_servers[index].busy_until {
                Some(_) => runtime_servers[index]
                    .busy_until
                    .map(|_| ())
                    .and_then(|_| {
                        events.iter().rev().find_map(|event| match event {
                            Event::Start {
                                customer,
                                server,
                                finish,
                                ..
                            } if *server == runtime_servers[index].id && *finish == time => {
                                Some(*customer)
                            }
                            _ => None,
                        })
                    })
                    .unwrap(),
                None => unreachable!(),
            };

            let server_id = runtime_servers[index].id;
            events.push(Event::Finish {
                time,
                customer: customer_id,
                server: server_id,
            });
            runtime_servers[index].busy_until = None;
            runtime_servers[index].idle_since = time;
            completed_customers += 1;
        }

        let mut opening: Vec<usize> = runtime_servers
            .iter()
            .enumerate()
            .filter_map(|(index, server)| {
                (!server.is_open && server.opens == time).then_some(index)
            })
            .collect();
        opening.sort_by_key(|&index| std::cmp::Reverse(runtime_servers[index].id));

        for index in opening {
            let server = &mut runtime_servers[index];
            server.is_open = true;
            server.idle_since = time;
            events.push(Event::Open {
                time,
                server: server.id,
            });
        }

        while customer_index < ordered_customers.len()
            && ordered_customers[customer_index].arrival == time
        {
            let customer = &ordered_customers[customer_index];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(customer_index);
            customer_index += 1;
        }

        while !waiting.is_empty() {
            let selected_server = runtime_servers
                .iter()
                .enumerate()
                .filter(|(_, server)| server.is_open && server.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected_server else {
                break;
            };

            let customer_index_for_start = waiting.remove(0);
            let customer = &ordered_customers[customer_index_for_start];
            let server_id = runtime_servers[server_index].id;
            let overhead = runtime_servers[server_index].overhead;

            let after_service = time.checked_add(customer.service).ok_or(Error::TimeOverflow {
                customer: customer.id,
                server: server_id,
                time,
            })?;
            let finish = after_service.checked_add(overhead).ok_or(Error::TimeOverflow {
                customer: customer.id,
                server: server_id,
                time,
            })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server_id,
                });
                runtime_servers[server_index].idle_since = time;
                completed_customers += 1;
            } else {
                runtime_servers[server_index].busy_until = Some(finish);
            }
        }

        if customer_index == ordered_customers.len()
            && waiting.is_empty()
            && completed_customers == ordered_customers.len()
            && runtime_servers.iter().all(|server| server.is_open)
        {
            break;
        }

        let next_arrival = ordered_customers
            .get(customer_index)
            .map(|customer| customer.arrival);
        let next_open = runtime_servers
            .iter()
            .filter(|server| !server.is_open)
            .map(|server| server.opens)
            .min();
        let next_finish = runtime_servers
            .iter()
            .filter_map(|server| server.busy_until)
            .min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .min();

        if let Some(next_time) = next_time {
            time = next_time;
        } else {
            break;
        }
    }

    Ok(events)
}
