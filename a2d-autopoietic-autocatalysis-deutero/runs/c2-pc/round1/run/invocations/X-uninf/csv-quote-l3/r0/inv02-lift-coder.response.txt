#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    let mut record: Vec<String> = Vec::new();
    let mut field = String::new();

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::Start;
    let mut record_started = false;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::Quoted { opening } => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
                let _ = opening;
            }
            State::Start | State::Unquoted | State::AfterQuote => {
                if b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
                    return Err(CsvError::BareCr {
                        record: record_index,
                        byte: i,
                    });
                }

                match state {
                    State::AfterQuote => {
                        if b == delimiter {
                            record.push(std::mem::take(&mut field));
                            state = State::Start;
                            record_started = true;
                            i += 1;
                        } else if b == b'\n' || b == b'\r' {
                            record.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut record));
                            record_index += 1;
                            record_started = false;
                            state = State::Start;
                            i += if b == b'\r' { 2 } else { 1 };
                        } else {
                            return Err(CsvError::TrailingByte {
                                record: record_index,
                                byte: i,
                            });
                        }
                    }
                    State::Start => {
                        if b == delimiter {
                            record.push(String::new());
                            record_started = true;
                            i += 1;
                        } else if b == quote {
                            state = State::Quoted { opening: i };
                            record_started = true;
                            i += 1;
                        } else if b == b'\n' || b == b'\r' {
                            record.push(String::new());
                            records.push(std::mem::take(&mut record));
                            record_index += 1;
                            record_started = false;
                            i += if b == b'\r' { 2 } else { 1 };
                        } else {
                            let ch = input[i..].chars().next().unwrap();
                            field.push(ch);
                            state = State::Unquoted;
                            record_started = true;
                            i += ch.len_utf8();
                        }
                    }
                    State::Unquoted => {
                        if b == quote {
                            return Err(CsvError::UnexpectedQuote {
                                record: record_index,
                                byte: i,
                            });
                        } else if b == delimiter {
                            record.push(std::mem::take(&mut field));
                            state = State::Start;
                            record_started = true;
                            i += 1;
                        } else if b == b'\n' || b == b'\r' {
                            record.push(std::mem::take(&mut field));
                            records.push(std::mem::take(&mut record));
                            record_index += 1;
                            record_started = false;
                            state = State::Start;
                            i += if b == b'\r' { 2 } else { 1 };
                        } else {
                            let ch = input[i..].chars().next().unwrap();
                            field.push(ch);
                            i += ch.len_utf8();
                        }
                    }
                    State::Quoted { .. } => unreachable!(),
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::AfterQuote | State::Unquoted => {
            record.push(field);
            records.push(record);
        }
        State::Start => {
            if record_started || !record.is_empty() {
                record.push(String::new());
                records.push(record);
            }
        }
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut records = records.into_iter();
            let header = records.next().unwrap();
            let remaining: Vec<Vec<String>> = records.collect();

            if config.duplicates == DuplicatePolicy::Reject {
                for second in 0..header.len() {
                    if let Some(first) = header[..second]
                        .iter()
                        .position(|value| value == &header[second])
                    {
                        return Err(CsvError::DuplicateHeader {
                            name: header[second].clone(),
                            first,
                            second,
                        });
                    }
                }
            }

            for (offset, row) in remaining.iter().enumerate() {
                if row.len() != header.len() {
                    return Err(CsvError::HeaderWidth {
                        record: offset + 1,
                        expected: header.len(),
                        actual: row.len(),
                    });
                }
            }

            (Some(header), remaining)
        }
    } else {
        (None, records)
    };

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return b[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let body = if b.first() == Some(&b'-') {
        &b[1..]
    } else {
        b
    };
    let Some(dot) = body.iter().position(|&c| c == b'.') else {
        return false;
    };
    if body[dot + 1..].len() != 2
        || !body[dot + 1..].iter().all(u8::is_ascii_digit)
    {
        return false;
    }
    let integral = &body[..dot];
    if integral.is_empty() {
        return false;
    }
    if integral == b"0" {
        return b.first() != Some(&b'-') || body[dot + 1..] != *b"00";
    }
    integral[0].is_ascii_digit()
        && integral[0] != b'0'
        && integral.iter().all(u8::is_ascii_digit)
}

fn integer_value(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn decimal_value(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let unsigned = if negative { &s[1..] } else { s };
    let dot = unsigned.find('.').unwrap();
    let whole: i128 = unsigned[..dot].parse().unwrap();
    let fraction: i128 = unsigned[dot + 1..].parse().unwrap();
    let magnitude = whole * 100 + fraction;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate(value: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if value.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(value.to_owned())),
        ColumnType::Integer => integer_value(value).map(Value::Integer),
        ColumnType::Decimal => decimal_value(value).map(Value::Decimal),
        ColumnType::Boolean => match value {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output_rows = Vec::new();
    let mut errors = Vec::new();

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed {
                break;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let column = resolved;
            let result = match resolved {
                None => Err(CellErrorKind::UnknownColumn),
                Some(index) => match row.get(index) {
                    None => Err(CellErrorKind::MissingCell),
                    Some(value) => evaluate(value, spec),
                },
            };

            match result {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column,
                        kind,
                    });
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow | ErrorPolicy::Stop => {
                            row_failed = true;
                        }
                    }
                }
            }

            if policy == ErrorPolicy::Stop && row_failed {
                output_rows.push(output);
                return TypedResult {
                    rows: output_rows,
                    errors,
                };
            }
        }

        output_rows.push(output);
    }

    TypedResult {
        rows: output_rows,
        errors,
    }
}
