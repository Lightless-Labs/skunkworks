#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && !(2..=2_147_483_648).contains(&value) || (value != 0 && !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let mut scan: Vec<(usize, u32)> = match side {
        Side::Left => row
            .iter()
            .enumerate()
            .filter(|(_, value)| **value != 0)
            .map(|(index, &value)| (index, value))
            .collect(),
        Side::Right => row
            .iter()
            .enumerate()
            .rev()
            .filter(|(_, value)| **value != 0)
            .map(|(index, &value)| (index, value))
            .collect(),
    };

    let mut outputs: Vec<u32> = Vec::with_capacity(scan.len());
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut i = 0;

    while i < scan.len() {
        if i + 1 < scan.len() && scan[i].1 == scan[i + 1].1 {
            let value = scan[i].1.checked_mul(2).ok_or(Error::MergeOverflow {
                first: scan[i].0,
                second: scan[i + 1].0,
            })?;
            let to = match side {
                Side::Left => outputs.len(),
                Side::Right => row.len() - 1 - outputs.len(),
            };
            merges.push(Merge {
                from: [scan[i].0, scan[i + 1].0],
                to,
                value,
            });
            outputs.push(value);
            score += value as u64;
            i += 2;
        } else {
            outputs.push(scan[i].1);
            i += 1;
        }
    }

    let mut cells = vec![0; row.len()];
    for (position, value) in outputs.into_iter().enumerate() {
        let destination = match side {
            Side::Left => position,
            Side::Right => row.len() - 1 - position,
        };
        cells[destination] = value;
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
