#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut bytes = Vec::with_capacity(self.pending.len() + input.len());
        bytes.extend_from_slice(&self.pending);
        bytes.extend_from_slice(input);

        let complete_len = bytes.len() / 5 * 5;
        let output = encode_bytes(&bytes[..complete_len], self.config.alphabet);
        self.pending = bytes[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = encode_bytes(&self.pending, self.config.alphabet);
        let symbol_count = (self.pending.len() * 8 + 4) / 5;
        if self.config.padding == Padding::Required && symbol_count != 0 {
            let padding_count = (8 - symbol_count % 8) % 8;
            output.extend(std::iter::repeat('=').take(padding_count));
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_bytes(bytes: &[u8], alphabet: Alphabet) -> String {
    if bytes.is_empty() {
        return String::new();
    }

    let table = alphabet_bytes(alphabet);
    let mut output = String::with_capacity((bytes.len() * 8 + 4) / 5);

    let mut accumulator: u32 = 0;
    let mut bits = 0usize;
    for &byte in bytes {
        accumulator = (accumulator << 8) | byte as u32;
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            output.push(table[((accumulator >> bits) & 31) as usize] as char);
        }
    }

    if bits != 0 {
        let value = (accumulator << (5 - bits)) & 31;
        output.push(table[value as usize] as char);
    }

    output
}

pub struct Decoder {
    config: Config,
    closed: bool,
    position: usize,
    data_count: usize,
    bit_buffer: u8,
    bit_count: u8,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
    last_data_offset: Option<usize>,
    last_symbol_value: u8,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            closed: false,
            position: 0,
            data_count: 0,
            bit_buffer: 0,
            bit_count: 0,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: None,
            last_data_offset: None,
            last_symbol_value: 0,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            let recognized = classify(byte, next.config.alphabet);

            match recognized {
                Symbol::Invalid => {
                    return Err(StreamError::InvalidByte { offset, byte });
                }
                Symbol::Data(value, canonical) => {
                    if next.config.canonical && byte != canonical {
                        return Err(StreamError::NonCanonical {
                            offset,
                            expected: canonical,
                        });
                    }
                    if next.first_padding_offset.is_some() {
                        return Err(StreamError::Padding { offset });
                    }

                    next.data_count += 1;
                    next.last_data_offset = Some(offset);
                    next.last_symbol_value = value;
                    next.bit_buffer = (next.bit_buffer << 5) | value;
                    next.bit_count += 5;
                    while next.bit_count >= 8 {
                        next.bit_count -= 8;
                        output.push((next.bit_buffer >> next.bit_count) as u8);
                        next.bit_buffer &= (1u16.checked_shl(next.bit_count as u32).unwrap_or(0) as u8)
                            .wrapping_sub(1);
                    }
                }
                Symbol::Padding => {
                    if next.config.padding == Padding::Omitted {
                        return Err(StreamError::Padding { offset });
                    }

                    if let Some(expected) = next.expected_padding {
                        if next.padding_count >= expected {
                            return Err(StreamError::Padding { offset });
                        }
                    } else {
                        let residue = next.data_count % 8;
                        let expected = match residue {
                            2 => 6,
                            4 => 4,
                            5 => 3,
                            7 => 1,
                            _ => return Err(StreamError::Padding { offset }),
                        };
                        next.expected_padding = Some(expected);
                        next.first_padding_offset = Some(offset);
                    }

                    next.padding_count += 1;
                }
            }

            next.position += 1;
        }

        self.copy_state_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let expected_padding = match residue {
            0 => Some(0),
            2 => Some(6),
            4 => Some(4),
            5 => Some(3),
            7 => Some(1),
            _ => None,
        };

        let expected_padding = match expected_padding {
            Some(value) => value,
            None => return Err(StreamError::InvalidLength { offset: self.position }),
        };

        if self.config.padding == Padding::Required && self.padding_count != expected_padding {
            return Err(StreamError::Padding {
                offset: self.first_padding_offset.unwrap_or(self.position),
            });
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused_bits != 0 {
            let mask = (1u8 << unused_bits) - 1;
            if self.last_symbol_value & mask != 0 {
                return Err(StreamError::TrailingBits {
                    offset: self.last_data_offset.unwrap_or(self.position),
                });
            }
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            closed: self.closed,
            position: self.position,
            data_count: self.data_count,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            last_data_offset: self.last_data_offset,
            last_symbol_value: self.last_symbol_value,
        }
    }

    fn copy_state_from(&mut self, other: Self) {
        self.closed = other.closed;
        self.position = other.position;
        self.data_count = other.data_count;
        self.bit_buffer = other.bit_buffer;
        self.bit_count = other.bit_count;
        self.first_padding_offset = other.first_padding_offset;
        self.padding_count = other.padding_count;
        self.expected_padding = other.expected_padding;
        self.last_data_offset = other.last_data_offset;
        self.last_symbol_value = other.last_symbol_value;
    }
}

enum Symbol {
    Data(u8, u8),
    Padding,
    Invalid,
}

fn classify(byte: u8, alphabet: Alphabet) -> Symbol {
    if byte == b'=' {
        return Symbol::Padding;
    }

    let upper = byte.to_ascii_uppercase();
    if let Some(value) = alphabet_bytes(alphabet).iter().position(|&b| b == upper) {
        Symbol::Data(value as u8, upper)
    } else {
        Symbol::Invalid
    }
}
