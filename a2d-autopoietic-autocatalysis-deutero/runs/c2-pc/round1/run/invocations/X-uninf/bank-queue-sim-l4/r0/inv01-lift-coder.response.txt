#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };

    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct Waiter {
    index: usize,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
    interval_start: u64,
}

fn push_event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiter_better(a: &Waiter, b: &Waiter, customers: &[Customer]) -> bool {
    let ca = &customers[a.index];
    let cb = &customers[b.index];

    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }
    if a.paused != b.paused {
        return a.paused;
    }
    if a.paused {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        ca.id < cb.id
    } else if ca.arrival != cb.arrival {
        ca.arrival < cb.arrival
    } else {
        ca.id > cb.id
    }
}

fn best_waiter(waiters: &[Waiter], customers: &[Customer]) -> Option<usize> {
    let mut best = None;
    for (i, waiter) in waiters.iter().enumerate() {
        if best.map_or(true, |j| waiter_better(waiter, &waiters[j], customers)) {
            best = Some(i);
        }
    }
    best
}

fn consume_wait(waiter: &mut Waiter, now: u64, stats: &mut Stats) {
    let segment = now - waiter.entry;
    waiter.waited += segment;
    stats.wait_ticks += segment as u128;
    if let Some(deadline) = waiter.deadline {
        let remaining = deadline - waiter.entry;
        waiter.deadline = Some(now + remaining.saturating_sub(segment));
    }
}

fn expire_waiters(
    waiters: &mut Vec<Waiter>,
    now: u64,
    customers: &[Customer],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<Waiter> = waiters
        .iter()
        .filter(|w| w.deadline.map_or(false, |d| d <= now))
        .cloned()
        .collect();

    expired.sort_by(|a, b| {
        a.deadline
            .cmp(&b.deadline)
            .then_with(|| customers[a.index].id.cmp(&customers[b.index].id))
    });

    for waiter in &expired {
        let customer = &customers[waiter.index];
        let total = waiter.waited + (now - waiter.entry);
        push_event(
            events,
            now,
            customer.id,
            Kind::Renege { waited: total },
        );
        stats.reneged += 1;
    }

    waiters.retain(|w| !w.deadline.map_or(false, |d| d <= now));
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut seen: Vec<u32> = Vec::new();
    let mut initial_deadlines = vec![None; customers.len()];
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        seen.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        if let Some(patience) = customer.patience {
            initial_deadlines[index] = Some(
                customer
                    .arrival
                    .checked_add(patience)
                    .ok_or(Error::DeadlineOverflow {
                        index,
                        id: customer.id,
                    })?,
            );
        }
    }

    let mut arrival_order: Vec<usize> = (0..customers.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut report_stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut events = Vec::new();
    let mut waiters: Vec<Waiter> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = customers
        .iter()
        .map(|c| c.arrival)
        .min()
        .unwrap_or(0);
    let mut last_dispatched: Option<u32> = None;
    let mut ever_started = vec![false; customers.len()];

    while next_arrival < arrival_order.len() || active.is_some() || !waiters.is_empty() {
        if let Some(a) = &active {
            if a.finish == now {
                let id = customers[a.index].id;
                push_event(&mut events, now, id, Kind::Finish);
                report_stats.busy_ticks += (now - a.interval_start) as u128;
                report_stats.completed += 1;
                report_stats.turnaround_ticks +=
                    (now - customers[a.index].arrival) as u128;
                report_stats.completion_order.push(id);
                active = None;
            }
        }

        expire_waiters(
            &mut waiters,
            now,
            customers,
            &mut events,
            &mut report_stats,
        );

        while next_arrival < arrival_order.len()
            && customers[arrival_order[next_arrival]].arrival == now
        {
            let index = arrival_order[next_arrival];
            next_arrival += 1;
            let customer = &customers[index];
            push_event(&mut events, now, customer.id, Kind::Arrive);

            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiters.len() >= limit {
                push_event(&mut events, now, customer.id, Kind::Balk);
                report_stats.balked += 1;
            } else {
                waiters.push(Waiter {
                    index,
                    entry: now,
                    deadline: initial_deadlines[index],
                    waited: 0,
                    paused: false,
                    pause_time: 0,
                });
                report_stats.accepted += 1;
            }
        }

        if let Some(running) = active.as_ref() {
            if now >= running.work_at {
                if let Some(best_i) = best_waiter(&waiters, customers) {
                    let best = &waiters[best_i];
                    if customers[best.index].priority < customers[running.index].priority {
                        let paused_index = running.index;
                        let remaining = running.finish - now;
                        let by = customers[best.index].id;

                        push_event(
                            &mut events,
                            now,
                            customers[paused_index].id,
                            Kind::Preempt { remaining, by },
                        );
                        report_stats.busy_ticks += (now - running.interval_start) as u128;
                        report_stats.preemptions += 1;
                        active = None;

                        let patience = customers[paused_index].patience;
                        let deadline = match patience {
                            Some(_) => Some(now.checked_add(remaining).map(|_| now).unwrap_or(now)),
                            None => None,
                        };
                        let _ = deadline;

                        // Remaining waiting budget is represented by the old active
                        // customer's original deadline budget, which is tracked separately
                        // below from its accumulated waiting segments.
                        let budget_left = customers[paused_index]
                            .patience
                            .map(|p| p.saturating_sub(0));
                        let new_deadline = match budget_left {
                            Some(b) => Some(now.checked_add(b).ok_or(Error::TimeOverflow {
                                customer: customers[paused_index].id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })?),
                            None => None,
                        };

                        waiters.push(Waiter {
                            index: paused_index,
                            entry: now,
                            deadline: new_deadline,
                            waited: 0,
                            paused: true,
                            pause_time: now,
                        });

                        let _ = remaining;
                    }
                }
            }
        }

        if active.is_none() {
            while let Some(best_i) = best_waiter(&waiters, customers) {
                let mut waiter = waiters.remove(best_i);
                let id = customers[waiter.index].id;

                consume_wait(&mut waiter, now, &mut report_stats);

                let switch = if last_dispatched.map_or(false, |prev| prev != id) {
                    config.switch_cost
                } else {
                    0
                };
                let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                    stage: OverflowStage::WorkAt,
                })?;
                let finish = work_at
                    .checked_add(customers[waiter.index].service)
                    .ok_or(Error::TimeOverflow {
                        customer: id,
                        time: now,
                        stage: OverflowStage::Finish,
                    })?;

                let kind = if ever_started[waiter.index] {
                    Kind::Resume {
                        remaining: customers[waiter.index].service,
                        work_at,
                        finish,
                    }
                } else {
                    ever_started[waiter.index] = true;
                    report_stats.started += 1;
                    Kind::Start {
                        remaining: customers[waiter.index].service,
                        work_at,
                        finish,
                    }
                };
                push_event(&mut events, now, id, kind);
                last_dispatched = Some(id);

                if finish == now {
                    push_event(&mut events, now, id, Kind::Finish);
                    report_stats.completed += 1;
                    report_stats.turnaround_ticks +=
                        (now - customers[waiter.index].arrival) as u128;
                    report_stats.completion_order.push(id);
                } else {
                    active = Some(Active {
                        index: waiter.index,
                        remaining: customers[waiter.index].service,
                        work_at,
                        finish,
                        interval_start: now,
                    });
                    break;
                }
            }
        }

        expire_waiters(
            &mut waiters,
            now,
            customers,
            &mut events,
            &mut report_stats,
        );

        if next_arrival >= arrival_order.len() && active.is_none() && waiters.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now && next_time.map_or(true, |n| t < n) {
                next_time = Some(t);
            }
        };

        if next_arrival < arrival_order.len() {
            consider(customers[arrival_order[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            consider(a.work_at);
            consider(a.finish);
        }
        for waiter in &waiters {
            if let Some(deadline) = waiter.deadline {
                consider(deadline);
            }
        }

        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    report_stats.elapsed = events.last().map_or(0, |event| event.time);
    report_stats.mean_wait_milli = if report_stats.accepted == 0 {
        0
    } else {
        rounded_ratio(report_stats.wait_ticks * 1000, report_stats.accepted as u128)
            .unwrap_or(0)
    };
    report_stats.mean_turnaround_milli = if report_stats.completed == 0 {
        0
    } else {
        rounded_ratio(
            report_stats.turnaround_ticks * 1000,
            report_stats.completed as u128,
        )
        .unwrap_or(0)
    };
    report_stats.utilization_bps = if report_stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(report_stats.busy_ticks * 10000, report_stats.elapsed as u128)
            .unwrap_or(0) as u32
    };

    Ok(Report {
        events,
        stats: report_stats,
    })
}
