fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn encoded(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sorted_evidence(e: &[Evidence]) -> Vec<Evidence> {
    let mut out = e.to_vec();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    out
}

fn evidence_lines(e: &[Evidence]) -> String {
    let mut out = String::new();
    for item in sorted_evidence(e) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(item.constraint));
        out.push_str(" via ");
        out.push_str(&item.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encoded(site), encoded(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encoded(package))
        }
        ResolveError::DuplicateVersion { package, version } => {
            format!("duplicate-version|package={}|version={}", encoded(package), version_text(*version))
        }
        ResolveError::DuplicateFeature { package, version, feature } => {
            format!(
                "duplicate-feature|package={}|version={}|feature={}",
                encoded(package),
                version_text(*version),
                encoded(feature)
            )
        }
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encoded(package))
        }
        ResolveError::Missing { package, evidence } => {
            format!("missing|package={}{}", encoded(package), evidence_lines(evidence))
        }
        ResolveError::NoMatch { package, requirements, evidence } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!("no-match|package={}|constraints={}{}", encoded(package), cs, evidence_lines(evidence))
        }
        ResolveError::Conflict { package, selected, requirements, evidence } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "conflict|package={}|selected={}|constraints={}{}",
                encoded(package),
                version_text(*selected),
                cs,
                evidence_lines(evidence)
            )
        }
        ResolveError::MissingFeature { package, selected, feature, path } => format!(
            "missing-feature|package={}|selected={}|feature={}{}\n  F:{} via {}",
            encoded(package),
            version_text(*selected),
            encoded(feature),
            "",
            encoded(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

    origins: std::collections::BTreeMap<String, Vec<Origin>>,
    requests: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    active_features: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    paths: std::collections::BTreeMap<String, Vec<String>>,
}

fn platform_name(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn platform_active(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!((p, t), (Platform::Linux, Target::Linux) | (Platform::Mac, Target::Mac) | (Platform::Windows, Target::Windows))
}

fn release_index<'a>(catalog: &'a [Package], names: &std::collections::BTreeMap<String, usize>, selected: &std::collections::BTreeMap<String, Version>, package: &str) -> Option<&'a Release> {
    let pi = *names.get(package)?;
    let v = *selected.get(package)?;
    catalog[pi].releases.iter().find(|r| r.version == v)
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!("root[{}]:{}@{}+{{{}}}", i, r.package, constraint_text(r.constraint), fs.join(","))
}

fn dep_token(source: &str, version: Version, j: usize, d: &Dependency) -> String {
    let mut fs = d.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        j,
        d.package,
        constraint_text(d.constraint),
        fs.join(","),
        d.gate.as_deref().unwrap_or("*"),
        platform_name(d.platform)
    )
}

fn compute_closure(
    catalog: &[Package],
    names: &std::collections::BTreeMap<String, usize>,
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    use std::collections::{BTreeMap, BTreeSet};
    let mut c = Closure::default();
    let mut queue: Vec<(String, String, Origin)> = Vec::new();

    for (i, r) in roots.iter().enumerate() {
        let path = vec![root_token(i, r)];
        queue.push((r.package.clone(), String::new(), Origin { constraint: r.constraint, path: path.clone() }));
        c.paths.entry(r.package.clone()).and_modify(|p| {
            if (path.len(), &path) < (p.len(), p) { *p = path.clone(); }
        }).or_insert(path);
        for f in &r.features {
            c.requests.entry(r.package.clone()).or_default().insert(f.clone());
        }
    }

    let mut processed_edges = BTreeSet::new();
    let mut processed_features = BTreeSet::new();

    loop {
        while let Some((dest, _, origin)) = queue.pop() {
            c.origins.entry(dest).or_default().push(origin);
        }

        let packages: Vec<String> = selected.keys().cloned().collect();
        let mut changed = false;

        for package in packages {
            let Some(rel) = release_index(catalog, names, selected, &package) else { continue };
            let reqs = c.requests.get(&package).cloned().unwrap_or_default();
            for feature in reqs {
                if processed_features.insert((package.clone(), feature.clone())) {
                    c.active_features.entry(package.clone()).or_default().insert(feature.clone());
                    if let Some(decl) = rel.features.iter().find(|f| f.name == feature) {
                        for enabled in &decl.enables {
                            if c.requests.entry(package.clone()).or_default().insert(enabled.clone()) {
                                changed = true;
                            }
                        }
                    }
                }
            }

            let active = c.active_features.get(&package).cloned().unwrap_or_default();
            for (j, dep) in rel.deps.iter().enumerate() {
                if !platform_active(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !active.contains(g))
                {
                    continue;
                }
                if processed_edges.insert((package.clone(), j)) {
                    let source_v = selected[&package];
                    let token = dep_token(&package, source_v, j, dep);
                    let source_path = c.paths.get(&package).cloned().unwrap_or_default();
                    let mut path = source_path;
                    path.push(token);
                    c.paths.entry(dep.package.clone()).and_modify(|p| {
                        if (path.len(), &path) < (p.len(), p) { *p = path.clone(); }
                    }).or_insert(path.clone());
                    queue.push((dep.package.clone(), package.clone(), Origin { constraint: dep.constraint, path }));
                    for f in &dep.features {
                        if c.requests.entry(dep.package.clone()).or_default().insert(f.clone()) {
                            changed = true;
                        }
                    }
                }
            }
        }

        if !queue.is_empty() {
            continue;
        }
        if !changed {
            break;
        }
    }

    for origins in c.origins.values_mut() {
        origins.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    }
    c
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    names: std::collections::BTreeMap<String, usize>,
    lock_map: std::collections::BTreeMap<String, Version>,
    attempts: usize,
    budget: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, std::collections::BTreeMap<String, Version>, Closure)>,
}

impl<'a> Search<'a> {
    fn inspect(&mut self, selected: &std::collections::BTreeMap<String, Version>) -> Result<(), ResolveError> {
        use std::collections::BTreeSet;
        let closure = compute_closure(self.catalog, &self.names, self.roots, selected, self.target);

        let mut active: BTreeSet<String> = closure.origins.keys().cloned().collect();
        active.extend(closure.requests.keys().cloned());
        for package in &active {
            if !self.names.contains_key(package) {
                let ev = closure.origins.get(package).cloned().unwrap_or_default().into_iter()
                    .map(|o| Evidence { constraint: o.constraint, path: o.path }).collect();
                return Err(ResolveError::Missing { package: package.clone(), evidence: ev });
            }
        }

        for (package, version) in selected {
            let fs = closure.active_features.get(package).cloned().unwrap_or_default();
            if let Some(rel) = release_index(self.catalog, &self.names, selected, package) {
                for feature in fs {
                    if !rel.features.iter().any(|f| f.name == feature) {
                        return Err(ResolveError::MissingFeature {
                            package: package.clone(),
                            selected: *version,
                            feature,
                            path: closure.paths.get(package).cloned().unwrap_or_default(),
                        });
                    }
                }
            }
        }

        for (package, version) in selected {
            let cs = closure.origins.get(package).cloned().unwrap_or_default().iter().map(|o| o.constraint).collect::<Vec<_>>();
            if !cs.iter().all(|c| matches(*c, *version)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: canonical_constraints(&cs),
                    evidence: closure.origins.get(package).cloned().unwrap_or_default().into_iter()
                        .map(|o| Evidence { constraint: o.constraint, path: o.path }).collect(),
                });
            }
        }

        let mut unselected: Vec<String> = active.into_iter().filter(|p| !selected.contains_key(p)).collect();
        unselected.sort();
        for package in &unselected {
            let cs = closure.origins.get(package).cloned().unwrap_or_default().iter().map(|o| o.constraint).collect::<Vec<_>>();
            let pi = self.names[package];
            let has = self.catalog[pi].releases.iter().any(|r| cs.iter().all(|c| matches(*c, r.version)));
            if !has {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical_constraints(&cs),
                    evidence: closure.origins.get(package).cloned().unwrap_or_default().into_iter()
                        .map(|o| Evidence { constraint: o.constraint, path: o.path }).collect(),
                });
            }
        }
        Ok(())
    }

    fn dfs(&mut self, selected: &mut std::collections::BTreeMap<String, Version>) -> Result<(), ResolveError> {
        if let Err(e) = self.inspect(selected) {
            if matches!(e, ResolveError::Limit { .. }) { return Err(e); }
            if self.first_error.is_none() { self.first_error = Some(e); }
            return Ok(());
        }
        let closure = compute_closure(self.catalog, &self.names, self.roots, selected, self.target);
        let mut active: Vec<String> = closure.origins.keys().cloned().collect();
        active.extend(closure.requests.keys().cloned());
        active.sort();
        active.dedup();
        let Some(package) = active.into_iter().find(|p| !selected.contains_key(p)) else {
            let cost = self.lock_map.iter().filter(|(p, v)| selected.get(*p) != Some(*v)).count();
            let replace = match &self.best {
                None => true,
                Some((bc, bs, _)) => {
                    cost < *bc || (cost == *bc && self.vector_cmp(selected, bs).is_gt())
                }
            };
            if replace { self.best = Some((cost, selected.clone(), closure)); }
            return Ok(());
        };

        let cs = closure.origins.get(&package).cloned().unwrap_or_default().iter().map(|o| o.constraint).collect::<Vec<_>>();
        let pi = self.names[&package];
        let mut candidates: Vec<Version> = self.catalog[pi].releases.iter()
            .filter(|r| cs.iter().all(|c| matches(*c, r.version)))
            .map(|r| r.version).collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.lock_map.get(&package) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit { attempts: self.budget });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            let result = self.dfs(selected);
            selected.remove(&package);
            result?;
        }
        Ok(())
    }

    fn vector_cmp(
        &self,
        a: &std::collections::BTreeMap<String, Version>,
        b: &std::collections::BTreeMap<String, Version>,
    ) -> std::cmp::Ordering {
        let names: Vec<String> = self.names.keys().cloned().collect();
        for name in names {
            let ord = match (a.get(&name), b.get(&name)) {
                (None, None) => std::cmp::Ordering::Equal,
                (None, Some(_)) => std::cmp::Ordering::Less,
                (Some(_), None) => std::cmp::Ordering::Greater,
                (Some(x), Some(y)) => x.cmp(y),
            };
            if !ord.is_eq() { return ord; }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{}]", i), name: r.package.clone() });
        }
        for (f, feature) in r.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName { site: format!("root[{}].feature[{}]", i, f), name: feature.clone() });
            }
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{}]", i), name: p.name.clone() });
        }
        for (j, rel) in p.releases.iter().enumerate() {
            for (k, dep) in rel.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].dep[{}]", i, j, k), name: dep.package.clone() });
                }
                if let Some(g) = &dep.gate {
                    if !valid_name(g) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k), name: g.clone() });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f), name: feature.clone() });
                    }
                }
            }
            for (f, feature) in rel.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].feature[{}]", i, j, f), name: feature.name.clone() });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].feature[{}].enable[{}]", i, j, f, e), name: enabled.clone() });
                    }
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for p in catalog {
        if !seen.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }
    for p in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion { package: p.name.clone(), version: r.version });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut features = std::collections::BTreeSet::new();
            for f in &r.features {
                if !features.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature { package: p.name.clone(), version: r.version, feature: f.name.clone() });
                }
            }
        }
    }
    for (i, l) in locks.iter().enumerate() {
        if !valid_name(&l.package) {
            return Err(ResolveError::InvalidName { site: format!("lock[{}]", i), name: l.package.clone() });
        }
    }
    let mut lock_seen = std::collections::BTreeSet::new();
    for l in locks {
        if !lock_seen.insert(l.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: l.package.clone() });
        }
    }

    let names: std::collections::BTreeMap<String, usize> =
        catalog.iter().enumerate().map(|(i, p)| (p.name.clone(), i)).collect();
    let lock_map = locks.iter().map(|l| (l.package.clone(), l.version)).collect();

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        names,
        lock_map,
        attempts: 0,
        budget: max_attempts,
        first_error: None,
        best: None,
    };
    let mut selected = std::collections::BTreeMap::new();
    search.dfs(&mut selected)?;
    let Some((_, best, final_closure)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: vec![],
            evidence: vec![],
        }));
    };

    let selected_vec = best.iter().map(|(package, version)| Selection { package: package.clone(), version: *version }).collect::<Vec<_>>();
    let mut changes = Vec::new();
    let mut all: std::collections::BTreeSet<String> = locks.iter().map(|l| l.package.clone()).collect();
    all.extend(best.keys().cloned());
    for package in all {
        let before = search.lock_map.get(&package).copied();
        let after = best.get(&package).copied();
        if before == after { continue; }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search.names.get(&package).is_some_and(|pi| catalog[*pi].releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let cs = final_closure.origins.get(&package).cloned().unwrap_or_default().iter().map(|o| o.constraint).collect::<Vec<_>>();
                    if !cs.iter().all(|c| matches(*c, old)) { ChangeReason::Rejected } else { ChangeReason::Rebalanced }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change { package, before, after, reason });
    }

    Ok(Resolution { selected: selected_vec, changes })
}

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn check_name(name: &str, site: String) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn evidence_lines(mut base: String, evidence: &[Evidence]) -> String {
    for e in evidence_sorted(evidence) {
        base.push_str("\n  C:");
        base.push_str(&constraint_text(e.constraint));
        base.push_str(" via ");
        base.push_str(&path_text(&e.path));
    }
    base
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => evidence_lines(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => evidence_lines(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => evidence_lines(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    active_features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    active_deps: BTreeMap<String, Vec<usize>>,
}

use std::collections::{BTreeMap, BTreeSet};

fn feature_index(release: &Release) -> BTreeMap<String, &Feature> {
    release
        .features
        .iter()
        .map(|f| (f.name.clone(), f))
        .collect()
}

fn target_name(target: Target) -> &'static str {
    match target {
        Target::Linux => "linux",
        Target::Mac => "mac",
        Target::Windows => "windows",
    }
}

fn platform_active(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!(
            (platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn sorted_feature_list(features: &[String]) -> Vec<String> {
    let mut f = features.to_vec();
    f.sort();
    f.dedup();
    f
}

fn root_token(i: usize, req: &Requirement) -> String {
    let features = sorted_feature_list(&req.features).join(",");
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        req.package,
        constraint_text(req.constraint),
        features
    )
}

fn dependency_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        sorted_feature_list(&dep.features).join(","),
        gate,
        platform
    )
}

fn compute_closure(
    roots: &[Requirement],
    catalog: &[Package],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let packages: BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut c = Closure {
        constraints: BTreeMap::new(),
        origins: BTreeMap::new(),
        requests: BTreeMap::new(),
        active_features: BTreeMap::new(),
        paths: BTreeMap::new(),
        active_deps: BTreeMap::new(),
    };

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        c.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        c.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: path.clone(),
            });
        c.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        let replace = c
            .paths
            .get(&root.package)
            .map(|old| (path.len(), &path) < (old.len(), old))
            .unwrap_or(true);
        if replace {
            c.paths.insert(root.package.clone(), path);
        }
    }

    loop {
        let mut changed = false;

        let package_names: Vec<String> = c.requests.keys().cloned().collect();
        for package in package_names {
            let Some(version) = selected.get(&package).copied() else {
                continue;
            };
            let Some(pkg) = packages.get(package.as_str()).copied() else {
                continue;
            };
            let Some(release) = pkg.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            let declarations = feature_index(release);
            let requested: Vec<String> = c
                .requests
                .get(&package)
                .into_iter()
                .flat_map(|s| s.iter().cloned())
                .collect();
            for feature in requested {
                if c.active_features
                    .entry(package.clone())
                    .or_default()
                    .insert(feature.clone())
                {
                    changed = true;
                }
                if let Some(decl) = declarations.get(&feature) {
                    for enabled in &decl.enables {
                        if c.requests
                            .entry(package.clone())
                            .or_default()
                            .insert(enabled.clone())
                        {
                            changed = true;
                        }
                    }
                }
            }
        }

        let selected_names: Vec<(String, Version)> = selected
            .iter()
            .map(|(p, v)| (p.clone(), *v))
            .collect();
        for (source, version) in selected_names {
            let Some(pkg) = packages.get(source.as_str()).copied() else {
                continue;
            };
            let Some(release) = pkg.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            let active_features = c
                .active_features
                .get(&source)
                .cloned()
                .unwrap_or_default();
            for (index, dep) in release.deps.iter().enumerate() {
                if !platform_active(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !active_features.contains(g))
                {
                    continue;
                }
                if c.active_deps
                    .entry(source.clone())
                    .or_default()
                    .contains(&index)
                {
                    continue;
                }
                c.active_deps
                    .entry(source.clone())
                    .or_default()
                    .push(index);
                changed = true;

                let token = dependency_token(&source, version, index, dep);
                let mut path = c.paths.get(&source).cloned().unwrap_or_default();
                path.push(token);
                let replace = c
                    .paths
                    .get(&dep.package)
                    .map(|old| (path.len(), &path) < (old.len(), old))
                    .unwrap_or(true);
                if replace {
                    c.paths.insert(dep.package.clone(), path.clone());
                    changed = true;
                }

                c.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                c.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path,
                    });
                let requests = c.requests.entry(dep.package.clone()).or_default();
                for feature in &dep.features {
                    if requests.insert(feature.clone()) {
                        changed = true;
                    }
                }
            }
        }

        if !changed {
            break;
        }
    }

    for values in c.constraints.values_mut() {
        *values = canonical_constraints(values);
    }
    for values in c.active_deps.values_mut() {
        values.sort_unstable();
    }
    c
}

fn node_error(
    roots: &[Requirement],
    catalog: &[Package],
    selected: &BTreeMap<String, Version>,
    target: Target,
    closure: &Closure,
) -> Option<ResolveError> {
    let packages: BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();

    let active: BTreeSet<String> = closure.constraints.keys().cloned().collect();
    if let Some(package) = active
        .iter()
        .find(|name| !packages.contains_key(name.as_str()))
    {
        return Some(ResolveError::Missing {
            package: package.clone(),
            evidence: evidence_for(closure, package),
        });
    }

    for (package, version) in selected {
        if let Some(requirements) = closure.constraints.get(package) {
            if !requirements.iter().all(|c| matches(*version, *c)) {
                return Some(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: requirements.clone(),
                    evidence: evidence_for(closure, package),
                });
            }
        }
    }

    for (package, version) in selected {
        let Some(pkg) = packages.get(package.as_str()).copied() else {
            continue;
        };
        let Some(release) = pkg.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        if let Some(features) = closure.active_features.get(package) {
            for feature in features {
                if !release.features.iter().any(|f| f.name == *feature) {
                    return Some(ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: closure.paths.get(package).cloned().unwrap_or_default(),
                    });
                }
            }
        }
    }

    for package in active {
        if selected.contains_key(&package) {
            continue;
        }
        let requirements = closure
            .constraints
            .get(&package)
            .cloned()
            .unwrap_or_default();
        let candidates = packages[package.as_str()]
            .releases
            .iter()
            .any(|r| requirements.iter().all(|c| matches(r.version, *c)));
        if !candidates {
            return Some(ResolveError::NoMatch {
                package: package.clone(),
                requirements,
                evidence: evidence_for(closure, &package),
            });
        }
    }

    let _ = (roots, target);
    None
}

fn evidence_for(closure: &Closure, package: &str) -> Vec<Evidence> {
    let mut out: Vec<Evidence> = closure
        .origins
        .get(package)
        .into_iter()
        .flat_map(|v| v.iter())
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

struct SearchState<'a> {
    roots: &'a [Requirement],
    catalog: &'a [Package],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
    limit: Option<ResolveError>,
}

fn solution_cost(selected: &BTreeMap<String, Version>, locks: &[Lock]) -> usize {
    locks
        .iter()
        .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
        .count()
}

fn solution_better(
    candidate: &BTreeMap<String, Version>,
    cost: usize,
    best: &(usize, BTreeMap<String, Version>),
    names: &[String],
) -> bool {
    if cost != best.0 {
        return cost < best.0;
    }
    for name in names {
        match (candidate.get(name), best.1.get(name)) {
            (Some(a), Some(b)) if a != b => return a > b,
            (Some(_), None) => return true,
            (None, Some(_)) => return false,
            _ => {}
        }
    }
    false
}

fn search(state: &mut SearchState<'_>, selected: &mut BTreeMap<String, Version>) {
    if state.limit.is_some() {
        return;
    }
    let closure = compute_closure(state.roots, state.catalog, selected, state.target);
    if let Some(error) = node_error(
        state.roots,
        state.catalog,
        selected,
        state.target,
        &closure,
    ) {
        if state.first_error.is_none() {
            state.first_error = Some(error);
        }
        return;
    }

    let active = closure.constraints.keys().cloned().collect::<BTreeSet<_>>();
    let next = active
        .iter()
        .find(|p| !selected.contains_key(*p))
        .cloned();
    let Some(package) = next else {
        let cost = solution_cost(selected, state.locks);
        let names: Vec<String> = state.catalog.iter().map(|p| p.name.clone()).collect();
        if state
            .best
            .as_ref()
            .map(|best| solution_better(selected, cost, best, &names))
            .unwrap_or(true)
        {
            state.best = Some((cost, selected.clone()));
        }
        return;
    };

    let Some(pkg) = state.catalog.iter().find(|p| p.name == package) else {
        return;
    };
    let requirements = closure
        .constraints
        .get(&package)
        .cloned()
        .unwrap_or_default();
    let mut candidates: Vec<Version> = pkg
        .releases
        .iter()
        .map(|r| r.version)
        .filter(|v| requirements.iter().all(|c| matches(*v, *c)))
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(lock) = state.locks.iter().find(|l| l.package == package) {
        if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
            let locked = candidates.remove(pos);
            candidates.insert(0, locked);
        }
    }

    for version in candidates {
        if state.attempts == state.max_attempts {
            state.limit = Some(ResolveError::Limit {
                attempts: state.max_attempts,
            });
            return;
        }
        state.attempts += 1;
        selected.insert(package.clone(), version);
        search(state, selected);
        selected.remove(&package);
        if state.limit.is_some() {
            return;
        }
    }
}

fn validate_input(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        check_name(&root.package, format!("root[{}]", i))?;
        for (f, feature) in root.features.iter().enumerate() {
            check_name(feature, format!("root[{}].feature[{}]", i, f))?;
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        check_name(&package.name, format!("catalog[{}]", i))?;
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                check_name(
                    &dep.package,
                    format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                )?;
                if let Some(gate) = &dep.gate {
                    check_name(
                        gate,
                        format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                    )?;
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    check_name(
                        feature,
                        format!(
                            "catalog[{}].release[{}].dep[{}].feature[{}]",
                            i, j, k, f
                        ),
                    )?;
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                check_name(
                    &feature.name,
                    format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                )?;
                for (e, enabled) in feature.enables.iter().enumerate() {
                    check_name(
                        enabled,
                        format!(
                            "catalog[{}].release[{}].feature[{}].enable[{}]",
                            i, j, f, e
                        ),
                    )?;
                }
            }
        }
    }

    let mut names = BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.as_str()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        check_name(&lock.package, format!("lock[{}]", i))?;
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

fn final_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    compute_closure(roots, catalog, selected, target)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_input(catalog, roots, locks)?;

    let names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    let mut state = SearchState {
        roots,
        catalog,
        locks,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        limit: None,
    };
    search(&mut state, &mut BTreeMap::new());

    if let Some(error) = state.limit {
        return Err(error);
    }
    let selected = if let Some((_, selection)) = state.best {
        selection
    } else {
        return Err(state.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_vec = selected
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();
    let closure = final_constraints(roots, catalog, &selected, target);

    let before: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let after = selected;
    let mut changed_names = BTreeSet::new();
    changed_names.extend(before.keys().cloned());
    changed_names.extend(after.keys().cloned());

    let mut changes = Vec::new();
    for package in changed_names {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !closure
                    .constraints
                    .get(&package)
                    .into_iter()
                    .flatten()
                    .all(|c| matches(old, *c))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }
    changes.sort_by(|a, b| a.package.cmp(&b.package));
    let _ = names;
    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
