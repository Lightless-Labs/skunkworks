pub const MAX_DIM: usize = 4096;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Axis {
    Row,
    Column,
}

#[derive(Clone, Copy, Debug, PartialEq)]
pub struct Entry {
    pub row: usize,
    pub col: usize,
    pub value: f64,
}

#[derive(Clone, Debug, PartialEq)]
pub struct Sparse {
    rows: usize,
    cols: usize,
    entries: Vec<Entry>,
}

#[derive(Clone, Debug, PartialEq)]
pub enum Error {
    Shape {
        axis: Axis,
        size: usize,
    },
    Bounds {
        entry: usize,
        axis: Axis,
        index: usize,
        limit: usize,
    },
    NonFinite {
        entry: usize,
    },
    SumOverflow {
        entry: usize,
    },
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        if rows > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Row,
                size: rows,
            });
        }
        if cols > MAX_DIM {
            return Err(Error::Shape {
                axis: Axis::Column,
                size: cols,
            });
        }

        for (index, entry) in entries.iter().enumerate() {
            if entry.row >= rows {
                return Err(Error::Bounds {
                    entry: index,
                    axis: Axis::Row,
                    index: entry.row,
                    limit: rows,
                });
            }
            if entry.col >= cols {
                return Err(Error::Bounds {
                    entry: index,
                    axis: Axis::Column,
                    index: entry.col,
                    limit: cols,
                });
            }
            if !entry.value.is_finite() {
                return Err(Error::NonFinite { entry: index });
            }
        }

        let mut sums: Vec<Entry> = Vec::new();
        for (index, entry) in entries.iter().enumerate() {
            let position = sums
                .iter()
                .position(|sum| sum.row == entry.row && sum.col == entry.col);
            if let Some(position) = position {
                let value = sums[position].value + entry.value;
                if !value.is_finite() {
                    return Err(Error::SumOverflow { entry: index });
                }
                sums[position].value = value;
            } else {
                sums.push(Entry {
                    row: entry.row,
                    col: entry.col,
                    value: 0.0 + entry.value,
                });
            }
        }

        sums.retain(|entry| entry.value != 0.0);
        sums.sort_by(|a, b| a.row.cmp(&b.row).then_with(|| a.col.cmp(&b.col)));

        Ok(Self {
            rows,
            cols,
            entries: sums,
        })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }
}
