#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_arrival = 0usize;
    let mut completed = 0usize;
    let mut time = 0u64;

    while completed < customers.len() || states.iter().any(|s| !s.opened) {
        // Finish all occupied servers due at this tick, in ascending server ID order.
        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| (state.busy_until == Some(time)).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| states[i].server.id);
        for i in finishing {
            // The customer ID is carried by its scheduled finish record below.
            // Locate it from the dispatch schedule.
            let (customer, _) = scheduled_finishes
                .remove(&(time, states[i].server.id))
                .expect("occupied server has a scheduled finish");
            events.push(Event::Finish {
                time,
                customer,
                server: states[i].server.id,
            });
            states[i].busy_until = None;
            states[i].idle_since = time;
            completed += 1;
        }

        // Open servers in descending server ID order.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.server.opens == time).then_some(i)
            })
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));
        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time,
                server: states[i].server.id,
            });
        }

        // Emit the complete arrival cohort in descending customer ID order.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = &arrivals[next_arrival];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(next_arrival);
            next_arrival += 1;
        }

        // Dispatch in FIFO order, recalculating server eligibility each time.
        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &arrivals[customer_index];
            let server = &states[server_index].server;
            let finish = time
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = time;
                completed += 1;
            } else {
                states[server_index].busy_until = Some(finish);
                scheduled_finishes.insert((finish, server.id), customer.id);
            }
        }

        if completed == customers.len() && states.iter().all(|s| s.opened) {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        if next_arrival < arrivals.len() {
            consider(arrivals[next_arrival].arrival);
        }
        for state in &states {
            if !state.opened {
                consider(state.server.opens);
            }
            if let Some(finish) = state.busy_until {
                consider(finish);
            }
        }

        time = next_time.expect("unfinished simulation has a future event");
    }

    Ok(events)
}

use std::collections::BTreeMap;

static mut UNUSED: () = ();
