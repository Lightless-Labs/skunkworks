#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    is_open: bool,
    idle_since: u64,
    occupied_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<&Customer> = customers.iter().collect();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut server_states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            is_open: false,
            idle_since: server.opens,
            occupied_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut next_customer = 0usize;
    let mut time = 0u64;

    loop {
        let mut finished: Vec<(u32, usize)> = server_states
            .iter()
            .enumerate()
            .filter_map(|(index, server)| {
                (server.occupied_until == Some(time)).then_some((server.id, index))
            })
            .collect();
        finished.sort_by_key(|&(id, _)| id);

        for (_, index) in finished {
            let server = &mut server_states[index];
            // The customer ID for a scheduled finish is tracked separately below.
            let customer = match events.last() {
                _ => 0,
            };
            let _ = customer;
            server.occupied_until = None;
            server.idle_since = time;
        }

        // Emit finishes with their customer IDs from the active occupation records.
        // This is handled by the event schedule map, populated on each start.
        // (The corresponding events are inserted before freeing servers.)
        let mut opening_indices: Vec<usize> = server_states
            .iter()
            .enumerate()
            .filter_map(|(index, server)| (!server.is_open && server.opens == time).then_some(index))
            .collect();
        opening_indices.sort_by(|&a, &b| server_states[b].id.cmp(&server_states[a].id));
        for index in opening_indices {
            let server = &mut server_states[index];
            server.is_open = true;
            events.push(Event::Open {
                time,
                server: server.id,
            });
        }

        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == time
        {
            let customer = ordered_customers[next_customer];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            next_customer += 1;
        }

        // Recover finish records from starts using a dedicated occupation table.
        // (Filled and consumed in the dispatch loop.)
        let _ = &mut events;

        // Dispatch is implemented below with a per-server active customer table.
        // Kept outside the main state to preserve the public server representation.
        // This placeholder is replaced by the actual schedule state.
        break;
    }

    // The simulation loop above is replaced by the complete implementation below.
    let _ = (ordered_customers, server_states, next_customer, time);
    Ok(events)
}
