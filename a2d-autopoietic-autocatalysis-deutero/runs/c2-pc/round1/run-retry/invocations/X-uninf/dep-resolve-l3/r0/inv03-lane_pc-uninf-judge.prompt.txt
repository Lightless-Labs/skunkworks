fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    (1..=24).contains(&bytes.len())
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(w) => v == w,
        Constraint::AtLeast(w) => v >= w,
        Constraint::Before(w) => v < w,
        Constraint::Caret(w) => {
            v >= w
                && if w.major > 0 {
                    v.major == w.major
                } else if w.minor > 0 {
                    v.major == 0 && v.minor == w.minor
                } else {
                    v == w
                }
        }
    }
}

fn matches_all(v: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|&c| matches(v, c))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn validate_names(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, req) in roots.iter().enumerate() {
        if !valid_name(&req.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: req.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    Ok(())
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn package(&self, name: &str) -> Option<&'a Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn inspect(&mut self, selected: &mut Vec<Selection>) -> Result<(), ResolveError> {
        let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
            std::collections::BTreeMap::new();
        for root in self.roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for selection in selected.iter() {
            if let Some(package) = self.package(&selection.package) {
                if let Some(release) = package
                    .releases
                    .iter()
                    .find(|r| r.version == selection.version)
                {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }

        let missing = active
            .keys()
            .filter(|name| self.package(name).is_none())
            .next()
            .cloned();
        if let Some(package) = missing {
            return Err(ResolveError::Missing { package });
        }

        selected.sort_by(|a, b| a.package.cmp(&b.package));
        for selection in selected.iter() {
            if let Some(constraints) = active.get(&selection.package) {
                if !matches_all(selection.version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: selection.package.clone(),
                        selected: selection.version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if selected.iter().any(|s| s.package == *name) {
                continue;
            }
            let package = self.package(name).expect("missing checked above");
            if !package
                .releases
                .iter()
                .any(|r| matches_all(r.version, constraints))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active
            .keys()
            .find(|name| !selected.iter().any(|s| s.package == ***name))
            .cloned();
        let Some(name) = next else {
            let mut complete = selected.clone();
            complete.sort_by(|a, b| a.package.cmp(&b.package));
            self.record_solution(complete);
            return Ok(());
        };

        let constraints = active.get(&name).expect("active package");
        let package = self.package(&name).expect("missing checked above");
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|&v| matches_all(v, constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(index) = candidates.iter().position(|&v| v == lock.version) {
                let locked = candidates.remove(index);
                candidates.insert(0, locked);
            }
        }

        let mut local_first_error = None;
        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.push(Selection {
                package: name.clone(),
                version,
            });
            let result = self.inspect(selected);
            selected.retain(|s| s.package != name);
            match result {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if local_first_error.is_none() {
                        local_first_error = Some(error);
                    }
                }
            }
        }
        if let Some(error) = local_first_error {
            Err(error)
        } else {
            Ok(())
        }
    }

    fn record_solution(&mut self, selected: Vec<Selection>) {
        let versions: std::collections::BTreeMap<&str, Version> = selected
            .iter()
            .map(|s| (s.package.as_str(), s.version))
            .collect();
        let cost = self
            .locks
            .iter()
            .filter(|lock| versions.get(lock.package.as_str()) != Some(&lock.version))
            .count();

        let better = match &self.best {
            None => true,
            Some((best_cost, best)) if cost < *best_cost => true,
            Some((best_cost, best)) if cost == *best_cost => {
                let names: std::collections::BTreeSet<&str> = self
                    .catalog
                    .iter()
                    .map(|p| p.name.as_str())
                    .collect();
                for name in names {
                    let a = versions.get(name).copied();
                    let b = best
                        .iter()
                        .find(|s| s.package == name)
                        .map(|s| s.version);
                    if a != b {
                        if let (Some(a), Some(b)) = (a, b) {
                            if a > b {
                                return self.set_best(cost, selected);
                            }
                        } else if a.is_some() {
                            return self.set_best(cost, selected);
                        }
                        return;
                    }
                }
                false
            }
            _ => false,
        };
        if better {
            self.best = Some((cost, selected));
        }
    }

    fn set_best(&mut self, cost: usize, selected: Vec<Selection>) {
        self.best = Some((cost, selected));
    }
}

    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_names(catalog, roots, locks)?;

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut selected = Vec::new();
    match search.inspect(&mut selected) {
        Err(ResolveError::Limit { attempts }) => {
            return Err(ResolveError::Limit { attempts });
        }
        Err(error) => search.first_error = Some(error),
        Ok(()) => {}
    }

    let Some((_, selected)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected_map: std::collections::BTreeMap<&str, Version> = selected
        .iter()
        .map(|s| (s.package.as_str(), s.version))
        .collect();
    let lock_map: std::collections::BTreeMap<&str, Version> = locks
        .iter()
        .map(|l| (l.package.as_str(), l.version))
        .collect();

    let mut names = std::collections::BTreeSet::new();
    names.extend(selected_map.keys().copied());
    names.extend(lock_map.keys().copied());

    let mut final_constraints: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        final_constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for selection in &selected {
        if let Some(package) = catalog.iter().find(|p| p.name == selection.package) {
            if let Some(release) = package
                .releases
                .iter()
                .find(|r| r.version == selection.version)
            {
                for dep in &release.deps {
                    final_constraints
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in final_constraints.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }

    let mut changes = Vec::new();
    for name in names {
        let before = lock_map.get(name).copied();
        let after = selected_map.get(name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == name)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if !matches_all(
                    old,
                    final_constraints
                        .get(name)
                        .map(Vec::as_slice)
                        .unwrap_or(&[]),
                ) {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name.to_string(),
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty()
        || bytes.len() > 24
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            out.push(*byte as char);
        } else {
            out.push('%');
            out.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            out.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn permits(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| permits(*c, version))
}

fn validate_and_index<'a>(
    catalog: &'a [Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(std::collections::BTreeMap<String, &'a Package>, std::collections::BTreeMap<String, Version>), ResolveError> {
    use std::collections::{BTreeMap, BTreeSet};

    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if lock_map.insert(lock.package.clone(), lock.version).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let index = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect::<BTreeMap<_, _>>();
    Ok((index, lock_map))
}

#[derive(Clone)]
struct SearchResult {
    assignments: std::collections::BTreeMap<String, Version>,
    constraints: std::collections::BTreeMap<String, Vec<Constraint>>,
}

struct Search<'a> {
    catalog: &'a std::collections::BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a std::collections::BTreeMap<String, Version>,
    attempts: usize,
    budget: usize,
    first_error: Option<ResolveError>,
    best: Option<SearchResult>,
    best_cost: usize,
    catalog_names: Vec<String>,
}

impl Search<'_> {
    fn inspect(
        &mut self,
        assignments: &std::collections::BTreeMap<String, Version>,
    ) -> Result<Option<std::collections::BTreeMap<String, Vec<Constraint>>>, ResolveError> {
        use std::collections::{BTreeMap, BTreeSet};

        let mut active: BTreeMap<String, Vec<Constraint>> = BTreeMap::new();
        for root in self.roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (package, version) in assignments {
            if let Some(definition) = self.catalog.get(package) {
                if let Some(release) = definition
                    .releases
                    .iter()
                    .find(|r| r.version == *version)
                {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }

        let missing = active
            .keys()
            .filter(|name| !self.catalog.contains_key(*name))
            .cloned()
            .collect::<BTreeSet<_>>();
        if let Some(package) = missing.into_iter().next() {
            return Err(ResolveError::Missing { package });
        }

        for (package, version) in assignments {
            if let Some(constraints) = active.get(package) {
                if !matches_all(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (package, constraints) in &active {
            if assignments.contains_key(package) {
                continue;
            }
            let has_candidate = self.catalog[package]
                .releases
                .iter()
                .any(|r| matches_all(r.version, constraints));
            if !has_candidate {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.clone(),
                });
            }
        }
        Ok(Some(active))
    }

    fn solution_cost(&self, assignments: &std::collections::BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(package, version)| assignments.get(*package) != Some(*version))
            .count()
    }

    fn better(
        &self,
        candidate: &std::collections::BTreeMap<String, Version>,
        candidate_cost: usize,
    ) -> bool {
        let Some(best) = &self.best else {
            return true;
        };
        if candidate_cost != self.best_cost {
            return candidate_cost < self.best_cost;
        }
        for name in &self.catalog_names {
            let a = candidate.get(name);
            let b = best.assignments.get(name);
            match (a, b) {
                (Some(a), Some(b)) if a != b => return a > b,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }

    fn remember(
        &mut self,
        assignments: &std::collections::BTreeMap<String, Version>,
        constraints: &std::collections::BTreeMap<String, Vec<Constraint>>,
    ) {
        let cost = self.solution_cost(assignments);
        if self.better(assignments, cost) {
            self.best_cost = cost;
            self.best = Some(SearchResult {
                assignments: assignments.clone(),
                constraints: constraints.clone(),
            });
        }
    }

    fn search(&mut self, assignments: &mut std::collections::BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = match self.inspect(assignments) {
            Ok(Some(active)) => active,
            Ok(None) => return Ok(()),
            Err(error) => {
                if self.first_error.is_none() {
                    self.first_error = Some(error);
                }
                return Ok(());
            }
        };

        let next = active
            .keys()
            .find(|package| !assignments.contains_key(*package))
            .cloned();
        let Some(package) = next else {
            self.remember(assignments, &active);
            return Ok(());
        };

        let constraints = &active[&package];
        let mut candidates = self.catalog[&package]
            .releases
            .iter()
            .filter(|r| matches_all(r.version, constraints))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(locked) = self.locks.get(&package) {
            if let Some(position) = candidates.iter().position(|v| v == locked) {
                let version = candidates.remove(position);
                candidates.insert(0, version);
            }
        }

        if candidates.is_empty() {
            let error = ResolveError::NoMatch {
                package,
                requirements: constraints.clone(),
            };
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        for candidate in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            assignments.insert(package.clone(), candidate);
            let result = self.search(assignments);
            assignments.remove(&package);
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    use std::collections::{BTreeMap, BTreeSet};

    let (catalog_index, lock_map) = validate_and_index(catalog, roots, locks)?;
    let mut names = catalog_index.keys().cloned().collect::<Vec<_>>();
    names.sort();

    let mut search = Search {
        catalog: &catalog_index,
        roots,
        locks: &lock_map,
        attempts: 0,
        budget: max_attempts,
        first_error: None,
        best: None,
        best_cost: usize::MAX,
        catalog_names: names,
    };

    search.search(&mut BTreeMap::new())?;
    let Some(best) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        }));
    };

    let selected = best
        .assignments
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut all_names = BTreeSet::new();
    all_names.extend(lock_map.keys().cloned());
    all_names.extend(best.assignments.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = best.assignments.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(locked), Some(_)) => {
                let exists = catalog_index
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == locked));
                if !exists {
                    ChangeReason::Unavailable
                } else if best
                    .constraints
                    .get(&package)
                    .is_some_and(|constraints| !matches_all(locked, constraints))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
