use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut field_started = false;
    let mut quote_open = 0usize;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: i });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                after_quote = false;
                field_started = false;
                i += 1;
            } else if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                after_quote = false;
                field_started = false;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record: record_index, byte: i });
            }
            continue;
        }

        if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
            }
            in_quotes = true;
            quote_open = i;
            field_started = true;
            i += 1;
        } else if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            field_started = false;
            i += if b == b'\r' { 2 } else { 1 };
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            field_started = true;
            i += ch.len_utf8();
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_open });
    }

    if !record.is_empty() || !field.is_empty() || field_started || after_quote {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|h| h == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|h| h == name)
            }
        }
    }
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name, second);
    }
    None
}

fn table_from_records(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut it = records.into_iter();
            (it.next(), it.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(h) = header.as_ref() {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(h) {
                return Err(error);
            }
        }
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    UnknownColumn,
    MissingCell,
    Empty,

    IntegerOverflow,

    InvalidBoolean,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start == 0 && b.len() == 1;
    }
    (b'1'..=b'9').contains(&b[start]) && b[start..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if b.len() < start + 4 || b.get(b.len() - 3) != Some(&b'.') {
        return false;
    }
    let integral = &b[start..b.len() - 3];
    if integral.is_empty() || !b[b.len() - 2..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    if integral == b"0" {
        return start == 0 || b[b.len() - 2..] != *b"00";
    }
    (b'1'..=b'9').contains(&integral[0]) && integral.iter().all(u8::is_ascii_digit)
}

fn typed_value(s: &str, kind: ColumnType, nullable: bool) -> Result<Value, CellErrorKind> {
    if s.is_empty() {
        if nullable {
            return Ok(Value::Null);
        }
        if kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }
    match kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(s) {
                return Err(CellErrorKind::InvalidInteger);
            }
            s.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) {
                return Err(CellErrorKind::InvalidDecimal);
            }
            let negative = s.starts_with('-');
            let body = if negative { &s[1..] } else { s };
            let dot = body.len() - 3;
            let digits = format!("{}{}", &body[..dot], &body[dot + 1..]);

            // Parse the magnitude against the sign-specific limit. The negative
            // limit is one greater than i64::MAX, allowing i64::MIN exactly.
            let limit: u64 = if negative {
                (i64::MAX as u64) + 1
            } else {
                i64::MAX as u64
            };
            let mut magnitude = 0u64;
            for digit in digits.bytes() {
                magnitude = magnitude
                    .checked_mul(10)
                    .and_then(|n| n.checked_add((digit - b'0') as u64))
                    .filter(|n| *n <= limit)
                    .ok_or(CellErrorKind::DecimalOverflow)?;
            }
            if negative {
                if magnitude == (i64::MAX as u64) + 1 {
                    Ok(Value::Decimal(i64::MIN))
                } else {
                    Ok(Value::Decimal(-(magnitude as i64)))
                }
            } else {
                Ok(Value::Decimal(magnitude as i64))
            }
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, value) = match resolved {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(i) => match row.get(i) {
                    None => (Some(i), Err(CellErrorKind::MissingCell)),
                    Some(cell) => (Some(i), typed_value(cell, spec.kind, spec.nullable)),
                },
            };
            match value {
                Ok(v) => output[spec_index] = Some(v),
                Err(kind) => {
                    result.errors.push(CellError { row: row_index, spec: spec_index, column, kind });
                    failed = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }
        result.rows.push(output);
        if failed && policy == ErrorPolicy::Stop {
            break;
        }
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding { Lf, CrLf }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode { Preserve, Canonical }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn write_duplicate(header: &[String]) -> Option<WriteError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(WriteError::DuplicateHeader { name: name.clone(), first, second });
        }
        seen.insert(name, second);
    }
    None
}

fn quote_field(field: &str, delimiter: u8, quote: u8, first_field: bool, sole: bool) -> String {
    let bytes = field.as_bytes();
    let quote_it = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field && field.starts_with('\u{feff}'))
        || (sole && field.is_empty());
    if !quote_it {
        return field.to_string();
    }
    let mut out = String::new();
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
        }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

fn serialize_record(fields: &[String], delimiter: u8, quote: u8, record_index: usize) -> String {
    let mut out = String::new();
    for (i, field) in fields.iter().enumerate() {
        if i > 0 {
            out.push(delimiter as char);
        }
        out.push_str(&quote_field(field, delimiter, quote, record_index == 0 && i == 0, fields.len() == 1));
    }
    out
}

fn row_cmp(a: &Vec<String>, b: &Vec<String>) -> std::cmp::Ordering {
    a.cmp(b)
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut originals: Vec<(usize, &Vec<String>)> = Vec::new();
    if let Some(header) = table.header.as_ref() {
        originals.push((0, header));
    }
    for (i, row) in table.rows.iter().enumerate() {
        originals.push((if table.header.is_some() { i + 1 } else { i }, row));
    }

    for (record, row) in &originals {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul { record: *record, column, byte });
            }
        }
    }

    if let Some(header) = table.header.as_ref() {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in &originals {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record: *record });
        }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some(error) = write_duplicate(table.header.as_ref().unwrap()) {
            return Err(error);
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(h) = &table.header { output_records.push(h.clone()); }
            output_records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut chosen: HashMap<&str, usize> = HashMap::new();
                for (i, name) in header.iter().enumerate() {
                    match table.duplicates {
                        DuplicatePolicy::KeepLast => { chosen.insert(name, i); }
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => { chosen.entry(name).or_insert(i); }
                    }
                }
                let mut indexes: Vec<usize> = chosen.into_values().collect();
                indexes.sort_by(|a, b| header[*a].as_bytes().cmp(header[*b].as_bytes()));
                let projected_header: Vec<String> = indexes.iter().map(|i| header[*i].clone()).collect();
                output_records.push(projected_header);
                let mut projected_rows: Vec<Vec<String>> = table.rows.iter()
                    .map(|r| indexes.iter().map(|i| r[*i].clone()).collect())
                    .collect();
                projected_rows.sort_by(row_cmp);
                output_records.extend(projected_rows);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(row_cmp);
                output_records = rows;
            }
        }
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    for (i, record) in output_records.iter().enumerate() {
        if i > 0 {
            out.push_str(match config.mode {
                WriteMode::Canonical => "\n",
                WriteMode::Preserve => match config.line_ending { LineEnding::Lf => "\n", LineEnding::CrLf => "\r\n" },
            });
        }
        out.push_str(&serialize_record(record, config.delimiter, config.quote, i));
    }
    let final_terminator = config.mode == WriteMode::Canonical || config.final_terminator;
    if final_terminator {
        out.push_str(match config.mode {
            WriteMode::Canonical => "\n",
            WriteMode::Preserve => match config.line_ending { LineEnding::Lf => "\n", LineEnding::CrLf => "\r\n" },
        });
    }
    Ok(out)
}
