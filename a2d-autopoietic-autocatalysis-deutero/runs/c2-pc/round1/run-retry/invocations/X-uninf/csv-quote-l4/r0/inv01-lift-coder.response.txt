#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart { Delimiter, Quote }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy { Reject, KeepFirst, KeepLast }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut field_started = false;

    enum State { Start, Unquoted, Quoted, Closed }
    let mut state = State::Start;
    let mut quote_start = 0usize;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if !matches!(state, State::Quoted) && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record: record_index, byte: i });
            }
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            state = State::Start;
            field_started = false;
            i += 2;
            record_index += 1;
            continue;
        }

        match state {
            State::Start => {
                if b == delimiter {
                    record.push(String::new());
                    i += 1;
                } else if b == quote {
                    quote_start = i;
                    field_started = true;
                    state = State::Quoted;
                    i += 1;
                } else if b == b'\n' {
                    record.push(String::new());
                    records.push(std::mem::take(&mut record));
                    i += 1;
                    record_index += 1;
                    field_started = false;
                } else {
                    field.push(b as char);
                    if b >= 0x80 {
                        let ch = input[i..].chars().next().unwrap();
                        field.pop();
                        field.push(ch);
                        i += ch.len_utf8();
                    } else {
                        i += 1;
                    }
                    field_started = true;
                    state = State::Unquoted;
                }
            }
            State::Unquoted => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    state = State::Start;
                    field_started = false;
                    i += 1;
                    record_index += 1;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
                } else {
                    if b >= 0x80 {
                        let ch = input[i..].chars().next().unwrap();
                        field.push(ch);
                        i += ch.len_utf8();
                    } else {
                        field.push(b as char);
                        i += 1;
                    }
                }
            }
            State::Quoted => {
                if b == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::Closed;
                        i += 1;
                    }
                } else if b >= 0x80 {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                } else {
                    field.push(b as char);
                    i += 1;
                }
            }
            State::Closed => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    state = State::Start;
                    field_started = false;
                    i += 1;
                    record_index += 1;
                } else {
                    return Err(CsvError::TrailingByte { record: record_index, byte: i });
                }
            }
        }
    }

    if matches!(state, State::Quoted) {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_start });
    }
    if !record.is_empty() || field_started || !field.is_empty() || !matches!(state, State::Start) {
        record.push(field);
        records.push(record);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject =>
                header.iter().position(|s| s == name),
        }
    }
}

fn table_from_records(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut iter = records.into_iter();
    let header = if config.headers { iter.next() } else { None };
    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen = std::collections::HashMap::<String, usize>::new();
            for (second, name) in h.iter().enumerate() {
                if let Some(&first) = seen.get(name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(), first, second,
                    });
                }
                seen.insert(name.clone(), second);
            }
        }
    }
    let rows: Vec<Vec<String>> = iter.collect();
    if let Some(ref h) = header {
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1, expected: h.len(), actual: row.len(),
                });
            }
        }
    }
    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), cached: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef { Index(usize), Name(String) }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType { Text, Integer, Decimal, Boolean }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn, MissingCell, Empty, InvalidInteger, IntegerOverflow,
    InvalidDecimal, DecimalOverflow, InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy { Collect, FirstPerRow, Stop }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start == 0 && b.len() == 1;
    }
    if !(b'1'..=b'9').contains(&b[start]) { return false; }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start >= b.len() { return false; }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    if dot == start || b.len() != dot + 3 { return false; }
    let integral = &b[start..dot];
    if integral == b"0" {
        if start == 1 { return false; }
    } else if integral.is_empty()
        || !(b'1'..=b'9').contains(&integral[0])
        || !integral[1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    b[dot + 1..].iter().all(u8::is_ascii_digit)
}

fn evaluate_cell(s: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if s.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Integer => {
            if !integer_syntax(s) { return Err(CellErrorKind::InvalidInteger); }
            s.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = s.starts_with('-');
            let unsigned = if neg { &s[1..] } else { s };
            let mut digits = unsigned.replace('.', "");
            if neg { digits.insert(0, '-'); }
            digits.parse::<i64>().map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        }
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut rows = Vec::new();
    let mut errors = Vec::new();
    for (row_idx, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut failed = false;
        for (spec_idx, spec) in columns.iter().enumerate() {
            if failed {
                out.push(None);
                continue;
            }
            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let result = match resolved {
                None => Err((None, CellErrorKind::UnknownColumn)),
                Some(col) => match row.get(col) {
                    None => Err((Some(col), CellErrorKind::MissingCell)),
                    Some(value) => evaluate_cell(value, spec).map_err(|kind| (Some(col), kind)),
                }
            };
            match result {
                Ok(value) => out.push(Some(value)),
                Err((col, kind)) => {
                    out.push(None);
                    errors.push(CellError { row: row_idx, spec: spec_idx, column: col, kind });
                    if policy != ErrorPolicy::Collect { failed = true; }
                    if policy == ErrorPolicy::Stop {
                        rows.push(out);
                        return TypedResult { rows, errors };
                    }
                }
            }
        }
        rows.push(out);
    }
    TypedResult { rows, errors }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding { Lf, CrLf }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode { Preserve, Canonical }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header { original.push(header); }
    original.extend(table.rows.iter());

    for (record, row) in original.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul { record, column, byte });
            }
        }
    }
    if let Some(header) = &table.header {
        for (idx, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: idx + 1, expected: header.len(), actual: row.len(),
                });
            }
        }
    }
    for (record, row) in original.iter().enumerate() {
        if row.is_empty() { return Err(WriteError::ZeroColumns { record }); }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header { output_records.push(header.clone()); }
            output_records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut seen = std::collections::HashMap::<String, usize>::new();
                for (idx, name) in header.iter().enumerate() {
                    if let Some(&first) = seen.get(name) {
                        if table.duplicates == DuplicatePolicy::Reject {
                            return Err(WriteError::DuplicateHeader {
                                name: name.clone(), first, second: idx,
                            });
                        }
                    } else {
                        seen.insert(name.clone(), idx);
                    }
                }
                let mut retained: Vec<usize> = Vec::new();
                for (idx, name) in header.iter().enumerate() {
                    let selected = match table.duplicates {
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject =>
                            seen.get(name) == Some(&idx),
                        DuplicatePolicy::KeepLast =>
                            header.iter().rposition(|n| n == name) == Some(idx),
                    };
                    if selected { retained.push(idx); }
                }
                retained.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));
                output_records.push(retained.iter().map(|&i| header[i].clone()).collect());
                let mut projected: Vec<Vec<String>> = table.rows.iter()
                    .map(|row| retained.iter().map(|&i| row[i].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
                output_records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| a.iter().map(String::as_bytes).cmp(b.iter().map(String::as_bytes)));
                output_records = rows;
            }
        }
    }

    if output_records.is_empty() { return Ok(String::new()); }
    let mut out = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let ending = match config.line_ending {
        LineEnding::Lf => "\n",
        LineEnding::CrLf => "\r\n",
    };
    for (ri, row) in output_records.iter().enumerate() {
        if ri > 0 {
            out.push_str(if canonical { "\n" } else { ending });
        }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 { out.push(config.delimiter as char); }
            let bytes = field.as_bytes();
            let quote_it = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || bytes.first().map_or(false, |&b| b == b' ' || b == b'\t')
                || bytes.last().map_or(false, |&b| b == b' ' || b == b'\t')
                || (ri == 0 && ci == 0 && field.starts_with('\u{feff}'))
                || (row.len() == 1 && field.is_empty());
            if quote_it {
                out.push(config.quote as char);
                for ch in field.chars() {
                    if ch as u32 == config.quote as u32 {
                        out.push(config.quote as char);
                    }
                    out.push(ch);
                }
                out.push(config.quote as char);
            } else {
                out.push_str(field);
            }
        }
    }
    if canonical || config.final_terminator {
        out.push_str(if canonical { "\n" } else { ending });
    }
    Ok(out)
}
