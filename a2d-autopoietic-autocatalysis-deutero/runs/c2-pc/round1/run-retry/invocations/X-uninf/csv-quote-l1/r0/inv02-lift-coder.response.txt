#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut index = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if index == bytes.len() {
        return Ok(Vec::new());
    }

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = State::FieldStart;
    let mut record_index = 0usize;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: index,
            });
        }

        let outside_quotes = !matches!(state, State::Quoted { .. });

        if outside_quotes && byte == b'\r' {
            if bytes.get(index + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: index,
                });
            }

            match state {
                State::FieldStart | State::Unquoted | State::AfterQuote => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    index += 2;
                    continue;
                }
                State::Quoted { .. } => unreachable!(),
            }
        }

        if outside_quotes && byte == b'\n' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            state = State::FieldStart;
            index += 1;
            continue;
        }

        match state {
            State::FieldStart => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    index += 1;
                }
                b'"' => {
                    state = State::Quoted { opening: index };
                    index += 1;
                }
                _ => {
                    let start = index;
                    while index < bytes.len()
                        && !matches!(bytes[index], b',' | b'\r' | b'\n' | b'"' | 0)
                    {
                        index += 1;
                    }
                    field.push_str(&input[start..index]);
                    state = State::Unquoted;
                }
            },
            State::Unquoted => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    index += 1;
                }
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: index,
                    });
                }
                _ => {
                    let start = index;
                    while index < bytes.len()
                        && !matches!(bytes[index], b',' | b'\r' | b'\n' | b'"' | 0)
                    {
                        index += 1;
                    }
                    field.push_str(&input[start..index]);
                }
            },
            State::Quoted { opening } => {
                if byte == b'"' {
                    if bytes.get(index + 1) == Some(&b'"') {
                        field.push('"');
                        index += 2;
                    } else {
                        state = State::AfterQuote;
                        index += 1;
                    }
                } else {
                    let start = index;
                    while index < bytes.len()
                        && bytes[index] != b'"'
                        && bytes[index] != 0
                    {
                        index += 1;
                    }
                    field.push_str(&input[start..index]);
                    if index == bytes.len() {
                        return Err(CsvError::UnclosedQuote {
                            record: record_index,
                            byte: opening,
                        });
                    }
                }
            }
            State::AfterQuote => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    index += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: index,
                    });
                }
            },
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if !matches!(state, State::FieldStart) || !record.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}
