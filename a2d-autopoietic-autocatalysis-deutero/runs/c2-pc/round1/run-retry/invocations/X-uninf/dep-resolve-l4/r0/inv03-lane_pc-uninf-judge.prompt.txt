fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

    out
}

        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn target_matches(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

fn target_text(platform: Platform) -> &'static str {
    match platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    active_deps: BTreeMap<String, BTreeSet<usize>>,
}

fn append_token(path: &[String], token: String) -> Vec<String> {
    let mut p = path.to_vec();
    p.push(token);
    p
}

fn canonical_paths_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn closure_for(
    roots: &[Requirement],
    catalog: &BTreeMap<String, &Package>,
    chosen: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut c = Closure::default();
    for (i, root) in roots.iter().enumerate() {
        let mut fs = root.features.clone();
        fs.sort();
        fs.dedup();
        let ftext = fs.join(",");
        let token = format!(
            "root[{i}]:{}@{}+{{{ftext}}}",
            root.package,
            constraint_text(root.constraint)
        );
        c.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        c.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        c.requests
            .entry(root.package.clone())
            .or_default()
            .extend(fs);
        let p = vec![token];
        match c.paths.get(&root.package) {
            Some(old) if !canonical_paths_better(&p, old) => {}
            _ => {
                c.paths.insert(root.package.clone(), p);
            }
        }
    }

    loop {
        let mut changed = false;

        let selected_names: Vec<String> = chosen.keys().cloned().collect();
        for name in selected_names {
            let Some(package) = catalog.get(&name) else {
                continue;
            };
            let version = chosen[&name];
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let mut feature_map: BTreeMap<&str, &Feature> = BTreeMap::new();
            for feature in &release.features {
                feature_map.insert(&feature.name, feature);
            }

            let requests = c.requests.get(&name).cloned().unwrap_or_default();
            let activated = c.activated.entry(name.clone()).or_default();
            for f in requests {
                if activated.insert(f) {
                    changed = true;
                }
            }

            let active_names = c.activated.get(&name).cloned().unwrap_or_default();
            for feature_name in active_names {
                if let Some(feature) = feature_map.get(feature_name.as_str()) {
                    let set = c.requests.entry(name.clone()).or_default();
                    for enabled in &feature.enables {
                        if set.insert(enabled.clone()) {
                            changed = true;
                        }
                    }
                }
            }

            let path = c.paths.get(&name).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !c.activated.get(&name).is_some_and(|s| s.contains(g)))
                {
                    continue;
                }
                if c.active_deps.entry(name.clone()).or_default().insert(j) {
                    changed = true;
                }
                let mut fs = dep.features.clone();
                fs.sort();
                fs.dedup();
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    name,
                    version_text(version),
                    j,
                    dep.package,
                    constraint_text(dep.constraint),
                    fs.join(","),
                    dep.gate.as_deref().unwrap_or("*"),
                    target_text(dep.platform)
                );
                let dep_path = append_token(&path, token);
                c.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                let origins = c.origins.entry(dep.package.clone()).or_default();
                if !origins.iter().any(|o| {
                    o.constraint == dep.constraint && o.path == dep_path
                }) {
                    origins.push(Origin {
                        constraint: dep.constraint,
                        path: dep_path.clone(),
                    });
                    changed = true;
                }
                let reqs = c.requests.entry(dep.package.clone()).or_default();
                for f in fs {
                    if reqs.insert(f) {
                        changed = true;
                    }
                }
                match c.paths.get(&dep.package) {
                    Some(old) if !canonical_paths_better(&dep_path, old) => {}
                    _ => {
                        if c.paths.get(&dep.package) != Some(&dep_path) {
                            c.paths.insert(dep.package.clone(), dep_path);
                            changed = true;
                        }
                    }
                }
            }
        }
        if !changed {
            break;
        }
    }

    for constraints in c.constraints.values_mut() {
        *constraints = canonical_constraints(std::mem::take(constraints));
    }
    c
}

fn evidence_for(c: &Closure, package: &str) -> Vec<Evidence> {
    let mut out: Vec<Evidence> = c
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, &'a Package>,
    target: Target,
    locks: &'a BTreeMap<String, Version>,
    attempts: usize,
    max_attempts: usize,
    best: Option<(usize, Vec<Version>, BTreeMap<String, Version>, Closure)>,
    catalog_names: Vec<String>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn inspect(&mut self, chosen: &BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let c = closure_for(self.roots, self.catalog, chosen, self.target);

        for package in c.constraints.keys() {
            if !self.catalog.contains_key(package) {
                return Err(ResolveError::Missing {
                    package: package.clone(),
                    evidence: evidence_for(&c, package),
                });
            }
        }

        for (package, version) in chosen {
            let Some(requirements) = c.constraints.get(package) else {
                continue;
            };
            if !requirements.iter().all(|r| matches(*version, *r)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: requirements.clone(),
                    evidence: evidence_for(&c, package),
                });
            }
        }

        for (package, version) in chosen {
            let Some(pkg) = self.catalog.get(package) else {
                continue;
            };
            let release = pkg.releases.iter().find(|r| r.version == *version).unwrap();
            let declared: BTreeSet<&str> =
                release.features.iter().map(|f| f.name.as_str()).collect();
            if let Some(feature) = c
                .activated
                .get(package)
                .into_iter()
                .flatten()
                .find(|f| !declared.contains(f.as_str()))
            {
                return Err(ResolveError::MissingFeature {
                    package: package.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: c.paths.get(package).cloned().unwrap_or_default(),
                });
            }
        }

        for (package, requirements) in &c.constraints {
            if chosen.contains_key(package) {
                continue;
            }
            let pkg = self.catalog[package];
            if !pkg.releases.iter().any(|r| {
                requirements.iter().all(|constraint| matches(r.version, *constraint))
            }) {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: requirements.clone(),
                    evidence: evidence_for(&c, package),
                });
            }
        }

        let next = c
            .constraints
            .keys()
            .find(|package| !chosen.contains_key(*package))
            .cloned();
        let Some(package) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| chosen.get(*name) != Some(*version))
                .count();
            let vector = self
                .catalog_names
                .iter()
                .map(|name| chosen.get(name).copied().unwrap_or(Version {
                    major: 0,
                    minor: 0,
                    patch: 0,
                }))
                .collect::<Vec<_>>();
            let replace = self.best.as_ref().is_none_or(|(bcost, bvec, ..)| {
                cost < *bcost || (cost == *bcost && vector > *bvec)
            });
            if replace {
                self.best = Some((cost, vector, chosen.clone(), c));
            }
            return Ok(());
        };

        let requirements = c.constraints[&package].clone();
        let pkg = self.catalog[&package];
        let mut candidates: Vec<Version> = pkg
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| requirements.iter().all(|r| matches(*v, *r)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(&package) {
            if let Some(i) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        let mut local_first_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            let mut child = chosen.clone();
            child.insert(package.clone(), candidate);
            match self.inspect(&child) {
                Ok(()) => {}
                Err(e @ ResolveError::Limit { .. }) => return Err(e),
                Err(e) => {
                    if local_first_error.is_none() {
                        local_first_error = Some(e);
                    }
                }
            }
        }
        if self.best.is_some() {
            Ok(())
        } else if let Some(e) = local_first_error {
            if self.first_error.is_none() {
                self.first_error = Some(e.clone());
            }
            Err(e)
        } else {
            Err(ResolveError::NoMatch {
                package,
                requirements,
                evidence: evidence_for(&c, &package),
            })
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: feature.clone(),
                });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{i}].release[{j}].dep[{k}]");
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base,
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].gate"),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].feature[{f}]"),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].feature[{f}]"),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{i}].release[{j}].feature[{f}].enable[{e}]"
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut names = BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }
    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    let mut search = Search {
        roots,
        catalog: &catalog_map,
        target,
        locks: &lock_map,
        attempts: 0,
        max_attempts,
        best: None,
        catalog_names,
        first_error: None,
    };

    match search.inspect(&BTreeMap::new()) {
        Err(ResolveError::Limit { attempts }) => return Err(ResolveError::Limit { attempts }),
        Err(error) => return Err(search.first_error.unwrap_or(error)),
        Ok(()) => {}
    }
    let Some((_, _, chosen, final_closure)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let mut selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    selected.sort_by(|a, b| a.package.cmp(&b.package));

    let before: BTreeMap<String, Version> = lock_map;
    let after = chosen;
    let mut all: BTreeSet<String> = before.keys().cloned().collect();
    all.extend(after.keys().cloned());
    let mut changes = Vec::new();
    for package in all {
        let b = before.get(&package).copied();
        let a = after.get(&package).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_closure
                    .constraints
                    .get(&package)
                    .into_iter()
                    .flatten()
                    .all(|c| matches(old, *c))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

            .then_with(|| a.path.cmp(&b.path))
    });
    v
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn evidence_lines(ev: &[Evidence]) -> String {
    evidence_sorted(ev)
        .iter()
        .map(|e| {
            format!(
                "\n  C:{} via {}",
                constraint_text(e.constraint),
                e.path.join(" -> ")
            )
        })
        .collect()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

        Target::Linux => "linux",
        Target::Mac => "mac",
        Target::Windows => "windows",

fn platform_active(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Closure {
    constraints: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    active_features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        fs.join(",")
    )
}

fn dep_token(src: &str, version: Version, j: usize, d: &Dependency) -> String {
    let mut fs = d.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        src,
        version_text(version),
        j,
        d.package,
        constraint_text(d.constraint),
        fs.join(","),
        d.gate.as_deref().unwrap_or("*"),
        match d.platform {
            Platform::Any => "any",
            Platform::Linux => "linux",
            Platform::Mac => "mac",
            Platform::Windows => "windows",
        }
    )
}

fn compute_closure(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut cl = Closure::default();
    for (i, r) in roots.iter().enumerate() {
        let p = root_token(i, r);
        cl.constraints
            .entry(r.package.clone())
            .or_default()
            .push(Origin {
                constraint: r.constraint,
                path: vec![p.clone()],
            });
        cl.requests
            .entry(r.package.clone())
            .or_default()
            .extend(r.features.iter().cloned());
        let paths = cl.paths.entry(r.package.clone()).or_default();
        if paths.is_empty() || path_better(&vec![p.clone()], paths) {
            *paths = vec![p];
        }
    }

    loop {
        let before = cl.clone();

        for (pkg, ver) in selected {
            if let Some(package) = catalog.get(pkg) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *ver) {
                    let reqs = cl.requests.get(pkg).cloned().unwrap_or_default();
                    let activated = cl.active_features.entry(pkg.clone()).or_default();
                    let mut queue: Vec<String> = reqs.into_iter().collect();
                    while let Some(f) = queue.pop() {
                        if !activated.insert(f.clone()) {
                            continue;
                        }
                        if let Some(decl) = release.features.iter().find(|d| d.name == f) {
                            for enabled in &decl.enables {
                                if !activated.contains(enabled) {
                                    queue.push(enabled.clone());
                                }
                            }
                        }
                    }
                }
            }
        }

        let active: Vec<(String, Version)> = selected
            .iter()
            .map(|(p, v)| (p.clone(), *v))
            .collect();
        for (src, ver) in active {
            let Some(package) = catalog.get(&src) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == ver) else {
                continue;
            };
            let features = cl.active_features.get(&src).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_active(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !features.contains(g))
                {
                    continue;
                }
                let token = dep_token(&src, ver, j, dep);
                let mut path = cl.paths.get(&src).cloned().unwrap_or_default();
                path.push(token);
                cl.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                cl.requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
                let dest_path = cl.paths.entry(dep.package.clone()).or_default();
                if dest_path.is_empty() || path_better(&path, dest_path) {
                    *dest_path = path;
                }
            }
        }

        if cl.constraints == before.constraints
            && cl.requests == before.requests
            && cl.active_features == before.active_features
            && cl.paths == before.paths
        {
            break;
        }
    }
    cl
}

fn requirements_for(cl: &Closure, package: &str) -> Vec<Constraint> {
    canonical_constraints(
        &cl.constraints
            .get(package)
            .into_iter()
            .flatten()
            .map(|o| o.constraint)
            .collect::<Vec<_>>(),
    )
}

fn evidence_for(cl: &Closure, package: &str) -> Vec<Evidence> {
    cl.constraints
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect()
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    catalog_names: Vec<String>,
    best: Option<(usize, Vec<Option<Version>>, BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let cl = compute_closure(self.catalog, self.roots, selected, self.target);

        for name in cl.constraints.keys() {
            if !self.catalog.contains_key(name) {
                return Err(ResolveError::Missing {
                    package: name.clone(),
                    evidence: evidence_for(&cl, name),
                });
            }
        }

        for (pkg, ver) in selected.iter() {
            let reqs = requirements_for(&cl, pkg);
            if reqs.iter().any(|c| !matches(*c, *ver)) {
                return Err(ResolveError::Conflict {
                    package: pkg.clone(),
                    selected: *ver,
                    requirements: reqs,
                    evidence: evidence_for(&cl, pkg),
                });
            }
        }

        for (pkg, ver) in selected.iter() {
            let Some(package) = self.catalog.get(pkg) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == *ver) else {
                continue;
            };
            if let Some(features) = cl.active_features.get(pkg) {
                for feature in features {
                    if !release.features.iter().any(|f| &f.name == feature) {
                        return Err(ResolveError::MissingFeature {
                            package: pkg.clone(),
                            selected: *ver,
                            feature: feature.clone(),
                            path: cl.paths.get(pkg).cloned().unwrap_or_default(),
                        });
                    }
                }
            }
        }

        let mut active: Vec<String> = cl.constraints.keys().cloned().collect();
        active.sort();
        for pkg in &active {
            if selected.contains_key(pkg) {
                continue;
            }
            let reqs = requirements_for(&cl, pkg);
            let package = self.catalog[pkg];
            if !package.releases.iter().any(|r| {
                reqs.iter().all(|c| matches(*c, r.version))
            }) {
                return Err(ResolveError::NoMatch {
                    package: pkg.clone(),
                    requirements: reqs,
                    evidence: evidence_for(&cl, pkg),
                });
            }
        }

        let next = active.into_iter().find(|p| !selected.contains_key(p));
        let Some(pkg) = next else {
            self.consider(selected, &cl);
            return Ok(());
        };

        let reqs = requirements_for(&cl, &pkg);
        let package = self.catalog[&pkg];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| reqs.iter().all(|c| matches(*c, *v)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == pkg) {
            if let Some(i) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(pkg.clone(), candidate);
            match self.visit(selected) {
                Ok(()) => {}
                Err(e @ ResolveError::Limit { .. }) => {
                    selected.remove(&pkg);
                    return Err(e);
                }
                Err(e) => self.record_error(e),
            }
            selected.remove(&pkg);
        }

        if self.best.is_some() {
            Ok(())
        } else {
            Err(self.first_error.clone().unwrap_or_else(|| ResolveError::NoMatch {
                package: pkg.clone(),
                requirements: reqs,
                evidence: evidence_for(&cl, &pkg),
            }))
        }
    }

    fn consider(&mut self, selected: &BTreeMap<String, Version>, cl: &Closure) {
        let cost = self
            .locks
            .iter()
            .filter(|l| selected.get(&l.package) != Some(&l.version))
            .count();
        let vector: Vec<Option<Version>> = self
            .catalog_names
            .iter()
            .map(|p| selected.get(p).copied())
            .collect();
        let better = self.best.as_ref().is_none_or(|(bc, bv, _, _)| {
            cost < *bc || (cost == *bc && vector > *bv)
        });
        if better {
            self.best = Some((cost, vector, selected.clone(), cl.clone()));
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: r.package.clone(),
            });
        }
        for (f, name) in r.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: p.name.clone(),
            });
        }
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: d.package.clone(),
                    });
                }
                if let Some(gate) = &d.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in d.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in r.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for p in catalog {
        if !seen_packages.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut seen_versions = BTreeSet::new();
        for r in &p.releases {
            if !seen_versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for p in catalog {
        for r in &p.releases {
            let mut seen = BTreeSet::new();
            for f in &r.features {
                if !seen.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }

    for (i, l) in locks.iter().enumerate() {
        if !valid_name(&l.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: l.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    for l in locks {
        if !seen_locks.insert(l.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: l.package.clone(),
            });
        }
    }

    let map: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let mut names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    names.sort();

    let mut search = Search {
        catalog: &map,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        catalog_names: names,
        best: None,
        first_error: None,
    };

    let mut selected = BTreeMap::new();
    match search.visit(&mut selected) {
        Err(e @ ResolveError::Limit { .. }) => return Err(e),
        Err(e) => {
            if search.best.is_none() {
                return Err(search.first_error.unwrap_or(e));
            }
        }
        Ok(()) => {}
    }

    let Some((_, _, chosen, final_closure)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: vec![],
            evidence: vec![],
        }));
    };

    let selected_out = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let lock_map: BTreeMap<&str, Version> = locks
        .iter()
        .map(|l| (l.package.as_str(), l.version))
        .collect();
    let mut all_names: BTreeSet<String> = lock_map.keys().map(|s| (*s).to_string()).collect();
    all_names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(package.as_str()).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if requirements_for(&final_closure, &package)
                    .iter()
                    .any(|c| !matches(*c, old))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}

        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn matches_all(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| matches(v, *c))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sorted_evidence(e: &[Evidence]) -> Vec<Evidence> {
    let mut out = e.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(out: &mut String, evidence: &[Evidence]) {
    for e in sorted_evidence(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut out = format!("missing|package={}", encode(package));
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let constraints = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut out = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints
            );
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let constraints = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut out = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints
            );
            append_evidence(&mut out, evidence);
            out
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct PackageInfo {
    constraints: Vec<Constraint>,
    origins: Vec<Origin>,
    features: Vec<String>,
    paths: Vec<Vec<String>>,
}

#[derive(Clone)]
struct Closure {
    info: std::collections::BTreeMap<String, PackageInfo>,
    activated: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
}

struct Search<'a> {
    catalog: &'a [Package],
    index: std::collections::BTreeMap<String, usize>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    attempts: usize,
    max_attempts: usize,
    first_failure: Option<ResolveError>,
    best: Option<(usize, Vec<Option<Version>>, std::collections::BTreeMap<String, Version>)>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn release(&self, package: &str, version: Version) -> Option<&'a Release> {
        let i = *self.index.get(package)?;
        self.catalog[i]
            .releases
            .iter()
            .find(|r| r.version == version)
    }

    fn platform_active(&self, p: Platform) -> bool {
        match p {
            Platform::Any => true,
            Platform::Linux => self.target == Target::Linux,
            Platform::Mac => self.target == Target::Mac,
            Platform::Windows => self.target == Target::Windows,
        }
    }

    fn token_feature_list(features: &[String]) -> String {
        let mut fs = features.to_vec();
        fs.sort();
        fs.dedup();
        fs.join(",")
    }

    fn root_token(i: usize, r: &Requirement) -> String {
        format!(
            "root[{}]:{}@{}+{{{}}}",
            i,
            r.package,
            constraint_text(r.constraint),
            Self::token_feature_list(&r.features)
        )
    }

    fn dep_token(source: &str, version: Version, j: usize, d: &Dependency) -> String {
        format!(
            "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
            source,
            version_text(version),
            j,
            d.package,
            constraint_text(d.constraint),
            Self::token_feature_list(&d.features),
            d.gate.as_deref().unwrap_or("*"),
            match d.platform {
                Platform::Any => "any",
                Platform::Linux => "linux",
                Platform::Mac => "mac",
                Platform::Windows => "windows",
            }
        )
    }

    fn closure(&self, selected: &std::collections::BTreeMap<String, Version>) -> Closure {
        use std::collections::{BTreeMap, BTreeSet, VecDeque};

        let mut info: BTreeMap<String, PackageInfo> = BTreeMap::new();
        let mut activated: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
        let mut queue: VecDeque<(String, String, Vec<String>)> = VecDeque::new();

        for (i, r) in self.roots.iter().enumerate() {
            let path = vec![Self::root_token(i, r)];
            let item = info.entry(r.package.clone()).or_default();
            item.constraints.push(r.constraint);
            item.origins.push(Origin {
                constraint: r.constraint,
                path: path.clone(),
            });
            item.paths.push(path.clone());
            for f in &r.features {
                queue.push_back((r.package.clone(), f.clone(), path.clone()));
            }
        }

        let mut processed_deps: BTreeSet<(String, usize)> = BTreeSet::new();
        while let Some((package, feature, path)) = queue.pop_front() {
            let item = info.entry(package.clone()).or_default();
            item.paths.push(path);
            let newly_active = activated
                .entry(package.clone())
                .or_default()
                .insert(feature.clone());
            if !newly_active {
                continue;
            }

            let Some(version) = selected.get(&package).copied() else {
                continue;
            };
            let Some(release) = self.release(&package, version) else {
                continue;
            };

            if let Some(decl) = release.features.iter().find(|f| f.name == feature) {
                for enabled in &decl.enables {
                    queue.push_back((package.clone(), enabled.clone(), Vec::new()));
                }
            }

            let active_features = activated.get(&package).cloned().unwrap_or_default();
            for (j, dep) in release.deps.iter().enumerate() {
                if !self.platform_active(dep.platform)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !active_features.contains(g))
                    || !processed_deps.insert((package.clone(), j))
                {
                    continue;
                }

                let canonical_source_path = info
                    .get(&package)
                    .and_then(|i| {
                        i.paths
                            .iter()
                            .filter(|p| !p.is_empty())
                            .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)))
                            .cloned()
                    })
                    .unwrap_or_default();

                let token = Self::dep_token(&package, version, j, dep);
                let mut dest_path = canonical_source_path;
                dest_path.push(token);

                let dest = info.entry(dep.package.clone()).or_default();
                dest.constraints.push(dep.constraint);
                dest.origins.push(Origin {
                    constraint: dep.constraint,
                    path: dest_path.clone(),
                });
                dest.paths.push(dest_path.clone());
                for f in &dep.features {
                    queue.push_back((dep.package.clone(), f.clone(), dest_path.clone()));
                }
            }
        }

        // Enabled features receive the incoming package path as provenance.
        // Resolve empty activation paths from the canonical paths collected for that package.
        for (package, features) in &activated {
            if let Some(item) = info.get(package) {
                let best = item
                    .paths
                    .iter()
                    .filter(|p| !p.is_empty())
                    .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)))
                    .cloned()
                    .unwrap_or_default();
                for _ in features {
                    // The actual paths are only needed for package-level diagnostics.
                    let _ = &best;
                }
            }
        }

        for item in info.values_mut() {
            item.constraints = canonical_constraints(&item.constraints);
            item.paths.sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)));
            item.paths.dedup();
            item.origins.sort_by(|a, b| {
                constraint_key(a.constraint)
                    .cmp(&constraint_key(b.constraint))
                    .then_with(|| a.path.cmp(&b.path))
            });
        }

        Closure { info, activated }
    }

    fn node_error(
        &self,
        selected: &std::collections::BTreeMap<String, Version>,
        closure: &Closure,
    ) -> Option<ResolveError> {
        for name in closure.info.keys() {
            if !self.index.contains_key(name) {
                let evidence = closure.info[name]
                    .origins
                    .iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path.clone(),
                    })
                    .collect();
                return Some(ResolveError::Missing {
                    package: name.clone(),
                    evidence,
                });
            }
        }

        for (package, version) in selected {
            if let Some(info) = closure.info.get(package) {
                if !matches_all(*version, &info.constraints) {
                    return Some(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: info.constraints.clone(),
                        evidence: info
                            .origins
                            .iter()
                            .map(|o| Evidence {
                                constraint: o.constraint,
                                path: o.path.clone(),
                            })
                            .collect(),
                    });
                }
            }
        }

        for (package, info) in &closure.info {
            if selected.contains_key(package) {
                continue;
            }
            let i = self.index[package];
            if !self.catalog[i]
                .releases
                .iter()
                .any(|r| matches_all(r.version, &info.constraints))
            {
                return Some(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: info.constraints.clone(),
                    evidence: info
                        .origins
                        .iter()
                        .map(|o| Evidence {
                            constraint: o.constraint,
                            path: o.path.clone(),
                        })
                        .collect(),
                });
            }
        }

        for (package, version) in selected {
            if let Some(info) = closure.info.get(package) {
                if let Some(features) = closure.activated.get(package) {
                    for feature in features {
                        let release = self.release(package, *version).unwrap();
                        if !release.features.iter().any(|f| f.name == *feature) {
                            let path = info
                                .paths
                                .iter()
                                .find(|p| !p.is_empty())
                                .cloned()
                                .unwrap_or_default();
                            return Some(ResolveError::MissingFeature {
                                package: package.clone(),
                                selected: *version,
                                feature: feature.clone(),
                                path,
                            });
                        }
                    }
                }
            }
        }
        None
    }

    fn vector(
        &self,
        selected: &std::collections::BTreeMap<String, Version>,
    ) -> Vec<Option<Version>> {
        self.catalog_names
            .iter()
            .map(|n| selected.get(n).copied())
            .collect()
    }

    fn solution_cost(&self, selected: &std::collections::BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|l| selected.get(&l.package) != Some(&l.version))
            .count()
    }

    fn record_solution(&mut self, selected: &std::collections::BTreeMap<String, Version>) {
        let cost = self.solution_cost(selected);
        let vector = self.vector(selected);
        let replace = match &self.best {
            None => true,
            Some((best_cost, best_vector, _)) => {
                cost < *best_cost || (cost == *best_cost && vector > *best_vector)
            }
        };
        if replace {
            self.best = Some((cost, vector, selected.clone()));
        }
    }

    fn search(&mut self, selected: &mut std::collections::BTreeMap<String, Version>) -> bool {
        let closure = self.closure(selected);
        if let Some(error) = self.node_error(selected, &closure) {
            if self.first_failure.is_none() {
                self.first_failure = Some(error);
            }
            return false;
        }

        let next = closure
            .info
            .keys()
            .find(|n| !selected.contains_key(*n))
            .cloned();

        let Some(package) = next else {
            self.record_solution(selected);
            return false;
        };

        let info = &closure.info[&package];
        let i = self.index[&package];
        let mut candidates: Vec<Version> = self.catalog[i]
            .releases
            .iter()
            .filter(|r| matches_all(r.version, &info.constraints))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self
            .locks
            .iter()
            .find(|l| l.package == package)
            .map(|l| l.version)
        {
            if let Some(pos) = candidates.iter().position(|v| *v == lock) {
                candidates.remove(pos);
                candidates.insert(0, lock);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                self.first_failure = Some(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
                return true;
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            if self.search(selected) {
                return true;
            }
            selected.remove(&package);
        }
        false
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: site.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", site),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", site, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = std::collections::BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let index: std::collections::BTreeMap<String, usize> = catalog
        .iter()
        .enumerate()
        .map(|(i, p)| (p.name.clone(), i))
        .collect();
    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();

    let mut search = Search {
        catalog,
        index,
        roots,
        locks,
        target,
        attempts: 0,
        max_attempts,
        first_failure: None,
        best: None,
        catalog_names,
    };

    let mut selected = std::collections::BTreeMap::new();
    let limit_hit = search.search(&mut selected);
    if limit_hit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }

    let Some((_, _, chosen)) = search.best else {
        return Err(search.first_failure.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_closure = search.closure(&chosen);
    let selected_vec = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut names = std::collections::BTreeSet::new();
    names.extend(locks.iter().map(|l| l.package.clone()));
    names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();

    for package in names {
        let before = locks
            .iter()
            .find(|l| l.package == package)
            .map(|l| l.version);
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .index
                    .get(&package)
                    .is_some_and(|i| catalog[*i].releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_closure
                    .info
                    .get(&package)
                    .is_some_and(|i| !matches_all(old, &i.constraints))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
