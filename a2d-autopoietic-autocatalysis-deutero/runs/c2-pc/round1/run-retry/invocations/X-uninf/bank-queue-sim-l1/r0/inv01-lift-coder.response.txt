#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0usize;
    let mut current_time = 0u64;
    let mut finish_time: Option<u64> = None;
    let mut last_finish_time: Option<u64> = None;
    let mut first_dispatch = true;

    while next_arrival < arrivals.len() || finish_time.is_some() || !waiting.is_empty() {
        let next_arrival_time = arrivals.get(next_arrival).map(|c| c.arrival);
        let next_time = match (finish_time, next_arrival_time) {
            (Some(finish), Some(arrival)) => finish.min(arrival),
            (Some(finish), None) => finish,
            (None, Some(arrival)) => arrival,
            (None, None) => break,
        };
        current_time = next_time;

        if finish_time == Some(current_time) {
            let customer = match events.last() {
                Some(Event {
                    customer,
                    kind: Kind::Start { .. },
                    ..
                }) => *customer,
                _ => {
                    let completed = events
                        .iter()
                        .rev()
                        .find(|event| matches!(event.kind, Kind::Start { finish, .. } if finish == current_time))
                        .expect("scheduled finish must have a corresponding start");
                    completed.customer
                }
            };
            events.push(Event {
                time: current_time,
                customer,
                kind: Kind::Finish,
            });
            finish_time = None;
            last_finish_time = Some(current_time);
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == current_time {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: current_time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        if finish_time.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = first_dispatch
                    || last_finish_time.map_or(true, |time| current_time > time);
                let base_finish = current_time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: current_time,
                    },
                )?;
                let finish = base_finish.checked_add(if wake { wake_cost } else { 0 }).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: current_time,
                    },
                )?;

                events.push(Event {
                    time: current_time,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });
                first_dispatch = false;

                if finish == current_time {
                    events.push(Event {
                        time: current_time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    last_finish_time = Some(current_time);
                } else {
                    finish_time = Some(finish);
                }
            }
        }
    }

    Ok(events)
}
