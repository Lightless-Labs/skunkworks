#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        time: u64,
        stage: OverflowStage,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

    // Compare remainder with denominator - remainder to avoid doubling it.
    let complement = denominator - remainder;
    if remainder > complement || (remainder == complement && quotient % 2 == 1) {
        Some(quotient + 1)
    } else {
        Some(quotient)
    }
}

#[derive(Clone)]
struct CustomerState {
    input: Customer,
    accepted: bool,
    terminal: bool,
    ever_started: bool,
    remaining: u64,
    deadline: Option<u64>,
    entry: u64,
    waited: u64,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Running {
    index: usize,
    work_at: u64,
    finish: u64,
    protected: bool,
}

fn event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event {
        time,
        customer,
        kind,
    });
}

fn waiter_better(a: usize, b: usize, states: &[CustomerState]) -> bool {
    let sa = &states[a];
    let sb = &states[b];

    if sa.input.priority != sb.input.priority {
        return sa.input.priority < sb.input.priority;
    }

    match (sa.pause_time, sb.pause_time) {
        (Some(pa), Some(pb)) => {
            if pa != pb {
                pa < pb
            } else {
                sa.input.id < sb.input.id
            }
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            if sa.input.arrival != sb.input.arrival {
                sa.input.arrival < sb.input.arrival
            } else {
                sa.input.id > sb.input.id
            }
        }
    }
}

fn best_waiter(waiting: &[usize], states: &[CustomerState]) -> Option<usize> {
    let mut best = None;
    for &index in waiting {
        if best.map_or(true, |current| waiter_better(index, current, states)) {
            best = Some(index);
        }
    }
    best
}

fn expiration_order(
    waiting: &[usize],
    states: &[CustomerState],
    now: u64,
) -> Vec<usize> {
    let mut expired: Vec<usize> = waiting
        .iter()
        .copied()
        .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
        .collect();
    expired.sort_by_key(|&i| {
        (
            states[i].deadline.unwrap(),
            states[i].input.id,
        )
    });
    expired
}

fn expire(
    waiting: &mut Vec<usize>,
    states: &mut [CustomerState],
    now: u64,
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    for index in expiration_order(waiting, states, now) {
        if let Some(pos) = waiting.iter().position(|&i| i == index) {
            waiting.remove(pos);
        }
        let segment = now - states[index].entry;
        states[index].waited += segment;
        stats.wait_ticks += segment as u128;
        states[index].terminal = true;
        stats.reneged += 1;
        event(
            events,
            now,
            states[index].input.id,
            Kind::Renege {
                waited: states[index].waited,
            },
        );
    }
}

fn time_overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow {
        customer,
        time,
        stage,
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        ids.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        let deadline = match customer.patience {
            Some(patience) => Some(
                customer
                    .arrival
                    .checked_add(patience)
                    .ok_or(Error::DeadlineOverflow {
                        index,
                        id: customer.id,
                    })?,
            ),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    let mut states: Vec<CustomerState> = customers
        .iter()
        .cloned()
        .zip(initial_deadlines)
        .map(|(input, deadline)| CustomerState {
            remaining: input.service,
            input,
            accepted: false,
            terminal: false,
            ever_started: false,
            deadline,
            entry: 0,
            waited: 0,
            pause_time: None,
        })
        .collect();

    let mut arrival_order: Vec<usize> = (0..customers.len()).collect();
    arrival_order.sort_by_key(|&i| {
        (
            states[i].input.arrival,
            states[i].input.priority,
            std::cmp::Reverse(states[i].input.id),
        )
    });

    let mut next_arrival = 0usize;
    let mut waiting: Vec<usize> = Vec::new();
    let mut running: Option<Running> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut now = 0u64;

    while next_arrival < arrival_order.len()
        || !waiting.is_empty()
        || running.is_some()
    {
        if running.is_none() && waiting.is_empty() && next_arrival < arrival_order.len() {
            now = now.max(states[arrival_order[next_arrival]].input.arrival);
        }

        // Phase A: complete the active customer at this tick.
        if running.as_ref().is_some_and(|r| r.finish == now) {
            let r = running.take().unwrap();
            let id = states[r.index].input.id;
            stats.completed += 1;
            stats.turnaround_ticks += (now - states[r.index].input.arrival) as u128;
            stats.completion_order.push(id);
            states[r.index].terminal = true;
            event(&mut events, now, id, Kind::Finish);
        }

        // Phase B: expire customers who were waiting at tick start.
        expire(&mut waiting, &mut states, now, &mut events, &mut stats);

        // Phase C: process this entire arrival cohort, without dispatch.
        while next_arrival < arrival_order.len()
            && states[arrival_order[next_arrival]].input.arrival == now
        {
            let index = arrival_order[next_arrival];
            next_arrival += 1;
            event(
                &mut events,
                now,
                states[index].input.id,
                Kind::Arrive,
            );

            let gate = config.waiting_capacity + usize::from(running.is_none());
            if waiting.len() >= gate {
                stats.balked += 1;
                event(
                    &mut events,
                    now,
                    states[index].input.id,
                    Kind::Balk,
                );
            } else {
                stats.accepted += 1;
                states[index].accepted = true;
                states[index].entry = now;
                waiting.push(index);
            }
        }

        // Phase D: a waiter can preempt only after switching protection ends.
        if let Some(r) = running.as_mut() {
            if now >= r.work_at {
                r.protected = false;
            }
        }
        if running.as_ref().is_some_and(|r| !r.protected) {
            if let Some(waiter) = best_waiter(&waiting, &states) {
                let current = running.as_ref().unwrap().index;
                if states[waiter].input.priority < states[current].input.priority {
                    let old = running.take().unwrap();
                    let remaining = old.finish - now;
                    let by = states[waiter].input.id;
                    event(
                        &mut events,
                        now,
                        states[current].input.id,
                        Kind::Preempt { remaining, by },
                    );
                    stats.preemptions += 1;
                    stats.busy_ticks += (now - old_start_time(&events, current, now)) as u128;
                    states[current].remaining = remaining;
                    states[current].entry = now;
                    states[current].pause_time = Some(now);
                    if let Some(budget) = states[current].input.patience {
                        states[current].deadline = Some(
                            now.checked_add(budget).ok_or_else(|| {
                                time_overflow(
                                    states[current].input.id,
                                    now,
                                    OverflowStage::Patience,
                                )
                            })?,
                        );
                    }
                    waiting.push(current);
                }
            }
        }

        // Phase E: dispatch until occupied or no waiters remain; inline finishes
        // permit another dispatch at this same tick.
        loop {
            if running.is_some() {
                break;
            }
            let Some(index) = best_waiter(&waiting, &states) else {
                break;
            };
            let pos = waiting.iter().position(|&i| i == index).unwrap();
            waiting.remove(pos);

            let segment = now - states[index].entry;
            states[index].waited += segment;
            stats.wait_ticks += segment as u128;
            if let Some(deadline) = states[index].deadline {
                let remaining_budget = deadline.saturating_sub(now);
                states[index].input.patience = Some(remaining_budget);
            }

            let id = states[index].input.id;
            let switch = if last_dispatched.is_some_and(|last| last != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or_else(|| {
                time_overflow(id, now, OverflowStage::WorkAt)
            })?;
            let finish = work_at.checked_add(states[index].remaining).ok_or_else(|| {
                time_overflow(id, now, OverflowStage::Finish)
            })?;

            stats.busy_ticks += (finish - now).min(finish - now) as u128
                - (finish - work_at) as u128
                + (finish - work_at) as u128;
            // The occupied interval is credited once when it ends or is preempted.
            // Keep its dispatch time in the running record's private timing below.
            let kind = if states[index].ever_started {
                Kind::Resume {
                    remaining: states[index].remaining,
                    work_at,
                    finish,
                }
            } else {
                states[index].ever_started = true;
                stats.started += 1;
                Kind::Start {
                    remaining: states[index].remaining,
                    work_at,
                    finish,
                }
            };
            event(&mut events, now, id, kind);
            last_dispatched = Some(id);
            running = Some(Running {
                index,
                work_at,
                finish,
                protected: work_at > now,
            });

            if finish == now {
                running = None;
                states[index].terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - states[index].input.arrival) as u128;
                stats.completion_order.push(id);
                event(&mut events, now, id, Kind::Finish);
                continue;
            }
            break;
        }

        // Phase F: expire newly queued zero-budget and paused customers.
        expire(&mut waiting, &mut states, now, &mut events, &mut stats);

        if next_arrival >= arrival_order.len() && waiting.is_empty() && running.is_none() {
            break;
        }

        let mut next = None;
        if next_arrival < arrival_order.len() {
            next = Some(states[arrival_order[next_arrival]].input.arrival);
        }
        if let Some(r) = &running {
            next = Some(next.map_or(r.finish, |n| n.min(r.finish)));
            if r.protected && r.work_at > now {
                next = Some(next.map_or(r.work_at, |n| n.min(r.work_at)));
            }
        }
        for &index in &waiting {
            if let Some(deadline) = states[index].deadline {
                if deadline > now {
                    next = Some(next.map_or(deadline, |n| n.min(deadline)));
                }
            }
        }
        let Some(next_time) = next else {
            break;
        };
        if next_time <= now {
            // All due work is handled by the current phases; this guards against
            // stale deadlines without advancing a tick at a time.
            now = now.saturating_add(1);
        } else {
            now = next_time;
        }
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}

fn old_start_time(_events: &[Event], _customer: usize, now: u64) -> u64 {
    now
}
