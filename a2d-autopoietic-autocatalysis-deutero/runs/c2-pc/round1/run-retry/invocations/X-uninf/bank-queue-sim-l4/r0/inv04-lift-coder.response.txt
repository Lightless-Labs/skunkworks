#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };
    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct Job {
    id: u32,
    arrival: u64,
    service: u64,
    priority: u8,
    remaining: u64,
    deadline: Option<u64>,
    wait_entry: u64,
    waited: u64,
    pause_time: Option<u64>,
    started: bool,
}

#[derive(Clone, Copy)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    interval_start: u64,
}

fn push_event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiter_better(a: &Job, b: &Job) -> bool {
    if a.priority != b.priority {
        return a.priority < b.priority;
    }
    match (a.pause_time, b.pause_time) {
        (Some(ta), Some(tb)) => ta < tb || (ta == tb && a.id < b.id),
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => a.arrival < b.arrival || (a.arrival == b.arrival && a.id > b.id),
    }
}

fn best_waiter(waiting: &[usize], jobs: &[Job]) -> Option<usize> {
    waiting.iter().copied().min_by(|&a, &b| {
        if waiter_better(&jobs[a], &jobs[b]) {
            std::cmp::Ordering::Less
        } else if waiter_better(&jobs[b], &jobs[a]) {
            std::cmp::Ordering::Greater
        } else {
            std::cmp::Ordering::Equal
        }
    })
}

fn checked_time(customer: u32, time: u64, amount: u64, stage: OverflowStage) -> Result<u64, Error> {
    time.checked_add(amount).ok_or(Error::TimeOverflow { customer, time, stage })
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut seen = Vec::<u32>::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&customer.id) {
            return Err(Error::DuplicateId { index, id: customer.id });
        }
        seen.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority { index, value: customer.priority });
        }
        let deadline = match customer.patience {
            Some(patience) => Some(customer.arrival.checked_add(patience).ok_or(
                Error::DeadlineOverflow { index, id: customer.id },
            )?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut jobs: Vec<Job> = customers.iter().zip(initial_deadlines).map(|(c, deadline)| Job {
        id: c.id,
        arrival: c.arrival,
        service: c.service,
        priority: c.priority,
        remaining: c.service,
        deadline,
        wait_entry: c.arrival,
        waited: 0,
        pause_time: None,
        started: false,
    }).collect();

    let mut arrival_order: Vec<usize> = (0..jobs.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        jobs[a].arrival.cmp(&jobs[b].arrival)
            .then_with(|| jobs[a].priority.cmp(&jobs[b].priority))
            .then_with(|| jobs[b].id.cmp(&jobs[a].id))
    });

    let mut events = Vec::new();
    let mut waiting = Vec::<usize>::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let (mut accepted, mut balked, mut reneged, mut completed, mut started, mut preemptions) =
        (0usize, 0usize, 0usize, 0usize, 0usize, 0usize);
    let (mut busy_ticks, mut wait_ticks, mut turnaround_ticks) = (0u128, 0u128, 0u128);
    let mut completion_order = Vec::new();

    if jobs.is_empty() {
        return Ok(Report {
            events,
            stats: Stats {
                accepted, balked, reneged, completed, started, preemptions,
                elapsed: 0, busy_ticks, wait_ticks, turnaround_ticks,
                mean_wait_milli: 0, mean_turnaround_milli: 0, utilization_bps: 0,
                completion_order,
            },
        });
    }

    let mut now = jobs[arrival_order[0]].arrival;
    loop {
        // Phase A
        if let Some(running) = active {
            if running.finish == now {
                let job = &jobs[running.index];
                push_event(&mut events, now, job.id, Kind::Finish);
                busy_ticks += (now - running.interval_start) as u128;
                turnaround_ticks += (now - job.arrival) as u128;
                completed += 1;
                completion_order.push(job.id);
                active = None;
            }
        }

        // Phase B
        let mut expired: Vec<usize> = waiting.iter().copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| jobs[a].deadline.cmp(&jobs[b].deadline)
            .then_with(|| jobs[a].id.cmp(&jobs[b].id)));
        for index in expired {
            if let Some(pos) = waiting.iter().position(|&i| i == index) {
                waiting.remove(pos);
                let segment = now - jobs[index].wait_entry;
                jobs[index].waited += segment;
                wait_ticks += segment as u128;
                push_event(&mut events, now, jobs[index].id, Kind::Renege { waited: jobs[index].waited });
                reneged += 1;
            }
        }

        // Phase C
        while next_arrival < arrival_order.len() && jobs[arrival_order[next_arrival]].arrival == now {
            let index = arrival_order[next_arrival];
            next_arrival += 1;
            push_event(&mut events, now, jobs[index].id, Kind::Arrive);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() < gate {
                accepted += 1;
                jobs[index].wait_entry = now;
                waiting.push(index);
            } else {
                balked += 1;
                push_event(&mut events, now, jobs[index].id, Kind::Balk);
            }
        }

        // Phase D
        if let Some(running) = active {
            if now >= running.work_at {
                if let Some(candidate) = best_waiter(&waiting, &jobs) {
                    if jobs[candidate].priority < jobs[running.index].priority {
                        let remaining = running.finish - now;
                        let paused_id = jobs[running.index].id;
                        let by = jobs[candidate].id;
                        push_event(&mut events, now, paused_id, Kind::Preempt { remaining, by });
                        busy_ticks += (now - running.interval_start) as u128;
                        active = None;

                        jobs[running.index].remaining = remaining;
                        jobs[running.index].wait_entry = now;
                        jobs[running.index].pause_time = Some(now);
                        jobs[running.index].deadline = match jobs[running.index].deadline {
                            Some(deadline) => {
                                let budget = deadline - running.interval_start;
                                Some(checked_time(paused_id, now, budget, OverflowStage::Patience)?)
                            }
                            None => None,
                        };
                        waiting.push(running.index);
                        preemptions += 1;
                    }
                }
            }
        }

        // Phase E
        loop {
            if active.is_some() {
                break;
            }
            let Some(index) = best_waiter(&waiting, &jobs) else { break };
            let pos = waiting.iter().position(|&i| i == index).unwrap();
            waiting.remove(pos);

            let segment = now - jobs[index].wait_entry;
            jobs[index].waited += segment;
            wait_ticks += segment as u128;

            let id = jobs[index].id;
            let dispatch_time = now;
            let switch = if last_dispatched.is_some() && last_dispatched != Some(id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = checked_time(id, dispatch_time, switch, OverflowStage::WorkAt)?;
            let finish = work_at.checked_add(jobs[index].remaining).ok_or(Error::TimeOverflow {
                customer: id,
                time: dispatch_time,
                stage: OverflowStage::Finish,
            })?;

            let kind = if jobs[index].started {
                Kind::Resume { remaining: jobs[index].remaining, work_at, finish }
            } else {
                jobs[index].started = true;
                started += 1;
                Kind::Start { remaining: jobs[index].remaining, work_at, finish }
            };
            push_event(&mut events, now, id, kind);
            last_dispatched = Some(id);
            active = Some(Active { index, work_at, finish, interval_start: now });

            if finish == now {
                push_event(&mut events, now, id, Kind::Finish);
                turnaround_ticks += (now - jobs[index].arrival) as u128;
                completed += 1;
                completion_order.push(id);
                active = None;
                continue;
            }
            break;
        }

        // Phase F
        let mut expired: Vec<usize> = waiting.iter().copied()
            .filter(|&i| jobs[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| jobs[a].deadline.cmp(&jobs[b].deadline)
            .then_with(|| jobs[a].id.cmp(&jobs[b].id)));
        for index in expired {
            if let Some(pos) = waiting.iter().position(|&i| i == index) {
                waiting.remove(pos);
                let segment = now - jobs[index].wait_entry;
                jobs[index].waited += segment;
                wait_ticks += segment as u128;
                push_event(&mut events, now, jobs[index].id, Kind::Renege { waited: jobs[index].waited });
                reneged += 1;
            }
        }

        if next_arrival == arrival_order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |current| current.min(time)));
            }
        };
        if next_arrival < arrival_order.len() {
            consider(jobs[arrival_order[next_arrival]].arrival);
        }
        if let Some(running) = active {
            consider(running.finish);
            if running.work_at > now {
                consider(running.work_at);
            }
        }
        for &index in &waiting {
            if let Some(deadline) = jobs[index].deadline {
                consider(deadline);
            }
        }
        let Some(time) = next_time else { break };
        now = time;
    }

    let elapsed = events.last().map_or(0, |event| event.time);
    let mean_wait_milli = if accepted == 0 { 0 } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 { 0 } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 { 0 } else {
        rounded_ratio(busy_ticks * 10_000, elapsed as u128).unwrap().min(u32::MAX as u128) as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted, balked, reneged, completed, started, preemptions, elapsed,
            busy_ticks, wait_ticks, turnaround_ticks, mean_wait_milli,
            mean_turnaround_milli, utilization_bps, completion_order,
        },
    })
}
