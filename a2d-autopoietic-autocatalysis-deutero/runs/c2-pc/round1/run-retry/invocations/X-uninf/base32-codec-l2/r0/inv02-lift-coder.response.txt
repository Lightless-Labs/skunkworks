#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

pub fn encode(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let symbols = alphabet_bytes(alphabet);
    let symbol_count = (input.len() * 8 + 4) / 5;
    let output_len = if padding == Padding::Required && symbol_count != 0 {
        (symbol_count + 7) / 8 * 8
    } else {
        symbol_count
    };

    let mut output = String::with_capacity(output_len);
    for symbol_index in 0..symbol_count {
        let mut value = 0u8;
        for bit_in_symbol in 0..5 {
            let bit_index = symbol_index * 5 + bit_in_symbol;
            value <<= 1;
            if bit_index < input.len() * 8 {
                let byte = input[bit_index / 8];
                value |= (byte >> (7 - bit_index % 8)) & 1;
            }
        }
        output.push(symbols[value as usize] as char);
    }

    if padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }

    output
}

pub fn decode(
    input: &str,
    alphabet: Alphabet,
    padding: Padding,
) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let symbols = alphabet_bytes(alphabet);

    // Lexical validation scans the complete input before other checks.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && !symbols.contains(&byte) {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let n = bytes.len();
    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);

    if padding == Padding::Omitted && p < n {
        return Err(DecodeError::Padding { offset: p });
    }

    if padding == Padding::Required && p < n {
        if let Some(offset) = bytes[p..].iter().position(|&byte| byte != b'=') {
            return Err(DecodeError::Padding { offset: p + offset });
        }
    }

    let d = p;
    let residue = d % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    if padding == Padding::Required {
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => unreachable!(),
        };
        if n % 8 != 0 || n - p != expected_padding {
            return Err(DecodeError::Padding { offset: p });
        }
    }

    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };
    if unused_bits != 0 {
        let last_value = symbols
            .iter()
            .position(|&symbol| symbol == bytes[d - 1])
            .expect("lexically validated symbol");
        if (last_value as u8) & ((1u8 << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits { offset: d - 1 });
        }
    }

    let mut output = Vec::with_capacity(d * 5 / 8);
    let mut accumulator = 0u32;
    let mut bit_count = 0usize;

    for &byte in &bytes[..d] {
        let value = symbols
            .iter()
            .position(|&symbol| symbol == byte)
            .expect("lexically validated symbol") as u32;
        accumulator = (accumulator << 5) | value;
        bit_count += 5;

        while bit_count >= 8 {
            bit_count -= 8;
            output.push((accumulator >> bit_count) as u8);
            accumulator &= (1u32 << bit_count).wrapping_sub(1);
        }
    }

    Ok(output)
}
