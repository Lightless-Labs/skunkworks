struct Version {
    revision: u64,
    value: Option<String>,
    deadline: Option<u64>,
}

impl Version {
    fn live(&self, now: u64) -> Option<&str> {
        match (&self.value, self.deadline) {
            (Some(value), Some(deadline)) if deadline <= now => None,
            (Some(value), _) => Some(value),
            (None, _) => None,
        }
    }
}

#[derive(Clone)]
struct Write {
    value: Option<String>,
    deadline: Option<u64>,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Write>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Write>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn transaction(&self, id: u64) -> Result<&Transaction, Error> {
        self.transactions.get(&id).ok_or(Error::UnknownTransaction(id))
    }

    fn transaction_mut(&mut self, id: u64) -> Result<&mut Transaction, Error> {
        self.transactions
            .get_mut(&id)
            .ok_or(Error::UnknownTransaction(id))
    }

    fn visible_write(&self, key: &str, tx: Option<u64>) -> Option<Write> {
        if let Some(id) = tx {
            if let Some(write) = self.transactions.get(&id)?.writes.get(key) {
                return Some(write.clone());
            }
        }
        let versions = self.data.get(key)?;
        let version = if let Some(id) = tx {
            let revision = self.transactions.get(&id)?.begin_revision;
            versions.iter().rev().find(|v| v.revision <= revision)?
        } else {
            versions.last()?
        };
        Some(Write {
            value: version.value.clone(),
            deadline: version.deadline,
        })
    }

    fn visible_value(&self, key: &str, tx: Option<u64>, now: u64) -> Option<String> {
        let write = self.visible_write(key, tx)?;
        match (write.value, write.deadline) {
            (Some(value), Some(deadline)) if deadline <= now => None,
            (Some(value), _) => Some(value),
            (None, _) => None,
        }
    }

    fn append(&mut self, key: String, write: Write, revision: u64) {
        self.data.entry(key).or_default().push(Version {
            revision,
            value: write.value,
            deadline: write.deadline,
        });
    }

    fn validate_key(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let bytes = name.as_bytes();
        for (i, byte) in bytes.iter().copied().enumerate() {
            let valid = if i == 0 {

                byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'_'
            };
            if !valid {
                return Err(Error::NameInvalid { at: i });
            }
        }
        Ok(())

        if let Some(end) = end {

                return Err(Error::InvalidRange);
            }
        }
        Ok(())

        key >= start

            && prefix.map_or(true, |prefix| key.starts_with(prefix))

                keys.extend(transaction.writes.keys().cloned());

        self.admit(now)?;
        if self.transactions.contains_key(&id) {

                writes: BTreeMap::new(),

        if !transaction.writes.is_empty() {

                self.data.get(key).and_then(|versions| {
                    versions
                        .last()
                        .filter(|v| v.revision > begin_revision)
                        .map(|_| key.clone())
                })
            });
            if let Some(key) = conflict {
                return Err(Error::Conflict { tx: id, key });
            }

            let writes = transaction.writes.clone();
            let revision = self.revision + 1;
            for (key, write) in writes {
                self.append(key, write, revision);
            }
            self.revision = revision;
        }

        self.transactions.remove(&id);
        Ok(self.revision)

            value: None,

        } else {
            self.revision += 1;
            self.append(key.to_string(), tombstone, self.revision);
        }
        Ok(existed)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        if let Some(id) = tx {
            self.transaction(id)?;
        }
        Self::validate_key(key)?;
        Ok(self.visible_value(key, tx, now))
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        if let Some(id) = tx {
            self.transaction(id)?;
        }
        Ok(self
            .selected_keys(tx)
            .iter()
            .filter(|key| self.visible_value(key, tx, now).as_deref() == Some(value))
            .count())
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        if let Some(id) = tx {
            self.transaction(id)?;
        }
        Self::validate_range(start, end)?;

        if let Some(cursor) = cursor {
            if cursor.start != start
                || cursor.end.as_deref() != end
                || cursor.prefix.as_deref() != prefix
                || cursor.offset != offset
                || cursor.last.is_empty()
                || !Self::matches(&cursor.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let keys: Vec<String> = self
            .selected_keys(tx)
            .into_iter()
            .filter(|key| Self::matches(key, start, end, prefix))
            .filter(|key| {
                cursor.map_or(true, |cursor| key.as_str() > cursor.last.as_str())
            })
            .filter(|key| self.visible_value(key, tx, now).is_some())
            .collect();

        let begin = if cursor.is_some() {
            0
        } else {
            offset.min(keys.len())
        };
        let finish = begin.saturating_add(limit).min(keys.len());
        let items: Vec<(String, String)> = keys[begin..finish]
            .iter()
            .filter_map(|key| {
                self.visible_value(key, tx, now)
                    .map(|value| (key.clone(), value))
            })
            .collect();

        let next = if finish < keys.len() && !items.is_empty() {
            Some(Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: items.last().unwrap().0.clone(),
            })
        } else {
            None
        };

        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        if let Some(id) = tx {
            self.transaction(id)?;
        }
        Self::validate_range(start, end)?;

        let mut result = Aggregate { count: 0, sum: 0 };
        for key in self.selected_keys(tx) {
            if !Self::matches(&key, start, end, prefix) {
                continue;
            }
            let Some(value) = self.visible_value(&key, tx, now) else {
                continue;
            };
            let parsed = parse_canonical_i64(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            result.sum = result
                .sum
                .checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            result.count += 1;
        }
        Ok(result)
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }

        let mut saved = BTreeMap::new();
        for key in self.data.keys() {
            let Some(version) = self.data.get(key).and_then(|v| v.last()) else {
                continue;
            };
            if version.live(now).is_some() {
                saved.insert(
                    key.clone(),
                    Write {
                        value: version.value.clone(),
                        deadline: version.deadline,
                    },
                );
            }
        }
        self.snapshots.insert(name.to_string(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let copied: BTreeMap<String, Write> = saved
            .into_iter()
            .filter(|(_, write)| {
                write.value.is_some()
                    && write.deadline.map_or(true, |deadline| deadline > now)
            })
            .collect();

        let mut current_live = BTreeSet::new();
        for key in self.data.keys() {
            if self
                .data
                .get(key)
                .and_then(|versions| versions.last())
                .and_then(|version| version.live(now))
                .is_some()
            {
                current_live.insert(key.clone());
            }
        }

        let union: BTreeSet<String> = current_live
            .iter()
            .cloned()
            .chain(copied.keys().cloned())
            .collect();

        if union.is_empty() {
            return Ok(self.revision);
        }

        let revision = self.revision + 1;
        for key in union {
            let write = copied.get(&key).cloned().unwrap_or(Write {
                value: None,
                deadline: None,
            });
            self.append(key, write, revision);
        }
        self.revision = revision;
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if !self.snapshots.contains_key(name) {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        self.snapshots.remove(name);
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;

        let mut retained: BTreeMap<String, BTreeSet<usize>> = BTreeMap::new();
        for (key, versions) in &self.data {
            if !versions.is_empty() {
                retained.entry(key.clone()).or_default().insert(versions.len() - 1);
            }
        }

        for transaction in self.transactions.values() {
            for (key, versions) in &self.data {
                if let Some(index) = versions
                    .iter()
                    .rposition(|version| version.revision <= transaction.begin_revision)
                {
                    retained.entry(key.clone()).or_default().insert(index);
                }
            }
        }

        let mut removed = 0;
        for (key, versions) in &mut self.data {
            let keep = retained.get(key);
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.is_some_and(|indices| indices.contains(&index));
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;

        let keys = self.data.len();
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self
            .data
            .values()
            .flat_map(|versions| versions.iter())
            .filter(|version| version.value.is_none())
            .count();
        let live_keys = self
            .data
            .values()
            .filter(|versions| versions.last().and_then(|v| v.live(now)).is_some())
            .count();
        let read_only_transactions = self
            .transactions
            .values()
            .filter(|transaction| transaction.read_only)
            .count();
        let staged_writes = self
            .transactions
            .values()
            .map(|transaction| transaction.writes.len())
            .sum();

        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys,
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions,
            staged_writes,
            snapshots: self.snapshots.len(),
        })
    }
}

fn parse_canonical_i64(value: &str) -> Option<i64> {
    if value == "0" {
        return Some(0);
    }

    let bytes = value.as_bytes();
    let digits = if bytes.first() == Some(&b'-') {
        &bytes[1..]
    } else {
        bytes
    };

    if digits.is_empty()
        || digits[0] == b'0'
        || !digits.iter().all(u8::is_ascii_digit)
    {
        return None;
    }
    if bytes.first() == Some(&b'-') && digits == b"0" {
        return None;
    }

    value.parse::<i64>().ok()
}

struct Entry {

    fn live(&self, now: u64) -> Option<&String> {
        match (&self.value, self.deadline) {

            (Some(value), _) => Some(value),
            (None, _) => None,
        }
    }

struct Transaction {

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: HashMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: HashMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn check_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn visible_entry(&self, tx: Option<u64>, key: &str) -> Option<&Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(entry) = transaction.writes.get(key) {
                return Some(entry);
            }
            self.data.get(key)?.iter()
                .rev()
                .find(|version| version.revision <= transaction.begin_revision)
                .map(|version| &version.entry)
        } else {
            self.data.get(key)?.last().map(|version| &version.entry)
        }
    }

    fn visible_map(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, String> {
        let mut result = BTreeMap::new();
        for key in self.data.keys() {
            if let Some(value) = self.visible_entry(tx, key).and_then(|e| e.live(now)) {
                result.insert(key.clone(), value.clone());
            }
        }
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                for (key, entry) in &transaction.writes {
                    if let Some(value) = entry.live(now) {
                        result.insert(key.clone(), value.clone());
                    } else {
                        result.remove(key);
                    }
                }
            }
        }
        result
    }

    fn append(&mut self, key: String, entry: Entry) -> u64 {
        self.revision += 1;
        self.data.entry(key).or_default().push(Version {
            revision: self.revision,
            entry,
        });
        self.revision
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    fn valid_key(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn selected_keys(
        &self,
        tx: Option<u64>,
        now: u64,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Vec<(String, String)> {
        self.visible_map(tx, now)
            .into_iter()
            .filter(|(key, _)| {
                key.as_str() >= start
                    && end.map_or(true, |bound| key.as_str() < bound)
                    && prefix.map_or(true, |p| key.starts_with(p))
            })
            .collect()
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let transaction = self
            .transactions
            .get(&id)
            .ok_or(Error::UnknownTransaction(id))?;

        if !transaction.writes.is_empty() {
            for key in transaction.writes.keys() {
                let latest = self.data.get(key).and_then(|versions| versions.last());
                if latest.map_or(false, |version| version.revision > transaction.begin_revision) {
                    return Err(Error::Conflict {
                        tx: id,
                        key: key.clone(),
                    });
                }
            }
            let writes = self.transactions.remove(&id).unwrap().writes;
            self.revision += 1;
            let revision = self.revision;
            for (key, entry) in writes {
                self.data.entry(key).or_default().push(Version {
                    revision,
                    entry,
                });
            }
            Ok(revision)
        } else {
            self.transactions.remove(&id);
            Ok(self.revision)
        }
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::valid_key(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let entry = match ttl {
            Some(0) => Entry::tombstone(),
            Some(duration) => Entry {
                value: Some(value.to_string()),
                deadline: Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            },
            None => Entry {
                value: Some(value.to_string()),
                deadline: None,
            },
        };

        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_string(), entry);
        } else {
            self.append(key.to_string(), entry);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::valid_key(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }

        let existed = self
            .visible_entry(tx, key)
            .and_then(|entry| entry.live(now))
            .is_some();

        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_string(), Entry::tombstone());
        } else {
            self.append(key.to_string(), Entry::tombstone());
        }
        Ok(existed)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::valid_key(key)?;
        Ok(self
            .visible_entry(tx, key)
            .and_then(|entry| entry.live(now))
            .cloned())
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Ok(self
            .visible_map(tx, now)
            .values()
            .filter(|found| found.as_str() == value)
            .count())
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if end.map_or(false, |bound| bound <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(cursor) = cursor {
            if cursor.start != start
                || cursor.end.as_deref() != end
                || cursor.prefix.as_deref() != prefix
                || cursor.offset != offset
                || cursor.last.is_empty()
                || cursor.last.as_str() < start
                || end.map_or(false, |bound| cursor.last.as_str() >= bound)
                || prefix.map_or(false, |p| !cursor.last.starts_with(p))
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let mut matching = self.selected_keys(tx, now, start, end, prefix);
        if let Some(cursor) = cursor {
            matching.retain(|(key, _)| key.as_str() > cursor.last.as_str());
        } else {
            matching = matching.into_iter().skip(offset).collect();
        }

        let take = matching.len().min(limit);
        let has_more = matching.len() > take;
        let items: Vec<_> = matching.into_iter().take(take).collect();
        let next = if has_more {
            items.last().map(|(last, _)| Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if end.map_or(false, |bound| bound <= start) {
            return Err(Error::InvalidRange);
        }

        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in self.selected_keys(tx, now, start, end, prefix) {
            let parsed = Self::parse_number(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum
                .checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    fn parse_number(value: &str) -> Option<i64> {
        let bytes = value.as_bytes();
        if bytes.is_empty() {
            return None;
        }
        let digits = if bytes[0] == b'-' {
            if bytes.len() < 2 {
                return None;
            }
            &bytes[1..]
        } else {
            bytes
        };
        if digits.is_empty() || !digits.iter().all(u8::is_ascii_digit) {
            return None;
        }
        if digits.len() > 1 && digits[0] == b'0' {
            return None;
        }
        if bytes[0] == b'-' && digits == b"0" {
            return None;
        }
        value.parse::<i64>().ok()
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let live = self
            .visible_map(None, now)
            .into_iter()
            .map(|(key, value)| {
                let entry = self.data.get(&key).unwrap().last().unwrap().entry.clone();
                let _ = value;
                (key, entry)
            })
            .collect();
        self.snapshots.insert(name.to_string(), live);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let copied: BTreeMap<String, Entry> = saved
            .into_iter()
            .filter(|(_, entry)| entry.live(now).is_some())
            .collect();
        let current = self.visible_map(None, now);
        let union: BTreeSet<String> = current.keys().chain(copied.keys()).cloned().collect();
        if union.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in union {
            let entry = copied
                .get(&key)
                .cloned()
                .unwrap_or_else(Entry::tombstone);
            self.data.entry(key).or_default().push(Version {
                revision,
                entry,
            });
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;
        for versions in self.data.values_mut() {
            let mut keep = BTreeSet::new();
            if let Some(last) = versions.len().checked_sub(1) {
                keep.insert(last);
            }
            for transaction in self.transactions.values() {
                if let Some(index) = versions
                    .iter()
                    .rposition(|version| version.revision <= transaction.begin_revision)
                {
                    keep.insert(index);
                }
            }
            let mut index = 0;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let keys = self.data.len();
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self
            .data
            .values()
            .flatten()
            .filter(|version| version.entry.value.is_none())
            .count();
        let live_keys = self.visible_map(None, now).len();
        let read_only_transactions = self
            .transactions
            .values()
            .filter(|transaction| transaction.read_only)
            .count();
        let staged_writes = self
            .transactions
            .values()
            .map(|transaction| transaction.writes.len())
            .sum();

        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys,
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions,
            staged_writes,
            snapshots: self.snapshots.len(),
        })
    }
}
