use std::collections::{BTreeMap, BTreeSet, HashMap};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Value {
    value: String,
    deadline: Option<u64>,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    value: Option<Value>,
}

#[derive(Clone)]
struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Option<Value>>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: HashMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Value>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: HashMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn ensure_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn check_key(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn newest_at(&self, key: &str, revision: Option<u64>) -> Option<&Version> {
        self.data.get(key)?.iter().rev().find(|v| {
            revision.map_or(true, |r| v.revision <= r)
        })
    }

    fn visible_value(&self, key: &str, tx: Option<u64>, now: u64) -> Option<String> {
        let version = match tx {
            Some(id) => self.newest_at(key, Some(self.transactions[&id].begin_revision)),
            None => self.newest_at(key, None),
        };
        let mut entry = version.and_then(|v| v.value.clone());
        if let Some(id) = tx {
            if let Some(w) = self.transactions[&id].writes.get(key) {
                entry = w.clone();
            }
        }
        entry.and_then(|v| {
            if v.deadline.map_or(true, |d| d > now) {
                Some(v.value)
            } else {
                None
            }
        })
    }

    fn visible_keys(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, String> {
        let mut keys: BTreeSet<String> = self.data.keys().cloned().collect();
        if let Some(id) = tx {
            keys.extend(self.transactions[&id].writes.keys().cloned());
        }
        keys.into_iter()
            .filter_map(|k| self.visible_value(&k, tx, now).map(|v| (k, v)))
            .collect()
    }

    fn append(&mut self, key: String, value: Option<Value>, revision: u64) {
        self.data
            .entry(key)
            .or_default()
            .push(Version { revision, value });
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let tx = self
            .transactions
            .get(&id)
            .ok_or(Error::UnknownTransaction(id))?;
        if tx.writes.is_empty() {
            self.transactions.remove(&id);
            return Ok(self.revision);
        }

        for key in tx.writes.keys() {
            if self
                .newest_at(key, None)
                .is_some_and(|v| v.revision > tx.begin_revision)
            {
                return Err(Error::Conflict {
                    tx: id,
                    key: key.clone(),
                });
            }
        }

        let writes = tx.writes.clone();
        self.revision += 1;
        let revision = self.revision;
        for (key, value) in writes {
            self.append(key, value, revision);
        }
        self.transactions.remove(&id);
        Ok(revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.ensure_tx(tx)?;
        Self::check_key(key)?;
        if let Some(id) = tx {
            if self.transactions[&id].read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let entry = match ttl {
            Some(0) => None,
            Some(d) => Some(Value {
                value: value.to_string(),
                deadline: Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            }),
            None => Some(Value {
                value: value.to_string(),
                deadline: None,
            }),
        };
        self.write(tx, key.to_string(), entry);
        Ok(())
    }

    fn write(&mut self, tx: Option<u64>, key: String, value: Option<Value>) {
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key, value);
        } else {
            self.revision += 1;
            let revision = self.revision;
            self.append(key, value, revision);
        }
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.ensure_tx(tx)?;
        Self::check_key(key)?;
        if let Some(id) = tx {
            if self.transactions[&id].read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let was_live = self.visible_value(key, tx, now).is_some();
        self.write(tx, key.to_string(), None);
        Ok(was_live)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.ensure_tx(tx)?;
        Self::check_key(key)?;
        Ok(self.visible_value(key, tx, now))
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        self.ensure_tx(tx)?;
        Ok(self
            .visible_keys(tx, now)
            .values()
            .filter(|v| v.as_str() == value)
            .count())
    }

    fn validate_range(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|e| e <= start) {
            Err(Error::InvalidRange)
        } else {
            Ok(())
        }
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.ensure_tx(tx)?;
        Self::validate_range(start, end)?;

        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let view = self.visible_keys(tx, now);
        let mut matching = view
            .into_iter()
            .filter(|(k, _)| {
                Self::matches(k, start, end, prefix)
                    && cursor.map_or(true, |c| k.as_str() > c.last.as_str())
            });

        if cursor.is_none() {
            for _ in 0..offset {
                if matching.next().is_none() {
                    return Ok(Page {
                        items: Vec::new(),
                        next: None,
                    });
                }
            }
        }

        let mut items = Vec::new();
        for _ in 0..limit {
            if let Some(item) = matching.next() {
                items.push(item);
            } else {
                break;
            }
        }
        let next = if !items.is_empty() && matching.next().is_some() {
            Some(Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: items.last().unwrap().0.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.ensure_tx(tx)?;
        Self::validate_range(start, end)?;
        let view = self.visible_keys(tx, now);
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in view {
            if !Self::matches(&key, start, end, prefix) {
                continue;
            }
            let parsed = parse_number(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum
                .checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            if !(ch.is_ascii_lowercase()
                || (at > 0 && (ch.is_ascii_digit() || ch == '_')))
            {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        Self::validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let saved = self.visible_keys(None, now).into_iter().map(|(k, v)| {
            let deadline = self
                .newest_at(&k, None)
                .and_then(|ver| ver.value.as_ref())
                .and_then(|v| v.deadline);
            (k, Value { value: v, deadline })
        }).collect();
        self.snapshots.insert(name.to_string(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        Self::validate_name(name)?;
        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let current = self.visible_keys(None, now);
        let saved_live: BTreeMap<String, Value> = saved
            .into_iter()
            .filter(|(_, v)| v.deadline.map_or(true, |d| d > now))
            .collect();
        let mut union: BTreeSet<String> = current.keys().cloned().collect();
        union.extend(saved_live.keys().cloned());
        if union.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in union {
            self.append(key.clone(), saved_live.get(&key).cloned(), revision);
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        Self::validate_name(name)?;
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let begin_revisions: Vec<u64> = self
            .transactions
            .values()
            .map(|tx| tx.begin_revision)
            .collect();
        let mut removed = 0;
        for versions in self.data.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() {
                keep.insert(versions.len() - 1);
            }
            for begin in &begin_revisions {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= *begin) {
                    keep.insert(index);
                }
            }
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let live_keys = self.visible_keys(None, now).len();
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self
            .data
            .values()
            .flat_map(|v| v.iter())
            .filter(|v| v.value.is_none())
            .count();
        let read_only_transactions = self
            .transactions
            .values()
            .filter(|tx| tx.read_only)
            .count();
        let staged_writes = self
            .transactions
            .values()
            .map(|tx| tx.writes.len())
            .sum();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions,
            staged_writes,
            snapshots: self.snapshots.len(),
        })
    }
}

fn parse_number(s: &str) -> Option<i64> {
    if s.is_empty() || s == "-0" {
        return None;
    }
    let digits = if let Some(rest) = s.strip_prefix('-') {
        rest
    } else {
        s
    };
    if digits.is_empty()
        || !digits.bytes().all(|b| b.is_ascii_digit())
        || (digits.len() > 1 && digits.starts_with('0'))
    {
        return None;
    }
    s.parse::<i64>().ok()
}
