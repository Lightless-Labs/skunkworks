#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    started: bool,
    pause_time: u64,
}

struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| b.started.cmp(&a.started))
        .then_with(|| {
            if a.started {
                a.pause_time
                    .cmp(&b.pause_time)
                    .then_with(|| a.customer.id.cmp(&b.customer.id))
            } else {
                a.customer
                    .arrival
                    .cmp(&b.customer.arrival)
                    .then_with(|| b.customer.id.cmp(&a.customer.id))
            }
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut seen = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut finished_count = 0usize;
    let mut now = 0u64;

    while finished_count < customers.len() {
        if let Some(active) = running.as_ref() {
            if active.finish == now {
                let id = active.customer.id;
                running = None;
                trace.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            trace.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                customer,
                started: false,
                pause_time: 0,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if now >= active.work_at {
                waiting.sort_by(waiting_cmp);
                if waiting
                    .first()
                    .is_some_and(|w| w.customer.priority < active.customer.priority)
                {
                    let by = waiting[0].customer.id;
                    let active = running.take().unwrap();
                    let remaining = active.finish - now;
                    trace.push(Event {
                        time: now,
                        customer: active.customer.id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    waiting.push(Waiting {
                        customer: active.customer,
                        started: true,
                        pause_time: now,
                    });
                }
            }
        }

        while running.is_none() && !waiting.is_empty() {
            waiting.sort_by(waiting_cmp);
            let item = waiting.remove(0);
            let id = item.customer.id;
            let cost = if last_dispatched.is_some_and(|last| last != id) {
                switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
            })?;
            let finish = work_at.checked_add(item.customer.service.min(
                if item.started {
                    // The stored remaining service is carried in customer.service below.
                    u64::MAX
                } else {
                    u64::MAX
                },
            )).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
            })?;

            // For fresh customers service is the full requirement; paused customers
            // store their remaining service in the service field.
            let remaining = item.customer.service;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
            })?;

            last_dispatched = Some(id);
            let kind = if item.started {
                Kind::Resume {
                    remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining,
                    work_at,
                    finish,
                }
            };
            trace.push(Event {
                time: now,
                customer: id,
                kind,
            });
            running = Some(Running {
                customer: item.customer,
                remaining,
                work_at,
                finish,
            });

            if finish == now {
                running = None;
                trace.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                finished_count += 1;
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let mut next_tick: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_tick = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            for time in [Some(active.finish), Some(active.work_at)].into_iter().flatten() {
                if time > now {
                    next_tick = Some(next_tick.map_or(time, |current| current.min(time)));
                }
            }
        }

        if let Some(tick) = next_tick {
            now = tick;
        } else {
            break;
        }
    }

    Ok(trace)
}
