#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{FEFF}') { 3 } else { 0 };

    if i == bytes.len() {
        return Ok(Vec::new());
    }

    #[derive(Clone, Copy, PartialEq, Eq)]
    enum State {
        StartField,
        Unquoted,
        Quoted,
        AfterQuote,
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut state = State::StartField;
    let mut opening_quote = 0;
    let mut record = 0;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if state != State::Quoted && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record, byte: i });
            }
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            state = State::StartField;
            i += 2;
            record += 1;
            continue;
        }

        match state {
            State::StartField => match b {
                b',' => {
                    fields.push(String::new());
                    i += 1;
                }
                b'\n' => {
                    fields.push(String::new());
                    records.push(std::mem::take(&mut fields));
                    i += 1;
                    record += 1;
                }
                b'"' => {
                    state = State::Quoted;
                    opening_quote = i;
                    i += 1;
                }
                _ => {
                    field.push(b as char);
                    state = State::Unquoted;
                    i += 1;
                }
            },
            State::Unquoted => match b {
                b',' => {
                    fields.push(std::mem::take(&mut field));
                    state = State::StartField;
                    i += 1;
                }
                b'\n' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    state = State::StartField;
                    i += 1;
                    record += 1;
                }
                b'"' => {
                    return Err(CsvError::UnexpectedQuote { record, byte: i });
                }
                _ => {
                    field.push(b as char);
                    i += 1;
                }
            },
            State::Quoted => {
                if b == b'"' {
                    if bytes.get(i + 1) == Some(&b'"') {
                        field.push('"');
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().expect("valid UTF-8");
                    field.push(ch);
                    i += ch.len_utf8();
                }
            }
            State::AfterQuote => match b {
                b',' => {
                    fields.push(std::mem::take(&mut field));
                    state = State::StartField;
                    i += 1;
                }
                b'\n' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    state = State::StartField;
                    i += 1;
                    record += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte { record, byte: i });
                }
            },
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: opening_quote,
        });
    }

    if state != State::StartField || !fields.is_empty() {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}
