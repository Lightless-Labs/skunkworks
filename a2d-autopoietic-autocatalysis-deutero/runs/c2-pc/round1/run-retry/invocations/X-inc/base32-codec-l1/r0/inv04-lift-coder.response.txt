#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

const ALPHABET: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";

pub fn encode(input: &[u8]) -> String {
    if input.is_empty() {
        return String::new();
    }

    let mut output = String::new();
    let mut accumulator: u16 = 0;
    let mut bits = 0usize;

    for &byte in input {
        accumulator = (accumulator << 8) | u16::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            let value = ((accumulator >> bits) & 0x1f) as usize;
            output.push(ALPHABET[value] as char);
        }
    }

    if bits > 0 {
        let value = ((accumulator << (5 - bits)) & 0x1f) as usize;
        output.push(ALPHABET[value] as char);
    }

    while output.len() % 8 != 0 {
        output.push('=');
    }

    output
}

pub fn decode(input: &str) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();

    // Lexical validation has priority over all structural errors.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && !ALPHABET.contains(&byte) {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);

    // Padding placement.
    if let Some(offset) = bytes[p..].iter().position(|&byte| byte != b'=') {
        return Err(DecodeError::Padding {
            offset: p + offset,
        });
    }

    let residue = p % 8;
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    let expected_padding = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => unreachable!(),
    };

    if n - p != expected_padding || n % 8 != 0 {
        return Err(DecodeError::Padding { offset: p });
    }

    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };

    if unused_bits > 0 {
        let final_value = ALPHABET
            .iter()
            .position(|&symbol| symbol == bytes[p - 1])
            .expect("lexical validation guarantees an alphabet symbol");
        let mask = (1usize << unused_bits) - 1;
        if final_value & mask != 0 {
            return Err(DecodeError::TrailingBits { offset: p - 1 });
        }
    }

    let decoded_capacity = (p / 8) * 5 + (p % 8) * 5 / 8;
    let mut output = Vec::with_capacity(decoded_capacity);
    let mut accumulator: u16 = 0;
    let mut bits = 0usize;

    for &symbol in &bytes[..p] {
        let value = ALPHABET
            .iter()
            .position(|&candidate| candidate == symbol)
            .expect("lexical validation guarantees an alphabet symbol") as u16;
        accumulator = (accumulator << 5) | value;
        bits += 5;

        while bits >= 8 {
            bits -= 8;
            output.push(((accumulator >> bits) & 0xff) as u8);
        }
    }

    Ok(output)
}
