#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = self.pending.clone();
        combined.extend_from_slice(input);

        let complete_len = (combined.len() / 5) * 5;
        let mut output = String::new();
        for chunk in combined[..complete_len].chunks_exact(5) {
            encode_symbols(chunk, 8, self.config.alphabet, &mut output);
        }

        self.pending = combined[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if !self.pending.is_empty() {
            let symbol_count = (self.pending.len() * 8 + 4) / 5;
            encode_symbols(
                &self.pending,
                symbol_count,
                self.config.alphabet,
                &mut output,
            );
            if self.config.padding == Padding::Required {
                while output.len() % 8 != 0 {
                    output.push('=');
                }
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_byte(alphabet: Alphabet, value: usize) -> u8 {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"[value],
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV"[value],
    }
}

fn encode_symbols(input: &[u8], symbol_count: usize, alphabet: Alphabet, output: &mut String) {
    let total_bits = input.len() * 8;
    for symbol_index in 0..symbol_count {
        let mut value = 0usize;
        for bit_index in 0..5 {
            let source_bit = symbol_index * 5 + bit_index;
            value <<= 1;
            if source_bit < total_bits {
                let byte = input[source_bit / 8];
                value |= ((byte >> (7 - (source_bit % 8))) & 1) as usize;
            }
        }
        output.push(alphabet_byte(alphabet, value) as char);
    }
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    bit_buffer: u32,
    bit_count: u8,
    last_data_offset: Option<usize>,
    tail_bits: u8,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            bit_buffer: 0,
            bit_count: 0,
            last_data_offset: None,
            tail_bits: 0,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: None,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            let recognized = classify(byte, next.config.alphabet);

            match recognized {
                Classified::Invalid => {
                    return Err(StreamError::InvalidByte { offset, byte });
                }
                Classified::Data { value, uppercase } => {
                    if next.config.canonical && byte != uppercase {
                        return Err(StreamError::NonCanonical {
                            offset,
                            expected: uppercase,
                        });
                    }
                    if next.first_padding_offset.is_some() {
                        return Err(StreamError::Padding { offset });
                    }

                    next.data_count += 1;
                    next.last_data_offset = Some(offset);
                    next.tail_bits = value;
                    next.bit_buffer = (next.bit_buffer << 5) | value as u32;
                    next.bit_count += 5;

                    while next.bit_count >= 8 {
                        next.bit_count -= 8;
                        output.push(((next.bit_buffer >> next.bit_count) & 0xff) as u8);
                    }
                    if next.bit_count == 0 {
                        next.bit_buffer = 0;
                    } else {
                        next.bit_buffer &= (1u32 << next.bit_count) - 1;
                    }
                }
                Classified::Padding => {
                    if next.config.padding == Padding::Omitted {
                        return Err(StreamError::Padding { offset });
                    }

                    if next.first_padding_offset.is_none() {
                        let residue = next.data_count % 8;
                        let expected = match residue {
                            2 => 6,
                            4 => 4,
                            5 => 3,
                            7 => 1,
                            _ => return Err(StreamError::Padding { offset }),
                        };
                        next.first_padding_offset = Some(offset);
                        next.expected_padding = Some(expected);
                    }

                    next.padding_count += 1;
                    if next.padding_count > next.expected_padding.unwrap_or(0) {
                        return Err(StreamError::Padding { offset });
                    }
                }
            }

            next.position += 1;
        }

        self.apply_state(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => return Err(StreamError::InvalidLength { offset: self.position }),
        };

        if self.config.padding == Padding::Required && self.padding_count != expected_padding {
            return Err(StreamError::Padding {
                offset: self.first_padding_offset.unwrap_or(self.position),
            });
        }

        let unused_bits = match residue {
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused_bits != 0 {
            let mask = (1u8 << unused_bits) - 1;
            if self.tail_bits & mask != 0 {
                return Err(StreamError::TrailingBits {
                    offset: self.last_data_offset.unwrap_or(self.position),
                });
            }
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> DecoderState {
        DecoderState {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            last_data_offset: self.last_data_offset,
            tail_bits: self.tail_bits,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
        }
    }

    fn apply_state(&mut self, state: DecoderState) {
        self.position = state.position;
        self.data_count = state.data_count;
        self.bit_buffer = state.bit_buffer;
        self.bit_count = state.bit_count;
        self.last_data_offset = state.last_data_offset;
        self.tail_bits = state.tail_bits;
        self.first_padding_offset = state.first_padding_offset;
        self.padding_count = state.padding_count;
        self.expected_padding = state.expected_padding;
    }
}

struct DecoderState {
    config: Config,
    position: usize,
    data_count: usize,
    bit_buffer: u32,
    bit_count: u8,
    last_data_offset: Option<usize>,
    tail_bits: u8,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
}

enum Classified {
    Data { value: u8, uppercase: u8 },
    Padding,
    Invalid,
}

fn classify(byte: u8, alphabet: Alphabet) -> Classified {
    if byte == b'=' {
        return Classified::Padding;
    }

    let uppercase = byte.to_ascii_uppercase();
    let value = match alphabet {
        Alphabet::Standard => match uppercase {
            b'A'..=b'Z' => Some(uppercase - b'A'),
            b'2'..=b'7' => Some(uppercase - b'2' + 26),
            _ => None,
        },
        Alphabet::Hex => match uppercase {
            b'0'..=b'9' => Some(uppercase - b'0'),
            b'A'..=b'V' => Some(uppercase - b'A' + 10),
            _ => None,
        },
    };

    match value {
        Some(value) => Classified::Data { value, uppercase },
        None => Classified::Invalid,
    }
}
