#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_order(constraint: &Constraint) -> (u8, Version) {
    match *constraint {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort_by_key(constraint_order);
    constraints.dedup();
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut constraints = constraints.to_vec();
    canonicalize(&mut constraints);
    constraints
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(text: &str) -> String {
    let mut encoded = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            encoded.push(*byte as char);
        } else {
            encoded.push('%');
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    encoded
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(wanted) => version == wanted,
        Constraint::AtLeast(minimum) => version >= minimum,
        Constraint::Before(limit) => version < limit,
        Constraint::Caret(base) => {
            version >= base
                && if base.major > 0 {
                    version.major == base.major
                } else if base.minor > 0 {
                    version.major == 0 && version.minor == base.minor
                } else {
                    version == base
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

fn validate_inputs(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn validate_duplicates(catalog: &[Package]) -> Result<(), ResolveError> {
    for i in 0..catalog.len() {
        if catalog[..i]
            .iter()
            .any(|previous| previous.name == catalog[i].name)
        {
            return Err(ResolveError::DuplicatePackage {
                package: catalog[i].name.clone(),
            });
        }
    }

    for package in catalog {
        for (i, release) in package.releases.iter().enumerate() {
            if package.releases[..i]
                .iter()
                .any(|previous| previous.version == release.version)
            {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    Ok(())
}

fn resolve_active(
    catalog: &[Package],
    constraints: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<Vec<Selection>, ResolveError> {
    use std::collections::BTreeSet;

    for requirements in constraints.values_mut() {
        canonicalize(requirements);
    }

    let active_names: BTreeSet<String> = constraints.keys().cloned().collect();

    for name in &active_names {
        if !catalog.iter().any(|package| package.name == *name) {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }
    }

    for name in &active_names {
        if let Some(version) = selected.get(name) {
            let requirements = constraints.get(name).cloned().unwrap_or_default();
            if !matches_all(*version, &requirements) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements,
                });
            }
        }
    }

    for name in &active_names {
        if selected.contains_key(name) {
            continue;
        }
        let requirements = constraints.get(name).cloned().unwrap_or_default();
        let package = catalog.iter().find(|package| package.name == *name).unwrap();
        if !package
            .releases
            .iter()
            .any(|release| matches_all(release.version, &requirements))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements,
            });
        }
    }

    let next = active_names
        .iter()
        .find(|name| !selected.contains_key(*name))
        .cloned();

    let Some(name) = next else {
        return Ok(selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect());
    };

    let requirements = constraints.get(&name).cloned().unwrap_or_default();
    let package = catalog.iter().find(|package| package.name == name).unwrap();
    let version = package
        .releases
        .iter()
        .filter(|release| matches_all(release.version, &requirements))
        .map(|release| release.version)
        .max()
        .unwrap();

    selected.insert(name.clone(), version);
    let release = package
        .releases
        .iter()
        .find(|release| release.version == version)
        .unwrap();

    for dependency in &release.deps {
        constraints
            .entry(dependency.package.clone())
            .or_default()
            .push(dependency.constraint);
    }

    resolve_active(catalog, constraints, selected)
}

pub fn resolve(catalog: &[Package], roots: &[Requirement]) -> Result<Vec<Selection>, ResolveError> {
    validate_inputs(catalog, roots)?;
    validate_duplicates(catalog)?;

    let mut constraints = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    resolve_active(
        catalog,
        &mut constraints,
        &mut std::collections::BTreeMap::new(),
    )
}
