#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName { site: String, name: String },
    DuplicatePackage { package: String },
    DuplicateVersion { package: String, version: Version },
    DuplicateFeature { package: String, version: Version, feature: String },
    DuplicateLock { package: String },
    Missing { package: String, evidence: Vec<Evidence> },
    NoMatch { package: String, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    Conflict { package: String, selected: Version, requirements: Vec<Constraint>, evidence: Vec<Evidence> },
    MissingFeature { package: String, selected: Version, feature: String, path: Vec<String> },
    Limit { attempts: usize },
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut e = evidence.to_vec();
    e.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    e
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } =>
            format!("invalid-name|site={}|name={}", encode(site), encode(name)),
        ResolveError::DuplicatePackage { package } =>
            format!("duplicate-package|package={}", encode(package)),
        ResolveError::DuplicateVersion { package, version } =>
            format!("duplicate-version|package={}|version={}", encode(package), version_text(*version)),
        ResolveError::DuplicateFeature { package, version, feature } =>
            format!("duplicate-feature|package={}|version={}|feature={}", encode(package), version_text(*version), encode(feature)),
        ResolveError::DuplicateLock { package } =>
            format!("duplicate-lock|package={}", encode(package)),
        ResolveError::Missing { package, evidence } =>
            format!("missing|package={}{}", encode(package), format_evidence(evidence)),
        ResolveError::NoMatch { package, requirements, evidence } =>
            format!("no-match|package={}|constraints={}{}", encode(package), constraints_text(requirements), format_evidence(evidence)),
        ResolveError::Conflict { package, selected, requirements, evidence } =>
            format!("conflict|package={}|selected={}|constraints={}{}", encode(package), version_text(*selected), constraints_text(requirements), format_evidence(evidence)),
        ResolveError::MissingFeature { package, selected, feature, path } =>
            format!("missing-feature|package={}|selected={}|feature={}\n  F:{} via {}", encode(package), version_text(*selected), encode(feature), encode(feature), path_text(path)),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs).iter().map(|c| constraint_text(*c)).collect::<Vec<_>>().join(",")
}

fn format_evidence(evidence: &[Evidence]) -> String {
    evidence_sorted(evidence)
        .iter()
        .map(|e| format!("\n  C:{} via {}", constraint_text(e.constraint), path_text(&e.path)))
        .collect::<String>()
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{}]", i), name: root.package.clone() });
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName { site: format!("root[{}].feature[{}]", i, f), name: name.clone() });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{}]", i), name: package.name.clone() });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName { site: base.clone(), name: dep.package.clone() });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName { site: format!("{}.gate", base), name: gate.clone() });
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("{}.feature[{}]", base, f), name: name.clone() });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].feature[{}]", i, j, f), name: feature.name.clone() });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{}].release[{}].feature[{}].enable[{}]", i, j, f, e), name: name.clone() });
                    }
                }
            }
        }
    }
    let mut names = std::collections::BTreeSet::new();
    for p in catalog {
        if !names.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }
    for p in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion { package: p.name.clone(), version: r.version });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut features = std::collections::BTreeSet::new();
            for f in &r.features {
                if !features.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature { package: p.name.clone(), version: r.version, feature: f.name.clone() });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName { site: format!("lock[{}]", i), name: lock.package.clone() });
        }
    }
    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: lock.package.clone() });
        }
    }
    Ok(())
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct NodeInfo {
    origins: std::collections::BTreeMap<String, Vec<Origin>>,
    requested: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    activated: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    paths: std::collections::BTreeMap<String, Vec<String>>,
}

fn target_name(t: Target) -> &'static str {
    match t { Target::Linux => "linux", Target::Mac => "mac", Target::Windows => "windows" }
}

fn platform_active(p: Platform, t: Target) -> bool {
    p == Platform::Any || matches!((p, t), (Platform::Linux, Target::Linux) | (Platform::Mac, Target::Mac) | (Platform::Windows, Target::Windows))
}

fn make_dep_token(source: &str, version: Version, index: usize, d: &Dependency) -> String {
    let fs = {
        let mut f = d.features.clone();
        f.sort();
        f.dedup();
        f.join(",")
    };
    format!("{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source, version_text(version), index, d.package, constraint_text(d.constraint), fs,
        d.gate.as_deref().unwrap_or("*"), match d.platform { Platform::Any => "any", Platform::Linux => "linux", Platform::Mac => "mac", Platform::Windows => "windows" })
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!("root[{}]:{}@{}+{{{}}}", i, r.package, constraint_text(r.constraint), fs.join(","))
}

fn compute_info(catalog: &[Package], roots: &[Requirement], selected: &std::collections::BTreeMap<String, Version>, target: Target) -> NodeInfo {
    use std::collections::{BTreeMap, BTreeSet, VecDeque};
    let packages: BTreeMap<&str, &Package> = catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut info = NodeInfo::default();
    let mut queue = VecDeque::new();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![root_token(i, root)];
        info.origins.entry(root.package.clone()).or_default().push(Origin { constraint: root.constraint, path: path.clone() });
        info.requested.entry(root.package.clone()).or_default().extend(root.features.iter().cloned());
        if info.paths.get(&root.package).map_or(true, |old| path.len() < old.len() || (path.len() == old.len() && path < *old)) {
            info.paths.insert(root.package.clone(), path);
        }
        queue.push_back(root.package.clone());
    }

    let mut processed = BTreeSet::<(String, String)>::new();
    while let Some(pkg_name) = queue.pop_front() {
        let Some(version) = selected.get(&pkg_name).copied() else { continue };
        let Some(pkg) = packages.get(pkg_name.as_str()).copied() else { continue };
        let Some(release) = pkg.releases.iter().find(|r| r.version == version) else { continue };

        let reqs = info.requested.get(&pkg_name).cloned().unwrap_or_default();
        let mut activated = info.activated.get(&pkg_name).cloned().unwrap_or_default();
        let mut changed = true;
        while changed {
            changed = false;
            let names: Vec<String> = activated.iter().cloned().collect();
            for name in names {
                if let Some(feature) = release.features.iter().find(|f| f.name == name) {
                    for enable in &feature.enables {
                        if activated.insert(enable.clone()) {
                            changed = true;
                        }
                    }
                }
            }
            for request in &reqs {
                if activated.insert(request.clone()) { changed = true; }
            }
        }
        let previous = info.activated.insert(pkg_name.clone(), activated.clone()).unwrap_or_default();
        let newly: Vec<String> = activated.difference(&previous).cloned().collect();

        for feature in newly {
            if let Some(decl) = release.features.iter().find(|f| f.name == feature) {
                for enable in &decl.enables {
                    info.activated.entry(pkg_name.clone()).or_default().insert(enable.clone());
                }
            }
        }

        for (index, dep) in release.deps.iter().enumerate() {
            if !platform_active(dep.platform, target) || dep.gate.as_ref().is_some_and(|g| !activated.contains(g)) {
                continue;
            }
            let key = (pkg_name.clone(), index.to_string());
            if !processed.insert(key) { continue; }
            let token = make_dep_token(&pkg_name, version, index, dep);
            let mut path = info.paths.get(&pkg_name).cloned().unwrap_or_default();
            path.push(token);
            info.origins.entry(dep.package.clone()).or_default().push(Origin { constraint: dep.constraint, path: path.clone() });
            info.requested.entry(dep.package.clone()).or_default().extend(dep.features.iter().cloned());
            if info.paths.get(&dep.package).map_or(true, |old| path.len() < old.len() || (path.len() == old.len() && path < *old)) {
                info.paths.insert(dep.package.clone(), path);
            }
            queue.push_back(dep.package.clone());
        }
    }

    // Revisit dependencies if a better canonical path or additional request arrived.
    // A bounded monotone pass settles paths and edge activation for the finite graph.
    let _ = target_name(target);
    info
}

fn evidence_for(info: &NodeInfo, package: &str) -> Vec<Evidence> {
    info.origins.get(package).into_iter().flatten().map(|o| Evidence { constraint: o.constraint, path: o.path.clone() }).collect()
}

fn canonical_requirements(info: &NodeInfo, package: &str) -> Vec<Constraint> {
    canonical_constraints(&info.origins.get(package).into_iter().flatten().map(|o| o.constraint).collect::<Vec<_>>())
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    budget: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, Vec<Option<Version>>, std::collections::BTreeMap<String, Version>, NodeInfo)>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn inspect(&mut self, selected: &mut std::collections::BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let info = compute_info(self.catalog, self.roots, selected, self.target);
        let active: Vec<String> = info.origins.keys().cloned().collect();
        let packages: std::collections::BTreeMap<&str, &Package> = self.catalog.iter().map(|p| (p.name.as_str(), p)).collect();

        let missing: Vec<String> = active.iter().filter(|n| !packages.contains_key(n.as_str())).cloned().collect();
        if let Some(package) = missing.into_iter().next() {
            return self.fail(ResolveError::Missing { evidence: evidence_for(&info, &package), package });
        }

        for package in selected.keys() {
            if !info.origins.contains_key(package) { continue; }
            let v = selected[package];
            let reqs = canonical_requirements(&info, package);
            if !reqs.iter().all(|c| matches(*c, v)) {
                return self.fail(ResolveError::Conflict { package: package.clone(), selected: v, requirements: reqs, evidence: evidence_for(&info, package) });
            }
        }

        for package in &active {
            if let Some(v) = selected.get(package) {
                let activated = info.activated.get(package).cloned().unwrap_or_default();
                let p = packages[package.as_str()];
                let rel = p.releases.iter().find(|r| r.version == *v).unwrap();
                for feature in activated {
                    if !rel.features.iter().any(|f| f.name == feature) {
                        return self.fail(ResolveError::MissingFeature { package: package.clone(), selected: *v, feature, path: info.paths.get(package).cloned().unwrap_or_default() });
                    }
                }
            }
        }

        for package in &active {
            if selected.contains_key(package) { continue; }
            let p = packages[package.as_str()];
            let reqs = canonical_requirements(&info, package);
            if !p.releases.iter().any(|r| reqs.iter().all(|c| matches(*c, r.version))) {
                return self.fail(ResolveError::NoMatch { package: package.clone(), requirements: reqs, evidence: evidence_for(&info, package) });
            }
        }

        let Some(package) = active.iter().find(|p| !selected.contains_key(*p)).cloned() else {
            let vector: Vec<Option<Version>> = self.catalog_names.iter().map(|n| selected.get(n).copied()).collect();
            let cost = self.locks.iter().filter(|l| selected.get(&l.package) != Some(&l.version)).count();
            let replace = self.best.as_ref().map_or(true, |(bc, bv, _, _)| cost < *bc || (cost == *bc && vector > *bv));
            if replace {
                self.best = Some((cost, vector, selected.clone(), info));
            }
            return Ok(());
        };

        let p = packages[package.as_str()];
        let reqs = canonical_requirements(&info, &package);
        let mut candidates: Vec<Version> = p.releases.iter().filter(|r| reqs.iter().all(|c| matches(*c, r.version))).map(|r| r.version).collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == package) {
            if let Some(i) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(i);
                candidates.insert(0, locked);
            }
        }
        for candidate in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit { attempts: self.budget });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            let result = self.inspect(selected);
            selected.remove(&package);
            if let Err(e) = result {
                if matches!(e, ResolveError::Limit { .. }) { return Err(e); }
                if self.first_error.is_none() { self.first_error = Some(e); }
            }
        }
        Ok(())
    }

    fn fail(&mut self, error: ResolveError) -> Result<(), ResolveError> {
        if self.first_error.is_none() { self.first_error = Some(error); }
        Ok(())
    }
}

pub fn resolve(catalog: &[Package], roots: &[Requirement], locks: &[Lock], target: Target, max_attempts: usize) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;
    let mut names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    names.sort();
    let mut search = Search {
        catalog, roots, locks, target, budget: max_attempts, attempts: 0,
        first_error: None, best: None, catalog_names: names,
    };
    let mut selected = std::collections::BTreeMap::new();
    search.inspect(&mut selected)?;
    if let Some((_, _, chosen, info)) = search.best {
        let selected_vec: Vec<Selection> = chosen.iter().map(|(package, version)| Selection { package: package.clone(), version: *version }).collect();
        let locks_by_name: std::collections::BTreeMap<&str, Version> = locks.iter().map(|l| (l.package.as_str(), l.version)).collect();
        let selected_by_name: std::collections::BTreeMap<&str, Version> = chosen.iter().map(|(p, v)| (p.as_str(), *v)).collect();
        let mut all: std::collections::BTreeSet<String> = locks.iter().map(|l| l.package.clone()).collect();
        all.extend(chosen.keys().cloned());
        let mut changes = Vec::new();
        for package in all {
            let before = locks_by_name.get(package.as_str()).copied();
            let after = selected_by_name.get(package.as_str()).copied();
            if before == after { continue; }
            let reason = match (before, after) {
                (None, Some(_)) => ChangeReason::Added,
                (Some(_), None) => ChangeReason::Removed,
                (Some(old), Some(_)) => {
                    let exists = catalog.iter().find(|p| p.name == package).is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                    if !exists { ChangeReason::Unavailable }
                    else if !canonical_requirements(&info, &package).iter().all(|c| matches(*c, old)) { ChangeReason::Rejected }
                    else { ChangeReason::Rebalanced }
                }
                _ => unreachable!(),
            };
            changes.push(Change { package, before, after, reason });
        }
        return Ok(Resolution { selected: selected_vec, changes });
    }
    Err(search.first_error.unwrap_or(ResolveError::NoMatch {
        package: roots.first().map(|r| r.package.clone()).unwrap_or_default(),
        requirements: Vec::new(),
        evidence: Vec::new(),
    }))
}
