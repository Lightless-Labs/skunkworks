fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(values: &[Constraint]) -> Vec<Constraint> {
    let mut out = values.to_vec();
    out.sort();
    out.dedup();
    out
}

fn constraints_text(values: &[Constraint]) -> String {
    canonical_constraints(values)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn constraint_matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn percent_encode(value: &str) -> String {
    let mut out = String::new();
    for b in value.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(line: &mut String, evidence: &[Evidence]) {
    for item in evidence_sorted(evidence) {
        line.push_str("\n  C:");
        line.push_str(&constraint_text(item.constraint));
        line.push_str(" via ");
        line.push_str(&item.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            percent_encode(site),
            percent_encode(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", percent_encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            percent_encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            percent_encode(package),
            version_text(*version),
            percent_encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", percent_encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut line = format!("missing|package={}", percent_encode(package));
            append_evidence(&mut line, evidence);
            line
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut line = format!(
                "no-match|package={}|constraints={}",
                percent_encode(package),
                constraints_text(requirements)
            );
            append_evidence(&mut line, evidence);
            line
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut line = format!(
                "conflict|package={}|selected={}|constraints={}",
                percent_encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut line, evidence);
            line
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            percent_encode(package),
            version_text(*selected),
            percent_encode(feature),
            percent_encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone, Default)]
struct BTreeStringSet(std::collections::BTreeSet<String>);

impl BTreeStringSet {
    fn insert(&mut self, value: String) -> bool {
        self.0.insert(value)
    }

    fn contains(&self, value: &str) -> bool {
        self.0.contains(value)
    }

    fn iter(&self) -> impl Iterator<Item = &String> {
        self.0.iter()
    }
}

#[derive(Clone, Default)]
struct Closure {
    states: std::collections::BTreeMap<String, PackageState>,
    active_deps: std::collections::BTreeSet<(String, usize)>,
    paths: std::collections::BTreeMap<String, Vec<String>>,
}

fn push_request(
    states: &mut std::collections::BTreeMap<String, PackageState>,
    package: String,
    constraint: Constraint,
    features: &[String],
    path: Vec<String>,
) {
    let state = states.entry(package).or_default();
    state.origins.push(Origin { constraint, path });
    for feature in features {
        state.features.insert(feature.clone());
    }
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn dependency_token(
    source: &str,
    version: Version,
    index: usize,
    dep: &Dependency,
) -> String {
    let features = {
        let mut f = dep.features.clone();
        f.sort();
        f.dedup();
        f.join(",")
    };
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features,
        gate,
        platform
    )
}

fn root_token(index: usize, req: &Requirement) -> String {
    let mut features = req.features.clone();
    features.sort();
    features.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        req.package,
        constraint_text(req.constraint),
        features.join(",")
    )
}

fn target_matches(platform: Platform, target: Target) -> bool {
    matches!(
        (platform, target),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn find_release<'a>(catalog: &'a [Package], name: &str, version: Version) -> Option<&'a Release> {
    catalog
        .iter()
        .find(|p| p.name == name)
        .and_then(|p| p.releases.iter().find(|r| r.version == version))
}

fn compute_closure(
    catalog: &[Package],
    roots: &[Requirement],
    assigned: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut closure = Closure::default();
    for (i, root) in roots.iter().enumerate() {
        push_request(
            &mut closure.states,
            root.package.clone(),
            root.constraint,
            &root.features,
            vec![root_token(i, root)],
        );
    }

    loop {
        let mut changed = false;
        let names: Vec<String> = closure.states.keys().cloned().collect();

        for name in names {
            let Some(version) = assigned.get(&name).copied() else {
                continue;
            };
            let Some(release) = find_release(catalog, &name, version) else {
                continue;
            };

            let requested: Vec<String> = closure.states[&name]
                .features
                .iter()
                .cloned()
                .collect();

            for feature_name in requested {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if closure.states.get_mut(&name).unwrap().features.insert(enabled.clone())
                        {
                            changed = true;
                        }
                    }
                }
            }

            let activated_features: Vec<String> =
                closure.states[&name].features.iter().cloned().collect();
            for (index, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !closure.states[&name].features.contains(g))
                {
                    continue;
                }

                let key = (name.clone(), index);
                if closure.active_deps.insert(key) {
                    changed = true;
                }

                let source_path = closure.paths.get(&name).cloned().unwrap_or_default();
                let token = dependency_token(&name, version, index, dep);
                let mut path = source_path;
                path.push(token);

                let state = closure.states.entry(dep.package.clone()).or_default();
                let before_origins = state.origins.len();
                state.origins.push(Origin {
                    constraint: dep.constraint,
                    path: path.clone(),
                });
                if state.origins.len() != before_origins {
                    changed = true;
                }
                for feature in &dep.features {
                    if state.features.insert(feature.clone()) {
                        changed = true;
                    }
                }

                if closure
                    .paths
                    .get(&dep.package)
                    .map_or(true, |old| path_better(&path, old))
                {
                    closure.paths.insert(dep.package.clone(), path);
                    changed = true;
                }
            }

            let _ = activated_features;
        }

        for (name, state) in &closure.states {
            for origin in &state.origins {
                let candidate = origin.path.clone();
                if closure
                    .paths
                    .get(name)
                    .map_or(true, |old| path_better(&candidate, old))
                {
                    closure.paths.insert(name.clone(), candidate);
                    changed = true;
                }
            }
        }

        if !changed {
            break;
        }
    }

    closure
}

fn canonical_evidence(state: &PackageState) -> Vec<Evidence> {
    let mut evidence: Vec<Evidence> = state
        .origins
        .iter()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect();
    evidence.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    evidence
}

fn active_constraints(state: &PackageState) -> Vec<Constraint> {
    canonical_constraints(&state.origins.iter().map(|o| o.constraint).collect::<Vec<_>>())
}

fn validation(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].gate",
                                i, j, k
                            ),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = std::collections::BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    Ok(())
}

fn node_error(
    catalog: &[Package],
    closure: &Closure,
    assigned: &std::collections::BTreeMap<String, Version>,
) -> Option<ResolveError> {
    for name in closure
        .states
        .keys()
        .filter(|name| !catalog.iter().any(|p| p.name == ***name))
    {
        return Some(ResolveError::Missing {
            package: name.clone(),
            evidence: canonical_evidence(&closure.states[name]),
        });
    }

    for (name, state) in &closure.states {
        if let Some(version) = assigned.get(name) {
            let constraints = active_constraints(state);
            if !constraints
                .iter()
                .all(|c| constraint_matches(*c, *version))
            {
                return Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints,
                    evidence: canonical_evidence(state),
                });
            }
        }
    }

    for (name, state) in &closure.states {
        if assigned.contains_key(name) {
            continue;
        }
        let constraints = active_constraints(state);
        let package = catalog.iter().find(|p| p.name == *name).unwrap();
        if !package.releases.iter().any(|release| {
            constraints
                .iter()
                .all(|c| constraint_matches(*c, release.version))
        }) {
            return Some(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints,
                evidence: canonical_evidence(state),
            });
        }
    }

    for (name, state) in &closure.states {
        let Some(version) = assigned.get(name) else {
            continue;
        };
        let release = find_release(catalog, name, *version).unwrap();
        for feature in state.features.iter() {
            if !release.features.iter().any(|f| f.name == *feature) {
                return Some(ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: closure.paths.get(name).cloned().unwrap_or_default(),
                });
            }
        }
    }

    None
}

struct SearchResult {
    best: Option<std::collections::BTreeMap<String, Version>>,
    first_error: Option<ResolveError>,
    best_cost: usize,
}

fn solution_cost(
    assigned: &std::collections::BTreeMap<String, Version>,
    locks: &[Lock],
) -> usize {
    locks
        .iter()
        .filter(|lock| assigned.get(&lock.package) != Some(&lock.version))
        .count()
}

fn better_vector(
    a: &std::collections::BTreeMap<String, Version>,
    b: &std::collections::BTreeMap<String, Version>,
    names: &[String],
) -> bool {
    for name in names {
        match (a.get(name), b.get(name)) {
            (Some(x), Some(y)) if x != y => return x > y,
            (Some(_), None) => return true,
            (None, Some(_)) => return false,
            _ => {}
        }
    }
    false
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
    attempts: &mut usize,
    assigned: &mut std::collections::BTreeMap<String, Version>,
    catalog_names: &[String],
    result: &mut SearchResult,
) -> Result<(), ResolveError> {
    let closure = compute_closure(catalog, roots, assigned, target);

    if let Some(error) = node_error(catalog, &closure, assigned) {
        if result.first_error.is_none() {
            result.first_error = Some(error);
        }
        return Ok(());
    }

    let next = closure
        .states
        .keys()
        .find(|name| !assigned.contains_key(*name))
        .cloned();

    let Some(package_name) = next else {
        let cost = solution_cost(assigned, locks);
        let replace = result.best.as_ref().map_or(true, |best| {
            cost < result.best_cost
                || (cost == result.best_cost && better_vector(assigned, best, catalog_names))
        });
        if replace {
            result.best = Some(assigned.clone());
            result.best_cost = cost;
        }
        return Ok(());
    };

    let state = &closure.states[&package_name];
    let constraints = active_constraints(state);
    let package = catalog.iter().find(|p| p.name == package_name).unwrap();
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|release| {
            constraints
                .iter()
                .all(|c| constraint_matches(*c, release.version))
        })
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    if let Some(lock) = locks.iter().find(|l| l.package == package_name) {
        if let Some(index) = candidates.iter().position(|version| *version == lock.version) {
            let version = candidates.remove(index);
            candidates.insert(0, version);
        }
    }

    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        assigned.insert(package_name.clone(), version);
        search(
            catalog,
            roots,
            locks,
            target,
            max_attempts,
            attempts,
            assigned,
            catalog_names,
            result,
        )?;
        assigned.remove(&package_name);
    }

    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validation(catalog, roots, locks)?;

    let mut catalog_names: Vec<String> = catalog.iter().map(|p| p.name.clone()).collect();
    catalog_names.sort();
    catalog_names.dedup();

    let mut assigned = std::collections::BTreeMap::new();
    let mut attempts = 0usize;
    let mut result = SearchResult {
        best: None,
        first_error: None,
        best_cost: usize::MAX,
    };

    search(
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        &mut attempts,
        &mut assigned,
        &catalog_names,
        &mut result,
    )?;

    let Some(best) = result.best else {
        return Err(result.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_closure = compute_closure(catalog, roots, &best, target);
    let mut before = std::collections::BTreeMap::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let mut names: std::collections::BTreeSet<String> = before.keys().cloned().collect();
    names.extend(best.keys().cloned());

    let mut changes = Vec::new();
    for package in names {
        let old = before.get(&package).copied();
        let new = best.get(&package).copied();
        if old == new {
            continue;
        }
        let reason = match (old, new) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old_version), Some(_)) => {
                if find_release(catalog, &package, old_version).is_none() {
                    ChangeReason::Unavailable
                } else {
                    let constraints = final_closure
                        .states
                        .get(&package)
                        .map(active_constraints)
                        .unwrap_or_default();
                    if !constraints
                        .iter()
                        .all(|c| constraint_matches(*c, old_version))
                    {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before: old,
            after: new,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
