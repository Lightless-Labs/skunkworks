fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut v = cs.to_vec();
    v.sort();
    v.dedup();
    v
}

fn constraint_list(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn all_match(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| matches(v, *c))
}

fn enc(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sort_evidence(ev: &[Evidence]) -> Vec<Evidence> {
    let mut out = ev.to_vec();
    out.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn evidence_lines(ev: &[Evidence]) -> String {
    let mut out = String::new();
    for e in sort_evidence(ev) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", enc(site), enc(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", enc(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            enc(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            enc(package),
            version_text(*version),
            enc(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", enc(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            enc(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            enc(package),
            constraint_list(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            enc(package),
            version_text(*selected),
            constraint_list(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            enc(package),
            version_text(*selected),
            enc(feature),
            enc(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone, Default)]
struct Active {
    origins: Vec<Origin>,
    requests: BTreeSet<String>,
    activated: BTreeSet<String>,
    paths: Vec<Vec<String>>,
}

struct Closure {
    active: BTreeMap<String, Active>,
    missing_feature: Option<(String, Version, String, Vec<String>)>,
}

fn platform_name(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn target_matches(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn feature_map(release: &Release) -> BTreeMap<&str, &Feature> {
    release.features.iter().map(|f| (f.name.as_str(), f)).collect()
}

fn closure(
    roots: &[Requirement],
    catalog: &BTreeMap<String, Package>,
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut active: BTreeMap<String, Active> = BTreeMap::new();

    for (i, root) in roots.iter().enumerate() {
        let fs = {
            let mut f = root.features.clone();
            f.sort();
            f.dedup();
            f.join(",")
        };
        let token = format!(
            "root[{}]:{}@{}+{{{}}}",
            i,
            root.package,
            constraint_text(root.constraint),
            fs
        );
        let entry = active.entry(root.package.clone()).or_default();
        entry.origins.push(Origin {
            constraint: root.constraint,
            path: vec![token.clone()],
        });
        entry.requests.extend(root.features.iter().cloned());
        entry.paths.push(vec![token]);
    }

    loop {
        let before = active.clone();

        let names: Vec<String> = active.keys().cloned().collect();
        for name in names {
            let Some(version) = selected.get(&name).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&name) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let fmap = feature_map(release);
            let mut progress = true;
            while progress {
                progress = false;
                let requested: Vec<String> = active[&name].requests.iter().cloned().collect();
                for feat_name in requested {
                    if active.get_mut(&name).unwrap().activated.insert(feat_name.clone()) {
                        progress = true;
                    }
                    if let Some(feat) = fmap.get(feat_name.as_str()) {
                        for enabled in &feat.enables {
                            if active
                                .get_mut(&name)
                                .unwrap()
                                .requests
                                .insert(enabled.clone())
                            {
                                progress = true;
                            }
                        }
                    }
                }
            }

            let activated = active[&name].activated.clone();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }
                let fs = {
                    let mut f = dep.features.clone();
                    f.sort();
                    f.dedup();
                    f.join(",")
                };
                let gate = dep.gate.as_deref().unwrap_or("*");
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    name,
                    version_text(version),
                    j,
                    dep.package,
                    constraint_text(dep.constraint),
                    fs,
                    gate,
                    platform_name(dep.platform)
                );
                let source_path = active[&name]
                    .paths
                    .iter()
                    .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)))
                    .cloned()
                    .unwrap_or_default();
                let mut path = source_path;
                path.push(token);
                let destination = active.entry(dep.package.clone()).or_default();
                if !destination.origins.iter().any(|o| {
                    o.constraint == dep.constraint && o.path == path
                }) {
                    destination.origins.push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                }
                if !destination.paths.contains(&path) {
                    destination.paths.push(path);
                }
                destination.requests.extend(dep.features.iter().cloned());
            }
        }

        if active == before {
            break;
        }
    }

    // Canonical provenance is the shortest, then lexicographically first path.
    for entry in active.values_mut() {
        entry
            .paths
            .sort_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)));
        entry.paths.dedup();
    }

    let mut missing_feature = None;
    for (name, state) in &active {
        let Some(version) = selected.get(name).copied() else {
            continue;
        };
        let Some(package) = catalog.get(name) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == version) else {
            continue;
        };
        let declared: BTreeSet<&str> = release.features.iter().map(|f| f.name.as_str()).collect();
        let path = state.paths.first().cloned().unwrap_or_default();
        for feature in &state.activated {
            if !declared.contains(feature.as_str()) {
                missing_feature = Some((name.clone(), version, feature.clone(), path.clone()));
                break;
            }
        }
        if missing_feature.is_some() {
            break;
        }
    }

    Closure {
        active,
        missing_feature,
    }
}

fn canonical_path(state: &Active) -> Vec<String> {
    state.paths.first().cloned().unwrap_or_default()
}

fn make_evidence(state: &Active) -> Vec<Evidence> {
    state
        .origins
        .iter()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect()
}

fn make_dynamic_error(
    active: &BTreeMap<String, Active>,
    catalog: &BTreeMap<String, Package>,
    selected: &BTreeMap<String, Version>,
    closure: &Closure,
) -> Option<ResolveError> {
    for package in active.keys() {
        if !catalog.contains_key(package) {
            return Some(ResolveError::Missing {
                package: package.clone(),
                evidence: make_evidence(&active[package]),
            });
        }
    }

    if let Some((package, version, feature, path)) = &closure.missing_feature {
        return Some(ResolveError::MissingFeature {
            package: package.clone(),
            selected: *version,
            feature: feature.clone(),
            path: path.clone(),
        });
    }

    for (package, version) in selected {
        let Some(state) = active.get(package) else {
            continue;
        };
        let constraints: Vec<Constraint> = state.origins.iter().map(|o| o.constraint).collect();
        if !all_match(*version, &constraints) {
            return Some(ResolveError::Conflict {
                package: package.clone(),
                selected: *version,
                requirements: canonical_constraints(&constraints),
                evidence: make_evidence(state),
            });
        }
    }

    for (package, state) in active {
        if selected.contains_key(package) {
            continue;
        }
        let constraints: Vec<Constraint> = state.origins.iter().map(|o| o.constraint).collect();
        let candidates = catalog[package]
            .releases
            .iter()
            .any(|r| all_match(r.version, &constraints));
        if !candidates {
            return Some(ResolveError::NoMatch {
                package: package.clone(),
                requirements: canonical_constraints(&constraints),
                evidence: make_evidence(state),
            });
        }
    }
    None
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, Package>,
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl Search<'_> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let cl = closure(self.roots, self.catalog, selected, self.target);
        if let Some(err) = make_dynamic_error(&cl.active, self.catalog, selected, &cl) {
            if self.first_error.is_none() {
                self.first_error = Some(err);
            }
            return Ok(());
        }

        let Some(package) = cl
            .active
            .keys()
            .find(|p| !selected.contains_key(*p))
            .cloned()
        else {
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let replace = self.best.as_ref().is_none_or(|(best_cost, best)| {
                if cost != *best_cost {
                    cost < *best_cost
                } else {
                    self.version_vector_cmp(selected, best).is_gt()
                }
            });
            if replace {
                self.best = Some((cost, selected.clone()));
            }
            return Ok(());
        };

        let constraints: Vec<Constraint> = cl.active[&package]
            .origins
            .iter()
            .map(|o| o.constraint)
            .collect();
        let mut candidates: Vec<Version> = self.catalog[&package]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| all_match(*v, &constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(locked) = self.locks.get(&package) {
            if let Some(i) = candidates.iter().position(|v| v == locked) {
                candidates.remove(i);
                candidates.insert(0, *locked);
            }
        }

        let mut branch_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            match self.visit(selected) {
                Err(e) => {
                    selected.remove(&package);
                    return Err(e);
                }
                Ok(()) => {}
            }
            selected.remove(&package);
            if branch_error.is_none() {
                branch_error = self.first_error.clone();
            }
        }
        if self.best.is_none() && self.first_error.is_none() {
            self.first_error = branch_error;
        }
        Ok(())
    }

    fn version_vector_cmp(
        &self,
        a: &BTreeMap<String, Version>,
        b: &BTreeMap<String, Version>,
    ) -> std::cmp::Ordering {
        let names: BTreeSet<&String> = self.catalog.keys().collect();
        for name in names {
            match (a.get(name), b.get(name)) {
                (Some(x), Some(y)) => {
                    let ord = x.cmp(y);
                    if !ord.is_eq() {
                        return ord;
                    }
                }
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (None, None) => {}
            }
        }
        std::cmp::Ordering::Equal
    }
}

fn validate_names(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Option<ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Some(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Some(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Some(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Some(ResolveError::InvalidName {
                        site: site.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Some(ResolveError::InvalidName {
                            site: format!("{}.gate", site),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Some(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", site, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Some(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Some(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Some(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    None
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    if let Some(e) = validate_names(catalog, roots, locks) {
        return Err(e);
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map: BTreeMap<String, Package> = catalog
        .iter()
        .cloned()
        .map(|p| (p.name.clone(), p))
        .collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        roots,
        catalog: &catalog_map,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut BTreeMap::new())?;

    let Some((_, chosen)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: roots
                .first()
                .map(|r| r.package.clone())
                .unwrap_or_default(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let final_closure = closure(roots, &catalog_map, &chosen, target);
    let mut all_names: BTreeSet<String> = locks.iter().map(|l| l.package.clone()).collect();
    all_names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints: Vec<Constraint> = final_closure
                        .active
                        .get(&package)
                        .map(|s| s.origins.iter().map(|o| o.constraint).collect())
                        .unwrap_or_default();
                    if !all_match(old, &constraints) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
