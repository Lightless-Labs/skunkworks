#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|item| item == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|item| item == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with_config(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

fn parse_records(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut field_started = false;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().expect("valid UTF-8 boundary");
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                fields.push(std::mem::take(&mut field));
                field_started = false;
                record_started = true;
                after_quote = false;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record_index += 1;
                field_started = false;
                record_started = false;
                after_quote = false;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if b == delimiter {
            fields.push(std::mem::take(&mut field));
            field_started = false;
            record_started = true;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            record_index += 1;
            field_started = false;
            record_started = false;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
            in_quotes = true;
            quote_open = i;
            record_started = true;
            i += 1;
        } else {
            let ch = input[i..].chars().next().expect("valid UTF-8 boundary");
            field.push(ch);
            field_started = true;
            record_started = true;
            i += ch.len_utf8();
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open,
        });
    }

    if record_started || field_started || !fields.is_empty() || !field.is_empty() || after_quote {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}

fn parse_with_config(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let records = parse_records(input, config.delimiter, config.quote)?;

    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut records = records;
            let header = records.remove(0);
            (Some(header), records)
        }
    } else {
        (None, records)
    };

    if let Some(header_values) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut first_seen = std::collections::HashMap::<&str, usize>::new();
            for (second, name) in header_values.iter().enumerate() {
                if let Some(&first) = first_seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second,
                    });
                }
                first_seen.insert(name, second);
            }
        }

        for (data_index, row) in rows.iter().enumerate() {
            if row.len() != header_values.len() {
                return Err(CsvError::HeaderWidth {
                    record: data_index + 1,
                    expected: header_values.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

fn valid_integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let digits = if bytes[0] == b'-' {
        &bytes[1..]
    } else {
        bytes
    };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return bytes[0] != b'-';
    }
    if !(b'1'..=b'9').contains(&digits[0]) {
        return false;
    }
    digits[1..].iter().all(u8::is_ascii_digit)
}

fn valid_decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    let digits = if bytes.first() == Some(&b'-') {
        &bytes[1..]
    } else {
        bytes
    };
    let Some(dot) = digits.iter().position(|&b| b == b'.') else {
        return false;
    };
    if digits[dot + 1..].len() != 2
        || !digits[dot + 1..].iter().all(u8::is_ascii_digit)
    {
        return false;
    }

    let whole = &digits[..dot];
    if whole == b"0" {
        return bytes.first() != Some(&b'-');
    }
    !whole.is_empty()
        && (b'1'..=b'9').contains(&whole[0])
        && whole[1..].iter().all(u8::is_ascii_digit)
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !valid_integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !valid_decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = s.starts_with('-');
    let spelling = if negative { &s[1..] } else { s };
    let (whole, fractional) = spelling.split_once('.').expect("validated decimal");
    let magnitude = whole
        .parse::<i128>()
        .expect("validated ASCII digits")
        .checked_mul(100)
        .and_then(|v| {
            let frac = i128::from(fractional.as_bytes()[0] - b'0') * 10
                + i128::from(fractional.as_bytes()[1] - b'0');
            v.checked_add(frac)
        })
        .expect("decimal spelling fits i128");
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    table: &Table,
    row: &[String],
    row_index: usize,
    spec_index: usize,
    spec: &ColumnSpec,
) -> Result<Value, CellError> {
    let column = match &spec.column {
        ColumnRef::Index(index) => Some(*index),
        ColumnRef::Name(name) => table.column_index(name),
    };

    let Some(column_index) = column else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: None,
            kind: CellErrorKind::UnknownColumn,
        });
    };

    let Some(cell) = row.get(column_index) else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: Some(column_index),
            kind: CellErrorKind::MissingCell,
        });
    };

    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column_index),
                kind: CellErrorKind::Empty,
            });
        }
    }

    let parsed = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.clone())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    };

    parsed.map_err(|kind| CellError {
        row: row_index,
        spec: spec_index,
        column: Some(column_index),
        kind,
    })
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut stop_row = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if stop_row {
                output.push(None);
                continue;
            }

            match evaluate_cell(table, row, row_index, spec_index, spec) {
                Ok(value) => output.push(Some(value)),
                Err(error) => {
                    output.push(None);
                    result.errors.push(error);
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow => stop_row = true,
                        ErrorPolicy::Stop => {
                            for _ in (spec_index + 1)..columns.len() {
                                output.push(None);
                            }
                            result.rows.push(output);
                            return result;
                        }
                    }
                }
            }
        }

        result.rows.push(output);
    }

    result
}
