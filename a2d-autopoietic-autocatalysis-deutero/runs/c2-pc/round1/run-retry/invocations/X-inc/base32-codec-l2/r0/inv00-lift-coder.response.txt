#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

fn symbols(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

pub fn encode(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let alphabet_symbols = symbols(alphabet);
    let mut output = String::new();
    let mut accumulator = 0u32;
    let mut bits = 0usize;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bits += 8;

        while bits >= 5 {
            bits -= 5;
            let value = ((accumulator >> bits) & 0x1f) as usize;
            output.push(alphabet_symbols[value] as char);
        }
    }

    if bits > 0 {
        let value = ((accumulator << (5 - bits)) & 0x1f) as usize;
        output.push(alphabet_symbols[value] as char);
    }

    if padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }

    output
}

pub fn decode(
    input: &str,
    alphabet: Alphabet,
    padding: Padding,
) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let alphabet_symbols = symbols(alphabet);

    // Lexical validation is performed across the entire input before any
    // padding or length rules are considered.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && !alphabet_symbols.contains(&byte) {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let n = bytes.len();
    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);

    match padding {
        Padding::Omitted if p < n => {
            return Err(DecodeError::Padding { offset: p });
        }
        Padding::Required if p < n => {
            if let Some(offset) = bytes[p..]
                .iter()
                .position(|&byte| byte != b'=')
                .map(|relative| p + relative)
            {
                return Err(DecodeError::Padding { offset });
            }
        }
        _ => {}
    }

    let data_len = p;
    let residue = data_len % 8;
    let required_padding = match residue {
        0 => 0,
        2 => 6,
        4 => 4,
        5 => 3,
        7 => 1,
        _ => return Err(DecodeError::InvalidLength { offset: n }),
    };

    if padding == Padding::Required {
        let actual_padding = n - p;
        if actual_padding != required_padding || n % 8 != 0 {
            return Err(DecodeError::Padding { offset: p });
        }
    }

    let trailing_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };

    if trailing_bits > 0 {
        let last_value = alphabet_symbols
            .iter()
            .position(|&symbol| symbol == bytes[data_len - 1])
            .expect("lexical validation guarantees alphabet symbols");
        if last_value & ((1usize << trailing_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits {
                offset: data_len - 1,
            });
        }
    }

    let mut output = Vec::with_capacity(data_len * 5 / 8);
    let mut accumulator = 0u32;
    let mut bits = 0usize;

    for &byte in &bytes[..data_len] {
        let value = alphabet_symbols
            .iter()
            .position(|&symbol| symbol == byte)
            .expect("lexical validation guarantees alphabet symbols") as u32;
        accumulator = (accumulator << 5) | value;
        bits += 5;

        while bits >= 8 {
            bits -= 8;
            output.push(((accumulator >> bits) & 0xff) as u8);
        }
    }

    Ok(output)
}
