impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|field| field == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|field| field == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;

    #[derive(Clone, Copy, PartialEq, Eq)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted,
        AfterQuote,
    }

    let mut state = State::FieldStart;
    let mut opening_quote = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != State::Quoted && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::FieldStart => {
                if b == delimiter {
                    record.push(String::new());
                    i += 1;
                } else if b == quote {
                    state = State::Quoted;
                    opening_quote = i;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    let begin = i;
                    while i < bytes.len()
                        && bytes[i] != delimiter
                        && bytes[i] != quote
                        && bytes[i] != b'\n'
                        && bytes[i] != b'\r'
                    {
                        if bytes[i] == 0 {
                            return Err(CsvError::Nul {
                                record: record_index,
                                byte: i,
                            });
                        }
                        i += 1;
                    }
                    field.push_str(&input[begin..i]);
                    state = State::Unquoted;
                }
            }
            State::Unquoted => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else {
                    let begin = i;
                    while i < bytes.len()
                        && bytes[i] != delimiter
                        && bytes[i] != quote
                        && bytes[i] != b'\n'
                        && bytes[i] != b'\r'
                    {
                        if bytes[i] == 0 {
                            return Err(CsvError::Nul {
                                record: record_index,
                                byte: i,
                            });
                        }
                        i += 1;
                    }
                    field.push_str(&input[begin..i]);
                }
            }
            State::Quoted => {
                if b == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    let begin = i;
                    while i < bytes.len() && bytes[i] != quote {
                        if bytes[i] == 0 {
                            return Err(CsvError::Nul {
                                record: record_index,
                                byte: i,
                            });
                        }
                        i += 1;
                    }
                    field.push_str(&input[begin..i]);
                }
            }
            State::AfterQuote => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if state == State::AfterQuote || state == State::Unquoted {
        record.push(field);
        records.push(record);
    } else if !record.is_empty() {
        record.push(String::new());
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers && !records.is_empty() {
        let mut iter = records.into_iter();
        let header = iter.next();
        (header, iter.collect::<Vec<_>>())
    } else {
        (None, records)
    };

    if let (Some(header), DuplicatePolicy::Reject) = (&header, config.duplicates) {
        let mut seen: Vec<(&str, usize)> = Vec::new();
        for (second, name) in header.iter().enumerate() {
            if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first: *first,
                    second,
                });
            }
            seen.push((name, second));
        }
    }

    if let Some(header) = &header {
        for (row_offset, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: row_offset + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

fn read_csv_validated(input: &str, config: Config) -> Result<Table, CsvError> {
    let records = parse_dialect(input, config.delimiter, config.quote)?;
    build_table(records, config)
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = read_csv_validated(&self.input, self.config);
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;
    read_csv_validated(input, config)
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let mut i = 0;
    if bytes[0] == b'-' {
        i = 1;
    }
    if i >= bytes.len() {
        return false;
    }
    if bytes[i] == b'0' {
        return i + 1 == bytes.len();
    }
    if !(b'1'..=b'9').contains(&bytes[i]) {
        return false;
    }
    bytes[i + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    let mut i = 0;
    if bytes.first() == Some(&b'-') {
        i = 1;
    }
    let integer_start = i;
    if i >= bytes.len() {
        return false;
    }
    if bytes[i] == b'0' {
        i += 1;
        if i < bytes.len() && bytes[i].is_ascii_digit() {
            return false;
        }
    } else if (b'1'..=b'9').contains(&bytes[i]) {
        i += 1;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
    } else {
        return false;
    }
    if i - integer_start == 0 || bytes.get(i) != Some(&b'.') {
        return false;
    }
    i += 1;
    bytes.len() == i + 2 && bytes[i].is_ascii_digit() && bytes[i + 1].is_ascii_digit()
        && !(bytes.first() == Some(&b'-')
            && bytes[1..i].iter().all(|b| *b == b'0')
            && bytes[i + 1..].iter().all(|b| *b == b'0'))
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let unsigned = if negative { &s[1..] } else { s };
    let dot = unsigned.find('.').unwrap();
    let whole: i128 = unsigned[..dot].parse().unwrap();
    let fraction: i128 = unsigned[dot + 1..].parse().unwrap();
    let magnitude = whole * 100 + fraction;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn evaluate_cell(
    row: &[String],
    table: &Table,
    spec: &ColumnSpec,
    row_index: usize,
    spec_index: usize,
) -> Result<Value, CellError> {
    let column = match &spec.column {
        ColumnRef::Index(index) => Some(*index),
        ColumnRef::Name(name) => table.column_index(name),
    };
    let Some(column) = column else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: None,
            kind: CellErrorKind::UnknownColumn,
        });
    };
    let Some(cell) = row.get(column) else {
        return Err(CellError {
            row: row_index,
            spec: spec_index,
            column: Some(column),
            kind: CellErrorKind::MissingCell,
        });
    };
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind: CellErrorKind::Empty,
            });
        }
    }

    let result = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.clone())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    };
    result.map_err(|kind| CellError {
        row: row_index,
        spec: spec_index,
        column: Some(column),
        kind,
    })
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut row_failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            match evaluate_cell(row, table, spec, row_index, spec_index) {
                Ok(value) => output[spec_index] = Some(value),
                Err(error) => {
                    result.errors.push(error);
                    row_failed = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }
        result.rows.push(output);
        if row_failed && policy == ErrorPolicy::Stop {
            break;
        }
    }
    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns {

        name: String,

fn duplicate_header(header: &[String]) -> Option<WriteError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
            return Some(WriteError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn row_nul(row: &[String], record: usize) -> Option<WriteError> {
    for (column, cell) in row.iter().enumerate() {
        if let Some(byte) = cell.as_bytes().iter().position(|b| *b == 0) {
            return Some(WriteError::Nul {
                record,
                column,
                byte,
            });
        }
    }
    None
}

fn field_needs_quote(
    field: &str,
    first_field_first_record: bool,
    sole_empty: bool,
    delimiter: u8,
    quote: u8,
) -> bool {
    let bytes = field.as_bytes();
    bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_first_record && field.starts_with('\u{feff}'))
        || (sole_empty && field.is_empty())
}

fn encode_field(
    output: &mut String,
    field: &str,
    first_field_first_record: bool,
    sole_empty: bool,
    delimiter: u8,
    quote: u8,
) {
    if !field_needs_quote(
        field,
        first_field_first_record,
        sole_empty,
        delimiter,
        quote,
    ) {
        output.push_str(field);
        return;
    }

    output.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            output.push(quote as char);
            output.push(quote as char);
        } else {
            output.push(ch);
        }
    }
    output.push(quote as char);
}

fn encode_records(
    records: &[Vec<String>],
    delimiter: u8,
    quote: u8,
    ending: &str,
    final_terminator: bool,
) -> String {
    let mut output = String::new();
    for (record_index, record) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(ending);
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                output.push(delimiter as char);
            }
            encode_field(
                &mut output,
                field,
                record_index == 0 && column_index == 0,
                record.len() == 1 && field.is_empty(),
                delimiter,
                quote,
            );
        }
    }
    if final_terminator && !records.is_empty() {
        output.push_str(ending);
    }
    output
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let order = left.as_bytes().cmp(right.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    if let Some(header) = &table.header {
        if let Some(error) = row_nul(header, 0) {
            return Err(error);
        }
    }
    let data_record_start = if table.header.is_some() { 1 } else { 0 };
    for (index, row) in table.rows.iter().enumerate() {
        if let Some(error) = row_nul(row, data_record_start + index) {
            return Err(error);
        }
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    if let Some(header) = &table.header {
        if header.is_empty() {
            return Err(WriteError::ZeroColumns { record: 0 });
        }
    }
    for (index, row) in table.rows.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: data_record_start + index,
            });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let (Some(header), DuplicatePolicy::Reject) = (&table.header, table.duplicates) {
            if let Some(WriteError::DuplicateHeader {
                name,
                first,
                second,
            }) = duplicate_header(header)
            {
                return Err(WriteError::DuplicateHeader {
                    name,
                    first,
                    second,
                });
            }
        }
    }

    let mut records = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                records.push(header.clone());
            }
            records.extend(table.rows.iter().cloned());
            let ending = match config.line_ending {
                LineEnding::Lf => "\n",
                LineEnding::CrLf => "\r\n",
            };
            Ok(encode_records(
                &records,
                config.delimiter,
                config.quote,
                ending,
                config.final_terminator,
            ))
        }
        WriteMode::Canonical => {
            let rows = if let Some(header) = &table.header {
                let mut retained = Vec::<(usize, &String)>::new();
                for (index, name) in header.iter().enumerate() {
                    let existing = retained.iter().position(|(_, prior)| *prior == name);
                    match (existing, table.duplicates) {
                        (None, _) => retained.push((index, name)),
                        (Some(position), DuplicatePolicy::KeepLast) => {
                            retained[position] = (index, name);
                        }
                        (Some(_), DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject) => {}
                    }
                }
                retained.sort_by(|(_, a), (_, b)| a.as_bytes().cmp(b.as_bytes()));
                let indexes: Vec<usize> = retained.iter().map(|(index, _)| *index).collect();
                let sorted_header: Vec<String> =
                    retained.iter().map(|(_, name)| (*name).clone()).collect();
                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| indexes.iter().map(|index| row[*index].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| compare_rows(a, b));
                records.push(sorted_header);
                records.extend(projected);
                records
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_rows(a, b));
                rows
            };
            Ok(encode_records(
                &rows,
                config.delimiter,
                config.quote,
                "\n",
                true,
            ))
        }
    }
}
