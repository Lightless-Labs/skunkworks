```rust
use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let other_half = denominator - r;
    Some(if r > other_half || (r == other_half && q % 2 == 1) {
        q + 1
    } else {
        q
    })
}

#[derive(Clone)]
struct State {
    customer: Customer,
    remaining: u64,
    fresh: bool,
    arrival: u64,
    entry: u64,
    deadline: Option<u64>,
    pause_time: u64,
    waited: u128,
}

struct Active {
    index: usize,
    dispatch: u64,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &State, b: &State) -> Ordering {
    a.customer.priority.cmp(&b.customer.priority).then_with(|| {
        match (a.fresh, b.fresh) {
            (false, true) => Ordering::Less,
            (true, false) => Ordering::Greater,
            (false, false) => a
                .pause_time
                .cmp(&b.pause_time)
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (true, true) => a
                .arrival
                .cmp(&b.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        }
    })
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn overflow(customer: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer, time, stage }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut seen: Vec<u32> = Vec::new();
    let mut deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        seen.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        let d = match c.patience {
            Some(p) => Some(
                c.arrival
                    .checked_add(p)
                    .ok_or(Error::DeadlineOverflow { index, id: c.id })?,
            ),
            None => None,
        };
        deadlines.push(d);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<Option<State>> = (0..customers.len()).map(|_| None).collect();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut events = Vec::new();
    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();
    let mut last_event_time = 0u64;

    loop {
        // Phase A: complete the running service before any other work.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let c = &customers[a.index];
            emit(&mut events, now, c.id, Kind::Finish);
            last_event_time = now;
            completed += 1;
            turnaround_ticks += (now - c.arrival) as u128;
            completion_order.push(c.id);
        }

        // Phase B: expire those already waiting at the beginning of this tick.
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| states[i].as_ref().unwrap().deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            states[a]
                .as_ref()
                .unwrap()
                .deadline
                .cmp(&states[b].as_ref().unwrap().deadline)
                .then_with(|| customers[a].id.cmp(&customers[b].id))
        });
        for i in expired {
            waiting.retain(|&x| x != i);
            let s = states[i].as_mut().unwrap();
            s.waited += (now - s.entry) as u128;
            wait_ticks += (now - s.entry) as u128;
            emit(
                &mut events,
                now,
                s.customer.id,
                Kind::Renege {
                    waited: s.waited as u64,
                },
            );
            last_event_time = now;
            reneged += 1;
            states[i] = None;
        }

        // Phase C: process the full arrival cohort without dispatching.
        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let i = order[next_arrival];
            next_arrival += 1;
            let c = &customers[i];
            emit(&mut events, now, c.id, Kind::Arrive);
            last_event_time = now;
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                emit(&mut events, now, c.id, Kind::Balk);
                last_event_time = now;
                balked += 1;
            } else {
                accepted += 1;
                states[i] = Some(State {
                    customer: c.clone(),
                    remaining: c.service,
                    fresh: true,
                    arrival: c.arrival,
                    entry: now,
                    deadline: deadlines[i],
                    pause_time: 0,
                    waited: 0,
                });
                waiting.push(i);
            }
        }

        // Phase D: preempt only after switching protection has ended.
        let can_preempt = active.as_ref().is_some_and(|a| now >= a.work_at);
        if can_preempt && !waiting.is_empty() {
            let best = *waiting
                .iter()
                .min_by(|&&a, &&b| waiter_cmp(states[a].as_ref().unwrap(), states[b].as_ref().unwrap()))
                .unwrap();
            let active_index = active.as_ref().unwrap().index;
            if customers[best].priority < customers[active_index].priority {
                let a = active.take().unwrap();
                let remaining = a.finish - now;
                emit(
                    &mut events,
                    now,
                    customers[a.index].id,
                    Kind::Preempt {
                        remaining,
                        by: customers[best].id,
                    },
                );
                last_event_time = now;
                preemptions += 1;
                busy_ticks += (now - a.dispatch) as u128;
                let s = states[a.index].as_mut().unwrap();
                s.remaining = remaining;
                s.fresh = false;
                s.entry = now;
                s.pause_time = now;
                if let Some(budget) = s.customer.patience {
                    s.deadline = Some(now.checked_add(budget).ok_or_else(|| {
                        overflow(s.customer.id, now, OverflowStage::Patience)
                    })?);
                } else {
                    s.deadline = None;
                }
                waiting.push(a.index);
            }
        }

        // Phase E: dispatch until an occupied interval remains, including inline finishes.
        loop {
            if active.is_some() || waiting.is_empty() {
                break;
            }
            let pos = (0..waiting.len())
                .min_by(|&x, &y| {
                    waiter_cmp(
                        states[waiting[x]].as_ref().unwrap(),
                        states[waiting[y]].as_ref().unwrap(),
                    )
                })
                .unwrap();
            let i = waiting.remove(pos);
            let s = states[i].as_mut().unwrap();
            let segment = now - s.entry;
            s.waited += segment as u128;
            wait_ticks += segment as u128;
            let remaining = s.remaining;
            let is_start = s.fresh;
            s.fresh = false;
            let id = s.customer.id;
            let cost = if last_dispatched.is_some_and(|previous| previous != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now
                .checked_add(cost)
                .ok_or_else(|| overflow(id, now, OverflowStage::WorkAt))?;
            let finish = work_at
                .checked_add(remaining)
                .ok_or_else(|| overflow(id, now, OverflowStage::Finish))?;
            emit(
                &mut events,
                now,
                id,
                if is_start {
                    started += 1;
                    Kind::Start {
                        remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Resume {
                        remaining,
                        work_at,
                        finish,
                    }
                },
            );
            last_event_time = now;
            last_dispatched = Some(id);
            active = Some(Active {
                index: i,
                dispatch: now,
                work_at,
                finish,
            });
            if finish == now {
                let a = active.take().unwrap();
                emit(&mut events, now, id, Kind::Finish);
                last_event_time = now;
                completed += 1;
                turnaround_ticks += (now - customers[a.index].arrival) as u128;
                completion_order.push(id);
                busy_ticks += (now - a.dispatch) as u128;
            }
        }

        // Phase F: expire waiters queued in this tick (including zero budgets).
        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| states[i].as_ref().unwrap().deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            states[a]
                .as_ref()
                .unwrap()
                .deadline
                .cmp(&states[b].as_ref().unwrap().deadline)
                .then_with(|| customers[a].id.cmp(&customers[b].id))
        });
        for i in expired {
            waiting.retain(|&x| x != i);
            let s = states[i].as_mut().unwrap();
            s.waited += (now - s.entry) as u128;
            wait_ticks += (now - s.entry) as u128;
            emit(
                &mut events,
                now,
                s.customer.id,
                Kind::Renege {
                    waited: s.waited as u64,
                },
            );
            last_event_time = now;
            reneged += 1;
            states[i] = None;
        }

        if next_arrival == order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next = None;
        let mut consider = |candidate: u64| {
            if candidate > now && next.map_or(true, |n| candidate < n) {
                next = Some(candidate);
            }
        };
        if next_arrival < order.len() {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = active.as_ref() {
            consider(a.finish);
            if a.work_at > now {
                consider(a.work_at);
            }
        }
        for &i in &waiting {
            if let Some(d) = states[i].as_ref().unwrap().deadline {
                consider(d);
            }
        }
        match next {
            Some(t) => now = t,
            None => break,
        }
    }

    let elapsed = if events.is_empty() { 0 } else { last_event_time };
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}
```
