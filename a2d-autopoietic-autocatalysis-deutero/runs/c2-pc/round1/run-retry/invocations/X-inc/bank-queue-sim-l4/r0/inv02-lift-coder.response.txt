#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(if round_up { q + 1 } else { q })
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    queue_entry: u64,
    waited: u64,
    remaining: u64,
    pause_time: Option<u64>,
    ever_started: bool,
}

#[derive(Clone)]
struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
    dispatch_time: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn waiting_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    a.customer.priority.cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(x), Some(y)) => x.cmp(&y).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => std::cmp::Ordering::Less,
            (None, Some(_)) => std::cmp::Ordering::Greater,
            (None, None) => a.customer.arrival.cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn checked_add(a: u64, b: u64) -> Option<u64> {
    a.checked_add(b)
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::new();
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index, id: c.id,
            })?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| customers[a].arrival.cmp(&customers[b].arrival)
        .then_with(|| customers[a].priority.cmp(&customers[b].priority))
        .then_with(|| customers[b].id.cmp(&customers[a].id)));

    let mut jobs: Vec<Option<Job>> = vec![None; customers.len()];
    let mut waiting: Vec<usize> = Vec::new();
    let mut running: Option<Running> = None;
    let mut cursor = 0usize;
    let mut now = 0u64;
    let mut have_now = false;
    let mut last_dispatch: Option<u32> = None;
    let mut events = Vec::new();

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();

    loop {
        if !have_now {
            if cursor >= order.len() {
                break;
            }
            now = customers[order[cursor]].arrival;
            have_now = true;
        }

        // Phase A: a completion releases the server before all other work.
        if running.as_ref().is_some_and(|r| r.finish == now) {
            let r = running.take().unwrap();
            busy_ticks += (now - r.dispatch_time) as u128;
            emit(&mut events, now, r.job.customer.id, Kind::Finish);
            completed += 1;
            turnaround_ticks += (now - r.job.customer.arrival) as u128;
            completion_order.push(r.job.customer.id);
        }

        // Phase B: expire customers that were waiting at the start of this tick.
        let mut expired: Vec<usize> = waiting.iter().copied()
            .filter(|&i| jobs[i].as_ref().unwrap().deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            jobs[a].as_ref().unwrap().deadline.cmp(&jobs[b].as_ref().unwrap().deadline)
                .then_with(|| jobs[a].as_ref().unwrap().customer.id.cmp(&jobs[b].as_ref().unwrap().customer.id))
        });
        for i in expired {
            waiting.retain(|&x| x != i);
            let j = jobs[i].as_mut().unwrap();
            let segment = now - j.queue_entry;
            j.waited += segment;
            wait_ticks += segment as u128;
            emit(&mut events, now, j.customer.id, Kind::Renege { waited: j.waited });
            reneged += 1;
        }

        // Phase C: process the complete cohort without dispatching mid-cohort.
        while cursor < order.len() && customers[order[cursor]].arrival == now {
            let i = order[cursor];
            cursor += 1;
            let c = &customers[i];
            emit(&mut events, now, c.id, Kind::Arrive);
            let server_idle = running.is_none();
            let limit = config.waiting_capacity + usize::from(server_idle);
            if waiting.len() >= limit {
                emit(&mut events, now, c.id, Kind::Balk);
                balked += 1;
            } else {
                let job = Job {
                    customer: c.clone(),
                    deadline: initial_deadlines[i],
                    queue_entry: now,
                    waited: 0,
                    remaining: c.service,
                    pause_time: None,
                    ever_started: false,
                };
                jobs[i] = Some(job);
                waiting.push(i);
                accepted += 1;
            }
        }

        // Phase D: compare against the best waiter before re-enqueuing the paused job.
        if let Some(r) = running.as_ref() {
            if now >= r.work_at {
                if let Some(&best) = waiting.iter().min_by(|&&a, &&b| {
                    waiting_cmp(jobs[a].as_ref().unwrap(), jobs[b].as_ref().unwrap())
                }) {
                    let waiter = jobs[best].as_ref().unwrap();
                    if waiter.customer.priority < r.job.customer.priority {
                        let by = waiter.customer.id;
                        let remaining = r.finish - now;
                        let old = running.take().unwrap();
                        busy_ticks += (now - old.dispatch_time) as u128;
                        emit(&mut events, now, old.job.customer.id, Kind::Preempt { remaining, by });
                        preemptions += 1;
                        let mut paused = old.job;
                        paused.remaining = remaining;
                        paused.queue_entry = now;
                        paused.pause_time = Some(now);
                        if let Some(budget) = paused.customer.patience {
                            paused.deadline = Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                                customer: paused.customer.id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })?);
                        }
                        let idx = customers.iter().position(|c| c.id == paused.customer.id).unwrap();
                        jobs[idx] = Some(paused);
                        waiting.push(idx);
                    }
                }
            }
        }

        // Phase E: dispatch until occupied, allowing inline zero-duration completions.
        loop {
            if running.is_some() || waiting.is_empty() {
                break;
            }
            let best_pos = waiting.iter().enumerate().min_by(|(_, &a), (_, &b)| {
                waiting_cmp(jobs[a].as_ref().unwrap(), jobs[b].as_ref().unwrap())
            }).map(|(p, _)| p).unwrap();
            let i = waiting.remove(best_pos);
            let mut job = jobs[i].take().unwrap();
            let segment = now - job.queue_entry;
            job.waited += segment;
            wait_ticks += segment as u128;

            let switch = if last_dispatch.is_some_and(|id| id != job.customer.id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = checked_add(now, switch).ok_or(Error::TimeOverflow {
                customer: job.customer.id, time: now, stage: OverflowStage::WorkAt,
            })?;
            let finish = checked_add(work_at, job.remaining).ok_or(Error::TimeOverflow {
                customer: job.customer.id, time: now, stage: OverflowStage::Finish,
            })?;
            let kind = if job.ever_started {
                Kind::Resume { remaining: job.remaining, work_at, finish }
            } else {
                started += 1;
                job.ever_started = true;
                Kind::Start { remaining: job.remaining, work_at, finish }
            };
            emit(&mut events, now, job.customer.id, kind);
            last_dispatch = Some(job.customer.id);
            running = Some(Running { job, work_at, finish, dispatch_time: now });

            if finish == now {
                let r = running.take().unwrap();
                emit(&mut events, now, r.job.customer.id, Kind::Finish);
                completed += 1;
                turnaround_ticks += (now - r.job.customer.arrival) as u128;
                completion_order.push(r.job.customer.id);
            }
        }

        // Phase F: expire any waiters created during this tick.
        let mut expired: Vec<usize> = waiting.iter().copied()
            .filter(|&i| jobs[i].as_ref().unwrap().deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            jobs[a].as_ref().unwrap().deadline.cmp(&jobs[b].as_ref().unwrap().deadline)
                .then_with(|| jobs[a].as_ref().unwrap().customer.id.cmp(&jobs[b].as_ref().unwrap().customer.id))
        });
        for i in expired {
            waiting.retain(|&x| x != i);
            let j = jobs[i].as_mut().unwrap();
            let segment = now - j.queue_entry;
            j.waited += segment;
            wait_ticks += segment as u128;
            emit(&mut events, now, j.customer.id, Kind::Renege { waited: j.waited });
            reneged += 1;
        }

        if cursor >= order.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut next: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now && next.is_none_or(|n| t < n) {
                next = Some(t);
            }
        };
        if cursor < order.len() {
            consider(customers[order[cursor]].arrival);
        }
        if let Some(r) = &running {
            consider(r.finish);
            if r.work_at > now {
                consider(r.work_at);
            }
        }
        for &i in &waiting {
            if let Some(d) = jobs[i].as_ref().unwrap().deadline {
                consider(d);
            }
        }
        match next {
            Some(t) => now = t,
            None => {
                // All remaining work must be actionable at the current time.
                have_now = false;
            }
        }
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted, balked, reneged, completed, started, preemptions, elapsed,
            busy_ticks, wait_ticks, turnaround_ticks, mean_wait_milli,
            mean_turnaround_milli, utilization_bps, completion_order,
        },
    })
}
