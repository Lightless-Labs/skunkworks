```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
    pub maintenance: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
    MaintenanceActive,
    MaintenanceRequired,
    AlreadyMode,
    ReconcileMismatch { coin: u16, expected: u16, found: u16 },
    ValueDrift { before: u64, after: u64 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
    Maintenance { enabled: bool },
    Reconciled { before: [u16; 4], after: [u16; 4] },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Audit {
    pub inserted: u64,
    pub refunded: u64,
    pub sales: u64,
    pub change: u64,
    pub adjustment: i64,
    pub cash: u64,
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    inserted: u64,
    refunded: u64,
    sales: u64,
    change: u64,
    adjustment: i64,
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMS.iter().position(|&d| d == coin)
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn tube_value(tubes: &[u16; 4]) -> u64 {
    tubes.iter().zip(DENOMS).map(|(&count, denom)| u64::from(count) * u64::from(denom)).sum()
}

fn saturating_add_signed(a: i64, b: i128) -> i64 {
    let result = i128::from(a) + b;
    if result > i128::from(i64::MAX) {
        i64::MAX
    } else if result < i128::from(i64::MIN) {
        i64::MIN
    } else {
        result as i64
    }
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashMap::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId { id: product.id.clone() });
            }
            if seen.insert(product.id.clone(), ()).is_some() {
                return Err(Error::DuplicateId { id: product.id.clone() });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id.clone() });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit { id: product.id.clone() });
            }
        }

        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }

    fn product_index(&self, id: &str) -> Option<usize> {
        self.products.iter().position(|p| p.id == id)
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if denomination_index(coin).is_none() {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        let credit = self.credit + coin;
        self.credit = credit;
        self.escrow.push(coin);
        self.inserted = self.inserted.saturating_add(u64::from(coin));
        self.log(EventKind::Inserted { coin, credit });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        let selected_id = self.products[index].id.clone();
        self.selected = Some(selected_id.clone());
        self.log(EventKind::Selected { id: selected_id });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let index = self.product_index(&id).expect("selected product must exist");
        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid { price, credit: self.credit });
        }

        let amount = self.credit - price;
        let change = if amount == 0 {
            Vec::new()
        } else {
            let mut remaining = amount;
            let mut counts = [0u16; 4];
            for i in (0..4).rev() {
                let denom = DENOMS[i];
                counts[i] = self.tubes[i].min(remaining / denom);
                remaining -= counts[i] * denom;
            }

            if remaining != 0 {
                let mut best: Option<[u16; 4]> = None;
                fn search(
                    i: usize,
                    remaining: u16,
                    tubes: &[u16; 4],
                    current: &mut [u16; 4],
                    best: &mut Option<[u16; 4]>,
                ) {
                    if i == 4 {
                        if remaining == 0 {
                            let better = match best {
                                None => true,
                                Some(previous) => {
                                    let n: u32 = current.iter().map(|&v| u32::from(v)).sum();
                                    let old_n: u32 = previous.iter().map(|&v| u32::from(v)).sum();
                                    n < old_n || (n == old_n && {
                                        for k in (0..4).rev() {
                                            if current[k] != previous[k] {
                                                return current[k] > previous[k];
                                            }
                                        }
                                        false
                                    })
                                }
                            };
                            if better {
                                *best = Some(*current);
                            }
                        }
                        return;
                    }
                    let denom = DENOMS[i];
                    let max = tubes[i].min(remaining / denom);
                    for count in 0..=max {
                        current[i] = count;
                        search(i + 1, remaining - count * denom, tubes, current, best);
                    }
                    current[i] = 0;
                }
                search(0, amount, &self.tubes, &mut [0; 4], &mut best);
                let Some(chosen) = best else {
                    return Err(Error::NoChange { amount });
                };
                counts = chosen;
            }

            let mut coins = Vec::new();
            for i in (0..4).rev() {
                for _ in 0..counts[i] {
                    coins.push(DENOMS[i]);
                }
            }
            coins
        };

        let mut proposed = self.tubes;
        for &coin in &change {
            let i = denomination_index(coin).expect("change denomination is valid");
            proposed[i] -= 1;
        }
        for &coin in &self.escrow {
            let i = denomination_index(coin).expect("escrow denomination is valid");
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        self.tubes = proposed;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        let change_value: u64 = change.iter().map(|&c| u64::from(c)).sum();
        self.change = self.change.saturating_add(change_value);
        self.log(EventKind::Vended { id: id.clone(), price, change: change.clone() });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let refunded = u64::from(self.credit);
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.refunded = self.refunded.saturating_add(refunded);
        self.log(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.cmp(&b.id));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged { id: id.to_owned(), old, new: price });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let stock = u32::from(self.products[index].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[index].stock = stock as u16;
        self.log(EventKind::Restocked { id: id.to_owned(), added: count, stock: stock as u16 });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value(&self.tubes) + u64::from(self.credit),
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let before_value = tube_value(&self.tubes);
        let after_value = tube_value(&actual);
        let diff = before_value.abs_diff(after_value);
        if diff > 20 {
            return Err(Error::ValueDrift { before: before_value, after: after_value });
        }

        let before = self.tubes;
        self.tubes = actual;
        self.adjustment = saturating_add_signed(
            self.adjustment,
            i128::from(after_value) - i128::from(before_value),
        );
        self.log(EventKind::Reconciled { before, after: actual });
        Ok(())
    }
}
```
