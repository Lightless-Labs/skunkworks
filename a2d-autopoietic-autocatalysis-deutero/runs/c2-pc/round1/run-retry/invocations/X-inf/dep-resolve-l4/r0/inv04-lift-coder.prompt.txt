fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),

fn constraint_key(c: Constraint) -> (u8, Version) {

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn percent_encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

    }
}

fn sorted_features(features: &[String]) -> Vec<String> {
    let mut result = features.to_vec();
    result.sort();
    result.dedup();
    result
}

fn token_root(i: usize, req: &Requirement) -> String {
    let features = sorted_features(&req.features).join(",");
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        req.package,
        constraint_text(req.constraint),
        features
    )
}

fn target_text(platform: Platform) -> &'static str {
    match platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn dependency_active_platform(platform: Platform, target: Target) -> bool {
    match platform {
        Platform::Any => true,
        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

impl Active {
    fn constraints(&self, package: &str) -> Vec<Constraint> {
        let cs = self
            .origins
            .get(package)
            .into_iter()
            .flatten()
            .map(|o| o.constraint)
            .collect::<Vec<_>>();
        canonical_constraints(&cs)
    }

    fn evidence(&self, package: &str) -> Vec<Evidence> {
        self.origins
            .get(package)
            .into_iter()
            .flatten()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect()
    }
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn put_path(paths: &mut BTreeMap<String, Vec<String>>, package: &str, path: Vec<String>) -> bool {
    match paths.get(package) {
        None => {
            paths.insert(package.to_string(), path);
            true
        }
        Some(old) if path_better(&path, old) => {
            paths.insert(package.to_string(), path);
            true
        }
        _ => false,
    }
}

fn compute_active(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let mut active = Active::default();
    let mut queue: Vec<(String, String)> = Vec::new();

    for (i, root) in roots.iter().enumerate() {
        let path = vec![token_root(i, root)];
        active
            .origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: path.clone(),
            });
        put_path(&mut active.paths, &root.package, path);
        for f in &root.features {
            if active
                .features
                .entry(root.package.clone())
                .or_default()
                .insert(f.clone())
            {
                queue.push((root.package.clone(), f.clone()));
            }
        }
    }

    let mut processed_features = BTreeSet::new();
    let mut processed_edges = BTreeSet::new();

    loop {
        let mut changed = false;

        let selected_names: Vec<String> = selected.keys().cloned().collect();
        for source in selected_names {
            let Some(source_version) = selected.get(&source).copied() else {
                continue;
            };
            let Some(pkg) = catalog.get(&source) else {
                continue;
            };
            let Some(release) = pkg.releases.iter().find(|r| r.version == source_version) else {
                continue;
            };

            let current_features = active
                .features
                .get(&source)
                .cloned()
                .unwrap_or_default();

            for feature_name in current_features.iter() {
                let key = (source.clone(), feature_name.clone());
                if !processed_features.insert(key) {
                    continue;
                }
                if let Some(feature) = release.features.iter().find(|f| f.name == *feature_name) {
                    for enabled in &feature.enables {
                        if active
                            .features
                            .entry(source.clone())
                            .or_default()
                            .insert(enabled.clone())
                        {
                            changed = true;
                        }
                    }
                }
            }

            for (j, dep) in release.deps.iter().enumerate() {
                if !dependency_active_platform(dep.platform, target) {
                    continue;
                }
                if dep
                    .gate
                    .as_ref()
                    .is_some_and(|gate| !current_features.contains(gate))
                {
                    continue;
                }
                if !processed_edges.insert((source.clone(), j)) {
                    continue;
                }

                let source_path = active.paths.get(&source).cloned().unwrap_or_default();
                let gate = dep.gate.as_deref().unwrap_or("*");
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    source,
                    version_text(source_version),
                    j,
                    dep.package,
                    constraint_text(dep.constraint),
                    sorted_features(&dep.features).join(","),
                    gate,
                    target_text(dep.platform)
                );
                let mut path = source_path;
                path.push(token);
                active
                    .origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                if put_path(&mut active.paths, &dep.package, path) {
                    changed = true;
                }
                for f in &dep.features {
                    if active
                        .features
                        .entry(dep.package.clone())
                        .or_default()
                        .insert(f.clone())
                    {
                        changed = true;
                    }
                }
                changed = true;
            }
        }

        // Feature requests can arrive before or after a package is selected.
        for (package, features) in active.features.clone() {
            if selected.contains_key(&package) {
                for feature in features {
                    if !processed_features.contains(&(package.clone(), feature.clone())) {
                        changed = true;
                    }
                }
            }
        }

        if !changed {
            break;
        }
    }

    // Rebuild dependency evidence paths using the final canonical source paths.
    // Each active dependency contributes exactly one origin.
    let mut rebuilt = Active::default();
    rebuilt.features = active.features.clone();
    rebuilt.paths = active.paths.clone();
    for (i, root) in roots.iter().enumerate() {
        rebuilt
            .origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![token_root(i, root)],
            });
    }
    for (source, version) in selected {
        let Some(pkg) = catalog.get(source) else {
            continue;
        };
        let Some(release) = pkg.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        let source_path = rebuilt
            .paths
            .get(source)
            .or_else(|| active.paths.get(source))
            .cloned()
            .unwrap_or_default();
        let source_features = active.features.get(source).cloned().unwrap_or_default();
        for (j, dep) in release.deps.iter().enumerate() {
            if !dependency_active_platform(dep.platform, target)
                || dep
                    .gate
                    .as_ref()
                    .is_some_and(|gate| !source_features.contains(gate))
            {
                continue;
            }
            let gate = dep.gate.as_deref().unwrap_or("*");
            let token = format!(
                "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                source,
                version_text(*version),
                j,
                dep.package,
                constraint_text(dep.constraint),
                sorted_features(&dep.features).join(","),
                gate,
                target_text(dep.platform)
            );
            let mut path = source_path.clone();
            path.push(token);
            rebuilt
                .origins
                .entry(dep.package.clone())
                .or_default()
                .push(Origin {
                    constraint: dep.constraint,
                    path,
                });
        }
    }
    rebuilt
}

fn validate_inputs(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut names = BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }
    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    for package in catalog {
        for release in &package.releases {
            let mut feature_names = BTreeSet::new();
            for feature in &release.features {
                if !feature_names.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    budget: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
    limit: bool,
}

impl<'a> Search<'a> {
    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn inspect(&mut self, selected: &mut BTreeMap<String, Version>) {
        if self.limit {
            return;
        }
        let active = compute_active(self.catalog, self.roots, selected, self.target);

        let missing = active
            .origins
            .keys()
            .filter(|p| !self.catalog.contains_key(*p))
            .next()
            .cloned();
        if let Some(package) = missing {
            self.record_error(ResolveError::Missing {
                evidence: active.evidence(&package),
                package,
            });
            return;
        }

        for (package, version) in selected.iter() {
            let constraints = active.constraints(package);
            if !constraints.iter().all(|c| matches(*version, *c)) {
                self.record_error(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: constraints,
                    evidence: active.evidence(package),
                });
                return;
            }
        }

        for (package, features) in &active.features {
            let Some(version) = selected.get(package) else {
                continue;
            };
            let release = self.catalog[package]
                .releases
                .iter()
                .find(|r| r.version == *version)
                .unwrap();
            for feature in features {
                if !release.features.iter().any(|f| f.name == *feature) {
                    self.record_error(ResolveError::MissingFeature {
                        package: package.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: active.paths.get(package).cloned().unwrap_or_default(),
                    });
                    return;
                }
            }
        }

        for package in active.origins.keys() {
            if selected.contains_key(package) {
                continue;
            }
            let constraints = active.constraints(package);
            let candidates = self.catalog[package]
                .releases
                .iter()
                .filter(|r| constraints.iter().all(|c| matches(r.version, *c)))
                .count();
            if candidates == 0 {
                self.record_error(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints,
                    evidence: active.evidence(package),
                });
                return;
            }
        }

        let next = active
            .origins
            .keys()
            .find(|p| !selected.contains_key(*p))
            .cloned();
        let Some(package) = next else {
            self.consider(selected);
            return;
        };

        let constraints = active.constraints(&package);
        let mut candidates: Vec<Version> = self.catalog[&package]
            .releases
            .iter()
            .filter(|r| constraints.iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(&package) {
            if let Some(index) = candidates.iter().position(|v| *v == *lock) {
                let v = candidates.remove(index);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.budget {
                self.limit = true;
                return;
            }
            self.attempts += 1;
            selected.insert(package.clone(), version);
            self.inspect(selected);
            selected.remove(&package);
            if self.limit {
                return;
            }
        }
    }

    fn consider(&mut self, selected: &BTreeMap<String, Version>) {
        let cost = self
            .locks
            .iter()
            .filter(|(p, v)| selected.get(*p) != Some(*v))
            .count();
        let better = match &self.best {
            None => true,
            Some((best_cost, best)) if cost < *best_cost => true,
            Some((best_cost, best)) if cost == *best_cost => {
                let names: BTreeSet<String> = self
                    .catalog
                    .keys()
                    .cloned()
                    .chain(selected.keys().cloned())
                    .chain(best.keys().cloned())
                    .collect();
                for name in names {
                    match (selected.get(&name), best.get(&name)) {
                        (Some(a), Some(b)) if a != b => {
                            if a > b {
                                break true;
                            } else {
                                break false;
                            }
                        }
                        (Some(_), None) => break true,
                        (None, Some(_)) => break false,
                        _ => {}
                    }
                }
                false
            }
            _ => false,
        };
        if better {
            self.best = Some((cost, selected.clone()));
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_inputs(catalog, roots, locks)?;

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks: lock_map.clone(),
        target,
        budget: max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        limit: false,
    };
    search.inspect(&mut BTreeMap::new());

    if search.limit {
        return Err(ResolveError::Limit {
            attempts: search.budget,
        });
    }

    let Some((_, selected_map)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_active = compute_active(&catalog_map, roots, &selected_map, target);
    let selected = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let names: BTreeSet<String> = lock_map
        .keys()
        .cloned()
        .chain(selected_map.keys().cloned())
        .collect();
    let mut changes = Vec::new();
    for package in names {
        let before = lock_map.get(&package).copied();
        let after = selected_map.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_active
                    .constraints(&package)
                    .iter()
                    .all(|c| matches(old, *c))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
