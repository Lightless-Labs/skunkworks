#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|v| v == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut field_started = false;

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::Start;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if !matches!(state, State::Quoted { .. }) && b == b'\r'
            && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n')
        {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::Start => {
                if b == delimiter {
                    record.push(String::new());
                    i += 1;
                } else if b == quote {
                    state = State::Quoted { opening: i };
                    field_started = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    field.clear();
                    field_started = false;
                    i += 1;
                } else if b == b'\r' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    field.clear();
                    field_started = false;
                    i += 2;
                } else {
                    field.push(b as char);
                    state = State::Unquoted;
                    field_started = true;
                    i += 1;
                }
            }
            State::Unquoted => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::Start;
                    field_started = false;
                    i += 2;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else {
                    field.push(b as char);
                    i += 1;
                }
            }
            State::Quoted { opening } => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    field.push(b as char);
                    let _ = opening;
                    i += 1;
                }
            }
            State::AfterQuote => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::Start;
                    field_started = false;
                    i += 2;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::Start if !field_started && record.is_empty() => {}
        _ => {
            record.push(field);
            records.push(record);
        }
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        if let Some(first) = header[..second].iter().position(|v| v == &header[second]) {
            return Some(CsvError::DuplicateHeader {
                name: header[second].clone(),
                first,
                second,
            });
        }
    }
    None
}

fn table_from_input(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;

    let records = parse_with_dialect(input, config.delimiter, config.quote)?;
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect())
        }
    } else {
        (None, records)
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(h) {
                return Err(error);
            }
        }
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(ref result) = self.cached {
            return result.clone();
        }
        let result = table_from_input(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    pub errors: Vec<CellError>,

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if start >= b.len() {
        return false;
    }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    let whole = &b[start..dot];
    let frac = &b[dot + 1..];
    if frac.len() != 2 || !frac.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if whole.is_empty() {
        return false;
    }
    if whole == b"0" {
        return !s.starts_with("-0.") || frac != b"00";
    }
    (b'1'..=b'9').contains(&whole[0]) && whole[1..].iter().all(u8::is_ascii_digit)
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let unsigned = if negative { &s[1..] } else { s };
    let dot = unsigned.find('.').unwrap();
    let whole = unsigned[..dot].parse::<i128>().unwrap();
    let frac = unsigned[dot + 1..].parse::<i128>().unwrap();
    let magnitude = whole * 100 + frac;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed {
                output.push(None);
                continue;
            }

            let column_index = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let error_or_value = match column_index {
                None => Err(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                }),
                Some(index) => match row.get(index) {
                    None => Err(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(index),
                        kind: CellErrorKind::MissingCell,
                    }),
                    Some(cell) if cell.is_empty() && spec.nullable => Ok(Value::Null),
                    Some(cell) if cell.is_empty() && spec.kind != ColumnType::Text => {
                        Err(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(index),
                            kind: CellErrorKind::Empty,
                        })
                    }
                    Some(cell) => match spec.kind {
                        ColumnType::Text => Ok(Value::Text(cell.clone())),
                        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
                        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
                        ColumnType::Boolean => match cell.as_str() {
                            "yes" | "1" => Ok(Value::Boolean(true)),
                            "no" | "0" => Ok(Value::Boolean(false)),
                            _ => Err(CellErrorKind::InvalidBoolean),
                        },
                    }
                    .map_err(|kind| CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(index),
                        kind,
                    }),
                },
            };

            match error_or_value {
                Ok(value) => output.push(Some(value)),
                Err(error) => {
                    result.errors.push(error);
                    output.push(None);
                    if policy != ErrorPolicy::Collect {
                        row_failed = true;
                    }
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(output);
                        return result;
                    }
                }
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::FirstPerRow {
            continue;
        }
    }

    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns {
        record: usize,
    },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn first_nul(table: &Table) -> Option<WriteError> {
    if let Some(header) = &table.header {
        for (column, field) in header.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Some(WriteError::Nul {
                    record: 0,
                    column,
                    byte,
                });
            }
        }
    }
    let offset = usize::from(table.header.is_some());
    for (row_index, row) in table.rows.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Some(WriteError::Nul {
                    record: row_index + offset,
                    column,
                    byte,
                });
            }
        }
    }
    None
}

fn encode_field(field: &str, delimiter: u8, quote: u8, first: bool, sole: bool) -> String {
    let bytes = field.as_bytes();
    let boundary_space = bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
        || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t');
    let starts_bom = first && field.starts_with('\u{feff}');
    let must_quote = field.as_bytes().contains(&delimiter)
        || field.as_bytes().contains(&quote)
        || field.contains('\r')
        || field.contains('\n')
        || boundary_space
        || starts_bom
        || (sole && field.is_empty());

    if !must_quote {
        return field.to_string();
    }

    let mut out = String::with_capacity(field.len() + 2);
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
    out
}

fn encode_records(
    records: &[Vec<String>],
    delimiter: u8,
    quote: u8,
    ending: &str,
    final_terminator: bool,
) -> String {
    if records.is_empty() {
        return String::new();
    }
    let mut out = String::new();
    for (ri, record) in records.iter().enumerate() {
        if ri > 0 {
            out.push_str(ending);
        }
        for (ci, field) in record.iter().enumerate() {
            if ci > 0 {
                out.push(delimiter as char);
            }
            let encoded = encode_field(field, delimiter, quote, ri == 0 && ci == 0, record.len() == 1);
            out.push_str(&encoded);
        }
    }
    if final_terminator {
        out.push_str(ending);
    }
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    if let Some(error) = first_nul(table) {
        return Err(error);
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    let mut original_records = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header.clone());
    }
    original_records.extend(table.rows.iter().cloned());

    for (record, row) in original_records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    let mut records = original_records;
    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(CsvError::DuplicateHeader {
                    name,
                    first,
                    second,
                }) = duplicate_header(header)
                {
                    return Err(WriteError::DuplicateHeader {
                        name,
                        first,
                        second,
                    });
                }
            }

            let mut selected: Vec<(String, usize)> = Vec::new();
            for (index, name) in header.iter().enumerate() {
                if let Some(existing) = selected.iter_mut().find(|(n, _)| n == name) {
                    if table.duplicates == DuplicatePolicy::KeepLast {
                        existing.1 = index;
                    }
                } else {
                    selected.push((name.clone(), index));
                }
            }
            selected.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
            let indices: Vec<usize> = selected.iter().map(|(_, i)| *i).collect();
            let mut canonical = Vec::with_capacity(records.len());
            canonical.push(indices.iter().map(|&i| header[i].clone()).collect());
            for row in &table.rows {
                canonical.push(indices.iter().map(|&i| row[i].clone()).collect());
            }
            let header_record = canonical.remove(0);
            canonical.sort_by(|a: &Vec<String>, b: &Vec<String>| {
                for (x, y) in a.iter().zip(b.iter()) {
                    let order = x.as_bytes().cmp(y.as_bytes());
                    if order != std::cmp::Ordering::Equal {
                        return order;
                    }
                }
                a.len().cmp(&b.len())
            });
            records.clear();
            records.push(header_record);
            records.extend(canonical);
        } else {
            records.sort_by(|a, b| {
                for (x, y) in a.iter().zip(b.iter()) {
                    let order = x.as_bytes().cmp(y.as_bytes());
                    if order != std::cmp::Ordering::Equal {
                        return order;
                    }
                }
                a.len().cmp(&b.len())
            });
        }

        Ok(encode_records(
            &records,
            config.delimiter,
            config.quote,
            "\n",
            true,
        ))
    } else {
        let ending = match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        };
        Ok(encode_records(
            &records,
            config.delimiter,
            config.quote,
            ending,
            config.final_terminator,
        ))
    }
}
