impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

fn validate_config(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| records_to_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record = 0usize;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut field_started = false;
    let mut record_has_content = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if in_quotes {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                after_quote = false;
                field_started = false;
                record_has_content = true;
                i += 1;
            } else if b == b'\n' || b == b'\r' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                after_quote = false;
                field_started = false;
                record_has_content = false;
                record += 1;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: i });
            }
            continue;
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            field_started = false;
            record_has_content = true;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            field_started = false;
            record_has_content = false;
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
            in_quotes = true;
            quote_start = i;
            field_started = true;
            record_has_content = true;
            i += 1;
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            field_started = true;
            record_has_content = true;
            i += ch.len_utf8();
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if record_has_content || !row.is_empty() || !field.is_empty() || after_quote || field_started {
        row.push(field);
        records.push(row);
    }
    Ok(records)
}

fn records_to_table(
    mut records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(h) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in h.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name, second));
            }
        }
        for (i, row) in records.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if failed && policy != ErrorPolicy::Collect {
                output.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let (value, error) = if let Some(column) = resolved {
                if let Some(cell) = row.get(column) {
                    evaluate_cell(cell, spec, row_index, spec_index, column)
                } else {
                    (
                        None,
                        Some(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(column),
                            kind: CellErrorKind::MissingCell,
                        }),
                    )
                }
            } else {
                (
                    None,
                    Some(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    }),
                )
            };

            output.push(value);
            if let Some(error) = error {
                failed = true;
                result.errors.push(error);
            }

            if policy == ErrorPolicy::Stop && failed {
                for _ in spec_index + 1..columns.len() {
                    output.push(None);
                }
                break;
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::Stop && failed {
            break;
        }
    }

    result
}

fn evaluate_cell(
    cell: &str,
    spec: &ColumnSpec,
    row: usize,
    spec_index: usize,
    column: usize,
) -> (Option<Value>, Option<CellError>) {
    let error = |kind| {
        Some(CellError {
            row,
            spec: spec_index,
            column: Some(column),
            kind,
        })
    };

    if cell.is_empty() {
        if spec.nullable {
            return (Some(Value::Null), None);
        }
        if spec.kind != ColumnType::Text {
            return (None, error(CellErrorKind::Empty));
        }
    }

    let parsed = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
        ColumnType::Integer => parse_integer(cell)
            .map(Value::Integer)
            .map_err(|kind| kind),
        ColumnType::Decimal => parse_decimal(cell)
            .map(Value::Decimal)
            .map_err(|kind| kind),
    };

    match parsed {
        Ok(value) => (Some(value), None),
        Err(kind) => (None, error(kind)),
    }
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(CellErrorKind::InvalidInteger);
    }
    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() || (start == 1 && bytes.get(1) == Some(&b'0')) {
        return Err(CellErrorKind::InvalidInteger);
    }
    if bytes[start] == b'0' {
        if start + 1 != bytes.len() {
            return Err(CellErrorKind::InvalidInteger);
        }
    } else if !(b'1'..=b'9').contains(&bytes[start]) {
        return Err(CellErrorKind::InvalidInteger);
    }
    if bytes[start + 1..].iter().any(|b| !b.is_ascii_digit()) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    let bytes = s.as_bytes();
    let start = usize::from(bytes.first() == Some(&b'-'));
    let dot = match bytes.iter().position(|&b| b == b'.') {
        Some(i) => i,
        None => return Err(CellErrorKind::InvalidDecimal),
    };
    if dot < start || bytes.len() != dot + 3 {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let integral = &bytes[start..dot];
    if integral.is_empty()
        || (integral[0] == b'0' && integral.len() != 1)
        || (integral[0] != b'0' && !(b'1'..=b'9').contains(&integral[0]))
        || integral.iter().any(|b| !b.is_ascii_digit())
        || !bytes[dot + 1..].iter().all(|b| b.is_ascii_digit())
    {
        return Err(CellErrorKind::InvalidDecimal);
    }
    if start == 1 && integral == b"0" && bytes[dot + 1..] == *b"00" {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let magnitude = format!("{}{}", &s[..dot], &s[dot + 1..]);
    magnitude
        .parse::<i64>()
        .map_err(|_| CellErrorKind::DecimalOverflow)
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_config(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header);
    }
    original_records.extend(table.rows.iter());

    for (record, fields) in original_records.iter().enumerate() {
        for (column, field) in fields.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, fields) in original_records.iter().enumerate() {
        if fields.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        let header = table.header.as_ref().unwrap();
        let mut seen: Vec<(&str, usize)> = Vec::new();
        for (second, name) in header.iter().enumerate() {
            if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
                return Err(WriteError::DuplicateHeader {
                    name: name.clone(),
                    first: *first,
                    second,
                });
            }
            seen.push((name, second));
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();

    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                output_records.push(header.clone());
            }
            output_records.extend(table.rows.iter().cloned());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained: Vec<(String, usize)> = Vec::new();
                for (index, name) in header.iter().enumerate() {
                    if let Some(pos) = retained.iter().position(|(prior, _)| prior == name) {
                        match table.duplicates {
                            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {}
                            DuplicatePolicy::KeepLast => retained[pos] = (name.clone(), index),
                        }
                    } else {
                        retained.push((name.clone(), index));
                    }
                }
                retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
                output_records.push(retained.iter().map(|(name, _)| name.clone()).collect());

                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| retained.iter().map(|(_, index)| row[*index].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| compare_rows(a, b));
                output_records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_rows(a, b));
                output_records.extend(rows);
            }
        }
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let line_ending: &[u8] = if canonical || config.line_ending == LineEnding::Lf {
        b"\n"
    } else {
        b"\r\n"
    };

    let mut output = String::new();
    for (record_index, record) in output_records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(std::str::from_utf8(line_ending).unwrap());
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            write_field(
                &mut output,
                field,
                config.delimiter,
                config.quote,
                record.len() == 1 && field.is_empty(),
                record_index == 0 && column_index == 0,
            );
        }
    }
    if canonical || config.final_terminator {
        output.push_str(std::str::from_utf8(line_ending).unwrap());
    }
    Ok(output)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let ordering = left.as_bytes().cmp(right.as_bytes());
        if ordering != std::cmp::Ordering::Equal {
            return ordering;
        }
    }
    a.len().cmp(&b.len())
}

fn write_field(
    output: &mut String,
    field: &str,
    delimiter: u8,
    quote: u8,
    sole_empty: bool,
    first_output_field: bool,
) {
    let bytes = field.as_bytes();
    let edge_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let starts_bom = first_output_field && bytes.starts_with(&[0xef, 0xbb, 0xbf]);
    let needs_quote = field.is_empty() && sole_empty
        || edge_space
        || starts_bom
        || bytes.iter().any(|&b| {
            b == delimiter || b == quote || b == b'\r' || b == b'\n'
        });

    if !needs_quote {
        output.push_str(field);
        return;
    }

    output.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            output.push(quote as char);
            output.push(quote as char);
        } else {
            output.push(ch);
        }
    }
    output.push(quote as char);
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut record = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                field.push(b);
                i += 1;
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(String::from_utf8(field).expect("CSV fields remain valid UTF-8"));
                field = Vec::new();
                after_quote = false;
                i += 1;
            } else if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
                row.push(String::from_utf8(field).expect("CSV fields remain valid UTF-8"));
                field = Vec::new();
                records.push(std::mem::take(&mut row));
                record += 1;
                after_quote = false;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: i });
            }
            continue;
        }

        if b == quote {
            if field.is_empty() {
                in_quotes = true;
                quote_start = i;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
        } else if b == delimiter {
            row.push(String::from_utf8(field).expect("CSV fields remain valid UTF-8"));
            field = Vec::new();
            i += 1;
        } else if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
            row.push(String::from_utf8(field).expect("CSV fields remain valid UTF-8"));
            field = Vec::new();
            records.push(std::mem::take(&mut row));
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else {
            field.push(b);
            i += 1;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if !row.is_empty() || !field.is_empty() || after_quote {
        row.push(String::from_utf8(field).expect("CSV fields remain valid UTF-8"));
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn build_table(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let mut records = records.into_iter();
    let header = if config.headers {
        records.next()
    } else {
        None
    };
    let rows: Vec<Vec<String>> = records.collect();

    if let Some(header) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: HashMap<&str, usize> = HashMap::new();
            for (second, name) in header.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second,
                    });
                }
                seen.insert(name.as_str(), second);
            }
        }

        let first_data_record = 1usize;
        for (offset, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: first_data_record + offset,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(value: &str) -> bool {
    let b = value.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start])
        && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_hundredths(value: &str) -> Option<Result<i64, CellErrorKind>> {
    let b = value.as_bytes();
    let negative = b.first() == Some(&b'-');
    let start = usize::from(negative);
    if start >= b.len() {
        return None;
    }
    let dot = b.iter().position(|&byte| byte == b'.')?;
    if dot < start || b.len() != dot + 3 {
        return None;
    }
    let integral = &b[start..dot];
    if integral.is_empty()
        || !integral.iter().all(u8::is_ascii_digit)
        || (integral.len() > 1 && integral[0] == b'0')
        || b[dot + 1..].iter().any(|byte| !byte.is_ascii_digit())
    {
        return None;
    }
    if negative && integral.iter().all(|&byte| byte == b'0') && b[dot + 1..].iter().all(|&x| x == b'0') {
        return None;
    }
    let magnitude = value[start..].replace(".", "");
    let signed = if negative {
        format!("-{magnitude}")
    } else {
        magnitude
    };
    Some(signed.parse::<i64>().map_err(|_| CellErrorKind::DecimalOverflow))
}

fn evaluate_cell(
    cell: &str,
    kind: ColumnType,
    nullable: bool,
) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if nullable {
            return Ok(Value::Null);
        }
        if kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match kind {
        ColumnType::Text => Ok(Value::Text(cell.to_owned())),
        ColumnType::Integer => {
            if !integer_syntax(cell) {
                return Err(CellErrorKind::InvalidInteger);
            }
            cell.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => match decimal_hundredths(cell) {
            None => Err(CellErrorKind::InvalidDecimal),
            Some(Ok(value)) => Ok(Value::Decimal(value)),
            Some(Err(error)) => Err(error),
        },
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };
    let mut stopped = false;

    for (row_index, row) in table.rows.iter().enumerate() {
        if stopped {
            break;
        }
        let mut output = vec![None; columns.len()];
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed {
                break;
            }

            let column_index = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let error = if column_index.is_none() {
                Some(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                })
            } else {
                let index = column_index.unwrap();
                match row.get(index) {
                    None => Some(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(index),
                        kind: CellErrorKind::MissingCell,
                    }),
                    Some(cell) => match evaluate_cell(cell, spec.kind, spec.nullable) {
                        Ok(value) => {
                            output[spec_index] = Some(value);
                            None
                        }
                        Err(kind) => Some(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(index),
                            kind,
                        }),
                    },
                }
            };

            if let Some(error) = error {
                result.errors.push(error);
                match policy {
                    ErrorPolicy::Collect => {}
                    ErrorPolicy::FirstPerRow => row_failed = true,
                    ErrorPolicy::Stop => {
                        row_failed = true;
                        stopped = true;
                    }
                }
            }
        }

        result.rows.push(output);
    }

    result
}

fn table_records(table: &Table) -> Vec<&Vec<String>> {
    let mut records = Vec::with_capacity(table.rows.len() + usize::from(table.header.is_some()));
    if let Some(header) = &table.header {
        records.push(header);
    }
    records.extend(table.rows.iter());
    records
}

fn duplicate_header(header: &[String]) -> Option<WriteError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(WriteError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name.as_str(), second);
    }
    None
}

fn quote_field(
    field: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole_empty: bool,
) -> String {
    let bytes = field.as_bytes();
    let needs_quote = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
        || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t')
        || (first_field_first_record && field.starts_with('\u{feff}'))
        || (sole_empty && field.is_empty());

    if !needs_quote {
        return field.to_owned();
    }

    let mut output = String::with_capacity(field.len() + 2);
    output.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            output.push(quote as char);
        }
        output.push(ch);
    }
    output.push(quote as char);
    output
}

fn compare_rows(a: &Vec<String>, b: &Vec<String>) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let order = left.as_bytes().cmp(right.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    for (record_index, row) in table_records(table).iter().enumerate() {
        for (column_index, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (offset, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: offset + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in table_records(table).iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(error) = duplicate_header(header) {
                    return Err(error);
                }
            }
        }
    }

    let mut records: Vec<Vec<String>> = if let Some(header) = &table.header {
        let mut records = Vec::with_capacity(table.rows.len() + 1);
        records.push(header.clone());
        records.extend(table.rows.clone());
        records
    } else {
        table.rows.clone()
    };

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            let mut selected: HashMap<&str, usize> = HashMap::new();
            for (index, name) in header.iter().enumerate() {
                match table.duplicates {
                    DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                        selected.entry(name.as_str()).or_insert(index);
                    }
                    DuplicatePolicy::KeepLast => {
                        selected.insert(name.as_str(), index);
                    }
                }
            }
            let mut columns: Vec<(&str, usize)> = selected.into_iter().collect();
            columns.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

            let sorted_header: Vec<String> =
                columns.iter().map(|(name, _)| (*name).to_owned()).collect();
            let mut projected = Vec::with_capacity(table.rows.len() + 1);
            projected.push(sorted_header);
            for row in &table.rows {
                projected.push(columns.iter().map(|(_, index)| row[*index].clone()).collect());
            }
            records = projected;
            records[1..].sort_by(compare_rows);
        } else {
            records.sort_by(compare_rows);
        }
    }

    let canonical = config.mode == WriteMode::Canonical;
    let terminator = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };

    let mut output = String::new();
    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(terminator);
        }
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            let first_field_first_record = record_index == 0 && column_index == 0;
            output.push_str(&quote_field(
                field,
                config.delimiter,
                config.quote,
                first_field_first_record,
                row.len() == 1,
            ));
        }
    }

    if canonical {
        if !records.is_empty() {
            output.push('\n');
        }
    } else if config.final_terminator && !records.is_empty() {
        output.push_str(terminator);
    }

    Ok(output)
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut field_started = false;

    #[derive(Clone, Copy)]
    enum State {
        Start,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::Start;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if !matches!(state, State::Quoted { .. }) && b == b'\r'
            && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n')
        {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::Start => {
                if b == delimiter {
                    record.push(String::new());
                    i += 1;
                } else if b == quote {
                    state = State::Quoted { opening: i };
                    field_started = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    field.clear();
                    field_started = false;
                    i += 1;
                } else if b == b'\r' {
                    record.push(String::new());
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    field.clear();
                    field_started = false;
                    i += 2;
                } else {
                    field.push(b as char);
                    state = State::Unquoted;
                    field_started = true;
                    i += 1;
                }
            }
            State::Unquoted => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::Start;
                    field_started = false;
                    i += 2;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else {
                    field.push(b as char);
                    i += 1;
                }
            }
            State::Quoted { opening } => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    field.push(b as char);
                    let _ = opening;
                    i += 1;
                }
            }
            State::AfterQuote => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::Start;
                    field_started = false;
                    i += 1;
                } else if b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(record);
                    record = Vec::new();
                    record_index += 1;
                    state = State::Start;
                    field_started = false;
                    i += 2;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::Start if !field_started && record.is_empty() => {}
        _ => {
            record.push(field);
            records.push(record);
        }
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        if let Some(first) = header[..second].iter().position(|v| v == &header[second]) {
            return Some(CsvError::DuplicateHeader {
                name: header[second].clone(),
                first,
                second,
            });
        }
    }
    None
}

fn table_from_input(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| CsvError::InvalidConfig { component })?;

    let records = parse_with_dialect(input, config.delimiter, config.quote)?;
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect())
        }
    } else {
        (None, records)
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(h) {
                return Err(error);
            }
        }
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(ref result) = self.cached {
            return result.clone();
        }
        let result = table_from_input(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if start >= b.len() {
        return false;
    }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    let whole = &b[start..dot];
    let frac = &b[dot + 1..];
    if frac.len() != 2 || !frac.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if whole.is_empty() {
        return false;
    }
    if whole == b"0" {
        return !s.starts_with("-0.") || frac != b"00";
    }
    (b'1'..=b'9').contains(&whole[0]) && whole[1..].iter().all(u8::is_ascii_digit)
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let unsigned = if negative { &s[1..] } else { s };
    let dot = unsigned.find('.').unwrap();
    let whole = unsigned[..dot].parse::<i128>().unwrap();
    let frac = unsigned[dot + 1..].parse::<i128>().unwrap();
    let magnitude = whole * 100 + frac;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed {
                output.push(None);
                continue;
            }

            let column_index = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let error_or_value = match column_index {
                None => Err(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                }),
                Some(index) => match row.get(index) {
                    None => Err(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(index),
                        kind: CellErrorKind::MissingCell,
                    }),
                    Some(cell) if cell.is_empty() && spec.nullable => Ok(Value::Null),
                    Some(cell) if cell.is_empty() && spec.kind != ColumnType::Text => {
                        Err(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(index),
                            kind: CellErrorKind::Empty,
                        })
                    }
                    Some(cell) => match spec.kind {
                        ColumnType::Text => Ok(Value::Text(cell.clone())),
                        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
                        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
                        ColumnType::Boolean => match cell.as_str() {
                            "yes" | "1" => Ok(Value::Boolean(true)),
                            "no" | "0" => Ok(Value::Boolean(false)),
                            _ => Err(CellErrorKind::InvalidBoolean),
                        },
                    }
                    .map_err(|kind| CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(index),
                        kind,
                    }),
                },
            };

            match error_or_value {
                Ok(value) => output.push(Some(value)),
                Err(error) => {
                    result.errors.push(error);
                    output.push(None);
                    if policy != ErrorPolicy::Collect {
                        row_failed = true;
                    }
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(output);
                        return result;
                    }
                }
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::FirstPerRow {
            continue;
        }
    }

    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode {
    Preserve,
    Canonical,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns {
        record: usize,
    },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn first_nul(table: &Table) -> Option<WriteError> {
    if let Some(header) = &table.header {
        for (column, field) in header.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Some(WriteError::Nul {
                    record: 0,
                    column,
                    byte,
                });
            }
        }
    }
    let offset = usize::from(table.header.is_some());
    for (row_index, row) in table.rows.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Some(WriteError::Nul {
                    record: row_index + offset,
                    column,
                    byte,
                });
            }
        }
    }
    None
}

fn encode_field(field: &str, delimiter: u8, quote: u8, first: bool, sole: bool) -> String {
    let bytes = field.as_bytes();
    let boundary_space = bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
        || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t');
    let starts_bom = first && field.starts_with('\u{feff}');
    let must_quote = field.as_bytes().contains(&delimiter)
        || field.as_bytes().contains(&quote)
        || field.contains('\r')
        || field.contains('\n')
        || boundary_space
        || starts_bom
        || (sole && field.is_empty());

    if !must_quote {
        return field.to_string();
    }

    let mut out = String::with_capacity(field.len() + 2);
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
    out
}

fn encode_records(
    records: &[Vec<String>],
    delimiter: u8,
    quote: u8,
    ending: &str,
    final_terminator: bool,
) -> String {
    if records.is_empty() {
        return String::new();
    }
    let mut out = String::new();
    for (ri, record) in records.iter().enumerate() {
        if ri > 0 {
            out.push_str(ending);
        }
        for (ci, field) in record.iter().enumerate() {
            if ci > 0 {
                out.push(delimiter as char);
            }
            let encoded = encode_field(field, delimiter, quote, ri == 0 && ci == 0, record.len() == 1);
            out.push_str(&encoded);
        }
    }
    if final_terminator {
        out.push_str(ending);
    }
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    if let Some(error) = first_nul(table) {
        return Err(error);
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    let mut original_records = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header.clone());
    }
    original_records.extend(table.rows.iter().cloned());

    for (record, row) in original_records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    let mut records = original_records;
    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(CsvError::DuplicateHeader {
                    name,
                    first,
                    second,
                }) = duplicate_header(header)
                {
                    return Err(WriteError::DuplicateHeader {
                        name,
                        first,
                        second,
                    });
                }
            }

            let mut selected: Vec<(String, usize)> = Vec::new();
            for (index, name) in header.iter().enumerate() {
                if let Some(existing) = selected.iter_mut().find(|(n, _)| n == name) {
                    if table.duplicates == DuplicatePolicy::KeepLast {
                        existing.1 = index;
                    }
                } else {
                    selected.push((name.clone(), index));
                }
            }
            selected.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
            let indices: Vec<usize> = selected.iter().map(|(_, i)| *i).collect();
            let mut canonical = Vec::with_capacity(records.len());
            canonical.push(indices.iter().map(|&i| header[i].clone()).collect());
            for row in &table.rows {
                canonical.push(indices.iter().map(|&i| row[i].clone()).collect());
            }
            let header_record = canonical.remove(0);
            canonical.sort_by(|a: &Vec<String>, b: &Vec<String>| {
                for (x, y) in a.iter().zip(b.iter()) {
                    let order = x.as_bytes().cmp(y.as_bytes());
                    if order != std::cmp::Ordering::Equal {
                        return order;
                    }
                }
                a.len().cmp(&b.len())
            });
            records.clear();
            records.push(header_record);
            records.extend(canonical);
        } else {
            records.sort_by(|a, b| {
                for (x, y) in a.iter().zip(b.iter()) {
                    let order = x.as_bytes().cmp(y.as_bytes());
                    if order != std::cmp::Ordering::Equal {
                        return order;
                    }
                }
                a.len().cmp(&b.len())
            });
        }

        Ok(encode_records(
            &records,
            config.delimiter,
            config.quote,
            "\n",
            true,
        ))
    } else {
        let ending = match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        };
        Ok(encode_records(
            &records,
            config.delimiter,
            config.quote,
            ending,
            config.final_terminator,
        ))
    }
}
