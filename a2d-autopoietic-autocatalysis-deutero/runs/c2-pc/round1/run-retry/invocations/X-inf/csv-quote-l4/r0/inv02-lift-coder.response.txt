#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

fn validate_config(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| records_to_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record = 0usize;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut field_started = false;
    let mut record_has_content = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if in_quotes {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                after_quote = false;
                field_started = false;
                record_has_content = true;
                i += 1;
            } else if b == b'\n' || b == b'\r' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                after_quote = false;
                field_started = false;
                record_has_content = false;
                record += 1;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: i });
            }
            continue;
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            field_started = false;
            record_has_content = true;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            field_started = false;
            record_has_content = false;
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
            in_quotes = true;
            quote_start = i;
            field_started = true;
            record_has_content = true;
            i += 1;
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            field_started = true;
            record_has_content = true;
            i += ch.len_utf8();
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if record_has_content || !row.is_empty() || !field.is_empty() || after_quote || field_started {
        row.push(field);
        records.push(row);
    }
    Ok(records)
}

fn records_to_table(
    mut records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(h) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();
            for (second, name) in h.iter().enumerate() {
                if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first: *first,
                        second,
                    });
                }
                seen.push((name, second));
            }
        }
        for (i, row) in records.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]

    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if failed && policy != ErrorPolicy::Collect {
                output.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let (value, error) = if let Some(column) = resolved {
                if let Some(cell) = row.get(column) {
                    evaluate_cell(cell, spec, row_index, spec_index, column)
                } else {
                    (
                        None,
                        Some(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(column),
                            kind: CellErrorKind::MissingCell,
                        }),
                    )
                }
            } else {
                (
                    None,
                    Some(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    }),
                )
            };

            output.push(value);
            if let Some(error) = error {
                failed = true;
                result.errors.push(error);
            }

            if policy == ErrorPolicy::Stop && failed {
                for _ in spec_index + 1..columns.len() {
                    output.push(None);
                }
                break;
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::Stop && failed {
            break;
        }
    }

    result
}

fn evaluate_cell(
    cell: &str,
    spec: &ColumnSpec,
    row: usize,
    spec_index: usize,
    column: usize,
) -> (Option<Value>, Option<CellError>) {
    let error = |kind| {
        Some(CellError {
            row,
            spec: spec_index,
            column: Some(column),
            kind,
        })
    };

    if cell.is_empty() {
        if spec.nullable {
            return (Some(Value::Null), None);
        }
        if spec.kind != ColumnType::Text {
            return (None, error(CellErrorKind::Empty));
        }
    }

    let parsed = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
        ColumnType::Integer => parse_integer(cell)
            .map(Value::Integer)
            .map_err(|kind| kind),
        ColumnType::Decimal => parse_decimal(cell)
            .map(Value::Decimal)
            .map_err(|kind| kind),
    };

    match parsed {
        Ok(value) => (Some(value), None),
        Err(kind) => (None, error(kind)),
    }
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(CellErrorKind::InvalidInteger);
    }
    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() || (start == 1 && bytes.get(1) == Some(&b'0')) {
        return Err(CellErrorKind::InvalidInteger);
    }
    if bytes[start] == b'0' {
        if start + 1 != bytes.len() {
            return Err(CellErrorKind::InvalidInteger);
        }
    } else if !(b'1'..=b'9').contains(&bytes[start]) {
        return Err(CellErrorKind::InvalidInteger);
    }
    if bytes[start + 1..].iter().any(|b| !b.is_ascii_digit()) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    let bytes = s.as_bytes();
    let start = usize::from(bytes.first() == Some(&b'-'));
    let dot = match bytes.iter().position(|&b| b == b'.') {
        Some(i) => i,
        None => return Err(CellErrorKind::InvalidDecimal),
    };
    if dot < start || bytes.len() != dot + 3 {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let integral = &bytes[start..dot];
    if integral.is_empty()
        || (integral[0] == b'0' && integral.len() != 1)
        || (integral[0] != b'0' && !(b'1'..=b'9').contains(&integral[0]))
        || integral.iter().any(|b| !b.is_ascii_digit())
        || !bytes[dot + 1..].iter().all(|b| b.is_ascii_digit())
    {
        return Err(CellErrorKind::InvalidDecimal);
    }
    if start == 1 && integral == b"0" && bytes[dot + 1..] == *b"00" {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let magnitude = format!("{}{}", &s[..dot], &s[dot + 1..]);
    magnitude
        .parse::<i64>()
        .map_err(|_| CellErrorKind::DecimalOverflow)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_config(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header);
    }
    original_records.extend(table.rows.iter());

    for (record, fields) in original_records.iter().enumerate() {
        for (column, field) in fields.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, fields) in original_records.iter().enumerate() {
        if fields.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        let header = table.header.as_ref().unwrap();
        let mut seen: Vec<(&str, usize)> = Vec::new();
        for (second, name) in header.iter().enumerate() {
            if let Some((_, first)) = seen.iter().find(|(prior, _)| *prior == name) {
                return Err(WriteError::DuplicateHeader {
                    name: name.clone(),
                    first: *first,
                    second,
                });
            }
            seen.push((name, second));
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();

    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                output_records.push(header.clone());
            }
            output_records.extend(table.rows.iter().cloned());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained: Vec<(String, usize)> = Vec::new();
                for (index, name) in header.iter().enumerate() {
                    if let Some(pos) = retained.iter().position(|(prior, _)| prior == name) {
                        match table.duplicates {
                            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {}
                            DuplicatePolicy::KeepLast => retained[pos] = (name.clone(), index),
                        }
                    } else {
                        retained.push((name.clone(), index));
                    }
                }
                retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
                output_records.push(retained.iter().map(|(name, _)| name.clone()).collect());

                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| retained.iter().map(|(_, index)| row[*index].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| compare_rows(a, b));
                output_records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_rows(a, b));
                output_records.extend(rows);
            }
        }
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let line_ending: &[u8] = if canonical || config.line_ending == LineEnding::Lf {
        b"\n"
    } else {
        b"\r\n"
    };

    let mut output = String::new();
    for (record_index, record) in output_records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(std::str::from_utf8(line_ending).unwrap());
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            write_field(
                &mut output,
                field,
                config.delimiter,
                config.quote,
                record.len() == 1 && field.is_empty(),
                record_index == 0 && column_index == 0,
            );
        }
    }
    if canonical || config.final_terminator {
        output.push_str(std::str::from_utf8(line_ending).unwrap());
    }
    Ok(output)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let ordering = left.as_bytes().cmp(right.as_bytes());
        if ordering != std::cmp::Ordering::Equal {
            return ordering;
        }
    }
    a.len().cmp(&b.len())
}

fn write_field(
    output: &mut String,
    field: &str,
    delimiter: u8,
    quote: u8,
    sole_empty: bool,
    first_output_field: bool,
) {
    let bytes = field.as_bytes();
    let edge_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
    let starts_bom = first_output_field && bytes.starts_with(&[0xef, 0xbb, 0xbf]);
    let needs_quote = field.is_empty() && sole_empty
        || edge_space
        || starts_bom
        || bytes.iter().any(|&b| {
            b == delimiter || b == quote || b == b'\r' || b == b'\n'
        });

    if !needs_quote {
        output.push_str(field);
        return;
    }

    output.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            output.push(quote as char);
            output.push(quote as char);
        } else {
            output.push(ch);
        }
    }
    output.push(quote as char);
}
