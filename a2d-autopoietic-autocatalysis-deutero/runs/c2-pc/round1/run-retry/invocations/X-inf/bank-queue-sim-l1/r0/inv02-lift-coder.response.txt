#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut queue: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0;
    let mut current_time = 0u64;
    let mut active: Option<(u64, u32)> = None;
    let mut previous_finish: Option<u64> = None;
    let mut first_dispatch = true;
    let mut finished_count = 0usize;

    while finished_count < customers.len() {
        let next_arrival_time = arrivals.get(next_arrival).map(|c| c.arrival);
        let next_finish_time = active.map(|(finish, _)| finish);

        let time = match (next_arrival_time, next_finish_time) {
            (Some(a), Some(f)) => a.min(f),
            (Some(a), None) => a,
            (None, Some(f)) => f,
            (None, None) => break,
        };
        current_time = time;

        if let Some((finish, id)) = active {
            if finish == time {
                events.push(Event {
                    time,
                    customer: id,
                    kind: Kind::Finish,
                });
                active = None;
                previous_finish = Some(time);
                finished_count += 1;
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            queue.push_back(customer);
            next_arrival += 1;
        }

        if active.is_none() {
            if let Some(customer) = queue.pop_front() {
                let wake = first_dispatch || previous_finish.map_or(false, |f| time > f);
                let service_end = time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    },
                )?;
                let finish = service_end.checked_add(if wake { wake_cost } else { 0 }).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    },
                )?;

                events.push(Event {
                    time,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });
                first_dispatch = false;

                if finish == time {
                    events.push(Event {
                        time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = Some(time);
                    finished_count += 1;

                    while let Some(customer) = queue.pop_front() {
                        let service_end = time.checked_add(customer.service).ok_or(
                            Error::TimeOverflow {
                                customer: customer.id,
                                time,
                            },
                        )?;
                        let finish = service_end.checked_add(0).ok_or(Error::TimeOverflow {
                            customer: customer.id,
                            time,
                        })?;

                        events.push(Event {
                            time,
                            customer: customer.id,
                            kind: Kind::Start {
                                finish,
                                wake: false,
                            },
                        });

                        if finish != time {
                            active = Some((finish, customer.id));
                            break;
                        }

                        events.push(Event {
                            time,
                            customer: customer.id,
                            kind: Kind::Finish,
                        });
                        previous_finish = Some(time);
                        finished_count += 1;
                    }
                } else {
                    active = Some((finish, customer.id));
                }
            }
        }
    }

    let _ = current_time;
    Ok(events)
}
