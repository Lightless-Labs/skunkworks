#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

impl Machine {
    const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains(&product.id.as_str()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !Self::DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = self.selected.as_ref().ok_or(Error::NoSelection)?;
        let product = self
            .products
            .iter()
            .find(|product| product.id == *selected_id)
            .expect("selected product is always in the catalog");

        let price = product.price;
        let id = product.id.clone();
        let credit = self.credit;
        if credit < price {
            return Err(Error::Underpaid { price, credit });
        }

        let amount = credit - price;
        let mut change_counts = [0u16; 4];
        let mut remaining = amount;

        for index in (0..4).rev() {
            let denomination = Self::DENOMINATIONS[index];
            let count = self.tubes[index].min(remaining / denomination);
            change_counts[index] = count;
            remaining -= count * denomination;
        }

        if remaining != 0 {
            if let Some(exact) = Self::best_exact_change(amount, self.tubes) {
                change_counts = exact;
            } else {
                return Err(Error::NoChange { amount });
            }
        }

        let mut proposed_tubes = self.tubes;
        for index in 0..4 {
            proposed_tubes[index] -= change_counts[index];
        }
        for &coin in &self.escrow {
            let index = Self::DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow contains only accepted denominations");
            proposed_tubes[index] += 1;
        }

        for (index, &count) in proposed_tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: Self::DENOMINATIONS[index],
                });
            }
        }

        let mut change = Vec::new();
        for index in (0..4).rev() {
            for _ in 0..change_counts[index] {
                change.push(Self::DENOMINATIONS[index]);
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let refunded = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        Ok(refunded)
    }

    fn best_exact_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
        fn search(
            index: usize,
            remaining: u16,
            tubes: &[u16; 4],
            counts: &mut [u16; 4],
            best: &mut Option<[u16; 4]>,
            best_total: &mut u16,
        ) {
            if index == 4 {
                if remaining == 0 {
                    let total: u16 = counts.iter().copied().sum();
                    let better = match best {
                        None => true,
                        Some(current) => {
                            total < *best_total
                                || (total == *best_total
                                    && (0..4).rev().find_map(|i| {
                                        (counts[i] != current[i]).then_some(counts[i] > current[i])
                                    }) == Some(true))
                        }
                    };
                    if better {
                        *best = Some(*counts);
                        *best_total = total;
                    }
                }
                return;
            }

            let denomination = Self::DENOMINATIONS[index];
            let max_count = tubes[index].min(remaining / denomination);
            for count in 0..=max_count {
                counts[index] = count;
                search(
                    index + 1,
                    remaining - count * denomination,
                    tubes,
                    counts,
                    best,
                    best_total,
                );
            }
            counts[index] = 0;
        }

        let mut best = None;
        let mut best_total = u16::MAX;
        search(
            0,
            amount,
            &tubes,
            &mut [0; 4],
            &mut best,
            &mut best_total,
        );
        best
    }
}
