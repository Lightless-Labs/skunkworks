use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|cell| cell == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|cell| cell == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = Vec::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut record = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                field.push(b);
                i += 1;
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                fields.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                after_quote = false;
                record_started = true;
                i += 1;
            } else if b == b'\n' || b == b'\r' {
                fields.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                records.push(fields);
                fields = Vec::new();
                after_quote = false;
                record_started = false;
                record += 1;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: i });
            }
            continue;
        }

        if b == b'\r' || b == b'\n' {
            fields.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            records.push(fields);
            fields = Vec::new();
            record_started = false;
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == delimiter {
            fields.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            record_started = true;
            i += 1;
        } else if b == quote {
            if field.is_empty() {
                in_quotes = true;
                quote_open = i;
                record_started = true;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
        } else {
            field.push(b);
            record_started = true;
            i += 1;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }
    if record_started || after_quote || !fields.is_empty() {
        fields.push(String::from_utf8(field).unwrap());
        records.push(fields);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            (Some(records[0].clone()), records[1..].to_vec())
        }
    } else {
        (None, records)
    };

    if let (Some(header), DuplicatePolicy::Reject) = (&header, config.duplicates) {
        let mut seen: HashMap<&str, usize> = HashMap::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            seen.insert(name.as_str(), second);
        }
    }

    if let Some(header) = &header {
        for (index, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef {
    Index(usize),
    Name(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {
    Text,
    Integer,
    Decimal,
    Boolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() {
        return false;
    }
    if bytes[start] == b'0' {
        return start + 1 == bytes.len();
    }
    (b'1'..=b'9').contains(&bytes[start])
        && bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.len() < 4 {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if bytes.len() < start + 4 {
        return false;
    }
    let dot = bytes.len() - 3;
    if bytes[dot] != b'.'
        || !bytes[dot + 1..].iter().all(u8::is_ascii_digit)
        || dot < start + 1
    {
        return false;
    }
    if bytes[start] == b'0' {
        if dot != start + 1 {
            return false;
        }
    } else if !(b'1'..=b'9').contains(&bytes[start])
        || !bytes[start + 1..dot].iter().all(u8::is_ascii_digit)
    {
        return false;
    }
    !(bytes[0] == b'-' && bytes[start] == b'0' && bytes[dot + 1..].iter().all(|&b| b == b'0'))
}

fn eval_cell(cell: &str, kind: ColumnType, nullable: bool) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if nullable {
            return Ok(Value::Null);
        }
        if kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }
    match kind {
        ColumnType::Text => Ok(Value::Text(cell.to_owned())),
        ColumnType::Integer => {
            if !integer_syntax(cell) {
                return Err(CellErrorKind::InvalidInteger);
            }
            cell.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(cell) {
                return Err(CellErrorKind::InvalidDecimal);
            }
            let negative = cell.starts_with('-');
            let unsigned = if negative { &cell[1..] } else { cell };
            let mut digits = unsigned.replace('.', "");
            if negative {
                digits.insert(0, '-');
            }
            digits
                .parse::<i64>()
                .map(Value::Decimal)
                .map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut had_error = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if policy == ErrorPolicy::FirstPerRow && had_error {
                break;
            }
            if policy == ErrorPolicy::Stop && !result.errors.is_empty() {
                break;
            }

            let (column_index, cell) = match &spec.column {
                ColumnRef::Index(index) => (*index, row.get(*index)),
                ColumnRef::Name(name) => match table.column_index(name) {
                    Some(index) => (index, row.get(index)),
                    None => {
                        result.errors.push(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: None,
                            kind: CellErrorKind::UnknownColumn,
                        });
                        had_error = true;
                        if policy == ErrorPolicy::Stop {
                            break;
                        }
                        continue;
                    }
                },
            };

            let Some(cell) = cell else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column_index),
                    kind: CellErrorKind::MissingCell,
                });
                had_error = true;
                if policy == ErrorPolicy::Stop {
                    break;
                }
                continue;
            };

            match eval_cell(cell, spec.kind, spec.nullable) {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind,
                    });
                    had_error = true;
                    if policy == ErrorPolicy::Stop {
                        break;
                    }
                }
            }

            if policy == ErrorPolicy::FirstPerRow && had_error {
                break;
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::Stop && !result.errors.is_empty() {
            break;
        }
    }

    result
}
