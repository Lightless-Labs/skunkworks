pub struct StreamReader {

    input: String,

        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with_config(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output_row = Vec::with_capacity(columns.len());
        let mut failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if failed {
                output_row.push(None);
                continue;
            }

            let (column_index, cell) = match &spec.column {
                ColumnRef::Index(index) => (*index, row.get(*index)),
                ColumnRef::Name(name) => match table.column_index(name) {
                    Some(index) => (index, row.get(index)),
                    None => {
                        output_row.push(None);
                        result.errors.push(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: None,
                            kind: CellErrorKind::UnknownColumn,
                        });
                        failed = policy != ErrorPolicy::Collect;
                        continue;
                    }
                },
            };

            let Some(cell) = cell else {
                output_row.push(None);
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column_index),
                    kind: CellErrorKind::MissingCell,
                });
                failed = policy != ErrorPolicy::Collect;
                continue;
            };

            match convert_cell(cell, spec) {
                Ok(value) => output_row.push(Some(value)),
                Err(kind) => {
                    output_row.push(None);
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind,
                    });
                    failed = policy != ErrorPolicy::Collect;
                }
            }
        }

        result.rows.push(output_row);
        if failed && policy == ErrorPolicy::Stop {
            break;
        }
    }

    result
}

fn convert_cell(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

fn valid_integer_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let digits = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return bytes[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn parse_integer(value: &str) -> Result<i64, CellErrorKind> {
    if !valid_integer_syntax(value) {
        return Err(CellErrorKind::InvalidInteger);
    }
    value
        .parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn valid_decimal_syntax(value: &str) -> bool {
    let bytes = value.as_bytes();
    if bytes.len() < 4 {
        return false;
    }
    let negative = bytes[0] == b'-';
    let start = usize::from(negative);
    let Some(dot) = bytes.iter().position(|&byte| byte == b'.') else {
        return false;
    };
    if dot < start || bytes.len() - dot != 3 {
        return false;
    }

    let integral = &bytes[start..dot];
    let fractional = &bytes[dot + 1..];
    if integral.is_empty()
        || fractional.len() != 2
        || !fractional.iter().all(u8::is_ascii_digit)
    {
        return false;
    }
    if integral == b"0" {
        if negative && fractional == b"00" {
            return false;
        }
    } else if integral[0].is_ascii_digit()
        && integral[0] != b'0'
        && integral.iter().all(u8::is_ascii_digit)
    {
        // Valid integral part.
    } else {
        return false;
    }
    true
}

fn parse_decimal(value: &str) -> Result<i64, CellErrorKind> {
    if !valid_decimal_syntax(value) {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = value.starts_with('-');
    let digits: String = value
        .bytes()
        .filter(|&byte| byte != b'-' && byte != b'.')
        .map(char::from)
        .collect();
    let magnitude = digits
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_config(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let records = parse_records(input, config.delimiter, config.quote)?;

    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(header_values) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen = std::collections::HashMap::<&str, usize>::new();
            for (second, name) in header_values.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second,
                    });
                }
                seen.insert(name, second);
            }
        }

        for (data_index, row) in rows.iter().enumerate() {
            if row.len() != header_values.len() {
                return Err(CsvError::HeaderWidth {
                    record: data_index + 1,
                    expected: header_values.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

fn parse_records(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::<Vec<String>>::new();
    let mut fields = Vec::<String>::new();
    let mut field = String::new();
    let mut index = start;
    let mut record_index = 0usize;
    let mut record_started = false;

    #[derive(Clone, Copy, PartialEq, Eq)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted,
        AfterQuote,
    }

    let mut state = State::FieldStart;
    let mut opening_quote = 0usize;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: index,
            });
        }

        if state != State::Quoted && byte == b'\r' {
            if bytes.get(index + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: index,
                });
            }
            fields.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut fields));
            index += 2;
            record_index += 1;
            record_started = false;
            state = State::FieldStart;
            continue;
        }

        match state {
            State::Quoted => {
                if byte == quote {
                    if bytes.get(index + 1) == Some(&quote) {
                        field.push(quote as char);
                        index += 2;
                    } else {
                        state = State::AfterQuote;
                        index += 1;
                    }
                } else {
                    let ch = input[index..].chars().next().expect("valid UTF-8 boundary");
                    field.push(ch);
                    index += ch.len_utf8();
                }
            }
            State::AfterQuote => {
                if byte == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    record_started = true;
                    index += 1;
                } else if byte == b'\n' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                    index += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: index,
                    });
                }
            }
            State::FieldStart => {
                if byte == delimiter {
                    fields.push(String::new());
                    record_started = true;
                    index += 1;
                } else if byte == b'\n' {
                    fields.push(String::new());
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    index += 1;
                } else if byte == quote {
                    opening_quote = index;
                    state = State::Quoted;
                    record_started = true;
                    index += 1;
                } else {
                    let ch = input[index..].chars().next().expect("valid UTF-8 boundary");
                    field.push(ch);
                    record_started = true;
                    state = State::Unquoted;
                    index += ch.len_utf8();
                }
            }
            State::Unquoted => {
                if byte == delimiter {
                    fields.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    record_started = true;
                    index += 1;
                } else if byte == b'\n' {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                    index += 1;
                } else if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: index,
                    });
                } else {
                    let ch = input[index..].chars().next().expect("valid UTF-8 boundary");
                    field.push(ch);
                    index += ch.len_utf8();
                }
            }
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening_quote,
        });
    }

    if record_started || !fields.is_empty() || !field.is_empty() || state == State::AfterQuote {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = Vec::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut record = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                field.push(b);
                i += 1;
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                fields.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                after_quote = false;
                record_started = true;
                i += 1;
            } else if b == b'\n' || b == b'\r' {
                fields.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                records.push(fields);
                fields = Vec::new();
                after_quote = false;
                record_started = false;
                record += 1;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: i });
            }
            continue;
        }

        if b == b'\r' || b == b'\n' {
            fields.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            records.push(fields);
            fields = Vec::new();
            record_started = false;
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == delimiter {
            fields.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            record_started = true;
            i += 1;
        } else if b == quote {
            if field.is_empty() {
                in_quotes = true;
                quote_open = i;
                record_started = true;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
        } else {
            field.push(b);
            record_started = true;
            i += 1;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }
    if record_started || after_quote || !fields.is_empty() {
        fields.push(String::from_utf8(field).unwrap());
        records.push(fields);
    }
    Ok(records)
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            (Some(records[0].clone()), records[1..].to_vec())
        }
    } else {
        (None, records)
    };

    if let (Some(header), DuplicatePolicy::Reject) = (&header, config.duplicates) {
        let mut seen: HashMap<&str, usize> = HashMap::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name.as_str()) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            seen.insert(name.as_str(), second);
        }
    }

    if let Some(header) = &header {
        for (index, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() {
        return false;
    }
    if bytes[start] == b'0' {
        return start + 1 == bytes.len();
    }
    (b'1'..=b'9').contains(&bytes[start])
        && bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.len() < 4 {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if bytes.len() < start + 4 {
        return false;
    }
    let dot = bytes.len() - 3;
    if bytes[dot] != b'.'
        || !bytes[dot + 1..].iter().all(u8::is_ascii_digit)
        || dot < start + 1
    {
        return false;
    }
    if bytes[start] == b'0' {
        if dot != start + 1 {
            return false;
        }
    } else if !(b'1'..=b'9').contains(&bytes[start])
        || !bytes[start + 1..dot].iter().all(u8::is_ascii_digit)
    {
        return false;
    }
    !(bytes[0] == b'-' && bytes[start] == b'0' && bytes[dot + 1..].iter().all(|&b| b == b'0'))
}

fn eval_cell(cell: &str, kind: ColumnType, nullable: bool) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if nullable {
            return Ok(Value::Null);
        }
        if kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }
    match kind {
        ColumnType::Text => Ok(Value::Text(cell.to_owned())),
        ColumnType::Integer => {
            if !integer_syntax(cell) {
                return Err(CellErrorKind::InvalidInteger);
            }
            cell.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(cell) {
                return Err(CellErrorKind::InvalidDecimal);
            }
            let negative = cell.starts_with('-');
            let unsigned = if negative { &cell[1..] } else { cell };
            let mut digits = unsigned.replace('.', "");
            if negative {
                digits.insert(0, '-');
            }
            digits
                .parse::<i64>()
                .map(Value::Decimal)
                .map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut had_error = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if policy == ErrorPolicy::FirstPerRow && had_error {
                break;
            }
            if policy == ErrorPolicy::Stop && !result.errors.is_empty() {
                break;
            }

            let (column_index, cell) = match &spec.column {
                ColumnRef::Index(index) => (*index, row.get(*index)),
                ColumnRef::Name(name) => match table.column_index(name) {
                    Some(index) => (index, row.get(index)),
                    None => {
                        result.errors.push(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: None,
                            kind: CellErrorKind::UnknownColumn,
                        });
                        had_error = true;
                        if policy == ErrorPolicy::Stop {
                            break;
                        }
                        continue;
                    }
                },
            };

            let Some(cell) = cell else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column_index),
                    kind: CellErrorKind::MissingCell,
                });
                had_error = true;
                if policy == ErrorPolicy::Stop {
                    break;
                }
                continue;
            };

            match eval_cell(cell, spec.kind, spec.nullable) {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind,
                    });
                    had_error = true;
                    if policy == ErrorPolicy::Stop {
                        break;
                    }
                }
            }

            if policy == ErrorPolicy::FirstPerRow && had_error {
                break;
            }
        }

        result.rows.push(output);
        if policy == ErrorPolicy::Stop && !result.errors.is_empty() {
            break;
        }
    }

    result
}

fn parse_configured(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    let mut records = parse_records(input, config.delimiter, config.quote)?;

    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(header) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: Vec<(&str, usize)> = Vec::new();

                return Err(CsvError::HeaderWidth {
                    record: data_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,

        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut record_started = false;

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        let outside = !matches!(state, State::Quoted { .. });
        if outside && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::FieldStart => {
                if b == delimiter {
                    row.push(String::new());
                    record_started = true;
                    i += 1;
                } else if b == quote {
                    record_started = true;
                    state = State::Quoted { opening: i };
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(String::new());
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    field.push(b as char);
                    record_started = true;
                    state = State::Unquoted;
                    i += 1;
                }
            }
            State::Unquoted => {
                if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
            }
            State::Quoted { opening } => {
                if b == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                    let _ = opening;
                }
            }
            State::AfterQuote => {
                if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    match state {
        State::Quoted { opening } => {
            return Err(CsvError::UnclosedQuote {
                record: record_index,
                byte: opening,
            });
        }
        State::AfterQuote => {
            row.push(field);
            records.push(row);
        }
        State::Unquoted => {
            row.push(field);
            records.push(row);
        }
        State::FieldStart => {
            if record_started || !row.is_empty() {
                row.push(String::new());
                records.push(row);
            }
        }
    }

    Ok(records)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef {
    Index(usize),
    Name(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {
    Text,
    Integer,
    Decimal,
    Boolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut out_row = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed && policy == ErrorPolicy::FirstPerRow {
                out_row.push(None);
                continue;
            }
            if policy == ErrorPolicy::Stop && !output.errors.is_empty() {
                out_row.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let result = match resolved {
                None => Err((None, CellErrorKind::UnknownColumn)),
                Some(index) => match row.get(index) {
                    None => Err((Some(index), CellErrorKind::MissingCell)),
                    Some(value) if value.is_empty() && spec.nullable => {
                        Ok(Value::Null)
                    }
                    Some(value) if value.is_empty() && spec.kind != ColumnType::Text => {
                        Err((Some(index), CellErrorKind::Empty))
                    }
                    Some(value) => match parse_typed(value, spec.kind) {
                        Ok(value) => Ok(value),
                        Err(kind) => Err((Some(index), kind)),
                    },
                },
            };

            match result {
                Ok(value) => out_row.push(Some(value)),
                Err((column, kind)) => {
                    out_row.push(None);
                    output.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column,
                        kind,
                    });
                    row_failed = true;
                }
            }
        }

        output.rows.push(out_row);
        if policy == ErrorPolicy::Stop && !output.errors.is_empty() {
            break;
        }
    }

    output
}

fn parse_typed(value: &str, kind: ColumnType) -> Result<Value, CellErrorKind> {
    match kind {
        ColumnType::Text => Ok(Value::Text(value.to_string())),
        ColumnType::Integer => parse_integer(value).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(value).map(Value::Decimal),
        ColumnType::Boolean => match value {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

fn parse_integer(value: &str) -> Result<i64, CellErrorKind> {
    let bytes = value.as_bytes();
    let mut start = 0;
    if bytes.first() == Some(&b'-') {
        start = 1;
    }
    let valid = if start >= bytes.len() {
        false
    } else if bytes[start] == b'0' {
        start == 0 && bytes.len() == 1
    } else {
        (b'1'..=b'9').contains(&bytes[start])
            && bytes[start + 1..].iter().all(u8::is_ascii_digit)
    };
    if !valid {
        return Err(CellErrorKind::InvalidInteger);
    }
    value
        .parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(value: &str) -> Result<i64, CellErrorKind> {
    let bytes = value.as_bytes();
    let mut pos = usize::from(bytes.first() == Some(&b'-'));
    let integral_start = pos;
    while pos < bytes.len() && bytes[pos].is_ascii_digit() {
        pos += 1;
    }

    let integral = &bytes[integral_start..pos];
    let integral_valid = if integral == b"0" {
        true
    } else {
        !integral.is_empty()
            && (b'1'..=b'9').contains(&integral[0])
            && integral[1..].iter().all(u8::is_ascii_digit)
    };

    let valid = integral_valid
        && bytes.get(pos) == Some(&b'.')
        && bytes.len() == pos + 3
        && bytes[pos + 1..].iter().all(u8::is_ascii_digit)
        && !(bytes.first() == Some(&b'-')
            && integral == b"0"
            && bytes.get(pos + 1) == Some(&b'0')
            && bytes.get(pos + 2) == Some(&b'0'));

    if !valid {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = bytes.first() == Some(&b'-');
    let whole: i128 = std::str::from_utf8(integral)
        .unwrap()
        .parse()
        .unwrap();
    let frac = ((bytes[pos + 1] - b'0') as i128) * 10
        + (bytes[pos + 2] - b'0') as i128;
    let magnitude = whole * 100 + frac;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}
