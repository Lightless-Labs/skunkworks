#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        time: u64,
        stage: OverflowStage,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = remainder > half || (denominator % 2 == 0 && remainder == half && quotient % 2 == 1);
    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct CustomerState {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    queue_since: u64,
    waited: u64,
    has_started: bool,
    paused: bool,
    pause_time: u64,
    terminal: bool,
    accepted: bool,
}

#[derive(Clone)]
struct Running {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch_time: u64,
}

fn priority_cmp(a: &CustomerState, b: &CustomerState) -> std::cmp::Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.paused, b.paused) {
            (true, false) => std::cmp::Ordering::Less,
            (false, true) => std::cmp::Ordering::Greater,
            (true, true) => a
                .pause_time
                .cmp(&b.pause_time)
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (false, false) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event {
        time,
        customer,
        kind,
    });
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let n = customers.len();
    let mut states: Vec<CustomerState> = customers
        .iter()
        .cloned()
        .map(|customer| CustomerState {
            deadline: customer.patience.map(|p| customer.arrival + p),
            remaining: customer.service,
            queue_since: customer.arrival,
            waited: 0,
            has_started: false,
            paused: false,
            pause_time: 0,
            terminal: false,
            accepted: false,
            customer,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..n).collect();
    arrivals.sort_by(|&a, &b| {
        states[a]
            .customer
            .arrival
            .cmp(&states[b].customer.arrival)
            .then_with(|| states[a].customer.priority.cmp(&states[b].customer.priority))
            .then_with(|| states[b].customer.id.cmp(&states[a].customer.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut now = 0u64;

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();

    loop {
        let mut did_phase_at_time = false;

        if let Some(active) = running.as_ref() {
            if active.finish == now {
                let active = running.take().unwrap();
                let id = states[active.index].customer.id;
                emit(&mut events, now, id, Kind::Finish);
                states[active.index].terminal = true;
                completed += 1;
                completion_order.push(id);
                turnaround_ticks += (now - states[active.index].customer.arrival) as u128;
                did_phase_at_time = true;
            }
        }

        let mut expired: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
        });
        for i in expired {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - states[i].queue_since;
                states[i].waited += segment;
                wait_ticks += segment as u128;
                states[i].terminal = true;
                reneged += 1;
                emit(
                    &mut events,
                    now,
                    states[i].customer.id,
                    Kind::Renege {
                        waited: states[i].waited,
                    },
                );
                did_phase_at_time = true;
            }
        }

        while next_arrival < n && states[arrivals[next_arrival]].customer.arrival == now {
            let i = arrivals[next_arrival];
            next_arrival += 1;
            let id = states[i].customer.id;
            emit(&mut events, now, id, Kind::Arrive);
            did_phase_at_time = true;

            let server_idle = running.is_none();
            let gate = config.waiting_capacity + usize::from(server_idle);
            if waiting.len() < gate {
                states[i].accepted = true;
                states[i].queue_since = now;
                waiting.push(i);
                accepted += 1;
            } else {
                balked += 1;
                emit(&mut events, now, id, Kind::Balk);
            }
        }

        if let Some(active) = running.as_ref() {
            if now >= active.work_at {
                let best = waiting.iter().copied().min_by(|&a, &b| {
                    priority_cmp(&states[a], &states[b])
                });
                if let Some(best_i) = best {
                    if states[best_i].customer.priority < states[active.index].customer.priority {
                        let active = running.take().unwrap();
                        let remaining = active.finish - now;
                        let paused_id = states[active.index].customer.id;
                        let by = states[best_i].customer.id;
                        emit(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt { remaining, by },
                        );
                        preemptions += 1;
                        busy_ticks += (now - active.dispatch_time) as u128;

                        states[active.index].remaining = remaining;
                        states[active.index].queue_since = now;
                        states[active.index].pause_time = now;
                        states[active.index].paused = true;
                        if let Some(budget) = states[active.index].customer.patience {
                            let remaining_budget = budget.saturating_sub(states[active.index].waited);
                            states[active.index].deadline = Some(
                                now.checked_add(remaining_budget).ok_or(Error::TimeOverflow {
                                    customer: paused_id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                })?,
                            );
                        }
                        waiting.push(active.index);
                        did_phase_at_time = true;
                    }
                }
            }
        }

        loop {
            if running.is_some() || waiting.is_empty() {
                break;
            }
            let selected = waiting
                .iter()
                .copied()
                .min_by(|&a, &b| priority_cmp(&states[a], &states[b]))
                .unwrap();
            waiting.retain(|&x| x != selected);

            let segment = now - states[selected].queue_since;
            states[selected].waited += segment;
            wait_ticks += segment as u128;
            if let Some(budget) = states[selected].customer.patience {
                let left = budget.saturating_sub(segment);
                states[selected].customer.patience = Some(left);
            }

            let id = states[selected].customer.id;
            let switch = if last_dispatched.is_some_and(|last| last != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(states[selected].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

            let is_first = !states[selected].has_started;
            let kind = if is_first {
                started += 1;
                states[selected].has_started = true;
                Kind::Start {
                    remaining: states[selected].remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Resume {
                    remaining: states[selected].remaining,
                    work_at,
                    finish,
                }
            };
            emit(&mut events, now, id, kind);
            last_dispatched = Some(id);
            running = Some(Running {
                index: selected,
                work_at,
                finish,
                dispatch_time: now,
            });
            did_phase_at_time = true;

            if finish == now {
                let active = running.take().unwrap();
                emit(&mut events, now, id, Kind::Finish);
                states[active.index].terminal = true;
                completed += 1;
                completion_order.push(id);
                turnaround_ticks += (now - states[active.index].customer.arrival) as u128;
                busy_ticks += (now - active.dispatch_time) as u128;
                continue;
            }
        }

        let mut expired_f: Vec<usize> = waiting
            .iter()
            .copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired_f.sort_by(|&a, &b| {
            states[a]
                .deadline
                .cmp(&states[b].deadline)
                .then_with(|| states[a].customer.id.cmp(&states[b].customer.id))
        });
        for i in expired_f {
            if let Some(pos) = waiting.iter().position(|&x| x == i) {
                waiting.remove(pos);
                let segment = now - states[i].queue_since;
                states[i].waited += segment;
                wait_ticks += segment as u128;
                states[i].terminal = true;
                reneged += 1;
                emit(
                    &mut events,
                    now,
                    states[i].customer.id,
                    Kind::Renege {
                        waited: states[i].waited,
                    },
                );
                did_phase_at_time = true;
            }
        }

        if next_arrival == n && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut next = None::<u64>;
        let mut consider = |time: u64| {
            if time > now && next.is_none_or(|old| time < old) {
                next = Some(time);
            }
        };
        if next_arrival < n {
            consider(states[arrivals[next_arrival]].customer.arrival);
        }
        if let Some(active) = running.as_ref() {
            consider(active.finish);
            if now < active.work_at {
                consider(active.work_at);
            }
        }
        for &i in &waiting {
            if let Some(deadline) = states[i].deadline {
                consider(deadline);
            }
        }
        if let Some(t) = next {
            now = t;
        } else if !did_phase_at_time {
            break;
        } else if now == u64::MAX {
            break;
        }
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10_000, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}
