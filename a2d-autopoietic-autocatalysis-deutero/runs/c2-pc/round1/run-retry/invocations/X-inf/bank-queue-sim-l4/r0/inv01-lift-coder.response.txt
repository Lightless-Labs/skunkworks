#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };
    Some(quotient + u128::from(round_up))
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    remaining: u64,
    started: bool,
    paused_at: Option<u64>,
    queue_since: u64,
    budget: Option<u64>,
    deadline: Option<u64>,
    waited: u64,
}

#[derive(Clone)]
struct Active {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn job_better(a: &Job, b: &Job) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }
    match (a.paused_at, b.paused_at) {
        (Some(x), Some(y)) => {
            if x != y { x < y } else { a.customer.id < b.customer.id }
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            if a.customer.arrival != b.customer.arrival {
                a.customer.arrival < b.customer.arrival
            } else {
                a.customer.id > b.customer.id
            }
        }
    }
}

fn time_error(id: u32, time: u64, stage: OverflowStage) -> Error {
    Error::TimeOverflow { customer: id, time, stage }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a].arrival.cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut first_dispatch = true;

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();

    let emit = |events: &mut Vec<Event>, time, customer, kind| {
        events.push(Event { time, customer, kind });
    };

    loop {
        if active.is_none() && waiting.is_empty() && next_arrival >= order.len() {
            break;
        }

        if active.is_none() && waiting.is_empty() {
            if next_arrival < order.len() {
                now = now.max(customers[order[next_arrival]].arrival);
            }
        }

        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let id = a.job.customer.id;
                let arrival = a.job.customer.arrival;
                active = None;
                emit(&mut events, now, id, Kind::Finish);
                completed += 1;
                turnaround_ticks += (now - arrival) as u128;
                completion_order.push(id);
            }
        }

        let mut expiring: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, j)| j.deadline.filter(|&d| d <= now).map(|d| (i, d)))
            .map(|(i, _)| i)
            .collect();
        expiring.sort_by(|&a, &b| {
            waiting[a].deadline.cmp(&waiting[b].deadline)
                .then(waiting[a].customer.id.cmp(&waiting[b].customer.id))
        });
        for i in expiring.into_iter().rev() {
            let job = waiting.remove(i);
            let segment = now - job.queue_since;
            let total_waited = job.waited + segment;
            wait_ticks += segment as u128;
            emit(&mut events, now, job.customer.id, Kind::Renege { waited: total_waited });
            reneged += 1;
        }

        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let c = customers[order[next_arrival]].clone();
            next_arrival += 1;
            emit(&mut events, now, c.id, Kind::Arrive);

            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                emit(&mut events, now, c.id, Kind::Balk);
                balked += 1;
            } else {
                accepted += 1;
                let deadline = c.patience.map(|p| c.arrival + p);
                waiting.push(Job {
                    remaining: c.service,
                    started: false,
                    paused_at: None,
                    queue_since: now,
                    budget: c.patience,
                    deadline,
                    waited: 0,
                    customer: c,
                });
            }
        }

        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                let best = waiting.iter().enumerate()
                    .min_by(|(_, x), (_, y)| if job_better(x, y) { std::cmp::Ordering::Less } else if job_better(y, x) { std::cmp::Ordering::Greater } else { std::cmp::Ordering::Equal })
                    .map(|(i, _)| i);
                if let Some(i) = best {
                    if waiting[i].customer.priority < a.job.customer.priority {
                        let remaining = a.finish - now;
                        let by = waiting[i].customer.id;
                        let mut paused = active.take().unwrap().job;
                        emit(&mut events, now, paused.customer.id, Kind::Preempt { remaining, by });
                        preemptions += 1;
                        busy_ticks += (now - (paused.queue_since - (paused.queue_since - paused.queue_since))) as u128;
                        paused.remaining = remaining;
                        paused.paused_at = Some(now);
                        paused.queue_since = now;
                        if let Some(budget) = paused.budget {
                            paused.deadline = Some(now.checked_add(budget).ok_or_else(|| time_error(paused.customer.id, now, OverflowStage::Patience))?);
                        }
                        waiting.push(paused);
                    }
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let best = waiting.iter().enumerate()
                .min_by(|(_, x), (_, y)| if job_better(x, y) { std::cmp::Ordering::Less } else if job_better(y, x) { std::cmp::Ordering::Greater } else { std::cmp::Ordering::Equal })
                .map(|(i, _)| i);
            let Some(i) = best else { break };
            let mut job = waiting.remove(i);
            let segment = now - job.queue_since;
            wait_ticks += segment as u128;
            job.waited += segment;
            if let Some(budget) = job.budget {
                job.budget = Some(budget - segment);
            }

            let switching = if first_dispatch {
                0
            } else if last_dispatch != Some(job.customer.id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switching)
                .ok_or_else(|| time_error(job.customer.id, now, OverflowStage::WorkAt))?;
            let finish = work_at.checked_add(job.remaining)
                .ok_or_else(|| time_error(job.customer.id, now, OverflowStage::Finish))?;

            let kind = if job.started {
                Kind::Resume { remaining: job.remaining, work_at, finish }
            } else {
                job.started = true;
                started += 1;
                Kind::Start { remaining: job.remaining, work_at, finish }
            };
            emit(&mut events, now, job.customer.id, kind);
            first_dispatch = false;
            last_dispatch = Some(job.customer.id);
            busy_ticks += switching as u128;

            if finish == now {
                emit(&mut events, now, job.customer.id, Kind::Finish);
                completed += 1;
                turnaround_ticks += (now - job.customer.arrival) as u128;
                completion_order.push(job.customer.id);
            } else {
                active = Some(Active { job, work_at, finish });
            }
        }

        let mut expiring: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, j)| j.deadline.filter(|&d| d <= now).map(|d| (i, d)))
            .map(|(i, _)| i)
            .collect();
        expiring.sort_by(|&a, &b| {
            waiting[a].deadline.cmp(&waiting[b].deadline)
                .then(waiting[a].customer.id.cmp(&waiting[b].customer.id))
        });
        for i in expiring.into_iter().rev() {
            let job = waiting.remove(i);
            let segment = now - job.queue_since;
            let total_waited = job.waited + segment;
            wait_ticks += segment as u128;
            emit(&mut events, now, job.customer.id, Kind::Renege { waited: total_waited });
            reneged += 1;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < order.len() {
            next_time = Some(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = active.as_ref() {
            next_time = Some(next_time.map_or(a.finish, |t| t.min(a.finish)));
            if a.work_at > now {
                next_time = Some(next_time.map_or(a.work_at, |t| t.min(a.work_at)));
            }
        }
        for job in &waiting {
            if let Some(d) = job.deadline {
                if d > now {
                    next_time = Some(next_time.map_or(d, |t| t.min(d)));
                }
            }
        }
        if let Some(t) = next_time {
            if t < now {
                break;
            }
            if let Some(a) = active.as_ref() {
                if t <= a.finish {
                    busy_ticks += (t - now) as u128;
                }
            }
            now = t;
        } else {
            break;
        }
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 { 0 } else { rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap() };
    let mean_turnaround_milli = if completed == 0 { 0 } else { rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap() };
    let utilization_bps = if elapsed == 0 { 0 } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128).unwrap().min(u32::MAX as u128) as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted, balked, reneged, completed, started, preemptions, elapsed,
            busy_ticks, wait_ticks, turnaround_ticks, mean_wait_milli,
            mean_turnaround_milli, utilization_bps, completion_order,
        },
    })
}
