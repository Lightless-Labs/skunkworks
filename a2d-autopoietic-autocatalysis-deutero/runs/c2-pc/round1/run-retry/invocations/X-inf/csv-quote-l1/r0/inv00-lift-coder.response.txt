#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut start = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = State::FieldStart;
    let mut quote_start = 0;
    let mut record_index = 0;

    while start < bytes.len() {
        let byte = bytes[start];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: start,
            });
        }

        if state != State::Quoted && byte == b'\r' {
            if bytes.get(start + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: start,
                });
            }
            finish_field(&mut record, &mut field);
            records.push(std::mem::take(&mut record));
            record_index += 1;
            start += 2;
            state = State::FieldStart;
            continue;
        }

        match state {
            State::Quoted => {
                if byte == b'"' {
                    if bytes.get(start + 1) == Some(&b'"') {
                        field.push('"');
                        start += 2;
                    } else {
                        state = State::AfterQuote;
                        start += 1;
                    }
                } else {
                    let ch = input[start..].chars().next().unwrap();
                    field.push(ch);
                    start += ch.len_utf8();
                }
            }
            State::AfterQuote => match byte {
                b',' => {
                    finish_field(&mut record, &mut field);
                    state = State::FieldStart;
                    start += 1;
                }
                b'\n' => {
                    finish_field(&mut record, &mut field);
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    start += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: start,
                    });
                }
            },
            State::FieldStart => match byte {
                b'"' => {
                    quote_start = start;
                    state = State::Quoted;
                    start += 1;
                }
                b',' => {
                    finish_field(&mut record, &mut field);
                    start += 1;
                }
                b'\n' => {
                    finish_field(&mut record, &mut field);
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    start += 1;
                }
                _ => {
                    let ch = input[start..].chars().next().unwrap();
                    field.push(ch);
                    state = State::Unquoted;
                    start += ch.len_utf8();
                }
            },
            State::Unquoted => match byte {
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: start,
                    });
                }
                b',' => {
                    finish_field(&mut record, &mut field);
                    state = State::FieldStart;
                    start += 1;
                }
                b'\n' => {
                    finish_field(&mut record, &mut field);
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = State::FieldStart;
                    start += 1;
                }
                _ => {
                    let ch = input[start..].chars().next().unwrap();
                    field.push(ch);
                    start += ch.len_utf8();
                }
            },
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if state != State::FieldStart || !record.is_empty() {
        finish_field(&mut record, &mut field);
        records.push(record);
    }

    Ok(records)
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum State {
    FieldStart,
    Unquoted,
    Quoted,
    AfterQuote,
}

fn finish_field(record: &mut Vec<String>, field: &mut String) {
    record.push(std::mem::take(field));
}
