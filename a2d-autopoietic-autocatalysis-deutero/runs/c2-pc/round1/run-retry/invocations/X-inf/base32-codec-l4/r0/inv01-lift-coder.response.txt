#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = Vec::with_capacity(self.pending.len() + input.len());
        combined.extend_from_slice(&self.pending);
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 5 * 5;
        let mut output = String::new();
        let alphabet = alphabet_bytes(self.config.alphabet);

        for chunk in combined[..complete_len].chunks_exact(5) {
            encode_symbols(chunk, 8, alphabet, &mut output);
        }

        self.pending.clear();
        self.pending.extend_from_slice(&combined[complete_len..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        let alphabet = alphabet_bytes(self.config.alphabet);
        if !self.pending.is_empty() {
            let symbol_count = (self.pending.len() * 8 + 4) / 5;
            encode_symbols(&self.pending, symbol_count, alphabet, &mut output);
            if self.config.padding == Padding::Required {
                while output.len() % 8 != 0 {
                    output.push('=');
                }
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_symbols(input: &[u8], count: usize, alphabet: &[u8; 32], output: &mut String) {
    for symbol_index in 0..count {
        let bit_start = symbol_index * 5;
        let mut value = 0u8;
        for bit_offset in 0..5 {
            value <<= 1;
            let bit_index = bit_start + bit_offset;
            if bit_index < input.len() * 8 {
                let byte = input[bit_index / 8];
                value |= (byte >> (7 - bit_index % 8)) & 1;
            }
        }
        output.push(alphabet[value as usize] as char);
    }
}

pub struct Decoder {
    config: Config,
    consumed: usize,
    data_count: usize,
    accumulator: u8,
    pending_bits: u8,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            consumed: 0,
            data_count: 0,
            accumulator: 0,
            pending_bits: 0,
            last_data_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.consumed;
            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                if next.first_padding_offset.is_none() {
                    let residue = next.data_count % 8;
                    let expected = match residue {
                        2 => 6,
                        4 => 4,
                        5 => 3,
                        7 => 1,
                        _ => return Err(StreamError::Padding { offset }),
                    };
                    next.first_padding_offset = Some(offset);
                    next.expected_padding = expected;
                }

                if next.padding_count >= next.expected_padding {
                    return Err(StreamError::Padding { offset });
                }
                next.padding_count += 1;
                next.consumed += 1;
                continue;
            }

            let original = byte;
            let upper = byte.to_ascii_uppercase();
            let value = match symbol_value(next.config.alphabet, upper) {
                Some(value) => value,
                None => {
                    return Err(StreamError::InvalidByte {
                        offset,
                        byte: original,
                    });
                }
            };

            if next.config.canonical && original.is_ascii_lowercase() {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: upper,
                });
            }

            if next.first_padding_offset.is_some() {
                return Err(StreamError::Padding { offset });
            }

            next.data_count += 1;
            next.last_data_offset = Some(offset);
            next.accumulator = (next.accumulator << 5) | value;
            next.pending_bits += 5;
            if next.pending_bits >= 8 {
                next.pending_bits -= 8;
                output.push(next.accumulator >> next.pending_bits);
                if next.pending_bits == 0 {
                    next.accumulator = 0;
                } else {
                    next.accumulator &= (1u8 << next.pending_bits) - 1;
                }
            }
            next.consumed += 1;
        }

        self.assign_state(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => {
                return Err(StreamError::InvalidLength {
                    offset: self.consumed,
                });
            }
        };

        if self.config.padding == Padding::Required
            && self.padding_count != expected_padding
        {
            return Err(StreamError::Padding {
                offset: self.first_padding_offset.unwrap_or(self.consumed),
            });
        }

        let unused = match residue {
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused != 0 && self.accumulator != 0 {
            return Err(StreamError::TrailingBits {
                offset: self
                    .last_data_offset
                    .unwrap_or(self.consumed.saturating_sub(1)),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> DecoderState {
        DecoderState {
            config: self.config,
            consumed: self.consumed,
            data_count: self.data_count,
            accumulator: self.accumulator,
            pending_bits: self.pending_bits,
            last_data_offset: self.last_data_offset,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
        }
    }

    fn assign_state(&mut self, state: DecoderState) {
        self.consumed = state.consumed;
        self.data_count = state.data_count;
        self.accumulator = state.accumulator;
        self.pending_bits = state.pending_bits;
        self.last_data_offset = state.last_data_offset;
        self.first_padding_offset = state.first_padding_offset;
        self.padding_count = state.padding_count;
        self.expected_padding = state.expected_padding;
    }
}

struct DecoderState {
    config: Config,
    consumed: usize,
    data_count: usize,
    accumulator: u8,
    pending_bits: u8,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: usize,
}

fn symbol_value(alphabet: Alphabet, byte: u8) -> Option<u8> {
    let alphabet = alphabet_bytes(alphabet);
    alphabet.iter().position(|&candidate| candidate == byte).map(|value| value as u8)
}
