#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = Vec::with_capacity(self.pending.len() + input.len());
        combined.extend_from_slice(&self.pending);
        combined.extend_from_slice(input);

        let block_bytes = (combined.len() / 5) * 5;
        let output = encode_bytes(self.config.alphabet, &combined[..block_bytes], false, self.config.padding);
        self.pending.clear();
        self.pending.extend_from_slice(&combined[block_bytes..]);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let output = encode_bytes(self.config.alphabet, &self.pending, true, self.config.padding);
        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_symbol(alphabet: Alphabet, value: u8) -> u8 {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"[value as usize],
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV"[value as usize],
    }
}

fn encode_bytes(alphabet: Alphabet, input: &[u8], final_fragment: bool, padding: Padding) -> String {
    if input.is_empty() {
        return String::new();
    }

    let symbol_count = (input.len() * 8 + 4) / 5;
    let mut output = String::with_capacity(if final_fragment && padding == Padding::Required {
        ((symbol_count + 7) / 8) * 8
    } else {
        symbol_count
    });

    let mut acc: u32 = 0;
    let mut bits = 0u8;
    for &byte in input {
        acc = (acc << 8) | u32::from(byte);
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            output.push(alphabet_symbol(alphabet, ((acc >> bits) & 31) as u8) as char);
        }
    }

    if bits > 0 {
        output.push(alphabet_symbol(alphabet, ((acc << (5 - bits)) & 31) as u8) as char);
    }

    if final_fragment && padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }
    output
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    pending_bits: u8,
    pending_value: u8,
    last_symbol_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            pending_bits: 0,
            pending_value: 0,
            last_symbol_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: None,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }
                let residue = next.data_count % 8;
                let expected = match residue {
                    0 => return Err(StreamError::Padding { offset }),
                    2 => 6,
                    4 => 4,
                    5 => 3,
                    7 => 1,
                    _ => return Err(StreamError::Padding { offset }),
                };
                if let Some(first_expected) = next.expected_padding {
                    if next.padding_count >= first_expected {
                        return Err(StreamError::Padding { offset });
                    }
                } else {
                    next.expected_padding = Some(expected);
                    next.first_padding_offset = Some(offset);
                }
                next.padding_count += 1;
                next.position += 1;
                continue;
            }

            let upper = byte.to_ascii_uppercase();
            let value = symbol_value(next.config.alphabet, upper)
                .ok_or(StreamError::InvalidByte { offset, byte })?;

            if next.config.canonical && byte != upper {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: upper,
                });
            }
            if next.first_padding_offset.is_some() {
                return Err(StreamError::Padding { offset });
            }

            next.data_count += 1;
            next.last_symbol_offset = Some(offset);
            next.pending_value = (next.pending_value << 5) | value;
            next.pending_bits += 5;
            while next.pending_bits >= 8 {
                next.pending_bits -= 8;
                output.push((next.pending_value >> next.pending_bits) as u8);
                next.pending_value &= (1u8 << next.pending_bits).wrapping_sub(1);
            }
            next.position += 1;
        }

        *self = next;
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        }

        if self.config.padding == Padding::Required {
            let expected = match residue {
                0 => 0,
                2 => 6,
                4 => 4,
                5 => 3,
                7 => 1,
                _ => unreachable!(),
            };
            if self.padding_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_padding_offset.unwrap_or(self.position),
                });
            }
        }

        let unused_bits = match residue {
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused_bits != 0 && (self.pending_value & ((1u8 << unused_bits) - 1)) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_symbol_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            pending_bits: self.pending_bits,
            pending_value: self.pending_value,
            last_symbol_offset: self.last_symbol_offset,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        }
    }
}

fn symbol_value(alphabet: Alphabet, byte: u8) -> Option<u8> {
    match alphabet {
        Alphabet::Standard => match byte {
            b'A'..=b'Z' => Some(byte - b'A'),
            b'2'..=b'7' => Some(byte - b'2' + 26),
            _ => None,
        },
        Alphabet::Hex => match byte {
            b'0'..=b'9' => Some(byte - b'0'),
            b'A'..=b'V' => Some(byte - b'A' + 10),
            _ => None,
        },
    }
}
