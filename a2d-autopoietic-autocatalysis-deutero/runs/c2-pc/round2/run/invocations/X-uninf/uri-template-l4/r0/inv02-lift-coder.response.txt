#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Debug, Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continue,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Query | Self::Continue => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Simple | Self::Reserved => None,
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continue => Some('&'),
        }
    }

    fn is_named(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continue)
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Operator,
    specs: Vec<Specifier>,
}

fn parse_expression(content: &str, start: usize, close: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, offset) = if let Some(&b) = bytes.first() {
        match b {
            b'+' => (Operator::Reserved, 1),
            b'#' => (Operator::Fragment, 1),
            b'.' => (Operator::Dot, 1),
            b'/' => (Operator::Slash, 1),
            b';' => (Operator::Semicolon, 1),
            b'?' => (Operator::Query, 1),
            b'&' => (Operator::Continue, 1),
            _ => (Operator::Simple, 0),
        }
    } else {
        (Operator::Simple, 0)
    };

    let rest = &content[offset..];
    if rest.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut entry_start = 0;

    for entry in rest.split(',') {
        if entry.is_empty() {
            return Err(Error::EmptyVariable {
                pos: start + offset + entry_start,
            });
        }

        let absolute_start = start + offset + entry_start;
        let entry_bytes = entry.as_bytes();
        let modifier_index = entry_bytes
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .unwrap_or(entry_bytes.len());
        let name = &entry[..modifier_index];

        if name.is_empty() {
            return Err(Error::InvalidName { pos: absolute_start });
        }

        let name_bytes = name.as_bytes();
        let first = name_bytes[0];
        if !first.is_ascii_alphabetic() && first != b'_' {
            return Err(Error::InvalidName { pos: absolute_start });
        }
        for (i, &b) in name_bytes.iter().enumerate().skip(1) {
            if !b.is_ascii_alphanumeric() && b != b'_' {
                return Err(Error::InvalidName {
                    pos: absolute_start + i,
                });
            }
        }

        let suffix = &entry[modifier_index..];
        let modifier = if suffix.is_empty() {
            Modifier::None
        } else if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let digit_bytes = digits.as_bytes();
            let valid = (1..=4).contains(&digit_bytes.len())
                && digit_bytes[0].is_ascii_digit()
                && digit_bytes[0] != b'0'
                && digit_bytes.iter().all(u8::is_ascii_digit);
            if !valid {
                return Err(Error::InvalidModifier {
                    pos: absolute_start + modifier_index,
                });
            }
            let value = digits.parse::<usize>().unwrap_or(0);
            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier {
                    pos: absolute_start + modifier_index,
                });
            }
            Modifier::Prefix(value)
        } else {
            return Err(Error::InvalidModifier {
                pos: absolute_start + modifier_index,
            });
        };

        if seen.iter().any(|previous| previous == name) {
            return Err(Error::DuplicateVariable { pos: absolute_start });
        }
        seen.push(name.to_owned());
        specs.push(Specifier {
            name: name.to_owned(),
            start: absolute_start,
            modifier,
        });

        entry_start += entry.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Option<Expression>)>, Error> {
    let bytes = template.as_bytes();
    let mut spans = Vec::new();
    let mut literal_start = 0;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;
                if let Some(relative_nested) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative_nested,
                    });
                }
                spans.push((literal_start, open, None));
                let expression = parse_expression(&template[open + 1..close], open + 1, close)?;
                spans.push((open, close + 1, Some(expression)));
                i = close + 1;
                literal_start = i;
            }
            _ => i += 1,
        }
    }

    spans.push((literal_start, template.len(), None));
    Ok(spans)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            result.push('%');
            result.push((bytes[i + 1] as char).to_ascii_uppercase());
            result.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            result.push(byte as char);
            i += 1;
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0F) as usize] as char);
            i += 1;
        }
    }

    result
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn scalar_field(name: &str, value: &str, operator: Operator) -> String {
    let encoded = encode(value, operator.reserved());
    if operator.is_named() {
        match operator {
            Operator::Semicolon if encoded.is_empty() => name.to_owned(),
            _ => format!("{name}={encoded}"),
        }
    } else {
        encoded
    }
}

fn named_encoded_field(name: &str, encoded: &str, operator: Operator) -> String {
    if operator.is_named() {
        match operator {
            Operator::Semicolon if encoded.is_empty() => name.to_owned(),
            _ => format!("{name}={encoded}"),
        }
    } else {
        encoded.to_owned()
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator;
    let mut fields = Vec::new();

    for spec in &expression.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(value) => {
                let truncated;
                let value = if let Modifier::Prefix(count) = spec.modifier {
                    truncated = value.chars().take(count).collect::<String>();
                    truncated.as_str()
                } else {
                    value.as_str()
                };
                fields.push(scalar_field(&spec.name, value, operator));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        fields.push(scalar_field(&spec.name, item, operator));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, operator.reserved()))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(named_encoded_field(&spec.name, &joined, operator));
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }
                for i in 0..entries.len() {
                    if entries[..i].iter().any(|(key, _)| key == &entries[i].0) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, operator.reserved());
                        let encoded_value = encode(value, operator.reserved());
                        let field = if matches!(operator, Operator::Semicolon)
                            && encoded_value.is_empty()
                        {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, operator.reserved()));
                        components.push(encode(value, operator.reserved()));
                    }
                    let joined = components.join(",");
                    fields.push(named_encoded_field(&spec.name, &joined, operator));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    if let Some(prefix) = operator.prefix() {
        output.push(prefix);
    }
    output.push_str(&fields.join(&operator.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let spans = parse_template(template)?;
    let mut output = String::new();

    for (start, end, expression) in spans {
        if let Some(expression) = expression {
            output.push_str(&expand_expression(&expression, vars)?);
        } else {
            output.push_str(&template[start..end]);
        }
    }

    Ok(output)
}
