#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression<'a> {
    operator: Option<char>,
    variables: Vec<Variable<'a>>,
}

#[derive(Debug, Clone)]
struct Variable<'a> {
    name: &'a str,
    pos: usize,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression<'_>, Error> {
    let content_start = open + 1;
    let content = &template[content_start..close];
    let bytes = content.as_bytes();

    let mut operator = None;
    let mut specs_start = content_start;

    if let Some(&first) = bytes.first() {
        let ch = first as char;
        if matches!(ch, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
            operator = Some(ch);
            specs_start += 1;
        }
    }

    if specs_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut used: Vec<&str> = Vec::new();
    let mut spec_start = specs_start;

    loop {
        let spec_end = template[spec_start..close]
            .find(',')
            .map(|relative| spec_start + relative)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name = &template[spec_start..spec_end];
        let name_bytes = name.as_bytes();

        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }

        for (index, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + index,
                });
            }
        }

        if used.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }

        used.push(name);
        variables.push(Variable {
            name,
            pos: spec_start,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' {
            if index + 2 < bytes.len()
                && bytes[index + 1].is_ascii_hexdigit()
                && bytes[index + 2].is_ascii_hexdigit()
            {
                output.push('%');
                output.push((bytes[index + 1] as char).to_ascii_uppercase());
                output.push((bytes[index + 2] as char).to_ascii_uppercase());
                index += 3;
                continue;
            }
            output.push_str("%25");
            index += 1;
            continue;
        }

        let unreserved =
            byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved = allow_reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    output
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|(binding_name, value)| (*binding_name == name).then_some(*value))
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let allow_reserved = matches!(operator, Some('+' | '#'));
    let mut fields = Vec::new();

    for variable in &expression.variables {
        let _ = variable.pos;
        if let Some(value) = lookup(vars, variable.name) {
            let encoded = encode_value(value, allow_reserved);
            let field = match operator {
                Some(';' | '?' | '&') => {
                    if operator == Some(';') && encoded.is_empty() {
                        variable.name.to_owned()
                    } else {
                        format!("{}={}", variable.name, encoded)
                    }
                }
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (separator, prefix) = match operator {
        Some('#') => (",", "#"),
        Some('.') => (".", "."),
        Some('/') => ("/", "/"),
        Some(';') => (";", ";"),
        Some('?') => ("&", "?"),
        Some('&') => ("&", "&"),
        _ => (",", ""),
    };

    let mut output = String::from(prefix);
    output.push_str(&fields.join(separator));
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let mut expressions = Vec::new();
    let mut index = 0;
    let bytes = template.as_bytes();

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let open = index;
                let Some(relative_close) = template[index + 1..].find('}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = index + 1 + relative_close;

                if let Some(relative_nested) = template[index + 1..close].find('{') {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + relative_nested,
                    });
                }

                let expression = parse_expression(template, open, close)?;
                expressions.push((open, close, expression));
                index = close + 1;
            }
            _ => {
                index += 1;
                while index < bytes.len() && bytes[index] != b'{' && bytes[index] != b'}' {
                    index += 1;
                }
            }
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    for (open, close, expression) in expressions {
        output.push_str(&template[cursor..open]);
        output.push_str(&expand_expression(&expression, vars));
        cursor = close + 1;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}
