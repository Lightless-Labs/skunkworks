#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

struct Entry {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    remaining: Option<u64>,
    started_at: u64,
}

pub struct Pool {
    entries: Vec<Entry>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_name(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job
                .tag
                .as_ref()
                .is_some_and(|tag| !valid_name(tag, 1, 8))
            {
                return Err(Error::BadTag(i));
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                let Some(&index) = ids.get(dependency) else {
                    return Err(Error::MissingDependency(i, j));
                };
                resolved.push(index);
            }
            dependencies.push(resolved);
        }

        fn visit(
            node: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return true;
            }
            if marks[node] == 2 {
                return false;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if visit(dep, deps, marks) {
                    return true;
                }
            }
            marks[node] = 2;
            false
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if visit(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Entry {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                dependencies: dependencies[i].clone(),
                remaining: None,
                started_at: 0,
            })
            .collect();

        Ok(Self {
            entries,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if self.has_run && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut time = if self.has_run {
            self.horizon
        } else {
            0
        };
        let mut first_phase = !self.has_run;

        loop {
            if first_phase || time > self.horizon || !self.has_run {
                self.process_time(time, &mut lines);
            } else if time > self.horizon {
                self.process_time(time, &mut lines);
            }

            if time == until {
                break;
            }

            let next = self.next_event(time);
            match next {
                Some(next_time) if next_time <= until => time = next_time,
                _ => break,
            }
            first_phase = false;
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.entries
            .iter()
            .find(|entry| entry.job.id == id)
            .map(|entry| entry.state.clone())
            .ok_or(Error::Unknown)
    }

    fn next_event(&self, time: u64) -> Option<u64> {
        let mut next = None;
        for entry in &self.entries {
            let candidate = match entry.state {
                State::Waiting { ready_at, .. } if ready_at > time => Some(ready_at),
                State::Running { finish_at, .. } if finish_at > time => Some(finish_at),
                _ => None,
            };
            if let Some(candidate) = candidate {
                next = Some(next.map_or(candidate, |current: u64| current.min(candidate)));
            }
        }
        next
    }

    fn process_time(&mut self, time: u64, lines: &mut Vec<String>) {
        let mut completions: Vec<usize> = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, entry)| match entry.state {
                State::Running { finish_at, .. } if finish_at == time => Some(i),
                _ => None,
            })
            .collect();
        completions.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

        for i in completions {
            let attempt = match self.entries[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let id = self.entries[i].job.id.clone();
            let success = attempt > self.entries[i].job.failures;
            if success {
                lines.push(format!("{time} finish {id} {attempt} ok"));
                self.entries[i].state = State::Succeeded;
                self.entries[i].remaining = None;
            } else {
                lines.push(format!("{time} finish {id} {attempt} fail"));
                self.entries[i].remaining = None;
                if attempt >= self.entries[i].job.max_attempts {
                    self.entries[i].state = State::Failed;
                } else {
                    let shift = u32::from(attempt - 1);
                    let delay = self
                        .retry_base
                        .checked_mul(1u64 << shift)
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready_at = time.saturating_add(delay);
                    lines.push(format!("{time} retry {id} {attempt} {ready_at}"));
                    self.entries[i].state = State::Waiting {
                        ready_at,
                        attempt,
                    };
                }
            }
        }

        loop {
            let mut newly_blocked = Vec::new();
            for i in 0..self.entries.len() {
                if !matches!(self.entries[i].state, State::Waiting { .. }) {
                    continue;
                }
                if let Some(&failed_dep) = self.entries[i].dependencies.iter().find(|&&dep| {
                    matches!(
                        self.entries[dep].state,
                        State::Failed | State::Blocked
                    )
                }) {
                    newly_blocked.push((i, failed_dep));
                }
            }
            if newly_blocked.is_empty() {
                break;
            }
            newly_blocked.sort_by(|(a, _), (b, _)| {
                self.entries[*a].job.id.cmp(&self.entries[*b].job.id)
            });
            for (i, dep) in newly_blocked {
                if !matches!(self.entries[i].state, State::Waiting { .. }) {
                    continue;
                }
                let id = self.entries[i].job.id.clone();
                let dependency = self.entries[dep].job.id.clone();
                lines.push(format!("{time} skip {id} {dependency}"));
                self.entries[i].state = State::Blocked;
            }
        }

        loop {
            let mut candidates: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, entry)| {
                    let State::Waiting { ready_at, .. } = entry.state else {
                        return None;
                    };
                    (ready_at <= time
                        && entry.dependencies.iter().all(|&dep| {
                            matches!(self.entries[dep].state, State::Succeeded)
                        }))
                    .then_some(i)
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, time);
                let pb = self.effective_priority(b, time);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut started_any = false;
            for candidate in candidates {
                let candidate_tag = self.entries[candidate].job.tag.as_deref();
                let tag_free = candidate_tag.is_none_or(|tag| {
                    !self.entries.iter().any(|entry| {
                        matches!(entry.state, State::Running { .. })
                            && entry.job.tag.as_deref() == Some(tag)
                    })
                });
                let running_count = self
                    .entries
                    .iter()
                    .filter(|entry| matches!(entry.state, State::Running { .. }))
                    .count();

                if running_count < self.capacity && tag_free {
                    self.start(candidate, time, false, lines);
                    started_any = true;
                    break;
                }

                let mut victims: Vec<usize> = self
                    .entries
                    .iter()
                    .enumerate()
                    .filter_map(|(i, entry)| {
                        if !matches!(entry.state, State::Running { .. })
                            || self.entries[candidate].job.priority
                                < entry.job.priority.saturating_add(2)
                        {
                            return None;
                        }
                        let after_count = running_count - 1;
                        let slot_free = after_count < self.capacity;
                        let tag_free_after = candidate_tag.is_none_or(|tag| {
                            !self.entries.iter().enumerate().any(|(j, other)| {
                                j != i
                                    && matches!(other.state, State::Running { .. })
                                    && other.job.tag.as_deref() == Some(tag)
                            })
                        });
                        (slot_free && tag_free_after).then_some(i)
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    let pa = self.entries[a].job.priority;
                    let pb = self.entries[b].job.priority;
                    pa.cmp(&pb)
                        .then_with(|| {
                            let ta = self.entries[a].started_at;
                            let tb = self.entries[b].started_at;
                            tb.cmp(&ta)
                        })
                        .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
                });

                if let Some(victim) = victims.first().copied() {
                    let State::Running {
                        attempt,
                        finish_at,
                    } = self.entries[victim].state
                    else {
                        continue;
                    };
                    let remaining = finish_at - time;
                    let victim_id = self.entries[victim].job.id.clone();
                    let candidate_id = self.entries[candidate].job.id.clone();
                    lines.push(format!(
                        "{time} preempt {victim_id} by {candidate_id}"
                    ));
                    self.entries[victim].state = State::Waiting {
                        ready_at: time,
                        attempt,
                    };
                    self.entries[victim].remaining = Some(remaining);
                    self.start(candidate, time, false, lines);
                    started_any = true;
                    break;
                }
            }

            if !started_any {
                break;
            }
        }
    }

    fn effective_priority(&self, index: usize, time: u64) -> u8 {
        let entry = &self.entries[index];
        let ready_at = match entry.state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => time,
        };
        if self.aging_ms == 0 {
            entry.job.priority
        } else {
            let age = time.saturating_sub(ready_at) / self.aging_ms;
            (u64::from(entry.job.priority) + age).min(9) as u8
        }
    }

    fn start(&mut self, index: usize, time: u64, _resumed: bool, lines: &mut Vec<String>) {
        let (prior_attempt, remaining) = match self.entries[index].state {
            State::Waiting { attempt, .. } => (attempt, self.entries[index].remaining),
            _ => return,
        };
        let (attempt, run_for) = if let Some(remaining) = remaining {
            (prior_attempt, remaining)
        } else {
            (prior_attempt + 1, self.entries[index].job.duration)
        };
        let finish_at = time.saturating_add(run_for);
        let id = self.entries[index].job.id.clone();
        lines.push(format!("{time} start {id} {attempt} {run_for}"));
        self.entries[index].state = State::Running { attempt, finish_at };
        self.entries[index].remaining = None;
        self.entries[index].started_at = time;
    }
}

fn valid_name(value: &str, min: usize, max: usize) -> bool {
    let bytes = value.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}
