#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
struct Variable<'a> {
    name: &'a str,
    start: usize,
}

struct Expression<'a> {
    operator: Option<u8>,
    variables: Vec<Variable<'a>>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;

    let operator = if content_start < close
        && matches!(
            bytes[content_start],
            b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' | b'!' | b'@' | b'=' | b'$' | b'|'
        )
    {
        let op = bytes[content_start];
        if !matches!(
            op,
            b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
        ) {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        content_start += 1;
        Some(op)
    } else {
        None
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let mut invalid = None;
        for (offset, &byte) in bytes[spec_start..spec_end].iter().enumerate() {
            if (offset == 0 && !is_name_start(byte))
                || (offset > 0 && !is_name_continue(byte))
            {
                invalid = Some(spec_start + offset);
                break;
            }
        }
        if let Some(pos) = invalid {
            return Err(Error::InvalidName { pos });
        }

        let name = &template[spec_start..spec_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        variables.push(Variable {
            name,
            start: spec_start,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        operator,
        variables,
    })
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' {
            if i + 2 < bytes.len()
                && hex_value(bytes[i + 1]).is_some()
                && hex_value(bytes[i + 2]).is_some()
            {
                out.push('%');
                out.push((bytes[i + 1] as char).to_ascii_uppercase());
                out.push((bytes[i + 2] as char).to_ascii_uppercase());
                i += 3;
                continue;
            }
            out.push_str("%25");
        } else if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn binding_value<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator.unwrap_or(0);
    let allow_reserved = matches!(operator, b'+' | b'#');
    let mut fields: Vec<(String, String)> = Vec::new();

    for variable in &expression.variables {
        if let Some(value) = binding_value(variable.name, vars) {
            let encoded = encode(value, allow_reserved);
            fields.push((variable.name.to_owned(), encoded));
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match operator {
        b'.' => (".", "."),
        b'/' => ("/", "/"),
        b';' => (";", ";"),
        b'?' => ("?", "&"),
        b'&' => ("&", "&"),
        _ => ("", ","),
    };

    let named = matches!(operator, b';' | b'?' | b'&');
    let mut out = String::from(prefix);

    for (index, (name, value)) in fields.iter().enumerate() {
        if index > 0 {
            out.push_str(separator);
        }
        if named {
            out.push_str(name);
            if operator != b';' || !value.is_empty() {
                out.push('=');
            }
        }
        out.push_str(value);
    }

    out
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut literal_start = 0;
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[cursor + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                {
                    Some(offset) => cursor + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let expression = parse_expression(template, open, close)?;
                expressions.push((literal_start, open, close + 1, expression));
                cursor = close + 1;
                literal_start = cursor;
            }
            _ => cursor += 1,
        }
    }

    let mut out = String::new();
    let mut copied_until = 0;
    for (start, open, end, expression) in expressions {
        debug_assert_eq!(start, copied_until);
        out.push_str(&template[copied_until..open]);
        out.push_str(&expand_expression(&expression, vars));
        copied_until = end;
    }
    out.push_str(&template[copied_until..]);
    Ok(out)
}
