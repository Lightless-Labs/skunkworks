#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

#[derive(Debug)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(&'a str, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let close = match bytes[open + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                {
                    Some(offset) => open + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let mut content_start = open + 1;
                let first = bytes.get(content_start).copied();
                let operator = match first {
                    Some(b'+') => Operator::Reserved,
                    Some(b'#') => Operator::Fragment,
                    Some(b'.') => Operator::Dot,
                    Some(b'/') => Operator::Slash,
                    Some(b';') => Operator::Semicolon,
                    Some(b'?') => Operator::Query,
                    Some(b'&') => Operator::Continuation,
                    Some(b'=') | Some(b'!') | Some(b'@') | Some(b'|') => {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                    _ => Operator::Simple,
                };

                if !matches!(operator, Operator::Simple) {
                    content_start += 1;
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut spec_start = content_start;
                let mut seen: Vec<&str> = Vec::new();

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&byte| byte == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let spec = &template[spec_start..spec_end];
                    let spec_bytes = spec.as_bytes();
                    let valid_first = |byte: u8| byte.is_ascii_alphabetic() || byte == b'_';

                    if !valid_first(spec_bytes[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }

                    for (offset, &byte) in spec_bytes.iter().enumerate().skip(1) {
                        if !(byte.is_ascii_alphanumeric() || byte == b'_') {
                            return Err(Error::InvalidName {
                                pos: spec_start + offset,
                            });
                        }
                    }

                    if seen.iter().any(|&name| name == spec) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(spec);
                    names.push((spec, spec_start));

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression {
                    start: open,
                    end: close,
                    operator,
                    names,
                });
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::new();
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut fields: Vec<(String, String)> = Vec::new();
        for (name, _) in expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(binding, _)| *binding == name) {
                let encoded = encode(value, matches!(expression.operator, Operator::Reserved | Operator::Fragment));
                fields.push((name.to_string(), encoded));
            }
        }

        if !fields.is_empty() {
            match expression.operator {
                Operator::Simple | Operator::Reserved => {
                    output.push_str(
                        &fields
                            .iter()
                            .map(|(_, value)| value.as_str())
                            .collect::<Vec<_>>()
                            .join(","),
                    );
                }
                Operator::Fragment => {
                    output.push('#');
                    output.push_str(
                        &fields
                            .iter()
                            .map(|(_, value)| value.as_str())
                            .collect::<Vec<_>>()
                            .join(","),
                    );
                }
                Operator::Dot | Operator::Slash => {
                    let separator = match expression.operator {
                        Operator::Dot => '.',
                        _ => '/',
                    };
                    output.push(separator);
                    let sep = separator.to_string();
                    output.push_str(
                        &fields
                            .iter()
                            .map(|(_, value)| value.as_str())
                            .collect::<Vec<_>>()
                            .join(&sep),
                    );
                }
                Operator::Semicolon => {
                    output.push(';');
                    let rendered: Vec<String> = fields
                        .iter()
                        .map(|(name, value)| {
                            if value.is_empty() {
                                name.clone()
                            } else {
                                format!("{name}={value}")
                            }
                        })
                        .collect();
                    output.push_str(&rendered.join(";"));
                }
                Operator::Query | Operator::Continuation => {
                    output.push(if matches!(expression.operator, Operator::Query) {
                        '?'
                    } else {
                        '&'
                    });
                    let rendered: Vec<String> = fields
                        .iter()
                        .map(|(name, value)| format!("{name}={value}"))
                        .collect();
                    output.push_str(&rendered.join("&"));
                }
            }
        }

        literal_start = expression.end + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2])) {
                output.push('%');
                output.push(hex_digit(high));
                output.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4));
            output.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    output
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
