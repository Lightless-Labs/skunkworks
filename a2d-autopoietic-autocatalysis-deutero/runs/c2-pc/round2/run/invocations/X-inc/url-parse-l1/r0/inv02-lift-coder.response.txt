#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }

    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map_or(bytes.len(), |offset| authority_start + offset);

    let authority = &bytes[authority_start..authority_end];
    let at_positions: Vec<usize> = authority
        .iter()
        .enumerate()
        .filter_map(|(offset, &byte)| (byte == b'@').then_some(offset))
        .collect();

    if at_positions.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_offset) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(input[authority_start..authority_start + at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_port = &authority[host_start_offset..];
    let host_start = authority_start + host_start_offset;
    let port_colon = host_port.iter().position(|&byte| byte == b':');
    let host_end_offset = port_colon.unwrap_or(host_port.len());
    let host_bytes = &host_port[..host_end_offset];

    if host_bytes.is_empty() {
        return Err(ParseError::InvalidHost { at: host_start });
    }

    for (offset, &byte) in host_bytes.iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_start + offset,
            });
        }
    }

    let port = if let Some(colon_offset) = port_colon {
        let port_start = host_start + colon_offset + 1;
        let port_bytes = &host_port[colon_offset + 1..];

        if port_bytes.is_empty() {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        for (offset, &byte) in port_bytes.iter().enumerate() {
            if !byte.is_ascii_digit() {
                return Err(ParseError::InvalidPort {
                    at: port_start + offset,
                });
            }
        }

        if port_bytes.len() >= 2 && port_bytes[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        let mut value: u32 = 0;
        for &byte in port_bytes {
            value = match value
                .checked_mul(10)
                .and_then(|current| current.checked_add(u32::from(byte - b'0')))
            {
                Some(value) => value,
                None => return Err(ParseError::InvalidPort { at: port_start }),
            };
        }

        if !(1..=65535).contains(&value) {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        Some(value as u16)
    } else {
        None
    };

    let component_bytes = &bytes[authority_end..];
    let fragment_offset = component_bytes.iter().position(|&byte| byte == b'#');
    let before_fragment_end = fragment_offset.unwrap_or(component_bytes.len());
    let before_fragment = &component_bytes[..before_fragment_end];

    let query_offset = before_fragment.iter().position(|&byte| byte == b'?');
    let path_end = query_offset.unwrap_or(before_fragment.len());
    let path_bytes = &before_fragment[..path_end];

    let path = if path_bytes.is_empty() {
        "/".to_owned()
    } else {
        String::from_utf8(path_bytes.to_vec()).expect("input was valid UTF-8")
    };

    let query = query_offset.map(|offset| {
        String::from_utf8(before_fragment[offset + 1..].to_vec())
            .expect("input was valid UTF-8")
    });

    let fragment = fragment_offset.map(|offset| {
        String::from_utf8(component_bytes[offset + 1..].to_vec())
            .expect("input was valid UTF-8")
    });

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: String::from_utf8(host_bytes.to_ascii_lowercase()).expect("ASCII host"),
        port,
        path,
        query,
        fragment,
    })
}
