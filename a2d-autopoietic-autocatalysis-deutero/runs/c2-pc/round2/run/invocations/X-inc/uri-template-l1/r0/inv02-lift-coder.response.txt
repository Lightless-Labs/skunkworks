#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

struct Expression {
    start: usize,
    end: usize,
    names: Vec<(usize, String)>,
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: open + 1 + offset,
        });
    }

    let mut content_start = open + 1;
    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator { pos: content_start });
    }

    let mut names = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = bytes[content_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| content_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        let valid_first = |b: u8| b.is_ascii_alphabetic() || b == b'_';
        let valid_rest = |b: u8| valid_first(b) || b.is_ascii_digit();

        if !valid_first(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        if let Some(offset) = name_bytes[1..].iter().position(|&b| !valid_rest(b)) {
            return Err(Error::InvalidName {
                pos: spec_start + 1 + offset,
            });
        }

        let name = String::from_utf8(name_bytes.to_vec()).expect("ASCII name");
        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        names.push((spec_start, name));

        if spec_end == close {
            break;
        }
        content_start = spec_end + 1;
        spec_start = content_start;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        names,
    })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(offset) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + offset;
                expressions.push(parse_expression(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn encode_value(value: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%'
            && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            result.push('%');
            result.push((bytes[i + 1] as char).to_ascii_uppercase());
            result.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            result.push(byte as char);
            i += 1;
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
            i += 1;
        }
    }

    result
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut result = String::new();
    let mut cursor = 0;

    for expression in expressions {
        result.push_str(&template[cursor..expression.start]);

        let mut fields = Vec::new();
        for (_, name) in expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(binding, _)| *binding == name) {
                fields.push(encode_value(value));
            }
        }
        if !fields.is_empty() {
            result.push_str(&fields.join(","));
        }

        cursor = expression.end;
    }

    result.push_str(&template[cursor..]);
    Ok(result)
}
