fn parse_atom(field: Field, bytes: &[u8], offset: usize) -> Result<(u32, bool), ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }
        let mut value = 0u32;
        for &b in bytes {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        return Ok((value, true));
    }

    if bytes.iter().all(u8::is_ascii_alphabetic) {
        if let Some(list) = names(field) {
            let word = bytes.iter().map(u8::to_ascii_uppercase).collect::<Vec<_>>();
            for (i, name) in list.iter().enumerate() {
                if word == name.as_bytes() {
                    return Ok((if field == Field::Month { i as u32 + 1 } else { i as u32 }, false));
                }
            }
        }
    }
    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let (_, min, max, width) = field_info(field);
    let mut selected = vec![false; (max + 1) as usize];
    let bytes = text.as_bytes();

    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let mut item_start = 0usize;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }
        let ib = item.as_bytes();
        let slash_positions: Vec<usize> = ib
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let (base, step_bytes, step_offset) = if let Some(&s) = slash_positions.first() {
            (&ib[..s], Some(&ib[s + 1..]), item_start + s + 1)
        } else {
            (ib, None, 0)
        };

        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();
        if hyphens.len() > 1
            || (is_star && !hyphens.is_empty())
            || (base.contains(&b'*') && !is_star)
            || (base.is_empty())
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let mut start = min;
        let mut end = max;
        if !is_star {
            if let Some(&h) = hyphens.first() {
                if h == 0 || h + 1 == base.len() || base[h + 1..].contains(&b'*') {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                let (a, _) = parse_atom(field, &base[..h], item_start)?;
                let (b, _) = parse_atom(field, &base[h + 1..], item_start + h + 1)?;
                if a < min || a > max {
                    return Err(ParseError::OutOfRange { field, offset: item_start });
                }
                if b < min || b > max {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start + h + 1,
                    });
                }
                if a > b {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + h + 1,
                    });
                }
                start = a;
                end = b;
            } else {
                let (v, _) = parse_atom(field, base, item_start)?;
                if v < min || v > max {
                    return Err(ParseError::OutOfRange { field, offset: item_start });
                }
                start = v;
                if step_bytes.is_none() {
                    end = v;
                }
            }
        }

        let step = if let Some(sb) = step_bytes {
            if sb.is_empty() || !sb.iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            if sb.len() > 1 && sb[0] == b'0' {
                return Err(ParseError::LeadingZero {
                    field,
                    offset: step_offset,
                });
            }
            let mut value = 0u32;
            for &b in sb {
                value = value
                    .checked_mul(10)
                    .and_then(|v| v.checked_add((b - b'0') as u32))
                    .ok_or(ParseError::NumberOverflow {
                        field,
                        offset: step_offset,
                    })?;
            }
            if value == 0 || value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            value
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }
        item_start += item.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = Vec::with_capacity(5);
    for (field, text) in fields.into_iter().zip(parts.iter()) {
        selected.push(parse_field(field, text)?);
    }
    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        unrestricted_days: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (index, min, max, _) = field_info(field);
        (min..=max)
            .filter(|&v| self.selected[index][v as usize])
            .map(|v| v as u8)
            .collect()
    }

    fn has(&self, field: Field, value: u32) -> bool {
        let (index, _, _, _) = field_info(field);
        self.selected[index].get(value as usize).copied().unwrap_or(false)
    }

    fn day_matches(&self, date: CivilDateTime, weekday: u8) -> bool {
        let dom = self.has(Field::DayOfMonth, date.day as u32);
        let dow = self.has(Field::DayOfWeek, weekday as u32);
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if date.day == days_in_month(date.year, date.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn validate(at: CivilDateTime) -> Result<(), DateError> {
        if !(2000..=2399).contains(&at.year) {
            return Err(DateError::InvalidDateTime { component: DatePart::Year });
        }
        if !(1..=12).contains(&at.month) {
            return Err(DateError::InvalidDateTime { component: DatePart::Month });
        }
        if at.day == 0 || at.day > days_in_month(at.year, at.month) {
            return Err(DateError::InvalidDateTime { component: DatePart::Day });
        }
        if at.hour > 23 {
            return Err(DateError::InvalidDateTime { component: DatePart::Hour });
        }
        if at.minute > 59 {
            return Err(DateError::InvalidDateTime { component: DatePart::Minute });
        }
        Ok(())
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        Self::validate(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.has(Field::Minute, at.minute as u32)
            && self.has(Field::Hour, at.hour as u32)
            && self.has(Field::Month, at.month as u32)
            && self.day_matches(at, weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        Self::validate(after)?;
        let mut current = after;
        loop {
            if current.year == 2399 && current.month == 12 && current.day == 31
                && current.hour == 23 && current.minute == 59
            {
                return Ok(None);
            }
            advance_minute(&mut current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn advance_minute(at: &mut CivilDateTime) {
    if at.minute < 59 {
        at.minute += 1;
        return;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return;
    }
    at.month = 1;
    at.year += 1;
}

        let d = domain(field);
        if value < d.min as u32 || value > d.max as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
        return Ok(Some(value));
    }
    if bytes.iter().all(u8::is_ascii_alphabetic) {
        if let Some(value) = name_value(field, token) {
            return Ok(Some(value as u32));
        }
    }
    Err(syntax_error(field, offset))
}

fn validate_step(field: Field, token: &str, offset: usize) -> Result<u32, ParseError> {
    let bytes = token.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax_error(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let d = domain(field);
    if value == 0 || value > (d.max - d.min + 1) as u32 {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_item(
    field: Field,
    item: &str,
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    let hyphen_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    if slash_positions.len() > 1 || hyphen_positions.len() > 1 {
        return Err(syntax_error(field, item_offset));
    }
    let slash = slash_positions.first().copied();
    let base_end = slash.unwrap_or(bytes.len());
    let base = &item[..base_end];
    if base.is_empty() {
        return Err(syntax_error(field, item_offset));
    }
    let step = if let Some(pos) = slash {
        let step_start = pos + 1;
        let step_token = &item[step_start..];
        if step_token.is_empty() {
            return Err(syntax_error(field, item_offset));
        }
        Some(validate_step(field, step_token, item_offset + step_start)?)
    } else {
        None
    };

    let d = domain(field);
    let (start, end) = if base == "*" {
        (d.min as u32, d.max as u32)
    } else {
        if base.as_bytes().contains(&b'*') {
            return Err(syntax_error(field, item_offset));
        }
        if let Some(hyphen) = hyphen_positions.first().copied() {
            if hyphen >= base_end || hyphen == 0 || hyphen + 1 >= base_end {
                return Err(syntax_error(field, item_offset));
            }
            let left = &base[..hyphen];
            let right = &base[hyphen + 1..];
            let start_value = validate_base_token(field, left, item_offset)?;
            let end_value = validate_base_token(field, right, item_offset + hyphen + 1)?;
            let start_value = start_value.ok_or_else(|| syntax_error(field, item_offset))?;
            let end_value = end_value
                .ok_or_else(|| syntax_error(field, item_offset + hyphen + 1))?;
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_offset + hyphen + 1,
                });
            }
            (start_value, end_value)
        } else {
            let value = validate_base_token(field, base, item_offset)?
                .ok_or_else(|| syntax_error(field, item_offset))?;
            (value, if step.is_some() { d.max as u32 } else { value })
        }
    };

    let step_size = step.unwrap_or(1);
    let mut value = start;
    while value <= end {
        selected[value as usize] = true;
        value += step_size;
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 32],
        vec![false; 13],
        vec![false; 7],
    ];
    let mut unrestricted_days = [false; 2];

    for (index, (&field, text)) in fields.iter().zip(parts.iter()).enumerate() {
        for (offset, &byte) in text.as_bytes().iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if field == Field::DayOfMonth {
            unrestricted_days[0] = *text == "*";
        } else if field == Field::DayOfWeek {
            unrestricted_days[1] = *text == "*";
        }

        let mut offset = 0;
        for item in text.split(',') {
            parse_item(field, item, offset, &mut selected[index])?;
            offset += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        unrestricted_days,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let d = domain(field);
        (d.min..=d.max)
            .filter(|&v| self.selected[field_index(field)][v as usize])
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        if after == max_datetime() {
            return Ok(None);
        }
        let mut current = after;
        loop {
            current = increment_minute(current);
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
            if current == max_datetime() {
                return Ok(None);
            }
        }
    }

    fn selected_value(&self, field: Field, value: u8) -> bool {
        self.selected[field_index(field)][value as usize]
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.selected_value(Field::Minute, at.minute)
            || !self.selected_value(Field::Hour, at.hour)
            || !self.selected_value(Field::Month, at.month)
        {
            return false;
        }

        let dom_unrestricted = self.unrestricted_days[0];
        let dow_unrestricted = self.unrestricted_days[1];
        let dom = self.selected_value(Field::DayOfMonth, at.day);
        let dow = self.selected_value(Field::DayOfWeek, weekday(at.year, at.month, at.day));

        match (dom_unrestricted, dow_unrestricted) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

    Month,
    Day,
    Hour,
    Minute,
}

pub enum DateError {

}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut total = 6u32;
    for y in 2000..year {
        total += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        total += days_in_month(year, m) as u32;
    }
    total += day as u32 - 1;
    (total % 7) as u8
}

fn increment_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}

fn max_datetime() -> CivilDateTime {
    CivilDateTime {
        year: 2399,
        month: 12,
        day: 31,
        hour: 23,
        minute: 59,
    }
}

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),

fn names(field: Field) -> &'static [(&'static str, u32)] {
    match field {
        Field::Month => &[
            ("JAN", 1), ("FEB", 2), ("MAR", 3), ("APR", 4),
            ("MAY", 5), ("JUN", 6), ("JUL", 7), ("AUG", 8),
            ("SEP", 9), ("OCT", 10), ("NOV", 11), ("DEC", 12),
        ],
        Field::DayOfWeek => &[
            ("SUN", 0), ("MON", 1), ("TUE", 2), ("WED", 3),
            ("THU", 4), ("FRI", 5), ("SAT", 6),
        ],
        _ => &[],
    }

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(field: Field, bytes: &[u8], offset: usize, step: bool) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax_error(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &byte in bytes {
        value = value.checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if step {
        let (_, max) = bounds(field);
        if value == 0 || value > max + 1 {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(value)
}

fn parse_endpoint(field: Field, bytes: &[u8], offset: usize) -> Result<u32, ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) {
        return parse_number(field, bytes, offset, false);
    }
    if bytes.iter().all(u8::is_ascii_alphabetic) {
        let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
        for &(name, value) in names(field) {
            if upper == name.as_bytes() {
                return Ok(value);
            }
        }
    }
    Err(syntax_error(field, offset))
}

fn parse_item(
    field: Field,
    text: &str,
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = text.as_bytes();
    if bytes.is_empty() {
        return Err(ParseError::EmptyItem { field, offset: item_offset });
    }

    let slash_positions: Vec<usize> = bytes.iter().enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();
    if slash_positions.len() > 1 {
        return Err(syntax_error(field, item_offset));
    }

    let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
        if slash == 0 || slash + 1 >= bytes.len() {
            return Err(syntax_error(field, item_offset));
        }
        (&bytes[..slash], Some(&bytes[slash + 1..]), item_offset + slash + 1)
    } else {
        (bytes, None, 0)
    };

    let is_star = base == b"*";
    let hyphens: Vec<usize> = base.iter().enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();
    if hyphens.len() > 1 || (is_star && !hyphens.is_empty()) {
        return Err(syntax_error(field, item_offset));
    }

    let (start, end, start_offset, end_offset) = if is_star {
        let (min, max) = bounds(field);
        (min, max, item_offset, item_offset)
    } else if let Some(&hyphen) = hyphens.first() {
        if hyphen == 0 || hyphen + 1 >= base.len() {
            return Err(syntax_error(field, item_offset));
        }
        let left = &base[..hyphen];
        let right = &base[hyphen + 1..];
        // Check endpoint forms before numeric validation, as required.
        let valid_endpoint = |part: &[u8]| {
            part.iter().all(u8::is_ascii_digit)
                || (part.iter().all(u8::is_ascii_alphabetic)
                    && names(field).iter().any(|(name, _)| {
                        part.eq_ignore_ascii_case(name.as_bytes())
                    }))
        };
        if !valid_endpoint(left) || !valid_endpoint(right) {
            return Err(syntax_error(field, item_offset));
        }
        (
            parse_endpoint(field, left, item_offset)?,
            parse_endpoint(field, right, item_offset + hyphen + 1)?,
            item_offset,
            item_offset + hyphen + 1,
        )
    } else {
        if !base.iter().all(u8::is_ascii_digit)
            && !(base.iter().all(u8::is_ascii_alphabetic)
                && names(field).iter().any(|(name, _)| {
                    base.eq_ignore_ascii_case(name.as_bytes())
                }))
        {
            return Err(syntax_error(field, item_offset));
        }
        let value = parse_endpoint(field, base, item_offset)?;
        let end = if step_bytes.is_some() {
            bounds(field).1
        } else {
            value
        };
        (value, end, item_offset, item_offset)
    };

    if start > end {
        return Err(ParseError::ReversedRange { field, offset: end_offset });
    }

    let step = if let Some(step_bytes) = step_bytes {
        Some(parse_number(field, step_bytes, step_offset, true)?)
    } else {
        None
    };
    let stride = step.unwrap_or(1);
    let mut value = start;
    while value <= end {
        let (min, _) = bounds(field);
        selected[(value - min) as usize] = true;
        match value.checked_add(stride) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }
    let _ = start_offset;
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (min, max) = bounds(FIELDS[i]);
        vec![false; (max - min + 1) as usize]
    });

    for (index, (&field, text)) in FIELDS.iter().zip(parts.iter()).enumerate() {
        for (offset, &byte) in text.as_bytes().iter().enumerate() {
            if !(byte.is_ascii_digit()
                || byte.is_ascii_alphabetic()
                || matches!(byte, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        let mut item_start = 0usize;
        for item in text.split(',') {
            parse_item(field, item, item_start, &mut selected[index])?;
            item_start += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        day_of_month_unrestricted: parts[2] == "*",
        day_of_week_unrestricted: parts[4] == "*",
    })
}
