#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn field_info(field: Field) -> (usize, u32, u32, u32) {
    match field {
        Field::Minute => (0, 0, 59, 60),
        Field::Hour => (1, 0, 23, 24),
        Field::DayOfMonth => (2, 1, 31, 31),
        Field::Month => (3, 1, 12, 12),
        Field::DayOfWeek => (4, 0, 6, 7),
    }
}

fn names(field: Field) -> Option<&'static [&'static str]> {
    match field {
        Field::Month => Some(&[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC",
        ]),
        Field::DayOfWeek => Some(&["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"]),
        _ => None,
    }
}

fn parse_atom(field: Field, bytes: &[u8], offset: usize) -> Result<(u32, bool), ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.iter().all(u8::is_ascii_digit) {
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }
        let mut value = 0u32;
        for &b in bytes {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        return Ok((value, true));
    }

    if bytes.iter().all(u8::is_ascii_alphabetic) {
        if let Some(list) = names(field) {
            let word = bytes.iter().map(u8::to_ascii_uppercase).collect::<Vec<_>>();
            for (i, name) in list.iter().enumerate() {
                if word == name.as_bytes() {
                    return Ok((if field == Field::Month { i as u32 + 1 } else { i as u32 }, false));
                }
            }
        }
    }
    Err(ParseError::InvalidSyntax { field, offset })
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let (_, min, max, width) = field_info(field);
    let mut selected = vec![false; (max + 1) as usize];
    let bytes = text.as_bytes();

    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let mut item_start = 0usize;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }
        let ib = item.as_bytes();
        let slash_positions: Vec<usize> = ib
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let (base, step_bytes, step_offset) = if let Some(&s) = slash_positions.first() {
            (&ib[..s], Some(&ib[s + 1..]), item_start + s + 1)
        } else {
            (ib, None, 0)
        };

        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();
        if hyphens.len() > 1
            || (is_star && !hyphens.is_empty())
            || (base.contains(&b'*') && !is_star)
            || (base.is_empty())
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let mut start = min;
        let mut end = max;
        if !is_star {
            if let Some(&h) = hyphens.first() {
                if h == 0 || h + 1 == base.len() || base[h + 1..].contains(&b'*') {
                    return Err(ParseError::InvalidSyntax {
                        field,
                        offset: item_start,
                    });
                }
                let (a, _) = parse_atom(field, &base[..h], item_start)?;
                let (b, _) = parse_atom(field, &base[h + 1..], item_start + h + 1)?;
                if a < min || a > max {
                    return Err(ParseError::OutOfRange { field, offset: item_start });
                }
                if b < min || b > max {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start + h + 1,
                    });
                }
                if a > b {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + h + 1,
                    });
                }
                start = a;
                end = b;
            } else {
                let (v, _) = parse_atom(field, base, item_start)?;
                if v < min || v > max {
                    return Err(ParseError::OutOfRange { field, offset: item_start });
                }
                start = v;
                if step_bytes.is_none() {
                    end = v;
                }
            }
        }

        let step = if let Some(sb) = step_bytes {
            if sb.is_empty() || !sb.iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            if sb.len() > 1 && sb[0] == b'0' {
                return Err(ParseError::LeadingZero {
                    field,
                    offset: step_offset,
                });
            }
            let mut value = 0u32;
            for &b in sb {
                value = value
                    .checked_mul(10)
                    .and_then(|v| v.checked_add((b - b'0') as u32))
                    .ok_or(ParseError::NumberOverflow {
                        field,
                        offset: step_offset,
                    })?;
            }
            if value == 0 || value > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            value
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }
        item_start += item.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = Vec::with_capacity(5);
    for (field, text) in fields.into_iter().zip(parts.iter()) {
        selected.push(parse_field(field, text)?);
    }
    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        unrestricted_days: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (index, min, max, _) = field_info(field);
        (min..=max)
            .filter(|&v| self.selected[index][v as usize])
            .map(|v| v as u8)
            .collect()
    }

    fn has(&self, field: Field, value: u32) -> bool {
        let (index, _, _, _) = field_info(field);
        self.selected[index].get(value as usize).copied().unwrap_or(false)
    }

    fn day_matches(&self, date: CivilDateTime, weekday: u8) -> bool {
        let dom = self.has(Field::DayOfMonth, date.day as u32);
        let dow = self.has(Field::DayOfWeek, weekday as u32);
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if date.day == days_in_month(date.year, date.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn validate(at: CivilDateTime) -> Result<(), DateError> {
        if !(2000..=2399).contains(&at.year) {
            return Err(DateError::InvalidDateTime { component: DatePart::Year });
        }
        if !(1..=12).contains(&at.month) {
            return Err(DateError::InvalidDateTime { component: DatePart::Month });
        }
        if at.day == 0 || at.day > days_in_month(at.year, at.month) {
            return Err(DateError::InvalidDateTime { component: DatePart::Day });
        }
        if at.hour > 23 {
            return Err(DateError::InvalidDateTime { component: DatePart::Hour });
        }
        if at.minute > 59 {
            return Err(DateError::InvalidDateTime { component: DatePart::Minute });
        }
        Ok(())
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        Self::validate(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.has(Field::Minute, at.minute as u32)
            && self.has(Field::Hour, at.hour as u32)
            && self.has(Field::Month, at.month as u32)
            && self.day_matches(at, weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        Self::validate(after)?;
        let mut current = after;
        loop {
            if current.year == 2399 && current.month == 12 && current.day == 31
                && current.hour == 23 && current.minute == 59
            {
                return Ok(None);
            }
            advance_minute(&mut current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn advance_minute(at: &mut CivilDateTime) {
    if at.minute < 59 {
        at.minute += 1;
        return;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return;
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return;
    }
    at.month = 1;
    at.year += 1;
}
