#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<String> = Vec::new();
        for product in &products {
            let valid_id = (1..=8).contains(&product.id.len())
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }

            if seen.iter().any(|id| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match &self.selected {
            Some(id) => id.clone(),
            None => return Err(Error::NoSelection),
        };

        let product = self
            .products
            .iter()
            .find(|product| product.id == selected_id)
            .expect("selected product must exist");
        let price = product.price;

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = match Self::make_change(amount, &self.tubes) {
            Some(change) => change,
            None => return Err(Error::NoChange { amount }),
        };

        let mut proposed_tubes = self.tubes;
        for (index, &coin) in DENOMINATIONS.iter().enumerate() {
            let removed = change.iter().filter(|&&c| c == coin).count() as u16;
            proposed_tubes[index] -= removed;
        }
        for &coin in &self.escrow {
            let index = DENOMINATIONS
                .iter()
                .position(|&denomination| denomination == coin)
                .expect("escrow contains only accepted denominations");
            proposed_tubes[index] += 1;
        }

        for (index, &count) in proposed_tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        self.tubes = proposed_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt {
            id: selected_id,
            price,
            change,
        })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        Ok(returned)
    }

    fn make_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
        let mut greedy_counts = [0_u16; 4];
        let mut remaining = amount;

        for index in (0..4).rev() {
            let coin = DENOMINATIONS[index];
            let count = tubes[index].min(remaining / coin);
            greedy_counts[index] = count;
            remaining -= count * coin;
        }

        if remaining == 0 {
            return Some(Self::counts_to_descending_coins(&greedy_counts));
        }

        let mut best: Option<[u16; 4]> = None;
        let mut counts = [0_u16; 4];
        Self::search_change(3, amount, tubes, &mut counts, &mut best);

        best.map(|counts| Self::counts_to_descending_coins(&counts))
    }

    fn search_change(
        index: i32,
        remaining: u16,
        tubes: &[u16; 4],
        counts: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
    ) {
        if index < 0 {
            if remaining == 0 {
                let replace = match best {
                    None => true,
                    Some(current) => {
                        let new_total: u32 = counts.iter().map(|&n| u32::from(n)).sum();
                        let old_total: u32 = current.iter().map(|&n| u32::from(n)).sum();
                        new_total < old_total
                            || (new_total == old_total
                                && Self::descending_lex_greater(counts, current))
                    }
                };
                if replace {
                    *best = Some(*counts);
                }
            }
            return;
        }

        let i = index as usize;
        let coin = DENOMINATIONS[i];
        let max_count = tubes[i].min(remaining / coin);
        for count in 0..=max_count {
            counts[i] = count;
            Self::search_change(
                index - 1,
                remaining - count * coin,
                tubes,
                counts,
                best,
            );
        }
        counts[i] = 0;
    }

    fn descending_lex_greater(left: &[u16; 4], right: &[u16; 4]) -> bool {
        for index in (0..4).rev() {
            if left[index] != right[index] {
                return left[index] > right[index];
            }
        }
        false
    }

    fn counts_to_descending_coins(counts: &[u16; 4]) -> Vec<u16> {
        let capacity: usize = counts.iter().map(|&count| usize::from(count)).sum();
        let mut coins = Vec::with_capacity(capacity);
        for index in (0..4).rev() {
            for _ in 0..counts[index] {
                coins.push(DENOMINATIONS[index]);
            }
        }
        coins
    }
}
