use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    if let Some((at, _)) = input
        .bytes()
        .enumerate()
        .find(|(_, b)| !(0x21..=0x7e).contains(b))
    {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(input.len());

    let authority = &input[authority_start..authority_end];
    let auth_bytes = authority.as_bytes();
    let ats: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();

    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + ats[1],
        });
    }

    let (userinfo, hostport, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            authority_start + at + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    let (host, port_text, port_start, ipv6) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        let suffix = &hostport[close + 1..];
        let (port, pstart) = if suffix.is_empty() {
            (None, hostport_start + close + 1)
        } else if suffix.starts_with(':') {
            (Some(&suffix[1..]), hostport_start + close + 2)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, pstart, true)
    } else {
        let split = hostport.as_bytes().iter().position(|&b| b == b':');
        let (host, port, pstart) = match split {
            Some(i) => (&hostport[..i], Some(&hostport[i + 1..]), hostport_start + i + 1),
            None => (hostport, None, hostport_start + hostport.len()),
        };
        if host.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in host.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + i,
                });
            }
        }
        (host.to_ascii_lowercase(), port, pstart, false)
    };

    let port = if let Some(port_text) = port_text {
        if port_text.is_empty() {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        if let Some(i) = port_text.bytes().position(|b| !b.is_ascii_digit()) {
            return Err(ParseError::InvalidPort {
                at: port_start + i,
            });
        }
        if port_text.len() >= 2 && port_text.starts_with('0') {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        let mut value: u32 = 0;
        for b in port_text.bytes() {
            value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
                Some(v) => v,
                None => return Err(ParseError::InvalidPort { at: port_start }),
            };
        }
        if !(1..=65535).contains(&value) {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        Some(value as u16)
    } else {
        None
    };

    let remainder = &input[authority_end..];
    let (before_fragment, fragment) = match remainder.find('#') {
        Some(i) => (&remainder[..i], Some(remainder[i + 1..].to_owned())),
        None => (remainder, None),
    };
    let (path_query, query) = match before_fragment.find('?') {
        Some(i) => (
            &before_fragment[..i],
            Some(before_fragment[i + 1..].to_owned()),
        ),
        None => (before_fragment, None),
    };
    let path = if path_query.is_empty() {
        "/".to_owned()
    } else {
        path_query.to_owned()
    };

    if let Some(ref value) = userinfo {
        validate_component(value, authority_start)?;
    }
    if !path_query.is_empty() {
        validate_component(&path, authority_end)?;
    }
    if let Some(ref value) = query {
        let offset = authority_end + before_fragment.find('?').unwrap_or(0) + 1;
        validate_component(value, offset)?;
    }
    if let Some(ref value) = fragment {
        let offset = authority_end + remainder.find('#').unwrap_or(0) + 1;
        validate_component(value, offset)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn validate_component(text: &str, offset: usize) -> Result<(), ParseError> {
    match decode_component(Component::Path, text) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: offset + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: offset + at }),
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    origins.push(i);
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = e.valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(at).copied().unwrap_or(input.len()),
            })
        }
    }
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let mut stack: Vec<&str> = Vec::new();
    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut terminal_navigation = false;

    for (i, raw) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape {
                at: at + segment_offset(path, i),
            },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 {
                at: at + segment_offset(path, i),
            },
        })?;
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        if is_dotdot {
            stack.pop();
        } else if !is_dot {
            stack.push(raw);
        }
        if i + 1 == segments.len() {
            terminal_navigation = is_dot || is_dotdot;
        }
    }

    if terminal_navigation {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn segment_offset(path: &str, segment_index: usize) -> usize {
    let mut offset = 1;
    for _ in 0..segment_index {
        match path[offset..].find('/') {
            Some(i) => offset += i + 1,
            None => return path.len(),
        }
    }
    offset
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    if let Some((at, _)) = reference
        .bytes()
        .enumerate()
        .find(|(_, b)| !(0x21..=0x7e).contains(b))
    {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let structural_end = reference
        .char_indices()
        .find(|(_, c)| matches!(c, '/' | '?' | '#'))
        .map(|(i, _)| i)
        .unwrap_or(reference.len());
    if reference[..structural_end].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (before_fragment, fragment) = match reference.find('#') {
        Some(i) => (&reference[..i], Some(reference[i + 1..].to_owned())),
        None => (reference, None),
    };
    let (ref_path, ref_query) = match before_fragment.find('?') {
        Some(i) => (&before_fragment[..i], Some(before_fragment[i + 1..].to_owned())),
        None => (before_fragment, None),
    };

    let (path, query) = if ref_path.is_empty() {
        (
            base_path,
            if ref_query.is_some() {
                ref_query
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
            format!("{slash}{ref_path}")
        };
        (merged, ref_query)
    };

    let mut constructed = format!("{}://", base_url.scheme);
    if let Some(userinfo) = &base_url.userinfo {
        constructed.push_str(userinfo);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&path);
    if let Some(query) = query {
        constructed.push('?');
        constructed.push_str(&query);
    }
    if let Some(fragment) = fragment {
        constructed.push('#');
        constructed.push_str(&fragment);
    }

    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}
