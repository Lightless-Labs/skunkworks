#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$') {
                        word.push(escaped);
                        active = true;
                        i = after;
                    } else if escaped == '\n' {
                        i = after;
                    } else {
                        word.push('\\');
                        active = true;
                        i = next_i;
                    }
                } else if c == '$' {
                    active = true;
                    if let Some((value, end)) = parse_expansion(input, i, env)? {
                        word.push_str(value);
                        i = end;
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if escaped != '\n' {
                        word.push(escaped);
                        active = true;
                    }
                    i = after;
                } else if c == '$' {
                    if let Some((value, end)) = parse_expansion(input, i, env)? {
                        word.push_str(value);
                        active = true;
                        i = end;
                    } else {
                        word.push('$');
                        active = true;
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

// Returns None for a literal dollar; otherwise the substituted value and byte offset after expansion.
fn parse_expansion<'a>(
    input: &str,
    dollar: usize,
    env: &'a [(&'a str, &'a str)],
) -> Result<Option<(&'a str, usize)>, SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok(None);
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let contents_start = after_dollar + 1;
        let mut cursor = contents_start;
        let mut close = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    let skipped = input[cursor..].chars().next().unwrap();
                    cursor += skipped.len_utf8();
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let Some(close_pos) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        };

        let contents = &input[contents_start..close_pos];
        let (name_end, name) = {
            let mut end = 0;
            let mut chars = contents.char_indices();
            if let Some((_, c)) = chars.next() {
                if !is_name_start(c) {
                    return Err(SplitError::InvalidExpansion { pos: dollar });
                }
                end = c.len_utf8();
                for (offset, c) in chars {
                    if is_name_continue(c) {
                        end = offset + c.len_utf8();
                    } else {
                        break;
                    }
                }
            } else {
                return Err(SplitError::InvalidExpansion { pos: dollar });
            }
            (end, &contents[..end])
        };

        let remainder = &contents[name_end..];
        let selected = if remainder.is_empty() {
            lookup(name, env).unwrap_or("")
        } else if let Some(default) = remainder.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value,
                _ => {
                    // Defaults can contain decoded escapes and are not borrowed from input.
                    // This branch is handled by the caller's owned expansion path below.
                    return Ok(Some((value_or_empty_placeholder(default), close_pos + 1)));
                }
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok(Some((selected, close_pos + 1)))
    } else if is_name_start(first) {
        let mut end = after_dollar + first.len_utf8();
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if is_name_continue(c) {
                end += c.len_utf8();
            } else {
                break;
            }
        }
        Ok(Some((lookup(&input[after_dollar..end], env).unwrap_or(""), end)))
    } else {
        Ok(None)
    }
}

// Placeholder for the default branch; replaced by the owned parser in split_with_defaults.
fn value_or_empty_placeholder(_text: &str) -> &'static str {
    ""
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    split_impl(input, env)
}

fn split_impl(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    // The parser below keeps expansion values owned so braced defaults can be decoded.
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$') {
                        word.push(escaped);
                        active = true;
                        i = after;
                    } else if escaped == '\n' {
                        i = after;
                    } else {
                        word.push('\\');
                        active = true;
                        i = next_i;
                    }
                } else if c == '$' {
                    let parsed = parse_expansion_owned(input, i, env)?;
                    if let Some((value, end)) = parsed {
                        word.push_str(&value);
                        active = true;
                        i = end;
                    } else {
                        word.push('$');
                        active = true;
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active = true;
                    i = next_i;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if escaped != '\n' {
                        word.push(escaped);
                        active = true;
                    }
                    i = after;
                } else if c == '$' {
                    if let Some((value, end)) = parse_expansion_owned(input, i, env)? {
                        word.push_str(&value);
                        active = true;
                        i = end;
                    } else {
                        word.push('$');
                        active = true;
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_expansion_owned(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<Option<(String, usize)>, SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok(None);
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let start = after_dollar + 1;
        let mut cursor = start;
        let mut close = None;
        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    let skipped = input[cursor..].chars().next().unwrap();
                    cursor += skipped.len_utf8();
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }
        let Some(close_pos) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        };

        let contents = &input[start..close_pos];
        let mut name_end = 0;
        let mut chars = contents.char_indices();
        let Some((_, first_name_char)) = chars.next() else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };
        if !is_name_start(first_name_char) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        name_end = first_name_char.len_utf8();
        for (offset, ch) in chars {
            if is_name_continue(ch) {
                name_end = offset + ch.len_utf8();
            } else {
                break;
            }
        }

        let name = &contents[..name_end];
        let rest = &contents[name_end..];
        let value = if rest.is_empty() {
            lookup(name, env).unwrap_or("").to_string()
        } else if let Some(default) = rest.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };
        Ok(Some((value, close_pos + 1)))
    } else if is_name_start(first) {
        let mut end = after_dollar + first.len_utf8();
        while end < input.len() {
            let ch = input[end..].chars().next().unwrap();
            if is_name_continue(ch) {
                end += ch.len_utf8();
            } else {
                break;
            }
        }
        Ok(Some((
            lookup(&input[after_dollar..end], env).unwrap_or("").to_string(),
            end,
        )))
    } else {
        Ok(None)
    }
}

fn decode_default(text: &str) -> String {
    let mut output = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        let next = i + c.len_utf8();
        if c == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            output.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            output.push(c);
            i = next;
        }
    }
    output
}
