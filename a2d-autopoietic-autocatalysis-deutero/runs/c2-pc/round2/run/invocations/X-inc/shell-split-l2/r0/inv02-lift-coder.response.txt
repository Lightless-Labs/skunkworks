#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn append_default(text: &str, out: &mut String) {
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().expect("valid character boundary");
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().expect("character exists");
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.push(next);
                        }
                        i = next_i + next.len_utf8();
                    } else {
                        word.push('\\');
                        i = next_i;
                    }
                } else if c == '$' {
                    active = true;
                    let (end, value) = parse_expansion(input, i, env)?;
                    word.push_str(&value);
                    i = end;
                } else {
                    word.push(c);
                    i = next_i;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().expect("character exists");
                    if next != '\n' {
                        active = true;
                        word.push(next);
                    }
                    i = next_i + next.len_utf8();
                } else if c == '$' {
                    active = true;
                    let (end, value) = parse_expansion(input, i, env)?;
                    word.push_str(&value);
                    i = end;
                } else {
                    active = true;
                    word.push(c);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((after_dollar, "$".to_string()));
    }

    let first = input[after_dollar..].chars().next().expect("character exists");
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut closing = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().expect("character exists");
            if c == '\\' {
                cursor += c.len_utf8();
                if cursor < input.len() {
                    let escaped = input[cursor..].chars().next().expect("character exists");
                    cursor += escaped.len_utf8();
                }
            } else if c == '}' {
                closing = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let close_pos = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close_pos];

        let mut chars = contents.char_indices();
        let mut name_end = 0;
        let mut valid_start = false;

        if let Some((_, c)) = chars.next() {
            if is_name_start(c) {
                valid_start = true;
                name_end = c.len_utf8();
                while let Some(&(offset, c)) = chars.peek() {
                    if is_name_continue(c) {
                        name_end = offset + c.len_utf8();
                        chars.next();
                    } else {
                        break;
                    }
                }
            }
        }

        if !valid_start {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &contents[..name_end];
        let remainder = &contents[name_end..];
        let value = if remainder.is_empty() {
            lookup(name, env).unwrap_or("").to_string()
        } else if let Some(default_text) = remainder.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => {
                    let mut decoded = String::new();
                    append_default(default_text, &mut decoded);
                    decoded
                }
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        Ok((close_pos + 1, value))
    } else if is_name_start(first) {
        let mut end = after_dollar + first.len_utf8();
        while end < input.len() {
            let c = input[end..].chars().next().expect("character exists");
            if is_name_continue(c) {
                end += c.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        Ok((end, lookup(name, env).unwrap_or("").to_string()))
    } else {
        Ok((after_dollar, "$".to_string()))
    }
}
