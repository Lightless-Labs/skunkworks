fn error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }

fn name_value(field: Field, s: &str) -> Option<u32> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",

        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,

    names
        .iter()
        .position(|n| n.eq_ignore_ascii_case(s))
        .map(|i| if field == Field::Month { i as u32 + 1 } else { i as u32 })

fn parse_number(field: Field, s: &str, offset: usize, step: bool) -> Result<u32, ParseError> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return Err(error(field, offset, 0));
    }
    if s.len() > 1 && s.as_bytes()[0] == b'0' {
        return Err(error(field, offset, 1));
    }
    let mut n = 0u32;
    for b in s.bytes() {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| error(field, offset, 2))?;
    }
    let (lo, hi) = bounds(field);
    if step {
        if n == 0 || n > width(field) {
            return Err(error(field, offset, 5));
        }
    } else if n < lo || n > hi {
        return Err(error(field, offset, 3));
    }
    Ok(n)
}

fn parse_endpoint(field: Field, s: &str, offset: usize) -> Result<(u32, bool), ParseError> {
    if s.bytes().all(|b| b.is_ascii_digit()) && !s.is_empty() {
        Ok((parse_number(field, s, offset, false)?, true))
    } else if let Some(v) = name_value(field, s) {
        Ok((v, false))
    } else {
        Err(error(field, offset, 0))
    }
}

fn parse_field(field: Field, text: &str, values: &mut [bool]) -> Result<(), ParseError> {
    for (i, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = bounds(field);
    let mut start = 0usize;
    loop {
        let end = text[start..]
            .find(',')
            .map(|i| start + i)
            .unwrap_or(text.len());
        let item = &text[start..end];
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: start,
            });
        }

        let bytes = item.as_bytes();
        let slash_positions: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, b)| (*b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(error(field, start, 0));
        }
        let (base, step_text, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), start + slash + 1)
        } else {
            (item, None, 0)
        };
        if base.is_empty() || step_text == Some("") {
            return Err(error(field, start, 0));
        }

        let hyphens: Vec<usize> = base
            .bytes()
            .enumerate()
            .filter_map(|(i, b)| (b == b'-').then_some(i))
            .collect();
        if hyphens.len() > 1 {
            return Err(error(field, start, 0));
        }

        let is_star = base == "*";
        let range = hyphens.first().copied();
        if is_star && range.is_some() {
            return Err(error(field, start, 0));
        }

        let mut parsed_start = min;
        let mut parsed_end = max;
        let mut numeric_start_offset = start;
        let mut numeric_end_offset = start;
        let mut numeric_start = false;
        let mut numeric_end = false;

        if !is_star {
            if let Some(h) = range {
                let left = &base[..h];
                let right = &base[h + 1..];
                if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
                    return Err(error(field, start, 0));
                }
                let (a, an) = parse_endpoint(field, left, start)?;
                let (b, bn) = parse_endpoint(field, right, start + h + 1)?;
                parsed_start = a;
                parsed_end = b;
                numeric_start = an;
                numeric_end = bn;
                numeric_start_offset = start;
                numeric_end_offset = start + h + 1;
            } else {
                let (a, n) = parse_endpoint(field, base, start)?;
                parsed_start = a;
                parsed_end = a;
                numeric_start = n;
                numeric_start_offset = start;
                numeric_end_offset = start;
            }
        }

        if range.is_some() && parsed_start > parsed_end {
            return Err(error(field, numeric_end_offset, 4));
        }

        let step = if let Some(s) = step_text {
            Some(parse_number(field, s, step_offset, true)?)
        } else {
            None
        };

        let (from, to) = if is_star {
            (min, max)
        } else if step.is_some() && range.is_none() {
            (parsed_start, max)
        } else {
            (parsed_start, parsed_end)
        };

        if let Some(step) = step {
            let mut n = from;

                match n.checked_add(step) {
                    Some(next) => n = next,
                    None => break,
                }
            }
        } else {
            for n in from..=to {
                values[(n - min) as usize] = true;
            }
        }

        let _ = (numeric_start, numeric_start_offset);
        if end == text.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.as_bytes().first() == Some(&b'@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let parts: Vec<&str> = expanded.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 31],
        vec![false; 12],
        vec![false; 7],
    ];
    for i in 0..5 {
        parse_field(FIELDS[i], parts[i], &mut sets[i])?;
    }

    Ok(Cron {
        sets,
        unrestricted_dom: parts[2] == "*",
        unrestricted_dow: parts[4] == "*",
    })
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_days(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if !(1..=month_days(at.year, at.month)).contains(&at.day) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_days(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((6 + days) % 7) as u8
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _) = bounds(field);
        self.sets[index]
            .iter()
            .enumerate()
            .filter_map(|(i, yes)| yes.then_some(min as u8 + i as u8))
            .collect()
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.sets[0][at.minute as usize]
            || !self.sets[1][at.hour as usize]
            || !self.sets[3][(at.month - 1) as usize]
        {
            return false;
        }
        let dom = self.sets[2][(at.day - 1) as usize];
        let dow = self.sets[4][weekday(at.year, at.month, at.day) as usize];
        if self.unrestricted_dom && self.unrestricted_dow {
            true
        } else if self.unrestricted_dom {
            dow
        } else if self.unrestricted_dow {
            dom
        } else if at.day == month_days(at.year, at.month) {
            dom && dow
        } else {
            dom || dow
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    fn times(&self) -> Vec<(u8, u8)> {

fn leap_year(year: u16) -> bool {

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;

    }
    for m in 1..month {
        days += u32::from(days_in_month(year, m));
    }
    days += u32::from(day - 1);
    ((6 + days) % 7) as u8
}

    let day_match = match (cron.unrestricted_days[0], cron.unrestricted_days[1]) {
        (true, true) => true,
        (true, false) => dow_match,
        (false, true) => dom,
        (false, false) if date.day == days_in_month(date.year, date.month) => dom && dow_match,
        (false, false) => dom || dow_match,

        return None;

    let minutes = cron.allowed(Field::Minute);
    for hour in hours {
        for &minute in &minutes {
            if strictly_after.map_or(true, |(h, m)| (hour, minute) > (h, m)) {
                return Some(CivilDateTime { hour, minute, ..date });
            }
        }
    }
    None
}

fn last_time(cron: &Cron, date: CivilDateTime, strictly_before: Option<(u8, u8)>) -> Option<CivilDateTime> {
    if !cron.selected[3][date.month as usize] {
        return None;
    }
    let dow = weekday(date.year, date.month, date.day);
    let dom = cron.selected[2][date.day as usize];
    let dow_match = cron.selected[4][dow as usize];
    let day_match = match (cron.unrestricted_days[0], cron.unrestricted_days[1]) {
        (true, true) => true,
        (true, false) => dow_match,
        (false, true) => dom,
        (false, false) if date.day == days_in_month(date.year, date.month) => dom && dow_match,
        (false, false) => dom || dow_match,
    };
    if !day_match {
        return None;
    }

    let hours = cron.allowed(Field::Hour);
    let minutes = cron.allowed(Field::Minute);
    for &hour in hours.iter().rev() {
        for &minute in minutes.iter().rev() {
            if strictly_before.map_or(true, |(h, m)| (hour, minute) < (h, m)) {
                return Some(CivilDateTime { hour, minute, ..date });
            }
        }
    }
    None
}

fn search_forward(cron: &Cron, from: CivilDateTime, inclusive: bool) -> Option<CivilDateTime> {
    let mut date = CivilDateTime {
        hour: 0,
        minute: 0,
        ..from
    };
    let bound = if inclusive {
        None
    } else {
        Some((from.hour, from.minute))
    };
    let mut first = true;
    loop {
        if let Some(found) = first_time(cron, date, if first { bound } else { None }) {
            if inclusive && found < from {
                // This path is only used for window starts, where the date time must be inclusive.
                if found == date {
                    return Some(found);
                }
            } else {
                return Some(found);
            }
        }
        first = false;
        date = next_day(date)?;
    }
}

fn search_backward(cron: &Cron, before: CivilDateTime) -> Option<CivilDateTime> {
    let mut date = CivilDateTime {
        hour: 0,
        minute: 0,
        ..before
    };
    let mut first = true;
    loop {
        if let Some(found) = last_time(
            cron,
            date,
            if first {
                Some((before.hour, before.minute))
            } else {
                None
            },
        ) {
            return Some(found);
        }
        first = false;
        date = previous_day(date)?;
    }
}
