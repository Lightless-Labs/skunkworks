#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn append_literal(word: &mut Word, c: char, wildcard_active: bool) {
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_expansion(word: &mut Word, value: &str, wildcard_active: bool) {
    for c in value.chars() {
        append_literal(word, c, wildcard_active);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.text.push(c);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        if next == '$' && input[after_next..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: next_i });
                        }
                        word.text.push(next);
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        word.text.push('\\');
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let contents_start = next_i + 1;
                        let mut j = contents_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let skipped = input[after..].chars().next().unwrap();
                                    j = after + skipped.len_utf8();
                                } else {
                                    j = after;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let body = &input[contents_start..close];
                        let bytes = body.as_bytes();
                        let mut name_end = 0;
                        let mut chars = body.char_indices();
                        if let Some((_, first)) = chars.next() {
                            if is_name_start(first) {
                                name_end = first.len_utf8();
                                for (off, ch) in chars {
                                    if is_name_continue(ch) {
                                        name_end = off + ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                            }
                        }
                        if name_end == 0 {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let name = &body[..name_end];
                        let selected = if name_end == body.len() {
                            lookup(name, env).unwrap_or("")
                        } else if bytes.get(name_end..name_end + 2) == Some(b":-") {
                            let value = lookup(name, env).unwrap_or("");
                            if value.is_empty() {
                                let default = &body[name_end + 2..];
                                let decoded = decode_default(default);
                                append_expansion(&mut word, &decoded, false);
                                i = close + 1;
                                active = true;
                                continue;
                            } else {
                                value
                            }
                        } else {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        };
                        append_expansion(&mut word, selected, false);
                        i = close + 1;
                        active = true;
                    } else if let Some(name_first) = input[next_i..].chars().next() {
                        if is_name_start(name_first) {
                            let mut end = next_i + name_first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let value = lookup(&input[next_i..end], env).unwrap_or("");
                            append_expansion(&mut word, value, false);
                            i = end;
                            active = true;
                        } else {
                            word.text.push('$');
                            i = next_i;
                        }
                    } else {
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    word.text.push(c);
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' || c == '"' {
                    quote = if c == '\'' { Quote::Single } else { Quote::Double };
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after;
                    } else {
                        append_literal(&mut word, next, false);
                        active = true;
                        i = after;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let dollar = i;
                        let contents_start = next_i + 1;
                        let mut j = contents_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let skipped = input[after..].chars().next().unwrap();
                                    j = after + skipped.len_utf8();
                                } else {
                                    j = after;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let body = &input[contents_start..close];
                        let bytes = body.as_bytes();
                        let mut name_end = 0;
                        let mut chars = body.char_indices();
                        if let Some((_, first)) = chars.next() {
                            if is_name_start(first) {
                                name_end = first.len_utf8();
                                for (off, ch) in chars {
                                    if is_name_continue(ch) {
                                        name_end = off + ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                            }
                        }
                        if name_end == 0 {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let name = &body[..name_end];
                        if name_end == body.len() {
                            let value = lookup(name, env).unwrap_or("");
                            append_expansion(&mut word, value, true);
                        } else if bytes.get(name_end..name_end + 2) == Some(b":-") {
                            let value = lookup(name, env).unwrap_or("");
                            if value.is_empty() {
                                let decoded = decode_default(&body[name_end + 2..]);
                                append_expansion(&mut word, &decoded, true);
                            } else {
                                append_expansion(&mut word, value, true);
                            }
                        } else {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        active = true;
                        i = close + 1;
                    } else if let Some(first) = input[next_i..].chars().next() {
                        if is_name_start(first) {
                            let mut end = next_i + first.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let value = lookup(&input[next_i..end], env).unwrap_or("");
                            append_expansion(&mut word, value, true);
                            active = true;
                            i = end;
                        } else {
                            append_literal(&mut word, '$', true);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append_literal(&mut word, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_literal(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }
    if active {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%') {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{word}'"))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let candidates = [
            bare_candidate(word),
            single_candidate(word),
            double_candidate(word),
        ];
        let mut best: Option<(usize, String)> = None;
        for candidate in candidates.into_iter().flatten() {
            let len = candidate.len();
            if best.as_ref().is_none_or(|(best_len, _)| len < *best_len) {
                best = Some((len, candidate));
            }
        }
        let chosen = best.map(|(_, text)| text).unwrap_or_else(|| {
            let mut out = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    out.push('\'');
                    out.push('\\');
                    out.push('\'');
                    out.push('\'');
                } else {
                    out.push(c);
                }
            }
            out.push('\'');
            out
        });
        encoded.push(chosen);
    }
    Ok(encoded.join(" "))
}
