#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobRecord {
    job: Job,
    dependency_indices: Vec<usize>,
    state: Option<State>,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<JobRecord>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut seen = std::collections::HashMap::<String, usize>::new();
        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if seen.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            seen.insert(job.id.clone(), i);
        }

        let mut dependency_indices = Vec::with_capacity(jobs.len());
        for (job_index, job) in jobs.iter().enumerate() {
            let mut local_seen = std::collections::HashSet::<&str>::new();
            let mut indices = Vec::with_capacity(job.dependencies.len());
            for (dependency_index, dependency) in job.dependencies.iter().enumerate() {
                if !local_seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(job_index, dependency_index));
                }
                match seen.get(dependency) {
                    Some(&index) => indices.push(index),
                    None => {
                        return Err(Error::MissingDependency(job_index, dependency_index));
                    }
                }
            }
            dependency_indices.push(indices);
        }

        let mut colors = vec![0u8; jobs.len()];
        fn visit(node: usize, edges: &[Vec<usize>], colors: &mut [u8]) -> bool {
            if colors[node] == 1 {
                return true;
            }
            if colors[node] == 2 {
                return false;
            }
            colors[node] = 1;
            for &next in &edges[node] {
                if visit(next, edges, colors) {
                    return true;
                }
            }
            colors[node] = 2;
            false
        }
        for i in 0..jobs.len() {
            if visit(i, &dependency_indices, &mut colors) {
                return Err(Error::Cycle);
            }
        }

        let records = jobs
            .into_iter()
            .zip(dependency_indices)
            .map(|(job, dependency_indices)| JobRecord {
                job,
                dependency_indices,
                state: None,
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs: records,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.cycle_started
            && self.jobs.iter().any(|record| {
                matches!(
                    record.state,
                    Some(State::Waiting { .. }) | Some(State::Running { .. })
                )
            })
        {
            return Err(Error::Busy);
        }

        for record in &mut self.jobs {
            record.state = Some(State::Waiting { ready_at: now });
            record.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (i, record) in self.jobs.iter().enumerate() {
            let ready_at = match record.state {
                Some(State::Waiting { ready_at }) if ready_at <= now => ready_at,
                _ => continue,
            };
            if !record
                .dependency_indices
                .iter()
                .all(|&dependency| matches!(self.jobs[dependency].state, Some(State::Succeeded)))
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                record.job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (record.job.priority as u64 + age).min(9) as u8
            };
            if best.map_or(true, |(_, priority)| effective > priority) {
                best = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = best else {
            return Ok(None);
        };

        let record = &mut self.jobs[index];
        record.attempts += 1;
        record.state = Some(State::Running {
            attempt: record.attempts,
        });
        Ok(Some(Ticket {
            id: record.job.id.clone(),
            attempt: record.attempts,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = self
            .jobs
            .iter()
            .position(|record| record.job.id == id)
            .ok_or(Error::Unknown)?;
        if !matches!(self.jobs[index].state, Some(State::Running { .. })) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.jobs[index].state = Some(State::Succeeded);
            return Ok(());
        }

        let attempts = self.jobs[index].attempts;
        if attempts == self.jobs[index].job.max_attempts {
            self.jobs[index].state = Some(State::Failed);
            self.propagate_blocked();
        } else {
            let shift = u32::from(attempts.saturating_sub(1));
            let delay = self
                .retry_base
                .checked_mul(1u64.checked_shl(shift).unwrap_or(u64::MAX))
                .unwrap_or(u64::MAX)
                .min(self.retry_cap);
            self.jobs[index].state = Some(State::Waiting {
                ready_at: now.saturating_add(delay),
            });
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let record = self
            .jobs
            .iter()
            .find(|record| record.job.id == id)
            .ok_or(Error::Unknown)?;
        record.state.clone().ok_or(Error::NoCycle)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();
            for (i, record) in self.jobs.iter().enumerate() {
                if !matches!(record.state, Some(State::Waiting { .. })) {
                    continue;
                }
                if record.dependency_indices.iter().any(|&dependency| {
                    matches!(
                        self.jobs[dependency].state,
                        Some(State::Failed) | Some(State::Blocked)
                    )
                }) {
                    newly_blocked.push(i);
                }
            }
            if newly_blocked.is_empty() {
                break;
            }
            for index in newly_blocked {
                self.jobs[index].state = Some(State::Blocked);
            }
        }
    }
}
