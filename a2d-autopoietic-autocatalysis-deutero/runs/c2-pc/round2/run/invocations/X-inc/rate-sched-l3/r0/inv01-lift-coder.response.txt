```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobState {
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<Job>,
    states: Vec<JobState>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            let valid_id = !job.id.is_empty()
                && job.id.len() <= 16
                && job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-');
            if !valid_id {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if ids.insert(job.id.as_str(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (d, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, d));
                }
                if !ids.contains_key(dep.as_str()) {
                    return Err(Error::MissingDependency(i, d));
                }
            }
        }

        let mut indegree = vec![0usize; jobs.len()];
        let mut children = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            for dep in &job.dependencies {
                let parent = ids[dep.as_str()];
                indegree[i] += 1;
                children[parent].push(i);
            }
        }
        let mut queue: std::collections::VecDeque<usize> = indegree
            .iter()
            .enumerate()
            .filter_map(|(i, &n)| (n == 0).then_some(i))
            .collect();
        let mut visited = 0;
        while let Some(node) = queue.pop_front() {
            visited += 1;
            for &child in &children[node] {
                indegree[child] -= 1;
                if indegree[child] == 0 {
                    queue.push_back(child);
                }
            }
        }
        if visited != jobs.len() {
            return Err(Error::Cycle);
        }

        let states = (0..jobs.len())
            .map(|_| JobState {
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            states,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.cycle_started
            && self.states.iter().any(|s| {
                matches!(s.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for state in &mut self.states {
            state.state = State::Waiting { ready_at: now };
            state.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut selected: Option<(usize, u8)> = None;
        for (i, job) in self.jobs.iter().enumerate() {
            let ready_at = match self.states[i].state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            let deps_succeeded = job.dependencies.iter().all(|dep| {
                let dep_index = self.jobs.iter().position(|j| j.id == *dep).unwrap();
                matches!(self.states[dep_index].state, State::Succeeded)
            });
            if !deps_succeeded {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (job.priority as u64 + age).min(9) as u8
            };
            if selected
                .map(|(_, priority)| effective > priority)
                .unwrap_or(true)
            {
                selected = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((i, _)) = selected else {
            return Ok(None);
        };

        self.states[i].attempts += 1;
        let attempt = self.states[i].attempts;
        self.states[i].state = State::Running { attempt };
        Ok(Some(Ticket {
            id: self.jobs[i].id.clone(),
            attempt,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(i) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        if !matches!(self.states[i].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.states[i].state = State::Succeeded;
            return Ok(());
        }

        let attempt = self.states[i].attempts;
        if attempt == self.jobs[i].max_attempts {
            self.states[i].state = State::Failed;
            self.propagate_blocked();
        } else {
            let shift = u32::from(attempt.saturating_sub(1));
            let delay = self
                .retry_base
                .checked_shl(shift)
                .unwrap_or(u64::MAX)
                .min(self.retry_cap);
            let ready_at = now.saturating_add(delay);
            self.states[i].state = State::Waiting { ready_at };
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(i) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        Ok(self.states[i].state.clone())
    }

    fn propagate_blocked(&mut self) {
        loop {
            let to_block: Vec<usize> = self
                .jobs
                .iter()
                .enumerate()
                .filter_map(|(i, job)| {
                    if !matches!(self.states[i].state, State::Waiting { .. }) {
                        return None;
                    }
                    let has_failed_dependency = job.dependencies.iter().any(|dep| {
                        let dep_index = self.jobs.iter().position(|j| j.id == *dep).unwrap();
                        matches!(
                            self.states[dep_index].state,
                            State::Failed | State::Blocked
                        )
                    });
                    has_failed_dependency.then_some(i)
                })
                .collect();

            if to_block.is_empty() {
                break;
            }
            for i in to_block {
                self.states[i].state = State::Blocked;
            }
        }
    }
}
```
