#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobData {
    job: Job,
    dependencies: Vec<usize>,
    state: Option<State>,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<JobData>,
    id_to_index: std::collections::HashMap<String, usize>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    has_time: bool,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut id_to_index = std::collections::HashMap::new();
        for (index, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(index));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(index));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(index));
            }
            if id_to_index.contains_key(&job.id) {
                return Err(Error::Duplicate(index));
            }
            id_to_index.insert(job.id.clone(), index);
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (job_index, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (dependency_index, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(job_index, dependency_index));
                }
                match id_to_index.get(dependency) {
                    Some(&index) => resolved.push(index),
                    None => return Err(Error::MissingDependency(job_index, dependency_index)),
                }
            }
            dependencies.push(resolved);
        }

        if has_cycle(&dependencies) {
            return Err(Error::Cycle);
        }

        let job_data = jobs
            .into_iter()
            .zip(dependencies)
            .map(|(job, dependencies)| JobData {
                job,
                dependencies,
                state: None,
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs: job_data,
            id_to_index,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_time: false,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.cycle_started
            && self.jobs.iter().any(|job| {
                matches!(
                    job.state,
                    Some(State::Waiting { .. } | State::Running { .. })
                )
            })
        {
            return Err(Error::Busy);
        }

        for job in &mut self.jobs {
            job.state = Some(State::Waiting { ready_at: now });
            job.attempts = 0;
        }
        self.cycle_started = true;
        self.record_time(now);
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (index, job) in self.jobs.iter().enumerate() {
            let ready_at = match job.state {
                Some(State::Waiting { ready_at }) if ready_at <= now => ready_at,
                _ => continue,
            };
            if !job.dependencies.iter().all(|&dep| {
                matches!(self.jobs[dep].state, Some(State::Succeeded))
            }) {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.job.priority
            } else {
                let gained = (now - ready_at) / self.aging_ms;
                (job.job.priority as u64 + gained).min(9) as u8
            };

            if best.map_or(true, |(best_index, best_priority)| {
                effective > best_priority
                    || (effective == best_priority && index < best_index)
            }) {
                best = Some((index, effective));
            }
        }

        let result = if let Some((index, _)) = best {
            let job = &mut self.jobs[index];
            job.attempts += 1;
            let attempt = job.attempts;
            job.state = Some(State::Running { attempt });
            Some(Ticket {
                id: job.job.id.clone(),
                attempt,
            })
        } else {
            None
        };

        self.record_time(now);
        Ok(result)
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = *self.id_to_index.get(id).ok_or(Error::Unknown)?;
        if !matches!(self.jobs[index].state, Some(State::Running { .. })) {
            return Err(Error::NotRunning);
        }

        if success {
            self.jobs[index].state = Some(State::Succeeded);
        } else if self.jobs[index].attempts == self.jobs[index].job.max_attempts {
            self.jobs[index].state = Some(State::Failed);
            self.propagate_blocked();
        } else {
            let delay = retry_delay(self.retry_base, self.retry_cap, self.jobs[index].attempts);
            let ready_at = now.saturating_add(delay);
            self.jobs[index].state = Some(State::Waiting { ready_at });
        }

        self.record_time(now);
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = *self.id_to_index.get(id).ok_or(Error::Unknown)?;
        self.jobs[index].state.clone().ok_or(Error::NoCycle)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if self.has_time && now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn record_time(&mut self, now: u64) {
        self.last_time = now;
        self.has_time = true;
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();
            for (index, job) in self.jobs.iter().enumerate() {
                if !matches!(job.state, Some(State::Waiting { .. })) {
                    continue;
                }
                if job.dependencies.iter().any(|&dep| {
                    matches!(
                        self.jobs[dep].state,
                        Some(State::Failed | State::Blocked)
                    )
                }) {
                    newly_blocked.push(index);
                }
            }
            if newly_blocked.is_empty() {
                break;
            }
            for index in newly_blocked {
                self.jobs[index].state = Some(State::Blocked);
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes.iter().all(|&byte| {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-'
        })
}

fn has_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(node: usize, dependencies: &[Vec<usize>], marks: &mut [u8]) -> bool {
        if marks[node] == 1 {
            return true;
        }
        if marks[node] == 2 {
            return false;
        }
        marks[node] = 1;
        for &dependency in &dependencies[node] {
            if visit(dependency, dependencies, marks) {
                return true;
            }
        }
        marks[node] = 2;
        false
    }

    let mut marks = vec![0; dependencies.len()];
    (0..dependencies.len()).any(|node| visit(node, dependencies, &mut marks))
}

fn retry_delay(base: u64, cap: u64, attempt: u8) -> u64 {
    let mut delay = base;
    for _ in 1..attempt {
        if delay >= cap || delay > cap / 2 {
            return cap;
        }
        delay *= 2;
    }
    delay.min(cap)
}
