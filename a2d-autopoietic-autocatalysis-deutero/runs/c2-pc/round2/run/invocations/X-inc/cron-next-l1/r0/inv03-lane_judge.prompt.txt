fn field_domain(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: start,
        });
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }

    if is_step {
        let (_, max) = field_domain(field);
        let width = max + 1 - field_domain(field).0;
        if value == 0 || value > width {
            return Err(ParseError::BadStep {
                field,
                offset: start,
            });
        }
    } else {
        let (min, max) = field_domain(field);
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: start,
            });
        }
    }

    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let item = &bytes[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &byte)| (byte == b'/').then_some(i))
        .collect();

    let (base_end, step) = if slash_positions.is_empty() {
        (item_end, None)
    } else if slash_positions.len() == 1 {
        let slash = slash_positions[0];
        if slash == 0 || slash + 1 == item.len() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let step_start = item_start + slash + 1;
        let parsed = parse_number(bytes, step_start, item_end, field, true)?;
        (item_start + slash, Some(parsed))
    } else {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    };

    let base = &bytes[item_start..base_end];
    let (domain_min, domain_max) = field_domain(field);

    let (start, end, range_end_offset) = if base == b"*" {
        (domain_min, domain_max, None)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &byte)| (byte == b'-').then_some(i))
            .collect();

        if hyphens.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        if let Some(&hyphen) = hyphens.first() {
            if hyphen == 0 || hyphen + 1 == base.len() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            let start_value =
                parse_number(bytes, item_start, item_start + hyphen, field, false)?;
            let end_start = item_start + hyphen + 1;
            let end_value = parse_number(bytes, end_start, base_end, field, false)?;
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: end_start,
                });
            }
            (start_value, end_value, Some(end_start))
        } else {
            let value = parse_number(bytes, item_start, base_end, field, false)?;
            (value, value, None)
        }
    };

    if step.is_none() {
        for value in start..=end {
            selected[value as usize] = true;
        }
    } else {
        let step = step.unwrap();
        let stepped_end = if base == b"*" {
            domain_max
        } else if range_end_offset.is_some() {
            end
        } else {
            domain_max
        };

        let mut value = start;
        while value <= stepped_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }
    }

    Ok(())
}

fn parse_field(
    text: &str,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut item_start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b',' {
            parse_item(bytes, item_start, index, field, selected)?;
            item_start = index + 1;
        }
    }
    parse_item(bytes, item_start, bytes.len(), field, selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = [
        vec![false; 60],
        vec![false; 24],
        vec![false; 32],
        vec![false; 13],
        vec![false; 7],
    ];
    let mut day_stars = [false; 2];

    for index in 0..5 {
        let field = field_kinds[index];
        if index == 2 {
            day_stars[0] = fields[index].as_bytes() == b"*";
        } else if index == 4 {
            day_stars[1] = fields[index].as_bytes() == b"*";
        }
        parse_field(fields[index], field, &mut selected[index])?;
    }

    Ok(Cron {
        selected,
        day_stars,
    })
}

fn bounds(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if end <= start {
        return Err(ParseError::InvalidSyntax { field, offset: start });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut value: u32 = 0;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    if is_step {
        let (min, max) = bounds(field);
        if value < 1 || value > (max - min + 1) as u32 {
            return Err(ParseError::BadStep { field, offset: start });
        }
    } else {
        let (min, max) = bounds(field);
        if value < min as u32 || value > max as u32 {
            return Err(ParseError::OutOfRange { field, offset: start });
        }
    }
    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (domain_min, domain_max) = bounds(field);
    let mut selected = vec![false; (domain_max + 1) as usize];
    let mut item_start = 0;

    for item_end in (0..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let base_end = slash_positions.first().copied().unwrap_or(item.len());
        let step_start = slash_positions.first().map(|p| p + 1);

        let base = &item[..base_end];
        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let syntax_ok = if is_star {
            true
        } else if hyphens.is_empty() {
            !base.is_empty() && base.iter().all(u8::is_ascii_digit)
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            h > 0
                && h + 1 < base.len()
                && base[..h].iter().all(u8::is_ascii_digit)
                && base[h + 1..].iter().all(u8::is_ascii_digit)
        } else {
            false
        };

        if !syntax_ok
            || slash_positions.len() > 1
            || step_start.is_some_and(|s| s >= item.len())
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let step = if let Some(s) = step_start {
            Some(parse_number(
                item,
                s,
                item.len(),
                field,
                true,
            )?)
        } else {
            None
        };

        let (start, end) = if is_star {
            (domain_min as u32, domain_max as u32)
        } else if hyphens.is_empty() {
            let n = parse_number(item, 0, base.len(), field, false)?;
            (
                n,
                if step.is_some() {
                    domain_max as u32
                } else {
                    n
                },
            )
        } else {
            let h = hyphens[0];
            let start = parse_number(item, 0, h, field, false)?;
            let end = parse_number(item, h + 1, base.len(), field, false)?;
            if start > end {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + h + 1,
                });
            }
            (start, end)
        };

        let stride = step.unwrap_or(1);
        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(stride) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &yes)| yes.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let field_types = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut parsed = Vec::with_capacity(5);
    for (part, field) in parts.iter().zip(field_types) {
        parsed.push(parse_field(part, field)?);
    }

    Ok(Cron {
        fields: parsed.try_into().ok().unwrap(),
        day_stars: [parts[2] == "*", parts[4] == "*"],
    })
}

            Field::Hour => 1,
            Field::DayOfMonth => 2,

    }
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let field = fields[i];
        vec![false; (field_max(field) - field_min(field) + 1) as usize]
    });
    let mut day_stars = [false; 2];

    for (index, (&field, text)) in fields.iter().zip(parts.iter()).enumerate() {
        for (offset, byte) in text.bytes().enumerate() {
            if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }

        if matches!(field, Field::DayOfMonth | Field::DayOfWeek) && *text == "*" {
            day_stars[if field == Field::DayOfMonth { 0 } else { 1 }] = true;
        }

        let mut item_start = 0usize;
        for item in text.split(',') {
            if item.is_empty() {
                return Err(ParseError::EmptyItem {
                    field,
                    offset: item_start,
                });
            }
            parse_item(field, item, item_start, &mut selected[index])?;
            item_start += item.len() + 1;
        }
    }

    Ok(Cron {
        selected,
        day_stars,
    })
}

fn parse_item(
    field: Field,
    item: &str,
    item_offset: usize,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
        .collect();

    if slash_positions.len() > 1 {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (base_end, step_start) = match slash_positions.first() {
        Some(&position) => (position, Some(position + 1)),
        None => (bytes.len(), None),
    };
    let base = &item[..base_end];

    let is_star = base == "*";
    let hyphens: Vec<usize> = base
        .as_bytes()
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
        .collect();

    let base_kind = if is_star {
        if !hyphens.is_empty() {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        BaseKind::Star
    } else if hyphens.is_empty() {
        if base.is_empty() || !base.bytes().all(|b| b.is_ascii_digit()) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        BaseKind::Single
    } else if hyphens.len() == 1 {
        let position = hyphens[0];
        if position == 0
            || position + 1 == base.len()
            || !base[..position].bytes().all(|b| b.is_ascii_digit())
            || !base[position + 1..].bytes().all(|b| b.is_ascii_digit())
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
        BaseKind::Range(position)
    } else {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    };

    if let Some(start) = step_start {
        if start >= item.len() || !item[start..].bytes().all(|b| b.is_ascii_digit()) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_offset,
            });
        }
    }

    let min = field_min(field) as u32;
    let max = field_max(field) as u32;
    let mut start_value = min;
    let mut end_value = max;

    match base_kind {
        BaseKind::Star => {}
        BaseKind::Single => {
            let value = parse_number(field, base, item_offset, false)?;
            check_domain(field, value, item_offset)?;
            start_value = value;
            if step_start.is_none() {
                end_value = value;
            }
        }
        BaseKind::Range(position) => {
            let left = &base[..position];
            let right = &base[position + 1..];
            let left_value = parse_number(field, left, item_offset, false)?;
            check_domain(field, left_value, item_offset)?;
            let right_offset = item_offset + position + 1;
            let right_value = parse_number(field, right, right_offset, false)?;
            check_domain(field, right_value, right_offset)?;
            if left_value > right_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: right_offset,
                });
            }
            start_value = left_value;
            end_value = right_value;
        }
    }

    let step = if let Some(step_offset) = step_start {
        let step_value = parse_number(field, &item[step_offset..], item_offset + step_offset, true)?;
        let width = max - min + 1;
        if step_value == 0 || step_value > width {
            return Err(ParseError::BadStep {
                field,
                offset: item_offset + step_offset,
            });
        }
        step_value
    } else {
        1
    };

    let mut value = start_value;
    while value <= end_value {
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

#[derive(Clone, Copy)]
enum BaseKind {
    Star,
    Single,
    Range(usize),
}

fn parse_number(
    field: Field,
    text: &str,
    offset: usize,
    is_step: bool,
) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &byte in bytes {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if is_step {
        let width = (field_max(field) - field_min(field) + 1) as u32;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset });
        }
    }
    Ok(value)
}

fn check_domain(field: Field, value: u32, offset: usize) -> Result<(), ParseError> {
    if value < field_min(field) as u32 || value > field_max(field) as u32 {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(())
}
