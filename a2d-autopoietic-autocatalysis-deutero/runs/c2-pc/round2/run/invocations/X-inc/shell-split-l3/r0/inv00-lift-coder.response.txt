#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append(word: &mut Word, s: &str, wildcards_active: bool) {
    for c in s.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if wildcards_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append(&mut word, &input[i..next_i], false);
                }
                i = next_i;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if matches!(n, '"' | '\\' | '$') {
                        append(&mut word, &input[next_i..after_n], false);
                        i = after_n;
                    } else if n == '\n' {
                        i = after_n;
                    } else {
                        append(&mut word, "\\", false);
                        i = next_i;
                    }
                } else if c == '$' {
                    if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if n == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                    }
                    if next_i < input.len() && input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let skipped = input[after..].chars().next().unwrap();
                                    j = after + skipped.len_utf8();
                                } else {
                                    j = after;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let body = &input[content_start..close];
                        let (name, default) = if let Some(k) = body.find(":-") {
                            (&body[..k], Some(&body[k + 2..]))
                        } else {
                            (body, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_continue)
                            || (default.is_none() && body.contains(':'))
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v).unwrap_or("");
                        if let Some(default) = default {
                            if value.is_empty() {
                                let decoded = decode_default(default);
                                append(&mut word, &decoded, false);
                            } else {
                                append(&mut word, value, false);
                            }
                        } else {
                            append(&mut word, value, false);
                        }
                        active = true;
                        i = close + 1;
                    } else if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if is_name_start(n) {
                            let mut end = next_i + n.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            let value = env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v).unwrap_or("");
                            append(&mut word, value, false);
                            active = true;
                            i = end;
                        } else {
                            append(&mut word, "$", false);
                            i = next_i;
                        }
                    } else {
                        append(&mut word, "$", false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else {
                    append(&mut word, &input[i..next_i], false);
                    i = next_i;
                }
                let _ = open;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after_n = next_i + n.len_utf8();
                    if n != '\n' {
                        append(&mut word, &input[next_i..after_n], false);
                        active = true;
                    }
                    i = after_n;
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if n == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                    }
                    if next_i < input.len() && input[next_i..].starts_with('{') {
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            let after = j + ch.len_utf8();
                            if ch == '\\' {
                                if after < input.len() {
                                    let skipped = input[after..].chars().next().unwrap();
                                    j = after + skipped.len_utf8();
                                } else {
                                    j = after;
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = after;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let body = &input[content_start..close];
                        let (name, default) = if let Some(k) = body.find(":-") {
                            (&body[..k], Some(&body[k + 2..]))
                        } else {
                            (body, None)
                        };
                        if name.is_empty() || !name.chars().next().is_some_and(is_name_start)
                            || !name.chars().all(is_name_continue)
                            || (default.is_none() && body.contains(':'))
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v).unwrap_or("");
                        if let Some(default) = default {
                            if value.is_empty() {
                                let decoded = decode_default(default);
                                append(&mut word, &decoded, true);
                            } else {
                                append(&mut word, value, true);
                            }
                        } else {
                            append(&mut word, value, true);
                        }
                        active = true;
                        i = close + 1;
                    } else if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if is_name_start(n) {
                            let mut end = next_i + n.len_utf8();
                            while end < input.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if is_name_continue(ch) {
                                    end += ch.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            let name = &input[next_i..end];
                            let value = env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v).unwrap_or("");
                            append(&mut word, value, true);
                            active = true;
                            i = end;
                        } else {
                            append(&mut word, "$", true);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append(&mut word, "$", true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append(&mut word, &input[i..next_i], true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
