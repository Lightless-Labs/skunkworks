#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append_char(word: &mut Word, c: char, active_glob: bool) {
    if active_glob && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn append_str(word: &mut Word, s: &str, active_glob: bool) {
    for c in s.chars() {
        append_char(word, c, active_glob);
    }
}

fn parse_braced(input: &str, dollar: usize) -> Result<(usize, String), SplitError> {
    let bytes = input.as_bytes();
    let mut close = None;
    let mut i = dollar + 2;

    while i < bytes.len() {
        if bytes[i] == b'\\' {
            i += 1;
            if i < bytes.len() {
                let c = input[i..].chars().next().unwrap();
                i += c.len_utf8();
            }
        } else if bytes[i] == b'}' {
            close = Some(i);
            break;
        } else {
            let c = input[i..].chars().next().unwrap();
            i += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let content = &input[dollar + 2..close];

    let name_end = content
        .char_indices()
        .find_map(|(offset, c)| (!is_name_continue(c)).then_some(offset))
        .unwrap_or(content.len());

    let name = &content[..name_end];
    if name.is_empty()
        || !name.chars().next().is_some_and(is_name_start)
    {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let remainder = &content[name_end..];
    let value = if remainder.is_empty() {
        lookup(name, &[]).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        decode_default(default)
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, value))
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }

    out
}

fn parse_braced_with_env(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let bytes = input.as_bytes();
    let mut close = None;
    let mut i = dollar + 2;

    while i < bytes.len() {
        if bytes[i] == b'\\' {
            i += 1;
            if i < bytes.len() {
                let c = input[i..].chars().next().unwrap();
                i += c.len_utf8();
            }
        } else if bytes[i] == b'}' {
            close = Some(i);
            break;
        } else {
            let c = input[i..].chars().next().unwrap();
            i += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let content = &input[dollar + 2..close];

    let name_end = content
        .char_indices()
        .find_map(|(offset, c)| (!is_name_continue(c)).then_some(offset))
        .unwrap_or(content.len());

    let name = &content[..name_end];
    if name.is_empty() || !name.chars().next().is_some_and(is_name_start) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let remainder = &content[name_end..];
    let value = if remainder.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = remainder.strip_prefix(":-") {
        match lookup(name, env) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0usize;
    let mut i = 0usize;
    let bytes = input.as_bytes();

    while i < bytes.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                    i = next_i;
                } else {
                    append_char(&mut current, c, false);
                    i = next_i;
                }
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= bytes.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_char(&mut current, next, false);
                        }
                        i = after_next;
                    } else {
                        append_char(&mut current, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, value) = parse_braced_with_env(input, i, env)?;
                        append_str(&mut current, &value, false);
                        active = true;
                        i = end;
                    } else if let Some(name_start) = input[next_i..].chars().next() {
                        if is_name_start(name_start) {
                            let mut end = next_i;
                            while end < bytes.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            append_str(&mut current, lookup(name, env).unwrap_or(""), false);
                            active = true;
                            i = end;
                        } else {
                            append_char(&mut current, '$', false);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append_char(&mut current, '$', false);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_char(&mut current, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' || c == '"' {
                    active = true;
                    quote = if c == '\'' { Quote::Single } else { Quote::Double };
                    quote_pos = i;
                    i = next_i;
                } else if c == '\\' {
                    if next_i >= bytes.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        append_char(&mut current, next, false);
                        active = true;
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, value) = parse_braced_with_env(input, i, env)?;
                        append_str(&mut current, &value, true);
                        active = true;
                        i = end;
                    } else if let Some(name_start) = input[next_i..].chars().next() {
                        if is_name_start(name_start) {
                            let mut end = next_i;
                            while end < bytes.len() {
                                let ch = input[end..].chars().next().unwrap();
                                if !is_name_continue(ch) {
                                    break;
                                }
                                end += ch.len_utf8();
                            }
                            let name = &input[next_i..end];
                            append_str(&mut current, lookup(name, env).unwrap_or(""), true);
                            active = true;
                            i = end;
                        } else {
                            append_char(&mut current, '$', true);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append_char(&mut current, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_char(&mut current, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }

    if active {
        words.push(current);
    }

    Ok(words)
}
