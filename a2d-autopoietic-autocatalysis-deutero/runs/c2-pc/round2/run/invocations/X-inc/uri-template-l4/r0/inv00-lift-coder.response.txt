#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Specifier {
    name: String,
    pos: usize,
    modifier: Modifier,
}

struct Expression {
    operator: Operator,
    specs: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, start: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = start + 1;
    let operator = if content_start < close {
        match bytes[content_start] {
            b'+' => {
                content_start += 1;
                Operator::Plus
            }
            b'#' => {
                content_start += 1;
                Operator::Hash
            }
            b'.' => {
                content_start += 1;
                Operator::Dot
            }
            b'/' => {
                content_start += 1;
                Operator::Slash
            }
            b';' => {
                content_start += 1;
                Operator::Semicolon
            }
            b'?' => {
                content_start += 1;
                Operator::Question
            }
            b'&' => {
                content_start += 1;
                Operator::Ampersand
            }
            b'!' | b'=' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: content_start });
            }
            _ => Operator::Simple,
        }
    } else {
        Operator::Simple
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut spec_start = content_start;

    loop {
        let mut end = spec_start;
        while end < close && bytes[end] != b',' {
            end += 1;
        }
        if end == spec_start {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let part = &template[spec_start..end];
        let part_bytes = part.as_bytes();
        let modifier_offset = part_bytes
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .unwrap_or(part_bytes.len());
        let name_bytes = &part_bytes[..modifier_offset];

        if name_bytes.is_empty() || !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (i, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_byte(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + i,
                });
            }
        }

        let modifier = if modifier_offset == part_bytes.len() {
            Modifier::None
        } else {
            let modifier_pos = spec_start + modifier_offset;
            let suffix = &part_bytes[modifier_offset..];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.first() == Some(&b':')
                && (2..=5).contains(&suffix.len())
                && suffix[1].is_ascii_digit()
                && suffix[1] != b'0'
                && suffix[1..].iter().all(u8::is_ascii_digit)
            {
                let mut n = 0usize;
                for &digit in &suffix[1..] {
                    n = n * 10 + usize::from(digit - b'0');
                }
                if n > 9999 {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                Modifier::Prefix(n)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        };

        let name = part[..modifier_offset].to_owned();
        if seen.iter().any(|previous| previous == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            pos: spec_start,
            modifier,
        });

        if end == close {
            break;
        }
        spec_start = end + 1;
    }

    Ok(Expression { operator, specs })
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> u8 {
    if value < 10 {
        b'0' + value
    } else {
        b'A' + value - 10
    }
}

fn encode(input: &str, reserved_allowed: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(char::from(hex_digit(high)));
                out.push(char::from(hex_digit(low)));
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved_allowed && is_reserved(byte)) {
            out.push(char::from(byte));
        } else {
            out.push('%');
            out.push(char::from(hex_digit(byte >> 4)));
            out.push(char::from(hex_digit(byte & 0x0f)));
        }
        i += 1;
    }
    out
}

fn encode_scalar_value(value: &str, modifier: Modifier, reserved: bool) -> String {
    let selected = match modifier {
        Modifier::Prefix(count) => value.chars().take(count).collect::<String>(),
        Modifier::None | Modifier::Explode => value.to_owned(),
    };
    encode(&selected, reserved)
}

fn field_value(operator: Operator, name: &str, value: String) -> String {
    match operator {
        Operator::Semicolon if value.is_empty() => name.to_owned(),
        Operator::Semicolon | Operator::Question | Operator::Ampersand => {
            format!("{name}={value}")
        }
        _ => value,
    }
}

fn operator_encoding(operator: Operator) -> bool {
    matches!(operator, Operator::Plus | Operator::Hash)
}

fn separator_and_prefix(operator: Operator) -> (&'static str, &'static str) {
    match operator {
        Operator::Simple | Operator::Plus => (",", ""),
        Operator::Hash => (",", "#"),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Question => ("&", "?"),
        Operator::Ampersand => ("&", "&"),
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields = Vec::<String>::new();
    let reserved = operator_encoding(expression.operator);

    for spec in &expression.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(scalar) => {
                let encoded = encode_scalar_value(scalar, spec.modifier, reserved);
                fields.push(field_value(expression.operator, &spec.name, encoded));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(field_value(expression.operator, &spec.name, encoded));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(field_value(expression.operator, &spec.name, joined));
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }
                for i in 0..entries.len() {
                    for j in 0..i {
                        if entries[i].0 == entries[j].0 {
                            return Err(Error::DuplicateKey { pos: spec.pos });
                        }
                    }
                }

                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let field = if matches!(expression.operator, Operator::Semicolon)
                            && encoded_value.is_empty()
                        {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(field_value(expression.operator, &spec.name, joined));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = separator_and_prefix(expression.operator);
    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let mut close = open + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }
                if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
                    return Err(Error::NestedOpen { pos: inner });
                }
                let expression = parse_expression(template, open, close)?;
                cursor = close + 1;
            }
            _ => {
                let start = cursor;
                while cursor < bytes.len() && bytes[cursor] != b'{' && bytes[cursor] != b'}' {
                    cursor += 1;
                }
                output.push_str(&template[start..cursor]);
                continue;
            }
        }

        if let Some(expression) = parse_expression(template, cursor - (bytes[cursor - 1] == b'}') as usize - 1, cursor - 1).ok() {
            let expanded = expand_expression(&expression, vars)?;
            output.push_str(&expanded);
        }
    }

    Ok(output)
}
