#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone)]
struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Expression {
    operator: Option<char>,
    specs: Vec<Specifier>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'_'
}

fn parse_expression(content: &str, content_start: usize, close_pos: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, body_start) = match bytes.first().copied() {
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
            (Some(bytes[0] as char), 1)
        }
        Some(b'=' | b'!' | b'@' | b'|' | b'$' | b'%' | b'^') => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => (None, 0),
    };

    if body_start == bytes.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let body = &content[body_start..];
    let body_offset = content_start + body_start;
    let mut specs = Vec::new();
    let mut used = std::collections::HashSet::new();
    let mut offset = 0;

    for part in body.split(',') {
        let start = body_offset + offset;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: start });
        }

        let part_bytes = part.as_bytes();
        let modifier_at = part_bytes
            .iter()
            .position(|&b| b == b'*' || b == b':')
            .unwrap_or(part_bytes.len());
        let name_bytes = &part_bytes[..modifier_at];

        if name_bytes.is_empty() || !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: start });
        }
        if let Some(i) = name_bytes.iter().position(|&b| !is_name_continue(b)) {
            return Err(Error::InvalidName { pos: start + i });
        }

        let name = std::str::from_utf8(name_bytes).unwrap().to_owned();
        let modifier = if modifier_at == part_bytes.len() {
            Modifier::None
        } else {
            let suffix = &part[modifier_at..];
            let modifier_pos = start + modifier_at;
            if suffix == "*" {
                Modifier::Explode
            } else if suffix.starts_with(':') {
                let digits = &suffix[1..];
                let valid = !digits.is_empty()
                    && digits.len() <= 4
                    && digits.as_bytes()[0].is_ascii_digit()
                    && digits.as_bytes()[0] != b'0'
                    && digits.bytes().all(|b| b.is_ascii_digit());
                if !valid {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                let n: usize = digits.parse().unwrap();
                if !(1..=9999).contains(&n) {
                    return Err(Error::InvalidModifier { pos: modifier_pos });
                }
                Modifier::Prefix(n)
            } else {
                return Err(Error::InvalidModifier { pos: modifier_pos });
            }
        };

        if !used.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        specs.push(Specifier {
            name,
            start,
            modifier,
        });

        offset += part.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn encode(input: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(h1), Some(h2)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(HEX[h1 as usize] as char);
                out.push(HEX[h2 as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn prefixed_chars(value: &str, count: usize) -> String {
    value.chars().take(count).collect()
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(key, _)| *key == name).map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: char) -> String {
    if operator == ';' && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expr.operator;
    let reserved = matches!(op, Some('+' | '#'));
    let separator = match op {
        Some('.') => '.',
        Some('/') => '/',
        Some(';') => ';',
        Some('?' | '&') => '&',
        _ => ',',
    };

    let mut fields = Vec::new();
    for spec in &expr.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        match value {
            Value::Scalar(raw) => {
                let value = match spec.modifier {
                    Modifier::Prefix(n) => prefixed_chars(raw, n),
                    Modifier::None | Modifier::Explode => raw.clone(),
                };
                fields.push(if matches!(op, Some(';' | '?' | '&')) {
                    named_field(&spec.name, encode(&value, reserved), op.unwrap())
                } else {
                    encode(&value, reserved)
                });
            }
            Value::List(items) => {
                if matches!(spec.modifier, Modifier::Prefix(_)) {
                    return Err(Error::TypeMismatch { pos: spec.start });
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if items.is_empty() {
                    continue;
                }
                if exploded {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if matches!(op, Some(';' | '?' | '&')) {
                            named_field(&spec.name, encoded, op.unwrap())
                        } else {
                            encoded
                        });
                    }
                } else {
                    let value = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(op, Some(';' | '?' | '&')) {
                        named_field(&spec.name, value, op.unwrap())
                    } else {
                        value
                    });
                }
            }
            Value::Map(entries) => {
                if matches!(spec.modifier, Modifier::Prefix(_)) {
                    return Err(Error::TypeMismatch { pos: spec.start });
                }
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }
                if entries.is_empty() {
                    continue;
                }

                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for (key, value) in entries {
                        let key = encode(key, reserved);
                        let value = encode(value, reserved);
                        fields.push(if op == Some(';') && value.is_empty() {
                            key
                        } else {
                            format!("{key}={value}")
                        });
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let value = components.join(",");
                    fields.push(if matches!(op, Some(';' | '?' | '&')) {
                        named_field(&spec.name, value, op.unwrap())
                    } else {
                        value
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    let prefix = match op {
        Some('#') => "#",
        Some('.') => ".",
        Some('/') => "/",
        Some(';') => ";",
        Some('?') => "?",
        Some('&') => "&",
        _ => "",
    };
    Ok(format!("{prefix}{}", fields.join(&separator.to_string())))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[cursor + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = cursor + 1 + relative_close;
                if let Some(relative_nested) = bytes[cursor + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: cursor + 1 + relative_nested,
                    });
                }

                let content = &template[cursor + 1..close];
                let parsed = parse_expression(content, open + 1, close)?;
                output.push_str(&expand_expression(&parsed, vars)?);
                cursor = close + 1;
            }
            _ => {
                let start = cursor;
                while cursor < bytes.len() && bytes[cursor] != b'{' && bytes[cursor] != b'}' {
                    cursor += 1;
                }
                output.push_str(&template[start..cursor]);
            }
        }
    }

    Ok(output)
}
