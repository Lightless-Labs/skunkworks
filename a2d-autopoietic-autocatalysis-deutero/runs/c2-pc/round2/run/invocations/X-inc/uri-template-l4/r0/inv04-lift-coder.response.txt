#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semi,
    Question,
    Amp,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug)]
struct Specifier<'a> {
    name: &'a str,
    modifier: Modifier,
    pos: usize,
}

#[derive(Debug)]
struct Expression<'a> {
    operator: Operator,
    specs: Vec<Specifier<'a>>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'_'
}

fn parse_specifier<'a>(text: &'a str, base: usize) -> Result<Specifier<'a>, Error> {
    let bytes = text.as_bytes();
    let modifier_at = bytes.iter().position(|&b| b == b'*' || b == b':');
    let name_end = modifier_at.unwrap_or(bytes.len());

    if name_end == 0 || !is_name_start(bytes[0]) {
        return Err(Error::InvalidName { pos: base });
    }
    for i in 1..name_end {
        if !is_name_continue(bytes[i]) {
            return Err(Error::InvalidName { pos: base + i });
        }
    }

    let modifier = if let Some(i) = modifier_at {
        let suffix = &text[i..];
        if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            if (1..=4).contains(&digits.len())
                && digits.as_bytes()[0].is_ascii_digit()
                && digits.as_bytes()[0] != b'0'
                && digits.bytes().all(|b| b.is_ascii_digit())
            {
                let n = digits
                    .bytes()
                    .fold(0usize, |n, b| n * 10 + (b - b'0') as usize);
                if (1..=9999).contains(&n) {
                    Modifier::Prefix(n)
                } else {
                    return Err(Error::InvalidModifier { pos: base + i });
                }
            } else {
                return Err(Error::InvalidModifier { pos: base + i });
            }
        } else {
            return Err(Error::InvalidModifier { pos: base + i });
        }
    } else {
        Modifier::None
    };

    Ok(Specifier {
        name: &text[..name_end],
        modifier,
        pos: base,
    })
}

fn parse_expression(
    content: &str,
    content_pos: usize,
    close_pos: usize,
) -> Result<Expression<'_>, Error> {
    let (operator, offset) = match content.as_bytes().first().copied() {
        Some(b'+') => (Operator::Plus, 1),
        Some(b'#') => (Operator::Hash, 1),
        Some(b'.') => (Operator::Dot, 1),
        Some(b'/') => (Operator::Slash, 1),
        Some(b';') => (Operator::Semi, 1),
        Some(b'?') => (Operator::Question, 1),
        Some(b'&') => (Operator::Amp, 1),
        _ => (Operator::Simple, 0),
    };

    let rest = &content[offset..];
    let rest_pos = content_pos + offset;
    if rest.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut used: Vec<&str> = Vec::new();
    let mut start = 0;
    loop {
        let end = rest[start..]
            .find(',')
            .map(|relative| start + relative)
            .unwrap_or(rest.len());
        if start == end {
            return Err(Error::EmptyVariable {
                pos: rest_pos + start,
            });
        }
        let spec = parse_specifier(&rest[start..end], rest_pos + start)?;
        if used.iter().any(|name| *name == spec.name) {
            return Err(Error::DuplicateVariable { pos: spec.pos });
        }
        used.push(spec.name);
        specs.push(spec);
        if end == rest.len() {
            break;
        }
        start = end + 1;
        if start == rest.len() {
            return Err(Error::EmptyVariable { pos: close_pos });
        }
    }

    Ok(Expression { operator, specs })
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression<'_>)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(relative) => i + 1 + relative,
                    None => return Err(Error::Unclosed { pos: open }),
                };
                if let Some(relative) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + relative });
                }
                let expression =
                    parse_expression(&template[i + 1..close], i + 1, close)?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/'
            | b'?'
            | b'#'
            | b'['
            | b']'
            | b'@'
            | b'!'
            | b'$'
            | b'&'
            | b'\''
            | b'('
            | b')'
            | b'*'
            | b'+'
            | b','
            | b';'
            | b'='
    )
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(n: u8) -> char {
    match n {
        0..=9 => (b'0' + n) as char,
        _ => (b'A' + n - 10) as char,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(hi), Some(lo)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                out.push('%');
                out.push(hex_digit(hi));
                out.push(hex_digit(lo));
                i += 3;
                continue;
            }
        }
        if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(hex_digit(b >> 4));
            out.push(hex_digit(b & 0x0f));
        }
        i += 1;
    }

    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if operator == Operator::Semi && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(
    expression: &Expression<'_>,
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let operator = expression.operator;
    let reserved = matches!(operator, Operator::Plus | Operator::Hash);
    let mut fields = Vec::new();

    for spec in &expression.specs {
        let Some(value) = lookup(spec.name, vars) else {
            continue;
        };

        let prefix = match spec.modifier {
            Modifier::Prefix(n) => {
                if let Value::Scalar(text) = value {
                    Some(text.chars().take(n).collect::<String>())
                } else {
                    return Err(Error::TypeMismatch { pos: spec.pos });
                }
            }
            _ => None,
        };

        match value {
            Value::Scalar(original) => {
                let selected = prefix.as_deref().unwrap_or(original);
                fields.push(match operator {
                    Operator::Semi | Operator::Question | Operator::Amp => {
                        named_field(spec.name, encode(selected, reserved), operator)
                    }
                    _ => encode(selected, reserved),
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if spec.modifier == Modifier::Explode {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(match operator {
                            Operator::Semi | Operator::Question | Operator::Amp => {
                                named_field(spec.name, encoded, operator)
                            }
                            _ => encoded,
                        });
                    }
                } else {
                    let value = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(match operator {
                        Operator::Semi | Operator::Question | Operator::Amp => {
                            named_field(spec.name, value, operator)
                        }
                        _ => value,
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys: Vec<&str> = Vec::new();
                for (key, _) in entries {
                    if keys.iter().any(|seen| *seen == key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                    keys.push(key);
                }
                if entries.is_empty() {
                    continue;
                }
                if spec.modifier == Modifier::Explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        fields.push(if operator == Operator::Semi && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        });
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let value = components.join(",");
                    fields.push(match operator {
                        Operator::Semi | Operator::Question | Operator::Amp => {
                            named_field(spec.name, value, operator)
                        }
                        _ => value,
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (separator, prefix) = match operator {
        Operator::Simple | Operator::Plus => (",", ""),
        Operator::Hash => (",", "#"),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semi => (";", ";"),
        Operator::Question => ("&", "?"),
        Operator::Amp => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::with_capacity(template.len());
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }
    output.push_str(&template[cursor..]);

    Ok(output)
}
