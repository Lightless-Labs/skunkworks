#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct Runtime {
    state: State,
    remaining: Option<u64>,
    segment_started: u64,
}

pub struct Pool {
    jobs: Vec<Job>,
    deps: Vec<Vec<usize>>,
    runtime: Vec<Runtime>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base < 1
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_name(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if !valid_name(tag, 1, 8) {
                    return Err(Error::BadTag(i));
                }
            }
            if ids.insert(job.id.as_str(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut deps = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match ids.get(dep.as_str()) {
                    Some(&idx) => deps[i].push(idx),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
        }

        fn visit(i: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[i] == 1 {
                return false;
            }
            if marks[i] == 2 {
                return true;
            }
            marks[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, marks) {
                    return false;
                }
            }
            marks[i] = 2;
            true
        }
        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &deps, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let runtime = jobs
            .iter()
            .map(|job| Runtime {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                remaining: None,
                segment_started: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            deps,
            runtime,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.jobs
            .iter()
            .position(|job| job.id == id)
            .map(|i| self.runtime[i].state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if self.has_run && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut now = if self.has_run { self.horizon } else { 0 };
        let mut first = !self.has_run;

        loop {
            if first || now > self.horizon || !self.has_run {
                self.process_timestamp(now, &mut lines);
            }
            first = false;

            if now == until {
                break;
            }

            let next = self.next_event(now, until);
            match next {
                Some(t) => now = t,
                None => break,
            }
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    fn next_event(&self, now: u64, until: u64) -> Option<u64> {
        let mut next = None;
        for r in &self.runtime {
            let t = match r.state {
                State::Waiting { ready_at, .. } => Some(ready_at),
                State::Running { finish_at, .. } => Some(finish_at),
                _ => None,
            };
            if let Some(t) = t {
                if t > now && t <= until && next.map_or(true, |n| t < n) {
                    next = Some(t);
                }
            }
        }
        next
    }

    fn process_timestamp(&mut self, now: u64, lines: &mut Vec<String>) {
        let mut completed: Vec<usize> = self
            .runtime
            .iter()
            .enumerate()
            .filter_map(|(i, r)| match r.state {
                State::Running { finish_at, .. } if finish_at == now => Some(i),
                _ => None,
            })
            .collect();
        completed.sort_by(|&a, &b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()));

        for i in completed {
            let attempt = match self.runtime[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let ok = attempt > self.jobs[i].failures;
            lines.push(format!(
                "{} finish {} {} {}",
                now,
                self.jobs[i].id,
                attempt,
                if ok { "ok" } else { "fail" }
            ));
            if ok {
                self.runtime[i].state = State::Succeeded;
                self.runtime[i].remaining = None;
            } else if attempt >= self.jobs[i].max_attempts {
                self.runtime[i].state = State::Failed;
                self.runtime[i].remaining = None;
            } else {
                let shift = (attempt - 1) as u32;
                let delay = self
                    .retry_base
                    .checked_shl(shift)
                    .unwrap_or(u64::MAX)
                    .min(self.retry_cap);
                let ready = now.saturating_add(delay);
                self.runtime[i].state = State::Waiting {
                    ready_at: ready,
                    attempt,
                };
                self.runtime[i].remaining = None;
                lines.push(format!("{} retry {} {} {}", now, self.jobs[i].id, attempt, ready));
            }
        }

        loop {
            let mut newly_blocked = Vec::new();
            for i in 0..self.jobs.len() {
                if matches!(self.runtime[i].state, State::Waiting { .. })
                    && self.deps[i].iter().any(|&d| {
                        matches!(self.runtime[d].state, State::Failed | State::Blocked)
                    })
                {
                    newly_blocked.push(i);
                }
            }
            if newly_blocked.is_empty() {
                break;
            }
            for &i in &newly_blocked {
                self.runtime[i].state = State::Blocked;
            }
            newly_blocked.sort_by(|&a, &b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()));
            for i in newly_blocked {
                let dep = self.deps[i]
                    .iter()
                    .filter(|&&d| matches!(self.runtime[d].state, State::Failed | State::Blocked))
                    .min_by(|&&a, &&b| self.jobs[a].id.as_bytes().cmp(self.jobs[b].id.as_bytes()))
                    .map(|&d| self.jobs[d].id.as_str())
                    .unwrap_or("");
                lines.push(format!("{} skip {} {}", now, self.jobs[i].id, dep));
            }
        }

        loop {
            let mut candidates: Vec<usize> = (0..self.jobs.len())
                .filter(|&i| match self.runtime[i].state {
                    State::Waiting { ready_at, .. } => {
                        ready_at <= now
                            && self.deps[i]
                                .iter()
                                .all(|&d| matches!(self.runtime[d].state, State::Succeeded))
                    }
                    _ => false,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, now);
                let pb = self.effective_priority(b, now);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut started = false;
            for candidate in candidates {
                if self.can_start_direct(candidate) {
                    self.start(candidate, now, lines);
                    started = true;
                    break;
                }
                if let Some(victim) = self.choose_victim(candidate) {
                    let remaining = match self.runtime[victim].state {
                        State::Running { finish_at, .. } => finish_at - now,
                        _ => 0,
                    };
                    let attempt = match self.runtime[victim].state {
                        State::Running { attempt, .. } => attempt,
                        _ => 0,
                    };
                    lines.push(format!(
                        "{} preempt {} by {}",
                        now, self.jobs[victim].id, self.jobs[candidate].id
                    ));
                    self.runtime[victim].state = State::Waiting {
                        ready_at: now,
                        attempt,
                    };
                    self.runtime[victim].remaining = Some(remaining);
                    self.start(candidate, now, lines);
                    started = true;
                    break;
                }
            }
            if !started {
                break;
            }
        }
    }

    fn effective_priority(&self, i: usize, now: u64) -> u8 {
        if self.aging_ms == 0 {
            return self.jobs[i].priority;
        }
        let ready = match self.runtime[i].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => now,
        };
        let age = now.saturating_sub(ready) / self.aging_ms;
        (self.jobs[i].priority as u64 + age).min(9) as u8
    }

    fn can_start_direct(&self, i: usize) -> bool {
        let running: Vec<usize> = self
            .runtime
            .iter()
            .enumerate()
            .filter_map(|(j, r)| matches!(r.state, State::Running { .. }).then_some(j))
            .collect();
        running.len() < self.capacity && self.tag_free(i, &running)
    }

    fn tag_free(&self, i: usize, running: &[usize]) -> bool {
        match &self.jobs[i].tag {
            None => true,
            Some(tag) => !running
                .iter()
                .any(|&j| self.jobs[j].tag.as_ref() == Some(tag)),
        }
    }

    fn choose_victim(&self, candidate: usize) -> Option<usize> {
        let running: Vec<usize> = self
            .runtime
            .iter()
            .enumerate()
            .filter_map(|(j, r)| matches!(r.state, State::Running { .. }).then_some(j))
            .collect();
        let candidate_tag = self.jobs[candidate].tag.as_ref();
        let mut eligible: Vec<usize> = running
            .iter()
            .copied()
            .filter(|&v| {
                self.jobs[candidate].priority >= self.jobs[v].priority.saturating_add(2)
                    && running.len() - 1 < self.capacity
                    && candidate_tag.map_or(true, |tag| {
                        !running
                            .iter()
                            .any(|&j| j != v && self.jobs[j].tag.as_ref() == Some(tag))
                    })
            })
            .collect();

        eligible.sort_by(|&a, &b| {
            self.jobs[a]
                .priority
                .cmp(&self.jobs[b].priority)
                .then_with(|| {
                    let sa = match self.runtime[a].state {
                        State::Running { .. } => self.runtime[a].segment_started,
                        _ => 0,
                    };
                    let sb = match self.runtime[b].state {
                        State::Running { .. } => self.runtime[b].segment_started,
                        _ => 0,
                    };
                    sb.cmp(&sa)
                })
                .then_with(|| self.jobs[b].id.as_bytes().cmp(self.jobs[a].id.as_bytes()))
        });
        eligible.into_iter().next()
    }

    fn start(&mut self, i: usize, now: u64, lines: &mut Vec<String>) {
        let (attempt, remaining) = match self.runtime[i].state {
            State::Waiting { attempt, .. } => {
                if let Some(rem) = self.runtime[i].remaining.take() {
                    (attempt, rem)
                } else {
                    (attempt + 1, self.jobs[i].duration)
                }
            }
            _ => return,
        };
        let finish_at = now.saturating_add(remaining);
        self.runtime[i].state = State::Running { attempt, finish_at };
        self.runtime[i].segment_started = now;
        lines.push(format!(
            "{} start {} {} {}",
            now, self.jobs[i].id, attempt, remaining
        ));
    }
}

fn valid_name(s: &str, min_len: usize, max_len: usize) -> bool {
    let b = s.as_bytes();
    (min_len..=max_len).contains(&b.len())
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}
