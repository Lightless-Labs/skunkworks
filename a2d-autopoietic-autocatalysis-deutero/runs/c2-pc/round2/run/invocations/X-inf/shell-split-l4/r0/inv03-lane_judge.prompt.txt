pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    Bare,
    Single,
    Double,
}

struct WordBuilder {
    text: String,
    glob: Vec<usize>,
    active: bool,
}

impl WordBuilder {
    fn new() -> Self {
        Self {
            text: String::new(),
            glob: Vec::new(),
            active: false,
        }
    }

    fn put(&mut self, value: &str, wildcards: bool) {
        for ch in value.chars() {
            if wildcards && matches!(ch, '*' | '?' | '[') {
                self.glob.push(self.text.len());
            }
            self.text.push(ch);
        }
    }

    fn push(&mut self, ch: char, wildcards: bool) {
        if wildcards && matches!(ch, '*' | '?' | '[') {
            self.glob.push(self.text.len());
        }
        self.text.push(ch);
    }

    fn flush(&mut self, output: &mut Vec<Word>) {
        if self.active {
            output.push(Word {
                text: std::mem::take(&mut self.text),
                glob: std::mem::take(&mut self.glob),
            });
            self.active = false;
        }
    }
}

fn is_name_start(ch: char) -> bool {
    ch.is_ascii_alphabetic() || ch == '_'
}

fn is_name_char(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    for &(key, value) in env {
        if key == name {
            return Some(value);
        }
    }
    None
}

fn find_brace_end(input: &str, mut i: usize) -> Option<usize> {
    while i < input.len() {
        let ch = input[i..].chars().next()?;
        let n = i + ch.len_utf8();
        if ch == '\\' {
            if n < input.len() {
                let escaped = input[n..].chars().next()?;
                i = n + escaped.len_utf8();
            } else {
                i = n;
            }
        } else if ch == '}' {
            return Some(i);
        } else {
            i = n;
        }
    }
    None
}

fn decode(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let ch = text[i..].chars().next().unwrap();
        let next = i + ch.len_utf8();
        if ch == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            result.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            result.push(ch);
            i = next;
        }
    }
    result
}

fn braced_value<'a>(
    input: &'a str,
    dollar: usize,
    start: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let end = find_brace_end(input, start)
        .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[start..end];

    let mut name_end = 0;
    for ch in contents.chars() {
        if (name_end == 0 && is_name_start(ch)) || (name_end != 0 && is_name_char(ch)) {
            name_end += ch.len_utf8();
        } else {
            break;
        }
    }
    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let trailing = &contents[name_end..];
    if trailing.is_empty() {
        return Ok((lookup(env, name).unwrap_or("").to_owned(), end + 1));
    }

    let default = trailing
        .strip_prefix(":-")
        .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
    if let Some(value) = lookup(env, name).filter(|value| !value.is_empty()) {
        Ok((value.to_owned(), end + 1))
    } else {
        Ok((decode(default), end + 1))
    }
}

fn variable_end(input: &str, name_start: usize) -> usize {
    let mut i = name_start;
    while i < input.len() {
        let ch = input[i..].chars().next().unwrap();
        if !is_name_char(ch) {
            break;
        }
        i += ch.len_utf8();
    }
    i
}

fn append_expansion(word: &mut WordBuilder, value: &str, quoted: bool) {
    word.active = true;
    word.put(value, !quoted);
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|byte| *byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut word = WordBuilder::new();
    let mut quote = Quote::Bare;
    let mut opening = 0usize;
    let mut cursor = 0usize;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let after = cursor + ch.len_utf8();

        match quote {
            Quote::Single => {
                if ch == '\'' {
                    quote = Quote::Bare;
                } else {
                    word.push(ch, false);
                }
                cursor = after;
            }
            Quote::Double => {
                if ch == '"' {
                    quote = Quote::Bare;
                    cursor = after;
                } else if ch == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[after..].chars().next().unwrap();
                    let end = after + escaped.len_utf8();
                    match escaped {
                        '"' | '\\' | '$' => {
                            word.push(escaped, false);
                            cursor = end;
                        }
                        '\n' => cursor = end,
                        _ => {
                            word.push('\\', false);
                            cursor = after;
                        }
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if input[after..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[after..].starts_with('{') {
                        let (value, end) = braced_value(input, cursor, after + 1, env)?;
                        append_expansion(&mut word, &value, true);
                        cursor = end;
                    } else if let Some(next) = input[after..].chars().next() {

                            cursor = end;
                        } else {
                            word.active = true;
                            word.push('$', false);
                            cursor = after;
                        }
                    } else {
                        word.active = true;
                        word.push('$', false);
                        cursor = after;
                    }
                } else {
                    word.push(ch, false);
                    cursor = after;
                }
            }
            Quote::Bare => {
                if matches!(ch, ' ' | '\t' | '\r' | '\n') {
                    word.flush(&mut result);
                    cursor = after;
                } else if ch == '\'' {
                    word.active = true;
                    quote = Quote::Single;
                    opening = cursor;
                    cursor = after;
                } else if ch == '"' {
                    word.active = true;
                    quote = Quote::Double;
                    opening = cursor;
                    cursor = after;
                } else if ch == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[after..].chars().next().unwrap();
                    let end = after + escaped.len_utf8();
                    if escaped == '\n' {
                        cursor = end;
                    } else {
                        word.active = true;
                        word.push(escaped, false);
                        cursor = end;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if input[after..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if input[after..].starts_with('{') {
                        let (value, end) = braced_value(input, cursor, after + 1, env)?;
                        append_expansion(&mut word, &value, false);
                        cursor = end;
                    } else if let Some(next) = input[after..].chars().next() {

                            cursor = end;
                        } else {

                            word.push('$', true);
                            cursor = after;
                        }
                    } else {
                        word.active = true;
                        word.push('$', true);
                        cursor = after;
                    }
                } else {
                    word.active = true;
                    word.push(ch, true);
                    cursor = after;
                }
            }
        }
    }

        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: opening }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: opening }),
        Quote::Bare => {}
    }

    word.flush(&mut result);
    Ok(result)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut encoded = String::new();
    for ch in word.chars() {
        if ch.is_ascii_alphanumeric() || "_-./:=,+@%".contains(ch) {
            encoded.push(ch);
        } else {
            encoded.push('\\');
            encoded.push(ch);
        }
    }
    Some(encoded)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        None
    } else {
        let mut encoded = String::from("\"");
        for ch in word.chars() {
            if matches!(ch, '\\' | '"' | '$') {
                encoded.push('\\');
            }
            encoded.push(ch);
        }
        encoded.push('"');
        Some(encoded)
    }
}

fn encode_word(word: &str) -> String {
    let candidates = [
        bare_candidate(word).map(|s| (0u8, s)),
        single_candidate(word).map(|s| (1u8, s)),
        double_candidate(word).map(|s| (2u8, s)),
    ];

    if let Some((_, selected)) = candidates
        .into_iter()
        .flatten()
        .min_by_key(|(priority, candidate)| (candidate.len(), *priority))
    {
        return selected;
    }

    let mut fallback = String::from("'");
    for ch in word.chars() {
        if ch == '\'' {
            fallback.push('\'');
            fallback.push('\\');
            fallback.push('\'');
            fallback.push('\'');
        } else {
            fallback.push(ch);
        }
    }
    fallback.push('\'');
    fallback
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|byte| *byte == 0) {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut output = String::new();
    for (index, word) in words.iter().enumerate() {
        if index != 0 {
            output.push(' ');
        }
        output.push_str(&encode_word(word));
    }
    Ok(output)
}

        }
    }

    fn env_value(&self, name: &str) -> Option<&'a str> {
        self.env
            .iter()
            .find_map(|(key, value)| (*key == name).then_some(*value))
    }

    fn parse_name_end(s: &str, start: usize) -> Option<usize> {
        let bytes = s.as_bytes();
        if start >= bytes.len()
            || !(bytes[start].is_ascii_alphabetic() || bytes[start] == b'_')
        {
            return None;
        }
        let mut i = start + 1;
        while i < bytes.len()
            && (bytes[i].is_ascii_alphanumeric() || bytes[i] == b'_')
        {
            i += 1;
        }
        Some(i)
    }

    fn decode_default(raw: &str) -> String {
        let mut out = String::new();
        let mut it = raw.chars();
        while let Some(ch) = it.next() {
            if ch == '\\' {
                if let Some(next) = it.next() {
                    out.push(next);
                }
            } else {
                out.push(ch);
            }
        }
        out
    }

    fn parse_braced(&mut self, dollar: usize, wildcard: bool) -> Result<(), SplitError> {
        let bytes = self.input.as_bytes();
        let mut close = self.pos + 2;
        while close < bytes.len() {
            if bytes[close] == b'\\' {
                close += 1;
                if close < bytes.len() {
                    let ch = self.input[close..].chars().next().unwrap();
                    close += ch.len_utf8();
                }
            } else if bytes[close] == b'}' {
                break;
            } else {
                let ch = self.input[close..].chars().next().unwrap();
                close += ch.len_utf8();
            }
        }

        if close >= bytes.len() {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        }

        let contents = &self.input[self.pos + 2..close];
        let name_end = Self::parse_name_end(contents, 0)
            .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
        let name = &contents[..name_end];

        let selected = if name_end == contents.len() {
            self.env_value(name).unwrap_or("")
        } else if contents[name_end..].starts_with(":-") {
            let raw_default = &contents[name_end + 2..];
            match self.env_value(name) {
                Some(value) if !value.is_empty() => value,
                _ => {
                    let decoded = Self::decode_default(raw_default);
                    self.push_expansion(&decoded, wildcard);
                    self.pos = close + 1;
                    self.active = true;
                    return Ok(());
                }
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        self.push_expansion(selected, wildcard);
        self.pos = close + 1;
        self.active = true;
        Ok(())
    }

    fn parse_dollar(&mut self, quote: Quote) -> Result<(), SplitError> {
        let dollar = self.pos;
        let next_pos = dollar + 1;
        let rest = &self.input[next_pos..];

        if rest.starts_with('(') {
            return Err(SplitError::CommandSubstitution { pos: dollar });
        }
        if rest.starts_with('{') {
            self.parse_braced(dollar, quote == Quote::None)?;
            return Ok(());
        }
        if let Some(end) = Self::parse_name_end(self.input, next_pos) {
            let name = &self.input[next_pos..end];
            let value = self.env_value(name).unwrap_or("");
            self.push_expansion(value, quote == Quote::None);
            self.pos = end;
            self.active = true;
            return Ok(());
        }

        self.push_scalar('$', quote == Quote::None);
        self.pos += 1;
        Ok(())
    }

    fn run(mut self) -> Result<Vec<Word>, SplitError> {
        let mut quote = Quote::None;
        let mut quote_start = 0;
        let bytes = self.input.as_bytes();

        while self.pos < bytes.len() {
            let ch = self.input[self.pos..].chars().next().unwrap();
            let width = ch.len_utf8();

            match quote {
                Quote::Single => {
                    if ch == '\'' {
                        quote = Quote::None;
                        self.pos += width;
                    } else {
                        self.push_scalar(ch, false);
                        self.pos += width;
                    }
                }
                Quote::Double => match ch {
                    '"' => {
                        quote = Quote::None;
                        self.pos += width;
                    }
                    '\\' => {
                        let slash = self.pos;
                        let next_pos = slash + 1;
                        if next_pos >= bytes.len() {
                            return Err(SplitError::TrailingEscape { pos: slash });
                        }
                        let next = self.input[next_pos..].chars().next().unwrap();
                        if matches!(next, '"' | '\\' | '$' | '\n') {
                            if next != '\n' {
                                self.push_scalar(next, false);
                            }
                            self.pos = next_pos + next.len_utf8();
                        } else {
                            self.push_scalar('\\', false);
                            self.pos += 1;
                        }
                    }
                    '$' => self.parse_dollar(Quote::Double)?,
                    '`' => return Err(SplitError::CommandSubstitution { pos: self.pos }),
                    _ => {
                        self.push_scalar(ch, false);
                        self.pos += width;
                    }
                },
                Quote::None => match ch {
                    ' ' | '\t' | '\r' | '\n' => {
                        self.finish_word();
                        self.pos += width;
                    }
                    '\'' | '"' => {
                        quote = if ch == '\'' { Quote::Single } else { Quote::Double };
                        quote_start = self.pos;
                        self.active = true;
                        self.pos += width;
                    }
                    '\\' => {
                        let slash = self.pos;
                        let next_pos = slash + 1;
                        if next_pos >= bytes.len() {
                            return Err(SplitError::TrailingEscape { pos: slash });
                        }
                        let next = self.input[next_pos..].chars().next().unwrap();
                        self.pos = next_pos + next.len_utf8();
                        if next != '\n' {
                            self.push_scalar(next, false);
                            self.active = true;
                        }
                    }
                    '$' => self.parse_dollar(Quote::None)?,
                    '`' => return Err(SplitError::CommandSubstitution { pos: self.pos }),
                    _ => {
                        self.push_scalar(ch, true);
                        self.active = true;
                        self.pos += width;
                    }
                },
            }
        }

        match quote {
            Quote::Single => {
                return Err(SplitError::UnterminatedSingle { pos: quote_start });
            }
            Quote::Double => {
                return Err(SplitError::UnterminatedDouble { pos: quote_start });
            }
            Quote::None => {}
        }

        self.finish_word();
        Ok(self.words)
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }
    Parser::new(input, env).run()
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|&b| b == 0) {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates: Vec<(usize, u8, String)> = Vec::new();

        if !word.is_empty() && !word.contains('\n') {
            let mut bare = String::new();
            for ch in word.chars() {
                if ch.is_ascii_alphanumeric() || "_-./:=,+@%".contains(ch) {
                    bare.push(ch);
                } else {
                    bare.push('\\');
                    bare.push(ch);
                }
            }
            candidates.push((bare.len(), 0, bare));
        }

        if !word.contains('\'') {
            candidates.push((word.len() + 2, 1, format!("'{}'", word)));
        }

        if !word.contains('`') {
            let mut double = String::new();
            double.push('"');
            for ch in word.chars() {
                if matches!(ch, '\\' | '"' | '$') {
                    double.push('\\');
                }
                double.push(ch);
            }
            double.push('"');
            candidates.push((double.len(), 2, double));
        }

        let choice = if let Some(best) = candidates
            .into_iter()
            .min_by_key(|(len, priority, _)| (*len, *priority))
        {
            best.2
        } else {
            let mut fallback = String::new();
            fallback.push('\'');
            for ch in word.chars() {
                if ch == '\'' {
                    fallback.push('\'');
                    fallback.push('\\');
                    fallback.push('\'');
                    fallback.push('\'');
                } else {
                    fallback.push(ch);
                }
            }
            fallback.push('\'');
            fallback
        };
        encoded.push(choice);
    }

    Ok(encoded.join(" "))
}

    let mut pos = 0;
    while pos < raw.len() {
        let ch = raw[pos..].chars().next().unwrap();
        if ch == '\\' {
            pos += 1;
            if pos < raw.len() {
                let escaped = raw[pos..].chars().next().unwrap();

            out.push(ch);

fn expansion_end(input: &str, mut pos: usize) -> Option<usize> {
    let bytes = input.as_bytes();
    while pos < bytes.len() {
        if bytes[pos] == b'}' {
            return Some(pos);
        }
        if bytes[pos] == b'\\' {
            pos += 1;
            if pos < bytes.len() {
                let ch = input[pos..].chars().next().unwrap();
                pos += ch.len_utf8();
            }
        } else {
            let ch = input[pos..].chars().next().unwrap();
            pos += ch.len_utf8();
        }
    }
    None
}

fn parse_expansion<'a>(
    input: &str,
    pos: &mut usize,
    env: &'a [(&str, &str)],
    glob_active: bool,
    output: &mut Word,
) -> Result<(), SplitError> {
    let dollar = *pos;
    let bytes = input.as_bytes();

    if bytes.get(dollar + 1) == Some(&b'{') {
        let body_start = dollar + 2;
        let close = expansion_end(input, body_start)
            .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let body = &input[body_start..close];

        let name_end = body
            .as_bytes()
            .first()
            .copied()
            .filter(|b| name_start(*b))
            .map(|_| {
                body.as_bytes()
                    .iter()
                    .position(|b| !name_continue(*b))
                    .unwrap_or(body.len())
            })
            .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
        let name = &body[..name_end];

        let value = if name_end == body.len() {
            lookup(env, name).unwrap_or("")
        } else if body[name_end..].starts_with(":-") {
            match lookup(env, name) {
                Some(value) if !value.is_empty() => value,
                _ => {
                    append(output, &decode_default(&body[name_end + 2..]), glob_active);
                    *pos = close + 1;
                    return Ok(());
                }
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        append(output, value, glob_active);
        *pos = close + 1;
        return Ok(());
    }

    let start = dollar + 1;
    if start < bytes.len() && name_start(bytes[start]) {
        let mut end = start + 1;
        while end < bytes.len() && name_continue(bytes[end]) {
            end += 1;
        }
        let value = lookup(env, &input[start..end]).unwrap_or("");
        append(output, value, glob_active);
        *pos = end;
    } else {
        output.text.push('$');
        *pos += 1;
    }
    Ok(())
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|b| *b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let bytes = input.as_bytes();
    let mut pos = 0;
    let mut quote = 0u8;
    let mut quote_start = 0;
    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;

    while pos < bytes.len() {
        let ch = input[pos..].chars().next().unwrap();
        let width = ch.len_utf8();

        if quote == b'\'' {
            if ch == '\'' {
                quote = 0;
                pos += 1;
            } else {
                word.text.push(ch);
                pos += width;
            }
            continue;
        }

        if quote == b'"' {
            match ch {
                '"' => {
                    quote = 0;
                    pos += 1;
                }
                '\\' => {
                    let slash = pos;
                    if slash + 1 >= bytes.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[slash + 1..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.text.push(next);
                        }
                        pos = slash + 1 + next.len_utf8();
                    } else {
                        word.text.push('\\');
                        pos += 1;
                    }
                }
                '$' => {
                    if input[pos + 1..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos });
                    }
                    parse_expansion(input, &mut pos, env, false, &mut word)?;
                    active = true;
                }
                '`' => return Err(SplitError::CommandSubstitution { pos }),
                _ => {
                    word.text.push(ch);
                    pos += width;
                }
            }
            continue;
        }

        match ch {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    words.push(std::mem::replace(
                        &mut word,
                        Word { text: String::new(), glob: Vec::new() },
                    ));
                    active = false;
                }
                pos += width;
            }
            '\'' | '"' => {
                quote = if ch == '\'' { b'\'' } else { b'"' };
                quote_start = pos;
                active = true;
                pos += 1;
            }
            '\\' => {
                let slash = pos;
                if slash + 1 >= bytes.len() {
                    return Err(SplitError::TrailingEscape { pos: slash });
                }
                let next = input[slash + 1..].chars().next().unwrap();
                pos = slash + 1 + next.len_utf8();
                if next != '\n' {
                    word.text.push(next);
                    active = true;
                }
            }
            '$' => {
                if input[pos + 1..].starts_with('(') {
                    return Err(SplitError::CommandSubstitution { pos });
                }
                parse_expansion(input, &mut pos, env, true, &mut word)?;
                active = true;
            }
            '`' => return Err(SplitError::CommandSubstitution { pos }),
            _ => {
                append(&mut word, &ch.to_string(), true);
                active = true;
                pos += width;
            }
        }
    }

    if quote == b'\'' {
        return Err(SplitError::UnterminatedSingle { pos: quote_start });
    }
    if quote == b'"' {
        return Err(SplitError::UnterminatedDouble { pos: quote_start });
    }
    if active {
        words.push(word);
    }
    Ok(words)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word, text) in words.iter().enumerate() {
        if let Some(pos) = text.as_bytes().iter().position(|b| *b == 0) {
            return Err(JoinError::Nul { word, pos });
        }
    }

    let mut output = String::new();
    for (index, text) in words.iter().enumerate() {
        if index > 0 {
            output.push(' ');
        }

        let bare = if !text.is_empty() && !text.contains('\n') {
            Some(text.chars().fold(String::new(), |mut s, ch| {
                if ch.is_ascii_alphanumeric() || "_-./:=,+@%".contains(ch) {
                    s.push(ch);
                } else {
                    s.push('\\');
                    s.push(ch);
                }
                s
            }))
        } else {
            None
        };

        let single = if !text.contains('\'') {
            Some(format!("'{}'", text))
        } else {
            None
        };

        let double = if !text.contains('`') {
            let mut s = String::from("\"");
            for ch in text.chars() {
                if matches!(ch, '\\' | '"' | '$') {
                    s.push('\\');
                }
                s.push(ch);
            }
            s.push('"');
            Some(s)
        } else {
            None
        };

        let chosen = [bare, single, double]
            .into_iter()
            .flatten()
            .min_by_key(String::len);

        if let Some(candidate) = chosen {
            output.push_str(&candidate);
        } else {
            output.push('\'');
            for ch in text.chars() {
                if ch == '\'' {
                    output.push_str("'\\''");
                } else {
                    output.push(ch);
                }
            }
            output.push('\'');
        }
    }
    Ok(output)
}

    UnterminatedSingle { pos: usize },

    UnterminatedExpansion { pos: usize },

    CommandSubstitution { pos: usize },

pub struct Word {

    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {

#[derive(Clone, Copy, PartialEq, Eq)]
enum Mode {
    Bare,
    Single,
    Double,
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append(word: &mut Word, s: &str, wildcard_active: bool) {
    let offset = word.text.len();
    word.text.push_str(s);
    if wildcard_active {
        for (i, c) in s.char_indices() {
            if matches!(c, '*' | '?' | '[') {
                word.glob.push(offset + i);
            }
        }
    }
}

fn lookup<'a>(env: &'a [(&'a str, &'a str)], name: &str) -> &'a str {
    env.iter()
        .find_map(|(k, v)| if *k == name { Some(*v) } else { None })
        .unwrap_or("")
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word { text: String::new(), glob: Vec::new() };
    let mut active = false;
    let mut mode = Mode::Bare;
    let mut quote_open = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match mode {
            Mode::Single => {
                if c == '\'' {
                    mode = Mode::Bare;
                } else {
                    append(&mut word, &input[i..i + n], false);
                }
                i += n;
            }
            Mode::Double => {
                if c == '"' {
                    mode = Mode::Bare;
                    i += n;
                } else if c == '\\' {
                    let p = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: p });
                    }
                    let next = input[i..].chars().next().unwrap();
                    let nn = next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append(&mut word, &input[i..i + nn], false);
                        }
                        i += nn;
                    } else {
                        append(&mut word, "\\", false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let dollar = i;
                        let mut close = None;
                        let mut j = i + n + 1;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            if ch == '\\' {
                                j += 1;
                                if j < input.len() {
                                    let escaped = input[j..].chars().next().unwrap();
                                    j += escaped.len_utf8();
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += ch.len_utf8();
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[dollar + 2..close];
                        let (name, default) = parse_braced(contents)
                            .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
                        let value = match default {
                            Some(text) => {
                                let found = env.iter().find_map(|(k, v)| {
                                    if *k == name { Some(*v) } else { None }
                                }).unwrap_or("");
                                if found.is_empty() {
                                    decode_default(text)
                                } else {
                                    found.to_string()
                                }
                            }
                            None => lookup(env, name).to_string(),
                        };
                        append(&mut word, &value, false);
                        active = true;
                        i = close + 1;
                    } else {
                        let start = i + n;
                        if start < input.len() {
                            let first = input[start..].chars().next().unwrap();
                            if is_name_start(first) {
                                let mut end = start + first.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if is_name_char(ch) {
                                        end += ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                let name = &input[start..end];
                                let value = lookup(env, name).to_string();
                                append(&mut word, &value, false);
                                active = true;
                                i = end;
                            } else {
                                append(&mut word, "$", false);
                                i += n;
                            }
                        } else {
                            append(&mut word, "$", false);
                            i += n;
                        }
                    }
                } else {
                    append(&mut word, &input[i..i + n], false);
                    i += n;
                }
            }
            Mode::Bare => {
                if is_sep(c) {
                    if active {
                        words.push(word);
                        word = Word { text: String::new(), glob: Vec::new() };
                        active = false;
                    }
                    i += n;
                } else if c == '\'' || c == '"' {
                    active = true;
                    mode = if c == '\'' { Mode::Single } else { Mode::Double };
                    quote_open = i;
                    i += n;
                } else if c == '\\' {
                    let p = i;
                    i += 1;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos: p });
                    }
                    let next = input[i..].chars().next().unwrap();
                    let nn = next.len_utf8();
                    if next != '\n' {
                        append(&mut word, &input[i..i + nn], false);
                        active = true;
                    }
                    i += nn;
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let dollar = i;
                        let mut close = None;
                        let mut j = i + n + 1;
                        while j < input.len() {
                            let ch = input[j..].chars().next().unwrap();
                            if ch == '\\' {
                                j += 1;
                                if j < input.len() {
                                    let escaped = input[j..].chars().next().unwrap();
                                    j += escaped.len_utf8();
                                }
                            } else if ch == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += ch.len_utf8();
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[dollar + 2..close];
                        let (name, default) = parse_braced(contents)
                            .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
                        let value = match default {
                            Some(text) => {
                                let found = env.iter().find_map(|(k, v)| {
                                    if *k == name { Some(*v) } else { None }
                                }).unwrap_or("");
                                if found.is_empty() {
                                    decode_default(text)
                                } else {
                                    found.to_string()
                                }
                            }
                            None => lookup(env, name).to_string(),
                        };
                        append(&mut word, &value, true);
                        active = true;
                        i = close + 1;
                    } else {
                        let start = i + n;
                        if start < input.len() {
                            let first = input[start..].chars().next().unwrap();
                            if is_name_start(first) {
                                let mut end = start + first.len_utf8();
                                while end < input.len() {
                                    let ch = input[end..].chars().next().unwrap();
                                    if is_name_char(ch) {
                                        end += ch.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                let name = &input[start..end];
                                let value = lookup(env, name).to_string();
                                append(&mut word, &value, true);
                                active = true;
                                i = end;
                            } else {
                                append(&mut word, "$", false);
                                active = true;
                                i += n;
                            }
                        } else {
                            append(&mut word, "$", false);
                            active = true;
                            i += n;
                        }
                    }
                } else {
                    append(&mut word, &input[i..i + n], true);
                    active = true;
                    i += n;
                }
            }
        }
    }

    match mode {
        Mode::Single => return Err(SplitError::UnterminatedSingle { pos: quote_open }),
        Mode::Double => return Err(SplitError::UnterminatedDouble { pos: quote_open }),
        Mode::Bare => {}
    }
    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_braced(contents: &str) -> Option<(&str, Option<&str>)> {
    let bytes = contents.as_bytes();
    let mut end = 0;
    while end < bytes.len() {
        let c = contents[end..].chars().next()?;
        if is_name_char(c) {
            end += c.len_utf8();
        } else {
            break;
        }
    }
    if end == 0 || !is_name_start(contents.chars().next()?) {
        return None;
    }
    let name = &contents[..end];
    if end == bytes.len() {
        return Some((name, None));
    }
    if contents[end..].starts_with(":-") {
        return Some((name, Some(&contents[end + 2..])));
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                out.push(next);
                i += next.len_utf8();
            }
        } else {
            out.push(c);
            i += c.len_utf8();
        }
    }
    out
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (wi, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|&b| b == 0) {
            return Err(JoinError::Nul { word: wi, pos });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let bare = bare_candidate(word);
        let single = if word.contains('\'') { None } else {
            Some(format!("'{}'", word))
        };
        let double = if word.contains('`') { None } else {
            let mut s = String::from("\"");
            for c in word.chars() {
                if matches!(c, '\\' | '"' | '$') {
                    s.push('\\');
                }
                s.push(c);
            }
            s.push('"');
            Some(s)
        };

        let fallback = if bare.is_none() && single.is_none() && double.is_none() {
            Some(fallback_candidate(word))
        } else {
            None
        };

        let chosen = if let Some(f) = fallback {
            f
        } else {
            let mut candidates: Vec<(usize, usize, String)> = Vec::new();
            if let Some(s) = bare { candidates.push((s.len(), 0, s)); }
            if let Some(s) = single { candidates.push((s.len(), 1, s)); }
            if let Some(s) = double { candidates.push((s.len(), 2, s)); }
            candidates.sort_by_key(|(len, priority, _)| (*len, *priority));
            candidates.remove(0).2
        };
        encoded.push(chosen);
    }
    Ok(encoded.join(" "))
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%') {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn fallback_candidate(word: &str) -> String {
    let mut out = String::from("'");
    for c in word.chars() {
        if c == '\'' {
            out.push('\'');
            out.push('\\');
            out.push('\'');
            out.push('\'');
        } else {
            out.push(c);
        }
    }
    out.push('\'');
    out
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum State {
    Plain,
    Single(usize),
    Double(usize),
}

fn separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\r' || ch == '\n'
}

fn name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn name_continue(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphanumeric()
}

fn push_text(word: &mut Word, text: &str, glob: bool) {
    let base = word.text.len();
    word.text.push_str(text);
    if glob {
        for (offset, ch) in text.char_indices() {
            if ch == '*' || ch == '?' || ch == '[' {
                word.glob.push(base + offset);
            }
        }
    }
}

fn environment_value<'a>(env: &'a [(&str, &str)], name: &str) -> &'a str {
    for (key, value) in env {
        if *key == name {
            return value;
        }
    }
    ""
}

fn scalar_at(s: &str, i: usize) -> (char, usize) {
    let ch = s[i..].chars().next().unwrap();
    (ch, ch.len_utf8())
}

fn expansion_end(input: &str, dollar: usize) -> Result<usize, SplitError> {
    let mut i = dollar + 2;
    while i < input.len() {
        let (ch, size) = scalar_at(input, i);
        if ch == '\\' {
            i += size;
            if i < input.len() {
                let (_, escaped_size) = scalar_at(input, i);
                i += escaped_size;
            }
        } else if ch == '}' {
            return Ok(i);
        } else {
            i += size;
        }
    }
    Err(SplitError::UnterminatedExpansion { pos: dollar })
}

fn braced_parts(contents: &str) -> Option<(&str, Option<&str>)> {
    let mut idx = 0;
    for (offset, ch) in contents.char_indices() {
        if offset == 0 {
            if !name_start(ch) {
                return None;
            }
            idx = ch.len_utf8();
        } else if name_continue(ch) {
            idx = offset + ch.len_utf8();
        } else {
            break;
        }
    }
    if idx == contents.len() {
        return Some((&contents[..idx], None));
    }
    if contents[idx..].starts_with(":-") {
        Some((&contents[..idx], Some(&contents[idx + 2..])))
    } else {
        None
    }
}

fn decode_text(s: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < s.len() {
        let (ch, size) = scalar_at(s, i);
        if ch == '\\' {
            i += size;
            if i < s.len() {
                let (next, next_size) = scalar_at(s, i);
                result.push(next);
                i += next_size;
            }
        } else {
            result.push(ch);
            i += size;
        }
    }
    result
}

fn do_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    if input[dollar + 1..].starts_with('{') {
        let close = expansion_end(input, dollar)?;
        let contents = &input[dollar + 2..close];
        let (name, default) = braced_parts(contents)
            .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
        let value = match default {
            None => environment_value(env, name).to_string(),
            Some(text) => {
                let found = environment_value(env, name);
                if found.is_empty() { decode_text(text) } else { found.to_string() }
            }
        };
        Ok((close + 1, value))
    } else {
        let mut i = dollar + 1;
        let (first, size) = scalar_at(input, i);
        if !name_start(first) {
            return Ok((dollar + 1, "$".to_string()));
        }
        i += size;
        while i < input.len() {
            let (ch, size) = scalar_at(input, i);
            if !name_continue(ch) {
                break;
            }
            i += size;
        }
        Ok((i, environment_value(env, &input[dollar + 1..i]).to_string()))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|b| *b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut output = Vec::new();
    let mut current = Word { text: String::new(), glob: Vec::new() };
    let mut active = false;
    let mut state = State::Plain;
    let mut i = 0;

    while i < input.len() {
        let (ch, size) = scalar_at(input, i);
        match state {
            State::Single(_) => {
                if ch == '\'' {
                    state = State::Plain;
                } else {
                    push_text(&mut current, &input[i..i + size], false);
                }
                i += size;
            }
            State::Double(_) => {
                if ch == '"' {
                    state = State::Plain;
                    i += size;
                } else if ch == '\\' {
                    let slash = i;
                    i += size;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let (next, next_size) = scalar_at(input, i);
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            push_text(&mut current, &input[i..i + next_size], false);
                        }
                        i += next_size;
                    } else {
                        push_text(&mut current, "\\", false);
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if ch == '$' {
                    if input[i + size..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + size..].starts_with('{') {
                        let (end, value) = do_expansion(input, i, env)?;
                        push_text(&mut current, &value, false);
                        active = true;
                        i = end;
                    } else if i + size < input.len()
                        && name_start(scalar_at(input, i + size).0)
                    {
                        let (end, value) = do_expansion(input, i, env)?;
                        push_text(&mut current, &value, false);
                        active = true;
                        i = end;
                    } else {
                        push_text(&mut current, "$", false);
                        i += size;
                    }
                } else {
                    push_text(&mut current, &input[i..i + size], false);
                    i += size;
                }
            }
            State::Plain => {
                if separator(ch) {
                    if active {
                        output.push(current);
                        current = Word { text: String::new(), glob: Vec::new() };
                        active = false;
                    }
                    i += size;
                } else if ch == '\'' {
                    state = State::Single(i);
                    active = true;
                    i += size;
                } else if ch == '"' {
                    state = State::Double(i);
                    active = true;
                    i += size;
                } else if ch == '\\' {
                    let slash = i;
                    i += size;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let (next, next_size) = scalar_at(input, i);
                    if next != '\n' {
                        push_text(&mut current, &input[i..i + next_size], false);
                        active = true;
                    }
                    i += next_size;
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if ch == '$' {
                    if input[i + size..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + size..].starts_with('{')
                        || (i + size < input.len()
                            && name_start(scalar_at(input, i + size).0))
                    {
                        let (end, value) = do_expansion(input, i, env)?;
                        push_text(&mut current, &value, true);
                        active = true;
                        i = end;
                    } else {
                        push_text(&mut current, "$", false);
                        active = true;
                        i += size;
                    }
                } else {
                    push_text(&mut current, &input[i..i + size], true);
                    active = true;
                    i += size;
                }
            }
        }
    }

    match state {
        State::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        State::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        State::Plain => {}
    }
    if active {
        output.push(current);
    }
    Ok(output)
}

fn bare_word(s: &str) -> Option<String> {
    if s.is_empty() || s.contains('\n') {
        return None;
    }
    let mut encoded = String::new();
    for ch in s.chars() {
        if ch.is_ascii_alphanumeric()
            || matches!(ch, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%')
        {
            encoded.push(ch);
        } else {
            encoded.push('\\');
            encoded.push(ch);
        }
    }
    Some(encoded)
}

fn single_word(s: &str) -> Option<String> {
    if s.contains('\'') {
        None
    } else {
        Some(format!("'{s}'"))
    }
}

fn double_word(s: &str) -> Option<String> {
    if s.contains('`') {
        return None;
    }
    let mut encoded = String::from("\"");
    for ch in s.chars() {
        if matches!(ch, '\\' | '"' | '$') {
            encoded.push('\\');
        }
        encoded.push(ch);
    }
    encoded.push('"');
    Some(encoded)
}

fn fallback_word(s: &str) -> String {
    let mut encoded = String::from("'");
    for ch in s.chars() {
        if ch == '\'' {
            encoded.push_str("'\\''");
        } else {
            encoded.push(ch);
        }
    }
    encoded.push('\'');
    encoded
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|b| *b == 0) {
            return Err(JoinError::Nul { word: word_index, pos });
        }
    }

    let mut result = String::new();
    for (index, word) in words.iter().enumerate() {
        if index > 0 {
            result.push(' ');
        }

        let bare = bare_word(word);
        let single = single_word(word);
        let double = double_word(word);
        let chosen = if bare.is_none() && single.is_none() && double.is_none() {
            fallback_word(word)
        } else {
            let mut candidates = Vec::new();
            if let Some(s) = bare { candidates.push((s.len(), 0, s)); }
            if let Some(s) = single { candidates.push((s.len(), 1, s)); }
            if let Some(s) = double { candidates.push((s.len(), 2, s)); }
            candidates.sort_by_key(|entry| (entry.0, entry.1));
            candidates.remove(0).2
        };
        result.push_str(&chosen);
    }
    Ok(result)
}
