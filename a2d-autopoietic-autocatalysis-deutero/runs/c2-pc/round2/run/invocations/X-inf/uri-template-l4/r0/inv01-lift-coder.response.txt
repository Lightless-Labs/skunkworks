#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

impl Operator {
    fn from_byte(byte: u8) -> Option<Self> {
        match byte {
            b'+' => Some(Self::Reserved),
            b'#' => Some(Self::Fragment),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Question),
            b'&' => Some(Self::Ampersand),
            _ => None,
        }
    }

    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Question | Self::Ampersand => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Simple | Self::Reserved => None,
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Question => Some('?'),
            Self::Ampersand => Some('&'),
        }
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }

    fn named(self) -> bool {
        matches!(self, Self::Semicolon | Self::Question | Self::Ampersand)
    }
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Spec<'a> {
    name: &'a str,
    start: usize,
    modifier: Modifier,
}

fn valid_name_byte(byte: u8, first: bool) -> bool {
    if first {
        byte.is_ascii_alphabetic() || byte == b'_'
    } else {
        byte.is_ascii_alphanumeric() || byte == b'_'
    }
}

fn parse_spec<'a>(part: &'a str, start: usize) -> Result<Spec<'a>, Error> {
    if part.is_empty() {
        return Err(Error::EmptyVariable { pos: start });
    }

    let bytes = part.as_bytes();
    let split = bytes
        .iter()
        .position(|&b| b == b'*' || b == b':')
        .unwrap_or(bytes.len());

    if split == 0 {
        return Err(Error::InvalidName { pos: start });
    }
    for (i, &byte) in bytes[..split].iter().enumerate() {
        if !valid_name_byte(byte, i == 0) {
            return Err(Error::InvalidName { pos: start + i });
        }
    }

    let modifier = if split == bytes.len() {
        Modifier::None
    } else if bytes[split] == b'*' && split + 1 == bytes.len() {
        Modifier::Explode
    } else if bytes[split] == b':' {
        let digits = &bytes[split + 1..];
        let valid = (1..=4).contains(&digits.len())
            && digits[0].is_ascii_digit()
            && digits[0] != b'0'
            && digits.iter().all(u8::is_ascii_digit);
        if !valid {
            return Err(Error::InvalidModifier { pos: start + split });
        }
        let mut number = 0usize;
        for &digit in digits {
            number = number * 10 + usize::from(digit - b'0');
        }
        Modifier::Prefix(number)
    } else {
        return Err(Error::InvalidModifier { pos: start + split });
    };

    Ok(Spec {
        name: &part[..split],
        start,
        modifier,
    })
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<(Operator, Vec<Spec<'a>>), Error> {
    let content_start = open + 1;
    let content = &template[content_start..close];
    let bytes = content.as_bytes();

    let (operator, body_start) = if let Some(&first) = bytes.first() {
        if let Some(operator) = Operator::from_byte(first) {
            (operator, 1)
        } else if matches!(first, b'=' | b'!' | b'@' | b'|' | b'$') {
            return Err(Error::UnsupportedOperator {
                pos: content_start,
            });
        } else {
            (Operator::Simple, 0)
        }
    } else {
        return Err(Error::EmptyExpression { pos: close });
    };

    if body_start == bytes.len() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let body = &content[body_start..];
    let body_start_pos = content_start + body_start;
    let mut specs = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut offset = 0;

    for part in body.split(',') {
        let spec = parse_spec(part, body_start_pos + offset)?;
        if seen.iter().any(|name| *name == spec.name) {
            return Err(Error::DuplicateVariable { pos: spec.start });
        }
        seen.push(spec.name);
        specs.push(spec);
        offset += part.len() + 1;
    }

    Ok((operator, specs))
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if hex_value(bytes[i + 1]).is_some() && hex_value(bytes[i + 2]).is_some() {
                out.push('%');
                out.push(bytes[i + 1].to_ascii_uppercase() as char);
                out.push(bytes[i + 2].to_ascii_uppercase() as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn scalar_prefix(value: &str, modifier: Modifier) -> String {
    match modifier {
        Modifier::Prefix(count) => value.chars().take(count).collect(),
        Modifier::None | Modifier::Explode => value.to_owned(),
    }
}

fn append_named_field(out: &mut Vec<String>, name: &str, value: String, operator: Operator) {
    if operator == Operator::Semicolon && value.is_empty() {
        out.push(name.to_owned());
    } else {
        out.push(format!("{name}={value}"));
    }
}

fn expand_expression(
    operator: Operator,
    specs: &[Spec<'_>],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let mut fields = Vec::new();
    let reserved = operator.reserved();

    for spec in specs {
        let Some(value) = lookup(spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(value) => {
                let value = scalar_prefix(value, spec.modifier);
                append_named_field(&mut fields, spec.name, encode(&value, reserved), operator);
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        if operator.named() {
                            append_named_field(&mut fields, spec.name, encoded, operator);
                        } else {
                            fields.push(encoded);
                        }
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    if operator.named() {
                        append_named_field(&mut fields, spec.name, joined, operator);
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut keys: Vec<&str> = Vec::new();
                for (key, _) in entries {
                    if keys.iter().any(|seen| *seen == key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                    keys.push(key);
                }
                if entries.is_empty() {
                    continue;
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let key = encode(key, reserved);
                        let value = encode(value, reserved);
                        if operator == Operator::Semicolon && value.is_empty() {
                            fields.push(key);
                        } else {
                            fields.push(format!("{key}={value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    if operator.named() {
                        append_named_field(&mut fields, spec.name, joined, operator);
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut result = String::new();
    if let Some(prefix) = operator.prefix() {
        result.push(prefix);
    }
    result.push_str(&fields.join(&operator.separator().to_string()));
    Ok(result)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut parsed = Vec::new();
    let mut literal_start = 0;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let mut close = i + 1;
                while close < bytes.len() && bytes[close] != b'}' {
                    close += 1;
                }
                if close == bytes.len() {
                    return Err(Error::Unclosed { pos: open });
                }

                if let Some(inner) = (open + 1..close).find(|&p| bytes[p] == b'{') {
                    return Err(Error::NestedOpen { pos: inner });
                }

                let (operator, specs) = parse_expression(template, open, close)?;
                parsed.push((literal_start, open, operator, specs));
                i = close + 1;
                literal_start = i;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for (start, open, operator, specs) in parsed {
        out.push_str(&template[cursor..open]);
        out.push_str(&expand_expression(operator, &specs, vars)?);
        cursor = start;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}

pub enum Value {

    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy)]
enum Op {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semi,
    Query,
    Amp,
}

impl Op {
    fn parse(b: u8) -> Option<Self> {
        Some(match b {
            b'+' => Self::Reserved,
            b'#' => Self::Fragment,
            b'.' => Self::Dot,
            b'/' => Self::Slash,
            b';' => Self::Semi,
            b'?' => Self::Query,
            b'&' => Self::Amp,
            _ => return None,
        })
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }

    fn named(self) -> bool {
        matches!(self, Self::Semi | Self::Query | Self::Amp)
    }

    fn separator(self) -> u8 {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => b',',
            Self::Dot => b'.',
            Self::Slash => b'/',
            Self::Semi => b';',
            Self::Query | Self::Amp => b'&',
        }
    }

    fn prefix(self) -> Option<u8> {
        match self {
            Self::Simple | Self::Reserved => None,
            Self::Fragment => Some(b'#'),
            Self::Dot => Some(b'.'),
            Self::Slash => Some(b'/'),
            Self::Semi => Some(b';'),
            Self::Query => Some(b'?'),
            Self::Amp => Some(b'&'),
        }
    }
}

#[derive(Clone, Copy)]
enum Mod {
    Plain,
    Explode,
    Prefix(usize),
}

struct Var<'a> {
    name: &'a str,
    pos: usize,
    modifier: Mod,
}

fn spec<'a>(s: &'a str, pos: usize) -> Result<Var<'a>, Error> {
    if s.is_empty() {
        return Err(Error::EmptyVariable { pos });
    }

    let bytes = s.as_bytes();
    let mark = bytes
        .iter()
        .position(|b| matches!(b, b'*' | b':'))
        .unwrap_or(bytes.len());

    let name = &bytes[..mark];
    if name.is_empty() {
        return Err(Error::InvalidName { pos });
    }
    for (i, b) in name.iter().enumerate() {
        let good = if i == 0 {
            b.is_ascii_alphabetic() || *b == b'_'
        } else {
            b.is_ascii_alphanumeric() || *b == b'_'
        };
        if !good {
            return Err(Error::InvalidName { pos: pos + i });
        }
    }

    let modifier = if mark == bytes.len() {
        Mod::Plain
    } else if bytes[mark] == b'*' && mark + 1 == bytes.len() {
        Mod::Explode
    } else if bytes[mark] == b':' {
        let digits = &bytes[mark + 1..];
        if !(1..=4).contains(&digits.len())
            || digits[0] == b'0'
            || !digits.iter().all(u8::is_ascii_digit)
        {
            return Err(Error::InvalidModifier { pos: pos + mark });
        }
        let n = digits
            .iter()
            .fold(0usize, |n, d| n * 10 + usize::from(d - b'0'));
        Mod::Prefix(n)
    } else {
        return Err(Error::InvalidModifier { pos: pos + mark });
    };

    Ok(Var {
        name: &s[..mark],
        pos,
        modifier,
    })
}

fn name_byte(b: u8, first: bool) -> bool {
    if first {
        b.is_ascii_alphabetic() || b == b'_'
    } else {
        b.is_ascii_alphanumeric() || b == b'_'
    }
}

fn parse(template: &str) -> Result<Vec<(usize, usize, Op, Vec<Var<'_>>)>, Error> {
    let b = template.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < b.len() {
        match b[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                i += 1;
                let mut close = i;
                while close < b.len() && b[close] != b'}' {
                    close += 1;
                }
                if close == b.len() {
                    return Err(Error::Unclosed { pos: open });
                }
                if let Some(inner) = (open + 1..close).find(|&p| b[p] == b'{') {
                    return Err(Error::NestedOpen { pos: inner });
                }

                let content = &template[open + 1..close];
                let bytes = content.as_bytes();
                let (op, offset) = if bytes.is_empty() {
                    return Err(Error::EmptyExpression { pos: close });
                } else if let Some(op) = Op::parse(bytes[0]) {
                    (op, 1)
                } else {
                    if matches!(bytes[0], b'=' | b'!' | b'@' | b'|' | b'$') {
                        return Err(Error::UnsupportedOperator { pos: open + 1 });
                    }
                    (Op::Simple, 0)
                };
                if offset == bytes.len() {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let body = &content[offset..];
                let body_pos = open + 1 + offset;
                let mut vars = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut part_start = 0;
                for part in body.split(',') {
                    let item = spec(part, body_pos + part_start)?;
                    if seen.contains(&item.name) {
                        return Err(Error::DuplicateVariable { pos: item.pos });
                    }
                    seen.push(item.name);
                    vars.push(item);
                    part_start += part.len() + 1;
                }

                result.push((open, close + 1, op, vars));
                i = close + 1;
            }
            _ => i += 1,
        }
    }
    Ok(result)
}

fn hex(b: u8) -> bool {
    b.is_ascii_hexdigit()
}

fn unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const H: &[u8] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let c = bytes[i];
        if c == b'%' && i + 2 < bytes.len() && hex(bytes[i + 1]) && hex(bytes[i + 2]) {
            out.push(b'%');
            out.push(bytes[i + 1].to_ascii_uppercase());
            out.push(bytes[i + 2].to_ascii_uppercase());
            i += 3;
        } else if unreserved(c) || (allow_reserved && reserved(c)) {
            out.push(c);
            i += 1;
        } else {
            out.push(b'%');
            out.push(H[(c >> 4) as usize]);
            out.push(H[(c & 15) as usize]);
            i += 1;
        }
    }
    String::from_utf8(out).expect("percent encoding emits ASCII")
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| value)
}

fn prefix(s: &str, modifier: Mod) -> String {
    match modifier {
        Mod::Prefix(n) => s.chars().take(n).collect(),
        Mod::Plain | Mod::Explode => s.to_owned(),
    }
}

fn named_scalar(name: &str, value: String, op: Op, fields: &mut Vec<String>) {
    if op == Op::Semi && value.is_empty() {
        fields.push(name.to_owned());
    } else {
        fields.push([name, "=", &value].concat());
    }
}

fn evaluate(op: Op, vars: &[Var<'_>], bindings: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields = Vec::new();
    for var in vars {
        let Some(value) = lookup(var.name, bindings) else {
            continue;
        };

        if matches!(var.modifier, Mod::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: var.pos });
        }

        match value {
            Value::Scalar(s) => {
                let encoded = encode(&prefix(s, var.modifier), op.reserved());
                named_scalar(var.name, encoded, op, &mut fields);
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if matches!(var.modifier, Mod::Explode) {
                    for item in items {
                        let encoded = encode(item, op.reserved());
                        if op.named() {
                            named_scalar(var.name, encoded, op, &mut fields);
                        } else {
                            fields.push(encoded);
                        }
                    }
                } else {
                    let value = items
                        .iter()
                        .map(|s| encode(s, op.reserved()))
                        .collect::<Vec<_>>()
                        .join(",");
                    if op.named() {
                        named_scalar(var.name, value, op, &mut fields);
                    } else {
                        fields.push(value);
                    }
                }
            }
            Value::Map(entries) => {
                let mut seen = Vec::<&str>::new();
                for (key, _) in entries {
                    if seen.contains(&key.as_str()) {
                        return Err(Error::DuplicateKey { pos: var.pos });
                    }
                    seen.push(key);
                }
                if entries.is_empty() {
                    continue;
                }
                if matches!(var.modifier, Mod::Explode) {
                    for (key, value) in entries {
                        let key = encode(key, op.reserved());
                        let value = encode(value, op.reserved());
                        if op == Op::Semi && value.is_empty() {
                            fields.push(key);
                        } else {
                            fields.push([&key, "=", &value].concat());
                        }
                    }
                } else {
                    let mut pieces = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        pieces.push(encode(key, op.reserved()));
                        pieces.push(encode(value, op.reserved()));
                    }
                    let value = pieces.join(",");
                    if op.named() {
                        named_scalar(var.name, value, op, &mut fields);
                    } else {
                        fields.push(value);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    if let Some(p) = op.prefix() {
        out.push(p as char);
    }
    for (i, field) in fields.iter().enumerate() {
        if i != 0 {
            out.push(op.separator() as char);
        }
        out.push_str(field);
    }
    Ok(out)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let parsed = parse(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (open, end, op, vars_in_expr) in parsed {
        output.push_str(&template[cursor..open]);
        output.push_str(&evaluate(op, &vars_in_expr, vars)?);
        cursor = end;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}
