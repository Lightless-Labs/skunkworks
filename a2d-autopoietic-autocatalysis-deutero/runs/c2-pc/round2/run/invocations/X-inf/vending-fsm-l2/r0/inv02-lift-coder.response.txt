```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = match &self.selected {
            Some(id) => id.clone(),
            None => return Err(Error::NoSelection),
        };
        let price = self
            .products
            .iter()
            .find(|product| product.id == id)
            .map(|product| product.price)
            .expect("selected product must belong to catalog");

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let change = choose_change(amount, self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for &coin in &change {
            let index = denomination_index(coin);
            proposed[index] -= 1;
        }
        for &coin in &self.escrow {
            let index = denomination_index(coin);
            proposed[index] += 1;
        }
        for (index, count) in proposed.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}

fn denomination_index(coin: u16) -> usize {
    match coin {
        1 => 0,
        3 => 1,
        4 => 2,
        10 => 3,
        _ => unreachable!(),
    }
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut greedy_counts = [0u16; 4];
    let mut remainder = amount;
    for index in (0..4).rev() {
        let coin = DENOMINATIONS[index];
        let take = tubes[index].min(remainder / coin);
        greedy_counts[index] = take;
        remainder -= take * coin;
    }
    if remainder == 0 {
        return Some(counts_to_descending_coins(greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    search_change(
        3,
        amount,
        tubes,
        [0; 4],
        &mut best,
    );
    best.map(counts_to_descending_coins)
}

fn search_change(
    index: isize,
    remainder: u16,
    tubes: [u16; 4],
    counts: [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index < 0 {
        if remainder == 0
            && best
                .as_ref()
                .map_or(true, |current| is_better(&counts, current))
        {
            *best = Some(counts);
        }
        return;
    }

    let i = index as usize;
    let coin = DENOMINATIONS[i];
    let max_take = tubes[i].min(remainder / coin);
    for take in 0..=max_take {
        let mut candidate = counts;
        candidate[i] = take;
        search_change(
            index - 1,
            remainder - take * coin,
            tubes,
            candidate,
            best,
        );
    }
}

fn is_better(candidate: &[u16; 4], current: &[u16; 4]) -> bool {
    let candidate_total: u32 = candidate.iter().map(|&count| u32::from(count)).sum();
    let current_total: u32 = current.iter().map(|&count| u32::from(count)).sum();
    if candidate_total != current_total {
        return candidate_total < current_total;
    }

    for index in (0..4).rev() {
        if candidate[index] != current[index] {
            return candidate[index] > current[index];
        }
    }
    false
}

fn counts_to_descending_coins(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        coins.extend(std::iter::repeat_n(
            DENOMINATIONS[index],
            usize::from(counts[index]),
        ));
    }
    coins
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    state: State,
}

const COINS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        for i in 0..products.len() {
            let p = &products[i];
            if p.id.is_empty()
                || p.id.len() > 8
                || !p.id.bytes().all(|byte| byte.is_ascii_lowercase())
            {
                return Err(Error::BadId { id: p.id.clone() });
            }
            if products[..i].iter().any(|earlier| earlier.id == p.id) {
                return Err(Error::DuplicateId { id: p.id.clone() });
            }
            if p.price == 0 || p.price > 100 {
                return Err(Error::BadPrice { id: p.id.clone() });
            }
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }

        Ok(Self {
            products,
            state: State {
                credit: 0,
                selected: None,
                escrow: Vec::new(),
                tubes,
            },
        })
    }

    pub fn state(&self) -> State {
        self.state.clone()
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !COINS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.state.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.state.credit += coin;
        self.state.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|p| p.id == id) {
            return Err(Error::UnknownProduct { id: id.to_string() });
        }
        if self.state.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.state.selected = Some(id.to_string());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = match self.state.selected.as_ref() {
            Some(id) => id.clone(),
            None => return Err(Error::NoSelection),
        };
        let price = match self.products.iter().find(|p| p.id == id) {
            Some(p) => p.price,
            None => return Err(Error::NoSelection),
        };
        if self.state.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.state.credit,
            });
        }
        let amount = self.state.credit - price;

        let change = if amount == 0 {
            Vec::new()
        } else {
            match_change(amount, self.state.tubes).ok_or(Error::NoChange { amount })?
        };

        let mut new_tubes = self.state.tubes;
        for &coin in &change {
            new_tubes[coin_index(coin)] -= 1;
        }
        for &coin in &self.state.escrow {
            new_tubes[coin_index(coin)] += 1;
        }
        for index in 0..4 {
            if new_tubes[index] > 100 {
                return Err(Error::TubeLimit { coin: COINS[index] });
            }
        }

        self.state.tubes = new_tubes;
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let result: Vec<u16> = self.state.escrow.iter().rev().copied().collect();
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();
        Ok(result)
    }
}

fn coin_index(coin: u16) -> usize {
    match coin {
        1 => 0,
        3 => 1,
        4 => 2,
        10 => 3,
        _ => 0,
    }
}

fn match_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    let mut remaining = amount;
    let mut greedy = [0u16; 4];
    for index in (0..4).rev() {
        let count = tubes[index].min(remaining / COINS[index]);
        greedy[index] = count;
        remaining -= count * COINS[index];
    }
    if remaining == 0 {
        return Some(to_coins(greedy));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut current = [0u16; 4];
    enumerate(3, amount, &tubes, &mut current, &mut best);
    best.map(to_coins)
}

fn enumerate(
    index: i32,
    remaining: u16,
    tubes: &[u16; 4],
    current: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index < 0 {
        if remaining == 0 {
            let replace = match best {
                None => true,
                Some(old) => {
                    let new_len: u32 = current.iter().map(|&x| u32::from(x)).sum();
                    let old_len: u32 = old.iter().map(|&x| u32::from(x)).sum();
                    new_len < old_len
                        || (new_len == old_len
                            && (3..=0).rev().any(|i| {
                                current[i] != old[i] && current[i] > old[i]
                            }))
                }
            };
            if replace {
                *best = Some(*current);
            }
        }
        return;
    }

    let i = index as usize;
    let max = tubes[i].min(remaining / COINS[i]);
    for count in 0..=max {
        current[i] = count;
        enumerate(index - 1, remaining - count * COINS[i], tubes, current, best);
    }
    current[i] = 0;
}

fn to_coins(counts: [u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for index in (0..4).rev() {
        for _ in 0..counts[index] {
            result.push(COINS[index]);
        }
    }
    result
}
```
