```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<usize>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&d| d == coin)
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn greedy_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut remaining = amount;
    let mut counts = [0u16; 4];
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / DENOMINATIONS[i]);
        counts[i] = take;
        remaining -= take * DENOMINATIONS[i];
    }
    (remaining == 0).then_some(counts)
}

fn optimal_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut best: Option<[u16; 4]> = None;
    let mut best_total = u32::MAX;

    for n10 in 0..=tubes[3].min(amount / 10) {
        let rem10 = amount - n10 * 10;
        for n4 in 0..=tubes[2].min(rem10 / 4) {
            let rem4 = rem10 - n4 * 4;
            for n3 in 0..=tubes[1].min(rem4 / 3) {
                let rem3 = rem4 - n3 * 3;
                if rem3 > tubes[0] {
                    continue;
                }

                let candidate = [rem3, n3, n4, n10];
                let total = candidate.iter().map(|&n| u32::from(n)).sum();
                let is_better = total < best_total
                    || (total == best_total
                        && best.as_ref().is_some_and(|old| {
                            (0..4)
                                .rev()
                                .find_map(|i| {
                                    (candidate[i] != old[i]).then_some(candidate[i] > old[i])
                                })
                                .unwrap_or(false)
                        }));
                if is_better {
                    best = Some(candidate);
                    best_total = total;
                }
            }
        }
    }
    best
}

fn change_coins(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for i in (0..4).rev() {
        coins.extend(std::iter::repeat_n(DENOMINATIONS[i], counts[i] as usize));
    }
    coins
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains(&product.id.as_str()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.map(|i| self.products[i].id.clone()),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if denomination_index(coin).is_none() {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected == Some(index) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(index);
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(index) = self.selected else {
            return Err(Error::NoSelection);
        };
        let product = &self.products[index];
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let amount = self.credit - product.price;
        let counts = if let Some(greedy) = greedy_change(amount, &self.tubes) {
            greedy
        } else if let Some(optimal) = optimal_change(amount, &self.tubes) {
            optimal
        } else {
            return Err(Error::NoChange { amount });
        };

        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= counts[i];
        }
        for &coin in &self.escrow {
            let i = denomination_index(coin).expect("escrow contains accepted coins");
            proposed[i] += 1;
        }
        for (i, &count) in proposed.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        let receipt = Receipt {
            id: product.id.clone(),
            price: product.price,
            change: change_coins(counts),
        };
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        Ok(returned)
    }
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    !id.is_empty() && id.len() <= 8 && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn coin_index(coin: u16) -> Option<usize> {
    match coin {
        1 => Some(0),
        3 => Some(1),
        4 => Some(2),
        10 => Some(3),
        _ => None,
    }
}

fn enumerate_change(
    amount: u16,
    tubes: &[u16; 4],
    denomination: usize,
    current: &mut [u16; 4],
    best: &mut Option<([u16; 4], u32)>,
) {
    if denomination == 0 {
        if amount > tubes[0] {
            return;
        }
        current[0] = amount;
        let total = current.iter().map(|&n| u32::from(n)).sum();
        let replace = match best {
            None => true,
            Some((old, old_total)) if total < *old_total => true,
            Some((old, old_total)) if total == *old_total => {
                (0..4)
                    .rev()
                    .find_map(|i| (current[i] != old[i]).then_some(current[i] > old[i]))
                    .unwrap_or(false)
            }
            _ => false,
        };
        if replace {
            *best = Some((*current, total));
        }
        return;
    }

    let coin = DENOMINATIONS[denomination];
    let max = tubes[denomination].min(amount / coin);
    for count in 0..=max {
        current[denomination] = count;
        enumerate_change(
            amount - count * coin,
            tubes,
            denomination - 1,
            current,
            best,
        );
    }
}

fn make_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut remaining = amount;
    let mut greedy = [0u16; 4];
    for i in (0..4).rev() {
        greedy[i] = tubes[i].min(remaining / DENOMINATIONS[i]);
        remaining -= greedy[i] * DENOMINATIONS[i];
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best = None;
    enumerate_change(amount, tubes, 3, &mut [0; 4], &mut best);
    best.map(|(counts, _)| counts)
}

fn sorted_coins(counts: [u16; 4]) -> Vec<u16> {
    (0..4)
        .rev()
        .flat_map(|i| std::iter::repeat_n(DENOMINATIONS[i], usize::from(counts[i])))
        .collect()
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        for i in 0..products.len() {
            let product = &products[i];
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if products[..i].iter().any(|p| p.id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if product.price == 0 || product.price > 100 {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if coin_index(coin).is_none() {
            return Err(Error::BadCoin { coin });
        }
        if self.credit + coin > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|p| p.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let product = self.products.iter().find(|p| p.id == id).unwrap();
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });
        }

        let amount = self.credit - product.price;
        let counts = make_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= counts[i];
        }
        for &coin in &self.escrow {
            proposed[coin_index(coin).unwrap()] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        let receipt = Receipt {
            id,
            price: product.price,
            change: sorted_coins(counts),
        };
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let result = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.escrow.clear();
        self.selected = None;
        Ok(result)
    }
}
```
