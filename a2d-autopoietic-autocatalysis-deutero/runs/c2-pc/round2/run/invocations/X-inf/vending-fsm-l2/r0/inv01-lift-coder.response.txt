```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    state: State,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid_id = (1..=8).contains(&product.id.len())
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for (index, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[index],
                });
            }
        }

        Ok(Self {
            products,
            state: State {
                credit: 0,
                selected: None,
                escrow: Vec::new(),
                tubes,
            },
        })
    }

    pub fn state(&self) -> State {
        self.state.clone()
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if self.state.credit + coin > 100 {
            return Err(Error::CreditLimit);
        }

        self.state.credit += coin;
        self.state.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(_) = self.products.iter().find(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.state.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.state.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.state.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let Some(product) = self.products.iter().find(|p| &p.id == selected_id) else {
            return Err(Error::NoSelection);
        };

        let id = product.id.clone();
        let price = product.price;
        let credit = self.state.credit;
        if credit < price {
            return Err(Error::Underpaid { price, credit });
        }
        let amount = credit - price;

        let change_counts = choose_change(amount, &self.state.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed_tubes = self.state.tubes;
        for i in 0..4 {
            proposed_tubes[i] -= change_counts[i];
            let deposited = self
                .state
                .escrow
                .iter()
                .filter(|&&coin| coin == DENOMINATIONS[i])
                .count() as u16;
            let proposed = proposed_tubes[i] + deposited;
            if proposed > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
            proposed_tubes[i] = proposed;
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..change_counts[i] {
                change.push(DENOMINATIONS[i]);
            }
        }

        self.state.tubes = proposed_tubes;
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned: Vec<u16> = self.state.escrow.iter().rev().copied().collect();
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();
        Ok(returned)
    }
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remainder = amount;
    for i in (0..4).rev() {
        let coin = DENOMINATIONS[i];
        greedy[i] = tubes[i].min(remainder / coin);
        remainder -= greedy[i] * coin;
    }
    if remainder == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u16::MAX;
    for tens in 0..=tubes[3].min(amount / 10) {
        let after_tens = amount - tens * 10;
        for fours in 0..=tubes[2].min(after_tens / 4) {
            let after_fours = after_tens - fours * 4;
            for threes in 0..=tubes[1].min(after_fours / 3) {
                let after_threes = after_fours - threes * 3;
                if after_threes > tubes[0] {
                    continue;
                }
                let candidate = [after_threes, threes, fours, tens];
                let count = candidate.iter().copied().sum::<u16>();
                let replace = match best {
                    None => true,
                    Some(current) => {
                        count < best_count
                            || (count == best_count
                                && (candidate[3], candidate[2], candidate[1], candidate[0])
                                    > (current[3], current[2], current[1], current[0]))
                    }
                };
                if replace {
                    best = Some(candidate);
                    best_count = count;
                }
            }
        }
    }
    best
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<usize>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        for i in 0..products.len() {
            let id = &products[i].id;
            if !(1..=8).contains(&id.len()) || !id.bytes().all(|b| b.is_ascii_lowercase()) {
                return Err(Error::BadId { id: id.clone() });
            }
            if products[..i].iter().any(|prior| prior.id == *id) {
                return Err(Error::DuplicateId { id: id.clone() });
            }
            if !(1..=100).contains(&products[i].price) {
                return Err(Error::BadPrice { id: id.clone() });
            }
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.map(|index| self.products[index].id.clone()),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if self.credit + coin > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected == Some(index) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(index);
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(index) = self.selected else {
            return Err(Error::NoSelection);
        };
        let product = &self.products[index];
        let price = product.price;
        let credit = self.credit;
        if credit < price {
            return Err(Error::Underpaid { price, credit });
        }

        let amount = credit - price;
        let counts = find_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;
        let mut proposed = self.tubes;

        for i in 0..4 {
            proposed[i] -= counts[i];
        }
        for &coin in &self.escrow {
            let i = DENOMINATIONS.iter().position(|&d| d == coin).unwrap();
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            change.extend(std::iter::repeat(DENOMINATIONS[i]).take(counts[i] as usize));
        }
        let receipt = Receipt {
            id: product.id.clone(),
            price,
            change,
        };
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        Ok(returned)
    }
}

fn find_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let greedy = greedy_change(amount, tubes);
    if value(&greedy) == amount {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = usize::MAX;
    enumerate(3, amount, tubes, &mut [0; 4], &mut best, &mut best_count);
    best
}

fn greedy_change(amount: u16, tubes: &[u16; 4]) -> [u16; 4] {
    let mut result = [0; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        result[i] = tubes[i].min(remaining / DENOMINATIONS[i]);
        remaining -= result[i] * DENOMINATIONS[i];
    }
    result
}

fn value(counts: &[u16; 4]) -> u16 {
    counts
        .iter()
        .zip(DENOMINATIONS)
        .map(|(&count, coin)| count * coin)
        .sum()
}

fn enumerate(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    current: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
    best_count: &mut usize,
) {
    if index < 0 {
        if remaining != 0 {
            return;
        }
        let count: usize = current.iter().map(|&n| n as usize).sum();
        let replace = match *best {
            None => true,
            Some(old) => {
                count < *best_count
                    || (count == *best_count
                        && (current[3], current[2], current[1], current[0])
                            > (old[3], old[2], old[1], old[0]))
            }
        };
        if replace {
            *best = Some(*current);
            *best_count = count;
        }
        return;
    }

    let i = index as usize;
    let denomination = DENOMINATIONS[i];
    let max_count = tubes[i].min(remaining / denomination);
    for count in 0..=max_count {
        current[i] = count;
        enumerate(
            index - 1,
            remaining - count * denomination,
            tubes,
            current,
            best,
            best_count,
        );
    }
    current[i] = 0;
}
```
