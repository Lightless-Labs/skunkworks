pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],

    pub id: String,

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<usize>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        for i in 0..products.len() {
            let id = &products[i].id;
            if !(1..=8).contains(&id.len()) || !id.bytes().all(|b| b.is_ascii_lowercase()) {
                return Err(Error::BadId { id: id.clone() });
            }
            if products[..i].iter().any(|prior| prior.id == *id) {
                return Err(Error::DuplicateId { id: id.clone() });
            }
            if !(1..=100).contains(&products[i].price) {
                return Err(Error::BadPrice { id: id.clone() });
            }
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.map(|index| self.products[index].id.clone()),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if self.credit + coin > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected == Some(index) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(index);
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(index) = self.selected else {
            return Err(Error::NoSelection);
        };
        let product = &self.products[index];
        let price = product.price;
        let credit = self.credit;
        if credit < price {
            return Err(Error::Underpaid { price, credit });
        }

        let amount = credit - price;
        let counts = find_change(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;
        let mut proposed = self.tubes;

        for i in 0..4 {
            proposed[i] -= counts[i];
        }
        for &coin in &self.escrow {
            let i = DENOMINATIONS.iter().position(|&d| d == coin).unwrap();
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            change.extend(std::iter::repeat(DENOMINATIONS[i]).take(counts[i] as usize));
        }
        let receipt = Receipt {
            id: product.id.clone(),
            price,
            change,
        };
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        Ok(returned)
    }
}

fn find_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let greedy = greedy_change(amount, tubes);
    if value(&greedy) == amount {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = usize::MAX;
    enumerate(3, amount, tubes, &mut [0; 4], &mut best, &mut best_count);
    best
}

fn greedy_change(amount: u16, tubes: &[u16; 4]) -> [u16; 4] {
    let mut result = [0; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        result[i] = tubes[i].min(remaining / DENOMINATIONS[i]);
        remaining -= result[i] * DENOMINATIONS[i];
    }
    result
}

fn value(counts: &[u16; 4]) -> u16 {
    counts
        .iter()
        .zip(DENOMINATIONS)
        .map(|(&count, coin)| count * coin)
        .sum()
}

fn enumerate(
    index: isize,
    remaining: u16,
    tubes: &[u16; 4],
    current: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
    best_count: &mut usize,
) {
    if index < 0 {
        if remaining != 0 {
            return;
        }
        let count: usize = current.iter().map(|&n| n as usize).sum();
        let replace = match *best {
            None => true,
            Some(old) => {
                count < *best_count
                    || (count == *best_count
                        && (current[3], current[2], current[1], current[0])
                            > (old[3], old[2], old[1], old[0]))
            }
        };
        if replace {
            *best = Some(*current);
            *best_count = count;
        }
        return;
    }

    let i = index as usize;
    let denomination = DENOMINATIONS[i];
    let max_count = tubes[i].min(remaining / denomination);
    for count in 0..=max_count {
        current[i] = count;
        enumerate(
            index - 1,
            remaining - count * denomination,
            tubes,
            current,
            best,
            best_count,
        );
    }
    current[i] = 0;
}

fn enumerate_change(
    amount: u16,
    tubes: &[u16; 4],
    denomination: usize,
    current: &mut [u16; 4],
    best: &mut Option<([u16; 4], u32)>,
) {
    if denomination == 0 {
        if amount > tubes[0] {
            return;
        }
        current[0] = amount;
        let total = current.iter().map(|&n| u32::from(n)).sum();
        let replace = match best {
            None => true,
            Some((old, old_total)) if total < *old_total => true,
            Some((old, old_total)) if total == *old_total => {
                (0..4)
                    .rev()
                    .find_map(|i| (current[i] != old[i]).then_some(current[i] > old[i]))
                    .unwrap_or(false)
            }
            _ => false,
        };
        if replace {
            *best = Some((*current, total));
        }
        return;
    }

    let coin = DENOMINATIONS[denomination];
    let max = tubes[denomination].min(amount / coin);
    for count in 0..=max {
        current[denomination] = count;
        enumerate_change(
            amount - count * coin,
            tubes,
            denomination - 1,
            current,
            best,
        );
    }
}

fn make_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut remaining = amount;
    let mut greedy = [0u16; 4];
    for i in (0..4).rev() {
        greedy[i] = tubes[i].min(remaining / DENOMINATIONS[i]);
        remaining -= greedy[i] * DENOMINATIONS[i];
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best = None;
    enumerate_change(amount, tubes, 3, &mut [0; 4], &mut best);
    best.map(|(counts, _)| counts)
}

fn sorted_coins(counts: [u16; 4]) -> Vec<u16> {
    (0..4)
        .rev()
        .flat_map(|i| std::iter::repeat_n(DENOMINATIONS[i], usize::from(counts[i])))
        .collect()
}

fn greedy_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut remaining = amount;
    let mut counts = [0u16; 4];
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / DENOMINATIONS[i]);
        counts[i] = take;
        remaining -= take * DENOMINATIONS[i];
    }
    (remaining == 0).then_some(counts)
}

fn optimal_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut best: Option<[u16; 4]> = None;
    let mut best_total = u32::MAX;

    for n10 in 0..=tubes[3].min(amount / 10) {
        let rem10 = amount - n10 * 10;
        for n4 in 0..=tubes[2].min(rem10 / 4) {
            let rem4 = rem10 - n4 * 4;
            for n3 in 0..=tubes[1].min(rem4 / 3) {
                let rem3 = rem4 - n3 * 3;
                if rem3 > tubes[0] {
                    continue;
                }

                let candidate = [rem3, n3, n4, n10];
                let total = candidate.iter().map(|&n| u32::from(n)).sum();
                let is_better = total < best_total
                    || (total == best_total
                        && best.as_ref().is_some_and(|old| {
                            (0..4)
                                .rev()
                                .find_map(|i| {
                                    (candidate[i] != old[i]).then_some(candidate[i] > old[i])
                                })
                                .unwrap_or(false)
                        }));
                if is_better {
                    best = Some(candidate);
                    best_total = total;
                }
            }
        }
    }
    best
}

fn change_coins(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();

            escrow: self.escrow.clone(),
            tubes: self.tubes,

fn denomination_index(coin: u16) -> usize {
    match coin {
        1 => 0,
        3 => 1,
        4 => 2,
        10 => 3,
        _ => unreachable!(),
    }
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut greedy_counts = [0u16; 4];
    let mut remainder = amount;
    for index in (0..4).rev() {
        let coin = DENOMINATIONS[index];
        let take = tubes[index].min(remainder / coin);
        greedy_counts[index] = take;
        remainder -= take * coin;
    }
    if remainder == 0 {
        return Some(counts_to_descending_coins(greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    search_change(
        3,
        amount,
        tubes,
        [0; 4],
        &mut best,
    );
    best.map(counts_to_descending_coins)
}

fn search_change(
    index: isize,
    remainder: u16,
    tubes: [u16; 4],
    counts: [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index < 0 {
        if remainder == 0
            && best
                .as_ref()
                .map_or(true, |current| is_better(&counts, current))
        {
            *best = Some(counts);
        }
        return;
    }

    let i = index as usize;
    let coin = DENOMINATIONS[i];
    let max_take = tubes[i].min(remainder / coin);
    for take in 0..=max_take {
        let mut candidate = counts;
        candidate[i] = take;
        search_change(
            index - 1,
            remainder - take * coin,
            tubes,
            candidate,
            best,
        );
    }
}

fn is_better(candidate: &[u16; 4], current: &[u16; 4]) -> bool {
    let candidate_total: u32 = candidate.iter().map(|&count| u32::from(count)).sum();
    let current_total: u32 = current.iter().map(|&count| u32::from(count)).sum();
    if candidate_total != current_total {
        return candidate_total < current_total;
    }

    for index in (0..4).rev() {
        if candidate[index] != current[index] {
            return candidate[index] > current[index];
        }
    }
    false
}

fn counts_to_descending_coins(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for index in (0..4).rev() {
        coins.extend(std::iter::repeat_n(
            DENOMINATIONS[index],
            usize::from(counts[index]),
        ));
    }
    coins
}

    let mut greedy = [0u16; 4];
    let mut remainder = amount;
    for i in (0..4).rev() {
        let coin = DENOMINATIONS[i];
        greedy[i] = tubes[i].min(remainder / coin);
        remainder -= greedy[i] * coin;

    if remainder == 0 {
        return Some(greedy);

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u16::MAX;
    for tens in 0..=tubes[3].min(amount / 10) {
        let after_tens = amount - tens * 10;

fn match_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    let mut remaining = amount;
    let mut greedy = [0u16; 4];
    for index in (0..4).rev() {
        let count = tubes[index].min(remaining / COINS[index]);
        greedy[index] = count;
        remaining -= count * COINS[index];
    }
    if remaining == 0 {
        return Some(to_coins(greedy));
    }

    let mut best: Option<[u16; 4]> = None;
    let mut current = [0u16; 4];
    enumerate(3, amount, &tubes, &mut current, &mut best);
    best.map(to_coins)
}

fn enumerate(
    index: i32,
    remaining: u16,
    tubes: &[u16; 4],
    current: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index < 0 {
        if remaining == 0 {
            let replace = match best {
                None => true,
                Some(old) => {
                    let new_len: u32 = current.iter().map(|&x| u32::from(x)).sum();
                    let old_len: u32 = old.iter().map(|&x| u32::from(x)).sum();
                    new_len < old_len
                        || (new_len == old_len
                            && (3..=0).rev().any(|i| {
                                current[i] != old[i] && current[i] > old[i]
                            }))
                }
            };
            if replace {
                *best = Some(*current);
            }
        }
        return;
    }

    let i = index as usize;
    let max = tubes[i].min(remaining / COINS[i]);
    for count in 0..=max {
        current[i] = count;
        enumerate(index - 1, remaining - count * COINS[i], tubes, current, best);
    }
    current[i] = 0;
}

    for index in (0..4).rev() {
        for _ in 0..counts[index] {

    }
