```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Variable<'a> {
    name: &'a str,
    start: usize,
}

#[derive(Debug)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<u8>,
    variables: Vec<Variable<'a>>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();
    let content_start = open + 1;

    if let Some(relative) = bytes[content_start..close].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: content_start + relative,
        });
    }

    let mut position = content_start;
    let operator = match bytes.get(position).copied() {
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
            let op = bytes[position];
            position += 1;
            if op != b'+' && op != b'#' {
                return Err(Error::UnsupportedOperator { pos: position - 1 });
            }
            Some(op)
        }
        _ => None,
    };

    if position == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = position;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|relative| spec_start + relative)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..spec_end];
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (offset, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = &template[spec_start..spec_end];
        if seen.iter().any(|&existing| existing == name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        variables.push(Variable {
            name,
            start: spec_start,
        });

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        variables,
    })
}

fn parse_template(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut position = 0;

    while position < bytes.len() {
        match bytes[position] {
            b'}' => return Err(Error::UnexpectedClose { pos: position }),
            b'{' => {
                let close = match bytes[position + 1..]
                    .iter()
                    .position(|&byte| byte == b'}')
                {
                    Some(relative) => position + 1 + relative,
                    None => return Err(Error::Unclosed { pos: position }),
                };
                let expression = parse_expression(template, position, close)?;
                position = expression.end;
                expressions.push(expression);
            }
            _ => position += 1,
        }
    }

    Ok(expressions)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'('
            | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_digit(value: u8) -> u8 {
    if value < 10 {
        b'0' + value
    } else {
        b'A' + (value - 10)
    }
}

fn push_percent_encoded(output: &mut String, byte: u8) {
    output.push('%');
    output.push(hex_digit(byte >> 4) as char);
    output.push(hex_digit(byte & 0x0f) as char);
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut position = 0;

    while position < bytes.len() {
        let byte = bytes[position];

        if byte == b'%'
            && position + 2 < bytes.len()
            && is_hex(bytes[position + 1])
            && is_hex(bytes[position + 2])
        {
            output.push('%');
            output.push((bytes[position + 1] as char).to_ascii_uppercase());
            output.push((bytes[position + 2] as char).to_ascii_uppercase());
            position += 3;
        } else if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
            position += 1;
        } else {
            push_percent_encoded(&mut output, byte);
            position += 1;
        }
    }

    output
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|&(binding_name, value)| (binding_name == name).then_some(value))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let mut position = 0;

    for expression in expressions {
        output.push_str(&template[position..expression.start]);

        let mut fields = Vec::new();
        for variable in &expression.variables {
            let _ = variable.start;
            if let Some(value) = lookup(vars, variable.name) {
                fields.push(encode_value(value, expression.operator.is_some()));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                output.push('#');
            }
            output.push_str(&fields.join(","));
        }
        position = expression.end;
    }

    output.push_str(&template[position..]);
    Ok(output)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy)]
struct ParsedExpression {
    open: usize,
    close: usize,
    operator: Option<u8>,
    spec_start: usize,
}

fn name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn name_continue(byte: u8) -> bool {
    name_start(byte) || byte.is_ascii_digit()
}

fn parse_one(template: &str, open: usize, close: usize) -> Result<ParsedExpression, Error> {
    let bytes = template.as_bytes();
    let begin = open + 1;

    if let Some(offset) = bytes[begin..close].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen { pos: begin + offset });
    }

    let (operator, spec_start) = match bytes.get(begin).copied() {
        Some(op @ (b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')) => {
            if op != b'+' && op != b'#' {
                return Err(Error::UnsupportedOperator { pos: begin });
            }
            (Some(op), begin + 1)
        }
        _ => (None, begin),
    };

    if spec_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut cursor = spec_start;
    let mut previous: Vec<&str> = Vec::new();

    while cursor <= close {
        let end = bytes[cursor..close]
            .iter()
            .position(|&b| b == b',')
            .map_or(close, |offset| cursor + offset);

        if cursor == end {
            return Err(Error::EmptyVariable { pos: cursor });
        }

        let spec = &bytes[cursor..end];
        if !name_start(spec[0]) {
            return Err(Error::InvalidName { pos: cursor });
        }
        for (i, &byte) in spec.iter().enumerate().skip(1) {
            if !name_continue(byte) {
                return Err(Error::InvalidName { pos: cursor + i });
            }
        }

        let name = &template[cursor..end];
        if previous.contains(&name) {
            return Err(Error::DuplicateVariable { pos: cursor });
        }
        previous.push(name);

        if end == close {
            break;
        }
        cursor = end + 1;
    }

    Ok(ParsedExpression {
        open,
        close,
        operator,
        spec_start,
    })
}

fn validate(template: &str) -> Result<Vec<ParsedExpression>, Error> {
    let bytes = template.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'}' {
            return Err(Error::UnexpectedClose { pos: i });
        }
        if bytes[i] == b'{' {
            let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                Some(offset) => i + 1 + offset,
                None => return Err(Error::Unclosed { pos: i }),
            };
            result.push(parse_one(template, i, close)?);
            i = close + 1;
        } else {
            i += 1;
        }
    }

    Ok(result)
}

fn reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'('
            | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn safe(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(h1), Some(h2)) = (hex(bytes[i + 1]), hex(bytes[i + 2])) {
                out.push('%');
                out.push((if h1 < 10 { b'0' + h1 } else { b'A' + h1 - 10 }) as char);
                out.push((if h2 < 10 { b'0' + h2 } else { b'A' + h2 - 10 }) as char);
                i += 3;
                continue;
            }
        }

        if safe(b) || (allow_reserved && reserved(b)) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push((if b >> 4 < 10 { b'0' + (b >> 4) } else { b'A' + (b >> 4) - 10 }) as char);
            out.push((if b & 15 < 10 { b'0' + (b & 15) } else { b'A' + (b & 15) - 10 }) as char);
        }
        i += 1;
    }

    out
}

fn value_for<'a>(bindings: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    for &(candidate, value) in bindings.iter().rev() {
        if candidate == name {
            return Some(value);
        }
    }
    None
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parsed = validate(template)?;
    let mut out = String::new();
    let mut cursor = 0;

    for expression in parsed {
        out.push_str(&template[cursor..expression.open]);
        let mut fields = Vec::new();
        let mut start = expression.spec_start;

        loop {
            let end = template.as_bytes()[start..expression.close]
                .iter()
                .position(|&b| b == b',')
                .map_or(expression.close, |offset| start + offset);
            let name = &template[start..end];
            if let Some(value) = value_for(vars, name) {
                fields.push(encode(value, expression.operator.is_some()));
            }
            if end == expression.close {
                break;
            }
            start = end + 1;
        }

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                out.push('#');
            }
            out.push_str(&fields.join(","));
        }
        cursor = expression.close + 1;
    }

    out.push_str(&template[cursor..]);
    Ok(out)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
enum Operator {
    Simple,
    Plus,
    Hash,
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    names: Vec<(usize, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;

                if let Some(relative_open) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_open,
                    });
                }

                let mut content_start = open + 1;
                let operator = match bytes.get(content_start).copied() {
                    Some(b'+') => {
                        content_start += 1;
                        Operator::Plus
                    }
                    Some(b'#') => {
                        content_start += 1;
                        Operator::Hash
                    }
                    Some(b'.' | b'/' | b';' | b'?' | b'&') => {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                    _ => Operator::Simple,
                };

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut seen: Vec<&[u8]> = Vec::new();
                let mut part_start = content_start;

                loop {
                    let part_end = bytes[part_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map_or(close, |offset| part_start + offset);

                    if part_start == part_end {
                        return Err(Error::EmptyVariable { pos: part_start });
                    }

                    let name_bytes = &bytes[part_start..part_end];
                    if !is_name_start(name_bytes[0]) {
                        return Err(Error::InvalidName { pos: part_start });
                    }
                    if let Some(offset) = name_bytes[1..].iter().position(|&b| !is_name_continue(b)) {
                        return Err(Error::InvalidName {
                            pos: part_start + 1 + offset,
                        });
                    }
                    if seen.iter().any(|name| *name == name_bytes) {
                        return Err(Error::DuplicateVariable { pos: part_start });
                    }

                    seen.push(name_bytes);
                    names.push((part_start, part_end));

                    if part_end == close {
                        break;
                    }
                    part_start = part_end + 1;
                }

                expressions.push(Expression {
                    start: open,
                    end: close + 1,
                    operator,
                    names,
                });
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    let mut output = String::with_capacity(template.len());
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let mut fields = Vec::new();
        for (start, end) in expression.names {
            let name = &template[start..end];
            if let Some(value) = vars
                .iter()
                .rev()
                .find_map(|(binding_name, value)| (*binding_name == name).then_some(*value))
            {
                fields.push(encode(value, !matches!(expression.operator, Operator::Simple)));
            }
        }

        if !fields.is_empty() {
            if matches!(expression.operator, Operator::Hash) {
                output.push('#');
            }
            for (index, field) in fields.iter().enumerate() {
                if index != 0 {
                    output.push(',');
                }
                output.push_str(field);
            }
        }

        literal_start = expression.end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

        _ => (b'A' + value - 10) as char,
    }
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut literal_start = 0;
    let mut scan = 0;
    let mut parsed = Vec::new();

    while scan < bytes.len() {
        if bytes[scan] == b'}' {
            return Err(Error::UnexpectedClose { pos: scan });
        }
        if bytes[scan] != b'{' {
            scan += 1;
            continue;
        }

        let opening = scan;
        let close = match bytes[opening + 1..].iter().position(|&b| b == b'}') {
            Some(offset) => opening + 1 + offset,
            None => return Err(Error::Unclosed { pos: opening }),
        };

        if let Some(offset) = bytes[opening + 1..close].iter().position(|&b| b == b'{') {
            return Err(Error::NestedOpen {
                pos: opening + 1 + offset,
            });
        }

        let mut start = opening + 1;
        let operator = match bytes.get(start).copied() {
            Some(op @ (b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')) => {
                start += 1;
                if op != b'+' && op != b'#' {
                    return Err(Error::UnsupportedOperator { pos: opening + 1 });
                }
                Some(op)
            }
            _ => None,
        };

        if start == close {
            return Err(Error::EmptyExpression { pos: close });
        }

        let mut names = Vec::new();
        let mut current = start;
        loop {
            let delimiter = bytes[current..close].iter().position(|&b| b == b',');
            let end = delimiter.map_or(close, |offset| current + offset);
            if current == end {
                return Err(Error::EmptyVariable { pos: current });
            }

            let first = bytes[current];
            if !(first.is_ascii_alphabetic() || first == b'_') {
                return Err(Error::InvalidName { pos: current });
            }

            for (offset, &byte) in bytes[current + 1..end].iter().enumerate() {
                if !(byte.is_ascii_alphanumeric() || byte == b'_') {
                    return Err(Error::InvalidName {
                        pos: current + 1 + offset,
                    });
                }
            }

            let name = &template[current..end];
            if names.iter().any(|(prior, _): &(&str, usize)| *prior == name) {
                return Err(Error::DuplicateVariable { pos: current });
            }
            names.push((name, current));

            match delimiter {
                Some(_) => current = end + 1,
                None => break,
            }
        }

        parsed.push((opening, close, operator, names));
        scan = close + 1;
    }

    for (opening, close, operator, names) in parsed {
        output.push_str(&template[literal_start..opening]);
        let mut fields = Vec::new();

        for (name, _) in names {
            if let Some((_, value)) = vars.iter().rev().find(|(binding, _)| *binding == name) {
                fields.push(encode_value(value, operator.is_some()));
            }
        }

        if !fields.is_empty() {
            if operator == Some(b'#') {
                output.push('#');
            }
            for (index, field) in fields.iter().enumerate() {
                if index != 0 {
                    output.push(',');
                }
                output.push_str(field);
            }
        }

        literal_start = close + 1;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn encode_value(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' && index + 2 < bytes.len() && is_hex_digit(bytes[index + 1]) && is_hex_digit(bytes[index + 2]) {
            out.push('%');
            out.push((bytes[index + 1] as char).to_ascii_uppercase());
            out.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
        } else if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            out.push(byte as char);
            index += 1;
        } else {
            out.push('%');
            out.push(hex_digit(byte >> 4));
            out.push(hex_digit(byte & 15));
            index += 1;
        }
    }
    out
}

fn is_hex_digit(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(byte, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'=')
}

fn hex_digit(value: u8) -> char {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    HEX[value as usize] as char
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Plus,
    Hash,
}

struct ParsedExpression<'a> {
    literal_start: usize,
    expression_end: usize,
    operator: Operator,
    names: Vec<&'a str>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let mut parsed = Vec::new();
    let bytes = template.as_bytes();
    let mut position = 0;
    let mut literal_start = 0;

    while position < bytes.len() {
        match bytes[position] {
            b'}' => return Err(Error::UnexpectedClose { pos: position }),
            b'{' => {
                let open = position;
                let close = match bytes[open + 1..].iter().position(|byte| *byte == b'}') {
                    Some(offset) => open + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                for inner in open + 1..close {
                    if bytes[inner] == b'{' {
                        return Err(Error::NestedOpen { pos: inner });
                    }
                }

                let mut content = open + 1;
                let operator = match bytes.get(content).copied() {
                    Some(b'+') => {
                        content += 1;
                        Operator::Plus
                    }
                    Some(b'#') => {
                        content += 1;
                        Operator::Hash
                    }
                    Some(b'.' | b'/' | b';' | b'?' | b'&') => {
                        return Err(Error::UnsupportedOperator { pos: content });
                    }
                    _ => Operator::Simple,
                };

                if content == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut names = Vec::new();
                let mut cursor = content;
                loop {
                    let end = bytes[cursor..close]
                        .iter()
                        .position(|byte| *byte == b',')
                        .map_or(close, |offset| cursor + offset);
                    if end == cursor {
                        return Err(Error::EmptyVariable { pos: cursor });
                    }

                    let token = &bytes[cursor..end];
                    if !token.first().copied().is_some_and(is_initial) {
                        return Err(Error::InvalidName { pos: cursor });
                    }
                    if let Some(offset) = token[1..].iter().position(|byte| !is_later(*byte)) {
                        return Err(Error::InvalidName {
                            pos: cursor + 1 + offset,
                        });
                    }

                    let name = &template[cursor..end];
                    if names.contains(&name) {
                        return Err(Error::DuplicateVariable { pos: cursor });
                    }
                    names.push(name);

                    if end == close {
                        break;
                    }
                    cursor = end + 1;
                }

                parsed.push(ParsedExpression {
                    literal_start,
                    expression_end: close + 1,
                    operator,
                    names,
                });
                position = close + 1;
                literal_start = position;
            }
            _ => position += 1,
        }
    }

    let mut output = String::new();
    for expression in parsed {
        let expression_start = expression.literal_start;
        let opening = template[expression_start..]
            .find('{')
            .map(|offset| expression_start + offset)
            .expect("parsed expression has an opening brace");
        output.push_str(&template[expression_start..opening]);

        let mut fields = Vec::new();
        for name in expression.names {
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                fields.push(encode_value(value, matches!(expression.operator, Operator::Plus | Operator::Hash)));
            }
        }

        if !fields.is_empty() {
            if matches!(expression.operator, Operator::Hash) {
                output.push('#');
            }
            for (index, field) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                output.push_str(&field);
            }
        }
    }
    output.push_str(&template[literal_start..]);
    Ok(output)
}

fn is_initial(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_later(byte: u8) -> bool {
    is_initial(byte) || byte.is_ascii_digit()
}

fn encode_value(value: &str, reserved: bool) -> String {
    let source = value.as_bytes();
    let mut result = String::new();
    let mut i = 0;

    while i < source.len() {
        let current = source[i];
        if current == b'%' && i + 2 < source.len() && source[i + 1].is_ascii_hexdigit() && source[i + 2].is_ascii_hexdigit() {
            result.push('%');
            result.push((source[i + 1] as char).to_ascii_uppercase());
            result.push((source[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let permitted = current.is_ascii_alphanumeric()
            || matches!(current, b'-' | b'.' | b'_' | b'~')
            || (reserved
                && matches!(
                    current,
                    b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
                        | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
                ));
        if permitted {
            result.push(current as char);
        } else {
            result.push('%');
            result.push(digit(current >> 4));
            result.push(digit(current & 15));
        }
        i += 1;
    }

    result
}

fn digit(value: u8) -> char {
    if value < 10 {
        (b'0' + value) as char
    } else {
        (b'A' + value - 10) as char
    }
}
```
