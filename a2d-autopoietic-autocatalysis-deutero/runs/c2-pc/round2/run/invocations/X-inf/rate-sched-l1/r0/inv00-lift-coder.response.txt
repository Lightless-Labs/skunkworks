```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum State {
    Queued,
    Cancelled,
    Completed,
}

struct Entry {
    job: Job,
    sequence: u64,
    state: State,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    entries: Vec<Entry>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            entries: Vec::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !Self::valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.entries.iter().any(|entry| entry.job.id == job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        let next_sequence = sequence.checked_add(1).unwrap_or(sequence);
        self.entries.push(Entry {
            job,
            sequence,
            state: State::Queued,
        });
        self.next_sequence = next_sequence;
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let entry = self
            .entries
            .iter_mut()
            .find(|entry| entry.job.id == id)
            .ok_or(Error::Unknown)?;

        match entry.state {
            State::Queued => entry.state = State::Cancelled,
            State::Cancelled => return Err(Error::Cancelled),
            State::Completed => return Err(Error::Completed),
        }

        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let selected = self
            .entries
            .iter()
            .enumerate()
            .filter(|(_, entry)| entry.state == State::Queued && entry.job.ready_at <= now)
            .map(|(index, entry)| {
                let effective_priority = self.effective_priority(&entry.job, now);
                (index, entry.sequence, effective_priority)
            })
            .max_by(|left, right| {
                left.2
                    .cmp(&right.2)
                    .then_with(|| right.1.cmp(&left.1))
            });

        let result = if let Some((index, _, effective_priority)) = selected {
            self.entries[index].state = State::Completed;
            Some(Dispatch {
                id: self.entries[index].job.id.clone(),
                effective_priority,
            })
        } else {
            None
        };

        self.last_time = now;
        Ok(result)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn valid_id(id: &str) -> bool {
        (1..=16).contains(&id.len())
            && id
                .bytes()
                .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-')
    }

    fn effective_priority(&self, job: &Job, now: u64) -> u8 {
        if self.aging_ms == 0 {
            return job.priority;
        }

        let gained = (now - job.ready_at) / self.aging_ms;
        if gained >= u64::from(9 - job.priority) {
            9
        } else {
            job.priority + gained as u8
        }
    }
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

struct Entry {
    job: Job,
    sequence: u64,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    sequence: u64,
    entries: Vec<Entry>,
    cancelled: Vec<String>,
    completed: Vec<String>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            sequence: 0,
            entries: Vec::new(),
            cancelled: Vec::new(),
            completed: Vec::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;
        if !Self::valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.entries.iter().any(|entry| entry.job.id == job.id) {
            return Err(Error::Duplicate);
        }

        self.entries.push(Entry {
            job,
            sequence: self.sequence,
        });
        self.sequence = self.sequence.saturating_add(1);
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        if self.cancelled.iter().any(|known| known == id) {
            return Err(Error::Cancelled);
        }
        if self.completed.iter().any(|known| known == id) {
            return Err(Error::Completed);
        }

        if let Some(index) = self.entries.iter().position(|entry| entry.job.id == id) {
            let removed = self.entries.remove(index);
            self.cancelled.push(removed.job.id);
            self.last_time = now;
            Ok(())
        } else {
            Err(Error::Unknown)
        }
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let selected = self
            .entries
            .iter()
            .enumerate()
            .filter(|(_, entry)| entry.job.ready_at <= now)
            .map(|(index, entry)| {
                let elapsed = now - entry.job.ready_at;
                let gained = if self.aging_ms == 0 {
                    0
                } else {
                    elapsed / self.aging_ms
                };
                let effective = if gained >= u64::from(9 - entry.job.priority) {
                    9
                } else {
                    entry.job.priority + gained as u8
                };
                (index, entry.sequence, effective)
            })
            .max_by(|a, b| a.2.cmp(&b.2).then_with(|| b.1.cmp(&a.1)));

        let result = if let Some((index, _, effective_priority)) = selected {
            let entry = self.entries.remove(index);
            self.completed.push(entry.job.id.clone());
            Some(Dispatch {
                id: entry.job.id,
                effective_priority,
            })
        } else {
            None
        };

        self.last_time = now;
        Ok(result)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn valid_id(id: &str) -> bool {
        (1..=16).contains(&id.len())
            && id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
    }
}
```
