    Cancelled,
    Completed,
}

    job: Job,
    sequence: u64,
    state: JobState,

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    entries: Vec<Entry>,

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {

            aging_ms,
            last_time: 0,
            next_sequence: 0,
            entries: Vec::new(),
        }

        if !Self::valid_id(&job.id) {
            return Err(Error::BadId);

            return Err(Error::BadPriority);
        }
        if self.entries.iter().any(|entry| entry.job.id == job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.checked_add(1).expect("sequence overflow");
        self.entries.push(Entry {
            job,
            sequence,
            state: JobState::Queued,
        });
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let Some(entry) = self.entries.iter_mut().find(|entry| entry.job.id == id) else {
            return Err(Error::Unknown);
        };

        match entry.state {
            JobState::Queued => entry.state = JobState::Cancelled,
            JobState::Cancelled => return Err(Error::Cancelled),
            JobState::Completed => return Err(Error::Completed),
        }

        self.last_time = now;
        Ok(())
    }

        self.check_time(now)?;

        let mut selected: Option<(usize, u8, u64)> = None;
        for (index, entry) in self.entries.iter().enumerate() {
            if entry.state != JobState::Queued || entry.job.ready_at > now {
                continue;

            let effective = self.effective_priority(&entry.job, now);
            match selected {

                Some((_, best_priority, best_sequence))
                    if effective > best_priority
                        || (effective == best_priority && entry.sequence < best_sequence) =>
                {
                    selected = Some((index, effective, entry.sequence));
                }
                _ => {}

            entry.state = JobState::Completed;

                id: entry.job.id.clone(),
                effective_priority,

            None

        Ok(result)

            Err(Error::TimeReversed)
        } else {
            Ok(())
        }

                .iter()
                .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-')

            return job.priority;
        }

    }

                }
            };

            let replace = match winner {
                None => true,
                Some((best_index, best_priority)) => {
                    effective > best_priority
                        || (effective == best_priority
                            && record.sequence < self.records[best_index].sequence)
                }
            };
            if replace {
                winner = Some((index, effective));
            }
        }

        self.clock = now;
        match winner {
            None => Ok(None),
            Some((index, effective_priority)) => {
                let record = &mut self.records[index];
                record.state = State::Completed;
                Ok(Some(Dispatch {
                    id: record.id.clone(),
                    effective_priority,
                }))
            }
        }
    }

    fn valid_id(id: &str) -> bool {
        let bytes = id.as_bytes();
        !bytes.is_empty()
            && bytes.len() <= 16
            && bytes.iter().all(|byte| {
                byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-'
            })
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

struct Entry {
    job: Job,
    sequence: u64,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    sequence: u64,
    entries: Vec<Entry>,
    cancelled: Vec<String>,
    completed: Vec<String>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            sequence: 0,
            entries: Vec::new(),
            cancelled: Vec::new(),
            completed: Vec::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;
        if !Self::valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.entries.iter().any(|entry| entry.job.id == job.id) {
            return Err(Error::Duplicate);
        }

        self.entries.push(Entry {
            job,
            sequence: self.sequence,
        });
        self.sequence = self.sequence.saturating_add(1);
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        if self.cancelled.iter().any(|known| known == id) {
            return Err(Error::Cancelled);
        }
        if self.completed.iter().any(|known| known == id) {
            return Err(Error::Completed);
        }

        if let Some(index) = self.entries.iter().position(|entry| entry.job.id == id) {
            let removed = self.entries.remove(index);
            self.cancelled.push(removed.job.id);
            self.last_time = now;
            Ok(())
        } else {
            Err(Error::Unknown)
        }
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let selected = self
            .entries
            .iter()
            .enumerate()
            .filter(|(_, entry)| entry.job.ready_at <= now)
            .map(|(index, entry)| {
                let elapsed = now - entry.job.ready_at;
                let gained = if self.aging_ms == 0 {
                    0
                } else {
                    elapsed / self.aging_ms
                };
                let effective = if gained >= u64::from(9 - entry.job.priority) {
                    9
                } else {
                    entry.job.priority + gained as u8
                };
                (index, entry.sequence, effective)
            })
            .max_by(|a, b| a.2.cmp(&b.2).then_with(|| b.1.cmp(&a.1)));

        let result = if let Some((index, _, effective_priority)) = selected {
            let entry = self.entries.remove(index);
            self.completed.push(entry.job.id.clone());
            Some(Dispatch {
                id: entry.job.id,
                effective_priority,
            })
        } else {
            None
        };

        self.last_time = now;
        Ok(result)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn valid_id(id: &str) -> bool {
        (1..=16).contains(&id.len())
            && id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
    }
}
