#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.bytes().position(|byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut word_active = false;
    let mut state = 0u8; // 0 bare, 1 single quoted, 2 double quoted
    let mut opening_quote = 0usize;
    let mut iter = input.char_indices();

    while let Some((offset, scalar)) = iter.next() {
        if state == 1 {
            if scalar == '\'' {
                state = 0;
            } else {
                current.push(scalar);
            }
            continue;
        }

        if state == 2 {
            match scalar {
                '"' => state = 0,
                '\\' => match iter.next() {
                    None => return Err(SplitError::TrailingEscape { pos: offset }),
                    Some((_, next)) if matches!(next, '"' | '\\' | '$' | '\n') => {
                        if next != '\n' {
                            current.push(next);
                        }
                    }
                    Some(_) => current.push('\\'),
                },
                _ => current.push(scalar),
            }
            continue;
        }

        match scalar {
            ' ' | '\t' | '\r' | '\n' => {
                if word_active {
                    words.push(std::mem::take(&mut current));
                    word_active = false;
                }
            }
            '\'' => {
                state = 1;
                opening_quote = offset;
                word_active = true;
            }
            '"' => {
                state = 2;
                opening_quote = offset;
                word_active = true;
            }
            '\\' => match iter.next() {
                None => return Err(SplitError::TrailingEscape { pos: offset }),
                Some((_, '\n')) => {}
                Some((_, next)) => {
                    current.push(next);
                    word_active = true;
                }
            },
            _ => {
                current.push(scalar);
                word_active = true;
            }
        }
    }

    if state == 1 {
        return Err(SplitError::UnterminatedSingle { pos: opening_quote });
    }
    if state == 2 {
        return Err(SplitError::UnterminatedDouble { pos: opening_quote });
    }
    if word_active {
        words.push(current);
    }
    Ok(words)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut word_is_active = false;
    let mut quote: Option<(char, usize)> = None;
    let mut i = 0;

    while i < input.len() {
        let ch = input[i..].chars().next().unwrap();
        let next_i = i + ch.len_utf8();

        if let Some((delimiter, _)) = quote {
            if delimiter == '\'' {
                if ch == '\'' {
                    quote = None;
                } else {
                    current.push(ch);
                }
                i = next_i;
                continue;
            }

            if ch == '"' {
                quote = None;
                i = next_i;
                continue;
            }
            if ch == '\\' {
                if next_i == input.len() {
                    return Err(SplitError::TrailingEscape { pos: i });
                }
                let next = input[next_i..].chars().next().unwrap();
                if matches!(next, '"' | '\\' | '$' | '\n') {
                    if next != '\n' {
                        current.push(next);
                    }
                    i = next_i + next.len_utf8();
                } else {
                    current.push('\\');
                    i = next_i;
                }
                continue;
            }
            current.push(ch);
            i = next_i;
            continue;
        }

        match ch {
            ' ' | '\t' | '\r' | '\n' => {
                if word_is_active {
                    words.push(std::mem::take(&mut current));
                    word_is_active = false;
                }
            }
            '\'' | '"' => {
                quote = Some((ch, i));
                word_is_active = true;
            }
            '\\' => {
                if next_i == input.len() {
                    return Err(SplitError::TrailingEscape { pos: i });
                }
                let next = input[next_i..].chars().next().unwrap();
                if next != '\n' {
                    current.push(next);
                    word_is_active = true;
                }
                i = next_i + next.len_utf8();
                continue;
            }
            _ => {
                current.push(ch);
                word_is_active = true;
            }
        }
        i = next_i;
    }

    if let Some((delimiter, pos)) = quote {
        return Err(if delimiter == '\'' {
            SplitError::UnterminatedSingle { pos }
        } else {
            SplitError::UnterminatedDouble { pos }
        });
    }

    if word_is_active {
        words.push(current);
    }
    Ok(words)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut mode = Mode::Bare;
    let mut quote_start = 0;

    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match mode {
            Mode::Bare => match ch {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    mode = Mode::Single;
                    quote_start = pos;
                }
                '"' => {
                    active = true;
                    mode = Mode::Double;
                    quote_start = pos;
                }
                '\\' => {
                    let Some((_, next)) = chars.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if next != '\n' {
                        word.push(next);
                        active = true;
                    }
                }
                _ => {
                    word.push(ch);
                    active = true;
                }
            },
            Mode::Single => {
                if ch == '\'' {
                    mode = Mode::Bare;
                } else {
                    word.push(ch);
                }
            }
            Mode::Double => match ch {
                '"' => mode = Mode::Bare,
                '\\' => {
                    let Some((_, next)) = chars.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    match next {
                        '"' | '\\' | '$' => word.push(next),
                        '\n' => {}
                        _ => {
                            word.push('\\');
                            word.push(next);
                        }
                    }
                }
                _ => word.push(ch),
            },
        }
    }

    match mode {
        Mode::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Mode::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Mode::Bare => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Mode {
    Bare,
    Single,
    Double,
}
