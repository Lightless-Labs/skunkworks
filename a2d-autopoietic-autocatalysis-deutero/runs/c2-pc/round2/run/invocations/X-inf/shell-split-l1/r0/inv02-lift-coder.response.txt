```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut mode = Mode::Bare;
    let mut quote_start = 0;

    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match mode {
            Mode::Bare => match ch {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    mode = Mode::Single;
                    quote_start = pos;
                }
                '"' => {
                    active = true;
                    mode = Mode::Double;
                    quote_start = pos;
                }
                '\\' => {
                    let Some((_, next)) = chars.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if next != '\n' {
                        word.push(next);
                        active = true;
                    }
                }
                _ => {
                    word.push(ch);
                    active = true;
                }
            },
            Mode::Single => {
                if ch == '\'' {
                    mode = Mode::Bare;
                } else {
                    word.push(ch);
                }
            }
            Mode::Double => match ch {
                '"' => mode = Mode::Bare,
                '\\' => {
                    let Some((_, next)) = chars.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    match next {
                        '"' | '\\' | '$' => word.push(next),
                        '\n' => {}
                        _ => {
                            word.push('\\');
                            word.push(next);
                        }
                    }
                }
                _ => word.push(ch),
            },
        }
    }

    match mode {
        Mode::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Mode::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Mode::Bare => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Mode {
    Bare,
    Single,
    Double,
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let bytes = input.as_bytes();
    let mut words = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut quote: Option<(u8, usize)> = None;
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        match quote.map(|q| q.0) {
            None => match b {
                b' ' | b'\t' | b'\r' | b'\n' => {
                    if active {
                        words.push(std::mem::take(&mut current));
                        active = false;
                    }
                    i += 1;
                }
                b'\'' | b'"' => {
                    quote = Some((b, i));
                    active = true;
                    i += 1;
                }
                b'\\' => {
                    let escape_pos = i;
                    i += 1;
                    if i == bytes.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    if bytes[i] == b'\n' {
                        i += 1;
                    } else {
                        let ch = input[i..].chars().next().unwrap();
                        current.push(ch);
                        active = true;
                        i += ch.len_utf8();
                    }
                }
                _ => {
                    let ch = input[i..].chars().next().unwrap();
                    current.push(ch);
                    active = true;
                    i += ch.len_utf8();
                }
            },
            Some((b'\'', _)) => {
                if b == b'\'' {
                    quote = None;
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    current.push(ch);
                    i += ch.len_utf8();
                }
            }
            Some((b'"', _)) => match b {
                b'"' => {
                    quote = None;
                    i += 1;
                }
                b'\\' => {
                    let escape_pos = i;
                    i += 1;
                    if i == bytes.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    match bytes[i] {
                        b'"' | b'\\' | b'$' => {
                            current.push(bytes[i] as char);
                            i += 1;
                        }
                        b'\n' => i += 1,
                        _ => {
                            current.push('\\');
                            let ch = input[i..].chars().next().unwrap();
                            current.push(ch);
                            i += ch.len_utf8();
                        }
                    }
                }
                _ => {
                    let ch = input[i..].chars().next().unwrap();
                    current.push(ch);
                    i += ch.len_utf8();
                }
            },
            _ => unreachable!(),
        }
    }

    if let Some((b'\'', pos)) = quote {
        return Err(SplitError::UnterminatedSingle { pos });
    }
    if let Some((b'"', pos)) = quote {
        return Err(SplitError::UnterminatedDouble { pos });
    }
    if active {
        words.push(current);
    }
    Ok(words)
}
```
