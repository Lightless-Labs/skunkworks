```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    #[derive(Clone, Copy, PartialEq, Eq)]
    enum Quote {
        None,
        Single,
        Double,
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match quote {
            Quote::Single => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
            }
            Quote::Double => {
                if ch == '"' {
                    quote = Quote::None;
                } else if ch == '\\' {
                    let Some(&(next_pos, next)) = chars.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if next == '"' || next == '\\' || next == '$' || next == '\n' {
                        chars.next();
                        if next != '\n' {
                            word.push(next);
                        }
                    } else {
                        word.push('\\');
                        let _ = next_pos;
                    }
                } else {
                    word.push(ch);
                }
            }
            Quote::None => match ch {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    quote = Quote::Single;
                    quote_start = pos;
                }
                '"' => {
                    active = true;
                    quote = Quote::Double;
                    quote_start = pos;
                }
                '\\' => {
                    let Some((_, next)) = chars.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if next != '\n' {
                        active = true;
                        word.push(next);
                    }
                }
                _ => {
                    active = true;
                    word.push(ch);
                }
            },
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.bytes().position(|byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut word_active = false;
    let mut state = 0u8; // 0 bare, 1 single quoted, 2 double quoted
    let mut opening_quote = 0usize;
    let mut iter = input.char_indices();

    while let Some((offset, scalar)) = iter.next() {
        if state == 1 {
            if scalar == '\'' {
                state = 0;
            } else {
                current.push(scalar);
            }
            continue;
        }

        if state == 2 {
            match scalar {
                '"' => state = 0,
                '\\' => match iter.next() {
                    None => return Err(SplitError::TrailingEscape { pos: offset }),
                    Some((_, next)) if matches!(next, '"' | '\\' | '$' | '\n') => {
                        if next != '\n' {
                            current.push(next);
                        }
                    }
                    Some(_) => current.push('\\'),
                },
                _ => current.push(scalar),
            }
            continue;
        }

        match scalar {
            ' ' | '\t' | '\r' | '\n' => {
                if word_active {
                    words.push(std::mem::take(&mut current));
                    word_active = false;
                }
            }
            '\'' => {
                state = 1;
                opening_quote = offset;
                word_active = true;
            }
            '"' => {
                state = 2;
                opening_quote = offset;
                word_active = true;
            }
            '\\' => match iter.next() {
                None => return Err(SplitError::TrailingEscape { pos: offset }),
                Some((_, '\n')) => {}
                Some((_, next)) => {
                    current.push(next);
                    word_active = true;
                }
            },
            _ => {
                current.push(scalar);
                word_active = true;
            }
        }
    }

    if state == 1 {
        return Err(SplitError::UnterminatedSingle { pos: opening_quote });
    }
    if state == 2 {
        return Err(SplitError::UnterminatedDouble { pos: opening_quote });
    }
    if word_active {
        words.push(current);
    }
    Ok(words)
}
```
