    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [u64; 5],
    day_stars: [bool; 2],
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn number(b: &[u8], a: usize, z: usize, field: Field) -> Result<u32, ParseError> {
    if z - a > 1 && b[a] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: a });
    }
    b[a..z].iter().try_fold(0u32, |v, &ch| {
        v.checked_mul(10)
            .and_then(|v| v.checked_add((ch - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: a })
    })
}

fn parse_one_field(s: &str, field: Field) -> Result<u64, ParseError> {
    let b = s.as_bytes();
    for (i, &ch) in b.iter().enumerate() {
        if !ch.is_ascii_digit() && !matches!(ch, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (lo, hi) = domain(field);
    let width = (hi - lo + 1) as u32;
    let mut mask = 0u64;
    let mut begin = 0;
    while begin <= b.len() {
        let end = b[begin..]
            .iter()
            .position(|&x| x == b',')
            .map_or(b.len(), |n| begin + n);
        if begin == end {
            return Err(ParseError::EmptyItem { field, offset: begin });
        }
        let item = &b[begin..end];
        let slash = item.iter().position(|&x| x == b'/');
        if slash.is_some_and(|p| item[p + 1..].contains(&b'/')) {
            return Err(ParseError::InvalidSyntax { field, offset: begin });
        }
        let base_end = slash.unwrap_or(item.len());
        if base_end == 0 || (slash.is_some() && base_end + 1 == item.len()) {
            return Err(ParseError::InvalidSyntax { field, offset: begin });
        }
        let base = &item[..base_end];
        let hyphens: Vec<usize> = base.iter().enumerate()
            .filter_map(|(i, &ch)| (ch == b'-').then_some(i)).collect();

        let (start, finish, single) = if base == b"*" {
            (lo as u32, hi as u32, false)
        } else if hyphens.is_empty() && base.iter().all(u8::is_ascii_digit) {
            let n = number(item, 0, base.len(), field)?;
            if n < lo as u32 || n > hi as u32 {
                return Err(ParseError::OutOfRange { field, offset: begin });
            }
            (n, n, true)
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            if h == 0 || h + 1 == base.len()
                || !base[..h].iter().all(u8::is_ascii_digit)
                || !base[h + 1..].iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax { field, offset: begin });
            }
            let x = number(item, 0, h, field)?;
            if x < lo as u32 || x > hi as u32 {
                return Err(ParseError::OutOfRange { field, offset: begin });
            }
            let y = number(item, h + 1, base.len(), field)?;
            if y < lo as u32 || y > hi as u32 {
                return Err(ParseError::OutOfRange { field, offset: begin + h + 1 });
            }
            if x > y {
                return Err(ParseError::ReversedRange { field, offset: begin + h + 1 });
            }
            (x, y, false)
        } else {
            return Err(ParseError::InvalidSyntax { field, offset: begin });
        };

        let step = if let Some(p) = slash {
            let n = number(item, p + 1, item.len(), field)?;
            if n == 0 || n > width {
                return Err(ParseError::BadStep { field, offset: begin + p + 1 });
            }
            n
        } else { 1 };
        let finish = if slash.is_some() && single { hi as u32 } else { finish };
        let mut v = start;
        while v <= finish {
            mask |= 1u64 << v;
            v += step;
        }

        if end == b.len() { break; }
        begin = end + 1;
    }
    Ok(mask)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let p: Vec<&str> = input.split(' ').collect();
    if p.len() != 5 {
        return Err(ParseError::FieldCount { found: p.len() });
    }
    let mut selected = [0u64; 5];
    for i in 0..5 {
        selected[i] = parse_one_field(p[i], FIELDS[i])?;
    }
    Ok(Cron {
        selected,
        day_stars: [p[2] == "*", p[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (lo, hi) = domain(field);
        (lo..=hi).filter(|&v| self.selected[index(field)] & (1u64 << v) != 0).collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        let wd = weekday(at.year, at.month, at.day);
        Ok([at.minute, at.hour, at.day, at.month, wd]
            .iter().enumerate()
            .all(|(i, &v)| self.selected[i] & (1u64 << v) != 0))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut x = after;
        while x.year < 2399 || x.month < 12 || x.day < 31 || x.hour < 23 || x.minute < 59 {
            x = tick(x);
            if self.matches(x)? { return Ok(Some(x)); }
        }
        Ok(None)
    }
}

fn index(f: Field) -> usize {
    match f { Field::Minute => 0, Field::Hour => 1, Field::DayOfMonth => 2, Field::Month => 3, Field::DayOfWeek => 4 }
}

    pub month: u8,

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart { Year, Month, Day, Hour, Minute }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError { InvalidDateTime { component: DatePart } }

fn is_leap(y: u16) -> bool { y % 4 == 0 && (y % 100 != 0 || y % 400 == 0) }
fn month_days(y: u16, m: u8) -> u8 {
    match m { 2 => if is_leap(y) { 29 } else { 28 }, 4 | 6 | 9 | 11 => 30, _ => 31 }
}
fn validate(x: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&x.year) { return Err(DateError::InvalidDateTime { component: DatePart::Year }); }
    if !(1..=12).contains(&x.month) { return Err(DateError::InvalidDateTime { component: DatePart::Month }); }
    if x.day == 0 || x.day > month_days(x.year, x.month) { return Err(DateError::InvalidDateTime { component: DatePart::Day }); }
    if x.hour > 23 { return Err(DateError::InvalidDateTime { component: DatePart::Hour }); }
    if x.minute > 59 { return Err(DateError::InvalidDateTime { component: DatePart::Minute }); }
    Ok(())
}
fn weekday(y: u16, m: u8, d: u8) -> u8 {
    let mut n = 0u32;
    for yy in 2000..y { n += if is_leap(yy) { 366 } else { 365 }; }
    for mm in 1..m { n += month_days(y, mm) as u32; }
    ((n + d as u32 + 5) % 7) as u8
}
fn tick(mut x: CivilDateTime) -> CivilDateTime {
    if x.minute < 59 { x.minute += 1; return x; }
    x.minute = 0;
    if x.hour < 23 { x.hour += 1; return x; }
    x.hour = 0;
    if x.day < month_days(x.year, x.month) { x.day += 1; return x; }
    x.day = 1;
    if x.month < 12 { x.month += 1; return x; }
    x.month = 1;
    x.year += 1;
    x
}

    EmptyItem { field: Field, offset: usize },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [u64; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

fn bounds(field: Field) -> (usize, u8, u8) {
    match field {
        Field::Minute => (0, 0, 59),
        Field::Hour => (1, 0, 23),
        Field::DayOfMonth => (2, 1, 31),
        Field::Month => (3, 1, 12),
        Field::DayOfWeek => (4, 0, 6),
    }
}

fn number(s: &str, offset: usize, field: Field) -> Result<u32, ParseError> {
    let b = s.as_bytes();
    if b.len() > 1 && b[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n = 0u32;
    for &digit in b {
        n = n.checked_mul(10)
            .and_then(|x| x.checked_add((digit - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(n)
}

fn parse_one(item: &str, offset: usize, field: Field, bits: &mut u64) -> Result<(), ParseError> {
    let raw = item.as_bytes();
    let slash: Vec<usize> = raw.iter().enumerate().filter_map(|(i, b)| (*b == b'/').then_some(i)).collect();
    if slash.len() > 1 {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    let (base, step_text) = if let Some(&i) = slash.first() {
        if i == 0 || i + 1 == raw.len() {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (&item[..i], Some(&item[i + 1..]))
    } else {
        (item, None)
    };

    let (idx, min, max) = bounds(field);
    let (start, end, base_num) = if base == "*" {
        (min as u32, max as u32, None)
    } else if base.contains('*') {
        return Err(ParseError::InvalidSyntax { field, offset });
    } else if let Some(i) = base.find('-') {
        if base[i + 1..].contains('-') || i == 0 || i + 1 == base.len() {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        let a = number(&base[..i], offset, field)?;
        let b = number(&base[i + 1..], offset + i + 1, field)?;
        for (v, o) in [(a, offset), (b, offset + i + 1)] {
            if v < min as u32 || v > max as u32 {
                return Err(ParseError::OutOfRange { field, offset: o });
            }
        }
        if a > b {
            return Err(ParseError::ReversedRange { field, offset: offset + i + 1 });
        }
        (a, b, None)
    } else {
        let a = number(base, offset, field)?;
        if a < min as u32 || a > max as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
        (a, if step_text.is_some() { max as u32 } else { a }, Some(a))
    };

    let step = if let Some(s) = step_text {
        let step_offset = offset + base.len() + 1;
        let n = number(s, step_offset, field)?;
        if n == 0 || n > (max - min + 1) as u32 {
            return Err(ParseError::BadStep { field, offset: step_offset });
        }
        n
    } else {
        1
    };

    let _ = (idx, base_num);
    let mut v = start;
    while v <= end {
        *bits |= 1u64 << v;
        v += step;
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<u64, ParseError> {
    for (i, b) in text.bytes().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }
    let mut bits = 0;
    let mut pos = 0;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: pos });
        }
        parse_one(item, pos, field, &mut bits)?;
        pos += item.len() + 1;
    }
    Ok(bits)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<_> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }
    let ids = [Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek];
    let mut values = [0; 5];
    for i in 0..5 {
        values[i] = parse_field(fields[i], ids[i])?;
    }
    Ok(Cron {
        values,
        day_of_month_is_star: fields[2] == "*",
        day_of_week_is_star: fields[4] == "*",
    })
}

            }
        }
    }

        2 => if is_leap(y) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(y: u16, m: u8, d: u8) -> u8 {
    let mut elapsed = 0u32;
    for year in 2000..y {
        elapsed += if is_leap(year) { 366 } else { 365 };
    }
    for month in 1..m {
        elapsed += month_days(y, month) as u32;
    }
    ((6 + elapsed + d as u32 - 1) % 7) as u8
}

fn next_minute(t: CivilDateTime) -> Option<CivilDateTime> {
    if t == (CivilDateTime { year: 2399, month: 12, day: 31, hour: 23, minute: 59 }) {
        return None;
    }
    let mut n = t;
    if n.minute < 59 {
        n.minute += 1;
    } else {
        n.minute = 0;
        if n.hour < 23 {
            n.hour += 1;
        } else {
            n.hour = 0;
            if n.day < month_days(n.year, n.month) {
                n.day += 1;
            } else {
                n.day = 1;
                if n.month < 12 {
                    n.month += 1;
                } else {
                    n.month = 1;
                    n.year += 1;
                }
            }
        }
    }
    Some(n)
}

    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<u8>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,

fn bounds(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }
    let mut value = 0u32;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }
    Ok(value)
}

fn parse_field(src: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = src.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = bounds(field);
    let width = (max - min + 1) as u32;
    let mut selected = vec![false; (max + 1) as usize];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map_or(bytes.len(), |n| item_start + n);
        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let (base_end, step_range) = if let Some(&slash) = slash_positions.first() {
            (slash, Some((slash + 1, item.len())))
        } else {
            (item.len(), None)
        };
        let base = &item[..base_end];
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        enum Base {
            Star,
            Single(usize, usize),
            Range(usize, usize, usize, usize),
        }

        let parsed_base = if base == b"*" {
            Base::Star
        } else if hyphens.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            Base::Single(0, base.len())
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            if h == 0
                || h + 1 == base.len()
                || !base[..h].iter().all(u8::is_ascii_digit)
                || !base[h + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            Base::Range(0, h, h + 1, base.len())
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        if base.is_empty() || step_range.is_some_and(|(a, b)| a == b) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (start, end, start_offset, end_offset, is_star, is_single) = match parsed_base {
            Base::Star => (min as u32, max as u32, item_start, item_start, true, false),
            Base::Single(a, b) => {
                let n = parse_number(item, a, b, field)?;
                if n < min as u32 || n > max as u32 {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start + a,
                    });
                }
                (n, n, item_start + a, item_start + a, false, true)
            }
            Base::Range(a, b, c, d) => {
                let x = parse_number(item, a, b, field)?;
                if x < min as u32 || x > max as u32 {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start + a,
                    });
                }
                let y = parse_number(item, c, d, field)?;
                if y < min as u32 || y > max as u32 {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start + c,
                    });
                }
                if x > y {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: item_start + c,
                    });
                }
                (x, y, item_start + a, item_start + c, false, false)
            }
        };

        let step = if let Some((a, b)) = step_range {
            let n = parse_number(item, a, b, field)?;
            if n == 0 || n > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: item_start + a,
                });
            }
            n
        } else {
            1
        };

        let actual_end = if step_range.is_some() && is_single {
            max as u32
        } else {
            end
        };
        let _ = (start_offset, end_offset, is_star);
        let mut value = start;
        while value <= actual_end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok((min..=max).filter(|&v| selected[v as usize]).collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut sets = Vec::with_capacity(5);
    for (i, field) in FIELDS.iter().copied().enumerate() {
        sets.push(parse_field(parts[i], field)?);
    }
    Ok(Cron {
        sets: sets.try_into().ok().unwrap(),
        day_of_month_is_star: parts[2] == "*",
        day_of_week_is_star: parts[4] == "*",
    })
}

    for m in 1..month {

    }
    days += (day - 1) as u32;

        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}

fn spec(field: Field) -> (usize, u32, u32) {
    match field {
        Field::Minute => (0, 0, 59),
        Field::Hour => (1, 0, 23),
        Field::DayOfMonth => (2, 1, 31),
        Field::Month => (3, 1, 12),

fn number(s: &[u8], a: usize, b: usize, f: Field) -> Result<u32, ParseError> {
    if b - a > 1 && s[a] == b'0' {
        return Err(ParseError::LeadingZero { field: f, offset: a });
    }
    let mut n = 0u32;
    for &x in &s[a..b] {
        n = n.checked_mul(10)
            .and_then(|v| v.checked_add((x - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field: f, offset: a })?;
    }
    Ok(n)
}

fn parse(s: &str, f: Field) -> Result<u64, ParseError> {
    let b = s.as_bytes();
    for (i, &x) in b.iter().enumerate() {
        if !x.is_ascii_digit() && !matches!(x, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field: f, offset: i });
        }
    }
    let (_, lo, hi) = spec(f);
    let mut mask = 0u64;
    let mut pos = 0;
    loop {
        let end = b[pos..].iter().position(|&x| x == b',').map(|x| pos + x).unwrap_or(b.len());
        if end == pos {
            return Err(ParseError::EmptyItem { field: f, offset: pos });
        }
        let item = &b[pos..end];
        let slash = item.iter().position(|&x| x == b'/');
        let base_end = slash.unwrap_or(item.len());
        let base = &item[..base_end];
        let hy: Vec<usize> = base.iter().enumerate().filter_map(|(i, &x)| (x == b'-').then_some(i)).collect();
        let (kind, a, z, c, e) = if base == b"*" {
            (0, None, None, None, None)
        } else if hy.is_empty() && !base.is_empty() {
            (1, Some(0), Some(base.len()), None, None)
        } else if hy.len() == 1 && hy[0] > 0 && hy[0] + 1 < base.len() {
            (2, Some(0), Some(hy[0]), Some(hy[0] + 1), Some(base.len()))
        } else {
            return Err(ParseError::InvalidSyntax { field: f, offset: pos });
        };
        if let Some(p) = slash {
            if p + 1 == item.len() || item[p + 1..].contains(&b'/') {
                return Err(ParseError::InvalidSyntax { field: f, offset: pos });
            }
        }

        let (mut first, mut last) = (lo, hi);
        if let (Some(a), Some(z)) = (a, z) {
            first = number(b, pos + a, pos + z, f)?;
            if first < lo || first > hi {
                return Err(ParseError::OutOfRange { field: f, offset: pos + a });
            }
            last = first;
        }
        if let (Some(c), Some(e)) = (c, e) {
            last = number(b, pos + c, pos + e, f)?;
            if last < lo || last > hi {
                return Err(ParseError::OutOfRange { field: f, offset: pos + c });
            }
            if first > last {
                return Err(ParseError::ReversedRange { field: f, offset: pos + c });
            }
        }

        let step = if let Some(p) = slash {
            let at = pos + p + 1;
            let n = number(b, at, item.len(), f)?;
            if n == 0 || n > hi - lo + 1 {
                return Err(ParseError::BadStep { field: f, offset: at });
            }
            n
        } else { 1 };

        if kind == 1 && slash.is_none() {
            mask |= 1u64 << first;
        } else {
            let mut v = first;
            while v <= last {
                mask |= 1u64 << v;
                v += step;
            }
        }
        if end == b.len() { break; }
        pos = end + 1;
    }
    Ok(mask)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let p: Vec<&str> = input.split(' ').collect();
    if p.len() != 5 {
        return Err(ParseError::FieldCount { found: p.len() });
    }
    let fs = [Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek];
    let mut sets = [0u64; 5];
    for i in 0..5 { sets[i] = parse(p[i], fs[i])?; }
    Ok(Cron { sets, day_of_month_is_star: p[2] == "*", day_of_week_is_star: p[4] == "*" })
}

                .collect();
            if hyphens.is_empty() {
                if base_end == 0 || item[..base_end].iter().any(|&b| !b.is_ascii_digit()) {
                    return Err(ParseError::InvalidSyntax { field, offset: item_start });
                }
                (1, 0, base_end, None)
            } else if hyphens.len() == 1 {
                let h = hyphens[0];
                if h == 0
                    || h + 1 == base_end
                    || item[..h].iter().any(|&b| !b.is_ascii_digit())
                    || item[h + 1..base_end].iter().any(|&b| !b.is_ascii_digit())
                {
                    return Err(ParseError::InvalidSyntax { field, offset: item_start });
                }
                (2, 0, h, Some((h + 1, base_end)))
            } else {
                return Err(ParseError::InvalidSyntax { field, offset: item_start });
            }
        };

        if slash_count > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: item_start });
        }

        let step_range = if let Some(s) = slash {
            if s + 1 == item.len()
                || item[s + 1..].iter().any(|&b| !b.is_ascii_digit())
                || item[..s].is_empty()
            {
                return Err(ParseError::InvalidSyntax { field, offset: item_start });
            }
            Some((s + 1, item.len()))
        } else {
            None
        };

        let (start, end) = match base_kind {
            0 => (min as u32, max as u32),
            1 => {
                let n = parse_number(&item[..base_end], 0, base_end, field)
                    .map_err(|e| rebase_error(e, item_start))?;
                if n < min as u32 || n > max as u32 {
                    return Err(ParseError::OutOfRange { field, offset: item_start });
                }
                (n, if step_range.is_some() { max as u32 } else { n })
            }
            _ => {
                let (a, b) = second.unwrap();
                let n1 = parse_number(&item[..base_end], first_start, first_end, field)
                    .map_err(|e| rebase_error(e, item_start))?;
                let n2 = parse_number(&item[..base_end], a, b, field)
                    .map_err(|e| rebase_error(e, item_start))?;
                for (n, off) in [(n1, first_start), (n2, a)] {
                    if n < min as u32 || n > max as u32 {
                        return Err(ParseError::OutOfRange { field, offset: item_start + off });
                    }
                }
                if n1 > n2 {
                    return Err(ParseError::ReversedRange { field, offset: item_start + a });
                }
                (n1, n2)
            }
        };

        let step = if let Some((s, e)) = step_range {
            let n = parse_number(item, s, e, field)
                .map_err(|err| rebase_error(err, item_start))?;
            let width = (max - min + 1) as u32;
            if n == 0 || n > width {
                return Err(ParseError::BadStep { field, offset: item_start + s });
            }
            n
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

fn rebase_error(error: ParseError, base: usize) -> ParseError {
    match error {
        ParseError::LeadingZero { field, offset } => ParseError::LeadingZero {
            field,
            offset: base + offset,
        },
        ParseError::NumberOverflow { field, offset } => ParseError::NumberOverflow {
            field,
            offset: base + offset,
        },
        other => other,
    }
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|_| Vec::new());
    for (i, field) in fields.into_iter().enumerate() {
        sets[i] = parse_field(parts[i], field)?;
    }
    Ok(Cron {
        sets,
        day_of_month_is_star: parts[2] == "*",
        day_of_week_is_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (index, min, max) = field_info(field);
        (min..=max)
            .filter(|&value| self.sets[index][value as usize])
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.sets[0][at.minute as usize]
            && self.sets[1][at.hour as usize]
            && self.sets[2][at.day as usize]
            && self.sets[3][at.month as usize]
            && self.sets[4][weekday as usize])
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if let Some(next) = increment_minute(current) {
                current = next;
                if self.matches(current)? {
                    return Ok(Some(current));
                }
            } else {
                return Ok(None);
            }
        }
    }
}

pub enum DatePart {
    Year,

    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn increment_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.year == 2399 && at.month == 12 && at.day == 31 && at.hour == 23 && at.minute == 59 {
        return None;
    }
    let mut next = at;
    if next.minute < 59 {
        next.minute += 1;
    } else {
        next.minute = 0;
        if next.hour < 23 {
            next.hour += 1;
        } else {
            next.hour = 0;
            if next.day < month_length(next.year, next.month) {
                next.day += 1;
            } else {
                next.day = 1;
                if next.month < 12 {
                    next.month += 1;
                } else {
                    next.month = 1;
                    next.year += 1;
                }
            }
        }
    }
    Some(next)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

fn field_info(field: Field) -> (usize, u32, u32) {
    match field {
        Field::Minute => (0, 0, 59),
        Field::Hour => (1, 0, 23),
        Field::DayOfMonth => (2, 1, 31),
        Field::Month => (3, 1, 12),
        Field::DayOfWeek => (4, 0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero {
            field,
            offset: start,
        });
    }
    let mut n = 0u32;
    for &b in &bytes[start..end] {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow {
                field,
                offset: start,
            })?;
    }
    Ok(n)
}

fn parse_field(text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (_, min, max) = field_info(field);
    let mut selected = vec![false; (max + 1) as usize];
    let mut item_start = 0;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash = item.iter().position(|&b| b == b'/');
        let base_end = slash.unwrap_or(item.len());

        let base = &item[..base_end];
        let (kind, first_span, second_span) = if base == b"*" {
            (0u8, None, None)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();
            if hyphens.is_empty() {
                (1u8, Some((0, base.len())), None)
            } else if hyphens.len() == 1
                && hyphens[0] > 0
                && hyphens[0] + 1 < base.len()
            {
                (
                    2u8,
                    Some((0, hyphens[0])),
                    Some((hyphens[0] + 1, base.len())),
                )
            } else {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        };

        if let Some(slash_pos) = slash {
            if slash_pos + 1 >= item.len()
                || item[slash_pos + 1..].iter().any(|&b| b == b'/')
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }

        let mut start_value = min;
        let mut end_value = max;

        if let Some((a, b)) = first_span {
            let n = parse_number(bytes, item_start + a, item_start + b, field)?;
            if n < min || n > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + a,
                });
            }
            start_value = n;
            end_value = n;
        }

        if let Some((a, b)) = second_span {
            let n = parse_number(bytes, item_start + a, item_start + b, field)?;
            if n < min || n > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + a,
                });
            }
            end_value = n;
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + a,
                });
            }
        }

        let step = if let Some(slash_pos) = slash {
            let step_start = item_start + slash_pos + 1;
            let step = parse_number(bytes, step_start, item.len(), field)?;
            let width = max - min + 1;
            if step == 0 || step > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_start,
                });
            }
            step
        } else {
            1
        };

        if kind == 1 && slash.is_none() {
            selected[start_value as usize] = true;
        } else {
            let mut value = start_value;
            while value <= end_value {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) if next > value => value = next,
                    _ => break,
                }
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,

        day_of_month_is_star: parts[2] == "*",
        day_of_week_is_star: parts[4] == "*",
    })
}
