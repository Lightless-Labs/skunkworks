fn err(field: Field, offset: usize, kind: ErrorKind) -> ParseError {
    match kind {
        ErrorKind::InvalidCharacter => ParseError::InvalidCharacter { field, offset },
        ErrorKind::EmptyItem => ParseError::EmptyItem { field, offset },
        ErrorKind::InvalidSyntax => ParseError::InvalidSyntax { field, offset },
        ErrorKind::LeadingZero => ParseError::LeadingZero { field, offset },
        ErrorKind::NumberOverflow => ParseError::NumberOverflow { field, offset },
        ErrorKind::OutOfRange => ParseError::OutOfRange { field, offset },
        ErrorKind::ReversedRange => ParseError::ReversedRange { field, offset },
        ErrorKind::BadStep => ParseError::BadStep { field, offset },
    }
}

enum ErrorKind {
    InvalidCharacter,
    EmptyItem,
    InvalidSyntax,
    LeadingZero,
    NumberOverflow,
    OutOfRange,
    ReversedRange,
    BadStep,
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    step: bool,
) -> Result<u32, ParseError> {
    let digits = &bytes[start..end];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(err(field, start, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in digits {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or_else(|| err(field, start, ErrorKind::NumberOverflow))?;
    }

    if step {
        let width = field_spec(field).1 - field_spec(field).0 + 1;
        if value == 0 || value > width {
            return Err(err(field, start, ErrorKind::BadStep));
        }
    } else {
        let (min, max) = field_spec(field);
        if value < min || value > max {
            return Err(err(field, start, ErrorKind::OutOfRange));
        }
    }
    Ok(value)
}

fn parse_field(source: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = source.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(err(field, offset, ErrorKind::InvalidCharacter));
        }
    }

    let (min, max) = field_spec(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let exact_star = bytes == b"*";

    let mut item_start = 0;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(err(field, item_start, ErrorKind::EmptyItem));
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let (base_end, step_start) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [slash] => (*slash, Some(slash + 1)),
            _ => return Err(err(field, item_start, ErrorKind::InvalidSyntax)),
        };

        let base = &item[..base_end];
        let base_kind;
        let first_number;
        let second_number;

        if base == b"*" {
            base_kind = 0u8;
            first_number = None;
            second_number = None;
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();
            match hyphens.as_slice() {
                [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {
                    base_kind = 1;
                    first_number = Some((0, base.len()));
                    second_number = None;
                }
                [hyphen]
                    if *hyphen > 0
                        && *hyphen + 1 < base.len()
                        && base[..*hyphen].iter().all(u8::is_ascii_digit)
                        && base[*hyphen + 1..].iter().all(u8::is_ascii_digit) =>
                {
                    base_kind = 2;
                    first_number = Some((0, *hyphen));
                    second_number = Some((*hyphen + 1, base.len()));
                }
                _ => return Err(err(field, item_start, ErrorKind::InvalidSyntax)),
            }
        }

        let (start, end) = match base_kind {
            0 => (min, max),
            1 => {
                let (a, b) = first_number.unwrap();
                let n = parse_number(bytes, item_start + a, item_start + b, field, false)?;
                (n, if step_start.is_some() { max } else { n })
            }
            _ => {
                let (a, b) = first_number.unwrap();
                let start = parse_number(bytes, item_start + a, item_start + b, field, false)?;
                let (a, b) = second_number.unwrap();
                let end = parse_number(bytes, item_start + a, item_start + b, field, false)?;
                if start > end {
                    return Err(err(field, item_start + a, ErrorKind::ReversedRange));
                }
                (start, end)
            }
        };

        let step = if let Some(step_offset) = step_start {
            if step_offset == item.len() {
                return Err(err(field, item_start, ErrorKind::InvalidSyntax));
            }
            parse_number(
                bytes,
                item_start + step_offset,
                item_end,
                field,
                true,
            )?
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[(value - min) as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok((
        selected
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
            .collect(),
        exact_star,
    ))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut values: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut stars = [false; 5];

    for (i, (source, field)) in parts.iter().zip(fields).enumerate() {
        let (selected, exact_star) = parse_field(source, field)?;
        values[i] = selected;
        stars[i] = exact_star;
    }

    Ok(Cron {
        values,
        day_of_month_is_star: stars[2],
        day_of_week_is_star: stars[4],
    })
}

    min: u32,
    max: u32,
}

const SPECS: [Spec; 5] = [
    Spec { field: Field::Minute, min: 0, max: 59 },
    Spec { field: Field::Hour, min: 0, max: 23 },
    Spec { field: Field::DayOfMonth, min: 1, max: 31 },

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn number(bytes: &[u8], offset: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n = 0u32;
    for &b in bytes {
        n = n.checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(n)
}

fn parse_spec(input: &str, spec: Spec) -> Result<Vec<u8>, ParseError> {
    let field = spec.field;
    let b = input.as_bytes();
    for (i, &ch) in b.iter().enumerate() {
        if !(ch.is_ascii_digit() || matches!(ch, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let mut bits = 0u64;
    let mut cursor = 0;
    loop {
        let end = b[cursor..].iter().position(|&x| x == b',')
            .map(|x| cursor + x).unwrap_or(b.len());
        let item = &b[cursor..end];
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: cursor });
        }

        let slash = item.iter().position(|&x| x == b'/');
        if slash.is_some_and(|s| item[s + 1..].contains(&b'/')) {
            return Err(invalid_syntax(field, cursor));
        }
        let (base, step_slice, step_offset) = match slash {
            Some(s) => (&item[..s], Some(&item[s + 1..]), cursor + s + 1),
            None => (item, None, end),
        };

        let (form, a, z, z_offset) = if base == b"*" {
            (0, None, None, cursor)
        } else {
            let hyphen = base.iter().position(|&x| x == b'-');
            match hyphen {
                Some(h) if base[h + 1..].contains(&b'-') => return Err(invalid_syntax(field, cursor)),
                Some(h) if h == 0 || h + 1 == base.len() => return Err(invalid_syntax(field, cursor)),
                Some(h) => (2, Some(&base[..h]), Some(&base[h + 1..]), cursor + h + 1),
                None if !base.is_empty() && base.iter().all(u8::is_ascii_digit) =>
                    (1, Some(base), None, cursor),
                None => return Err(invalid_syntax(field, cursor)),
            }
        };

        if step_slice.is_some_and(|s| s.is_empty() || !s.iter().all(u8::is_ascii_digit)) {
            return Err(invalid_syntax(field, cursor));
        }

        let (start, end_value) = if form == 0 {
            (spec.min, spec.max)
        } else {
            let av = number(a.unwrap(), cursor, field)?;
            if av < spec.min || av > spec.max {
                return Err(ParseError::OutOfRange { field, offset: cursor });
            }
            if form == 1 {
                (av, if step_slice.is_some() { spec.max } else { av })
            } else {
                let zv = number(z.unwrap(), z_offset, field)?;
                if zv < spec.min || zv > spec.max {
                    return Err(ParseError::OutOfRange { field, offset: z_offset });
                }
                if av > zv {
                    return Err(ParseError::ReversedRange { field, offset: z_offset });
                }
                (av, zv)
            }
        };

        let step = match step_slice {
            Some(s) => {
                let n = number(s, step_offset, field)?;
                if n == 0 || n > spec.max - spec.min + 1 {
                    return Err(ParseError::BadStep { field, offset: step_offset });
                }
                n
            }
            None => 1,
        };

        let mut v = start;
        while v <= end_value {
            bits |= 1u64 << v;
            v += step;
        }

        if end == b.len() {
            break;
        }
        cursor = end + 1;
    }

    Ok((0..64).filter(|&n| bits & (1u64 << n) != 0).map(|n| n as u8).collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let mut fields: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for i in 0..5 {
        fields[i] = parse_spec(parts[i], SPECS[i])?;
    }
    let bits = std::array::from_fn(|i| {
        fields[i].iter().fold(0u64, |acc, &v| acc | (1u64 << v))
    });
    Ok(Cron {
        bits,
        literal_day_stars: [parts[2] == "*", parts[4] == "*"],
    })
}

        let (index, max) = match field {
            Field::Minute => (0, 59),
            Field::Hour => (1, 23),
            Field::DayOfMonth => (2, 31),
            Field::Month => (3, 12),
            Field::DayOfWeek => (4, 6),
        };
        (0..=max).filter(|&v| self.bits[index] & (1u64 << v) != 0).collect()
    }
}

        } else {
            let mut found = None;
            for (i, &b) in base.iter().enumerate() {
                if b == b'-' {
                    if found.is_some() {
                        return Err(make_error(field, item_offset, 2));
                    }
                    found = Some(i);
                }
            }
            match found {
                None if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => (1, None),
                Some(i)
                    if i > 0
                        && i + 1 < base.len()
                        && base[..i].iter().all(u8::is_ascii_digit)
                        && base[i + 1..].iter().all(u8::is_ascii_digit) =>
                {
                    (2, Some(i))
                }
                _ => return Err(make_error(field, item_offset, 2)),
            }
        };

        let step_range = if let Some(slash_at) = slash {
            let from = slash_at + 1;
            if from == item_bytes.len()
                || !item_bytes[from..].iter().all(u8::is_ascii_digit)
            {
                return Err(make_error(field, item_offset, 2));
            }
            Some((from, item_bytes.len()))
        } else {
            None
        };

        let (start_value, end_value) = match shape {
            0 => (min, max),
            1 => {
                let n = number(item_bytes, 0, base_end, field)?;
                if n < min || n > max {
                    return Err(make_error(field, item_offset, 5));
                }
                (n, if step_range.is_some() { max } else { n })
            }
            _ => {
                let h = hyphen.unwrap();
                let a = number(item_bytes, 0, h, field)?;
                if a < min || a > max {
                    return Err(make_error(field, item_offset, 5));
                }
                let b = number(item_bytes, h + 1, base_end, field)?;
                if b < min || b > max {
                    return Err(make_error(field, item_offset + h + 1, 5));
                }
                if a > b {
                    return Err(make_error(field, item_offset + h + 1, 6));
                }
                (a, b)
            }
        };

        let step = if let Some((from, to)) = step_range {
            let n = number(item_bytes, from, to, field)?;
            if n == 0 || n > width {
                return Err(make_error(field, item_offset + from, 7));
            }
            n
        } else {
            1
        };

        let mut n = start_value;
        while n <= end_value {
            *bits |= 1u64 << n;
            match n.checked_add(step) {
                Some(next) => n = next,
                None => break,
            }
        }

        if item_offset == 0 && item_bytes == b"*" {
            any_exact_star = true;
        }
        item_offset += item_bytes.len() + 1;
    }

    Ok(any_exact_star && text == "*")
}

fn parse_field(text: &str, field: Field) -> Result<(u64, bool), ParseError> {
    let mut bits = 0u64;
    let star = parse_one(text, field, &mut bits)?;
    Ok((bits, star))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }

    let order = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    let mut parsed = Vec::new();
    for (field, text) in order.into_iter().zip(fields.iter()) {
        parsed.push(parse_field(text, field)?);
    }

    Ok(Cron {
        minute: parsed[0].0,
        hour: parsed[1].0 as u32,
        day_of_month: parsed[2].0 as u32,
        month: parsed[3].0 as u16,
        day_of_week: parsed[4].0 as u8,
        day_of_month_star: parsed[2].1,
        day_of_week_star: parsed[4].1,
    })
}

fn make_error(field: Field, offset: usize, variant: u8) -> ParseError {

        0 => ParseError::InvalidCharacter { field, offset },
        1 => ParseError::EmptyItem { field, offset },
        2 => ParseError::InvalidSyntax { field, offset },
        3 => ParseError::LeadingZero { field, offset },
        4 => ParseError::NumberOverflow { field, offset },
        5 => ParseError::OutOfRange { field, offset },
        6 => ParseError::ReversedRange { field, offset },
        _ => ParseError::BadStep { field, offset },

fn number(
    bytes: &[u8],
    begin: usize,
    end: usize,
    field: Field,
    as_step: bool,
) -> Result<u32, ParseError> {
    if end - begin > 1 && bytes[begin] == b'0' {
        return Err(make_error(field, begin, 3));
    }
    let mut n = 0u32;
    for &b in &bytes[begin..end] {
        n = n
            .checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u32))
            .ok_or_else(|| make_error(field, begin, 4))?;
    }
    if as_step {
        let width = bounds(field).1 - bounds(field).0 + 1;
        if !(1..=width).contains(&n) {
            return Err(make_error(field, begin, 7));
        }
    } else {
        let (min, max) = bounds(field);
        if !(min..=max).contains(&n) {
            return Err(make_error(field, begin, 5));
        }
    }
    Ok(n)
}

fn parse_one(
    src: &str,
    field: Field,
    bits: &mut u64,
) -> Result<(), ParseError> {
    let b = src.as_bytes();
    for (i, &x) in b.iter().enumerate() {
        if !(x.is_ascii_digit() || matches!(x, b'*' | b'-' | b'/' | b',')) {
            return Err(make_error(field, i, 0));
        }
    }
    let (min, max) = bounds(field);
    let mut cursor = 0usize;
    loop {
        let end = b[cursor..]
            .iter()
            .position(|&x| x == b',')
            .map_or(b.len(), |p| cursor + p);
        if cursor == end {
            return Err(make_error(field, cursor, 1));
        }
        let item = &b[cursor..end];
        let slash = item.iter().position(|&x| x == b'/');
        if item.iter().filter(|&&x| x == b'/').count() > 1 {
            return Err(make_error(field, cursor, 2));
        }
        let base_end = slash.unwrap_or(item.len());
        let step_start = slash.map(|x| x + 1);
        let base = &item[..base_end];

        let (kind, first, second) = if base == b"*" {
            (0, None, None)
        } else {
            let hyphens: Vec<usize> = base.iter().enumerate()
                .filter_map(|(i, &x)| (x == b'-').then_some(i)).collect();
            match hyphens.as_slice() {
                [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) =>
                    (1, Some((0, base.len())), None),
                [h] if *h > 0 && *h + 1 < base.len()
                    && base[..*h].iter().all(u8::is_ascii_digit)
                    && base[*h + 1..].iter().all(u8::is_ascii_digit) =>
                    (2, Some((0, *h)), Some((*h + 1, base.len()))),
                _ => return Err(make_error(field, cursor, 2)),
            }
        };

        let (start, finish) = match kind {
            0 => (min, max),
            1 => {
                let (a, z) = first.unwrap();
                let n = number(b, cursor + a, cursor + z, field, false)?;
                (n, if slash.is_some() { max } else { n })
            }
            _ => {
                let (a, z) = first.unwrap();
                let lo = number(b, cursor + a, cursor + z, field, false)?;
                let (a, z) = second.unwrap();
                let hi = number(b, cursor + a, cursor + z, field, false)?;
                if lo > hi {
                    return Err(make_error(field, cursor + a, 6));
                }
                (lo, hi)
            }
        };

        let stride = if let Some(pos) = step_start {
            if pos == item.len() {
                return Err(make_error(field, cursor, 2));
            }
            number(b, cursor + pos, end, field, true)?
        } else {
            1
        };
        let mut n = start;
        while n <= finish {
            *bits |= 1u64 << (n - min);
            match n.checked_add(stride) {
                Some(next) => n = next,
                None => break,
            }
        }
        if end == b.len() { break; }
        cursor = end + 1;
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let fields = [Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek];
    let mut selected = [0u64; 5];
    let mut stars = [false; 5];
    for i in 0..5 {
        stars[i] = parts[i].as_bytes() == b"*";
        parse_one(parts[i], fields[i], &mut selected[i])?;
    }
    Ok(Cron { selected, dom_star: stars[2], dow_star: stars[4] })
}

    InvalidCharacter,
    EmptyItem,
    InvalidSyntax,
    LeadingZero,
    NumberOverflow,
    OutOfRange,

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    let digits = &bytes[start..end];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(error(field, start, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &digit in digits {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((digit - b'0') as u32))
            .ok_or_else(|| error(field, start, ErrorKind::NumberOverflow))?;
    }
    Ok(value)
}

fn parse_field(text: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit() || matches!(byte, b'*' | b'-' | b'/' | b',')) {
            return Err(error(field, offset, ErrorKind::InvalidCharacter));
        }
    }

    let (minimum, maximum) = bounds(field);
    let width = maximum - minimum + 1;
    let mut selected = vec![false; (maximum + 1) as usize];

    let mut item_start = 0;
    for item in text.split(',') {
        let item_bytes = item.as_bytes();
        let item_end = item_start + item_bytes.len();
        if item_bytes.is_empty() {
            return Err(error(field, item_start, ErrorKind::EmptyItem));
        }

        let slash_positions: Vec<usize> = item_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        let base_end = if slash_positions.is_empty() {
            item_bytes.len()
        } else if slash_positions.len() == 1 {
            slash_positions[0]
        } else {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        };

        let base = &item_bytes[..base_end];
        let range_positions: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let shape = if base == b"*" {
            0
        } else if range_positions.is_empty() && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            1
        } else if range_positions.len() == 1
            && range_positions[0] > 0
            && range_positions[0] + 1 < base.len()
            && base[..range_positions[0]].iter().all(u8::is_ascii_digit)
            && base[range_positions[0] + 1..].iter().all(u8::is_ascii_digit)
        {
            2
        } else {
            return Err(error(field, item_start, ErrorKind::InvalidSyntax));
        };

        let mut step = None;
        if !slash_positions.is_empty() {
            let step_start = slash_positions[0] + 1;
            if step_start >= item_bytes.len()
                || !item_bytes[step_start..].iter().all(u8::is_ascii_digit)
            {
                return Err(error(field, item_start, ErrorKind::InvalidSyntax));
            }
            step = Some((step_start, item_bytes.len()));
        }

        let (mut start_value, mut end_value) = match shape {
            0 => (minimum, maximum),
            1 => {
                let value = parse_number(item_bytes, 0, base_end, field)?;
                if value < minimum || value > maximum {
                    return Err(error(field, item_start, ErrorKind::OutOfRange));
                }
                (value, if step.is_some() { maximum } else { value })
            }
            _ => {
                let hyphen = range_positions[0];
                let start = parse_number(item_bytes, 0, hyphen, field)?;
                if start < minimum || start > maximum {
                    return Err(error(field, item_start, ErrorKind::OutOfRange));
                }
                let end = parse_number(item_bytes, hyphen + 1, base_end, field)?;
                if end < minimum || end > maximum {
                    return Err(error(field, item_start + hyphen + 1, ErrorKind::OutOfRange));
                }
                if start > end {
                    return Err(error(field, item_start + hyphen + 1, ErrorKind::ReversedRange));
                }
                (start, end)
            }
        };

        let stride = if let Some((step_start, step_end)) = step {
            let value = parse_number(item_bytes, step_start, step_end, field)?;
            if value == 0 || value > width {
                return Err(error(field, item_start + step_start, ErrorKind::BadStep));
            }
            value
        } else {
            1
        };

        while start_value <= end_value {
            selected[start_value as usize] = true;
            match start_value.checked_add(stride) {
                Some(next) => start_value = next,
                None => break,
            }
        }

        item_start = item_end + 1;
    }

    let values = selected
        .into_iter()
        .enumerate()
        .filter_map(|(value, yes)| yes.then_some(value as u8))
        .collect();

    Ok((values, text == "*"))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = Vec::with_capacity(5);
    let mut day_stars = [false; 2];

    for (index, (&field, text)) in fields.iter().zip(pieces.iter()).enumerate() {
        let (values, is_star) = parse_field(text, field)?;
        if index == 2 {
            day_stars[0] = is_star;
        } else if index == 4 {
            day_stars[1] = is_star;
        }
        selected.push(values);
    }

    let selected: [Vec<u8>; 5] = selected.try_into().ok().unwrap();
    Ok(Cron {
        selected,
        day_stars,
    })
}

    min: u32,
    max: u32,
}

const DOMAINS: [Domain; 5] = [
    Domain { field: Field::Minute, min: 0, max: 59 },
    Domain { field: Field::Hour, min: 0, max: 23 },
    Domain { field: Field::DayOfMonth, min: 1, max: 31 },

fn err(field: Field, offset: usize, make: fn(Field, usize) -> ParseError) -> ParseError {
    make(field, offset)
}

fn syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(bytes: &[u8], start: usize, field: Field) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax(field, start));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }
    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }
    Ok(value)
}

fn parse_field(text: &str, domain: Domain) -> Result<Vec<u8>, ParseError> {
    let field = domain.field;
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut selected = vec![false; (domain.max + 1) as usize];
    let mut item_start = 0usize;

    while item_start <= bytes.len() {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());
        let item = &bytes[item_start..item_end];

        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: item_start });
        }

        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        if slash_positions.len() > 1 {
            return Err(syntax(field, item_start));
        }
        let (base, step_part, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), item_start + slash + 1)
        } else {
            (item, None, item_end)
        };

        let (kind, first, second, second_offset) = if base == b"*" {
            (0u8, None, None, item_start)
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();
            if hyphens.len() > 1 {
                return Err(syntax(field, item_start));
            }
            if let Some(&hyphen) = hyphens.first() {
                if hyphen == 0 || hyphen + 1 == base.len() {
                    return Err(syntax(field, item_start));
                }
                (2u8, Some(&base[..hyphen]), Some(&base[hyphen + 1..]), item_start + hyphen + 1)
            } else if !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
                (1u8, Some(base), None, item_start)
            } else {
                return Err(syntax(field, item_start));
            }
        };

        if step_part.is_some_and(|s| s.is_empty() || !s.iter().all(u8::is_ascii_digit)) {
            return Err(syntax(field, item_start));
        }

        let mut start = domain.min;
        let mut end = domain.max;
        if kind != 0 {
            let first_bytes = first.unwrap();
            start = parse_number(first_bytes, item_start, field)?;
            if start < domain.min || start > domain.max {
                return Err(ParseError::OutOfRange { field, offset: item_start });
            }
            if kind == 2 {
                let second_bytes = second.unwrap();
                end = parse_number(second_bytes, second_offset, field)?;
                if end < domain.min || end > domain.max {
                    return Err(ParseError::OutOfRange { field, offset: second_offset });
                }
                if start > end {
                    return Err(ParseError::ReversedRange { field, offset: second_offset });
                }
            } else {
                end = if step_part.is_some() { domain.max } else { start };
            }
        }

        let step = if let Some(step_bytes) = step_part {
            let value = parse_number(step_bytes, step_offset, field)?;
            if value == 0 || value > domain.max - domain.min + 1 {
                return Err(ParseError::BadStep { field, offset: step_offset });
            }
            value
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(value, &yes)| yes.then_some(value as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let mut selected = Vec::with_capacity(5);
    for (part, domain) in parts.iter().zip(DOMAINS) {
        selected.push(parse_field(part, domain)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_star: [parts[2].as_bytes() == b"*", parts[4].as_bytes() == b"*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.selected[index].clone()
    }
}
