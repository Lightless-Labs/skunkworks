```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width)
        || !(1..=8).contains(&board.height)
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn line_indices(board: &Board, direction: Direction, line: usize) -> Vec<usize> {
    match direction {
        Direction::Left => (0..board.width).map(|x| line * board.width + x).collect(),
        Direction::Right => (0..board.width)
            .rev()
            .map(|x| line * board.width + x)
            .collect(),
        Direction::Up => (0..board.height).map(|y| y * board.width + line).collect(),
        Direction::Down => (0..board.height)
            .rev()
            .map(|y| y * board.width + line)
            .collect(),
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0; board.cells.len()];
    let mut score = 0u64;
    let mut merges = Vec::new();

    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line in 0..line_count {
        let indices = line_indices(board, direction, line);
        let tiles: Vec<(usize, u32)> = indices
            .iter()
            .filter_map(|&index| {
                let value = board.cells[index];
                (value != 0).then_some((index, value))
            })
            .collect();

        let mut output = Vec::new();
        let mut i = 0;
        while i < tiles.len() {
            if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
                let value = tiles[i].1;
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: tiles[i].0,
                        second: tiles[i + 1].0,
                    });
                }
                let merged_value = value * 2;
                let to = indices[output.len()];
                merges.push(Merge {
                    from: [tiles[i].0, tiles[i + 1].0],
                    to,
                    value: merged_value,
                });
                score += u64::from(merged_value);
                output.push(merged_value);
                i += 2;
            } else {
                output.push(tiles[i].1);
                i += 1;
            }
        }

        for (position, value) in output.into_iter().enumerate() {
            cells[indices[position]] = value;
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut legal = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(result) if result.changed => legal.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(legal)
}
```

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if board.width == 0
        || board.width > 8
        || board.height == 0
        || board.height > 8
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }
    for index in 0..board.cells.len() {
        let value = board.cells[index];
        if value != 0 && (value < 2 || value > 2_147_483_648 || (value & (value - 1)) != 0) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn coordinates(board: &Board, direction: Direction, line: usize, offset: usize) -> usize {
    match direction {
        Direction::Left => line * board.width + offset,
        Direction::Right => line * board.width + (board.width - 1 - offset),
        Direction::Up => offset * board.width + line,
        Direction::Down => (board.height - 1 - offset) * board.width + line,
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };
    let line_length = match direction {
        Direction::Left | Direction::Right => board.width,
        Direction::Up | Direction::Down => board.height,
    };

    let mut result_cells = vec![0u32; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    for line in 0..line_count {
        let mut inputs: Vec<(usize, u32)> = Vec::new();
        for offset in 0..line_length {
            let index = coordinates(board, direction, line, offset);
            let value = board.cells[index];
            if value != 0 {
                inputs.push((index, value));
            }
        }

        let mut cursor = 0;
        let mut output_offset = 0;
        while cursor < inputs.len() {
            let (index, value) = inputs[cursor];
            let target = coordinates(board, direction, line, output_offset);
            if cursor + 1 < inputs.len() && inputs[cursor + 1].1 == value {
                if value == 2_147_483_648 {
                    let (second, _) = inputs[cursor + 1];
                    return Err(Error::MergeOverflow { first: index, second });
                }
                let doubled = value * 2;
                result_cells[target] = doubled;
                merges.push(Merge {
                    from: [index, inputs[cursor + 1].0],
                    to: target,
                    value: doubled,
                });
                score += doubled as u64;
                cursor += 2;
            } else {
                result_cells[target] = value;
                cursor += 1;
            }
            output_offset += 1;
        }
    }

    let changed = result_cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells: result_cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut directions = Vec::new();
    let ordered = [Direction::Up, Direction::Left, Direction::Down, Direction::Right];
    for direction in ordered {
        if let Ok(result) = move_board(board, direction) {
            if result.changed {
                directions.push(direction);
            }
        }
    }
    Ok(directions)
}
```
