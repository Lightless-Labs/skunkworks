```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width)
        || !(1..=8).contains(&board.height)
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn lines(board: &Board, direction: Direction) -> Vec<Vec<usize>> {
    let mut result = Vec::new();
    match direction {
        Direction::Left | Direction::Right => {
            for row in 0..board.height {
                let indices: Vec<usize> = (0..board.width)
                    .map(|step| {
                        let col = if direction == Direction::Left {
                            step
                        } else {
                            board.width - 1 - step
                        };
                        row * board.width + col
                    })
                    .collect();
                result.push(indices);
            }
        }
        Direction::Up | Direction::Down => {
            for col in 0..board.width {
                let indices: Vec<usize> = (0..board.height)
                    .map(|step| {
                        let row = if direction == Direction::Up {
                            step
                        } else {
                            board.height - 1 - step
                        };
                        row * board.width + col
                    })
                    .collect();
                result.push(indices);
            }
        }
    }
    result
}

fn move_validated(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    let mut cells = vec![0; board.cells.len()];
    let mut score = 0u64;
    let mut merges = Vec::new();

    for line in lines(board, direction) {
        let occupied: Vec<(usize, u32)> = line
            .iter()
            .filter_map(|&index| {
                let value = board.cells[index];
                (value != 0).then_some((index, value))
            })
            .collect();

        let mut outputs: Vec<(u32, Option<[usize; 2]>)> = Vec::new();
        let mut i = 0;
        while i < occupied.len() {
            let (first_index, first_value) = occupied[i];
            if i + 1 < occupied.len() && occupied[i + 1].1 == first_value {
                let second_index = occupied[i + 1].0;
                if first_value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: first_index,
                        second: second_index,
                    });
                }
                let value = first_value * 2;
                outputs.push((value, Some([first_index, second_index])));
                score += u64::from(value);
                i += 2;
            } else {
                outputs.push((first_value, None));
                i += 1;
            }
        }

        for (position, (value, source)) in outputs.into_iter().enumerate() {
            let to = line[position];
            cells[to] = value;
            if let Some(from) = source {
                merges.push(Merge { from, to, value });
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    move_validated(board, direction)
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut result = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        if let Ok(moved) = move_validated(board, direction) {
            if moved.changed {
                result.push(direction);
            }
        }
    }
    Ok(result)
}
```

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if board.width == 0
        || board.width > 8
        || board.height == 0
        || board.height > 8
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }

    for index in 0..board.cells.len() {
        let value = board.cells[index];
        if value != 0
            && (value > 2_147_483_648 || value < 2 || value.count_ones() != 1)
        {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn coord_index(board: &Board, row: usize, col: usize) -> usize {
    row * board.width + col
}

fn process_line(
    board: &Board,
    cells: &mut [u32],
    score: &mut u64,
    merges: &mut Vec<Merge>,
    indices: &[usize],
) -> Result<(), Error> {
    let mut inputs = Vec::new();
    for &index in indices {
        let value = board.cells[index];
        if value != 0 {
            inputs.push((index, value));
        }
    }

    let mut outputs: Vec<(u32, Option<[usize; 2]>)> = Vec::new();
    let mut cursor = 0;
    while cursor < inputs.len() {
        let (first, value) = inputs[cursor];
        if cursor + 1 < inputs.len() && inputs[cursor + 1].1 == value {
            let second = inputs[cursor + 1].0;
            let doubled = value.checked_mul(2).ok_or(Error::MergeOverflow {
                first,
                second,
            })?;
            outputs.push((doubled, Some([first, second])));
            *score += doubled as u64;
            cursor += 2;
        } else {
            outputs.push((value, None));
            cursor += 1;
        }
    }

    for (position, (value, sources)) in outputs.into_iter().enumerate() {
        let target = indices[position];
        cells[target] = value;
        if let Some(from) = sources {
            merges.push(Merge {
                from,
                to: target,
                value,
            });
        }
    }
    Ok(())
}

fn move_validated(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    let mut cells = vec![0; board.cells.len()];
    let mut score = 0;
    let mut merges = Vec::new();

    match direction {
        Direction::Left | Direction::Right => {
            for row in 0..board.height {
                let indices: Vec<usize> = (0..board.width)
                    .map(|step| {
                        let col = match direction {
                            Direction::Left => step,
                            Direction::Right => board.width - step - 1,
                            _ => unreachable!(),
                        };
                        coord_index(board, row, col)
                    })
                    .collect();
                process_line(board, &mut cells, &mut score, &mut merges, &indices)?;
            }
        }
        Direction::Up | Direction::Down => {
            for col in 0..board.width {
                let indices: Vec<usize> = (0..board.height)
                    .map(|step| {
                        let row = match direction {
                            Direction::Up => step,
                            Direction::Down => board.height - step - 1,
                            _ => unreachable!(),
                        };
                        coord_index(board, row, col)
                    })
                    .collect();
                process_line(board, &mut cells, &mut score, &mut merges, &indices)?;
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    move_validated(board, direction)
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let order = [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ];
    let mut legal = Vec::new();
    for direction in order {
        match move_validated(board, direction) {
            Ok(result) if result.changed => legal.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(legal)
}
```
