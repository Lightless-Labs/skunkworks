```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().is_some_and(|tier| tier.up_to.is_some()) {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn add_band(
    total: &mut u64,
    bands: &mut Vec<Band>,
    index: usize,
    units: u64,
    unit: u64,
    start: u64,
) -> Result<(), Error> {
    if units == 0 {
        return Ok(());
    }

    let amount = units
        .checked_mul(unit)
        .and_then(|product| product.checked_add(start))
        .ok_or(Error::Overflow { index })?;
    *total = total
        .checked_add(amount)
        .ok_or(Error::Overflow { index })?;
    bands.push(Band {
        index,
        units,
        amount,
    });
    Ok(())
}

fn flat_charge(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    let mut amount = 0;
    let mut bands = Vec::new();
    add_band(&mut amount, &mut bands, 0, quantity, unit, start)?;
    Ok(Charge { amount, bands })
}

fn graduated_charge(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    let mut remaining = quantity;
    let mut previous = 0u64;
    let mut amount = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 {
            break;
        }

        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous;
                let allocated = remaining.min(width);
                previous = bound;
                allocated
            }
            None => remaining,
        };

        add_band(
            &mut amount,
            &mut bands,
            index,
            units,
            tier.unit,
            tier.start,
        )?;
        remaining -= units;
    }

    Ok(Charge { amount, bands })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => flat_charge(quantity, *unit, *start),
        Price::Tiers(tiers) => graduated_charge(quantity, tiers),
    }
}

fn volume_charge(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let index = tiers
        .iter()
        .position(|tier| tier.up_to.is_none_or(|bound| bound >= quantity))
        .expect("validated schedules have a final unbounded tier");
    let tier = &tiers[index];
    let mut amount = 0;
    let mut bands = Vec::new();
    add_band(
        &mut amount,
        &mut bands,
        index,
        quantity,
        tier.unit,
        tier.start,
    )?;
    Ok(Charge { amount, bands })
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Flat { unit, start }, _) => flat_charge(quantity, *unit, *start)?,
        (Price::Tiers(tiers), Mode::Graduated) => graduated_charge(quantity, tiers)?,
        (Price::Tiers(tiers), Mode::Volume) => volume_charge(quantity, tiers)?,
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .expect("minimum adjustment cannot overflow the u64 minimum");
    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn check_schedule(tiers: &[Tier]) -> Result<(), Error> {
    match tiers.len() {
        0 => return Err(Error::Empty),
        1..=64 => {}
        _ => return Err(Error::TooMany),
    }

    let mut last_bound = 0u64;
    for (i, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= last_bound {
                return Err(Error::Bound { index: i });
            }
            last_bound = bound;
        } else if i != tiers.len() - 1 {
            return Err(Error::Unbounded { index: i });
        }
    }

    if tiers[tiers.len() - 1].up_to.is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn price_band(index: usize, units: u64, tier: &Tier) -> Result<Band, Error> {
    let product = units
        .checked_mul(tier.unit)
        .ok_or(Error::Overflow { index })?;
    let amount = product
        .checked_add(tier.start)
        .ok_or(Error::Overflow { index })?;
    Ok(Band {
        index,
        units,
        amount,
    })
}

fn charge_flat(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }
    let product = quantity.checked_mul(unit).ok_or(Error::Overflow { index: 0 })?;
    let amount = product
        .checked_add(start)
        .ok_or(Error::Overflow { index: 0 })?;
    Ok(Charge {
        amount,
        bands: vec![Band {
            index: 0,
            units: quantity,
            amount,
        }],
    })
}

fn graduate(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    check_schedule(tiers)?;
    let mut bands = Vec::new();
    let mut sum = 0u64;
    let mut lower = 0u64;
    let mut remaining = quantity;

    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 {
            break;
        }
        let units = match tier.up_to {
            Some(upper) => {
                let width = upper - lower;
                lower = upper;
                remaining.min(width)
            }
            None => remaining,
        };
        if units != 0 {
            let band = price_band(index, units, tier)?;
            sum = sum
                .checked_add(band.amount)
                .ok_or(Error::Overflow { index })?;
            bands.push(band);
            remaining -= units;
        }
    }
    Ok(Charge { amount: sum, bands })
}

fn volume(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    check_schedule(tiers)?;
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let mut chosen = tiers.len() - 1;
    for (i, tier) in tiers.iter().enumerate() {
        if tier.up_to.map_or(true, |bound| quantity <= bound) {
            chosen = i;
            break;
        }
    }
    let band = price_band(chosen, quantity, &tiers[chosen])?;
    Ok(Charge {
        amount: band.amount,
        bands: vec![band],
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => charge_flat(quantity, *unit, *start),
        Price::Tiers(tiers) => graduate(quantity, tiers),
    }
}

pub fn bill(
    quantity: u64,
    price: &Price,
    mode: Mode,
    minimum: u64,
) -> Result<Bill, Error> {
    let charge = match price {
        Price::Flat { unit, start } => charge_flat(quantity, *unit, *start)?,
        Price::Tiers(tiers) => match mode {
            Mode::Graduated => graduate(quantity, tiers)?,
            Mode::Volume => volume(quantity, tiers)?,
        },
    };
    let minimum_adjustment = if quantity == 0 {
        0
    } else {
        minimum.saturating_sub(charge.amount)
    };
    let total = charge.amount + minimum_adjustment;
    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```
