```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous_bound = 0;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous_bound {
                    return Err(Error::Bound { index });
                }
                previous_bound = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().and_then(|tier| tier.up_to).is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn checked_band_amount(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    units
        .checked_mul(unit)
        .and_then(|product| product.checked_add(start))
        .ok_or(Error::Overflow { index })
}

fn graduated(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }
            let amount = checked_band_amount(quantity, *unit, *start, 0)?;
            Ok(Charge {
                amount,
                bands: vec![Band {
                    index: 0,
                    units: quantity,
                    amount,
                }],
            })
        }
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            let mut remaining = quantity;
            let mut previous_bound = 0;
            let mut total = 0u64;
            let mut bands = Vec::new();

            for (index, tier) in tiers.iter().enumerate() {
                if remaining == 0 {
                    break;
                }
                let units = match tier.up_to {
                    Some(bound) => {
                        let width = bound - previous_bound;
                        previous_bound = bound;
                        remaining.min(width)
                    }
                    None => remaining,
                };
                if units == 0 {
                    continue;
                }
                let amount = checked_band_amount(units, tier.unit, tier.start, index)?;
                total = total
                    .checked_add(amount)
                    .ok_or(Error::Overflow { index })?;
                bands.push(Band {
                    index,
                    units,
                    amount,
                });
                remaining -= units;
            }

            Ok(Charge {
                amount: total,
                bands,
            })
        }
    }
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    graduated(quantity, price)
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (&mode, price) {
        (_, Price::Flat { .. }) | (Mode::Graduated, _) => graduated(quantity, price)?,
        (Mode::Volume, Price::Tiers(tiers)) => {
            validate_tiers(tiers)?;
            if quantity == 0 {
                Charge {
                    amount: 0,
                    bands: Vec::new(),
                }
            } else {
                let (index, tier) = tiers
                    .iter()
                    .enumerate()
                    .find(|(_, tier)| tier.up_to.map_or(true, |bound| bound >= quantity))
                    .expect("validated tiers end in an unbounded tier");
                let amount = checked_band_amount(quantity, tier.unit, tier.start, index)?;
                Charge {
                    amount,
                    bands: vec![Band {
                        index,
                        units: quantity,
                        amount,
                    }],
                }
            }
        }
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .expect("minimum adjustment cannot make total exceed minimum or charge");
    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate(price: &Price) -> Result<(), Error> {
    let Price::Tiers(tiers) = price else {
        return Ok(());
    };
    match tiers.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }

    let mut last_finite = 0;
    for (index, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= last_finite {
                return Err(Error::Bound { index });
            }
            last_finite = bound;
        } else if index != tiers.len() - 1 {
            return Err(Error::Unbounded { index });
        }
    }
    if tiers[tiers.len() - 1].up_to.is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn amount(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    match units.checked_mul(unit) {
        Some(product) => product.checked_add(start).ok_or(Error::Overflow { index }),
        None => Err(Error::Overflow { index }),
    }
}

fn charge_band(index: usize, units: u64, unit: u64, start: u64) -> Result<Band, Error> {
    Ok(Band {
        index,
        units,
        amount: amount(units, unit, start, index)?,
    })
}

fn calculate(quantity: u64, price: &Price, mode: Mode) -> Result<Charge, Error> {
    validate(price)?;
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                })
            } else {
                let band = charge_band(0, quantity, *unit, *start)?;
                Ok(Charge {
                    amount: band.amount,
                    bands: vec![band],
                })
            }
        }
        Price::Tiers(tiers) => {
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }

            if mode == Mode::Volume {
                let index = tiers
                    .iter()
                    .position(|tier| tier.up_to.is_none_or(|bound| quantity <= bound))
                    .unwrap();
                let tier = &tiers[index];
                let band = charge_band(index, quantity, tier.unit, tier.start)?;
                return Ok(Charge {
                    amount: band.amount,
                    bands: vec![band],
                });
            }

            let mut remaining = quantity;
            let mut lower = 0u64;
            let mut total = 0u64;
            let mut bands = Vec::new();
            for (index, tier) in tiers.iter().enumerate() {
                if remaining == 0 {
                    break;
                }
                let units = tier
                    .up_to
                    .map(|bound| remaining.min(bound - lower))
                    .unwrap_or(remaining);
                if let Some(bound) = tier.up_to {
                    lower = bound;
                }
                if units != 0 {
                    let band = charge_band(index, units, tier.unit, tier.start)?;
                    total = total
                        .checked_add(band.amount)
                        .ok_or(Error::Overflow { index })?;
                    bands.push(band);
                    remaining -= units;
                }
            }
            Ok(Charge {
                amount: total,
                bands,
            })
        }
    }
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    calculate(quantity, price, Mode::Graduated)
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = calculate(quantity, price, mode)?;
    let minimum_adjustment = if quantity == 0 {
        0
    } else {
        minimum.saturating_sub(charge.amount)
    };
    let total = charge.amount + minimum_adjustment;
    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```
