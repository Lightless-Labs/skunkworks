```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    ExactPayment { price: u16, credit: u16 },
    TenderMismatch {
        index: usize,
        expected: u16,
        found: u16,
    },
}

pub struct Machine {
    products: HashMap<String, u16>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
}

impl Machine {
    pub fn new(products: Vec<Product>) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut catalog = HashMap::with_capacity(products.len());
        for product in products {
            let valid_id = (1..=8).contains(&product.id.len())
                && product.id.bytes().all(|byte| byte.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId { id: product.id });
            }
            if catalog.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id });
            }
            catalog.insert(product.id, product.price);
        }

        Ok(Self {
            products: catalog,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !matches!(coin, 1 | 3 | 4 | 10) {
            return Err(Error::BadCoin { coin });
        }
        if self.credit + coin > 100 {
            return Err(Error::CreditLimit);
        }
        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.contains_key(id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.as_ref().ok_or(Error::NoSelection)?;
        let price = *self.products.get(id).ok_or(Error::NoSelection)?;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        if self.credit > price {
            return Err(Error::ExactPayment {
                price,
                credit: self.credit,
            });
        }

        let mut expected = Vec::new();
        let mut remaining = price;
        for denomination in [10_u16, 4, 3, 1] {
            while remaining >= denomination {
                expected.push(denomination);
                remaining -= denomination;
            }
        }

        let mut found = self.escrow.clone();
        found.sort_unstable_by(|a, b| b.cmp(a));
        for index in 0..expected.len().max(found.len()) {
            if expected.get(index) != found.get(index) {
                return Err(Error::TenderMismatch {
                    index,
                    expected: expected[index],
                    found: found[index],
                });
            }
        }

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: Vec::new(),
        };
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(coins)
    }
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    ExactPayment { price: u16, credit: u16 },
    TenderMismatch {
        index: usize,
        expected: u16,
        found: u16,
    },
}

pub struct Machine {
    products: Vec<Product>,
    selected: Option<usize>,
    escrow: Vec<u16>,
    credit: u16,
}

impl Machine {
    pub fn new(products: Vec<Product>) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        for i in 0..products.len() {
            let product = &products[i];
            if !(1..=8).contains(&product.id.len())
                || !product.id.bytes().all(|byte| byte.is_ascii_lowercase())
            {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if products[..i].iter().any(|prior| prior.id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        Ok(Self {
            products,
            selected: None,
            escrow: Vec::new(),
            credit: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self
                .selected
                .map(|index| self.products[index].id.clone()),
            escrow: self.escrow.clone(),
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if coin != 1 && coin != 3 && coin != 4 && coin != 10 {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(index) = self.products.iter().position(|product| product.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected == Some(index) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(index);
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(index) = self.selected else {
            return Err(Error::NoSelection);
        };
        let product = &self.products[index];
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        if self.credit > price {
            return Err(Error::ExactPayment {
                price,
                credit: self.credit,
            });
        }

        let mut expected = Vec::new();
        let mut remaining = price;
        for denomination in [10_u16, 4, 3, 1] {
            let count = remaining / denomination;
            expected.extend(std::iter::repeat_n(denomination, usize::from(count)));
            remaining %= denomination;
        }
        let mut actual = self.escrow.clone();
        actual.sort_unstable_by(|left, right| right.cmp(left));

        for index in 0..expected.len().min(actual.len()) {
            if expected[index] != actual[index] {
                return Err(Error::TenderMismatch {
                    index,
                    expected: expected[index],
                    found: actual[index],
                });
            }
        }

        let receipt = Receipt {
            id: product.id.clone(),
            price,
            change: Vec::new(),
        };
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.selected = None;
        self.credit = 0;
        Ok(returned)
    }
}
```
