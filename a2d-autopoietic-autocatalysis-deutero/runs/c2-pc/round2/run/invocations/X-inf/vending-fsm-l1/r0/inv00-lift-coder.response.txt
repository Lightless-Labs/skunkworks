```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    ExactPayment { price: u16, credit: u16 },
    TenderMismatch {
        index: usize,
        expected: u16,
        found: u16,
    },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
}

impl Machine {
    pub fn new(products: Vec<Product>) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<String> = Vec::new();
        for product in &products {
            let valid_id = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.iter().any(|id| id == &product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(product.id.clone());
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !matches!(coin, 1 | 3 | 4 | 10) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.products.iter().any(|product| product.id == id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match self.selected.as_deref() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };
        let product = self
            .products
            .iter()
            .find(|product| product.id == selected_id)
            .expect("selected product is always in the catalog");
        let price = product.price;

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        if self.credit > price {
            return Err(Error::ExactPayment {
                price,
                credit: self.credit,
            });
        }

        let mut expected = Vec::new();
        let mut remaining = price;
        for denomination in [10u16, 4, 3, 1] {
            while remaining >= denomination {
                expected.push(denomination);
                remaining -= denomination;
            }
        }

        let mut found = self.escrow.clone();
        found.sort_unstable_by(|a, b| b.cmp(a));
        for index in 0..expected.len() {
            if expected[index] != found[index] {
                return Err(Error::TenderMismatch {
                    index,
                    expected: expected[index],
                    found: found[index],
                });
            }
        }

        let receipt = Receipt {
            id: selected_id.to_owned(),
            price,
            change: Vec::new(),
        };
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    ExactPayment { price: u16, credit: u16 },
    TenderMismatch {
        index: usize,
        expected: u16,
        found: u16,
    },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<usize>,
    escrow: Vec<u16>,
}

impl Machine {
    pub fn new(products: Vec<Product>) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        for index in 0..products.len() {
            let id = &products[index].id;
            if id.is_empty()
                || id.len() > 8
                || !id.bytes().all(|byte| byte.is_ascii_lowercase())
            {
                return Err(Error::BadId { id: id.clone() });
            }
            if products[..index].iter().any(|previous| previous.id == *id) {
                return Err(Error::DuplicateId { id: id.clone() });
            }
            if !(1..=100).contains(&products[index].price) {
                return Err(Error::BadPrice { id: id.clone() });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self
                .selected
                .map(|index| self.products[index].id.clone()),
            escrow: self.escrow.clone(),
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if coin != 1 && coin != 3 && coin != 4 && coin != 10 {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let index = match self.products.iter().position(|product| product.id == id) {
            Some(index) => index,
            None => return Err(Error::UnknownProduct { id: id.to_owned() }),
        };
        if self.selected == Some(index) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(index);
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let index = match self.selected {
            Some(index) => index,
            None => return Err(Error::NoSelection),
        };
        let product = &self.products[index];
        let price = product.price;

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        if self.credit > price {
            return Err(Error::ExactPayment {
                price,
                credit: self.credit,
            });
        }

        let mut canonical = Vec::new();
        let mut remainder = price;
        while remainder >= 10 {
            canonical.push(10);
            remainder -= 10;
        }
        while remainder >= 4 {
            canonical.push(4);
            remainder -= 4;
        }
        while remainder >= 3 {
            canonical.push(3);
            remainder -= 3;
        }
        while remainder >= 1 {
            canonical.push(1);
            remainder -= 1;
        }

        let mut tender = self.escrow.clone();
        tender.sort_unstable_by(|left, right| right.cmp(left));
        let mismatch = canonical
            .iter()
            .copied()
            .zip(tender.iter().copied())
            .enumerate()
            .find(|(_, (expected, found))| expected != found);
        if let Some((position, (expected, found))) = mismatch {
            return Err(Error::TenderMismatch {
                index: position,
                expected,
                found,
            });
        }

        let receipt = Receipt {
            id: product.id.clone(),
            price,
            change: Vec::new(),
        };
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        Ok(coins)
    }
}
```
