#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
    Month,
    Period,
    Credit { index: usize },
    CreditOrder { index: usize },
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().and_then(|tier| tier.up_to).is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn band_amount(index: usize, units: u64, unit: u64, start: u64) -> Result<u64, Error> {
    let product = units.checked_mul(unit).ok_or(Error::Overflow { index })?;
    product.checked_add(start).ok_or(Error::Overflow { index })
}

fn add_amount(total: u64, amount: u64, index: usize) -> Result<u64, Error> {
    total.checked_add(amount).ok_or(Error::Overflow { index })
}

fn flat_charge(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }
    let amount = band_amount(0, quantity, unit, start)?;
    Ok(Charge {
        amount,
        bands: vec![Band {
            index: 0,
            units: quantity,
            amount,
        }],
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => flat_charge(quantity, *unit, *start),
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }

            let mut remaining = quantity;
            let mut previous = 0u64;
            let mut total = 0u64;
            let mut bands = Vec::new();

            for (index, tier) in tiers.iter().enumerate() {
                let units = match tier.up_to {
                    Some(bound) => {
                        let width = bound - previous;
                        let allocated = remaining.min(width);
                        previous = bound;
                        allocated
                    }
                    None => remaining,
                };

                if units > 0 {
                    let amount = band_amount(index, units, tier.unit, tier.start)?;
                    total = add_amount(total, amount, index)?;
                    bands.push(Band {
                        index,
                        units,
                        amount,
                    });
                    remaining -= units;
                }
            }

            Ok(Charge { amount: total, bands })
        }
    }
}

fn volume_charge(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    for (index, tier) in tiers.iter().enumerate() {
        if tier.up_to.map_or(true, |bound| quantity <= bound) {
            let amount = band_amount(index, quantity, tier.unit, tier.start)?;
            return Ok(Charge {
                amount,
                bands: vec![Band {
                    index,
                    units: quantity,
                    amount,
                }],
            });
        }
    }

    unreachable!()
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(tiers), Mode::Volume) => volume_charge(quantity, tiers)?,
        (_, Mode::Graduated) => quote(quantity, price)?,
        (Price::Flat { unit, start }, Mode::Volume) => flat_charge(quantity, *unit, *start)?,
    };

    let minimum_adjustment = if quantity > 0 && minimum > charge.amount {
        minimum - charge.amount
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .ok_or(Error::Overflow {
            index: charge.bands.last().map_or(0, |band| band.index),
        })?;

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if !matches!(days, 28 | 29 | 30 | 31) {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut previous_credit_end = start;
    let mut credited_days = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_credit_end {
            return Err(Error::CreditOrder { index });
        }
        credited_days += credit.end - credit.start;
        previous_credit_end = credit.end;
    }

    let active_days = end - start - credited_days;
    let numerator = u128::from(fee) * u128::from(active_days);
    let denominator = u128::from(days);
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;

    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => {
            if remainder == 0 { quotient } else { quotient + 1 }
        }
        Rounding::HalfEven => {
            let twice = remainder * 2;
            if twice < denominator {
                quotient
            } else if twice > denominator || quotient % 2 == 1 {
                quotient + 1
            } else {
                quotient
            }
        }
    };

    Ok(Proration {
        active_days,
        amount: rounded as u64,
    })
}

pub struct DayRange {

    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
    Month,
    Period,
    Credit { index: usize },
    CreditOrder { index: usize },
}

fn checked_band(index: usize, units: u64, tier: &Tier) -> Result<u64, Error> {
    let product = u128::from(units) * u128::from(tier.unit);
    let sum = product + u128::from(tier.start);
    u64::try_from(sum).map_err(|_| Error::Overflow { index })
}

fn validate(prices: &[Tier]) -> Result<(), Error> {
    if prices.len() == 0 {
        return Err(Error::Empty);
    }
    if prices.len() > 64 {
        return Err(Error::TooMany);
    }
    let mut prior = 0;
    for (i, tier) in prices.iter().enumerate() {
        match tier.up_to {
            Some(limit) if limit > prior => prior = limit,
            Some(_) => return Err(Error::Bound { index: i }),
            None if i != prices.len() - 1 => return Err(Error::Unbounded { index: i }),
            None => {}
        }
    }
    if prices[prices.len() - 1].up_to.is_some() {
        Err(Error::MissingFinal)
    } else {
        Ok(())
    }
}

fn flat(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge { amount: 0, bands: vec![] });
    }
    let product = u128::from(quantity) * u128::from(unit);
    let amount = u64::try_from(product + u128::from(start))
        .map_err(|_| Error::Overflow { index: 0 })?;
    Ok(Charge {
        amount,
        bands: vec![Band { index: 0, units: quantity, amount }],
    })
}

fn graduated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate(tiers)?;
    let mut remaining = quantity;
    let mut lower = 0u64;
    let mut total = 0u64;
    let mut bands = Vec::new();

    for (index, tier) in tiers.iter().enumerate() {
        let allocated = match tier.up_to {
            Some(bound) => {
                let size = bound - lower;
                lower = bound;
                remaining.min(size)
            }
            None => remaining,
        };
        if allocated != 0 {
            let amount = checked_band(index, allocated, tier)?;
            total = total
                .checked_add(amount)
                .ok_or(Error::Overflow { index })?;
            bands.push(Band { index, units: allocated, amount });
            remaining -= allocated;
        }
    }
    Ok(Charge { amount: total, bands })
}

fn volume(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate(tiers)?;
    if quantity == 0 {
        return Ok(Charge { amount: 0, bands: vec![] });
    }
    let index = tiers
        .iter()
        .position(|tier| tier.up_to.map_or(true, |bound| quantity <= bound))
        .unwrap();
    let tier = &tiers[index];
    let amount = checked_band(index, quantity, tier)?;
    Ok(Charge {
        amount,
        bands: vec![Band { index, units: quantity, amount }],
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => flat(quantity, *unit, *start),
        Price::Tiers(tiers) => graduated(quantity, tiers),
    }
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(tiers), Mode::Volume) => volume(quantity, tiers)?,
        (Price::Flat { unit, start }, Mode::Volume) => flat(quantity, *unit, *start)?,
        (_, Mode::Graduated) => quote(quantity, price)?,
    };
    let minimum_adjustment = if quantity != 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .ok_or(Error::Overflow {
            index: charge.bands.last().map_or(0, |band| band.index),
        })?;
    Ok(Bill { charge, minimum_adjustment, total })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }
    let mut cursor = start;
    let mut credit_total = 0u32;
    for (i, range) in credits.iter().enumerate() {
        if range.start < start || range.start >= range.end || range.end > end {
            return Err(Error::Credit { index: i });
        }
        if range.start < cursor {
            return Err(Error::CreditOrder { index: i });
        }
        credit_total += range.end - range.start;
        cursor = range.end;
    }
    let active_days = end - start - credit_total;

    let numerator = u128::from(fee) * u128::from(active_days);
    let denominator = u128::from(days);
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let result = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            match (remainder * 2).cmp(&denominator) {
                std::cmp::Ordering::Less => quotient,
                std::cmp::Ordering::Greater => quotient + 1,
                std::cmp::Ordering::Equal if quotient % 2 == 1 => quotient + 1,
                std::cmp::Ordering::Equal => quotient,
            }
        }
    };
    Ok(Proration {
        active_days,
        amount: result as u64,
    })
}
