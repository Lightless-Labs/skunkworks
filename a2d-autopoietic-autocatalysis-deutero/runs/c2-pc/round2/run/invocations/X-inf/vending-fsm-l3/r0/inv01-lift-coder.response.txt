```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    products: Vec<Product>,
    state: State,
    events: Vec<Event>,
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn coin_index(coin: u16) -> Option<usize> {
    DENOMS.iter().position(|&d| d == coin)
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = std::collections::HashSet::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if !seen.insert(product.id.clone()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            products,
            state: State {
                credit: 0,
                selected: None,
                escrow: Vec::new(),
                tubes,
            },
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        self.state.clone()
    }

    fn product_index(&self, id: &str) -> Option<usize> {
        self.products.iter().position(|p| p.id == id)
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if coin_index(coin).is_none() {
            return Err(Error::BadCoin { coin });
        }
        let credit = self.state.credit as u32 + coin as u32;
        if credit > 100 {
            return Err(Error::CreditLimit);
        }
        self.state.credit = credit as u16;
        self.state.escrow.push(coin);
        self.log(EventKind::Inserted {
            coin,
            credit: self.state.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.product_index(id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.state.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.state.selected = Some(id.to_owned());
        self.log(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = match self.state.selected.as_deref() {
            Some(id) => id.to_owned(),
            None => return Err(Error::NoSelection),
        };
        let index = self.product_index(&id).expect("selected product exists");
        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        let credit = self.state.credit;
        if credit < price {
            return Err(Error::Underpaid { price, credit });
        }
        let amount = credit - price;
        let change = match choose_change(amount, &self.state.tubes) {
            Some(coins) => coins,
            None => return Err(Error::NoChange { amount }),
        };

        let mut proposed = self.state.tubes;
        for &coin in &change {
            let i = coin_index(coin).expect("change uses valid denominations");
            proposed[i] -= 1;
        }
        for &coin in &self.state.escrow {
            let i = coin_index(coin).expect("escrow contains valid denominations");
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        self.products[index].stock -= 1;
        self.state.tubes = proposed;
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();
        let receipt = Receipt {
            id: id.clone(),
            price,
            change: change.clone(),
        };
        self.log(EventKind::Vended {
            id,
            price,
            change,
        });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.state.escrow.iter().rev().copied().collect();
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let index = match self.product_index(id) {
            Some(index) => index,
            None => return Err(Error::UnknownProduct { id: id.to_owned() }),
        };
        if self.state.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = match self.product_index(id) {
            Some(index) => index,
            None => return Err(Error::UnknownProduct { id: id.to_owned() }),
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let stock = self.products[index].stock as u32 + count as u32;
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[index].stock = stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut remaining = amount as usize;
    let mut greedy = [0usize; 4];
    for i in (0..4).rev() {
        let denomination = DENOMS[i] as usize;
        let take = (tubes[i] as usize).min(remaining / denomination);
        greedy[i] = take;
        remaining -= take * denomination;
    }
    if remaining == 0 {
        return Some(counts_to_descending(&greedy));
    }

    let target = amount as usize;
    let mut best: Option<[usize; 4]> = None;
    search_change(0, target, tubes, [0; 4], &mut best);

    best.map(|counts| counts_to_descending(&counts))
}

fn search_change(
    index: usize,
    remaining: usize,
    tubes: &[u16; 4],
    counts: [usize; 4],
    best: &mut Option<[usize; 4]>,
) {
    if index == 4 {
        if remaining == 0 {
            let replace = match best {
                None => true,
                Some(current) => {
                    let count = counts.iter().sum::<usize>();
                    let current_count = current.iter().sum::<usize>();
                    count < current_count
                        || (count == current_count
                            && (0..4).rev().find_map(|i| {
                                if counts[i] != current[i] {
                                    Some(counts[i] > current[i])
                                } else {
                                    None
                                }
                            }) == Some(true))
                }
            };
            if replace {
                *best = Some(counts);
            }
        }
        return;
    }

    let denomination = DENOMS[index] as usize;
    let max_take = (tubes[index] as usize).min(remaining / denomination);
    for take in 0..=max_take {
        let mut next = counts;
        next[index] = take;
        search_change(
            index + 1,
            remaining - take * denomination,
            tubes,
            next,
            best,
        );
    }
}

fn counts_to_descending(counts: &[usize; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            coins.push(DENOMS[i]);
        }
    }
    coins
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    catalog: Vec<Product>,
    credit: u16,
    selected_id: Option<String>,
    escrow_coins: Vec<u16>,
    tube_counts: [u16; 4],
    log: Vec<Event>,
}

const DENOMINATIONS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }
        let mut seen: Vec<&str> = Vec::new();
        for p in &products {
            if !(1..=8).contains(&p.id.len())
                || !p.id.bytes().all(|b| b.is_ascii_lowercase())
            {
                return Err(Error::BadId { id: p.id.clone() });
            }
            if seen.iter().any(|id| *id == p.id) {
                return Err(Error::DuplicateId { id: p.id.clone() });
            }
            seen.push(&p.id);
            if p.price == 0 || p.price > 100 {
                return Err(Error::BadPrice { id: p.id.clone() });
            }
            if p.stock > 100 {
                return Err(Error::StockLimit { id: p.id.clone() });
            }
        }
        for (i, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }
        Ok(Self {
            catalog: products,
            credit: 0,
            selected_id: None,
            escrow_coins: Vec::new(),
            tube_counts: tubes,
            log: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected_id.clone(),
            escrow: self.escrow_coins.clone(),
            tubes: self.tube_counts,
        }
    }

    fn find_product(&self, id: &str) -> Option<usize> {
        self.catalog.iter().position(|p| p.id == id)
    }

    fn push_event(&mut self, kind: EventKind) {
        self.log.push(Event {
            seq: self.log.len(),
            kind,
        });
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMINATIONS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if self.credit as u32 + coin as u32 > 100 {
            return Err(Error::CreditLimit);
        }
        let credit = self.credit + coin;
        self.escrow_coins.push(coin);
        self.credit = credit;
        self.push_event(EventKind::Inserted { coin, credit });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.find_product(id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected_id.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected_id = Some(id.to_owned());
        self.push_event(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self
            .selected_id
            .clone()
            .ok_or(Error::NoSelection)?;
        let index = self.find_product(&id).expect("selection is catalogued");
        let p = &self.catalog[index];
        if p.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = p.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        let amount = self.credit - price;
        let chosen = find_change(amount, self.tube_counts)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tube_counts;
        for coin in &chosen {
            let i = DENOMINATIONS.iter().position(|d| d == coin).unwrap();
            proposed[i] -= 1;
        }
        for coin in &self.escrow_coins {
            let i = DENOMINATIONS.iter().position(|d| d == coin).unwrap();
            proposed[i] = proposed[i].checked_add(1).unwrap();
        }
        for (i, count) in proposed.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        self.catalog[index].stock -= 1;
        self.tube_counts = proposed;
        self.credit = 0;
        self.selected_id = None;
        self.escrow_coins.clear();
        self.push_event(EventKind::Vended {
            id: id.clone(),
            price,
            change: chosen.clone(),
        });
        Ok(Receipt {
            id,
            price,
            change: chosen,
        })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow_coins.iter().rev().copied().collect();
        self.escrow_coins.clear();
        self.selected_id = None;
        self.credit = 0;
        self.push_event(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut result = self.catalog.clone();
        result.sort_by(|a, b| a.id.cmp(&b.id));
        result
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let index = self
            .find_product(id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.catalog[index].price;
        self.catalog[index].price = price;
        self.push_event(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = self
            .find_product(id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let updated = self.catalog[index].stock as u32 + count as u32;
        if updated > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.catalog[index].stock = updated as u16;
        self.push_event(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: updated as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.log
    }
}

fn find_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }
    let mut counts = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let denom = DENOMINATIONS[i];
        let take = tubes[i].min(remaining / denom);
        counts[i] = take;
        remaining -= take * denom;
    }
    if remaining == 0 {
        return Some(expand(counts));
    }

    let mut best: Option<([u16; 4], u16)> = None;
    enumerate(0, amount, tubes, [0; 4], &mut best);
    best.map(|(c, _)| expand(c))
}

fn enumerate(
    i: usize,
    remaining: u16,
    tubes: [u16; 4],
    counts: [u16; 4],
    best: &mut Option<([u16; 4], u16)>,
) {
    if i == 4 {
        if remaining != 0 {
            return;
        }
        let total: u16 = counts.iter().sum();
        let replace = match best {
            None => true,
            Some((old, old_total)) => {
                total < *old_total
                    || (total == *old_total && {
                        let mut comparison = std::cmp::Ordering::Equal;
                        for j in (0..4).rev() {
                            if counts[j] != old[j] {
                                comparison = counts[j].cmp(&old[j]);
                                break;
                            }
                        }
                        comparison == std::cmp::Ordering::Greater
                    })
            }
        };
        if replace {
            *best = Some((counts, total));
        }
        return;
    }

    let denom = DENOMINATIONS[i];
    let max = tubes[i].min(remaining / denom);
    for take in 0..=max {
        let mut next = counts;
        next[i] = take;
        enumerate(i + 1, remaining - take * denom, tubes, next, best);
    }
}

fn expand(counts: [u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            result.push(DENOMINATIONS[i]);
        }
    }
    result
}
```
