```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

struct Specifier<'a> {
    name: &'a str,
    start: usize,
}

struct Expression<'a> {
    operator: Operator,
    specifiers: Vec<Specifier<'a>>,
}

fn is_supported_operator(byte: u8) -> bool {
    matches!(byte, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
}

fn is_reserved_operator(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let content_start = open + 1;
    let content = &template[content_start..close];

    if let Some(relative) = content.as_bytes().iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen {
            pos: content_start + relative,
        });
    }

    let bytes = content.as_bytes();
    let (operator, names_start) = if let Some(&first) = bytes.first() {
        if is_supported_operator(first) {
            let operator = match first {
                b'+' => Operator::Reserved,
                b'#' => Operator::Fragment,
                b'.' => Operator::Dot,
                b'/' => Operator::Slash,
                b';' => Operator::Semicolon,
                b'?' => Operator::Query,
                b'&' => Operator::Continuation,
                _ => unreachable!(),
            };
            (operator, content_start + 1)
        } else if matches!(first, b'=' | b'!' | b'@' | b'|' | b'%') {
            return Err(Error::UnsupportedOperator { pos: content_start });
        } else {
            (Operator::Simple, content_start)
        }
    } else {
        return Err(Error::EmptyExpression { pos: close });
    };

    if names_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let names = &template[names_start..close];
    let mut specifiers = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut offset = names_start;

    for name in names.split(',') {
        if name.is_empty() {
            return Err(Error::EmptyVariable { pos: offset });
        }

        let name_bytes = name.as_bytes();
        let valid_first = |b: u8| b.is_ascii_alphabetic() || b == b'_';
        let valid_rest = |b: u8| b.is_ascii_alphanumeric() || b == b'_';

        if !valid_first(name_bytes[0]) {
            return Err(Error::InvalidName { pos: offset });
        }
        if let Some(index) = name_bytes.iter().enumerate().skip(1).find_map(|(i, &b)| {
            if valid_rest(b) { None } else { Some(i) }
        }) {
            return Err(Error::InvalidName { pos: offset + index });
        }

        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: offset });
        }
        seen.push(name);
        specifiers.push(Specifier { name, start: offset });
        offset += name.len() + 1;
    }

    Ok(Expression {
        operator,
        specifiers,
    })
}

fn parse_template(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let open = index;
                let Some(relative_close) = bytes[index + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = index + 1 + relative_close;
                expressions.push(parse_expression(template, open, close)?);
                index = close + 1;
            }
            _ => index += 1,
        }
    }

    Ok(expressions)
}

fn encode_value(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' && index + 2 < bytes.len()
            && bytes[index + 1].is_ascii_hexdigit()
            && bytes[index + 2].is_ascii_hexdigit()
        {
            result.push('%');
            result.push((bytes[index + 1] as char).to_ascii_uppercase());
            result.push((bytes[index + 2] as char).to_ascii_uppercase());
            index += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~');
        if unreserved || (reserved && is_reserved_operator(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    result
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|&(binding, value)| (binding == name).then_some(value))
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let mut fields = Vec::new();
    for specifier in &expression.specifiers {
        if let Some(value) = lookup(vars, specifier.name) {
            let encoded = encode_value(
                value,
                matches!(expression.operator, Operator::Reserved | Operator::Fragment),
            );
            let field = match expression.operator {
                Operator::Semicolon if encoded.is_empty() => specifier.name.to_owned(),
                Operator::Semicolon | Operator::Query | Operator::Continuation => {
                    format!("{}={}", specifier.name, encoded)
                }
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continuation => ("&", "&"),
    };

    let mut result = String::from(prefix);
    result.push_str(&fields.join(separator));
    result
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;
    let mut expr_index = 0;

    while cursor < bytes.len() {
        if bytes[cursor] == b'{' {
            let close = bytes[cursor + 1..]
                .iter()
                .position(|&b| b == b'}')
                .map(|relative| cursor + 1 + relative)
                .expect("template was validated");
            output.push_str(&template[if cursor == 0 { 0 } else { 0 }..0]);
            output.push_str(&expand_expression(&expressions[expr_index], vars));
            expr_index += 1;
            cursor = close + 1;
        } else {
            let start = cursor;
            while cursor < bytes.len() && bytes[cursor] != b'{' {
                cursor += 1;
            }
            output.push_str(&template[start..cursor]);
        }
    }

    Ok(output)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

struct ParsedExpression<'a> {
    open: usize,
    close: usize,
    operator: Operator,
    names: Vec<&'a str>,
}

fn supported_operator(byte: u8) -> Option<Operator> {
    match byte {
        b'+' => Some(Operator::Reserved),
        b'#' => Some(Operator::Fragment),
        b'.' => Some(Operator::Dot),
        b'/' => Some(Operator::Slash),
        b';' => Some(Operator::Semicolon),
        b'?' => Some(Operator::Query),
        b'&' => Some(Operator::Continuation),
        _ => None,
    }
}

fn parse_one(template: &str, open: usize, close: usize) -> Result<ParsedExpression<'_>, Error> {
    let start = open + 1;
    let bytes = template.as_bytes();

    if let Some(inner) = bytes[start..close].iter().position(|&b| b == b'{') {
        return Err(Error::NestedOpen { pos: start + inner });
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let operator = supported_operator(bytes[start]).unwrap_or(Operator::Simple);
    if matches!(bytes[start], b'=' | b'!' | b'@' | b'|' | b'%') {
        return Err(Error::UnsupportedOperator { pos: start });
    }
    let names_start = if matches!(operator, Operator::Simple) { start } else { start + 1 };
    if names_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut segment_start = names_start;

    for end in names_start..=close {
        if end == close || bytes[end] == b',' {
            if end == segment_start {
                return Err(Error::EmptyVariable { pos: segment_start });
            }

            let name = &template[segment_start..end];
            let name_bytes = name.as_bytes();
            for (offset, &b) in name_bytes.iter().enumerate() {
                let valid = if offset == 0 {
                    b.is_ascii_alphabetic() || b == b'_'
                } else {
                    b.is_ascii_alphanumeric() || b == b'_'
                };
                if !valid {
                    return Err(Error::InvalidName { pos: segment_start + offset });
                }
            }

            if !seen.insert(name) {
                return Err(Error::DuplicateVariable { pos: segment_start });
            }
            names.push(name);
            segment_start = end + 1;
        }
    }

    Ok(ParsedExpression { open, close, operator, names })
}

fn parse(template: &str) -> Result<Vec<ParsedExpression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'}' {
            return Err(Error::UnexpectedClose { pos: i });
        }
        if bytes[i] != b'{' {
            i += 1;
            continue;
        }

        let open = i;
        let Some(relative) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
            return Err(Error::Unclosed { pos: open });
        };
        let close = i + 1 + relative;
        expressions.push(parse_one(template, open, close)?);
        i = close + 1;
    }
    Ok(expressions)
}

fn reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else {
            let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
            if unreserved || (allow_reserved && reserved(b)) {
                out.push(b as char);
            } else {
                out.push('%');
                out.push(HEX[(b >> 4) as usize] as char);
                out.push(HEX[(b & 15) as usize] as char);
            }
            i += 1;
        }
    }
    out
}

fn get_value<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    for &(binding, value) in vars.iter().rev() {
        if binding == name {
            return Some(value);
        }
    }
    None
}

fn render(expression: &ParsedExpression<'_>, vars: &[(&str, &str)]) -> String {
    let mut fields = Vec::new();
    for name in &expression.names {
        let Some(value) = get_value(vars, name) else {
            continue;
        };

        let encoded = encode(
            value,
            matches!(expression.operator, Operator::Reserved | Operator::Fragment),
        );
        let field = match expression.operator {
            Operator::Semicolon if encoded.is_empty() => (*name).to_string(),
            Operator::Semicolon | Operator::Query | Operator::Continuation => {
                format!("{name}={encoded}")
            }
            _ => encoded,
        };
        fields.push(field);
    }

    if fields.is_empty() {
        return String::new();
    }
    let (prefix, separator) = match expression.operator {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continuation => ("&", "&"),
    };
    format!("{prefix}{}", fields.join(separator))
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parsed = parse(template)?;
    let bytes = template.as_bytes();
    let mut result = String::new();
    let mut cursor = 0;

    for expression in &parsed {
        result.push_str(&template[cursor..expression.open]);
        result.push_str(&render(expression, vars));
        cursor = expression.close + 1;
    }
    result.push_str(&template[cursor..]);
    Ok(result)
}
```
