    pub scheme: String,
    pub userinfo: Option<String>,

    pub query: Option<String>,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            if hi.is_none() || lo.is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hi.unwrap() << 4) | lo.unwrap());
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = origins.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn validate_component(component: Component, raw: &str, base: usize) -> Result<(), ParseError> {
    match decode_component(component, raw) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: base + at }),
    }
}

fn parse_port(s: &str, at: usize) -> Result<u16, ParseError> {
    if s.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    if let Some(i) = s.as_bytes().iter().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: at + i });
    }
    let bytes = s.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }
    let mut value: u32 = 0;
    for &b in bytes {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) if v <= 65535 => v,
            _ => return Err(ParseError::InvalidPort { at }),
        };
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(value as u16)
}

fn parse_authority(authority: &str, authority_at: usize) -> Result<(Option<String>, String, Option<u16>, bool), ParseError> {
    let ats: Vec<usize> = authority.match_indices('@').map(|(i, _)| i).collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo { at: authority_at + ats[1] });
    }
    let (userinfo, hostport, hostport_at) = if let Some(&pos) = ats.first() {
        if pos == 0 {
            return Err(ParseError::InvalidUserInfo { at: authority_at });
        }
        (
            Some(authority[..pos].to_owned()),
            &authority[pos + 1..],
            authority_at + pos + 1,
        )
    } else {
        (None, authority, authority_at)
    };

    if hostport.starts_with('[') {
        let open_at = hostport_at;
        let Some(close) = hostport.find(']') else {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        }
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_at + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_at + close + 1,
            });
        };
        return Ok((userinfo, interior.to_ascii_lowercase(), port, true));
    }

    let colon = hostport.find(':');
    let (host, port, host_len) = if let Some(pos) = colon {
        (&hostport[..pos], Some(&hostport[pos + 1..]), pos)
    } else {
        (hostport, None, hostport.len())
    };
    if host.is_empty() {
        return Err(ParseError::InvalidHost { at: hostport_at });
    }
    for (i, b) in host.bytes().enumerate() {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost { at: hostport_at + i });
        }
    }
    let parsed_port = if let Some(port) = port {
        Some(parse_port(port, hostport_at + host_len + 1)?)
    } else {
        None
    };
    Ok((userinfo, host.to_ascii_lowercase(), parsed_port, false))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let bytes = input.as_bytes();
    let Some(colon) = bytes.iter().position(|&b| b == b':') else {
        return Err(ParseError::InvalidScheme { at: input.len() });
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let tail = &input[auth_start..];
    let auth_len = tail
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(tail.len());
    let authority = &tail[..auth_len];
    let auth_end = auth_start + auth_len;
    let (userinfo, host, port, ipv6) = parse_authority(authority, auth_start)?;

    let rem = &input[auth_end..];
    let fragment_pos = rem.find('#');
    let before_fragment = fragment_pos.map(|i| &rem[..i]).unwrap_or(rem);
    let fragment = fragment_pos.map(|i| rem[i + 1..].to_owned());
    let query_pos = before_fragment.find('?');
    let raw_path = query_pos.map(|i| &before_fragment[..i]).unwrap_or(before_fragment);
    let query = query_pos.map(|i| before_fragment[i + 1..].to_owned());
    let path = if raw_path.is_empty() { "/".to_owned() } else { raw_path.to_owned() };

    if let Some(ref u) = userinfo {
        let at = auth_start;
        validate_component(Component::UserInfo, u, at)?;
    }
    if !raw_path.is_empty() {
        validate_component(Component::Path, raw_path, auth_end)?;
    }
    if let (Some(q), Some(qi)) = (&query, query_pos) {
        validate_component(Component::Query, q, auth_end + qi + 1)?;
    }
    if let (Some(f), Some(fi)) = (&fragment, fragment_pos) {
        validate_component(Component::Fragment, f, auth_end + fi + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded = decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;
    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded[1..].split('/').collect();
    let terminal_navigation = decoded_segments
        .last()
        .is_some_and(|s| *s == "." || *s == "..");
    let mut stack: Vec<&str> = Vec::new();
    for (raw, dec) in raw_segments.iter().zip(decoded_segments.iter()) {
        if *dec == "." {
            continue;
        } else if *dec == ".." {
            stack.pop();
        } else {
            stack.push(raw);
        }
    }
    if terminal_navigation {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn scan_reference(input: &str) -> Result<(), ParseError> {
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn append_optional(out: &mut String, prefix: char, value: &Option<String>) {
    if let Some(value) = value {
        out.push(prefix);
        out.push_str(value);
    }
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    scan_reference(reference).map_err(ResolveError::Reference)?;

    let first_delimiter = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let fragment_pos = reference.find('#');
    let before_fragment = fragment_pos.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = fragment_pos.map(|i| before_fragment_or_tail(reference, i + 1).to_owned());
    let query_pos = before_fragment.find('?');
    let ref_path = query_pos.map(|i| &before_fragment[..i]).unwrap_or(before_fragment);
    let ref_query = query_pos.map(|i| before_fragment[i + 1..].to_owned());

    let (selected_path, selected_query) = if ref_path.is_empty() {
        (base_path.clone(), if query_pos.is_some() { ref_query } else { base_url.query.clone() })
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_path[..slash], ref_path)
        };
        (merged, ref_query)
    };

    let mut absolute = format!("{}://", base_url.scheme);
    if let Some(userinfo) = &base_url.userinfo {
        absolute.push_str(userinfo);
        absolute.push('@');
    }
    if base_url.ipv6 {
        absolute.push('[');
        absolute.push_str(&base_url.host);
        absolute.push(']');
    } else {
        absolute.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        absolute.push(':');
        absolute.push_str(&port.to_string());
    }
    absolute.push_str(&selected_path);
    append_optional(&mut absolute, '?', &selected_query);
    append_optional(&mut absolute, '#', &fragment);

    let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

fn before_fragment_or_tail(input: &str, offset: usize) -> &str {
    &input[offset..]
}

pub struct QueryMap {

#[derive(Debug, Clone, PartialEq, Eq)]

    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },

pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),

fn form_decode(s: &str) -> String {
    s.replace('+', " ")
}

fn query_decode(s: &str, base: usize) -> Result<String, QueryError> {
    decode_component(Component::Query, &form_decode(s)).map_err(|e| match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    })
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut start = 0;
    for field in input.split('&') {
        let field_start = start;
        start += field.len() + 1;
        if field.is_empty() {
            continue;
        }
        let eq = field.find('=');
        let (raw_key, raw_value, value_offset) = if let Some(i) = eq {
            (&field[..i], &field[i + 1..], i + 1)
        } else {
            (field, "", field.len())
        };
        let key = query_decode(raw_key, field_start)?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }
        let value = query_decode(raw_value, field_start + value_offset)?;
        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }
    Ok(QueryMap { entries })
}

fn form_encode(s: &str) -> String {
    let mut out = String::with_capacity(s.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else if b == b' ' {
            out.push('+');
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(&str, Vec<&str>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, accumulated)) = merged.iter_mut().find(|(k, _)| *k == key) {
            accumulated.extend(values.iter().map(String::as_str));
        } else {
            merged.push((key.as_str(), values.iter().map(String::as_str).collect()));
        }
    }
    let mut fields = Vec::new();
    for (key, values) in merged {
        let key = form_encode(key);
        for value in values {
            fields.push(format!("{}={}", key, form_encode(value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = if let Some(query) = &url.query {
        Some(parse_query(query).map_err(CanonicalError::Query)?)
    } else {
        None
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, userinfo).unwrap();
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host).unwrap();
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        if !((url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443)) {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let segments: Vec<String> = path
        .split('/')
        .map(|segment| {
            decode_component(Component::Path, segment)
                .map(|decoded| encode_unreserved(&decoded))
                .unwrap()
        })
        .collect();
    out.push_str(&segments.join("/"));

    if let Some(map) = &query_map {
        out.push('?');
        out.push_str(&serialise_query(map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment).unwrap();
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn encode_unreserved(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn from_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let source = input.as_bytes();
    let mut decoded = Vec::with_capacity(source.len());
    let mut source_offsets = Vec::with_capacity(source.len());
    let mut i = 0;
    while i < source.len() {
        if source[i] == b'%' {
            if i + 2 >= source.len()
                || from_hex(source[i + 1]).is_none()
                || from_hex(source[i + 2]).is_none()
            {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((from_hex(source[i + 1]).unwrap() << 4) | from_hex(source[i + 2]).unwrap());
            source_offsets.push(i);
            i += 3;
        } else {
            decoded.push(source[i]);
            source_offsets.push(i);
            i += 1;
        }
    }
    let text = match std::str::from_utf8(&decoded) {
        Ok(text) => text,
        Err(error) => {
            return Err(CodecError::InvalidUtf8 {
                at: source_offsets[error.valid_up_to()],
            });
        }
    };
    Ok(text.to_owned())
}

pub fn encode_component(component: Component, text: &str) -> String {
    let extra = match component {
        Component::UserInfo => b":",
        Component::Path => b"/",
        Component::Query => b"/?",
        Component::Fragment => b"/?#",
    };
    let mut result = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || extra.contains(&byte)
        {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 15) as usize] as char);
        }
    }
    result
}

fn validate_raw(component: Component, raw: &str, offset: usize) -> Result<(), ParseError> {
    decode_component(component, raw).map(|_| ()).map_err(|error| match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    })
}

fn parse_port(text: &str, offset: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: offset });
    }
    if let Some(pos) = text.bytes().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: offset + pos });
    }
    if text.len() > 1 && text.starts_with('0') {
        return Err(ParseError::InvalidPort { at: offset });
    }
    let mut n = 0u32;
    for b in text.bytes() {
        n = n
            .checked_mul(10)
            .and_then(|n| n.checked_add((b - b'0') as u32))
            .filter(|n| *n <= 65535)
            .ok_or(ParseError::InvalidPort { at: offset })?;
    }
    if n == 0 {
        return Err(ParseError::InvalidPort { at: offset });
    }
    Ok(n as u16)
}

fn authority_parts(authority: &str, offset: usize) -> Result<(Option<String>, String, Option<u16>, bool), ParseError> {
    let mut at_pos = None;
    for (i, b) in authority.bytes().enumerate() {
        if b == b'@' {
            if at_pos.is_some() {
                return Err(ParseError::InvalidUserInfo { at: offset + i });
            }
            at_pos = Some(i);
        }
    }
    let (userinfo, hp, hp_offset) = match at_pos {
        Some(i) if i == 0 => return Err(ParseError::InvalidUserInfo { at: offset }),
        Some(i) => (Some(authority[..i].to_owned()), &authority[i + 1..], offset + i + 1),
        None => (None, authority, offset),
    };

    if hp.starts_with('[') {
        let close = hp.find(']').ok_or(ParseError::InvalidIpv6 { at: hp_offset })?;
        let host = &hp[1..close];
        Ipv6Addr::from_str(host).map_err(|_| ParseError::InvalidIpv6 { at: hp_offset })?;
        let suffix = &hp[close + 1..];
        let port = match suffix.as_bytes().first() {
            None => None,
            Some(b':') => Some(parse_port(&suffix[1..], hp_offset + close + 2)?),
            Some(_) => return Err(ParseError::InvalidHost { at: hp_offset + close + 1 }),
        };
        return Ok((userinfo, host.to_ascii_lowercase(), port, true));
    }

    let split = hp.find(':');
    let (host, port_text, host_size) = if let Some(i) = split {
        (&hp[..i], Some(&hp[i + 1..]), i)
    } else {
        (hp, None, hp.len())
    };
    if host.is_empty() {
        return Err(ParseError::InvalidHost { at: hp_offset });
    }
    for (i, byte) in host.bytes().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost { at: hp_offset + i });
        }
    }
    let port = port_text
        .map(|p| parse_port(p, hp_offset + host_size + 1))
        .transpose()?;
    Ok((userinfo, host.to_ascii_lowercase(), port, false))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Some((at, _)) = input.bytes().enumerate().find(|(_, b)| !(0x21..=0x7e).contains(b)) {
        return Err(ParseError::InvalidByte { at });
    }
    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        if !(bytes[i].is_ascii_alphanumeric() || matches!(bytes[i], b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }
    let authority_start = colon + 3;
    let tail = &input[authority_start..];
    let authority_len = tail.bytes().position(|b| matches!(b, b'/' | b'?' | b'#')).unwrap_or(tail.len());
    let authority = &tail[..authority_len];
    let authority_end = authority_start + authority_len;
    let (userinfo, host, port, ipv6) = authority_parts(authority, authority_start)?;

    let remainder = &input[authority_end..];
    let hash = remainder.find('#');
    let pre_fragment = hash.map(|i| &remainder[..i]).unwrap_or(remainder);
    let fragment = hash.map(|i| remainder[i + 1..].to_owned());
    let question = pre_fragment.find('?');
    let raw_path = question.map(|i| &pre_fragment[..i]).unwrap_or(pre_fragment);
    let query = question.map(|i| pre_fragment[i + 1..].to_owned());
    let path = if raw_path.is_empty() { "/".to_owned() } else { raw_path.to_owned() };

    if let Some(value) = &userinfo {
        validate_raw(Component::UserInfo, value, authority_start)?;
    }
    if !raw_path.is_empty() {
        validate_raw(Component::Path, raw_path, authority_end)?;
    }
    if let (Some(value), Some(i)) = (&query, question) {
        validate_raw(Component::Query, value, authority_end + i + 1)?;
    }
    if let (Some(value), Some(i)) = (&fragment, hash) {
        validate_raw(Component::Fragment, value, authority_end + i + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded = decode_component(Component::Path, path).map_err(|error| match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;
    let raw = path[1..].split('/').collect::<Vec<_>>();
    let decoded_parts = decoded[1..].split('/').collect::<Vec<_>>();
    let needs_trailing_slash = matches!(decoded_parts.last(), Some(&".") | Some(&".."));
    let mut stack: Vec<&str> = Vec::new();
    for (raw_segment, decoded_segment) in raw.into_iter().zip(decoded_parts) {
        match decoded_segment {
            "." => {}
            ".." => {
                stack.pop();
            }
            _ => stack.push(raw_segment),
        }
    }
    if needs_trailing_slash {
        stack.push("");
    }
    Ok(if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    })
}

fn check_reference_bytes(reference: &str) -> Result<(), ParseError> {
    for (at, byte) in reference.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    check_reference_bytes(reference).map_err(ResolveError::Reference)?;

    let structural_end = reference.bytes().position(|b| matches!(b, b'/' | b'?' | b'#')).unwrap_or(reference.len());
    if reference[..structural_end].contains(':') {
        let mut result = parse_url(reference).map_err(ResolveError::Reference)?;
        result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
        return Ok(result);
    }
    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut result = parse_url(&constructed).map_err(ResolveError::Reference)?;
        result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
        return Ok(result);
    }

    let hash = reference.find('#');
    let without_fragment = hash.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = hash.map(|i| reference[i + 1..].to_owned());
    let question = without_fragment.find('?');
    let reference_path = question.map(|i| &without_fragment[..i]).unwrap_or(without_fragment);
    let reference_query = question.map(|i| without_fragment[i + 1..].to_owned());

    let (path, query) = if reference_path.is_empty() {
        (base_path.clone(), if question.is_some() { reference_query } else { base_url.query.clone() })
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let directory_end = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_path[..directory_end], reference_path)
        };
        (merged, reference_query)
    };

    let mut constructed = format!("{}://", base_url.scheme);
    if let Some(userinfo) = &base_url.userinfo {
        constructed.push_str(userinfo);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&path);
    if let Some(query) = query {
        constructed.push('?');
        constructed.push_str(&query);
    }
    if let Some(fragment) = fragment {
        constructed.push('#');
        constructed.push_str(&fragment);
    }
    let mut result = parse_url(&constructed).map_err(ResolveError::Reference)?;
    result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
    Ok(result)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn decode_form(s: &str, base: usize) -> Result<String, QueryError> {
    let replaced = s.replace('+', " ");
    decode_component(Component::Query, &replaced).map_err(|e| match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    })
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut output: Vec<(String, Vec<String>)> = Vec::new();
    let mut offset = 0;
    for field in input.split('&') {
        let field_offset = offset;
        offset += field.len() + 1;
        if field.is_empty() {
            continue;
        }
        let equals = field.find('=');
        let (key_raw, value_raw, value_offset) = match equals {
            Some(i) => (&field[..i], &field[i + 1..], i + 1),
            None => (field, "", field.len()),
        };
        let key = decode_form(key_raw, field_offset)?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_offset });
        }
        let value = decode_form(value_raw, field_offset + value_offset)?;
        if let Some((_, values)) = output.iter_mut().find(|(existing, _)| existing == &key) {
            values.push(value);
        } else {
            output.push((key, vec![value]));
        }
    }
    Ok(QueryMap { entries: output })
}

fn form_encode(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut output = String::new();
    for &byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else if byte == b' ' {
            output.push('+');
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 15) as usize] as char);
        }
    }
    output
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut keys: Vec<(&str, Vec<&str>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        match keys.iter_mut().find(|(seen, _)| *seen == key) {
            Some((_, accumulated)) => accumulated.extend(values.iter().map(String::as_str)),
            None => keys.push((key, values.iter().map(String::as_str).collect())),
        }
    }
    let mut fields = Vec::new();
    for (key, values) in keys {
        for value in values {
            fields.push(format!("{}={}", form_encode(key), form_encode(value)));
        }
    }
    fields.join("&")
}

fn encode_unreserved(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut output = String::new();
    for &byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 15) as usize] as char);
        }
    }
    output
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query = match &url.query {
        Some(raw) => Some(parse_query(raw).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut output = format!("{}://", url.scheme);
    if let Some(raw) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, raw).unwrap();
        output.push_str(&encode_component(Component::UserInfo, &decoded));
        output.push('@');
    }
    if url.ipv6 {
        let address = Ipv6Addr::from_str(&url.host).unwrap();
        output.push('[');
        output.push_str(&address.to_string());
        output.push(']');
    } else {
        output.push_str(&url.host);
    }
    if let Some(port) = url.port {
        if !((url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443)) {
            output.push(':');
            output.push_str(&port.to_string());
        }
    }

    let encoded_path = path
        .split('/')
        .map(|segment| {
            let decoded = decode_component(Component::Path, segment).unwrap();
            encode_unreserved(&decoded)
        })
        .collect::<Vec<_>>()
        .join("/");
    output.push_str(&encoded_path);

    if let Some(query) = query {
        output.push('?');
        output.push_str(&serialise_query(&query));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment).unwrap();
        output.push('#');
        output.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(output)
}
