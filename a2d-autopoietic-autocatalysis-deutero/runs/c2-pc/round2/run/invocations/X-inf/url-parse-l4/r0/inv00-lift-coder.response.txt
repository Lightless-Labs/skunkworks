use std::str::FromStr;

pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            if hi.is_none() || lo.is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hi.unwrap() << 4) | lo.unwrap());
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::new();
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const H: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(H[(b >> 4) as usize] as char);
            out.push(H[(b & 15) as usize] as char);
        }
    }
    out
}

fn map_codec_parse(e: CodecError, offset: usize) -> ParseError {
    match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn validate_port(raw: &str, start: usize) -> Result<u16, ParseError> {
    if raw.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in raw.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    if raw.len() >= 2 && raw.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut n: u32 = 0;
    for b in raw.bytes() {
        n = n * 10 + (b - b'0') as u32;
        if n > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if n == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(n as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let after = colon + 3;
    let authority_end = bytes[after..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|p| after + p)
        .unwrap_or(bytes.len());
    let authority = &input[after..authority_end];
    let ats: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| if b == b'@' { Some(i) } else { None })
        .collect();
    if ats.len() > 1 {
        return Err(ParseError::InvalidUserInfo { at: after + ats[1] });
    }
    let (userinfo, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: after });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };
    let hostport = &authority[hostport_start..];
    let host_start = after + hostport_start;

    let (host, port, ipv6) = if hostport.starts_with('[') {
        let open = host_start;
        let close = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &hostport[1..close];
        if interior.contains('%') || Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(validate_port(&suffix[1..], host_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport.find(':');
        let (raw_host, port) = if let Some(i) = split {
            (
                &hostport[..i],
                Some(validate_port(&hostport[i + 1..], host_start + i + 1)?),
            )
        } else {
            (hostport, None)
        };
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, b) in raw_host.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + i,
                });
            }
        }
        (raw_host.to_ascii_lowercase(), port, false)
    };

    let mut fragment = None;
    let before_fragment_end = if let Some(rel) = input[authority_end..].find('#') {
        let start = authority_end + rel;
        fragment = Some(input[start + 1..].to_owned());
        start
    } else {
        input.len()
    };
    let mut query = None;
    let path_end;
    if let Some(rel) = input[authority_end..before_fragment_end].find('?') {
        let start = authority_end + rel;
        query = Some(input[start + 1..before_fragment_end].to_owned());
        path_end = start;
    } else {
        path_end = before_fragment_end;
    }
    let path = if authority_end == path_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    if let Some(ui) = &userinfo {
        decode_component(Component::UserInfo, ui)
            .map_err(|e| map_codec_parse(e, after))?;
    }
    if path != "/" || authority_end != path_end {
        decode_component(Component::Path, &path)
            .map_err(|e| map_codec_parse(e, authority_end))?;
    }
    if let Some(q) = &query {
        let q_offset = path_end + 1;
        decode_component(Component::Query, q)
            .map_err(|e| map_codec_parse(e, q_offset))?;
    }
    if let Some(f) = &fragment {
        let f_offset = before_fragment_end + 1;
        decode_component(Component::Fragment, f)
            .map_err(|e| map_codec_parse(e, f_offset))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn map_codec_path(e: CodecError) -> PathError {
    match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let last_is_navigation = segments.last().map(|s| {
        decode_component(Component::Path, s)
            .map(|d| d == "." || d == "..")
            .unwrap_or(false)
    }).unwrap_or(false);

    let mut stack: Vec<&str> = Vec::new();
    for seg in segments {
        let decoded = decode_component(Component::Path, seg).map_err(map_codec_path)?;
        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(seg);
        }
    }
    if last_is_navigation {
        stack.push("");
    }
    Ok(format!("/{}", stack.join("/")))
}

fn scan_reference(input: &str) -> Result<(), ParseError> {
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn append_query_fragment(out: &mut String, query: Option<&str>, fragment: Option<&str>) {
    if let Some(q) = query {
        out.push('?');
        out.push_str(q);
    }
    if let Some(f) = fragment {
        out.push('#');
        out.push_str(f);
    }
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    scan_reference(reference).map_err(ResolveError::Reference)?;

    if reference.is_empty() {
        let mut out = base_url.clone();
        out.path = base_path;
        return Ok(out);
    }

    let first_structural = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_structural].contains(':') {
        let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&absolute).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    let fragment_pos = reference.find('#');
    let fragment = fragment_pos.map(|i| &reference[i + 1..]);
    let before_fragment = fragment_pos.map(|i| &reference[..i]).unwrap_or(reference);
    let query_pos = before_fragment.find('?');
    let reference_query = query_pos.map(|i| &before_fragment[i + 1..]);
    let reference_path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_path.clone(),
            if query_pos.is_none() {
                base_url.query.as_deref()
            } else {
                reference_query
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let cut = base_path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_path[..=cut], reference_path)
        };
        (merged, reference_query)
    };

    let mut constructed = format!("{}://", base_url.scheme);
    if let Some(ui) = &base_url.userinfo {
        constructed.push_str(ui);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(p) = base_url.port {
        constructed.push(':');
        constructed.push_str(&p.to_string());
    }
    constructed.push_str(&selected_path);
    append_query_fragment(&mut constructed, selected_query, fragment);

    let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
    u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
    Ok(u)
}

pub struct QueryMap {

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn map_codec_query(e: CodecError, offset: usize) -> QueryError {
    match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

fn plus_to_space(s: &str) -> String {
    s.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0;
    for field in input.split('&') {
        let field_len = field.len();
        if !field.is_empty() {
            let eq = field.find('=');
            let (raw_key, raw_value, value_offset) = match eq {
                Some(i) => (&field[..i], &field[i + 1..], i + 1),
                None => (field, "", field.len()),
            };
            let key_input = plus_to_space(raw_key);
            let key = decode_component(Component::Query, &key_input)
                .map_err(|e| map_codec_query(e, field_start))?;
            if key.is_empty() {
                return Err(QueryError::EmptyKey { at: field_start });
            }
            let value_input = plus_to_space(raw_value);
            let value = decode_component(Component::Query, &value_input)
                .map_err(|e| map_codec_query(e, field_start + value_offset))?;

            if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
                values.push(value);
            } else {
                entries.push((key, vec![value]));
            }
        }
        field_start += field_len + 1;
    }
    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut out = String::new();
    const H: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(H[(b >> 4) as usize] as char);
            out.push(H[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(&str, Vec<&str>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| *k == key) {
            existing.extend(values.iter().map(String::as_str));
        } else {
            merged.push((key, values.iter().map(String::as_str).collect()));
        }
    }
    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", form_encode(key), form_encode(value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let mut url = parse_url(input).map_err(CanonicalError::Parse)?;
    url.path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(raw) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, raw)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, 0)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host)
            .expect("parsed IPv6 address remains valid");
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        let default = (url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443);
        if !default {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut canonical_segments = Vec::new();
    for segment in url.path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, 0)))?;
        canonical_segments.push(encode_unreserved(&decoded));
    }
    out.push_str(&canonical_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(raw) = &url.fragment {
        let decoded = decode_component(Component::Fragment, raw)
            .map_err(|e| CanonicalError::Parse(map_codec_parse(e, 0)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn encode_unreserved(text: &str) -> String {
    let mut out = String::new();
    const H: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(H[(b >> 4) as usize] as char);
            out.push(H[(b & 15) as usize] as char);
        }
    }
    out
}

    InvalidScheme { at: usize },

    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let src = input.as_bytes();
    let mut out = Vec::with_capacity(src.len());
    let mut offsets = Vec::with_capacity(src.len());
    let mut i = 0;
    while i < src.len() {
        if src[i] != b'%' {
            out.push(src[i]);
            offsets.push(i);
            i += 1;
        } else {
            if src.len() - i < 3 {
                return Err(CodecError::BadEscape { at: i });
            }
            let h = hex_value(src[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let l = hex_value(src[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            out.push(h * 16 + l);
            offsets.push(i);
            i += 3;
        }
    }
    let decoded = std::str::from_utf8(&out).map_err(|e| CodecError::InvalidUtf8 {
        at: offsets[e.valid_up_to()],
    })?;
    Ok(decoded.to_owned())
}

fn hex_value(b: u8) -> Option<u8> {
    if b.is_ascii_digit() {
        Some(b - b'0')
    } else if (b'a'..=b'f').contains(&b) {
        Some(b - b'a' + 10)
    } else if (b'A'..=b'F').contains(&b) {
        Some(b - b'A' + 10)
    } else {
        None
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut result = String::new();
    for &b in text.as_bytes() {
        let preserve = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => b == b'/' || b == b'?',
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if preserve {
            result.push(b as char);
        } else {
            result.push('%');
            result.push(b"0123456789ABCDEF"[(b >> 4) as usize] as char);
            result.push(b"0123456789ABCDEF"[(b & 15) as usize] as char);
        }
    }
    result
}

fn parse_port(raw: &str, offset: usize) -> Result<u16, ParseError> {
    if raw.is_empty() {
        return Err(ParseError::InvalidPort { at: offset });
    }
    for (i, b) in raw.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: offset + i });
        }
    }
    if raw.len() > 1 && raw.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: offset });
    }
    let mut value: u32 = 0;
    for b in raw.bytes() {
        value = value * 10 + (b - b'0') as u32;
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: offset });
        }
    }
    if value == 0 {
        Err(ParseError::InvalidPort { at: offset })
    } else {
        Ok(value as u16)
    }
}

fn codec_parse_error(e: CodecError, offset: usize) -> ParseError {
    match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn check_ascii_url(input: &str) -> Result<(), ParseError> {
    for (at, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    check_ascii_url(input)?;
    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(p) => p,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|b| matches!(*b, b'/' | b'?' | b'#'))
        .map(|p| auth_start + p)
        .unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];
    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + at_positions[1],
        });
    }
    let (userinfo, hp_index) = match at_positions.first() {
        Some(&i) if i == 0 => return Err(ParseError::InvalidUserInfo { at: auth_start }),
        Some(&i) => (Some(authority[..i].to_owned()), i + 1),
        None => (None, 0),
    };
    let hp = &authority[hp_index..];
    let hp_abs = auth_start + hp_index;

    let (host, port, ipv6) = if hp.starts_with('[') {
        let open = hp_abs;
        let Some(close) = hp.find(']') else {
            return Err(ParseError::InvalidIpv6 { at: open });
        };
        let literal = &hp[1..close];
        if literal.contains('%') || Ipv6Addr::from_str(literal).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        let tail = &hp[close + 1..];
        let port = if tail.is_empty() {
            None
        } else if let Some(p) = tail.strip_prefix(':') {
            Some(parse_port(p, hp_abs + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hp_abs + close + 1,
            });
        };
        (literal.to_ascii_lowercase(), port, true)
    } else {
        let colon_at = hp.find(':');
        let (raw_host, port) = if let Some(c) = colon_at {
            (&hp[..c], Some(parse_port(&hp[c + 1..], hp_abs + c + 1)?))
        } else {
            (hp, None)
        };
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: hp_abs });
        }
        for (i, b) in raw_host.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: hp_abs + i });
            }
        }
        (raw_host.to_ascii_lowercase(), port, false)
    };

    let fragment_at = input[auth_end..].find('#').map(|i| auth_end + i);
    let comp_end = fragment_at.unwrap_or(input.len());
    let fragment = fragment_at.map(|i| input[i + 1..].to_owned());
    let query_at = input[auth_end..comp_end].find('?').map(|i| auth_end + i);
    let path_end = query_at.unwrap_or(comp_end);
    let query = query_at.map(|i| input[i + 1..comp_end].to_owned());
    let path = if path_end == auth_end {
        "/".to_owned()
    } else {
        input[auth_end..path_end].to_owned()
    };

    if let Some(v) = &userinfo {
        decode_component(Component::UserInfo, v)
            .map_err(|e| codec_parse_error(e, auth_start))?;
    }
    if path_end != auth_end {
        decode_component(Component::Path, &path)
            .map_err(|e| codec_parse_error(e, auth_end))?;
    }
    if let Some(v) = &query {
        decode_component(Component::Query, v)
            .map_err(|e| codec_parse_error(e, query_at.unwrap() + 1))?;
    }
    if let Some(v) = &fragment {
        decode_component(Component::Fragment, v)
            .map_err(|e| codec_parse_error(e, fragment_at.unwrap() + 1))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn path_codec(e: CodecError) -> PathError {
    match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(path_codec)?;
    let parts: Vec<&str> = path[1..].split('/').collect();
    let trailing_navigation = parts.last().is_some_and(|s| {
        matches!(decode_component(Component::Path, s), Ok(ref d) if d == "." || d == "..")
    });
    let mut stack: Vec<&str> = Vec::new();
    for part in parts {
        match decode_component(Component::Path, part).map_err(path_codec)?.as_str() {
            "." => {}
            ".." => {
                stack.pop();
            }
            _ => stack.push(part),
        }
    }
    if trailing_navigation {
        stack.push("");
    }
    Ok(format!("/{}", stack.join("/")))
}

fn check_reference_bytes(reference: &str) -> Result<(), ParseError> {
    for (i, b) in reference.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    check_reference_bytes(reference).map_err(ResolveError::Reference)?;

    let first_delimiter = reference
        .find(|c: char| matches!(c, '/' | '?' | '#'))
        .unwrap_or(reference.len());
    if reference[..first_delimiter].contains(':') {
        let mut result = parse_url(reference).map_err(ResolveError::Reference)?;
        result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
        return Ok(result);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut result = parse_url(&absolute).map_err(ResolveError::Reference)?;
        result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
        return Ok(result);
    }

    let fragment_idx = reference.find('#');
    let frag = fragment_idx.map(|i| &reference[i + 1..]);
    let no_frag = fragment_idx.map(|i| &reference[..i]).unwrap_or(reference);
    let query_idx = no_frag.find('?');
    let ref_query = query_idx.map(|i| &no_frag[i + 1..]);
    let ref_path = query_idx.map(|i| &no_frag[..i]).unwrap_or(no_frag);

    let (path, query) = if ref_path.is_empty() {
        (base_path.clone(), if query_idx.is_none() { base_url.query.as_deref() } else { ref_query })
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_path[..=slash], ref_path)
        };
        (merged, ref_query)
    };

    let mut raw = format!("{}://", base_url.scheme);
    if let Some(ui) = &base_url.userinfo {
        raw.push_str(ui);
        raw.push('@');
    }
    if base_url.ipv6 {
        raw.push('[');
        raw.push_str(&base_url.host);
        raw.push(']');
    } else {
        raw.push_str(&base_url.host);
    }
    if let Some(p) = base_url.port {
        raw.push(':');
        raw.push_str(&p.to_string());
    }
    raw.push_str(&path);
    if let Some(q) = query {
        raw.push('?');
        raw.push_str(q);
    }
    if let Some(f) = frag {
        raw.push('#');
        raw.push_str(f);
    }
    let mut result = parse_url(&raw).map_err(ResolveError::Reference)?;
    result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
    Ok(result)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn query_codec(e: CodecError, offset: usize) -> QueryError {
    match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

fn form_decode(raw: &str) -> String {
    raw.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut result = Vec::<(String, Vec<String>)>::new();
    let mut cursor = 0usize;
    for field in input.split('&') {
        if !field.is_empty() {
            let eq = field.find('=').unwrap_or(field.len());
            let key_raw = &field[..eq];
            let value_raw = if eq < field.len() { &field[eq + 1..] } else { "" };
            let key = decode_component(Component::Query, &form_decode(key_raw))
                .map_err(|e| query_codec(e, cursor))?;
            if key.is_empty() {
                return Err(QueryError::EmptyKey { at: cursor });
            }
            let value_offset = cursor + if eq < field.len() { eq + 1 } else { field.len() };
            let value = decode_component(Component::Query, &form_decode(value_raw))
                .map_err(|e| query_codec(e, value_offset))?;
            if let Some((_, values)) = result.iter_mut().find(|(existing, _)| *existing == key) {
                values.push(value);
            } else {
                result.push((key, vec![value]));
            }
        }
        cursor += field.len() + 1;
    }
    Ok(QueryMap { entries: result })
}

fn form_encode(text: &str) -> String {
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(b"0123456789ABCDEF"[(b >> 4) as usize] as char);
            out.push(b"0123456789ABCDEF"[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut groups: Vec<(&str, Vec<&str>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        let index = groups.iter().position(|(k, _)| *k == key);
        if let Some(i) = index {
            groups[i].1.extend(values.iter().map(|v| v.as_str()));
        } else {
            groups.push((key, values.iter().map(|v| v.as_str()).collect()));
        }
    }
    let mut fields = Vec::new();
    for (key, values) in groups {
        for value in values {
            fields.push(format!("{}={}", form_encode(key), form_encode(value)));
        }
    }
    fields.join("&")
}

fn encode_unreserved(text: &str) -> String {
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(b"0123456789ABCDEF"[(b >> 4) as usize] as char);
            out.push(b"0123456789ABCDEF"[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let mut url = parse_url(input).map_err(CanonicalError::Parse)?;
    url.path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query = match url.query.as_deref() {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, userinfo)
            .expect("parse_url already validated userinfo");
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if url.ipv6 {
        let address = Ipv6Addr::from_str(&url.host).expect("parse_url validated IPv6");
        out.push('[');
        out.push_str(&address.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        if !((url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443)) {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }
    let mut pieces = Vec::new();
    for segment in url.path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .expect("parse_url and normalise_path validated path");
        pieces.push(encode_unreserved(&decoded));
    }
    out.push_str(&pieces.join("/"));
    if let Some(map) = query {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .expect("parse_url validated fragment");
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}
