#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();
        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();
        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(text, 0, split, 60, 60, &mut minutes)?;
        parse_field(text, split + 1, text.len(), 24, 24, &mut hours)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    text: &str,
    start: usize,
    end: usize,
    max_value: usize,
    max_step: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut item_start = start;
    loop {
        let item_end = bytes[item_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(end);
        parse_item(text, item_start, item_end, max_value, max_step, selected)?;
        if item_end == end {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(())
}

fn parse_item(
    text: &str,
    start: usize,
    end: usize,
    max_value: usize,
    max_step: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let b = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }

    let mut slash = None;
    for i in start..end {
        if b[i] == b'/' {
            if slash.is_some() {
                return Err(CronError::Syntax(i));
            }
            slash = Some(i);
        }
    }

    let base_end = slash.unwrap_or(end);
    let step_start = slash.map(|i| i + 1);
    let base_is_star = base_end == start + 1 && b[start] == b'*';
    let dash = (start..base_end).find(|&i| b[i] == b'-');
    let is_range = dash.is_some();

    if slash.is_some() && !base_is_star && !is_range {
        return Err(CronError::Syntax(slash.unwrap()));
    }

    let (lo, hi) = if base_is_star {
        (0, max_value - 1)
    } else if let Some(d) = dash {
        let left = parse_number(text, start, d, max_value)?;
        let right = parse_number(text, d + 1, base_end, max_value)?;
        if left > right {
            return Err(CronError::Descending(d));
        }
        (left, right)
    } else {
        let n = parse_number(text, start, base_end, max_value)?;
        (n, n)
    };

    let step = if let Some(s) = step_start {
        parse_step(text, s, end, max_step)?
    } else {
        1
    };

    let mut n = lo;
    while n <= hi {
        selected[n] = true;
        match n.checked_add(step) {
            Some(next) => n = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_number(text: &str, start: usize, end: usize, limit: usize) -> Result<usize, CronError> {
    let b = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !b[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && b[start] == b'0' {
        return Err(CronError::Number(start));
    }
    let mut value = 0usize;
    for i in start..end {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b[i] - b'0') as usize)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(start)),
        };
    }
    if value >= limit {
        return Err(CronError::Bounds(start));
    }
    Ok(value)
}

fn parse_step(text: &str, start: usize, end: usize, max_step: usize) -> Result<usize, CronError> {
    let b = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !b[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && b[start] == b'0' {
        return Err(CronError::Number(start));
    }
    let mut value = 0usize;
    for i in start..end {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b[i] - b'0') as usize)) {
            Some(v) => v,
            None => return Err(CronError::Step(start)),
        };
    }
    if value == 0 || value > max_step {
        return Err(CronError::Step(start));
    }
    Ok(value)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let owned_id = id.to_owned();
        self.ids.push(owned_id.clone());
        self.jobs.push(Job {
            id: owned_id,
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower_exclusive = self.last_poll;
        let upper_minute = now / 60_000;
        let mut out: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let first = match lower_exclusive {
                Some(last) => last / 60_000 + 1,
                None => 0,
            };
            if first > upper_minute {
                continue;
            }
            let matching = |m: u64| job.cron.matches(m);
            match job.policy {
                Policy::Skip => {
                    if now % 60_000 == 0
                        && now / 60_000 >= first
                        && matching(now / 60_000)
                    {
                        out.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Once => {
                    if let Some(m) = (first..=upper_minute).rev().find(|&m| matching(m)) {
                        out.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: m * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::All => {
                    for m in first..=upper_minute {
                        if matching(m) {
                            out.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: m * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        out.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(out.into_iter().map(|entry| entry.0).collect())
    }
}

impl Default for Calendar {
    fn default() -> Self {
        Self::new()
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minute_mask: u64,
    hour_mask: u32,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let b = text.as_bytes();
        for (i, &ch) in b.iter().enumerate() {
            if ch.is_ascii_whitespace() && ch != b' ' {
                return Err(CronError::Layout(i));
            }
        }
        let mut split = None;
        for (i, &ch) in b.iter().enumerate() {
            if ch == b' ' {
                if split.is_some() {
                    return Err(CronError::Layout(i));
                }
                split = Some(i);
            }
        }
        let split = split.ok_or(CronError::Layout(text.len()))?;
        if split == 0 || split + 1 == b.len() {
            return Err(CronError::Layout(split));
        }

        let mut minute_mask = 0u64;
        let mut hour_mask = 0u32;
        parse_into(text, 0, split, 59, 60, &mut minute_mask)?;
        parse_into(text, split + 1, b.len(), 23, 24, &mut hour_mask)?;
        Ok(Self {
            minute_mask,
            hour_mask,
        })
    }

    pub fn matches(&self, minute: u64) -> bool {
        let m = minute % 60;
        let h = (minute / 60) % 24;
        (self.minute_mask & (1u64 << m)) != 0 && (self.hour_mask & (1u32 << h)) != 0
    }
}

fn parse_into(
    text: &str,
    start: usize,
    end: usize,
    max: usize,
    step_max: usize,
    mask: &mut u64,
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut pos = start;
    loop {
        let item_end = (pos..end).find(|&i| bytes[i] == b',').unwrap_or(end);
        if pos == item_end {
            return Err(CronError::Syntax(pos));
        }
        parse_one(text, pos, item_end, max, step_max, mask)?;
        if item_end == end {
            return Ok(());
        }
        pos = item_end + 1;
    }
}

fn parse_one(
    text: &str,
    start: usize,
    end: usize,
    max: usize,
    step_max: usize,
    mask: &mut u64,
) -> Result<(), CronError> {
    let b = text.as_bytes();
    let mut slash = None;
    for i in start..end {
        if b[i] == b'/' {
            if slash.is_some() {
                return Err(CronError::Syntax(i));
            }
            slash = Some(i);
        }
    }
    let base_end = slash.unwrap_or(end);
    let star = base_end == start + 1 && b[start] == b'*';
    let dash = (start..base_end).find(|&i| b[i] == b'-');

    if slash.is_some() && !star && dash.is_none() {
        return Err(CronError::Syntax(slash.unwrap()));
    }

    let (lo, hi) = if star {
        (0, max)
    } else if let Some(d) = dash {
        let lo = parse_fragment(text, start, d, max)?;
        let hi = parse_fragment(text, d + 1, base_end, max)?;
        if lo > hi {
            return Err(CronError::Descending(d));
        }
        (lo, hi)
    } else {
        let n = parse_fragment(text, start, base_end, max)?;
        (n, n)
    };

    let step = if let Some(slash_at) = slash {
        parse_step_fragment(text, slash_at + 1, end, step_max)?
    } else {
        1
    };
    let mut v = lo;
    while v <= hi {
        *mask |= 1u64 << v;
        v += step;
    }
    Ok(())
}

fn parse_fragment(text: &str, start: usize, end: usize, max: usize) -> Result<usize, CronError> {
    let b = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !b[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && b[start] == b'0' {
        return Err(CronError::Number(start));
    }
    let mut n = 0usize;
    for i in start..end {
        n = n.saturating_mul(10).saturating_add((b[i] - b'0') as usize);
        if n > max {
            return Err(CronError::Bounds(start));
        }
    }
    Ok(n)
}

fn parse_step_fragment(
    text: &str,
    start: usize,
    end: usize,
    max: usize,
) -> Result<usize, CronError> {
    let b = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !b[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && b[start] == b'0' {
        return Err(CronError::Number(start));
    }
    let mut n = 0usize;
    for i in start..end {
        n = n.saturating_mul(10).saturating_add((b[i] - b'0') as usize);
        if n > max {
            return Err(CronError::Step(start));
        }
    }
    if n == 0 {
        return Err(CronError::Step(start));
    }
    Ok(n)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Entry {
    id: String,
    mask_min: u64,
    mask_hour: u32,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    entries: Vec<Entry>,
    reserved: Vec<String>,
    watermark: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            reserved: Vec::new(),
            watermark: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if !(1..=16).contains(&id.len())
            || !id
                .bytes()
                .all(|x| x.is_ascii_lowercase() || x.is_ascii_digit() || x == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved.iter().any(|x| x == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        self.reserved.push(id.to_owned());
        self.entries.push(Entry {
            id: id.to_owned(),
            mask_min: cron.minute_mask,
            mask_hour: cron.hour_mask,
            priority,
            policy,
            order: self.entries.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if self.watermark.is_some_and(|last| now < last) {
            return Err(Error::TimeReversed);
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first = self.watermark.map_or(0, |last| last / 60_000 + 1);
        let last = now / 60_000;
        let mut results: Vec<(Occurrence, u8, usize)> = Vec::new();

        for entry in &self.entries {
            match entry.policy {
                Policy::Skip => {
                    if now % 60_000 == 0 && now / 60_000 >= first && entry_matches(entry, now / 60_000) {
                        results.push((Occurrence { id: entry.id.clone(), at: now }, entry.priority, entry.order));
                    }
                }
                Policy::Once => {
                    if first <= last {
                        for m in (first..=last).rev() {
                            if entry_matches(entry, m) {
                                results.push((Occurrence { id: entry.id.clone(), at: m * 60_000 }, entry.priority, entry.order));
                                break;
                            }
                        }
                    }
                }
                Policy::All => {
                    if first <= last {
                        for m in first..=last {
                            if entry_matches(entry, m) {
                                results.push((Occurrence { id: entry.id.clone(), at: m * 60_000 }, entry.priority, entry.order));
                            }
                        }
                    }
                }
            }
        }

        results.sort_by(|a, b| {
            a.0.at.cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.watermark = Some(now);
        Ok(results.into_iter().map(|x| x.0).collect())
    }
}

fn entry_matches(entry: &Entry, minute: u64) -> bool {
    let m = minute % 60;
    let h = (minute / 60) % 24;
    entry.mask_min & (1u64 << m) != 0 && entry.mask_hour & (1u32 << h) != 0
}

impl Default for Calendar {
    fn default() -> Self {
        Self::new()
    }
}
