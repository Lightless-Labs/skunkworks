#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(text, 0, split, 60, 60, &mut minutes)?;
        parse_field(text, split + 1, bytes.len(), 24, 24, &mut hours)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    text: &str,
    start: usize,
    end: usize,
    max: usize,
    step_max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut item_start = start;

    loop {
        let mut item_end = item_start;
        while item_end < end && bytes[item_end] != b',' {
            item_end += 1;
        }

        parse_item(text, item_start, item_end, max, step_max, selected)?;

        if item_end == end {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(())
}

fn parse_item(
    text: &str,
    start: usize,
    end: usize,
    max: usize,
    step_max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }

    let mut slash = None;
    for i in start..end {
        if bytes[i] == b'/' {
            if slash.is_some() {
                return Err(CronError::Syntax(i));
            }
            slash = Some(i);
        }
    }

    let base_end = slash.unwrap_or(end);
    let step_start = slash.map(|i| i + 1);
    let has_step = slash.is_some();

    if has_step {
        let first = bytes[start];
        let base_is_star = base_end == start + 1 && first == b'*';
        let base_is_range = bytes[start..base_end].contains(&b'-');
        if !base_is_star && !base_is_range {
            return Err(CronError::Syntax(slash.unwrap()));
        }
    }

    let mut lo = 0usize;
    let mut hi = max - 1;

    if bytes[start] == b'*' {
        if base_end != start + 1 {
            return Err(CronError::Syntax(start + 1));
        }
    } else {
        let dash = bytes[start..base_end]
            .iter()
            .position(|&b| b == b'-')
            .map(|p| start + p);

        if let Some(dash) = dash {
            lo = parse_number(bytes, start, dash, max)?;
            hi = parse_number(bytes, dash + 1, base_end, max)?;
            if lo > hi {
                return Err(CronError::Descending(dash));
            }
        } else {
            lo = parse_number(bytes, start, base_end, max)?;
            hi = lo;
        }
    }

    let step = if let Some(step_start) = step_start {
        parse_number(bytes, step_start, end, step_max + 1)?
    } else {
        1
    };

    if has_step && (step == 0 || step > step_max) {
        return Err(CronError::Step(step_start.unwrap()));
    }

    let mut value = lo;
    while value <= hi {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }

    let _ = text;
    Ok(())
}

fn parse_number(bytes: &[u8], start: usize, end: usize, max: usize) -> Result<usize, CronError> {
    if start == end {
        return Err(CronError::Syntax(start));
    }

    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }

    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for i in start..end {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((bytes[i] - b'0') as usize))
        {
            Some(v) => v,
            None => return Err(CronError::Bounds(start)),
        };
        if value >= max {
            return Err(CronError::Bounds(start));
        }
    }
    Ok(value)
}

    TimeRange,
    BadId,
    BadPriority,
    Duplicate,

}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll;
        let first_minute = match lower {
            None => 0,
            Some(t) => t / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let exact_now_boundary = now % 60_000 == 0;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        if first_minute <= last_minute {
            for job in &self.jobs {
                let mut matching = Vec::new();
                let mut minute = first_minute;
                while minute <= last_minute {
                    if job.cron.matches(minute) {
                        let at = minute * 60_000;
                        if lower.map_or(true, |t| at > t) && at <= now {
                            matching.push(at);
                        }
                    }
                    minute += 1;
                }

                let chosen: Vec<u64> = match job.policy {
                    Policy::All => matching,
                    Policy::Once => matching.last().copied().into_iter().collect(),
                    Policy::Skip => {
                        if exact_now_boundary && matching.last().copied() == Some(now) {
                            vec![now]
                        } else {
                            Vec::new()
                        }
                    }
                };

                for at in chosen {
                    emitted.push((
                        Occurrence {
                            id: job.id.clone(),
                            at,
                        },
                        job.priority,
                        job.order,
                    ));
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|(occurrence, _, _)| occurrence).collect())
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(&bytes[..split], 0, 60, &mut minutes)?;
        parse_field(&bytes[split + 1..], split + 1, 24, &mut hours)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &[u8],
    offset: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;
    loop {
        let end = field[start..]
            .iter()
            .position(|&b| b == b',')
            .map_or(field.len(), |p| start + p);

        if start == end {
            return Err(CronError::Syntax(offset + start));
        }
        parse_item(&field[start..end], offset + start, limit, selected)?;

        if end == field.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

fn number(fragment: &[u8], start: usize) -> Result<usize, CronError> {
    if fragment.is_empty() {
        return Err(CronError::Syntax(start));
    }

    for (i, &b) in fragment.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(CronError::Syntax(start + i));
        }
    }

    if fragment.len() > 1 && fragment[0] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for &b in fragment {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as usize))
            .ok_or(CronError::Bounds(start))?;
    }
    Ok(value)
}

fn parse_item(
    item: &[u8],
    offset: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let slash = item.iter().position(|&b| b == b'/');
    if let Some(s) = slash {
        if let Some(second) = item[s + 1..].iter().position(|&b| b == b'/') {
            return Err(CronError::Syntax(offset + s + 1 + second));
        }
    }

    let base_end = slash.unwrap_or(item.len());
    let base = &item[..base_end];
    let max = limit - 1;

    enum Base {
        Star,
        Range(usize, usize),
        Single(usize),
    }

    let parsed_base = if base == b"*" {
        Base::Star
    } else if let Some(dash) = base.iter().position(|&b| b == b'-') {
        let left = number(&base[..dash], offset)?;
        if left > max {
            return Err(CronError::Bounds(offset));
        }
        let right_start = offset + dash + 1;
        let right = number(&base[dash + 1..], right_start)?;
        if right > max {
            return Err(CronError::Bounds(right_start));
        }
        if left > right {
            return Err(CronError::Descending(offset + dash));
        }
        Base::Range(left, right)
    } else {
        let value = number(base, offset)?;
        if value > max {
            return Err(CronError::Bounds(offset));
        }
        Base::Single(value)
    };

    let step = if let Some(slash_at) = slash {
        if !matches!(parsed_base, Base::Star | Base::Range(_, _)) {
            return Err(CronError::Syntax(offset + slash_at));
        }
        let step_start = offset + slash_at + 1;
        let value = number(&item[slash_at + 1..], step_start)?;
        let step_max = if limit == 60 { 60 } else { 24 };
        if value == 0 || value > step_max {
            return Err(CronError::Step(step_start));
        }
        value
    } else {
        1
    };

    let (begin, end) = match parsed_base {
        Base::Star => (0, max),
        Base::Range(a, b) => (a, b),
        Base::Single(n) => (n, n),
    };

    let mut n = begin;
    loop {
        selected[n] = true;
        if end - n < step {
            break;
        }
        n += step;
    }
    Ok(())
}

            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_poll = self.last_poll.is_none();
        let previous = self.last_poll.unwrap_or(0);
        let upper_minute = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {

                None => 0,
                Some(t) => t / 60_000 + 1,

                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,

                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Once => {
                    let mut latest = None;
                    for minute in first..=upper_minute {
                        if job.cron.matches(minute) {
                            latest = Some(minute);
                        }
                    }
                    if let Some(minute) = latest {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0
                        && now / 60_000 >= first
                        && job.cron.matches(now / 60_000)
                    {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|item| item.0).collect())
    }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minute_mask: u64,
    hour_mask: u32,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();
        for (i, &byte) in bytes.iter().enumerate() {
            if byte.is_ascii_whitespace() && byte != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let separators: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();
        if separators.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if separators.len() > 1 {
            return Err(CronError::Layout(separators[1]));
        }
        let sep = separators[0];
        if sep == 0 || sep + 1 == bytes.len() {
            return Err(CronError::Layout(sep));
        }

        let minute_mask = parse_field(bytes, 0, sep, 60, 60)?;
        let hour_mask = parse_field(bytes, sep + 1, bytes.len(), 24, 24)?;
        Ok(Self {
            minute_mask,
            hour_mask: hour_mask as u32,
        })
    }

    pub fn matches(&self, minute: u64) -> bool {
        (self.minute_mask & (1u64 << (minute % 60))) != 0
            && (self.hour_mask & (1u32 << ((minute / 60) % 24))) != 0
    }
}

fn parse_field(
    bytes: &[u8],
    start: usize,
    end: usize,
    max: usize,
    step_limit: usize,
) -> Result<u64, CronError> {
    let mut mask = 0u64;
    let mut item_start = start;
    loop {
        let mut item_end = item_start;
        while item_end < end && bytes[item_end] != b',' {
            item_end += 1;
        }
        mask |= parse_item(bytes, item_start, item_end, max, step_limit)?;
        if item_end == end {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(mask)
}

fn parse_item(
    bytes: &[u8],
    start: usize,
    end: usize,
    max: usize,
    step_limit: usize,
) -> Result<u64, CronError> {
    if start == end {
        return Err(CronError::Syntax(start));
    }

    let slash = bytes[start..end]
        .iter()
        .position(|&b| b == b'/')
        .map(|p| start + p);
    if let Some(slash) = slash {
        if let Some(second) = bytes[slash + 1..end]
            .iter()
            .position(|&b| b == b'/')
        {
            return Err(CronError::Syntax(slash + 1 + second));
        }
        let base = &bytes[start..slash];
        let valid_step_base = base == b"*" || base.contains(&b'-');
        if !valid_step_base {
            return Err(CronError::Syntax(slash));
        }
    }

    let base_end = slash.unwrap_or(end);
    let base = &bytes[start..base_end];
    let (lo, hi) = if base == b"*" {
        (0, max - 1)
    } else if let Some(dash_pos) = base.iter().position(|&b| b == b'-') {
        let dash = start + dash_pos;
        let lo = parse_number(bytes, start, dash, max)?;
        let hi = parse_number(bytes, dash + 1, base_end, max)?;
        if lo > hi {
            return Err(CronError::Descending(dash));
        }
        (lo, hi)
    } else {
        let n = parse_number(bytes, start, base_end, max)?;
        (n, n)
    };

    let step = if let Some(slash) = slash {
        let step = parse_number(bytes, slash + 1, end, usize::MAX)?;
        if step == 0 || step > step_limit {
            return Err(CronError::Step(slash + 1));
        }
        step
    } else {
        1
    };

    let mut mask = 0u64;
    let mut n = lo;
    while n <= hi {
        mask |= 1u64 << n;
        match n.checked_add(step) {
            Some(next) if next > n => n = next,
            _ => break,
        }
    }
    Ok(mask)
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    max: usize,
) -> Result<usize, CronError> {
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut n = 0usize;
    for &b in &bytes[start..end] {
        n = match n.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as usize)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(start)),
        };
        if n >= max {
            return Err(CronError::Bounds(start));
        }
    }
    Ok(n)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.len() == 0
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        self.jobs.push(Job {
            id: id.to_string(),
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last;
        let first = match lower {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let final_minute = now / 60_000;
        let mut output: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let mut hits = Vec::new();
            if first <= final_minute {
                for minute in first..=final_minute {
                    if job.cron.matches(minute) {
                        let at = minute * 60_000;
                        if lower.map_or(true, |time| at > time) && at <= now {
                            hits.push(at);
                        }
                    }
                }
            }

            match job.policy {
                Policy::All => {
                    for at in hits {
                        output.push((Occurrence { id: job.id.clone(), at }, job.priority, job.order));
                    }
                }
                Policy::Once => {
                    if let Some(at) = hits.last().copied() {
                        output.push((Occurrence { id: job.id.clone(), at }, job.priority, job.order));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 && hits.last().copied() == Some(now) {
                        output.push((Occurrence { id: job.id.clone(), at: now }, job.priority, job.order));
                    }
                }
            }
        }

        output.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last = Some(now);
        Ok(output.into_iter().map(|x| x.0).collect())
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();
        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();
        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(text, 0, split, 60, 60, &mut minutes)?;
        parse_field(text, split + 1, text.len(), 24, 24, &mut hours)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    text: &str,
    start: usize,
    end: usize,
    max_value: usize,
    max_step: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut item_start = start;
    loop {
        let item_end = bytes[item_start..end]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(end);
        parse_item(text, item_start, item_end, max_value, max_step, selected)?;
        if item_end == end {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(())
}

fn parse_item(
    text: &str,
    start: usize,
    end: usize,
    max_value: usize,
    max_step: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let b = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }

    let mut slash = None;
    for i in start..end {
        if b[i] == b'/' {
            if slash.is_some() {
                return Err(CronError::Syntax(i));
            }
            slash = Some(i);
        }
    }

    let base_end = slash.unwrap_or(end);
    let step_start = slash.map(|i| i + 1);
    let base_is_star = base_end == start + 1 && b[start] == b'*';
    let dash = (start..base_end).find(|&i| b[i] == b'-');
    let is_range = dash.is_some();

    if slash.is_some() && !base_is_star && !is_range {
        return Err(CronError::Syntax(slash.unwrap()));
    }

    let (lo, hi) = if base_is_star {
        (0, max_value - 1)
    } else if let Some(d) = dash {
        let left = parse_number(text, start, d, max_value)?;
        let right = parse_number(text, d + 1, base_end, max_value)?;
        if left > right {
            return Err(CronError::Descending(d));
        }
        (left, right)
    } else {
        let n = parse_number(text, start, base_end, max_value)?;
        (n, n)
    };

    let step = if let Some(s) = step_start {
        parse_step(text, s, end, max_step)?
    } else {
        1
    };

    let mut n = lo;
    while n <= hi {
        selected[n] = true;
        match n.checked_add(step) {
            Some(next) => n = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_number(text: &str, start: usize, end: usize, limit: usize) -> Result<usize, CronError> {
    let b = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !b[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && b[start] == b'0' {
        return Err(CronError::Number(start));
    }
    let mut value = 0usize;
    for i in start..end {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b[i] - b'0') as usize)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(start)),
        };
    }
    if value >= limit {
        return Err(CronError::Bounds(start));
    }
    Ok(value)
}

fn parse_step(text: &str, start: usize, end: usize, max_step: usize) -> Result<usize, CronError> {
    let b = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !b[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && b[start] == b'0' {
        return Err(CronError::Number(start));

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {

    pub at: u64,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let owned_id = id.to_owned();
        self.ids.push(owned_id.clone());
        self.jobs.push(Job {
            id: owned_id,
            cron,
            priority,
            policy,
            order: self.jobs.len(),
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower_exclusive = self.last_poll;
        let upper_minute = now / 60_000;
        let mut out: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let first = match lower_exclusive {
                Some(last) => last / 60_000 + 1,
                None => 0,
            };
            if first > upper_minute {
                continue;
            }
            let matching = |m: u64| job.cron.matches(m);
            match job.policy {
                Policy::Skip => {
                    if now % 60_000 == 0
                        && now / 60_000 >= first
                        && matching(now / 60_000)
                    {
                        out.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Once => {
                    if let Some(m) = (first..=upper_minute).rev().find(|&m| matching(m)) {
                        out.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: m * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::All => {
                    for m in first..=upper_minute {
                        if matching(m) {
                            out.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: m * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        out.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(out.into_iter().map(|entry| entry.0).collect())
    }
}

impl Default for Calendar {
    fn default() -> Self {
        Self::new()
    }
}
