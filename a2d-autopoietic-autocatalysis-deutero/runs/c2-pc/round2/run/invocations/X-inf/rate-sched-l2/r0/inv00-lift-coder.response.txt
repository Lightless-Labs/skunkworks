#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(&bytes[..split], 0, 60, &mut minutes)?;
        parse_field(&bytes[split + 1..], split + 1, 24, &mut hours)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &[u8],
    offset: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;
    loop {
        let end = field[start..]
            .iter()
            .position(|&b| b == b',')
            .map_or(field.len(), |p| start + p);

        if start == end {
            return Err(CronError::Syntax(offset + start));
        }
        parse_item(&field[start..end], offset + start, limit, selected)?;

        if end == field.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

fn number(fragment: &[u8], start: usize) -> Result<usize, CronError> {
    if fragment.is_empty() {
        return Err(CronError::Syntax(start));
    }

    for (i, &b) in fragment.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(CronError::Syntax(start + i));
        }
    }

    if fragment.len() > 1 && fragment[0] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for &b in fragment {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as usize))
            .ok_or(CronError::Bounds(start))?;
    }
    Ok(value)
}

fn parse_item(
    item: &[u8],
    offset: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let slash = item.iter().position(|&b| b == b'/');
    if let Some(s) = slash {
        if let Some(second) = item[s + 1..].iter().position(|&b| b == b'/') {
            return Err(CronError::Syntax(offset + s + 1 + second));
        }
    }

    let base_end = slash.unwrap_or(item.len());
    let base = &item[..base_end];
    let max = limit - 1;

    enum Base {
        Star,
        Range(usize, usize),
        Single(usize),
    }

    let parsed_base = if base == b"*" {
        Base::Star
    } else if let Some(dash) = base.iter().position(|&b| b == b'-') {
        let left = number(&base[..dash], offset)?;
        if left > max {
            return Err(CronError::Bounds(offset));
        }
        let right_start = offset + dash + 1;
        let right = number(&base[dash + 1..], right_start)?;
        if right > max {
            return Err(CronError::Bounds(right_start));
        }
        if left > right {
            return Err(CronError::Descending(offset + dash));
        }
        Base::Range(left, right)
    } else {
        let value = number(base, offset)?;
        if value > max {
            return Err(CronError::Bounds(offset));
        }
        Base::Single(value)
    };

    let step = if let Some(slash_at) = slash {
        if !matches!(parsed_base, Base::Star | Base::Range(_, _)) {
            return Err(CronError::Syntax(offset + slash_at));
        }
        let step_start = offset + slash_at + 1;
        let value = number(&item[slash_at + 1..], step_start)?;
        let step_max = if limit == 60 { 60 } else { 24 };
        if value == 0 || value > step_max {
            return Err(CronError::Step(step_start));
        }
        value
    } else {
        1
    };

    let (begin, end) = match parsed_base {
        Base::Star => (0, max),
        Base::Range(a, b) => (a, b),
        Base::Single(n) => (n, n),
    };

    let mut n = begin;
    loop {
        selected[n] = true;
        if end - n < step {
            break;
        }
        n += step;
    }
    Ok(())
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    added_after: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.ids.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
            added_after: self.last_poll,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_poll = self.last_poll.is_none();
        let previous = self.last_poll.unwrap_or(0);
        let upper_minute = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let lower_exclusive = match (first_poll, job.added_after) {
                (true, _) => None,
                (false, Some(added)) => Some(previous.max(added)),
                (false, None) => Some(previous),
            };

            let mut first = match lower_exclusive {
                None => 0,
                Some(t) => t / 60_000 + 1,
            };
            if let Some(added) = job.added_after {
                first = first.max(added / 60_000 + 1);
            }

            if first > upper_minute {
                continue;
            }

            match job.policy {
                Policy::All => {
                    for minute in first..=upper_minute {
                        if job.cron.matches(minute) {
                            emitted.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Once => {
                    let mut latest = None;
                    for minute in first..=upper_minute {
                        if job.cron.matches(minute) {
                            latest = Some(minute);
                        }
                    }
                    if let Some(minute) = latest {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0
                        && now / 60_000 >= first
                        && job.cron.matches(now / 60_000)
                    {
                        emitted.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: now,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|item| item.0).collect())
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: u64,
    hours: u32,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let b = text.as_bytes();
        for (i, &byte) in b.iter().enumerate() {
            if byte.is_ascii_whitespace() && byte != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let mut separator = None;
        let mut count = 0usize;
        for (i, &byte) in b.iter().enumerate() {
            if byte == b' ' {
                count += 1;
                if separator.is_none() {
                    separator = Some(i);
                } else {
                    return Err(CronError::Layout(i));
                }
            }
        }
        let Some(separator) = separator else {
            return Err(CronError::Layout(text.len()));
        };
        if count != 1 || separator == 0 || separator + 1 == b.len() {
            return Err(CronError::Layout(separator));
        }

        let mut minutes = 0u64;
        let mut hours = 0u32;
        parse_bits(&b[..separator], 0, 60, &mut minutes)?;
        parse_bits(&b[separator + 1..], separator + 1, 24, &mut hours)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes & (1u64 << (minute % 60)) != 0
            && self.hours & (1u32 << ((minute / 60) % 24)) != 0
    }
}

fn parse_num(bytes: &[u8], start: usize) -> Result<usize, CronError> {
    if bytes.is_empty() {
        return Err(CronError::Syntax(start));
    }
    for (i, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(CronError::Syntax(start + i));
        }
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as usize))
            .ok_or(CronError::Bounds(start))?;
    }
    Ok(value)
}

fn parse_bits(
    field: &[u8],
    offset: usize,
    limit: usize,
    bits: &mut u64,
) -> Result<(), CronError> {
    let mut pos = 0;
    loop {
        let end = field[pos..]
            .iter()
            .position(|&b| b == b',')
            .map_or(field.len(), |i| pos + i);
        parse_bit_item(&field[pos..end], offset + pos, limit, bits)?;
        if end == field.len() {
            return Ok(());
        }
        pos = end + 1;
    }
}

fn parse_bit_item(
    item: &[u8],
    offset: usize,
    limit: usize,
    bits: &mut u64,
) -> Result<(), CronError> {
    let slash = item.iter().position(|&b| b == b'/');
    if let Some(s) = slash {
        if let Some(i) = item[s + 1..].iter().position(|&b| b == b'/') {
            return Err(CronError::Syntax(offset + s + 1 + i));
        }
    }

    let base_end = slash.unwrap_or(item.len());
    let base = &item[..base_end];
    let max = limit - 1;
    let (from, to, is_range_or_star) = if base == b"*" {
        (0, max, true)
    } else if let Some(dash) = base.iter().position(|&b| b == b'-') {
        let a = parse_num(&base[..dash], offset)?;
        if a > max {
            return Err(CronError::Bounds(offset));
        }
        let right_pos = offset + dash + 1;
        let b = parse_num(&base[dash + 1..], right_pos)?;
        if b > max {
            return Err(CronError::Bounds(right_pos));
        }
        if a > b {
            return Err(CronError::Descending(offset + dash));
        }
        (a, b, true)
    } else {
        let n = parse_num(base, offset)?;
        if n > max {
            return Err(CronError::Bounds(offset));
        }
        (n, n, false)
    };

    let step = match slash {
        None => 1,
        Some(s) => {
            if !is_range_or_star {
                return Err(CronError::Syntax(offset + s));
            }
            let step_pos = offset + s + 1;
            let value = parse_num(&item[s + 1..], step_pos)?;
            if value == 0 || value > limit {
                return Err(CronError::Step(step_pos));
            }
            value
        }
    };

    let mut n = from;
    loop {
        *bits |= 1u64 << n;
        if to - n < step {
            break;
        }
        n += step;
    }
    Ok(())
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Entry {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    insertion: usize,
    created_after: Option<u64>,
}

pub struct Calendar {
    entries: Vec<Entry>,
    reserved: Vec<String>,
    watermark: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            reserved: Vec::new(),
            watermark: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if !(1..=16).contains(&id.len())
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved.iter().any(|value| value == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let insertion = self.entries.len();
        self.reserved.push(id.to_owned());
        self.entries.push(Entry {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            insertion,
            created_after: self.watermark,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if self.watermark.is_some_and(|last| now < last) {
            return Err(Error::TimeReversed);
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first = self.watermark.is_none();
        let last = self.watermark.unwrap_or(0);
        let end = now / 60_000;
        let mut result: Vec<(Occurrence, u8, usize)> = Vec::new();

        for entry in &self.entries {
            let lower = if first {
                0
            } else {
                last / 60_000 + 1
            };
            let lower = entry
                .created_after
                .map_or(lower, |t| lower.max(t / 60_000 + 1));
            if lower > end {
                continue;
            }

            match entry.policy {
                Policy::Skip => {
                    if now % 60_000 == 0 && now / 60_000 >= lower {
                        let minute = now / 60_000;
                        if entry.cron.matches(minute) {
                            result.push((
                                Occurrence {
                                    id: entry.id.clone(),
                                    at: now,
                                },
                                entry.priority,
                                entry.insertion,
                            ));
                        }
                    }
                }
                Policy::Once => {
                    let latest = (lower..=end)
                        .rev()
                        .find(|&minute| entry.cron.matches(minute));
                    if let Some(minute) = latest {
                        result.push((
                            Occurrence {
                                id: entry.id.clone(),
                                at: minute * 60_000,
                            },
                            entry.priority,
                            entry.insertion,
                        ));
                    }
                }
                Policy::All => {
                    for minute in lower..=end {
                        if entry.cron.matches(minute) {
                            result.push((
                                Occurrence {
                                    id: entry.id.clone(),
                                    at: minute * 60_000,
                                },
                                entry.priority,
                                entry.insertion,
                            ));
                        }
                    }
                }
            }
        }

        result.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.watermark = Some(now);
        Ok(result.into_iter().map(|row| row.0).collect())
    }
}
