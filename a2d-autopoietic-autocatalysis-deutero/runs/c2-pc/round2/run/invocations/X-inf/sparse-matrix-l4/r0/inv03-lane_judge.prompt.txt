        fn err(line:usize,kind:TextError)->Error{Error::Text{line,kind}}
        fn dec(s:&str)->Option<usize>{
            if s.is_empty()||!s.bytes().all(|b|b.is_ascii_digit())||(s.len()>1&&s.starts_with('0')){None}else{s.parse().ok()}
        }
        if !text.ends_with('\n'){return Err(err(1+text.bytes().filter(|b|*b==b'\n').count(),TextError::Newline));}
        let lines:Vec<&str>=text[..text.len()-1].split('\n').collect();
        let h:Vec<&str>=lines[0].split(' ').collect();
        if h.len()!=4||h.iter().any(|x|x.is_empty())||h[0]!="SM1"{return Err(err(1,TextError::Header));}
        let rows=dec(h[1]).ok_or_else(||err(1,TextError::Decimal))?;
        let cols=dec(h[2]).ok_or_else(||err(1,TextError::Decimal))?;
        let n=dec(h[3]).ok_or_else(||err(1,TextError::Decimal))?;
        if rows>MAX_DIM||cols>MAX_DIM{return Err(err(1,TextError::Dimension));}
        if n!=lines.len()-1{return Err(err(1,TextError::Count));}
        let mut rp=vec![0;rows+1];let mut ci=Vec::with_capacity(n);let mut vs=Vec::with_capacity(n);
        let mut prior=None;let mut cr=0;
        for (i,line) in lines.iter().enumerate().skip(1){
            let ln=i+1;let f:Vec<&str>=line.split(' ').collect();
            if f.len()!=3||f.iter().any(|x|x.is_empty()){return Err(err(ln,TextError::Record));}
            let r=dec(f[0]).ok_or_else(||err(ln,TextError::Decimal))?;
            let c=dec(f[1]).ok_or_else(||err(ln,TextError::Decimal))?;
            if f[2].len()!=16||!f[2].bytes().all(|b|b.is_ascii_digit()||(b'a'..=b'f').contains(&b)){return Err(err(ln,TextError::Hex));}
            let v=f64::from_bits(u64::from_str_radix(f[2],16).unwrap());
            if r>=rows{return Err(err(ln,TextError::RowBounds));}
            if c>=cols{return Err(err(ln,TextError::ColumnBounds));}
            if !v.is_finite(){return Err(err(ln,TextError::NonFinite));}
            if v==0.0{return Err(err(ln,TextError::Zero));}
            if prior.is_some_and(|p|(r,c)<=p){return Err(err(ln,TextError::Order));}
            while cr<r{rp[cr+1]=ci.len();cr+=1;}
            ci.push(c);vs.push(v);prior=Some((r,c));
        }
        while cr<rows{rp[cr+1]=ci.len();cr+=1;}
        Ok(Csr{rows,cols,row_ptr:rp,col_idx:ci,values:vs})
    }
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        for (i, e) in entries.iter().enumerate() {
            if e.row >= rows {
                return Err(Error::Bounds { entry: i, axis: Axis::Row, index: e.row, limit: rows });
            }
            if e.col >= cols {
                return Err(Error::Bounds { entry: i, axis: Axis::Column, index: e.col, limit: cols });
            }
            if !e.value.is_finite() {
                return Err(Error::NonFinite { entry: i });
            }
        }

        let mut sums: Vec<((usize, usize), f64)> = Vec::new();
        for (i, e) in entries.iter().enumerate() {
            if let Some((_, sum)) = sums.iter_mut().find(|((r, c), _)| *r == e.row && *c == e.col) {
                *sum += e.value;
                if !sum.is_finite() {
                    return Err(Error::SumOverflow { entry: i });
                }
            } else {
                sums.push(((e.row, e.col), 0.0));
                let sum = &mut sums.last_mut().unwrap().1;
                *sum += e.value;
                if !sum.is_finite() {
                    return Err(Error::SumOverflow { entry: i });
                }
            }
        }
        sums.retain(|(_, v)| *v != 0.0);
        sums.sort_by_key(|((r, c), _)| (*r, *c));
        let canonical = sums.into_iter().map(|((row, col), value)| Entry { row, col, value }).collect();
        Ok(Self { rows, cols, entries: canonical })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }

    pub fn to_csr(&self) -> Csr {
        let mut row_ptr = vec![0; self.rows + 1];
        for e in &self.entries {
            row_ptr[e.row + 1] += 1;
        }
        for r in 0..self.rows {
            row_ptr[r + 1] += row_ptr[r];
        }
        Csr {
            rows: self.rows,
            cols: self.cols,
            row_ptr,
            col_idx: self.entries.iter().map(|e| e.col).collect(),
            values: self.entries.iter().map(|e| e.value).collect(),
        }
    }

impl Csr {
    pub fn from_parts(rows: usize, cols: usize, row_ptr: Vec<usize>, col_idx: Vec<usize>, values: Vec<f64>) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        if row_ptr.len() != rows + 1 {
            return Err(Error::RowPtrLength { expected: rows + 1, actual: row_ptr.len() });
        }
        if col_idx.len() != values.len() {
            return Err(Error::ColumnValueLength { columns: col_idx.len(), values: values.len() });
        }
        if row_ptr[0] != 0 {
            return Err(Error::RowPtrStart { found: row_ptr[0] });
        }
        if row_ptr[rows] != values.len() {
            return Err(Error::RowPtrEnd { found: row_ptr[rows], expected: values.len() });
        }
        for r in 0..rows {
            if row_ptr[r] > row_ptr[r + 1] {
                return Err(Error::RowPtrOrder { row: r });
            }
        }
        for r in 0..rows {
            let mut previous = None;
            for s in row_ptr[r]..row_ptr[r + 1] {
                let col = col_idx[s];
                if col >= cols {
                    return Err(Error::ColumnBounds { slot: s, column: col, limit: cols });
                }
                if previous.is_some_and(|p| col <= p) {
                    return Err(Error::ColumnOrder { slot: s });
                }
                if !values[s].is_finite() {
                    return Err(Error::NonFiniteValue { slot: s });
                }
                if values[s] == 0.0 {
                    return Err(Error::ExplicitZero { slot: s });
                }
                previous = Some(col);
            }
        }
        Ok(Self { rows, cols, row_ptr, col_idx, values })
    }

    pub fn shape(&self) -> (usize, usize) { (self.rows, self.cols) }
    pub fn row_ptr(&self) -> &[usize] { &self.row_ptr }
    pub fn col_indices(&self) -> &[usize] { &self.col_idx }
    pub fn values(&self) -> &[f64] { &self.values }

    pub fn to_coo(&self) -> Sparse {
        let mut entries = Vec::with_capacity(self.values.len());
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                entries.push(Entry { row: r, col: self.col_idx[s], value: self.values[s] });
            }
        }
        Sparse { rows: self.rows, cols: self.cols, entries }
    }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        if x.len() != self.cols {
            return Err(Error::VectorLength { expected: self.cols, actual: x.len() });
        }
        for (i, v) in x.iter().enumerate() {
            if !v.is_finite() {
                return Err(Error::NonFiniteVector { index: i });
            }
        }
        let mut out = Vec::with_capacity(self.rows);
        for r in 0..self.rows {
            let mut acc = 0.0;
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                let c = self.col_idx[s];
                let p = self.values[s] * x[c];
                if !p.is_finite() {
                    return Err(Error::ProductOverflow { row: r, column: c });
                }
                acc += p;
                if !acc.is_finite() {
                    return Err(Error::AccumulationOverflow { row: r, column: c });
                }
            }
            out.push(if acc == 0.0 { 0.0 } else { acc });
        }
        Ok(out)
    }

    pub fn transpose(&self) -> Csr {
        let mut counts = vec![0usize; self.cols];
        for &c in &self.col_idx { counts[c] += 1; }
        let mut row_ptr = vec![0usize; self.cols + 1];
        for r in 0..self.cols { row_ptr[r + 1] = row_ptr[r] + counts[r]; }
        let mut next = row_ptr[..self.cols].to_vec();
        let mut col_idx = vec![0usize; self.values.len()];
        let mut values = vec![0.0; self.values.len()];
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                let c = self.col_idx[s];
                let at = next[c];
                col_idx[at] = r;
                values[at] = self.values[s];
                next[c] += 1;
            }
        }
        Csr { rows: self.cols, cols: self.rows, row_ptr, col_idx, values }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch { left_cols: self.cols, right_rows: rhs.rows });
        }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for r in 0..self.rows {
            let mut acc: Vec<(usize, f64)> = Vec::new();
            for ls in self.row_ptr[r]..self.row_ptr[r + 1] {
                let k = self.col_idx[ls];
                for rs in rhs.row_ptr[k]..rhs.row_ptr[k + 1] {
                    let c = rhs.col_idx[rs];
                    let p = self.values[ls] * rhs.values[rs];
                    if !p.is_finite() {
                        return Err(Error::MultiplyOverflow { row: r, inner: k, column: c });
                    }
                    if let Some((_, a)) = acc.iter_mut().find(|(col, _)| *col == c) {
                        *a += p;
                        if !a.is_finite() {
                            return Err(Error::MultiplySumOverflow { row: r, inner: k, column: c });
                        }
                    } else {
                        acc.push((c, 0.0));
                        let a = &mut acc.last_mut().unwrap().1;
                        *a += p;
                        if !a.is_finite() {
                            return Err(Error::MultiplySumOverflow { row: r, inner: k, column: c });
                        }
                    }
                }
            }
            acc.retain(|(_, v)| *v != 0.0);
            acc.sort_by_key(|(c, _)| *c);
            for (c, v) in acc {
                col_idx.push(c);
                values.push(v);
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: rhs.cols, row_ptr, col_idx, values })
    }

    pub fn prune(&self, tolerance: f64) -> Result<Csr, Error> {
        if !tolerance.is_finite() || tolerance < 0.0 {
            return Err(Error::Tolerance);
        }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                if self.values[s].abs() > tolerance {
                    col_idx.push(self.col_idx[s]);
                    values.push(self.values[s]);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: self.cols, row_ptr, col_idx, values })
    }

    pub fn write_text(&self) -> String {
        use std::fmt::Write;
        let mut out = String::new();
        let _ = writeln!(&mut out, "SM1 {} {} {}", self.rows, self.cols, self.values.len());
        for r in 0..self.rows {
            for s in self.row_ptr[r]..self.row_ptr[r + 1] {
                let _ = writeln!(&mut out, "{} {} {:016x}", r, self.col_idx[s], self.values[s].to_bits());
            }
        }
        out
    }

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        fn text_err(line: usize, kind: TextError) -> Error { Error::Text { line, kind } }
        fn decimal(s: &str) -> Option<usize> {
            if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) { return None; }
            if s.len() > 1 && s.as_bytes()[0] == b'0' { return None; }
            s.parse::<usize>().ok()
        }

        if !text.ends_with('\n') {
            return Err(text_err(1 + text.bytes().filter(|&b| b == b'\n').count(), TextError::Newline));
        }
        let body = &text[..text.len() - 1];
        let lines: Vec<&str> = body.split('\n').collect();
        let header: Vec<&str> = lines[0].split(' ').collect();
        if header.len() != 4 || header.iter().any(|s| s.is_empty()) || header[0] != "SM1" {
            return Err(text_err(1, TextError::Header));
        }
        let rows = decimal(header[1]).ok_or_else(|| text_err(1, TextError::Decimal))?;
        let cols = decimal(header[2]).ok_or_else(|| text_err(1, TextError::Decimal))?;
        let nnz = decimal(header[3]).ok_or_else(|| text_err(1, TextError::Decimal))?;
        if rows > MAX_DIM || cols > MAX_DIM {
            return Err(text_err(1, TextError::Dimension));
        }
        if nnz != lines.len() - 1 {
            return Err(text_err(1, TextError::Count));
        }

        let mut row_ptr = vec![0usize; rows + 1];
        let mut col_idx = Vec::with_capacity(nnz);
        let mut values = Vec::with_capacity(nnz);
        let mut previous: Option<(usize, usize)> = None;
        let mut current_row = 0usize;

        for (i, line) in lines.iter().enumerate().skip(1) {
            let line_no = i + 1;
            let fields: Vec<&str> = line.split(' ').collect();
            if fields.len() != 3 || fields.iter().any(|s| s.is_empty()) {
                return Err(text_err(line_no, TextError::Record));
            }
            let row = decimal(fields[0]).ok_or_else(|| text_err(line_no, TextError::Decimal))?;
            let col = decimal(fields[1]).ok_or_else(|| text_err(line_no, TextError::Decimal))?;
            if fields[2].len() != 16 || !fields[2].bytes().all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b)) {
                return Err(text_err(line_no, TextError::Hex));
            }
            let bits = u64::from_str_radix(fields[2], 16).unwrap();
            let value = f64::from_bits(bits);
            if row >= rows { return Err(text_err(line_no, TextError::RowBounds)); }
            if col >= cols { return Err(text_err(line_no, TextError::ColumnBounds)); }
            if !value.is_finite() { return Err(text_err(line_no, TextError::NonFinite)); }
            if value == 0.0 { return Err(text_err(line_no, TextError::Zero)); }
            if previous.is_some_and(|p| (row, col) <= p) {
                return Err(text_err(line_no, TextError::Order));
            }
            while current_row < row {
                row_ptr[current_row + 1] = col_idx.len();
                current_row += 1;
            }
            col_idx.push(col);
            values.push(value);
            previous = Some((row, col));
        }
        while current_row < rows {
            row_ptr[current_row + 1] = col_idx.len();
            current_row += 1;
        }
        Ok(Csr { rows, cols, row_ptr, col_idx, values })
    }

fn shape_check(r: usize, c: usize) -> Result<(), Error> {
    if r > MAX_DIM { Err(Error::Shape { axis: Axis::Row, size: r }) }
    else if c > MAX_DIM { Err(Error::Shape { axis: Axis::Column, size: c }) }
    else { Ok(()) }
}

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        shape_check(rows, cols)?;
        for (i, e) in entries.iter().enumerate() {
            if e.row >= rows { return Err(Error::Bounds { entry: i, axis: Axis::Row, index: e.row, limit: rows }); }
            if e.col >= cols { return Err(Error::Bounds { entry: i, axis: Axis::Column, index: e.col, limit: cols }); }
            if !e.value.is_finite() { return Err(Error::NonFinite { entry: i }); }
        }
        let mut data = entries.to_vec();
        data.sort_by_key(|e| (e.row, e.col));
        let mut out = Vec::new();
        let mut i = 0;
        while i < data.len() {
            let row = data[i].row;
            let col = data[i].col;
            let mut sum = 0.0;
            let mut j = i;
            while j < data.len() && data[j].row == row && data[j].col == col {
                sum += data[j].value;
                if !sum.is_finite() {
                    // Recompute in original order to report the exact offending occurrence.
                    let mut original_sum = 0.0;
                    for (k, e) in entries.iter().enumerate() {
                        if e.row == row && e.col == col {
                            original_sum += e.value;
                            if !original_sum.is_finite() { return Err(Error::SumOverflow { entry: k }); }
                        }
                    }
                }
                j += 1;
            }
            if sum != 0.0 { out.push(Entry { row, col, value: sum }); }
            i = j;
        }
        Ok(Self { shape: (rows, cols), data: out })
    }

    pub fn shape(&self) -> (usize, usize) { self.shape }
    pub fn entries(&self) -> &[Entry] { &self.data }

    pub fn to_csr(&self) -> Csr {
        let (rows, cols) = self.shape;
        let mut rp = vec![0; rows + 1];
        for e in &self.data { rp[e.row + 1] += 1; }
        for r in 0..rows { rp[r + 1] += rp[r]; }
        Csr { shape: self.shape, rp, ci: self.data.iter().map(|e| e.col).collect(), v: self.data.iter().map(|e| e.value).collect() }
    }
}

impl Csr {
    pub fn from_parts(rows: usize, cols: usize, rp: Vec<usize>, ci: Vec<usize>, v: Vec<f64>) -> Result<Self, Error> {
        shape_check(rows, cols)?;
        if rp.len() != rows + 1 { return Err(Error::RowPtrLength { expected: rows + 1, actual: rp.len() }); }
        if ci.len() != v.len() { return Err(Error::ColumnValueLength { columns: ci.len(), values: v.len() }); }
        if rp[0] != 0 { return Err(Error::RowPtrStart { found: rp[0] }); }
        if rp[rows] != v.len() { return Err(Error::RowPtrEnd { found: rp[rows], expected: v.len() }); }
        for r in 0..rows { if rp[r] > rp[r + 1] { return Err(Error::RowPtrOrder { row: r }); } }
        for r in 0..rows {
            let mut prev = None;
            for s in rp[r]..rp[r + 1] {
                let c = ci[s];
                if c >= cols { return Err(Error::ColumnBounds { slot: s, column: c, limit: cols }); }
                if prev.is_some_and(|p| c <= p) { return Err(Error::ColumnOrder { slot: s }); }
                if !v[s].is_finite() { return Err(Error::NonFiniteValue { slot: s }); }
                if v[s] == 0.0 { return Err(Error::ExplicitZero { slot: s }); }
                prev = Some(c);
            }
        }
        Ok(Self { shape: (rows, cols), rp, ci, v })
    }

    pub fn shape(&self) -> (usize, usize) { self.shape }
    pub fn row_ptr(&self) -> &[usize] { &self.rp }
    pub fn col_indices(&self) -> &[usize] { &self.ci }
    pub fn values(&self) -> &[f64] { &self.v }

    pub fn to_coo(&self) -> Sparse {
        let mut data = Vec::with_capacity(self.v.len());
        for r in 0..self.shape.0 {
            for s in self.rp[r]..self.rp[r + 1] {
                data.push(Entry { row: r, col: self.ci[s], value: self.v[s] });
            }
        }
        Sparse { shape: self.shape, data }
    }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        let (rows, cols) = self.shape;
        if x.len() != cols { return Err(Error::VectorLength { expected: cols, actual: x.len() }); }
        for (i, y) in x.iter().enumerate() { if !y.is_finite() { return Err(Error::NonFiniteVector { index: i }); } }
        let mut out = Vec::with_capacity(rows);
        for r in 0..rows {
            let mut sum = 0.0;
            for s in self.rp[r]..self.rp[r + 1] {
                let c = self.ci[s];
                let p = self.v[s] * x[c];
                if !p.is_finite() { return Err(Error::ProductOverflow { row: r, column: c }); }
                sum += p;
                if !sum.is_finite() { return Err(Error::AccumulationOverflow { row: r, column: c }); }
            }
            out.push(if sum == 0.0 { 0.0 } else { sum });
        }
        Ok(out)
    }

    pub fn transpose(&self) -> Csr {
        let (rows, cols) = self.shape;
        let mut rp = vec![0; cols + 1];
        for &c in &self.ci { rp[c + 1] += 1; }
        for r in 0..cols { rp[r + 1] += rp[r]; }
        let mut next = rp[..cols].to_vec();
        let mut ci = vec![0; self.v.len()];
        let mut v = vec![0.0; self.v.len()];
        for r in 0..rows {
            for s in self.rp[r]..self.rp[r + 1] {
                let c = self.ci[s];
                let d = next[c];
                next[c] += 1;
                ci[d] = r;
                v[d] = self.v[s];
            }
        }
        Csr { shape: (cols, rows), rp, ci, v }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        let (rows, inner) = self.shape;
        let (rr, cols) = rhs.shape;
        if inner != rr { return Err(Error::DimensionMismatch { left_cols: inner, right_rows: rr }); }
        let mut rp = vec![0];
        let mut ci = Vec::new();
        let mut v = Vec::new();
        for r in 0..rows {
            // Accumulators indexed directly by output column, but visited in required
            // left-slot then right-slot order.
            let mut sums = vec![0.0; cols];
            let mut touched = vec![false; cols];
            for a in self.rp[r]..self.rp[r + 1] {
                let k = self.ci[a];
                for b in rhs.rp[k]..rhs.rp[k + 1] {
                    let c = rhs.ci[b];
                    let p = self.v[a] * rhs.v[b];
                    if !p.is_finite() { return Err(Error::MultiplyOverflow { row: r, inner: k, column: c }); }
                    sums[c] += p;
                    touched[c] = true;
                    if !sums[c].is_finite() { return Err(Error::MultiplySumOverflow { row: r, inner: k, column: c }); }
                }
            }
            for c in 0..cols {
                if touched[c] && sums[c] != 0.0 { ci.push(c); v.push(sums[c]); }
            }
            rp.push(v.len());
        }
        Ok(Csr { shape: (rows, cols), rp, ci, v })
    }

    pub fn prune(&self, tolerance: f64) -> Result<Csr, Error> {
        if !tolerance.is_finite() || tolerance < 0.0 { return Err(Error::Tolerance); }
        let (rows, cols) = self.shape;
        let mut rp = vec![0];
        let mut ci = Vec::new();
        let mut v = Vec::new();
        for r in 0..rows {
            for s in self.rp[r]..self.rp[r + 1] {
                if self.v[s].abs() > tolerance { ci.push(self.ci[s]); v.push(self.v[s]); }
            }
            rp.push(v.len());
        }
        Ok(Csr { shape: (rows, cols), rp, ci, v })
    }

    pub fn write_text(&self) -> String {
        let (rows, cols) = self.shape;
        let mut s = format!("SM1 {} {} {}\n", rows, cols, self.v.len());
        for r in 0..rows {
            for i in self.rp[r]..self.rp[r + 1] {
                s.push_str(&format!("{} {} {:016x}\n", r, self.ci[i], self.v[i].to_bits()));
            }
        }
        s
    }

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        fn err(line: usize, kind: TextError) -> Error { Error::Text { line, kind } }
        fn dec(s: &str) -> Option<usize> {
            if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) || (s.len() > 1 && s.starts_with('0')) { None }
            else { s.parse().ok() }
        }
        if !text.ends_with('\n') {
            return Err(err(1 + text.bytes().filter(|&b| b == b'\n').count(), TextError::Newline));
        }
        let lines: Vec<&str> = text[..text.len() - 1].split('\n').collect();
        let h: Vec<&str> = lines[0].split(' ').collect();
        if h.len() != 4 || h.iter().any(|x| x.is_empty()) || h[0] != "SM1" { return Err(err(1, TextError::Header)); }
        let rows = dec(h[1]).ok_or_else(|| err(1, TextError::Decimal))?;
        let cols = dec(h[2]).ok_or_else(|| err(1, TextError::Decimal))?;
        let n = dec(h[3]).ok_or_else(|| err(1, TextError::Decimal))?;
        if rows > MAX_DIM || cols > MAX_DIM { return Err(err(1, TextError::Dimension)); }
        if n != lines.len() - 1 { return Err(err(1, TextError::Count)); }

        let mut rp = vec![0; rows + 1];
        let mut ci = Vec::with_capacity(n);
        let mut v = Vec::with_capacity(n);
        let mut prev = None;
        let mut next_row = 0;
        for (idx, line) in lines.iter().enumerate().skip(1) {
            let ln = idx + 1;
            let f: Vec<&str> = line.split(' ').collect();
            if f.len() != 3 || f.iter().any(|x| x.is_empty()) { return Err(err(ln, TextError::Record)); }
            let r = dec(f[0]).ok_or_else(|| err(ln, TextError::Decimal))?;
            let c = dec(f[1]).ok_or_else(|| err(ln, TextError::Decimal))?;
            if f[2].len() != 16 || !f[2].bytes().all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b)) {
                return Err(err(ln, TextError::Hex));
            }
            let bits = u64::from_str_radix(f[2], 16).map_err(|_| err(ln, TextError::Hex))?;
            let value = f64::from_bits(bits);
            if r >= rows { return Err(err(ln, TextError::RowBounds)); }
            if c >= cols { return Err(err(ln, TextError::ColumnBounds)); }
            if !value.is_finite() { return Err(err(ln, TextError::NonFinite)); }
            if value == 0.0 { return Err(err(ln, TextError::Zero)); }
            if prev.is_some_and(|p| (r, c) <= p) { return Err(err(ln, TextError::Order)); }
            while next_row < r {
                rp[next_row + 1] = ci.len();
                next_row += 1;
            }
            ci.push(c);
            v.push(value);
            prev = Some((r, c));
        }
        while next_row < rows {
            rp[next_row + 1] = ci.len();
            next_row += 1;
        }
        Ok(Csr { shape: (rows, cols), rp, ci, v })
    }
}

    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        for (i, e) in entries.iter().enumerate() {
            if e.row >= rows {
                return Err(Error::Bounds { entry: i, axis: Axis::Row, index: e.row, limit: rows });
            }
            if e.col >= cols {
                return Err(Error::Bounds { entry: i, axis: Axis::Column, index: e.col, limit: cols });

            if !e.value.is_finite() {

        for (i, e) in entries.iter().enumerate() {
            let sum = sums.entry((e.row, e.col)).or_insert(0.0);
            *sum = *sum + e.value;
            if !sum.is_finite() {
                return Err(Error::SumOverflow { entry: i });
            }

            if value == 0.0 { None } else { Some(Entry { row, col, value }) }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {

    pub fn transpose(&self) -> Csr {
        let mut counts = vec![0usize; self.cols];
        for &c in &self.col_idx {
            counts[c] += 1;
        }

                col_idx[dest] = r;
                values[dest] = self.values[s];
                next[c] += 1;
            }
        }
        Csr { rows: self.cols, cols: self.rows, row_ptr, col_idx, values }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch { left_cols: self.cols, right_rows: rhs.rows });
        }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for r in 0..self.rows {
            let mut sums: BTreeMap<usize, f64> = BTreeMap::new();
            for a in self.row_ptr[r]..self.row_ptr[r + 1] {
                let k = self.col_idx[a];
                for b in rhs.row_ptr[k]..rhs.row_ptr[k + 1] {
                    let c = rhs.col_idx[b];
                    let p = self.values[a] * rhs.values[b];
                    if !p.is_finite() {
                        return Err(Error::MultiplyOverflow { row: r, inner: k, column: c });

                    let acc = sums.entry(c).or_insert(0.0);

                    if !acc.is_finite() {
                        return Err(Error::MultiplySumOverflow { row: r, inner: k, column: c });
                    }
                }
            }
            for (c, v) in sums {
                if v != 0.0 {
                    col_idx.push(c);
                    values.push(v);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: rhs.cols, row_ptr, col_idx, values })

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        fn fail(line: usize, kind: TextError) -> Error {
            Error::Text { line, kind }
        }
        fn decimal(token: &str) -> Option<usize> {
            if token.is_empty() || !token.bytes().all(|b| b.is_ascii_digit()) {
                return None;
            }
            if token.len() > 1 && token.as_bytes()[0] == b'0' {
                return None;
            }
            token.parse::<usize>().ok()
        }
        fn fields<'a>(line: &'a str, count: usize) -> Option<Vec<&'a str>> {
            let parts: Vec<_> = line.split(' ').collect();
            if parts.len() == count && parts.iter().all(|s| !s.is_empty()) {
                Some(parts)
            } else {
                None
            }
        }

        if !text.ends_with('\n') {
            return Err(fail(1 + text.bytes().filter(|&b| b == b'\n').count(), TextError::Newline));
        }
        let body = &text[..text.len() - 1];
        let lines: Vec<&str> = body.split('\n').collect();
        let header = fields(lines[0], 4).ok_or_else(|| fail(1, TextError::Header))?;
        if header[0] != "SM1" {
            return Err(fail(1, TextError::Header));
        }
        let rows = decimal(header[1]).ok_or_else(|| fail(1, TextError::Decimal))?;
        let cols = decimal(header[2]).ok_or_else(|| fail(1, TextError::Decimal))?;
        let nnz = decimal(header[3]).ok_or_else(|| fail(1, TextError::Decimal))?;
        if rows > MAX_DIM || cols > MAX_DIM {
            return Err(fail(1, TextError::Dimension));
        }
        if nnz != lines.len() - 1 {
            return Err(fail(1, TextError::Count));
        }

        let mut parsed = Vec::with_capacity(nnz);
        for (i, line) in lines.iter().enumerate().skip(1) {
            let line_no = i + 1;
            let f = fields(line, 3).ok_or_else(|| fail(line_no, TextError::Record))?;
            let row = decimal(f[0]).ok_or_else(|| fail(line_no, TextError::Decimal))?;
            let col = decimal(f[1]).ok_or_else(|| fail(line_no, TextError::Decimal))?;
            if f[2].len() != 16 || !f[2].bytes().all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b)) {
                return Err(fail(line_no, TextError::Hex));
            }
            let bits = u64::from_str_radix(f[2], 16).map_err(|_| fail(line_no, TextError::Hex))?;
            let value = f64::from_bits(bits);
            if row >= rows {
                return Err(fail(line_no, TextError::RowBounds));
            }
            if col >= cols {
                return Err(fail(line_no, TextError::ColumnBounds));
            }
            if !value.is_finite() {
                return Err(fail(line_no, TextError::NonFinite));
            }
            if value == 0.0 {
                return Err(fail(line_no, TextError::Zero));
            }
            if let Some(&(pr, pc, _)) = parsed.last() {
                if (row, col) <= (pr, pc) {
                    return Err(fail(line_no, TextError::Order));
                }
            }
            parsed.push((row, col, value));
        }

        let mut row_ptr = vec![0usize; rows + 1];
        for &(r, _, _) in &parsed {
            row_ptr[r + 1] += 1;
        }
        for r in 0..rows {
            row_ptr[r + 1] += row_ptr[r];
        }
        let col_idx = parsed.iter().map(|&(_, c, _)| c).collect();
        let values = parsed.iter().map(|&(_, _, v)| v).collect();
        Ok(Csr { rows, cols, row_ptr, col_idx, values })
    }

fn check_shape(rows: usize, cols: usize) -> Result<(), Error> {
    if rows > MAX_DIM {
        Err(Error::Shape { axis: Axis::Row, size: rows })
    } else if cols > MAX_DIM {
        Err(Error::Shape { axis: Axis::Column, size: cols })
    } else {
        Ok(())
    }

impl Sparse {
    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        for (i, e) in entries.iter().enumerate() {
            if e.row >= rows {
                return Err(Error::Bounds { entry: i, axis: Axis::Row, index: e.row, limit: rows });
            }
            if e.col >= cols {
                return Err(Error::Bounds { entry: i, axis: Axis::Column, index: e.col, limit: cols });
            }
            if !e.value.is_finite() {
                return Err(Error::NonFinite { entry: i });
            }
        }

        let mut sums: Vec<(usize, usize, f64)> = Vec::new();
        for (i, e) in entries.iter().enumerate() {
            if let Some((_, _, sum)) = sums.iter_mut().find(|(r, c, _)| *r == e.row && *c == e.col) {
                *sum = *sum + e.value;
                if !sum.is_finite() {
                    return Err(Error::SumOverflow { entry: i });
                }
            } else {
                sums.push((e.row, e.col, 0.0 + e.value));
            }
        }
        sums.sort_by_key(|(r, c, _)| (*r, *c));
        let entries = sums.into_iter()
            .filter(|(_, _, v)| *v != 0.0)
            .map(|(row, col, value)| Entry { row, col, value })
            .collect();
        Ok(Self { rows, cols, entries })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)
    }

    pub fn entries(&self) -> &[Entry] {
        &self.entries
    }
}

impl Sparse {
    pub fn to_csr(&self) -> Csr {
        let mut row_ptr = vec![0; self.rows + 1];
        for e in &self.entries {
            row_ptr[e.row + 1] += 1;
        }
        for r in 0..self.rows {
            row_ptr[r + 1] += row_ptr[r];
        }
        Csr {
            rows: self.rows,
            cols: self.cols,
            row_ptr,
            col_idx: self.entries.iter().map(|e| e.col).collect(),
            values: self.entries.iter().map(|e| e.value).collect(),
        }
    }

    pub fn from_parts(
        rows: usize,
        cols: usize,
        row_ptr: Vec<usize>,
        col_idx: Vec<usize>,
        values: Vec<f64>,
    ) -> Result<Self, Error> {
        check_shape(rows, cols)?;
        let expected = rows + 1;
        if row_ptr.len() != expected {
            return Err(Error::RowPtrLength { expected, actual: row_ptr.len() });
        }
        if col_idx.len() != values.len() {
            return Err(Error::ColumnValueLength { columns: col_idx.len(), values: values.len() });
        }
        if row_ptr[0] != 0 {
            return Err(Error::RowPtrStart { found: row_ptr[0] });
        }
        if row_ptr[rows] != values.len() {
            return Err(Error::RowPtrEnd { found: row_ptr[rows], expected: values.len() });
        }
        for r in 0..rows {
            if row_ptr[r] > row_ptr[r + 1] {
                return Err(Error::RowPtrOrder { row: r });
            }
        }
        for r in 0..rows {
            let start = row_ptr[r];
            let end = row_ptr[r + 1];
            let mut previous = None;
            for slot in start..end {
                let col = col_idx[slot];
                if col >= cols {
                    return Err(Error::ColumnBounds { slot, column: col, limit: cols });
                }
                if previous.is_some_and(|p| col <= p) {
                    return Err(Error::ColumnOrder { slot });
                }
                let value = values[slot];
                if !value.is_finite() {
                    return Err(Error::NonFiniteValue { slot });
                }
                if value == 0.0 {
                    return Err(Error::ExplicitZero { slot });
                }
                previous = Some(col);
            }
        }
        Ok(Self { rows, cols, row_ptr, col_idx, values })
    }

    pub fn shape(&self) -> (usize, usize) {
        (self.rows, self.cols)

                entries.push(Entry { row: r, col: self.col_idx[slot], value: self.values[slot] });
            }

    pub fn mul_vector(&self, x: &[f64]) -> Result<Vec<f64>, Error> {
        if x.len() != self.cols {
            return Err(Error::VectorLength { expected: self.cols, actual: x.len() });
        }
        for (index, value) in x.iter().enumerate() {
            if !value.is_finite() {
                return Err(Error::NonFiniteVector { index });
            }
        }
        let mut result = Vec::with_capacity(self.rows);
        for r in 0..self.rows {
            let mut sum = 0.0;
            for slot in self.row_ptr[r]..self.row_ptr[r + 1] {
                let col = self.col_idx[slot];
                let p = self.values[slot] * x[col];
                if !p.is_finite() {
                    return Err(Error::ProductOverflow { row: r, column: col });
                }
                sum += p;
                if !sum.is_finite() {
                    return Err(Error::AccumulationOverflow { row: r, column: col });
                }
            }
            result.push(if sum == 0.0 { 0.0 } else { sum });
        }
        Ok(result)
    }

impl Csr {
    pub fn transpose(&self) -> Csr {
        let mut counts = vec![0usize; self.cols];
        for &c in &self.col_idx {
            counts[c] += 1;
        }
        let mut row_ptr = vec![0usize; self.cols + 1];
        for r in 0..self.cols {
            row_ptr[r + 1] = row_ptr[r] + counts[r];
        }
        let mut next = row_ptr[..self.cols].to_vec();
        let mut col_idx = vec![0usize; self.values.len()];
        let mut values = vec![0.0; self.values.len()];
        for r in 0..self.rows {
            for slot in self.row_ptr[r]..self.row_ptr[r + 1] {
                let c = self.col_idx[slot];
                let dst = next[c];
                next[c] += 1;
                col_idx[dst] = r;
                values[dst] = self.values[slot];
            }
        }
        Csr { rows: self.cols, cols: self.rows, row_ptr, col_idx, values }
    }

    pub fn multiply(&self, rhs: &Csr) -> Result<Csr, Error> {
        if self.cols != rhs.rows {
            return Err(Error::DimensionMismatch { left_cols: self.cols, right_rows: rhs.rows });
        }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for r in 0..self.rows {
            let mut accum: Vec<(usize, f64)> = Vec::new();
            for left_slot in self.row_ptr[r]..self.row_ptr[r + 1] {
                let k = self.col_idx[left_slot];
                for right_slot in rhs.row_ptr[k]..rhs.row_ptr[k + 1] {
                    let c = rhs.col_idx[right_slot];
                    let fields = (r, k, c);
                    let p = self.values[left_slot] * rhs.values[right_slot];
                    if !p.is_finite() {
                        return Err(Error::MultiplyOverflow { row: fields.0, inner: fields.1, column: fields.2 });
                    }
                    let target = if let Some(i) = accum.iter().position(|(col, _)| *col == c) {
                        i
                    } else {
                        accum.push((c, 0.0));
                        accum.len() - 1
                    };
                    accum[target].1 += p;
                    if !accum[target].1.is_finite() {
                        return Err(Error::MultiplySumOverflow { row: r, inner: k, column: c });
                    }
                }
            }
            accum.sort_by_key(|(c, _)| *c);
            for (c, value) in accum {
                if value != 0.0 {
                    col_idx.push(c);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: rhs.cols, row_ptr, col_idx, values })
    }
}

impl Csr {
    pub fn prune(&self, tolerance: f64) -> Result<Csr, Error> {
        if !tolerance.is_finite() || tolerance < 0.0 {
            return Err(Error::Tolerance);
        }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for r in 0..self.rows {
            for slot in self.row_ptr[r]..self.row_ptr[r + 1] {
                let value = self.values[slot];
                if value.abs() > tolerance {
                    col_idx.push(self.col_idx[slot]);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: self.cols, row_ptr, col_idx, values })
    }

    pub fn write_text(&self) -> String {
        let mut out = format!("SM1 {} {} {}\n", self.rows, self.cols, self.values.len());
        for r in 0..self.rows {
            for slot in self.row_ptr[r]..self.row_ptr[r + 1] {
                out.push_str(&format!("{} {} {:016x}\n", r, self.col_idx[slot], self.values[slot].to_bits()));
            }
        }
        out
    }

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        fn fail(line: usize, kind: TextError) -> Error {
            Error::Text { line, kind }
        }
        fn decimal(s: &str) -> Option<usize> {
            if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
                return None;
            }
            if s.len() > 1 && s.as_bytes()[0] == b'0' {
                return None;
            }
            s.parse().ok()
        }
        if !text.ends_with('\n') {
            let line = 1 + text.bytes().filter(|b| *b == b'\n').count();
            return Err(fail(line, TextError::Newline));
        }
        let body = &text[..text.len() - 1];
        let lines: Vec<&str> = body.split('\n').collect();
        let header: Vec<&str> = lines[0].split(' ').collect();
        if header.len() != 4 || header.iter().any(|s| s.is_empty()) || header[0] != "SM1" {
            return Err(fail(1, TextError::Header));
        }
        let rows = decimal(header[1]).ok_or_else(|| fail(1, TextError::Decimal))?;
        let cols = decimal(header[2]).ok_or_else(|| fail(1, TextError::Decimal))?;
        let nnz = decimal(header[3]).ok_or_else(|| fail(1, TextError::Decimal))?;
        if rows > MAX_DIM || cols > MAX_DIM {
            return Err(fail(1, TextError::Dimension));
        }
        if nnz != lines.len() - 1 {
            return Err(fail(1, TextError::Count));
        }

        let mut row_ptr = vec![0usize; rows + 1];
        let mut col_idx = Vec::with_capacity(nnz);
        let mut values = Vec::with_capacity(nnz);
        let mut previous: Option<(usize, usize)> = None;
        let mut current_row = 0usize;

        for (i, line) in lines.iter().enumerate().skip(1) {
            let line_no = i + 1;
            let fields: Vec<&str> = line.split(' ').collect();
            if fields.len() != 3 || fields.iter().any(|s| s.is_empty()) {
                return Err(fail(line_no, TextError::Record));
            }
            let row = decimal(fields[0]).ok_or_else(|| fail(line_no, TextError::Decimal))?;
            let col = decimal(fields[1]).ok_or_else(|| fail(line_no, TextError::Decimal))?;
            let bits = fields[2];
            if bits.len() != 16 || !bits.bytes().all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b)) {
                return Err(fail(line_no, TextError::Hex));
            }
            let raw = u64::from_str_radix(bits, 16).map_err(|_| fail(line_no, TextError::Hex))?;
            let value = f64::from_bits(raw);
            if row >= rows {
                return Err(fail(line_no, TextError::RowBounds));
            }
            if col >= cols {
                return Err(fail(line_no, TextError::ColumnBounds));
            }
            if !value.is_finite() {
                return Err(fail(line_no, TextError::NonFinite));
            }
            if value == 0.0 {
                return Err(fail(line_no, TextError::Zero));
            }
            let coord = (row, col);
            if previous.is_some_and(|p| coord <= p) {
                return Err(fail(line_no, TextError::Order));
            }
            while current_row < row {
                row_ptr[current_row + 1] = col_idx.len();
                current_row += 1;
            }
            col_idx.push(col);
            values.push(value);
            previous = Some(coord);
        }
        while current_row < rows {
            row_ptr[current_row + 1] = col_idx.len();
            current_row += 1;
        }
        Ok(Csr { rows, cols, row_ptr, col_idx, values })
    }
}

    pub fn from_coo(rows: usize, cols: usize, entries: &[Entry]) -> Result<Self, Error> {

        for (i, entry) in entries.iter().enumerate() {
            if entry.row >= rows {
                return Err(Error::Bounds { entry: i, axis: Axis::Row, index: entry.row, limit: rows });

            let mut touched = Vec::<usize>::new();
            for left_slot in self.row_ptr[row]..self.row_ptr[row + 1] {

                        positions[column] = index;
                        sums.push(0.0);
                        touched.push(column);
                        index
                    } else {
                        pos
                    };
                    sums[acc_index] = sums[acc_index] + product;
                    if !sums[acc_index].is_finite() {
                        return Err(Error::MultiplySumOverflow { row, inner, column });
                    }
                }
            }
            touched.sort_unstable();
            for column in touched {
                let value = sums[positions[column]];
                if value != 0.0 {
                    col_idx.push(column);
                    values.push(value);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: rhs.cols, row_ptr, col_idx, values })
    }

    pub fn prune(&self, tolerance: f64) -> Result<Csr, Error> {
        if !tolerance.is_finite() || tolerance < 0.0 {
            return Err(Error::Tolerance);
        }
        let mut row_ptr = Vec::with_capacity(self.rows + 1);
        let mut col_idx = Vec::new();
        let mut values = Vec::new();
        row_ptr.push(0);
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                if self.values[slot].abs() > tolerance {
                    col_idx.push(self.col_idx[slot]);
                    values.push(self.values[slot]);
                }
            }
            row_ptr.push(values.len());
        }
        Ok(Csr { rows: self.rows, cols: self.cols, row_ptr, col_idx, values })
    }

    pub fn write_text(&self) -> String {
        let mut text = format!("SM1 {} {} {}\n", self.rows, self.cols, self.values.len());
        for row in 0..self.rows {
            for slot in self.row_ptr[row]..self.row_ptr[row + 1] {
                text.push_str(&format!("{} {} {:016x}\n", row, self.col_idx[slot], self.values[slot].to_bits()));
            }
        }
        text
    }

    pub fn parse_text(text: &str) -> Result<Csr, Error> {
        fn error(line: usize, kind: TextError) -> Error {
            Error::Text { line, kind }
        }
        fn parse_decimal(token: &str) -> Option<usize> {
            let bytes = token.as_bytes();
            if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) || (bytes.len() > 1 && bytes[0] == b'0') {
                return None;
            }
            token.parse().ok()
        }
        fn split_exact(line: &str, n: usize) -> Option<Vec<&str>> {
            let fields: Vec<_> = line.split(' ').collect();
            if fields.len() == n && fields.iter().all(|field| !field.is_empty()) {
                Some(fields)
            } else {
                None
            }
        }

        if !text.ends_with('\n') {
            return Err(error(1 + text.bytes().filter(|&byte| byte == b'\n').count(), TextError::Newline));
        }
        let lines: Vec<&str> = text[..text.len() - 1].split('\n').collect();
        let header = split_exact(lines[0], 4).ok_or_else(|| error(1, TextError::Header))?;
        if header[0] != "SM1" {
            return Err(error(1, TextError::Header));
        }
        let rows = parse_decimal(header[1]).ok_or_else(|| error(1, TextError::Decimal))?;
        let cols = parse_decimal(header[2]).ok_or_else(|| error(1, TextError::Decimal))?;
        let count = parse_decimal(header[3]).ok_or_else(|| error(1, TextError::Decimal))?;
        if rows > MAX_DIM || cols > MAX_DIM {
            return Err(error(1, TextError::Dimension));
        }
        if count != lines.len() - 1 {
            return Err(error(1, TextError::Count));
        }

        let mut records: Vec<(usize, usize, f64)> = Vec::with_capacity(count);
        for line_index in 1..lines.len() {
            let line_number = line_index + 1;
            let fields = split_exact(lines[line_index], 3).ok_or_else(|| error(line_number, TextError::Record))?;
            let row = parse_decimal(fields[0]).ok_or_else(|| error(line_number, TextError::Decimal))?;
            let column = parse_decimal(fields[1]).ok_or_else(|| error(line_number, TextError::Decimal))?;
            let token = fields[2].as_bytes();
            if token.len() != 16 || !token.iter().all(|&b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b)) {
                return Err(error(line_number, TextError::Hex));
            }
            let bits = u64::from_str_radix(fields[2], 16).map_err(|_| error(line_number, TextError::Hex))?;
            let value = f64::from_bits(bits);
            if row >= rows {
                return Err(error(line_number, TextError::RowBounds));
            }
            if column >= cols {
                return Err(error(line_number, TextError::ColumnBounds));
            }
            if !value.is_finite() {
                return Err(error(line_number, TextError::NonFinite));
            }
            if value == 0.0 {
                return Err(error(line_number, TextError::Zero));
            }
            if records.last().is_some_and(|&(old_row, old_col, _)| (row, column) <= (old_row, old_col)) {
                return Err(error(line_number, TextError::Order));
            }
            records.push((row, column, value));
        }

        let mut row_ptr = vec![0usize; rows + 1];
        for &(row, _, _) in &records {
            row_ptr[row + 1] += 1;
        }
        for row in 0..rows {
            row_ptr[row + 1] += row_ptr[row];
        }
        let mut col_idx = Vec::with_capacity(records.len());
        let mut values = Vec::with_capacity(records.len());
        for (_, column, value) in records {
            col_idx.push(column);
            values.push(value);
        }
        Ok(Csr { rows, cols, row_ptr, col_idx, values })
    }
}
