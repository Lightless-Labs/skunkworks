#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
enum Phase {
    Waiting {
        ready_at: u64,
        attempt: u8,
        remaining: Option<u64>,
    },
    Running {
        attempt: u8,
        finish_at: u64,
        started_at: u64,
    },
    Succeeded,
    Failed,
    Blocked,
}

struct Entry {
    job: Job,
    phase: Phase,
}

pub struct Pool {
    entries: Vec<Entry>,
    by_id: std::collections::HashMap<String, usize>,
    deps: Vec<Vec<usize>>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    initialized: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }
        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if tag.is_empty()
                    || tag.len() > 8
                    || !tag
                        .bytes()
                        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
                {
                    return Err(Error::BadTag(i));
                }
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut deps = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (k, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling.as_str()) {
                    return Err(Error::DuplicateDependency(i, k));
                }
                if let Some(&d) = by_id.get(spelling) {
                    deps[i].push(d);
                } else {
                    return Err(Error::MissingDependency(i, k));
                }
            }
        }

        let mut indegree = vec![0usize; jobs.len()];
        let mut reverse = vec![Vec::new(); jobs.len()];
        for (i, ds) in deps.iter().enumerate() {
            indegree[i] = ds.len();
            for &d in ds {
                reverse[d].push(i);
            }
        }
        let mut queue: std::collections::VecDeque<usize> = indegree
            .iter()
            .enumerate()
            .filter_map(|(i, &n)| if n == 0 { Some(i) } else { None })
            .collect();
        let mut visited = 0usize;
        while let Some(i) = queue.pop_front() {
            visited += 1;
            for &child in &reverse[i] {
                indegree[child] -= 1;
                if indegree[child] == 0 {
                    queue.push_back(child);
                }
            }
        }
        if visited != jobs.len() {
            return Err(Error::Cycle);
        }

        let entries = jobs
            .into_iter()
            .map(|job| Entry {
                phase: Phase::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                    remaining: None,
                },
                job,
            })
            .collect();
        Ok(Self {
            entries,
            by_id,
            deps,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            initialized: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &i = self.by_id.get(id).ok_or(Error::Unknown)?;
        match self.entries[i].phase {
            Phase::Waiting {
                ready_at, attempt, ..
            } => Ok(State::Waiting { ready_at, attempt }),
            Phase::Running {
                attempt, finish_at, ..
            } => Ok(State::Running { attempt, finish_at }),
            Phase::Succeeded => Ok(State::Succeeded),
            Phase::Failed => Ok(State::Failed),
            Phase::Blocked => Ok(State::Blocked),
        }
    }

    fn run_count(&self) -> usize {
        self.entries
            .iter()
            .filter(|e| matches!(e.phase, Phase::Running { .. }))
            .count()
    }

    fn tag_busy(&self, candidate: usize) -> bool {
        self.entries.iter().enumerate().any(|(i, e)| {
            i != candidate
                && matches!(e.phase, Phase::Running { .. })
                && self.entries[candidate].job.tag.is_some()
                && e.job.tag == self.entries[candidate].job.tag
        })
    }

    fn start(&mut self, i: usize, now: u64, lines: &mut Vec<String>) {
        let (attempt, duration) = match self.entries[i].phase.clone() {
            Phase::Waiting {
                attempt, remaining, ..
            } => match remaining {
                Some(d) => (attempt, d),
                None => (attempt + 1, self.entries[i].job.duration),
            },
            _ => return,
        };
        let finish_at = now.saturating_add(duration);
        self.entries[i].phase = Phase::Running {
            attempt,
            finish_at,
            started_at: now,
        };
        lines.push(format!(
            "{} start {} {} {}",
            now, self.entries[i].job.id, attempt, duration
        ));
    }

    fn process_event(&mut self, now: u64, lines: &mut Vec<String>) {
        let mut finishes: Vec<usize> = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| match e.phase {
                Phase::Running { finish_at, .. } if finish_at == now => Some(i),
                _ => None,
            })
            .collect();
        finishes.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

        for i in finishes {
            let attempt = match self.entries[i].phase {
                Phase::Running { attempt, .. } => attempt,
                _ => 0,
            };
            if attempt > self.entries[i].job.failures {
                lines.push(format!(
                    "{} finish {} {} ok",
                    now, self.entries[i].job.id, attempt
                ));
                self.entries[i].phase = Phase::Succeeded;
            } else {
                lines.push(format!(
                    "{} finish {} {} fail",
                    now, self.entries[i].job.id, attempt
                ));
                if attempt >= self.entries[i].job.max_attempts {
                    self.entries[i].phase = Phase::Failed;
                } else {
                    let mut delay = self.retry_base;
                    for _ in 1..attempt {
                        delay = delay.saturating_mul(2).min(self.retry_cap);
                    }
                    let ready = now.saturating_add(delay);
                    self.entries[i].phase = Phase::Waiting {
                        ready_at: ready,
                        attempt,
                        remaining: None,
                    };
                    lines.push(format!(
                        "{} retry {} {} {}",
                        now, self.entries[i].job.id, attempt, ready
                    ));
                }
            }
        }

        loop {
            let mut newly = Vec::new();
            for i in 0..self.entries.len() {
                if let Phase::Waiting {
                    ready_at,
                    attempt,
                    remaining,
                } = self.entries[i].phase
                {
                    if self.deps[i]
                        .iter()
                        .any(|&d| matches!(self.entries[d].phase, Phase::Failed | Phase::Blocked))
                    {
                        newly.push((i, ready_at, attempt, remaining));
                    }
                }
            }
            if newly.is_empty() {
                break;
            }
            newly.sort_by(|a, b| self.entries[a.0].job.id.cmp(&self.entries[b.0].job.id));
            for (i, _, _, _) in newly {
                let dep = self.deps[i]
                    .iter()
                    .filter(|&&d| {
                        matches!(self.entries[d].phase, Phase::Failed | Phase::Blocked)
                    })
                    .min_by(|&&a, &&b| self.entries[a].job.id.cmp(&self.entries[b].job.id))
                    .copied();
                if let Some(d) = dep {
                    lines.push(format!(
                        "{} skip {} {}",
                        now, self.entries[i].job.id, self.entries[d].job.id
                    ));
                }
                self.entries[i].phase = Phase::Blocked;
            }
        }

        loop {
            let mut candidates: Vec<usize> = (0..self.entries.len())
                .filter(|&i| {
                    matches!(
                        self.entries[i].phase,
                        Phase::Waiting { ready_at, .. } if ready_at <= now
                    ) && self.deps[i]
                        .iter()
                        .all(|&d| matches!(self.entries[d].phase, Phase::Succeeded))
                })
                .collect();
            candidates.sort_by(|&a, &b| {
                let effective = |i: usize| {
                    let base = self.entries[i].job.priority as u64;
                    if self.aging_ms == 0 {
                        base
                    } else {
                        let ready = match self.entries[i].phase {
                            Phase::Waiting { ready_at, .. } => ready_at,
                            _ => now,
                        };
                        (base + (now - ready) / self.aging_ms).min(9)
                    }
                };
                effective(b)
                    .cmp(&effective(a))
                    .then_with(|| a.cmp(&b))
            });

            let mut choice = None;
            for candidate in candidates {
                let enough_slots = self.run_count() < self.capacity;
                let no_tag_conflict = !self.tag_busy(candidate);
                if enough_slots && no_tag_conflict {
                    choice = Some((candidate, None));
                    break;
                }

                let mut victims: Vec<usize> = self
                    .entries
                    .iter()
                    .enumerate()
                    .filter_map(|(i, e)| match e.phase {
                        Phase::Running { started_at, .. }
                            if self.entries[candidate].job.priority
                                >= e.job.priority.saturating_add(2) =>
                        {
                            Some((i, started_at))
                        }
                        _ => None,
                    })
                    .map(|x| x.0)
                    .collect();
                victims.sort_by(|&a, &b| {
                    self.entries[a]
                        .job
                        .priority
                        .cmp(&self.entries[b].job.priority)
                        .then_with(|| {
                            let sa = match self.entries[a].phase {
                                Phase::Running { started_at, .. } => started_at,
                                _ => 0,
                            };
                            let sb = match self.entries[b].phase {
                                Phase::Running { started_at, .. } => started_at,
                                _ => 0,
                            };
                            sb.cmp(&sa)
                        })
                        .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
                });

                for victim in victims {
                    let conflict_after = self.entries[candidate].job.tag.as_ref().is_some_and(|tag| {
                        self.entries.iter().enumerate().any(|(i, e)| {
                            i != victim
                                && matches!(e.phase, Phase::Running { .. })
                                && e.job.tag.as_ref() == Some(tag)
                        })
                    });
                    if conflict_after {
                        continue;
                    }
                    if self.run_count() - 1 >= self.capacity {
                        continue;
                    }
                    choice = Some((candidate, Some(victim)));
                    break;
                }
                if choice.is_some() {
                    break;
                }
            }

            let Some((candidate, victim)) = choice else {
                break;
            };
            if let Some(victim) = victim {
                let (attempt, finish_at) = match self.entries[victim].phase {
                    Phase::Running {
                        attempt, finish_at, ..
                    } => (attempt, finish_at),
                    _ => continue,
                };
                let remaining = finish_at - now;
                self.entries[victim].phase = Phase::Waiting {
                    ready_at: now,
                    attempt,
                    remaining: Some(remaining),
                };
                lines.push(format!(
                    "{} preempt {} by {}",
                    now, self.entries[victim].job.id, self.entries[candidate].job.id
                ));
            }
            self.start(candidate, now, lines);
        }
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.initialized && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if self.initialized && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut next_allowed = if self.initialized {
            self.horizon.saturating_add(1)
        } else {
            0
        };
        loop {
            let mut next_event = None;
            for e in &self.entries {
                let time = match e.phase {
                    Phase::Running { finish_at, .. } => Some(finish_at),
                    Phase::Waiting { ready_at, .. } => Some(ready_at),
                    _ => None,
                };
                if let Some(time) = time {
                    if time >= next_allowed {
                        next_event = Some(next_event.map_or(time, |old: u64| old.min(time)));
                    }
                }
            }
            let now = if !self.initialized && next_allowed == 0 {
                0
            } else if let Some(t) = next_event {
                t
            } else {
                break;
            };
            if now > until {
                break;
            }
            self.process_event(now, &mut lines);
            self.initialized = true;
            next_allowed = now.saturating_add(1);
        }
        self.horizon = until;
        self.initialized = true;
        Ok(lines)
    }
}

pub struct Pool {

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if let Some(tag) = &job.tag {
                if tag.is_empty()
                    || tag.len() > 8
                    || !tag
                        .bytes()
                        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
                {
                    return Err(Error::BadTag(i));
                }
            }
            if by_id.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut deps = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (k, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling.as_str()) {
                    return Err(Error::DuplicateDependency(i, k));
                }
                match by_id.get(spelling) {
                    Some(&idx) => deps[i].push(idx),
                    None => return Err(Error::MissingDependency(i, k)),
                }
            }
        }

        fn visit(i: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[i] == 1 {
                return false;
            }
            if marks[i] == 2 {
                return true;
            }
            marks[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, marks) {
                    return false;
                }
            }
            marks[i] = 2;
            true
        }
        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if !visit(i, &deps, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let entries = jobs
            .into_iter()
            .map(|job| Entry {
                phase: Phase::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                    remaining: None,
                },
                job,
            })
            .collect();

        Ok(Self {
            entries,
            by_id,
            deps,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            initialized: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &i = self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(match self.entries[i].phase {
            Phase::Waiting { ready_at, attempt, .. } => State::Waiting { ready_at, attempt },
            Phase::Running { attempt, finish_at, .. } => State::Running { attempt, finish_at },
            Phase::Succeeded => State::Succeeded,
            Phase::Failed => State::Failed,
            Phase::Blocked => State::Blocked,
        })
    }

    fn id_cmp(&self, a: usize, b: usize) -> std::cmp::Ordering {
        self.entries[a].job.id.as_bytes().cmp(self.entries[b].job.id.as_bytes())
    }

    fn deps_succeeded(&self, i: usize) -> bool {
        self.deps[i]
            .iter()
            .all(|&d| matches!(self.entries[d].phase, Phase::Succeeded))
    }

    fn occupied_tag(&self, tag: &str) -> bool {
        self.entries.iter().any(|e| {
            matches!(e.phase, Phase::Running { .. }) && e.job.tag.as_deref() == Some(tag)
        })
    }

    fn running_count(&self) -> usize {
        self.entries
            .iter()
            .filter(|e| matches!(e.phase, Phase::Running { .. }))
            .count()
    }

    fn retry_ready(&self, t: u64, attempt: u8) -> u64 {
        let mut delay = self.retry_base;
        for _ in 1..attempt {
            delay = delay.saturating_mul(2).min(self.retry_cap);
        }
        t.saturating_add(delay)
    }

    fn start(&mut self, i: usize, t: u64, lines: &mut Vec<String>) {
        let (attempt, duration) = match self.entries[i].phase.clone() {
            Phase::Waiting { attempt, remaining, .. } => {
                if let Some(rem) = remaining {
                    (attempt, rem)
                } else {
                    (attempt + 1, self.entries[i].job.duration)
                }
            }
            _ => return,
        };
        let finish_at = t.saturating_add(duration);
        self.entries[i].phase = Phase::Running {
            attempt,
            finish_at,
            started_at: t,
        };
        lines.push(format!(
            "{} start {} {} {}",
            t, self.entries[i].job.id, attempt, duration
        ));
    }

    fn process_time(&mut self, t: u64, lines: &mut Vec<String>) {
        let mut completed: Vec<usize> = self
            .entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| match e.phase {
                Phase::Running { finish_at, .. } if finish_at == t => Some(i),
                _ => None,
            })
            .collect();
        completed.sort_by(|&a, &b| self.id_cmp(a, b));

        for i in completed {
            let (attempt, _) = match self.entries[i].phase {
                Phase::Running { attempt, finish_at, .. } => (attempt, finish_at),
                _ => continue,
            };
            let ok = attempt > self.entries[i].job.failures;
            lines.push(format!(
                "{} finish {} {} {}",
                t,
                self.entries[i].job.id,
                attempt,
                if ok { "ok" } else { "fail" }
            ));
            if ok {
                self.entries[i].phase = Phase::Succeeded;
            } else if attempt >= self.entries[i].job.max_attempts {
                self.entries[i].phase = Phase::Failed;
            } else {
                let ready = self.retry_ready(t, attempt);
                self.entries[i].phase = Phase::Waiting {
                    ready_at: ready,
                    attempt,
                    remaining: None,
                };
                lines.push(format!(
                    "{} retry {} {} {}",
                    t, self.entries[i].job.id, attempt, ready
                ));
            }
        }

        loop {
            let mut newly = Vec::new();
            for i in 0..self.entries.len() {
                if let Phase::Waiting { ready_at, attempt, remaining } = self.entries[i].phase {
                    if self.deps[i].iter().any(|&d| {
                        matches!(self.entries[d].phase, Phase::Failed | Phase::Blocked)
                    }) {
                        newly.push((i, ready_at, attempt, remaining));
                    }
                }
            }
            if newly.is_empty() {
                break;
            }
            newly.sort_by(|a, b| self.id_cmp(a.0, b.0));
            for (i, _, _, _) in &newly {
                let dependency = self.deps[*i]
                    .iter()
                    .filter(|&&d| {
                        matches!(self.entries[d].phase, Phase::Failed | Phase::Blocked)
                    })
                    .min_by(|&&a, &&b| self.id_cmp(a, b))
                    .copied();
                if let Some(d) = dependency {
                    lines.push(format!(
                        "{} skip {} {}",
                        t, self.entries[*i].job.id, self.entries[d].job.id
                    ));
                }
                self.entries[*i].phase = Phase::Blocked;
            }
        }

        loop {
            let mut candidates: Vec<usize> = (0..self.entries.len())
                .filter(|&i| {
                    matches!(
                        self.entries[i].phase,
                        Phase::Waiting { ready_at, .. } if ready_at <= t
                    ) && self.deps_succeeded(i)
                })
                .collect();
            candidates.sort_by(|&a, &b| {
                let priority = |i: usize| -> u8 {
                    let base = self.entries[i].job.priority;
                    if self.aging_ms == 0 {
                        base
                    } else {
                        let ready_at = match self.entries[i].phase {
                            Phase::Waiting { ready_at, .. } => ready_at,
                            _ => t,
                        };
                        let age = (t - ready_at) / self.aging_ms;
                        (base as u64 + age).min(9) as u8
                    }
                };
                priority(b)
                    .cmp(&priority(a))
                    .then_with(|| a.cmp(&b))
            });

            let mut started = false;
            for candidate in candidates {
                let tag = self.entries[candidate].job.tag.clone();
                let slots_free = self.running_count() < self.capacity;
                let tag_free = tag
                    .as_deref()
                    .map(|tag| !self.occupied_tag(tag))
                    .unwrap_or(true);
                if slots_free && tag_free {
                    self.start(candidate, t, lines);
                    started = true;
                    break;
                }

                let mut victims: Vec<usize> = self
                    .entries
                    .iter()
                    .enumerate()
                    .filter_map(|(i, e)| match e.phase {
                        Phase::Running { started_at, .. }
                            if self.entries[candidate].job.priority
                                >= e.job.priority.saturating_add(2) =>
                        {
                            Some((i, started_at))
                        }
                        _ => None,
                    })
                    .map(|x| x.0)
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.entries[a]
                        .job
                        .priority
                        .cmp(&self.entries[b].job.priority)
                        .then_with(|| {
                            let sa = match self.entries[a].phase {
                                Phase::Running { started_at, .. } => started_at,
                                _ => 0,
                            };
                            let sb = match self.entries[b].phase {
                                Phase::Running { started_at, .. } => started_at,
                                _ => 0,
                            };
                            sb.cmp(&sa)
                        })
                        .then_with(|| self.id_cmp(b, a))
                });

                for victim in victims {
                    let victim_tag = self.entries[victim].job.tag.clone();
                    let tag_after = tag.as_deref().map_or(true, |wanted| {
                        victim_tag.as_deref() == Some(wanted) || !self.occupied_tag(wanted)
                    });
                    let slot_after = self.running_count() - 1 < self.capacity;
                    if !tag_after || !slot_after {
                        continue;
                    }

                    let (attempt, finish_at) = match self.entries[victim].phase {
                        Phase::Running { attempt, finish_at, .. } => (attempt, finish_at),
                        _ => continue,
                    };
                    let remaining = finish_at - t;
                    self.entries[victim].phase = Phase::Waiting {
                        ready_at: t,
                        attempt,
                        remaining: Some(remaining),
                    };
                    lines.push(format!(
                        "{} preempt {} by {}",
                        t, self.entries[victim].job.id, self.entries[candidate].job.id
                    ));
                    self.start(candidate, t, lines);
                    started = true;
                    break;
                }
                if started {
                    break;
                }
            }
            if !started {
                break;
            }
        }
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.initialized && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if self.initialized && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut next = if self.initialized {
            self.horizon.checked_add(1).unwrap_or(u64::MAX)
        } else {
            0
        };

        loop {
            let mut event_time: Option<u64> = None;
            for entry in &self.entries {
                match entry.phase {
                    Phase::Running { finish_at, .. } if finish_at >= next => {
                        event_time = Some(event_time.map_or(finish_at, |x| x.min(finish_at)));
                    }
                    Phase::Waiting { ready_at, .. } if ready_at >= next => {
                        event_time = Some(event_time.map_or(ready_at, |x| x.min(ready_at)));
                    }
                    _ => {}
                }
            }
            let t = if !self.initialized && next == 0 {
                Some(0)
            } else {
                event_time
            };
            let Some(t) = t else { break };
            if t > until {
                break;
            }
            self.process_time(t, &mut lines);
            next = t.saturating_add(1);
            self.initialized = true;
        }

        self.horizon = until;
        self.initialized = true;
        Ok(lines)
    }
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|s| !valid_tag(s)) {
                return Err(Error::BadTag(i));
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (k, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep) {
                    return Err(Error::DuplicateDependency(i, k));
                }
                if !by_id.contains_key(dep) {
                    return Err(Error::MissingDependency(i, k));
                }
            }
        }

        let mut indegree = vec![0usize; jobs.len()];
        let mut outgoing = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            indegree[i] = job.dependencies.len();
            for dep in &job.dependencies {
                outgoing[by_id[dep]].push(i);
            }
        }
        let mut queue: std::collections::VecDeque<usize> = indegree
            .iter()
            .enumerate()
            .filter_map(|(i, &d)| (d == 0).then_some(i))
            .collect();
        let mut visited = 0;
        while let Some(i) = queue.pop_front() {
            visited += 1;
            for &next in &outgoing[i] {
                indegree[next] -= 1;
                if indegree[next] == 0 {
                    queue.push_back(next);
                }
            }
        }
        if visited != jobs.len() {
            return Err(Error::Cycle);
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(input_order, job)| Entry {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                input_order,
                remaining: None,
                segment_started: 0,
            })
            .collect();

        Ok(Self {
            entries,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &i = self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(self.entries[i].state.clone())
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if self.has_run && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut now = if self.has_run {
            self.next_event_after(self.horizon, until)
        } else {
            Some(0)
        };

        while let Some(t) = now {
            if t > until {
                break;
            }
            self.process_time(t, &mut lines);
            now = self.next_event_after(t, until);
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    fn next_event_after(&self, after: u64, until: u64) -> Option<u64> {
        let mut next = None;
        let mut consider = |t: u64| {
            if t > after && t <= until {
                next = Some(next.map_or(t, |old: u64| old.min(t)));
            }
        };
        for entry in &self.entries {
            match entry.state {
                State::Waiting { ready_at, .. } => consider(ready_at),
                State::Running { finish_at, .. } => consider(finish_at),
                _ => {}
            }
        }
        next
    }

    fn process_time(&mut self, t: u64, lines: &mut Vec<String>) {
        let mut completed: Vec<usize> = self.entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| match e.state {
                State::Running { finish_at, .. } if finish_at == t => Some(i),
                _ => None,
            })
            .collect();
        completed.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

        for i in completed {
            let attempt = match self.entries[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            if attempt > self.entries[i].job.failures {
                self.entries[i].state = State::Succeeded;
                lines.push(format!("{t} finish {} {attempt} ok", self.entries[i].job.id));
            } else {
                lines.push(format!("{t} finish {} {attempt} fail", self.entries[i].job.id));
                if attempt >= self.entries[i].job.max_attempts {
                    self.entries[i].state = State::Failed;
                } else {
                    let shift = (attempt - 1) as u32;
                    let delay = self
                        .retry_base
                        .checked_shl(shift)
                        .unwrap_or(u64::MAX)
                        .min(self.retry_cap);
                    let ready = t.saturating_add(delay);
                    self.entries[i].state = State::Waiting {
                        ready_at: ready,
                        attempt,
                    };
                    self.entries[i].remaining = None;
                    lines.push(format!("{t} retry {} {attempt} {ready}", self.entries[i].job.id));
                }
            }
        }

        let mut newly_blocked = Vec::new();
        loop {
            let mut changed = false;
            for i in 0..self.entries.len() {
                if !matches!(self.entries[i].state, State::Waiting { .. }) {
                    continue;
                }
                let blocked_by = self.entries[i]
                    .job
                    .dependencies
                    .iter()
                    .find(|dep| {
                        let di = self.by_id[*dep];
                        matches!(
                            self.entries[di].state,
                            State::Failed | State::Blocked
                        )
                    })
                    .cloned();
                if let Some(dep) = blocked_by {
                    self.entries[i].state = State::Blocked;
                    newly_blocked.push((i, dep));
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
        newly_blocked.sort_by(|(a, _), (b, _)| self.entries[*a].job.id.cmp(&self.entries[*b].job.id));
        for (i, dep) in newly_blocked {
            lines.push(format!("{t} skip {} {dep}", self.entries[i].job.id));
        }

        self.schedule(t, lines);
    }

    fn schedule(&mut self, t: u64, lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = self.entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| match e.state {
                    State::Waiting { ready_at, .. }
                        if ready_at <= t
                            && e.job.dependencies.iter().all(|dep| {
                                matches!(self.entries[self.by_id[dep]].state, State::Succeeded)
                            }) =>
                    {
                        Some(i)
                    }
                    _ => None,
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, t);
                let pb = self.effective_priority(b, t);
                pb.cmp(&pa).then_with(|| {
                    self.entries[a].input_order.cmp(&self.entries[b].input_order)
                })
            });

            let mut made_start = false;
            for candidate in candidates {
                if self.can_start_direct(candidate) {
                    self.start(candidate, t, lines);
                    made_start = true;
                    break;
                }

                let victims = self.possible_victims(candidate);
                if let Some(victim) = victims.first().copied() {
                    let victim_id = self.entries[victim].job.id.clone();
                    let candidate_id = self.entries[candidate].job.id.clone();
                    let finish = match self.entries[victim].state {
                        State::Running { finish_at, .. } => finish_at,
                        _ => continue,
                    };
                    let remaining = finish - t;
                    let attempt = match self.entries[victim].state {
                        State::Running { attempt, .. } => attempt,
                        _ => continue,
                    };
                    self.entries[victim].state = State::Waiting {
                        ready_at: t,
                        attempt,
                    };
                    self.entries[victim].remaining = Some(remaining);
                    lines.push(format!("{t} preempt {victim_id} by {candidate_id}"));
                    self.start(candidate, t, lines);
                    made_start = true;
                    break;
                }
            }
            if !made_start {
                break;
            }
        }
    }

    fn effective_priority(&self, i: usize, t: u64) -> u8 {
        let ready_at = match self.entries[i].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => t,
        };
        let base = self.entries[i].job.priority;
        if self.aging_ms == 0 {
            base
        } else {
            let age = (t - ready_at) / self.aging_ms;
            (base as u64 + age).min(9) as u8
        }
    }

    fn can_start_direct(&self, candidate: usize) -> bool {
        let running = self.entries.iter().filter(|e| matches!(e.state, State::Running { .. })).count();
        if running >= self.capacity {
            return false;
        }
        let tag = self.entries[candidate].job.tag.as_ref();
        !self.entries.iter().any(|e| {
            matches!(e.state, State::Running { .. })
                && tag.is_some()
                && e.job.tag.as_ref() == tag
        })
    }

    fn possible_victims(&self, candidate: usize) -> Vec<usize> {
        let candidate_tag = self.entries[candidate].job.tag.as_ref();
        let running: Vec<usize> = self.entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| matches!(e.state, State::Running { .. }).then_some(i))
            .collect();
        let occupied_tag_conflict = candidate_tag.is_some()
            && running.iter().any(|&i| self.entries[i].job.tag.as_ref() == candidate_tag);
        let at_capacity = running.len() >= self.capacity;
        let mut victims: Vec<usize> = running
            .into_iter()
            .filter(|&i| {
                let victim = &self.entries[i];
                candidate_tag.is_some()
                    && victim.job.tag.as_ref() == candidate_tag
                    || at_capacity
            })
            .filter(|&i| {
                self.entries[candidate].job.priority
                    >= self.entries[i].job.priority.saturating_add(2)
            })
            .filter(|&i| {
                let remaining_running = self.entries
                    .iter()
                    .filter(|e| matches!(e.state, State::Running { .. }))
                    .count()
                    - 1;
                let slot_free = remaining_running < self.capacity;
                let tag_free = candidate_tag.is_none()
                    || !self.entries.iter().enumerate().any(|(j, e)| {
                        j != i
                            && matches!(e.state, State::Running { .. })
                            && e.job.tag.as_ref() == candidate_tag
                    });
                slot_free && tag_free
            })
            .collect();

        victims.sort_by(|&a, &b| {
            self.entries[a]
                .job
                .priority
                .cmp(&self.entries[b].job.priority)
                .then_with(|| {
                    let sa = match self.entries[a].state {
                        State::Running { .. } => self.entries[a].segment_started,
                        _ => 0,
                    };
                    let sb = match self.entries[b].state {
                        State::Running { .. } => self.entries[b].segment_started,
                        _ => 0,
                    };
                    sb.cmp(&sa)
                })
                .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
        });
        let _ = occupied_tag_conflict;
        victims
    }

    fn start(&mut self, i: usize, t: u64, lines: &mut Vec<String>) {
        let (attempt, remaining) = match self.entries[i].state {
            State::Waiting { attempt, .. } => {
                if let Some(remaining) = self.entries[i].remaining.take() {
                    (attempt, remaining)
                } else {
                    let next = attempt.saturating_add(1);
                    (next, self.entries[i].job.duration)
                }
            }
            _ => return,
        };
        let finish_at = t.saturating_add(remaining);
        self.entries[i].state = State::Running { attempt, finish_at };
        self.entries[i].segment_started = t;
        lines.push(format!("{t} start {} {attempt} {remaining}", self.entries[i].job.id));
    }
}

fn valid_id(s: &str) -> bool {
    !s.is_empty()

        && s.bytes().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')

    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|s| !valid_tag(s)) {
                return Err(Error::BadTag(i));
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            for (k, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep) {
                    return Err(Error::DuplicateDependency(i, k));
                }
                if !by_id.contains_key(dep) {
                    return Err(Error::MissingDependency(i, k));
                }
            }
        }

        let n = jobs.len();
        let mut color = vec![0u8; n];
        let mut deps: Vec<Vec<usize>> = jobs
            .iter()
            .map(|job| job.dependencies.iter().map(|d| by_id[d]).collect())
            .collect();
        fn visit(i: usize, deps: &[Vec<usize>], color: &mut [u8]) -> bool {
            if color[i] == 1 {
                return false;
            }
            if color[i] == 2 {
                return true;
            }
            color[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, color) {
                    return false;
                }
            }
            color[i] = 2;
            true
        }
        for i in 0..n {
            if !visit(i, &deps, &mut color) {
                return Err(Error::Cycle);
            }
        }

        let entries = jobs
            .into_iter()
            .map(|job| Entry {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                job,
                remaining: None,
                segment_started: 0,
            })
            .collect();

        Ok(Self {
            entries,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        Ok(self.entries[*self.by_id.get(id).ok_or(Error::Unknown)?].state.clone())
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if self.has_run && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut trace = Vec::new();
        let mut time = if self.has_run {
            self.next_event(self.horizon, until)
        } else {
            Some(0)
        };
        while let Some(t) = time {
            self.process(t, &mut trace);
            time = self.next_event(t, until);
        }
        self.horizon = until;
        self.has_run = true;
        Ok(trace)
    }

    fn next_event(&self, after: u64, limit: u64) -> Option<u64> {
        self.entries
            .iter()
            .filter_map(|e| match e.state {
                State::Waiting { ready_at, .. } if ready_at > after => Some(ready_at),
                State::Running { finish_at, .. } if finish_at > after => Some(finish_at),
                _ => None,
            })
            .filter(|&t| t <= limit)
            .min()
    }

    fn process(&mut self, t: u64, trace: &mut Vec<String>) {
        let mut finished: Vec<usize> = self.entries
            .iter()
            .enumerate()
            .filter_map(|(i, e)| match e.state {
                State::Running { finish_at, .. } if finish_at == t => Some(i),
                _ => None,
            })
            .collect();
        finished.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

        for i in finished {
            let attempt = match self.entries[i].state {
                State::Running { attempt, .. } => attempt,
                _ => continue,
            };
            let id = self.entries[i].job.id.clone();
            if attempt > self.entries[i].job.failures {
                self.entries[i].state = State::Succeeded;
                trace.push(format!("{t} finish {id} {attempt} ok"));
            } else {
                trace.push(format!("{t} finish {id} {attempt} fail"));
                if attempt == self.entries[i].job.max_attempts {
                    self.entries[i].state = State::Failed;
                } else {
                    let shift = (attempt - 1).min(63) as u32;
                    let delay = self.retry_base.saturating_mul(1u64 << shift).min(self.retry_cap);
                    let ready = t.saturating_add(delay);
                    self.entries[i].state = State::Waiting { ready_at: ready, attempt };
                    self.entries[i].remaining = None;
                    trace.push(format!("{t} retry {id} {attempt} {ready}"));
                }
            }
        }

        let mut skipped = Vec::<(usize, String)>::new();
        loop {
            let mut added = false;
            for i in 0..self.entries.len() {
                if !matches!(self.entries[i].state, State::Waiting { .. }) {
                    continue;
                }
                if let Some(dep) = self.entries[i].job.dependencies.iter().find(|dep| {
                    matches!(self.entries[self.by_id[*dep]].state, State::Failed | State::Blocked)
                }) {
                    let dep = dep.clone();
                    self.entries[i].state = State::Blocked;
                    skipped.push((i, dep));
                    added = true;
                }
            }
            if !added {
                break;
            }
        }
        skipped.sort_by(|(a, _), (b, _)| self.entries[*a].job.id.cmp(&self.entries[*b].job.id));
        for (i, dep) in skipped {
            trace.push(format!("{t} skip {} {dep}", self.entries[i].job.id));
        }

        self.schedule(t, trace);
    }

    fn schedule(&mut self, t: u64, trace: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = self.entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| match e.state {
                    State::Waiting { ready_at, .. }
                        if ready_at <= t
                            && e.job.dependencies.iter().all(|dep| {
                                matches!(self.entries[self.by_id[dep]].state, State::Succeeded)
                            }) => Some(i),
                    _ => None,
                })
                .collect();
            candidates.sort_by(|&a, &b| {
                self.priority(a, t)
                    .cmp(&self.priority(b, t))
                    .reverse()
                    .then_with(|| a.cmp(&b))
            });

            let mut acted = false;
            for c in candidates {
                if self.direct(c) {
                    self.start(c, t, trace);
                    acted = true;
                    break;
                }
                let mut choices: Vec<usize> = self.entries
                    .iter()
                    .enumerate()
                    .filter_map(|(i, e)| matches!(e.state, State::Running { .. }).then_some(i))
                    .filter(|&v| {
                        self.entries[c].job.priority >= self.entries[v].job.priority + 2
                            && self.can_free(c, v)
                    })
                    .collect();
                choices.sort_by(|&a, &b| {
                    self.entries[a]
                        .job
                        .priority
                        .cmp(&self.entries[b].job.priority)
                        .then_with(|| {
                            self.entries[b]
                                .segment_started
                                .cmp(&self.entries[a].segment_started)
                        })
                        .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
                });
                if let Some(v) = choices.first().copied() {
                    let finish = match self.entries[v].state {
                        State::Running { finish_at, .. } => finish_at,
                        _ => continue,
                    };
                    let attempt = match self.entries[v].state {
                        State::Running { attempt, .. } => attempt,
                        _ => continue,
                    };
                    let rem = finish - t;
                    let victim_id = self.entries[v].job.id.clone();
                    let candidate_id = self.entries[c].job.id.clone();
                    self.entries[v].state = State::Waiting { ready_at: t, attempt };
                    self.entries[v].remaining = Some(rem);
                    trace.push(format!("{t} preempt {victim_id} by {candidate_id}"));
                    self.start(c, t, trace);
                    acted = true;
                    break;
                }
            }
            if !acted {
                break;
            }
        }
    }

    fn priority(&self, i: usize, t: u64) -> u8 {
        let ready = match self.entries[i].state {
            State::Waiting { ready_at, .. } => ready_at,
            _ => t,
        };
        if self.aging_ms == 0 {
            self.entries[i].job.priority
        } else {
            (self.entries[i].job.priority as u64 + (t - ready) / self.aging_ms).min(9) as u8
        }
    }

    fn direct(&self, c: usize) -> bool {
        let count = self.entries.iter().filter(|e| matches!(e.state, State::Running { .. })).count();
        count < self.capacity
            && (self.entries[c].job.tag.is_none()
                || !self.entries.iter().any(|e| {
                    matches!(e.state, State::Running { .. })
                        && e.job.tag == self.entries[c].job.tag
                }))
    }

    fn can_free(&self, c: usize, v: usize) -> bool {
        let candidate_tag = self.entries[c].job.tag.as_ref();
        let running_count = self.entries.iter().filter(|e| matches!(e.state, State::Running { .. })).count();
        let slot_ok = running_count - 1 < self.capacity;
        let tag_ok = candidate_tag.is_none()
            || !self.entries.iter().enumerate().any(|(i, e)| {
                i != v && matches!(e.state, State::Running { .. }) && e.job.tag.as_ref() == candidate_tag
            });
        slot_ok && tag_ok
    }

    fn start(&mut self, i: usize, t: u64, trace: &mut Vec<String>) {
        let (attempt, duration) = match self.entries[i].state {
            State::Waiting { attempt, .. } => match self.entries[i].remaining.take() {
                Some(rem) => (attempt, rem),
                None => (attempt + 1, self.entries[i].job.duration),
            },
            _ => return,
        };
        self.entries[i].state = State::Running {
            attempt,
            finish_at: t.saturating_add(duration),
        };
        self.entries[i].segment_started = t;
        trace.push(format!("{t} start {} {attempt} {duration}", self.entries[i].job.id));
    }
}

enum Phase {
    Waiting {
        ready_at: u64,
        attempt: u8,
        remaining: Option<u64>,
    },
    Running {
        attempt: u8,
        finish_at: u64,
        started_at: u64,
        remaining: u64,
    },
    Succeeded,
    Failed,
    Blocked,
}

struct Entry {
    job: Job,
    dependencies: Vec<usize>,
    phase: Phase,
}

pub struct Pool {
    entries: Vec<Entry>,
    by_id: std::collections::HashMap<String, usize>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    has_run: bool,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut by_id = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_text(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job.tag.as_ref().is_some_and(|t| !valid_text(t, 1, 8)) {
                return Err(Error::BadTag(i));
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut resolved = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut deps = Vec::with_capacity(job.dependencies.len());
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match by_id.get(dep) {
                    Some(&idx) => deps.push(idx),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            resolved.push(deps);
        }

        let mut indegree: Vec<usize> = resolved.iter().map(Vec::len).collect();
        let mut outgoing = vec![Vec::new(); jobs.len()];
        for (i, deps) in resolved.iter().enumerate() {
            for &d in deps {
                outgoing[d].push(i);
            }
        }
        let mut queue = std::collections::VecDeque::new();
        for (i, &n) in indegree.iter().enumerate() {
            if n == 0 {
                queue.push_back(i);
            }
        }
        let mut visited = 0;
        while let Some(i) = queue.pop_front() {
            visited += 1;
            for &next in &outgoing[i] {
                indegree[next] -= 1;
                if indegree[next] == 0 {
                    queue.push_back(next);
                }
            }
        }
        if visited != jobs.len() {
            return Err(Error::Cycle);
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Entry {
                phase: Phase::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                    remaining: None,
                },
                job,
                dependencies: resolved[i].clone(),
            })
            .collect();

        Ok(Self {
            entries,
            by_id,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            has_run: false,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &i = self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(match self.entries[i].phase {
            Phase::Waiting {
                ready_at, attempt, ..
            } => State::Waiting { ready_at, attempt },
            Phase::Running {
                attempt, finish_at, ..
            } => State::Running { attempt, finish_at },
            Phase::Succeeded => State::Succeeded,
            Phase::Failed => State::Failed,
            Phase::Blocked => State::Blocked,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.has_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }

        let mut lines = Vec::new();
        let mut time = if self.has_run {
            self.horizon.checked_add(1)
        } else {
            Some(0)
        };
        let mut current = time.take();

        while let Some(t) = current {
            if t > until {
                break;
            }

            let mut completions: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| match e.phase {
                    Phase::Running { finish_at, .. } if finish_at == t => Some(i),
                    _ => None,
                })
                .collect();
            completions.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));

            for i in completions {
                let (attempt, _) = match self.entries[i].phase {
                    Phase::Running {
                        attempt, remaining, ..
                    } => (attempt, remaining),
                    _ => unreachable!(),
                };
                let id = self.entries[i].job.id.clone();
                if attempt > self.entries[i].job.failures {
                    self.entries[i].phase = Phase::Succeeded;
                    lines.push(format!("{t} finish {id} {attempt} ok"));
                } else {
                    lines.push(format!("{t} finish {id} {attempt} fail"));
                    if attempt >= self.entries[i].job.max_attempts {
                        self.entries[i].phase = Phase::Failed;
                    } else {
                        let shift = u32::from(attempt.saturating_sub(1));
                        let delay = self
                            .retry_base
                            .checked_shl(shift)
                            .unwrap_or(u64::MAX)
                            .min(self.retry_cap);
                        let ready_at = t.saturating_add(delay);
                        self.entries[i].phase = Phase::Waiting {
                            ready_at,
                            attempt,
                            remaining: None,
                        };
                        lines.push(format!("{t} retry {id} {attempt} {ready_at}"));
                    }
                }
            }

            let mut newly_blocked = Vec::new();
            loop {
                let mut added = Vec::new();
                for (i, entry) in self.entries.iter().enumerate() {
                    if matches!(entry.phase, Phase::Waiting { .. })
                        && entry.dependencies.iter().any(|&d| {
                            matches!(self.entries[d].phase, Phase::Failed | Phase::Blocked)
                        })
                    {
                        added.push(i);
                    }
                }
                if added.is_empty() {
                    break;
                }
                for i in added {
                    self.entries[i].phase = Phase::Blocked;
                    newly_blocked.push(i);
                }
            }
            newly_blocked.sort_by(|&a, &b| self.entries[a].job.id.cmp(&self.entries[b].job.id));
            for i in newly_blocked {
                let dep = self.entries[i]
                    .dependencies
                    .iter()
                    .filter(|&&d| matches!(self.entries[d].phase, Phase::Failed | Phase::Blocked))
                    .min_by_key(|&&d| self.entries[d].job.id.as_str())
                    .map(|&d| self.entries[d].job.id.clone())
                    .unwrap_or_default();
                lines.push(format!("{t} skip {} {dep}", self.entries[i].job.id));
            }

            self.schedule(t, &mut lines);

            let next_wait = self
                .entries
                .iter()
                .filter_map(|e| match e.phase {
                    Phase::Waiting { ready_at, .. } if ready_at > t => Some(ready_at),
                    _ => None,
                })
                .min();
            let next_finish = self
                .entries
                .iter()
                .filter_map(|e| match e.phase {
                    Phase::Running { finish_at, .. } if finish_at > t => Some(finish_at),
                    _ => None,
                })
                .min();
            current = match (next_wait, next_finish) {
                (Some(a), Some(b)) => Some(a.min(b)),
                (Some(a), None) => Some(a),
                (None, Some(b)) => Some(b),
                (None, None) => None,
            };
        }

        self.horizon = until;
        self.has_run = true;
        Ok(lines)
    }

    fn schedule(&mut self, t: u64, lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, e)| {
                    let ready_at = match e.phase {
                        Phase::Waiting { ready_at, .. } if ready_at <= t => ready_at,
                        _ => return None,
                    };
                    if !e.dependencies.iter().all(|&d| matches!(self.entries[d].phase, Phase::Succeeded)) {
                        return None;
                    }
                    let priority = if self.aging_ms == 0 {
                        u64::from(e.job.priority)
                    } else {
                        u64::from(e.job.priority)
                            .saturating_add((t - ready_at) / self.aging_ms)
                            .min(9)
                    };
                    Some((i, priority))
                })
                .collect::<Vec<_>>()
                .into_iter()
                .map(|x| x.0)
                .collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, t);
                let pb = self.effective_priority(b, t);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut started = false;
            for candidate in candidates {
                let tag = self.entries[candidate].job.tag.clone();
                let running: Vec<usize> = self
                    .entries
                    .iter()
                    .enumerate()
                    .filter_map(|(i, e)| matches!(e.phase, Phase::Running { .. }).then_some(i))
                    .collect();

                let slot_free = running.len() < self.capacity;
                let tag_free = tag.as_ref().is_none_or(|wanted| {
                    !running.iter().any(|&i| self.entries[i].job.tag.as_ref() == Some(wanted))
                });

                if slot_free && tag_free {
                    self.start(candidate, t, lines);
                    started = true;
                    break;
                }

                let mut victims: Vec<usize> = running
                    .into_iter()
                    .filter(|&v| {
                        let victim_tag = self.entries[v].job.tag.as_ref();
                        let after_count = self.running_count() - 1;
                        let slot_ok = after_count < self.capacity;
                        let tag_ok = tag.as_ref().is_none_or(|wanted| {
                            victim_tag == Some(wanted)
                                || !self.entries.iter().enumerate().any(|(i, e)| {
                                    i != v
                                        && matches!(e.phase, Phase::Running { .. })
                                        && e.job.tag.as_ref() == Some(wanted)
                                })
                        });
                        slot_ok
                            && tag_ok
                            && self.entries[candidate].job.priority
                                >= self.entries[v].job.priority.saturating_add(2)
                    })
                    .collect();

                victims.sort_by(|&a, &b| {
                    self.entries[a]
                        .job
                        .priority
                        .cmp(&self.entries[b].job.priority)
                        .then_with(|| {
                            let sa = match self.entries[a].phase {
                                Phase::Running { started_at, .. } => started_at,
                                _ => 0,
                            };
                            let sb = match self.entries[b].phase {
                                Phase::Running { started_at, .. } => started_at,
                                _ => 0,
                            };
                            sb.cmp(&sa)
                        })
                        .then_with(|| self.entries[b].job.id.cmp(&self.entries[a].job.id))
                });

                if let Some(victim) = victims.first().copied() {
                    let remaining = match self.entries[victim].phase {
                        Phase::Running { finish_at, .. } => finish_at - t,
                        _ => 0,
                    };
                    let attempt = match self.entries[victim].phase {
                        Phase::Running { attempt, .. } => attempt,
                        _ => 0,
                    };
                    let victim_id = self.entries[victim].job.id.clone();
                    let candidate_id = self.entries[candidate].job.id.clone();
                    lines.push(format!("{t} preempt {victim_id} by {candidate_id}"));
                    self.entries[victim].phase = Phase::Waiting {
                        ready_at: t,
                        attempt,
                        remaining: Some(remaining),
                    };
                    self.start(candidate, t, lines);
                    started = true;
                    break;
                }
            }

            if !started {
                break;
            }
        }
    }

    fn start(&mut self, i: usize, t: u64, lines: &mut Vec<String>) {
        let (attempt, remaining) = match self.entries[i].phase {
            Phase::Waiting {
                attempt, remaining, ..
            } => match remaining {
                Some(r) => (attempt, r),
                None => (attempt.saturating_add(1), self.entries[i].job.duration),
            },
            _ => unreachable!(),
        };
        let finish_at = t.saturating_add(remaining);
        let id = self.entries[i].job.id.clone();
        lines.push(format!("{t} start {id} {attempt} {remaining}"));
        self.entries[i].phase = Phase::Running {
            attempt,
            finish_at,
            started_at: t,
            remaining,
        };
    }

    fn running_count(&self) -> usize {
        self.entries
            .iter()
            .filter(|e| matches!(e.phase, Phase::Running { .. }))
            .count()
    }

    fn effective_priority(&self, i: usize, t: u64) -> u64 {
        let ready_at = match self.entries[i].phase {
            Phase::Waiting { ready_at, .. } => ready_at,
            _ => t,
        };
        if self.aging_ms == 0 {
            u64::from(self.entries[i].job.priority)
        } else {
            (u64::from(self.entries[i].job.priority) + (t - ready_at) / self.aging_ms).min(9)
        }
    }
}

fn valid_text(s: &str, min: usize, max: usize) -> bool {
    let bytes = s.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
enum Phase {
    Waiting { ready: u64, attempt: u8, saved: Option<u64> },
    Running { attempt: u8, finish: u64, since: u64 },
    Succeeded,
    Failed,
    Blocked,
}

struct Node {
    job: Job,
    deps: Vec<usize>,
    phase: Phase,
}

pub struct Pool {
    nodes: Vec<Node>,
    index: std::collections::HashMap<String, usize>,
    capacity: usize,
    aging: u64,
    base: u64,
    cap: u64,
    horizon: u64,
    initialized: bool,
}

impl Pool {
    pub fn new(jobs: Vec<Job>, capacity: usize, aging_ms: u64, retry_base: u64, retry_cap: u64) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity) || retry_base == 0 || retry_base > retry_cap || retry_cap > 1_000_000_000 {
            return Err(Error::Config);
        }

        let mut index = std::collections::HashMap::new();
        for (i, j) in jobs.iter().enumerate() {
            if !valid(&j.id, 1, 16) { return Err(Error::BadId(i)); }
            if j.priority > 9 { return Err(Error::BadPriority(i)); }
            if !(1..=32).contains(&j.max_attempts) { return Err(Error::BadAttempts(i)); }
            if j.release_at > 1_000_000_000 || j.duration == 0 || j.duration > 1_000_000_000 {
                return Err(Error::BadTime(i));
            }
            if j.tag.as_ref().is_some_and(|s| !valid(s, 1, 8)) { return Err(Error::BadTag(i)); }
            if index.contains_key(&j.id) { return Err(Error::Duplicate(i)); }
            index.insert(j.id.clone(), i);
        }

        let mut deps = Vec::new();
        for (i, j) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut row = Vec::new();
            for (k, name) in j.dependencies.iter().enumerate() {
                if !seen.insert(name.as_str()) { return Err(Error::DuplicateDependency(i, k)); }
                row.push(*index.get(name).ok_or(Error::MissingDependency(i, k))?);
            }
            deps.push(row);
        }

        let mut marks = vec![0u8; jobs.len()];
        fn visit(i: usize, deps: &[Vec<usize>], marks: &mut [u8]) -> bool {
            if marks[i] == 1 { return false; }
            if marks[i] == 2 { return true; }
            marks[i] = 1;
            for &d in &deps[i] {
                if !visit(d, deps, marks) { return false; }
            }
            marks[i] = 2;
            true
        }
        for i in 0..jobs.len() {
            if !visit(i, &deps, &mut marks) { return Err(Error::Cycle); }
        }

        let nodes = jobs.into_iter().enumerate().map(|(i, job)| Node {
            phase: Phase::Waiting { ready: job.release_at, attempt: 0, saved: None },
            job,
            deps: deps[i].clone(),
        }).collect();

        Ok(Self { nodes, index, capacity, aging: aging_ms, base: retry_base, cap: retry_cap, horizon: 0, initialized: false })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        let &i = self.index.get(id).ok_or(Error::Unknown)?;
        Ok(match self.nodes[i].phase {
            Phase::Waiting { ready, attempt, .. } => State::Waiting { ready_at: ready, attempt },
            Phase::Running { attempt, finish, .. } => State::Running { attempt, finish_at: finish },
            Phase::Succeeded => State::Succeeded,
            Phase::Failed => State::Failed,
            Phase::Blocked => State::Blocked,
        })
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if self.initialized && until < self.horizon { return Err(Error::TimeReversed); }
        if until > 1_000_000_000 { return Err(Error::TimeRange); }

        let mut trace = Vec::new();
        let mut now = if self.initialized { self.horizon.checked_add(1) } else { Some(0) };

        while let Some(t) = now {
            if t > until { break; }
            let mut finishes: Vec<usize> = self.nodes.iter().enumerate().filter_map(|(i,n)| {
                match n.phase { Phase::Running { finish, .. } if finish == t => Some(i), _ => None }
            }).collect();
            finishes.sort_by(|&a,&b| self.nodes[a].job.id.cmp(&self.nodes[b].job.id));

            for i in finishes {
                let attempt = match self.nodes[i].phase { Phase::Running { attempt, .. } => attempt, _ => 0 };
                let id = self.nodes[i].job.id.clone();
                if attempt > self.nodes[i].job.failures {
                    self.nodes[i].phase = Phase::Succeeded;
                    trace.push(format!("{t} finish {id} {attempt} ok"));
                } else {
                    trace.push(format!("{t} finish {id} {attempt} fail"));
                    if attempt == self.nodes[i].job.max_attempts {
                        self.nodes[i].phase = Phase::Failed;
                    } else {
                        let exp = u32::from(attempt - 1);
                        let wait = self.base.checked_mul(1u64.checked_shl(exp).unwrap_or(u64::MAX)).unwrap_or(u64::MAX).min(self.cap);
                        let ready = t.saturating_add(wait);
                        self.nodes[i].phase = Phase::Waiting { ready, attempt, saved: None };
                        trace.push(format!("{t} retry {id} {attempt} {ready}"));
                    }
                }
            }

            let mut skips = Vec::new();
            loop {
                let add: Vec<usize> = self.nodes.iter().enumerate().filter_map(|(i,n)| {
                    if matches!(n.phase, Phase::Waiting { .. }) && n.deps.iter().any(|&d| matches!(self.nodes[d].phase, Phase::Failed | Phase::Blocked)) { Some(i) } else { None }
                }).collect();
                if add.is_empty() { break; }
                for i in add {
                    self.nodes[i].phase = Phase::Blocked;
                    skips.push(i);
                }
            }
            skips.sort_by(|&a,&b| self.nodes[a].job.id.cmp(&self.nodes[b].job.id));
            for i in skips {
                let dependency = self.nodes[i].deps.iter().filter(|&&d| matches!(self.nodes[d].phase, Phase::Failed | Phase::Blocked))
                    .min_by_key(|&&d| self.nodes[d].job.id.as_str()).map(|&d| self.nodes[d].job.id.clone()).unwrap_or_default();
                trace.push(format!("{t} skip {} {dependency}", self.nodes[i].job.id));
            }

            self.schedule(t, &mut trace);
            let w = self.nodes.iter().filter_map(|n| match n.phase { Phase::Waiting { ready, .. } if ready > t => Some(ready), _ => None }).min();
            let f = self.nodes.iter().filter_map(|n| match n.phase { Phase::Running { finish, .. } if finish > t => Some(finish), _ => None }).min();
            now = match (w,f) { (Some(a),Some(b)) => Some(a.min(b)), (Some(a),None) => Some(a), (None,Some(b)) => Some(b), _ => None };
        }

        self.horizon = until;
        self.initialized = true;
        Ok(trace)
    }

    fn schedule(&mut self, t: u64, out: &mut Vec<String>) {
        loop {
            let mut eligible: Vec<usize> = self.nodes.iter().enumerate().filter_map(|(i,n)| {
                match n.phase {
                    Phase::Waiting { ready, .. } if ready <= t && n.deps.iter().all(|&d| matches!(self.nodes[d].phase, Phase::Succeeded)) => Some(i),
                    _ => None
                }
            }).collect();
            eligible.sort_by(|&a,&b| self.rank(b,t).cmp(&self.rank(a,t)).then_with(|| a.cmp(&b)));

            let mut changed = false;
            for c in eligible {
                let tag = self.nodes[c].job.tag.as_ref();
                let running: Vec<usize> = self.nodes.iter().enumerate().filter_map(|(i,n)| matches!(n.phase, Phase::Running { .. }).then_some(i)).collect();
                let slot = running.len() < self.capacity;
                let resource = tag.is_none_or(|tag| !running.iter().any(|&i| self.nodes[i].job.tag.as_ref() == Some(tag)));
                if slot && resource {
                    self.begin(c,t,out);
                    changed = true;
                    break;
                }

                let mut candidates: Vec<usize> = running.into_iter().filter(|&v| {
                    let after = self.nodes.iter().enumerate().any(|(i,n)| i != v && matches!(n.phase, Phase::Running { .. }) && tag.is_some_and(|x| self.nodes[i].job.tag.as_ref() == Some(x)));
                    self.nodes[c].job.priority >= self.nodes[v].job.priority.saturating_add(2)
                        && self.running_count() - 1 < self.capacity
                        && !after
                }).collect();
                candidates.sort_by(|&a,&b| {
                    self.nodes[a].job.priority.cmp(&self.nodes[b].job.priority)
                        .then_with(|| self.started(b).cmp(&self.started(a)))
                        .then_with(|| self.nodes[b].job.id.cmp(&self.nodes[a].job.id))
                });
                if let Some(v) = candidates.first().copied() {
                    let remain = match self.nodes[v].phase { Phase::Running { finish, .. } => finish-t, _ => 0 };
                    let attempt = match self.nodes[v].phase { Phase::Running { attempt, .. } => attempt, _ => 0 };
                    out.push(format!("{t} preempt {} by {}", self.nodes[v].job.id, self.nodes[c].job.id));
                    self.nodes[v].phase = Phase::Waiting { ready: t, attempt, saved: Some(remain) };
                    self.begin(c,t,out);
                    changed = true;
                    break;
                }
            }
            if !changed { break; }
        }
    }

    fn begin(&mut self, i: usize, t: u64, out: &mut Vec<String>) {
        let (attempt, duration) = match self.nodes[i].phase {
            Phase::Waiting { attempt, saved, .. } => saved.map_or((attempt.saturating_add(1), self.nodes[i].job.duration), |r| (attempt,r)),
            _ => unreachable!(),
        };
        let finish = t.saturating_add(duration);
        out.push(format!("{t} start {} {attempt} {duration}", self.nodes[i].job.id));
        self.nodes[i].phase = Phase::Running { attempt, finish, since: t };
    }

    fn rank(&self, i: usize, t: u64) -> u64 {
        let ready = match self.nodes[i].phase { Phase::Waiting { ready, .. } => ready, _ => t };
        if self.aging == 0 { u64::from(self.nodes[i].job.priority) }
        else { (u64::from(self.nodes[i].job.priority) + (t-ready)/self.aging).min(9) }
    }
    fn started(&self, i: usize) -> u64 {
        match self.nodes[i].phase { Phase::Running { since, .. } => since, _ => 0 }
    }
    fn running_count(&self) -> usize {
        self.nodes.iter().filter(|n| matches!(n.phase, Phase::Running { .. })).count()
    }
}

fn valid(s: &str, min: usize, max: usize) -> bool {
    (min..=max).contains(&s.len()) && s.bytes().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}
