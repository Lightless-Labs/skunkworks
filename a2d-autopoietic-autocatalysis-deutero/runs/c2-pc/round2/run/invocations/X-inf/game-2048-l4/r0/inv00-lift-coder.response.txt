#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
    ScoreOverflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width)
        || !(1..=8).contains(&board.height)
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn moved_line(
    board: &Board,
    direction: Direction,
    line: usize,
) -> Result<(Vec<(usize, u32)>, Vec<Merge>, u64), Error> {
    let (length, indices): (usize, Vec<usize>) = match direction {
        Direction::Left => (
            board.width,
            (0..board.width).map(|x| line * board.width + x).collect(),
        ),
        Direction::Right => (
            board.width,
            (0..board.width)
                .rev()
                .map(|x| line * board.width + x)
                .collect(),
        ),
        Direction::Up => (
            board.height,
            (0..board.height).map(|y| y * board.width + line).collect(),
        ),
        Direction::Down => (
            board.height,
            (0..board.height)
                .rev()
                .map(|y| y * board.width + line)
                .collect(),
        ),
    };
    let _ = length;
    let tiles: Vec<(usize, u32)> = indices
        .iter()
        .filter_map(|&i| (board.cells[i] != 0).then_some((i, board.cells[i])))
        .collect();

    let mut outputs = Vec::new();
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut i = 0;
    while i < tiles.len() {
        if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
            let (first, second) = (tiles[i].0, tiles[i + 1].0);
            let value = tiles[i].1;
            if value == 2_147_483_648 {
                return Err(Error::MergeOverflow { first, second });
            }
            let doubled = value * 2;
            let to = indices[outputs.len()];
            score = score.checked_add(doubled as u64).ok_or(Error::ScoreOverflow)?;
            outputs.push((to, doubled));
            merges.push(Merge {
                from: [first, second],
                to,
                value: doubled,
            });
            i += 2;
        } else {
            outputs.push((indices[outputs.len()], tiles[i].1));
            i += 1;
        }
    }
    Ok((outputs, merges, score))
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };
    for line in 0..line_count {
        let (outputs, line_merges, line_score) = moved_line(board, direction, line)?;
        for (index, value) in outputs {
            cells[index] = value;
        }
        score = score.checked_add(line_score).ok_or(Error::ScoreOverflow)?;
        merges.extend(line_merges);
    }
    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut result = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(m) if m.changed => result.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(result)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Spawn {
    pub index: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SpawnResult {
    pub board: Board,
    pub rng: u64,
    pub spawn: Option<Spawn>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Turn {
    pub board: Board,
    pub rng: u64,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
    pub spawn: Option<Spawn>,
}

fn draw(rng: u64) -> u64 {
    rng.wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let empty: Vec<usize> = board
        .cells
        .iter()
        .enumerate()
        .filter_map(|(i, &value)| (value == 0).then_some(i))
        .collect();
    if empty.is_empty() {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }
    let a = draw(rng);
    let index = empty[(a % empty.len() as u64) as usize];
    let b = draw(a);
    let value = if b % 10 == 0 { 4 } else { 2 };
    let mut result = board.clone();
    result.cells[index] = value;
    Ok(SpawnResult {
        board: result,
        rng: b,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }
    let spawned = spawn_tile(&moved.board, rng)?;
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub board: Board,
    pub score: u64,
    pub rng: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Game {
    current: State,
    undo_state: Option<State>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Command {
    Move(Direction),
    Undo,
    Check,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Event {
    Moved(Turn),
    Undone(bool),
    Checked(bool),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Replay {
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ReplayError {
    pub index: Option<usize>,
    pub error: Error,
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

pub fn new_game(initial: State) -> Result<Game, Error> {
    validate(&initial.board)?;
    Ok(Game {
        current: initial,
        undo_state: None,
    })
}

pub fn snapshot(game: &Game) -> State {
    game.current.clone()
}

pub fn step(game: &mut Game, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(&game.current.board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng: game.current.rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }
    let score = game
        .current
        .score
        .checked_add(moved.score)
        .ok_or(Error::ScoreOverflow)?;
    let spawned = spawn_tile(&moved.board, game.current.rng)?;
    let previous = game.current.clone();
    game.current = State {
        board: spawned.board.clone(),
        score,
        rng: spawned.rng,
    };
    game.undo_state = Some(previous);
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

pub fn undo(game: &mut Game) -> bool {
    if let Some(previous) = game.undo_state.take() {
        game.current = previous;
        true
    } else {
        false
    }
}

pub fn can_undo(game: &Game) -> bool {
    game.undo_state.is_some()
}

pub fn game_over(game: &Game) -> bool {
    legal_moves(&game.current.board)
        .map(|moves| moves.is_empty())
        .unwrap_or(true)
}

pub fn replay(initial: State, commands: &[Command]) -> Result<Replay, ReplayError> {
    let mut game = match new_game(initial.clone()) {
        Ok(game) => game,
        Err(error) => {
            return Err(ReplayError {
                index: None,
                error,
                state: initial,
                can_undo: false,
                events: Vec::new(),
            });
        }
    };
    let mut events = Vec::new();
    for (index, command) in commands.iter().enumerate() {
        match command {
            Command::Move(direction) => match step(&mut game, *direction) {
                Ok(turn) => events.push(Event::Moved(turn)),
                Err(error) => {
                    return Err(ReplayError {
                        index: Some(index),
                        error,
                        state: snapshot(&game),
                        can_undo: can_undo(&game),
                        events,
                    });
                }
            },
            Command::Undo => events.push(Event::Undone(undo(&mut game))),
            Command::Check => events.push(Event::Checked(game_over(&game))),
        }
    }
    Ok(Replay {
        state: snapshot(&game),
        can_undo: can_undo(&game),
        events,
    })
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
    ScoreOverflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if board.width == 0
        || board.width > 8
        || board.height == 0
        || board.height > 8
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }
    for (index, value) in board.cells.iter().copied().enumerate() {
        if value != 0 && (value > 2_147_483_648 || value < 2 || value & (value - 1) != 0) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn scan_line(board: &Board, indices: Vec<usize>) -> Result<(Vec<(usize, u32)>, Vec<Merge>, u64), Error> {
    let mut input = Vec::new();
    for &index in &indices {
        let value = board.cells[index];
        if value != 0 {
            input.push((index, value));
        }
    }

    let mut output = Vec::new();
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut cursor = 0usize;
    while cursor < input.len() {
        let destination = indices[output.len()];
        if cursor + 1 < input.len() && input[cursor].1 == input[cursor + 1].1 {
            let first = input[cursor].0;
            let second = input[cursor + 1].0;
            let value = input[cursor].1;
            if value == 2_147_483_648 {
                return Err(Error::MergeOverflow { first, second });
            }
            let merged = value * 2;
            score = score.checked_add(merged as u64).ok_or(Error::ScoreOverflow)?;
            output.push((destination, merged));
            merges.push(Merge {
                from: [first, second],
                to: destination,
                value: merged,
            });
            cursor += 2;
        } else {
            output.push((destination, input[cursor].1));
            cursor += 1;
        }
    }
    Ok((output, merges, score))
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };
    let mut cells = vec![0; board.cells.len()];
    let mut score = 0u64;
    let mut merges = Vec::new();

    for line in 0..line_count {
        let indices: Vec<usize> = match direction {
            Direction::Left => (0..board.width).map(|x| line * board.width + x).collect(),
            Direction::Right => (0..board.width).rev().map(|x| line * board.width + x).collect(),
            Direction::Up => (0..board.height).map(|y| y * board.width + line).collect(),
            Direction::Down => (0..board.height).rev().map(|y| y * board.width + line).collect(),
        };
        let (output, line_merges, line_score) = scan_line(board, indices)?;
        for (index, value) in output {
            cells[index] = value;
        }
        score = score.checked_add(line_score).ok_or(Error::ScoreOverflow)?;
        merges.extend(line_merges);
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut legal = Vec::new();
    for direction in [Direction::Up, Direction::Left, Direction::Down, Direction::Right] {
        match move_board(board, direction) {
            Ok(result) if result.changed => legal.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(legal)
}

    pub score: u64,

    pub spawn: Option<Spawn>,

fn next_rng(rng: u64) -> u64 {
    rng.wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let count = board.cells.iter().filter(|&&cell| cell == 0).count();
    if count == 0 {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let first_draw = next_rng(rng);
    let rank = (first_draw % count as u64) as usize;
    let index = board
        .cells
        .iter()
        .enumerate()
        .filter(|(_, value)| **value == 0)
        .nth(rank)
        .map(|(index, _)| index)
        .unwrap();
    let second_draw = next_rng(first_draw);
    let value = if second_draw % 10 == 0 { 4 } else { 2 };
    let mut result = board.clone();
    result.cells[index] = value;
    Ok(SpawnResult {
        board: result,
        rng: second_draw,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let result = move_board(board, direction)?;
    if !result.changed {
        return Ok(Turn {
            board: result.board,
            rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }
    let spawned = spawn_tile(&result.board, rng)?;
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: result.score,
        changed: true,
        merges: result.merges,
        spawn: spawned.spawn,
    })
}

pub struct Game {
    current: State,

}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Command {
    Move(Direction),
    Undo,
    Check,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Event {
    Moved(Turn),
    Undone(bool),
    Checked(bool),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Replay {
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ReplayError {
    pub index: Option<usize>,
    pub error: Error,
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

pub fn new_game(initial: State) -> Result<Game, Error> {
    validate(&initial.board)?;
    Ok(Game {
        current: initial,
        saved: None,
    })
}

pub fn snapshot(game: &Game) -> State {
    game.current.clone()
}

pub fn step(game: &mut Game, direction: Direction) -> Result<Turn, Error> {
    let result = move_board(&game.current.board, direction)?;
    if !result.changed {
        return Ok(Turn {
            board: result.board,
            rng: game.current.rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let total_score = game
        .current
        .score
        .checked_add(result.score)
        .ok_or(Error::ScoreOverflow)?;
    let spawned = spawn_tile(&result.board, game.current.rng)?;
    let before = game.current.clone();
    game.current = State {
        board: spawned.board.clone(),
        score: total_score,
        rng: spawned.rng,
    };
    game.saved = Some(before);

    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: result.score,
        changed: true,
        merges: result.merges,
        spawn: spawned.spawn,
    })
}

pub fn undo(game: &mut Game) -> bool {
    match game.saved.take() {
        Some(state) => {
            game.current = state;
            true
        }
        None => false,
    }
}

pub fn can_undo(game: &Game) -> bool {
    game.saved.is_some()
}

pub fn game_over(game: &Game) -> bool {
    match legal_moves(&game.current.board) {
        Ok(moves) => moves.is_empty(),
        Err(_) => true,
    }
}

pub fn replay(initial: State, commands: &[Command]) -> Result<Replay, ReplayError> {
    let mut game = match new_game(initial.clone()) {
        Ok(game) => game,
        Err(error) => {
            return Err(ReplayError {
                index: None,
                error,
                state: initial,
                can_undo: false,
                events: Vec::new(),
            });
        }
    };
    let mut events = Vec::new();
    for (index, command) in commands.iter().enumerate() {
        match command {
            Command::Move(direction) => match step(&mut game, *direction) {
                Ok(turn) => events.push(Event::Moved(turn)),
                Err(error) => {
                    return Err(ReplayError {
                        index: Some(index),
                        error,
                        state: snapshot(&game),
                        can_undo: can_undo(&game),
                        events,
                    });
                }
            },
            Command::Undo => events.push(Event::Undone(undo(&mut game))),
            Command::Check => events.push(Event::Checked(game_over(&game))),
        }
    }
    Ok(Replay {
        state: snapshot(&game),
        can_undo: can_undo(&game),
        events,
    })
}
