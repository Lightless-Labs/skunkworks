    Up,
    Left,
    Down,
    Right,
}

pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
    ScoreOverflow,
}

    pub height: usize,

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Turn {

    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }
    let expected = board.width * board.height;
    if board.cells.len() != expected {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn line_indices(board: &Board, direction: Direction, line: usize) -> Vec<usize> {
    match direction {
        Direction::Left => (0..board.width).map(|x| line * board.width + x).collect(),
        Direction::Right => (0..board.width)
            .rev()
            .map(|x| line * board.width + x)
            .collect(),
        Direction::Up => (0..board.height).map(|y| y * board.width + line).collect(),
        Direction::Down => (0..board.height)
            .rev()
            .map(|y| y * board.width + line)
            .collect(),
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line in 0..line_count {
        let indices = line_indices(board, direction, line);
        let tiles: Vec<(usize, u32)> = indices
            .iter()
            .filter_map(|&index| {
                let value = board.cells[index];
                (value != 0).then_some((index, value))
            })
            .collect();

        let mut output = 0usize;
        let mut i = 0usize;
        while i < tiles.len() {
            let (first, value) = tiles[i];
            if i + 1 < tiles.len() && tiles[i + 1].1 == value {
                let second = tiles[i + 1].0;
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first, second });
                }
                let doubled = value * 2;
                score = score.checked_add(doubled as u64).ok_or(Error::ScoreOverflow)?;
                let to = indices[output];
                cells[to] = doubled;
                merges.push(Merge {
                    from: [first, second],
                    to,
                    value: doubled,
                });
                output += 1;
                i += 2;
            } else {
                cells[indices[output]] = value;
                output += 1;
                i += 1;
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut result = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(moved) if moved.changed => result.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(result)
}

fn next_rng(rng: u64) -> u64 {
    rng.wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let empty: Vec<usize> = board
        .cells
        .iter()
        .enumerate()
        .filter_map(|(i, &value)| (value == 0).then_some(i))
        .collect();

    if empty.is_empty() {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let a = next_rng(rng);
    let index = empty[(a % empty.len() as u64) as usize];
    let b = next_rng(a);
    let value = if b % 10 == 0 { 4 } else { 2 };
    let mut spawned_board = board.clone();
    spawned_board.cells[index] = value;

    Ok(SpawnResult {
        board: spawned_board,
        rng: b,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let spawned = spawn_tile(&moved.board, rng)?;
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

pub fn new_game(initial: State) -> Result<Game, Error> {
    validate(&initial.board)?;
    Ok(Game {
        current: initial,
        previous: None,
    })
}

pub fn snapshot(game: &Game) -> State {
    game.current.clone()
}

pub fn step(game: &mut Game, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(&game.current.board, direction)?;
    let cumulative = game
        .current
        .score
        .checked_add(moved.score)
        .ok_or(Error::ScoreOverflow)?;

    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng: game.current.rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let spawned = spawn_tile(&moved.board, game.current.rng)?;
    let turn = Turn {
        board: spawned.board.clone(),
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    };
    let old = game.current.clone();
    game.current = State {
        board: spawned.board,
        score: cumulative,
        rng: spawned.rng,
    };
    game.previous = Some(old);
    Ok(turn)
}

pub fn undo(game: &mut Game) -> bool {
    if let Some(previous) = game.previous.take() {
        game.current = previous;
        true
    } else {
        false
    }
}

pub fn can_undo(game: &Game) -> bool {
    game.previous.is_some()
}

pub fn game_over(game: &Game) -> bool {
    legal_moves(&game.current.board)
        .map(|moves| moves.is_empty())
        .unwrap_or(false)
}

pub fn replay(initial: State, commands: &[Command]) -> Result<Replay, ReplayError> {
    let mut game = match new_game(initial.clone()) {
        Ok(game) => game,
        Err(error) => {
            return Err(ReplayError {
                index: None,
                error,
                state: initial,
                can_undo: false,
                events: Vec::new(),
            });
        }
    };

    let mut events = Vec::new();
    for (index, command) in commands.iter().enumerate() {
        match command {
            Command::Move(direction) => match step(&mut game, *direction) {
                Ok(turn) => events.push(Event::Moved(turn)),
                Err(error) => {
                    return Err(ReplayError {
                        index: Some(index),
                        error,
                        state: snapshot(&game),
                        can_undo: can_undo(&game),
                        events,
                    });
                }
            },
            Command::Undo => events.push(Event::Undone(undo(&mut game))),
            Command::Check => events.push(Event::Checked(game_over(&game))),
        }
    }

    Ok(Replay {
        state: snapshot(&game),
        can_undo: can_undo(&game),
        events,
    })
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }
    if board.cells.len() != board.width * board.height {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn advance(state: u64) -> u64 {
    state.wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}

fn process_line(
    board: &Board,
    direction: Direction,
    line: usize,
    output: &mut [u32],
    merges: &mut Vec<Merge>,
    score: &mut u64,
) -> Result<(), Error> {
    let len = match direction {
        Direction::Left | Direction::Right => board.width,
        Direction::Up | Direction::Down => board.height,
    };
    let mut indices = Vec::with_capacity(len);
    for offset in 0..len {
        let index = match direction {
            Direction::Left => line * board.width + offset,
            Direction::Right => line * board.width + (board.width - 1 - offset),
            Direction::Up => offset * board.width + line,
            Direction::Down => (board.height - 1 - offset) * board.width + line,
        };
        indices.push(index);
    }

    let mut pending: Option<(usize, u32)> = None;
    let mut write = 0usize;
    for &index in &indices {
        let value = board.cells[index];
        if value == 0 {
            continue;
        }
        if let Some((first, previous_value)) = pending {
            if previous_value == value {
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first,
                        second: index,
                    });
                }
                let merged_value = value * 2;
                *score = score
                    .checked_add(merged_value as u64)
                    .ok_or(Error::ScoreOverflow)?;
                output[indices[write]] = merged_value;
                merges.push(Merge {
                    from: [first, index],
                    to: indices[write],
                    value: merged_value,
                });
                write += 1;
                pending = None;
                continue;
            }
            output[indices[write]] = previous_value;
            write += 1;
        }
        pending = Some((index, value));
    }
    if let Some((_, value)) = pending {
        output[indices[write]] = value;
    }
    Ok(())
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut output = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let lines = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };
    for line in 0..lines {
        process_line(
            board,
            direction,
            line,
            &mut output,
            &mut merges,
            &mut score,
        )?;
    }

    let changed = output != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells: output,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut legal = Vec::new();
    for direction in [Direction::Up, Direction::Left, Direction::Down, Direction::Right] {
        match move_board(board, direction) {
            Ok(result) if result.changed => legal.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(legal)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let empty_count = board.cells.iter().filter(|&&value| value == 0).count();
    if empty_count == 0 {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let first_draw = advance(rng);
    let rank = (first_draw % empty_count as u64) as usize;
    let index = board
        .cells
        .iter()
        .enumerate()
        .filter(|(_, value)| **value == 0)
        .nth(rank)
        .map(|(index, _)| index)
        .expect("rank is within the empty cell count");

    let second_draw = advance(first_draw);
    let value = if second_draw % 10 == 0 { 4 } else { 2 };
    let mut cells = board.cells.clone();
    cells[index] = value;

    Ok(SpawnResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        rng: second_draw,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn snapshot(game: &Game) -> State {
    game.current.clone()

pub fn step(game: &mut Game, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(&game.current.board, direction)?;
    let total = game
        .current
        .score
        .checked_add(moved.score)
        .ok_or(Error::ScoreOverflow)?;

    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng: game.current.rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let spawned = spawn_tile(&moved.board, game.current.rng)?;
    let prior = game.current.clone();
    let turn = Turn {
        board: spawned.board.clone(),
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    };
    game.current = State {
        board: spawned.board,
        score: total,
        rng: spawned.rng,
    };
    game.previous = Some(prior);
    Ok(turn)
}

pub fn undo(game: &mut Game) -> bool {
    let Some(prior) = game.previous.take() else {
        return false;
    };
    game.current = prior;
    true
}

pub fn can_undo(game: &Game) -> bool {
    game.previous.is_some()
}

pub fn game_over(game: &Game) -> bool {
    match legal_moves(&game.current.board) {
        Ok(moves) => moves.is_empty(),
        Err(_) => false,
    }
}

pub fn replay(initial: State, commands: &[Command]) -> Result<Replay, ReplayError> {
    let mut game = match new_game(initial.clone()) {
        Ok(game) => game,
        Err(error) => {
            return Err(ReplayError {
                index: None,
                error,
                state: initial,
                can_undo: false,
                events: Vec::new(),
            });
        }
    };

    let mut events = Vec::with_capacity(commands.len());
    for (index, command) in commands.iter().enumerate() {
        let event = match command {
            Command::Move(direction) => match step(&mut game, *direction) {
                Ok(turn) => Event::Moved(turn),
                Err(error) => {
                    return Err(ReplayError {
                        index: Some(index),
                        error,
                        state: snapshot(&game),
                        can_undo: can_undo(&game),
                        events,
                    });
                }
            },
            Command::Undo => Event::Undone(undo(&mut game)),
            Command::Check => Event::Checked(game_over(&game)),
        };
        events.push(event);
    }

    Ok(Replay {
        state: snapshot(&game),
        can_undo: can_undo(&game),
        events,
    })
}
