#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
    ScoreOverflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }
    if board.cells.len() != board.width * board.height {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn line_indices(board: &Board, direction: Direction, line: usize) -> Vec<usize> {
    match direction {
        Direction::Left => (0..board.width).map(|x| line * board.width + x).collect(),
        Direction::Right => (0..board.width)
            .rev()
            .map(|x| line * board.width + x)
            .collect(),
        Direction::Up => (0..board.height)
            .map(|y| y * board.width + line)
            .collect(),
        Direction::Down => (0..board.height)
            .rev()
            .map(|y| y * board.width + line)
            .collect(),
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line in 0..line_count {
        let indices = line_indices(board, direction, line);
        let tiles: Vec<(usize, u32)> = indices
            .iter()
            .filter_map(|&i| (board.cells[i] != 0).then_some((i, board.cells[i])))
            .collect();

        let mut out_pos = 0;
        let mut i = 0;
        while i < tiles.len() {
            let (first_index, value) = tiles[i];
            if i + 1 < tiles.len() && tiles[i + 1].1 == value {
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: first_index,
                        second: tiles[i + 1].0,
                    });
                }
                let doubled = value * 2;
                let to = indices[out_pos];
                cells[to] = doubled;
                merges.push(Merge {
                    from: [first_index, tiles[i + 1].0],
                    to,
                    value: doubled,
                });
                score += doubled as u64;
                out_pos += 1;
                i += 2;
            } else {
                cells[indices[out_pos]] = value;
                out_pos += 1;
                i += 1;
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut result = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(moved) if moved.changed => result.push(direction),
            _ => {}
        }
    }
    Ok(result)
}

}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SpawnResult {
    pub board: Board,
    pub rng: u64,
    pub spawn: Option<Spawn>,
}

fn draw(state: u64) -> u64 {
    state
        .wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let empty: Vec<usize> = board
        .cells
        .iter()
        .enumerate()
        .filter_map(|(i, &v)| (v == 0).then_some(i))
        .collect();

    if empty.is_empty() {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let a = draw(rng);
    let index = empty[(a % empty.len() as u64) as usize];
    let b = draw(a);
    let value = if b % 10 == 0 { 4 } else { 2 };
    let mut spawned_board = board.clone();
    spawned_board.cells[index] = value;

    Ok(SpawnResult {
        board: spawned_board,
        rng: b,
        spawn: Some(Spawn { index, value }),
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Turn {
    pub board: Board,
    pub rng: u64,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
    pub spawn: Option<Spawn>,
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let spawned = spawn_tile(&moved.board, rng)?;
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub board: Board,
    pub score: u64,
    pub rng: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Game {
    current: State,
    undo_state: Option<State>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Command {
    Move(Direction),
    Undo,
    Check,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Event {
    Moved(Turn),
    Undone(bool),
    Checked(bool),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Replay {
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ReplayError {
    pub index: Option<usize>,
    pub error: Error,
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

pub fn new_game(initial: State) -> Result<Game, Error> {
    validate(&initial.board)?;
    Ok(Game {
        current: initial,
        undo_state: None,
    })
}

pub fn snapshot(game: &Game) -> State {
    game.current.clone()
}

pub fn step(game: &mut Game, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(&game.current.board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng: game.current.rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let cumulative_score = game
        .current
        .score
        .checked_add(moved.score)
        .ok_or(Error::ScoreOverflow)?;
    let spawned = spawn_tile(&moved.board, game.current.rng)?;
    let previous = game.current.clone();

    game.current = State {
        board: spawned.board.clone(),
        score: cumulative_score,
        rng: spawned.rng,
    };
    game.undo_state = Some(previous);

    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

pub fn undo(game: &mut Game) -> bool {
    if let Some(previous) = game.undo_state.take() {
        game.current = previous;
        true
    } else {
        false
    }
}

pub fn can_undo(game: &Game) -> bool {
    game.undo_state.is_some()
}

pub fn game_over(game: &Game) -> bool {
    legal_moves(&game.current.board)
        .map(|moves| moves.is_empty())
        .unwrap_or(true)
}

pub fn replay(initial: State, commands: &[Command]) -> Result<Replay, ReplayError> {
    let mut game = match new_game(initial.clone()) {
        Ok(game) => game,
        Err(error) => {
            return Err(ReplayError {
                index: None,
                error,
                state: initial,
                can_undo: false,
                events: Vec::new(),
            });
        }
    };

    let mut events = Vec::new();
    for (index, command) in commands.iter().enumerate() {
        match command {
            Command::Move(direction) => match step(&mut game, *direction) {
                Ok(turn) => events.push(Event::Moved(turn)),
                Err(error) => {
                    return Err(ReplayError {
                        index: Some(index),
                        error,
                        state: snapshot(&game),
                        can_undo: can_undo(&game),
                        events,
                    });
                }
            },
            Command::Undo => events.push(Event::Undone(undo(&mut game))),
            Command::Check => events.push(Event::Checked(game_over(&game))),
        }
    }

    Ok(Replay {
        state: snapshot(&game),
        can_undo: can_undo(&game),
        events,
    })
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction {
    Up,
    Left,
    Down,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
    ScoreOverflow,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if board.width == 0
        || board.width > 8
        || board.height == 0
        || board.height > 8
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }

    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0
            && (value < 2 || value > 2_147_483_648 || value & (value - 1) != 0)
        {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn coordinates(board: &Board, direction: Direction, line: usize) -> Vec<usize> {
    let mut result = Vec::new();
    match direction {
        Direction::Left => {
            for x in 0..board.width {
                result.push(line * board.width + x);
            }
        }
        Direction::Right => {
            for x in (0..board.width).rev() {
                result.push(line * board.width + x);
            }
        }
        Direction::Up => {
            for y in 0..board.height {
                result.push(y * board.width + line);
            }
        }
        Direction::Down => {
            for y in (0..board.height).rev() {
                result.push(y * board.width + line);
            }
        }
    }
    result
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let horizontal = matches!(direction, Direction::Left | Direction::Right);
    let lines = if horizontal { board.height } else { board.width };
    let mut output = vec![0u32; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    for line in 0..lines {
        let indices = coordinates(board, direction, line);
        let mut values = Vec::new();
        for &index in &indices {
            if board.cells[index] != 0 {
                values.push((board.cells[index], index));
            }
        }

        let mut source = 0usize;
        let mut destination = 0usize;
        while source < values.len() {
            let (value, first_index) = values[source];
            if source + 1 < values.len() && values[source + 1].0 == value {
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: first_index,
                        second: values[source + 1].1,
                    });
                }
                let merged = value * 2;
                let to = indices[destination];
                output[to] = merged;
                merges.push(Merge {
                    from: [first_index, values[source + 1].1],
                    to,
                    value: merged,
                });
                score += merged as u64;
                source += 2;
            } else {
                output[indices[destination]] = value;
                source += 1;
            }
            destination += 1;
        }
    }

    let changed = output != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells: output,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut legal = Vec::new();
    for direction in [Direction::Up, Direction::Left, Direction::Down, Direction::Right] {
        if let Ok(result) = move_board(board, direction) {
            if result.changed {
                legal.push(direction);
            }
        }
    }
    Ok(legal)
}

pub struct SpawnResult {
    pub board: Board,
    pub rng: u64,
    pub spawn: Option<Spawn>,
}

fn next_rng(rng: u64) -> u64 {
    rng.wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let mut empties = Vec::new();
    for (index, &value) in board.cells.iter().enumerate() {
        if value == 0 {
            empties.push(index);
        }
    }
    if empties.is_empty() {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let first_draw = next_rng(rng);
    let index = empties[(first_draw % empties.len() as u64) as usize];
    let second_draw = next_rng(first_draw);
    let value = if second_draw % 10 == 0 { 4 } else { 2 };
    let mut result_board = board.clone();
    result_board.cells[index] = value;

    Ok(SpawnResult {
        board: result_board,
        rng: second_draw,
        spawn: Some(Spawn { index, value }),
    })
}

    Undone(bool),
    Checked(bool),

    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct ReplayError {
    pub index: Option<usize>,
    pub error: Error,
    pub state: State,
    pub can_undo: bool,
    pub events: Vec<Event>,
}

pub fn new_game(initial: State) -> Result<Game, Error> {
    validate(&initial.board)?;
    Ok(Game {
        current: initial,
        previous: None,
    })
}

pub fn snapshot(game: &Game) -> State {
    game.current.clone()
}

pub fn step(game: &mut Game, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(&game.current.board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng: game.current.rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let new_score = game
        .current
        .score
        .checked_add(moved.score)
        .ok_or(Error::ScoreOverflow)?;
    let spawned = spawn_tile(&moved.board, game.current.rng)?;
    let before = game.current.clone();
    game.current = State {
        board: spawned.board.clone(),
        score: new_score,
        rng: spawned.rng,
    };
    game.previous = Some(before);

    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}

pub fn undo(game: &mut Game) -> bool {
    match game.previous.take() {
        Some(previous) => {
            game.current = previous;
            true
        }
        None => false,
    }
}

pub fn can_undo(game: &Game) -> bool {
    game.previous.is_some()
}

pub fn game_over(game: &Game) -> bool {
    legal_moves(&game.current.board)
        .map(|moves| moves.is_empty())
        .unwrap_or(true)
}

pub fn replay(initial: State, commands: &[Command]) -> Result<Replay, ReplayError> {
    let mut game = match new_game(initial.clone()) {
        Ok(game) => game,
        Err(error) => {
            return Err(ReplayError {
                index: None,
                error,
                state: initial,
                can_undo: false,
                events: Vec::new(),
            });
        }
    };

    let mut events = Vec::new();
    for (index, command) in commands.iter().enumerate() {
        let event = match command {
            Command::Move(direction) => match step(&mut game, *direction) {
                Ok(turn) => Event::Moved(turn),
                Err(error) => {
                    return Err(ReplayError {
                        index: Some(index),
                        error,
                        state: snapshot(&game),
                        can_undo: can_undo(&game),
                        events,
                    });
                }
            },
            Command::Undo => Event::Undone(undo(&mut game)),
            Command::Check => Event::Checked(game_over(&game)),
        };
        events.push(event);
    }

    Ok(Replay {
        state: snapshot(&game),
        can_undo: can_undo(&game),
        events,
    })
}
