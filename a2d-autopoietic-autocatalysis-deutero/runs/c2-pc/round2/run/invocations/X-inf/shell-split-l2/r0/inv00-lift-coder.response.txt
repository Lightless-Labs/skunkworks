```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0;
    let mut iter = input.char_indices().peekable();

    while let Some((pos, ch)) = iter.next() {
        match quote {
            Quote::Single => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
            }
            Quote::Double => match ch {
                '"' => quote = Quote::None,
                '\\' => {
                    let Some(&(next_pos, next)) = iter.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        iter.next();
                        if next != '\n' {
                            word.push(next);
                        }
                    } else {
                        word.push('\\');
                        let _ = next_pos;
                    }
                }
                '$' => {
                    active = true;
                    expand(input, &mut iter, pos, env, &mut word)?;
                }
                _ => word.push(ch),
            },
            Quote::None => match ch {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    quote = Quote::Single;
                    quote_pos = pos;
                }
                '"' => {
                    active = true;
                    quote = Quote::Double;
                    quote_pos = pos;
                }
                '\\' => {
                    let Some(&(_, next)) = iter.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    iter.next();
                    if next != '\n' {
                        active = true;
                        word.push(next);
                    }
                }
                '$' => {
                    active = true;
                    expand(input, &mut iter, pos, env, &mut word)?;
                }
                _ => {
                    active = true;
                    word.push(ch);
                }
            },
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }
    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(
    input: &str,
    iter: &mut std::iter::Peekable<std::str::CharIndices<'_>>,
    dollar_pos: usize,
    env: &[(&str, &str)],
    word: &mut String,
) -> Result<(), SplitError> {
    let Some(&(_, next)) = iter.peek() else {
        word.push('$');
        return Ok(());
    };

    if next == '{' {
        iter.next();
        let content_start = iter.peek().map_or(input.len(), |(p, _)| *p);
        let mut scan = content_start;
        let mut close = None;
        while scan < input.len() {
            let ch = input[scan..].chars().next().unwrap();
            if ch == '\\' {
                scan += 1;
                if scan < input.len() {
                    scan += input[scan..].chars().next().unwrap().len_utf8();
                }
            } else if ch == '}' {
                close = Some(scan);
                break;
            } else {
                scan += ch.len_utf8();
            }
        }
        let Some(close_pos) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
        };

        let contents = &input[content_start..close_pos];
        let (name, default) = if let Some(op_pos) = contents.find(":-") {
            (&contents[..op_pos], Some(&contents[op_pos + 2..]))
        } else {
            (contents, None)
        };

        if !valid_name(name) {
            return Err(SplitError::InvalidExpansion { pos: dollar_pos });
        }
        if default.is_none() && contents.contains(':') {
            return Err(SplitError::InvalidExpansion { pos: dollar_pos });
        }

        while iter.peek().is_some_and(|(p, _)| *p <= close_pos) {
            iter.next();
        }

        let value = env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value);
        match default {
            Some(text) if value.is_none_or(str::is_empty) => word.push_str(&decode_default(text)),
            Some(_) => word.push_str(value.unwrap()),
            None => word.push_str(value.unwrap_or("")),
        }
        return Ok(());
    }

    if is_name_start(next) {
        let mut name = String::new();
        while let Some(&(_, ch)) = iter.peek() {
            if is_name_continue(ch) {
                name.push(ch);
                iter.next();
            } else {
                break;
            }
        }
        if let Some((_, value)) = env.iter().find(|(key, _)| **key == name) {
            word.push_str(value);
        }
        return Ok(());
    }

    word.push('$');
    Ok(())
}

fn valid_name(name: &str) -> bool {
    let mut chars = name.chars();
    matches!(chars.next(), Some(ch) if is_name_start(ch))
        && chars.all(is_name_continue)
}

fn is_name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn is_name_continue(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn decode_default(text: &str) -> String {
    let mut decoded = String::new();
    let mut chars = text.chars();
    while let Some(ch) = chars.next() {
        if ch == '\\' {
            if let Some(next) = chars.next() {
                decoded.push(next);
            } else {
                decoded.push('\\');
            }
        } else {
            decoded.push(ch);
        }
    }
    decoded
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    let bytes = input.as_bytes();
    if let Some(pos) = bytes.iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut mode = 0u8; // 0 bare, 1 single-quoted, 2 double-quoted
    let mut opening = 0usize;
    let mut index = 0usize;

    while index < input.len() {
        let ch = input[index..].chars().next().unwrap();
        let width = ch.len_utf8();

        if mode == 1 {
            if ch == '\'' {
                mode = 0;
            } else {
                current.push(ch);
            }
            index += width;
            continue;
        }

        if mode == 2 {
            match ch {
                '"' => {
                    mode = 0;
                    index += width;
                }
                '\\' => {
                    let esc_pos = index;
                    index += width;
                    if index == input.len() {
                        return Err(SplitError::TrailingEscape { pos: esc_pos });
                    }
                    let next = input[index..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        index += next.len_utf8();
                        if next != '\n' {
                            current.push(next);
                        }
                    } else {
                        current.push('\\');
                    }
                }
                '$' => {
                    active = true;
                    index = expand_at(input, index, env, &mut current)?;
                }
                _ => {
                    current.push(ch);
                    index += width;
                }
            }
            continue;
        }

        match ch {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    result.push(std::mem::take(&mut current));
                    active = false;
                }
                index += width;
            }
            '\'' | '"' => {
                active = true;
                mode = if ch == '\'' { 1 } else { 2 };
                opening = index;
                index += width;
            }
            '\\' => {
                let esc_pos = index;
                index += width;
                if index == input.len() {
                    return Err(SplitError::TrailingEscape { pos: esc_pos });
                }
                let next = input[index..].chars().next().unwrap();
                index += next.len_utf8();
                if next != '\n' {
                    active = true;
                    current.push(next);
                }
            }
            '$' => {
                active = true;
                index = expand_at(input, index, env, &mut current)?;
            }
            _ => {
                active = true;
                current.push(ch);
                index += width;
            }
        }
    }

    if mode == 1 {
        return Err(SplitError::UnterminatedSingle { pos: opening });
    }
    if mode == 2 {
        return Err(SplitError::UnterminatedDouble { pos: opening });
    }
    if active {
        result.push(current);
    }
    Ok(result)
}

fn expand_at(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
    output: &mut String,
) -> Result<usize, SplitError> {
    let mut index = dollar + 1;
    if index == input.len() {
        output.push('$');
        return Ok(index);
    }
    let first = input[index..].chars().next().unwrap();
    if first == '{' {
        index += 1;
        let contents_start = index;
        let mut close = None;
        while index < input.len() {
            let ch = input[index..].chars().next().unwrap();
            if ch == '\\' {
                index += 1;
                if index < input.len() {
                    index += input[index..].chars().next().unwrap().len_utf8();
                }
            } else if ch == '}' {
                close = Some(index);
                break;
            } else {
                index += ch.len_utf8();
            }
        }
        let Some(close_at) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        };
        let body = &input[contents_start..close_at];
        let operator = body.find(":-");
        let (name, default) = match operator {
            Some(at) => (&body[..at], Some(&body[at + 2..])),
            None => (body, None),
        };
        if !valid_name(name) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        if default.is_none() && body.contains(':') {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        let found = env.iter().find(|(key, _)| *key == name).map(|(_, val)| *val);
        if let Some(default_text) = default {
            if found.is_some_and(|value| !value.is_empty()) {
                output.push_str(found.unwrap());
            } else {
                output.push_str(&decode(default_text));
            }
        } else if let Some(value) = found {
            output.push_str(value);
        }
        Ok(close_at + 1)
    } else if name_start(first) {
        let start = index;
        while index < input.len() {
            let ch = input[index..].chars().next().unwrap();
            if name_continue(ch) {
                index += ch.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[start..index];
        if let Some((_, value)) = env.iter().find(|(key, _)| *key == name) {
            output.push_str(value);
        }
        Ok(index)
    } else {
        output.push('$');
        Ok(dollar + 1)
    }
}

fn name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn name_continue(ch: char) -> bool {
    name_start(ch) || ch.is_ascii_digit()
}

fn valid_name(s: &str) -> bool {
    let mut chars = s.chars();
    matches!(chars.next(), Some(ch) if name_start(ch)) && chars.all(name_continue)
}

fn decode(text: &str) -> String {
    let mut chars = text.chars();
    let mut output = String::new();
    while let Some(ch) = chars.next() {
        if ch == '\\' {
            if let Some(next) = chars.next() {
                output.push(next);
            } else {
                output.push('\\');
            }
        } else {
            output.push(ch);
        }
    }
    output
}
```
