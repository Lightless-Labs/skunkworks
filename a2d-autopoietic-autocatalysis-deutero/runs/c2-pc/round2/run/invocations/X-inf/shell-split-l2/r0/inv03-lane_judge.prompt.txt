#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    let bytes = input.as_bytes();
    if let Some(pos) = bytes.iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut mode = 0u8; // 0 bare, 1 single-quoted, 2 double-quoted
    let mut opening = 0usize;
    let mut index = 0usize;

    while index < input.len() {
        let ch = input[index..].chars().next().unwrap();
        let width = ch.len_utf8();

        if mode == 1 {
            if ch == '\'' {
                mode = 0;
            } else {
                current.push(ch);
            }
            index += width;
            continue;
        }

        if mode == 2 {
            match ch {
                '"' => {
                    mode = 0;
                    index += width;
                }
                '\\' => {
                    let esc_pos = index;
                    index += width;
                    if index == input.len() {
                        return Err(SplitError::TrailingEscape { pos: esc_pos });
                    }
                    let next = input[index..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        index += next.len_utf8();
                        if next != '\n' {
                            current.push(next);
                        }
                    } else {
                        current.push('\\');
                    }
                }
                '$' => {
                    active = true;
                    index = expand_at(input, index, env, &mut current)?;
                }
                _ => {
                    current.push(ch);
                    index += width;
                }
            }
            continue;
        }

        match ch {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    result.push(std::mem::take(&mut current));
                    active = false;
                }
                index += width;
            }
            '\'' | '"' => {
                active = true;
                mode = if ch == '\'' { 1 } else { 2 };
                opening = index;
                index += width;
            }
            '\\' => {
                let esc_pos = index;
                index += width;
                if index == input.len() {
                    return Err(SplitError::TrailingEscape { pos: esc_pos });
                }
                let next = input[index..].chars().next().unwrap();
                index += next.len_utf8();
                if next != '\n' {
                    active = true;
                    current.push(next);
                }
            }
            '$' => {
                active = true;
                index = expand_at(input, index, env, &mut current)?;
            }
            _ => {
                active = true;
                current.push(ch);
                index += width;
            }
        }
    }

    if mode == 1 {
        return Err(SplitError::UnterminatedSingle { pos: opening });
    }
    if mode == 2 {
        return Err(SplitError::UnterminatedDouble { pos: opening });
    }
    if active {
        result.push(current);
    }
    Ok(result)
}

fn expand_at(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
    output: &mut String,
) -> Result<usize, SplitError> {
    let mut index = dollar + 1;
    if index == input.len() {
        output.push('$');
        return Ok(index);
    }
    let first = input[index..].chars().next().unwrap();
    if first == '{' {
        index += 1;
        let contents_start = index;
        let mut close = None;
        while index < input.len() {
            let ch = input[index..].chars().next().unwrap();
            if ch == '\\' {
                index += 1;
                if index < input.len() {
                    index += input[index..].chars().next().unwrap().len_utf8();
                }
            } else if ch == '}' {
                close = Some(index);
                break;
            } else {
                index += ch.len_utf8();
            }
        }
        let Some(close_at) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar });
        };
        let body = &input[contents_start..close_at];
        let operator = body.find(":-");
        let (name, default) = match operator {
            Some(at) => (&body[..at], Some(&body[at + 2..])),
            None => (body, None),
        };
        if !valid_name(name) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        if default.is_none() && body.contains(':') {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        let found = env.iter().find(|(key, _)| *key == name).map(|(_, val)| *val);
        if let Some(default_text) = default {
            if found.is_some_and(|value| !value.is_empty()) {
                output.push_str(found.unwrap());
            } else {
                output.push_str(&decode(default_text));
            }
        } else if let Some(value) = found {
            output.push_str(value);
        }
        Ok(close_at + 1)
    } else if name_start(first) {
        let start = index;
        while index < input.len() {
            let ch = input[index..].chars().next().unwrap();
            if name_continue(ch) {
                index += ch.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[start..index];
        if let Some((_, value)) = env.iter().find(|(key, _)| *key == name) {
            output.push_str(value);
        }
        Ok(index)
    } else {
        output.push('$');
        Ok(dollar + 1)
    }
}

fn name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn name_continue(ch: char) -> bool {
    name_start(ch) || ch.is_ascii_digit()
}

fn valid_name(s: &str) -> bool {
    let mut chars = s.chars();
    matches!(chars.next(), Some(ch) if name_start(ch)) && chars.all(name_continue)
}

fn decode(text: &str) -> String {
    let mut chars = text.chars();
    let mut output = String::new();
    while let Some(ch) = chars.next() {
        if ch == '\\' {
            if let Some(next) = chars.next() {
                output.push(next);
            } else {
                output.push('\\');
            }
        } else {
            output.push(ch);
        }
    }
    output
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum State {
    Bare,
    Single(usize),
    Double(usize),
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.bytes().position(|b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut state = State::Bare;
    let mut offset = 0;

    while offset < input.len() {
        let ch = input[offset..].chars().next().unwrap();
        let width = ch.len_utf8();

        match state {
            State::Single(start) => {
                if ch == '\'' {
                    state = State::Bare;
                } else {
                    current.push(ch);
                }
                offset += width;
                if offset == input.len() {
                    let _ = start;
                }
            }
            State::Double(start) => {
                if ch == '"' {
                    state = State::Bare;
                    offset += width;
                } else if ch == '\\' {
                    let esc_at = offset;
                    offset += width;
                    if offset == input.len() {
                        return Err(SplitError::TrailingEscape { pos: esc_at });
                    }
                    let escaped = input[offset..].chars().next().unwrap();
                    if matches!(escaped, '"' | '\\' | '$') {
                        current.push(escaped);
                        active = true;
                        offset += escaped.len_utf8();
                    } else if escaped == '\n' {
                        offset += escaped.len_utf8();
                    } else {
                        current.push('\\');
                        active = true;
                    }
                } else if ch == '$' {
                    if let Some((end, value)) = expansion(input, offset, env)? {
                        current.push_str(&value);
                        active = true;
                        offset = end;
                    } else {
                        current.push('$');
                        active = true;
                        offset += 1;
                    }
                } else {
                    current.push(ch);
                    active = true;
                    offset += width;
                }
                let _ = start;
            }
            State::Bare => match ch {
                '\'' => {
                    state = State::Single(offset);
                    active = true;
                    offset += width;
                }
                '"' => {
                    state = State::Double(offset);
                    active = true;
                    offset += width;
                }
                '\\' => {
                    let esc_at = offset;
                    offset += width;
                    if offset == input.len() {
                        return Err(SplitError::TrailingEscape { pos: esc_at });
                    }
                    let escaped = input[offset..].chars().next().unwrap();
                    if escaped != '\n' {
                        current.push(escaped);
                        active = true;
                    }
                    offset += escaped.len_utf8();
                }
                '$' => {
                    if let Some((end, value)) = expansion(input, offset, env)? {
                        current.push_str(&value);
                        active = true;
                        offset = end;
                    } else {
                        current.push('$');
                        active = true;
                        offset += 1;
                    }
                }
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        result.push(std::mem::take(&mut current));
                        active = false;
                    }
                    offset += width;
                }
                _ => {
                    current.push(ch);
                    active = true;
                    offset += width;
                }
            },
        }
    }

    match state {
        State::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        State::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        State::Bare => {}
    }

    if active {
        result.push(current);
    }
    Ok(result)
}

fn expansion(
    input: &str,
    pos: usize,
    env: &[(&str, &str)],
) -> Result<Option<(usize, String)>, SplitError> {
    let start = pos + 1;
    if start == input.len() {
        return Ok(None);
    }
    let first = input[start..].chars().next().unwrap();
    if first == '{' {
        let body_start = start + 1;
        let mut p = body_start;
        let mut end = None;
        while p < input.len() {
            let c = input[p..].chars().next().unwrap();
            let w = c.len_utf8();
            if c == '\\' {
                p += w;
                if p < input.len() {
                    p += input[p..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                end = Some(p);
                break;
            } else {
                p += w;
            }
        }
        let close = end.ok_or(SplitError::UnterminatedExpansion { pos })?;
        let body = &input[body_start..close];
        let mut n = 0;
        for (idx, c) in body.char_indices() {
            if (idx == 0 && is_start(c)) || (idx > 0 && is_continue(c)) {
                n = idx + c.len_utf8();
            } else {
                break;
            }
        }
        if n == 0 {
            return Err(SplitError::InvalidExpansion { pos });
        }
        let name = &body[..n];
        let suffix = &body[n..];
        let default = if suffix.is_empty() {
            None
        } else if let Some(rest) = suffix.strip_prefix(":-") {
            Some(rest)
        } else {
            return Err(SplitError::InvalidExpansion { pos });
        };
        let value = env
            .iter()
            .find(|(key, _)| *key == name)
            .map(|(_, value)| *value)
            .unwrap_or("");
        let chosen = match default {
            Some(d) if value.is_empty() => decode(d),
            _ => value.to_owned(),
        };
        Ok(Some((close + 1, chosen)))
    } else if is_start(first) {
        let mut p = start;
        while p < input.len() {
            let c = input[p..].chars().next().unwrap();
            if is_continue(c) {
                p += c.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[start..p];
        let value = env
            .iter()
            .find(|(key, _)| *key == name)
            .map(|(_, value)| *value)
            .unwrap_or("");
        Ok(Some((p, value.to_owned())))
    } else {
        Ok(None)
    }
}

fn is_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_continue(c: char) -> bool {
    is_start(c) || c.is_ascii_digit()
}

fn decode(s: &str) -> String {
    let mut out = String::new();
    let mut p = 0;
    while p < s.len() {
        let c = s[p..].chars().next().unwrap();
        if c == '\\' {
            p += 1;
            if p < s.len() {
                let escaped = s[p..].chars().next().unwrap();
                out.push(escaped);
                p += escaped.len_utf8();
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
            p += c.len_utf8();
        }
    }
    out
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_pos = 0;
    let mut iter = input.char_indices().peekable();

    while let Some((pos, ch)) = iter.next() {
        match quote {
            Quote::Single => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
            }
            Quote::Double => match ch {
                '"' => quote = Quote::None,
                '\\' => {
                    let Some(&(next_pos, next)) = iter.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        iter.next();
                        if next != '\n' {
                            word.push(next);
                        }
                    } else {
                        word.push('\\');
                        let _ = next_pos;
                    }
                }
                '$' => {
                    active = true;
                    expand(input, &mut iter, pos, env, &mut word)?;
                }
                _ => word.push(ch),
            },
            Quote::None => match ch {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    quote = Quote::Single;
                    quote_pos = pos;
                }
                '"' => {
                    active = true;
                    quote = Quote::Double;
                    quote_pos = pos;
                }
                '\\' => {
                    let Some(&(_, next)) = iter.peek() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    iter.next();
                    if next != '\n' {
                        active = true;
                        word.push(next);
                    }
                }
                '$' => {
                    active = true;
                    expand(input, &mut iter, pos, env, &mut word)?;
                }
                _ => {
                    active = true;
                    word.push(ch);
                }
            },
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Quote::None => {}
    }
    if active {
        words.push(word);
    }
    Ok(words)
}

fn expand(
    input: &str,
    iter: &mut std::iter::Peekable<std::str::CharIndices<'_>>,
    dollar_pos: usize,
    env: &[(&str, &str)],
    word: &mut String,
) -> Result<(), SplitError> {
    let Some(&(_, next)) = iter.peek() else {
        word.push('$');
        return Ok(());
    };

    if next == '{' {
        iter.next();
        let content_start = iter.peek().map_or(input.len(), |(p, _)| *p);
        let mut scan = content_start;
        let mut close = None;
        while scan < input.len() {
            let ch = input[scan..].chars().next().unwrap();
            if ch == '\\' {
                scan += 1;
                if scan < input.len() {
                    scan += input[scan..].chars().next().unwrap().len_utf8();
                }
            } else if ch == '}' {
                close = Some(scan);
                break;
            } else {
                scan += ch.len_utf8();
            }
        }
        let Some(close_pos) = close else {
            return Err(SplitError::UnterminatedExpansion { pos: dollar_pos });
        };

        let contents = &input[content_start..close_pos];
        let (name, default) = if let Some(op_pos) = contents.find(":-") {
            (&contents[..op_pos], Some(&contents[op_pos + 2..]))
        } else {
            (contents, None)
        };

        if !valid_name(name) {
            return Err(SplitError::InvalidExpansion { pos: dollar_pos });
        }
        if default.is_none() && contents.contains(':') {
            return Err(SplitError::InvalidExpansion { pos: dollar_pos });
        }

        while iter.peek().is_some_and(|(p, _)| *p <= close_pos) {
            iter.next();
        }

        let value = env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value);
        match default {
            Some(text) if value.is_none_or(str::is_empty) => word.push_str(&decode_default(text)),
            Some(_) => word.push_str(value.unwrap()),
            None => word.push_str(value.unwrap_or("")),
        }
        return Ok(());
    }

    if is_name_start(next) {
        let mut name = String::new();
        while let Some(&(_, ch)) = iter.peek() {
            if is_name_continue(ch) {
                name.push(ch);
                iter.next();
            } else {
                break;
            }
        }
        if let Some((_, value)) = env.iter().find(|(key, _)| **key == name) {
            word.push_str(value);
        }
        return Ok(());
    }

    word.push('$');
    Ok(())
}

fn valid_name(name: &str) -> bool {
    let mut chars = name.chars();
    matches!(chars.next(), Some(ch) if is_name_start(ch))
        && chars.all(is_name_continue)
}

fn is_name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn is_name_continue(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn decode_default(text: &str) -> String {
    let mut decoded = String::new();
    let mut chars = text.chars();
    while let Some(ch) = chars.next() {
        if ch == '\\' {
            if let Some(next) = chars.next() {
                decoded.push(next);
            } else {
                decoded.push('\\');
            }
        } else {
            decoded.push(ch);
        }
    }
    decoded
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    let mut out = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote: Option<(char, usize)> = None;
    let mut k = 0usize;

    while k < chars.len() {
        let (pos, c) = chars[k];
        let next = if k + 1 < chars.len() { chars[k + 1].0 } else { input.len() };

        if let Some((q, _)) = quote {
            if q == '\'' {
                if c == '\'' {
                    quote = None;
                } else {
                    word.push(c);
                }
                k += 1;
                continue;
            }

            if c == '"' {
                quote = None;
                k += 1;
            } else if c == '\\' {
                if k + 1 == chars.len() {
                    return Err(SplitError::TrailingEscape { pos });
                }
                let nc = chars[k + 1].1;
                if matches!(nc, '"' | '\\' | '$' | '\n') {
                    if nc != '\n' {
                        word.push(nc);
                    }
                    k += 2;
                } else {
                    word.push('\\');
                    k += 1;
                }
            } else if c == '$' {
                let (value, consumed) = expand_from(input, pos, env)?;
                if consumed == pos + 1 {
                    word.push('$');
                    k += 1;
                } else {
                    word.push_str(&value);
                    active = true;
                    while k < chars.len() && chars[k].0 < consumed {
                        k += 1;
                    }
                }
            } else {
                word.push(c);
                k += 1;
            }
            continue;
        }

        if matches!(c, ' ' | '\t' | '\r' | '\n') {
            if active {
                out.push(std::mem::take(&mut word));
                active = false;
            }
            k += 1;
        } else if c == '\'' || c == '"' {
            quote = Some((c, pos));
            active = true;
            k += 1;
        } else if c == '\\' {
            if k + 1 == chars.len() {
                return Err(SplitError::TrailingEscape { pos });
            }
            let nc = chars[k + 1].1;
            if nc == '\n' {
                k += 2;
            } else {
                word.push(nc);
                active = true;
                k += 2;
            }
        } else if c == '$' {
            let (value, consumed) = expand_from(input, pos, env)?;
            if consumed == pos + 1 {
                word.push('$');
                active = true;
                k += 1;
            } else {
                word.push_str(&value);
                active = true;
                while k < chars.len() && chars[k].0 < consumed {
                    k += 1;
                }
            }
        } else {
            word.push(c);
            active = true;
            k += 1;
        }
    }

    if let Some((q, pos)) = quote {
        return Err(if q == '\'' {
            SplitError::UnterminatedSingle { pos }
        } else {
            SplitError::UnterminatedDouble { pos }
        });
    }

    if active {
        out.push(word);
    }
    Ok(out)
}

fn expand_from(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(String, usize), SplitError> {
    let after = dollar + 1;
    if after == input.len() {
        return Ok(("$".to_string(), after));
    }
    let first = input[after..].chars().next().unwrap();

    if first == '{' {
        let start = after + 1;
        let mut p = start;
        let mut close = None;
        while p < input.len() {
            let c = input[p..].chars().next().unwrap();
            let q = p + c.len_utf8();
            if c == '\\' && q < input.len() {
                let escaped = input[q..].chars().next().unwrap();
                p = q + escaped.len_utf8();
            } else if c == '}' {
                close = Some(p);
                break;
            } else {
                p = q;
            }
        }
        let end = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let body = &input[start..end];

        let mut split_at = body.len();
        for (idx, c) in body.char_indices() {
            if !is_name_char_for_position(c, idx == 0) {
                split_at = idx;
                break;
            }
        }
        if split_at == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }
        let name = &body[..split_at];
        let suffix = &body[split_at..];
        let use_default = if suffix.is_empty() {
            false
        } else if let Some(text) = suffix.strip_prefix(":-") {
            let found = env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v);
            if found.map_or(true, str::is_empty) {
                let decoded = decode(text);
                return Ok((decoded, end + 1));
            }
            return Ok((found.unwrap().to_string(), end + 1));
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        let value = if use_default {
            String::new()
        } else {
            env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v).unwrap_or("").to_string()
        };
        Ok((value, end + 1))
    } else if is_start(first) {
        let mut end = after + first.len_utf8();
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if is_start(c) || c.is_ascii_digit() {
                end += c.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after..end];
        let value = env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v).unwrap_or("");
        Ok((value.to_string(), end))
    } else {
        Ok(("$".to_string(), after))
    }
}

fn decode(text: &str) -> String {
    let mut output = String::new();
    let mut iter = text.chars();
    while let Some(c) = iter.next() {
        if c == '\\' {
            if let Some(next) = iter.next() {
                output.push(next);
            } else {
                output.push('\\');
            }
        } else {
            output.push(c);
        }
    }
    output
}

fn is_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char_for_position(c: char, first: bool) -> bool {
    if first { is_start(c) } else { is_start(c) || c.is_ascii_digit() }
}

/// Returns Some(value) and the end offset when expansion syntax is consumed.
/// Returns None and the offset after '$' when the dollar is literal.
fn expansion<'a>(
    input: &'a str,
    dollar: usize,
    env: &'a [(&'a str, &'a str)],
) -> Result<(Option<&'a str>, usize), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((None, after_dollar));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut j = content_start;
        let mut close = None;

        while j < input.len() {
            let c = input[j..].chars().next().unwrap();
            let next = j + c.len_utf8();
            if c == '\\' {
                if next < input.len() {
                    let escaped = input[next..].chars().next().unwrap();
                    j = next + escaped.len_utf8();
                } else {
                    j = next;
                }
            } else if c == '}' {
                close = Some(j);
                break;
            } else {
                j = next;
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let (name_end, has_default) = {
            let mut p = 0usize;
            let mut chars = contents.char_indices();
            match chars.next() {
                Some((_, c)) if is_name_start(c) => p = c.len_utf8(),
                _ => return Err(SplitError::InvalidExpansion { pos: dollar }),
            }
            for (offset, c) in chars {
                if is_name_continue(c) {
                    p = offset + c.len_utf8();
                } else {
                    break;
                }
            }

            let rest = &contents[p..];
            if rest.is_empty() {
                (p, false)
            } else if rest.starts_with(":-") {
                (p, true)
            } else {
                return Err(SplitError::InvalidExpansion { pos: dollar });
            }
        };

        let name = &contents[..name_end];
        let value = lookup(env, name);
        if has_default && value.map_or(true, |v| v.is_empty()) {
            let text_start = content_start + name_end + 2;
            let text_end = close;
            let decoded = decode_default(&input[text_start..text_end]);
            return Ok((Some(Box::leak(decoded.into_boxed_str())), close + 1));
        }
        return Ok((Some(value.unwrap_or("")), close + 1));
    }

    if is_name_start(first) {
        let mut end = after_dollar + first.len_utf8();
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if is_name_continue(c) {
                end += c.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        return Ok((Some(lookup(env, name).unwrap_or("")), end));
    }

    Ok((None, after_dollar))
}

fn lookup<'a>(env: &'a [(&str, &'a str)], name: &str) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0usize;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        let next = i + c.len_utf8();
        if c == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            result.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            result.push(c);
            i = next;
        }
    }
    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0;
    let mut i = 0;

    while i < input.len() {
        let ch = input[i..].chars().next().unwrap();
        let next_i = i + ch.len_utf8();

        match quote {
            Quote::Single => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
                i = next_i;
            }
            Quote::Double => {
                match ch {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let after = next_i + escaped.len_utf8();
                        match escaped {
                            '"' | '\\' | '$' => {
                                word.push(escaped);
                                active = true;
                                i = after;
                            }
                            '\n' => {
                                i = after;
                            }
                            _ => {
                                word.push('\\');
                                active = true;
                                i = next_i;
                            }
                        }
                    }
                    '$' => {
                        let (consumed, value) = parse_expansion(input, i, env)?;
                        if let Some((end, value)) = consumed.zip(value) {
                            word.push_str(&value);
                            active = true;
                            i = end;
                        } else {
                            word.push('$');
                            active = true;
                            i = next_i;
                        }
                    }
                    _ => {
                        word.push(ch);
                        active = true;
                        i = next_i;
                    }
                }
            }
            Quote::None => {
                match ch {
                    '\'' | '"' => {
                        quote = if ch == '\'' { Quote::Single } else { Quote::Double };
                        quote_start = i;
                        active = true;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let after = next_i + escaped.len_utf8();
                        if escaped != '\n' {
                            word.push(escaped);
                            active = true;
                        }
                        i = after;
                    }
                    '$' => {
                        let (consumed, value) = parse_expansion(input, i, env)?;
                        if let Some((end, value)) = consumed.zip(value) {
                            word.push_str(&value);
                            active = true;
                            i = end;
                        } else {
                            word.push('$');
                            active = true;
                            i = next_i;
                        }
                    }

// Returns (Some(end), Some(value)) for an expansion, and (None, None) for a literal dollar.
fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(Option<usize>, Option<String>), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((None, None));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let contents_start = after_dollar + 1;
        let mut cursor = contents_start;
        let mut closing = None;
        while cursor < input.len() {
            let ch = input[cursor..].chars().next().unwrap();
            let end = cursor + ch.len_utf8();
            if ch == '\\' {
                if end < input.len() {
                    let skipped = input[end..].chars().next().unwrap();
                    cursor = end + skipped.len_utf8();
                } else {
                    cursor = end;
                }
            } else if ch == '}' {
                closing = Some(cursor);
                break;
            } else {
                cursor = end;
            }
        }
        let close = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[contents_start..close];

        let (name_end, name) = parse_name(contents, 0);
        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let selected_default = if name_end == contents.len() {
            None
        } else if contents[name_end..].starts_with(":-") {
            Some(&contents[name_end + 2..])
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        let value = lookup(env, name);
        let result = if let Some(default) = selected_default {
            if value.is_empty() {
                decode_default(default)
            } else {
                value.to_string()
            }
        } else {
            value.to_string()
        };
        Ok((Some(close + 1), Some(result)))
    } else if is_name_start(first) {
        let (end, name) = parse_name(input, after_dollar);
        Ok((Some(end), Some(lookup(env, name).to_string())))
    } else {
        Ok((None, None))
    }
}

fn parse_name(s: &str, start: usize) -> (usize, &str) {
    let mut end = start;
    let mut first = true;
    while end < s.len() {
        let ch = s[end..].chars().next().unwrap();
        if if first { is_name_start(ch) } else { is_name_continue(ch) } {
            end += ch.len_utf8();
            first = false;
        } else {
            break;
        }
    }
    (end, &s[start..end])
}

fn is_name_start(ch: char) -> bool {
    ch == '_' || ch.is_ascii_alphabetic()
}

fn is_name_continue(ch: char) -> bool {
    is_name_start(ch) || ch.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> &'a str {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
        .unwrap_or("")
}

        } else {

    }

}
