#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn append(word: &mut Word, value: &str, wildcards_active: bool) {
    for c in value.chars() {
        let offset = word.text.len();
        word.text.push(c);
        if wildcards_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(offset);
        }
    }
}

fn append_char(word: &mut Word, c: char, wildcards_active: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if wildcards_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn find_closing_brace(input: &str, start: usize) -> Option<usize> {
    let mut i = start;
    while i < input.len() {
        let c = input[i..].chars().next()?;
        if c == '\\' {
            i += c.len_utf8();
            if i < input.len() {
                let next = input[i..].chars().next()?;
                i += next.len_utf8();
            }
        } else if c == '}' {
            return Some(i);
        } else {
            i += c.len_utf8();
        }
    }
    None
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < text.len() {
                let next = text[i..].chars().next().unwrap();
                result.push(next);
                i += next.len_utf8();
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
            i += c.len_utf8();
        }
    }
    result
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_open = dollar + 2;
    let close = find_closing_brace(input, after_open)
        .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_open..close];

    let name_end = contents
        .char_indices()
        .take_while(|(_, c)| is_name_continue(*c))
        .map(|(i, c)| i + c.len_utf8())
        .last()
        .unwrap_or(0);

    if name_end == 0 || !is_name_start(contents.chars().next().unwrap()) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let suffix = &contents[name_end..];
    let value = if suffix.is_empty() {
        lookup(name, env).unwrap_or("").to_string()
    } else if let Some(default) = suffix.strip_prefix(":-") {
        let found = lookup(name, env).unwrap_or("");
        if found.is_empty() {
            decode_default(default)
        } else {
            found.to_string()
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(&mut word, c, false);
                }
                i = next_i;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        append_char(&mut word, next, false);
                        active = true;
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        append_char(&mut word, '\\', false);
                        active = true;
                        i = next_i;
                    }
                } else if c == '$' {
                    if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if n == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        if is_name_start(n) {
                            let mut end = next_i + n.len_utf8();
                            while end < input.len() {
                                let x = input[end..].chars().next().unwrap();
                                if !is_name_continue(x) {
                                    break;
                                }
                                end += x.len_utf8();
                            }
                            let name = &input[next_i..end];
                            append(&mut word, lookup(name, env).unwrap_or(""), false);
                            active = true;
                            i = end;
                        } else if n == '{' {
                            let (value, end) = parse_expansion(input, i, env)?;
                            append(&mut word, &value, false);
                            active = true;
                            i = end;
                        } else {
                            append_char(&mut word, '$', false);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append_char(&mut word, '$', false);
                        active = true;
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else {
                    append_char(&mut word, c, false);
                    active = true;
                    i = next_i;
                }
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' || c == '"' {
                    quote = if c == '\'' { Quote::Single } else { Quote::Double };
                    quote_start = i;
                    active = true;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        append_char(&mut word, next, false);
                        active = true;
                        i = after_next;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if n == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                        if is_name_start(n) {
                            let mut end = next_i + n.len_utf8();
                            while end < input.len() {
                                let x = input[end..].chars().next().unwrap();
                                if !is_name_continue(x) {
                                    break;
                                }
                                end += x.len_utf8();
                            }
                            let name = &input[next_i..end];
                            append(&mut word, lookup(name, env).unwrap_or(""), true);
                            active = true;
                            i = end;
                        } else if n == '{' {
                            let (value, end) = parse_expansion(input, i, env)?;
                            append(&mut word, &value, true);
                            active = true;
                            i = end;
                        } else {
                            append_char(&mut word, '$', true);
                            active = true;
                            i = next_i;
                        }
                    } else {
                        append_char(&mut word, '$', true);
                        active = true;
                        i = next_i;
                    }
                } else {
                    append_char(&mut word, c, true);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }

        words.push(word);
    }
    Ok(words)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append(word: &mut Word, s: &str, wildcards_active: bool) {
    for c in s.chars() {
        if wildcards_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append(&mut word, &input[i..i + n], false);
                }
                i += n;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i += n;
                } else if c == '\\' {
                    let slash = i;
                    i += n;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append(&mut word, &input[i..i + next.len_utf8()], false);
                        }
                        i += next.len_utf8();
                    } else {
                        append(&mut word, "\\", false);
                    }
                } else if c == '$' {
                    let next_i = i + n;
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, name, default) = parse_braced(input, i)?;
                        let value = select_value(name, default, env);
                        active = true;
                        append(&mut word, &value, false);
                        i = end;
                    } else if let Some((end, name)) = parse_simple_name(input, next_i) {
                        active = true;
                        let value = lookup(env, name).unwrap_or("");
                        append(&mut word, value, false);
                        i = end;
                    } else {
                        active = true;
                        append(&mut word, "$", false);
                        i += n;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else {
                    append(&mut word, &input[i..i + n], false);
                    i += n;
                }
                let _ = open;
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        result.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i += n;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i += n;
                } else if c == '\\' {
                    let slash = i;
                    i += n;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next == '\n' {
                        i += next.len_utf8();
                    } else {
                        active = true;
                        append(&mut word, &input[i..i + next.len_utf8()], false);
                        i += next.len_utf8();
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_i = i + n;
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, name, default) = parse_braced(input, i)?;
                        active = true;
                        let value = select_value(name, default, env);
                        append(&mut word, &value, true);
                        i = end;
                    } else if let Some((end, name)) = parse_simple_name(input, next_i) {
                        active = true;
                        let value = lookup(env, name).unwrap_or("");
                        append(&mut word, value, true);
                        i = end;
                    } else {
                        active = true;
                        append(&mut word, "$", true);
                        i += n;
                    }
                } else {
                    active = true;
                    append(&mut word, &input[i..i + n], true);
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }
    if active {
        result.push(word);
    }
    Ok(result)
}

fn parse_simple_name(input: &str, start: usize) -> Option<(usize, &str)> {
    let first = input.get(start..)?.chars().next()?;
    if !is_name_start(first) {
        return None;
    }
    let mut end = start + first.len_utf8();
    while end < input.len() {
        let c = input[end..].chars().next().unwrap();
        if !is_name_continue(c) {
            break;
        }
        end += c.len_utf8();
    }
    Some((end, &input[start..end]))
}

fn parse_braced(input: &str, dollar: usize) -> Result<(usize, &str, Option<&str>), SplitError> {
    let content_start = dollar + 2;
    let mut cursor = content_start;
    let mut close = None;
    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        if c == '\\' {
            cursor += c.len_utf8();
            if cursor < input.len() {
                let next = input[cursor..].chars().next().unwrap();
                cursor += next.len_utf8();
            }
        } else if c == '}' {
            close = Some(cursor);
            break;
        } else {
            cursor += c.len_utf8();
        }
    }
    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[content_start..close];

    let name_end_rel = contents
        .char_indices()
        .find_map(|(i, c)| if !is_name_continue(c) { Some(i) } else { None })
        .unwrap_or(contents.len());

    let first = contents.chars().next();
    if !first.map(is_name_start).unwrap_or(false) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    let name_end = name_end_rel;
    let name = &contents[..name_end];
    let rest = &contents[name_end..];
    let default = if rest.is_empty() {
        None
    } else if let Some(text) = rest.strip_prefix(":-") {
        Some(text)
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, name, default))
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn select_value(name: &str, default: Option<&str>, env: &[(&str, &str)]) -> String {
    match default {
        None => lookup(env, name).unwrap_or("").to_string(),
        Some(text) => match lookup(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(text),
        },
    }
}

    let bytes = input.as_bytes();
    if p >= bytes.len() || !name_start(bytes[p]) {
        return None;
    }
    let start = p;
    p += 1;
    while p < bytes.len() && name_byte(bytes[p]) {
        p += 1;
    }
    Some((p, &input[start..p]))
}

fn brace(input: &str, dollar: usize) -> Result<(usize, &str, Option<&str>), SplitError> {
    let bytes = input.as_bytes();
    let mut p = dollar + 2;
    while p < bytes.len() {
        if bytes[p] == b'\\' {
            p += 1;
            if p < bytes.len() {
                let c = input[p..].chars().next().unwrap();
                p += c.len_utf8();
            }
        } else if bytes[p] == b'}' {
            break;
        } else {
            let c = input[p..].chars().next().unwrap();
            p += c.len_utf8();
        }
    }
    if p == bytes.len() {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    }

    let body = &input[dollar + 2..p];
    let (name_end, name) = parse_name(body, 0)
        .ok_or(SplitError::InvalidExpansion { pos: dollar })?;
    let tail = &body[name_end..];
    let default = if tail.is_empty() {
        None
    } else if let Some(t) = tail.strip_prefix(":-") {
        Some(t)
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    Ok((p + 1, name, default))
}

fn decoded(s: &str) -> String {
    let mut result = String::new();
    let mut p = 0;
    while p < s.len() {
        let c = s[p..].chars().next().unwrap();
        p += c.len_utf8();
        if c == '\\' && p < s.len() {
            let next = s[p..].chars().next().unwrap();
            result.push(next);
            p += next.len_utf8();
        } else {
            result.push(c);
        }
    }
    result
}

fn expand(name: &str, default: Option<&str>, env: &[(&str, &str)]) -> String {

        None => env_value(env, name).unwrap_or("").to_owned(),
        Some(d) => match env_value(env, name) {
            Some(v) if !v.is_empty() => v.to_owned(),
            _ => decoded(d),
        },
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut out = Vec::new();
    let mut word = Word { text: String::new(), glob: Vec::new() };
    let mut started = false;
    let mut mode = Mode::Bare;
    let mut p = 0;

    while p < input.len() {
        let c = input[p..].chars().next().unwrap();
        let size = c.len_utf8();

        match mode {
            Mode::Single(_) => {
                if c == '\'' {
                    mode = Mode::Bare;
                } else {
                    push(&mut word, &input[p..p + size], false);
                }
                p += size;
            }
            Mode::Double(_) => {
                if c == '"' {
                    mode = Mode::Bare;
                    p += size;
                } else if c == '\\' {
                    let slash = p;
                    p += size;
                    if p == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[p..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            push(&mut word, &input[p..p + next.len_utf8()], false);
                        }
                        p += next.len_utf8();
                    } else {
                        push(&mut word, "\\", false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: p });
                } else if c == '$' {
                    let q = p + size;
                    if input[q..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: p });
                    }
                    if input[q..].starts_with('{') {
                        let (end, name, default) = brace(input, p)?;
                        let val = expand(name, default, env);
                        started = true;
                        push(&mut word, &val, false);
                        p = end;
                    } else if let Some((end, name)) = parse_name(input, q) {
                        started = true;
                        let val = env_value(env, name).unwrap_or("");
                        push(&mut word, val, false);
                        p = end;
                    } else {
                        started = true;
                        push(&mut word, "$", false);
                        p += size;
                    }
                } else {
                    push(&mut word, &input[p..p + size], false);
                    p += size;
                }
            }
            Mode::Bare => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if started {
                        out.push(word);
                        word = Word { text: String::new(), glob: Vec::new() };
                        started = false;
                    }
                    p += size;
                } else if c == '\'' {
                    started = true;
                    mode = Mode::Single(p);
                    p += size;
                } else if c == '"' {
                    started = true;
                    mode = Mode::Double(p);
                    p += size;
                } else if c == '\\' {
                    let slash = p;
                    p += size;
                    if p == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[p..].chars().next().unwrap();
                    if next == '\n' {
                        p += next.len_utf8();
                    } else {
                        started = true;
                        push(&mut word, &input[p..p + next.len_utf8()], false);
                        p += next.len_utf8();
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: p });
                } else if c == '$' {
                    let q = p + size;
                    if input[q..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: p });
                    }
                    if input[q..].starts_with('{') {
                        let (end, name, default) = brace(input, p)?;
                        let val = expand(name, default, env);
                        started = true;
                        push(&mut word, &val, true);
                        p = end;
                    } else if let Some((end, name)) = parse_name(input, q) {
                        started = true;
                        let val = env_value(env, name).unwrap_or("");
                        push(&mut word, val, true);
                        p = end;
                    } else {
                        started = true;
                        push(&mut word, "$", true);
                        p += size;
                    }
                } else {
                    started = true;
                    push(&mut word, &input[p..p + size], true);
                    p += size;
                }
            }
        }
    }

    match mode {
        Mode::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Mode::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Mode::Bare => {}
    }
    if started {
        out.push(word);
    }
    Ok(out)
}

fn expansion(
    src: &str,
    dollar: usize,
    env: &[(&str, &str)],

    while p < src.len() {
        let ch = src[p..].chars().next().unwrap();
        if ch == '\\' {
            p += 1;
            if p < src.len() {
                p += src[p..].chars().next().unwrap().len_utf8();

            p += ch.len_utf8();

    if bytes.first().copied().map(name_start) != Some(true) {

    }

    while n < bytes.len() && name_continue(bytes[n]) {
        n += 1;
    }
    let name = &body[..n];

    let result = if n == bytes.len() {
        val.to_string()
    } else if bytes.get(n..n + 2) == Some(b":-") {
        if val.is_empty() {
            let default = &body[n + 2..];
            let mut decoded = String::new();
            let mut it = default.chars();
            while let Some(ch) = it.next() {
                if ch == '\\' {
                    if let Some(next) = it.next() {
                        decoded.push(next);
                    } else {
                        decoded.push('\\');
                    }
                } else {
                    decoded.push(ch);
                }
            }
            decoded
        } else {
            val.to_string()
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((result, p + 1))
}

                '`' => return Err(SplitError::CommandSubstitution { pos: p }),
                '$' if next < input.len() => {
                    let c = input[next..].chars().next().unwrap();
                    if c == '(' {
                        return Err(SplitError::CommandSubstitution { pos: p });
                    }
                    if c == '{' {
                        let (v, end) = expansion(input, p, env)?;
                        add(&mut current, &v, false);
                        has_word = true;
                        p = end;
                    } else if c == '_' || c.is_ascii_alphabetic() {
                        let mut end = next;
                        while end < input.len() {
                            let b = input.as_bytes()[end];
                            if !name_continue(b) {
                                break;
                            }
                            end += 1;
                        }
                        let name = &input[next..end];
                        let v = env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v).unwrap_or("");
                        add(&mut current, v, false);
                        has_word = true;
                        p = end;
                    } else {
                        add(&mut current, "$", false);
                        has_word = true;
                        p = next;
                    }
                }
                '$' => {
                    add(&mut current, "$", false);
                    has_word = true;
                    p = next;
                }
                _ => {
                    add(&mut current, &input[p..next], false);
                    has_word = true;
                    p = next;
                }
            },
            Mode::Bare => {
                if matches!(ch, ' ' | '\t' | '\r' | '\n') {
                    if has_word {
                        out.push(current);
                        current = Word { text: String::new(), glob: Vec::new() };
                        has_word = false;
                    }
                    p = next;
                } else if matches!(ch, '\'' | '"') {
                    mode = if ch == '\'' { Mode::Single } else { Mode::Double };
                    opened_at = p;
                    has_word = true;
                    p = next;
                } else if ch == '\\' {
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: p });
                    }
                    let escaped = input[next..].chars().next().unwrap();
                    if escaped == '\n' {
                        p = next + 1;
                    } else {
                        add(&mut current, &input[next..next + escaped.len_utf8()], false);
                        has_word = true;
                        p = next + escaped.len_utf8();
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: p });
                } else if ch == '$' && next < input.len() {
                    let c = input[next..].chars().next().unwrap();
                    if c == '(' {
                        return Err(SplitError::CommandSubstitution { pos: p });
                    }
                    if c == '{' {
                        let (v, end) = expansion(input, p, env)?;
                        add(&mut current, &v, true);
                        has_word = true;
                        p = end;
                    } else if c == '_' || c.is_ascii_alphabetic() {
                        let mut end = next;
                        while end < input.len() && name_continue(input.as_bytes()[end]) {
                            end += 1;
                        }
                        let name = &input[next..end];
                        let v = env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v).unwrap_or("");
                        add(&mut current, v, true);
                        has_word = true;
                        p = end;
                    } else {
                        add(&mut current, "$", true);
                        has_word = true;
                        p = next;
                    }
                } else {
                    add(&mut current, &input[p..next], true);
                    has_word = true;
                    p = next;
                }
            }
        }
    }

    match mode {
        Mode::Single => return Err(SplitError::UnterminatedSingle { pos: opened_at }),
        Mode::Double => return Err(SplitError::UnterminatedDouble { pos: opened_at }),
        Mode::Bare => {}
    }
    if has_word {
        out.push(current);
    }
    Ok(out)
}
