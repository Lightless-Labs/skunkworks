#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
    Month,
    Period,
    Credit { index: usize },
    CreditOrder { index: usize },
    InvoiceDiscount,
    InvalidName { index: usize },
    DuplicateName { index: usize },
    Discount,
    CombinedOverflow,

    InvoiceOverflow,

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers.last().unwrap().up_to.is_some() {
        return Err(Error::MissingFinal);
    }
    Ok(())
}

fn checked_band_amount(units: u64, unit: u64, start: u64, index: usize) -> Result<u64, Error> {
    units
        .checked_mul(unit)
        .and_then(|amount| amount.checked_add(start))
        .ok_or(Error::Overflow { index })
}

fn flat_charge(quantity: u64, unit: u64, start: u64) -> Result<Charge, Error> {
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }
    let amount = checked_band_amount(quantity, unit, start, 0)?;
    Ok(Charge {
        amount,
        bands: vec![Band {
            index: 0,
            units: quantity,
            amount,
        }],
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => flat_charge(quantity, *unit, *start),
        Price::Tiers(tiers) => {
            validate_tiers(tiers)?;
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }

            let mut remaining = quantity;
            let mut previous = 0u64;
            let mut amount = 0u64;
            let mut bands = Vec::new();

            for (index, tier) in tiers.iter().enumerate() {
                let units = match tier.up_to {
                    Some(bound) => {
                        let width = bound - previous;
                        previous = bound;
                        remaining.min(width)
                    }
                    None => remaining,
                };

                if units > 0 {
                    let band_amount =
                        checked_band_amount(units, tier.unit, tier.start, index)?;
                    amount = amount
                        .checked_add(band_amount)
                        .ok_or(Error::Overflow { index })?;
                    bands.push(Band {
                        index,
                        units,
                        amount: band_amount,
                    });
                    remaining -= units;
                }
                if remaining == 0 {
                    break;
                }
            }

            Ok(Charge { amount, bands })
        }
    }
}

fn volume_charge(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    if quantity == 0 {
        return Ok(Charge {
            amount: 0,
            bands: Vec::new(),
        });
    }

    let mut selected = tiers.len() - 1;
    for (index, tier) in tiers.iter().enumerate() {
        if tier.up_to.map_or(true, |bound| quantity <= bound) {
            selected = index;
            break;
        }
    }

    let tier = &tiers[selected];
    let amount = checked_band_amount(quantity, tier.unit, tier.start, selected)?;
    Ok(Charge {
        amount,
        bands: vec![Band {
            index: selected,
            units: quantity,
            amount,
        }],
    })
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(tiers), Mode::Volume) => volume_charge(quantity, tiers)?,
        _ => quote(quantity, price)?,
    };
    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .ok_or(Error::CombinedOverflow)?;
    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if !(28..=31).contains(&days) {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut previous_end = start;
    let mut excluded = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end {
            return Err(Error::CreditOrder { index });
        }
        excluded += credit.end - credit.start;
        previous_end = credit.end;
    }

    let active_days = (end - start) - excluded;
    let numerator = (fee as u128) * (active_days as u128);
    let denominator = days as u128;
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            let twice = remainder * 2;
            if twice < denominator {
                quotient
            } else if twice > denominator || quotient % 2 != 0 {
                quotient + 1
            } else {
                quotient
            }
        }
    };
    Ok(Proration {
        active_days,
        amount: rounded as u64,
    })
}

fn discount_amount(base: u64, bps: u32) -> u64 {
    ((base as u128) * (bps as u128) / 10_000u128) as u64
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 32
        && !bytes.iter().any(|byte| {
            *byte <= 0x1f || *byte == 0x7f || *byte == b'|' || *byte == b':'
        })
}

pub fn invoice(
    meters: &[Meter],
    invoice_discount_bps: u32,
    requested_credit: u64,
) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 {
        return Err(Error::InvoiceDiscount);
    }

    for (index, meter) in meters.iter().enumerate() {
        if !valid_name(&meter.name) {
            return Err(Error::InvalidName { index });
        }
        if meters[..index].iter().any(|earlier| earlier.name == meter.name) {
            return Err(Error::DuplicateName { index });
        }
    }

    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;

    for (index, meter) in meters.iter().enumerate() {
        let result = (|| {
            if meter.discount_bps > 10_000 {
                return Err(Error::Discount);
            }
            let bill = bill(
                meter.quantity,
                &meter.price,
                meter.mode.clone(),
                meter.minimum,
            )?;
            let prorated = prorate(
                meter.monthly_fee,
                meter.days,
                meter.start,
                meter.end,
                &meter.credits,
                meter.rounding.clone(),
            )?;
            let base = bill
                .total
                .checked_add(prorated.amount)
                .ok_or(Error::CombinedOverflow)?;
            let discount = discount_amount(base, meter.discount_bps);
            let total = base - discount;
            Ok(InvoiceLine {
                name: meter.name.clone(),
                usage: bill.charge.amount,
                minimum_adjustment: bill.minimum_adjustment,
                monthly: prorated.amount,
                discount,
                total,
            })
        })()
        .map_err(|source| Error::Meter {
            index,
            source: Box::new(source),
        })?;

        subtotal = subtotal
            .checked_add(result.total)
            .ok_or(Error::InvoiceOverflow)?;
        lines.push(result);
    }

    lines.sort_by(|a, b| a.name.as_bytes().cmp(b.name.as_bytes()));

    let discount = discount_amount(subtotal, invoice_discount_bps);
    let after_discount = subtotal - discount;
    let credit = requested_credit.min(after_discount);
    let total = after_discount - credit;

    let mut rendered = String::new();
    for line in &lines {
        rendered.push_str(&line.name.as_bytes().len().to_string());
        rendered.push(':');
        rendered.push_str(&line.name);
        rendered.push('|');
        rendered.push_str(&line.usage.to_string());
        rendered.push('|');
        rendered.push_str(&line.minimum_adjustment.to_string());
        rendered.push('|');
        rendered.push_str(&line.monthly.to_string());
        rendered.push('|');
        rendered.push_str(&line.discount.to_string());
        rendered.push('|');
        rendered.push_str(&line.total.to_string());
        rendered.push('\n');
    }
    rendered.push_str(&format!(
        "TOTAL|{}|{}|{}|{}\n",
        subtotal, discount, credit, total
    ));

    Ok(Invoice {
        lines,
        subtotal,
        discount,
        credit,
        total,
        rendered,
    })
}

pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
    Month,
    Period,
    Credit { index: usize },

fn validate_schedule(tiers: &[Tier]) -> Result<(), Error> {
    match tiers.len() {
        0 => return Err(Error::Empty),
        65.. => return Err(Error::TooMany),
        _ => {}
    }

    let mut last_finite = 0;
    for (i, tier) in tiers.iter().enumerate() {
        if let Some(bound) = tier.up_to {
            if bound <= last_finite {
                return Err(Error::Bound { index: i });
            }
            last_finite = bound;
        } else if i != tiers.len() - 1 {
            return Err(Error::Unbounded { index: i });
        }
    }
    if tiers[tiers.len() - 1].up_to.is_some() {
        Err(Error::MissingFinal)
    } else {
        Ok(())
    }
}

fn amount(units: u64, tier: &Tier, index: usize) -> Result<u64, Error> {
    units
        .checked_mul(tier.unit)
        .and_then(|n| n.checked_add(tier.start))
        .ok_or(Error::Overflow { index })
}

fn graduated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_schedule(tiers)?;
    let mut bands = Vec::new();
    let mut total = 0u64;
    let mut low = 0u64;
    let mut left = quantity;

    for (i, tier) in tiers.iter().enumerate() {
        let allocated = match tier.up_to {
            Some(high) => {
                let width = high - low;
                low = high;
                left.min(width)
            }
            None => left,
        };
        if allocated != 0 {
            let charge = amount(allocated, tier, i)?;
            total = total.checked_add(charge).ok_or(Error::Overflow { index: i })?;
            bands.push(Band {
                index: i,
                units: allocated,
                amount: charge,
            });
            left -= allocated;
        }
        if left == 0 {
            break;
        }
    }
    Ok(Charge { amount: total, bands })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge { amount: 0, bands: vec![] });
            }
            let charge = quantity.checked_mul(*unit)
                .and_then(|n| n.checked_add(*start))
                .ok_or(Error::Overflow { index: 0 })?;
            Ok(Charge {
                amount: charge,
                bands: vec![Band { index: 0, units: quantity, amount: charge }],
            })
        }
        Price::Tiers(tiers) => graduated(quantity, tiers),
    }
}

fn volume(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_schedule(tiers)?;
    if quantity == 0 {
        return Ok(Charge { amount: 0, bands: vec![] });
    }
    let index = tiers.iter().position(|tier| {
        tier.up_to.map_or(true, |bound| quantity <= bound)
    }).unwrap();
    let selected = &tiers[index];
    let charge = amount(quantity, selected, index)?;
    Ok(Charge {
        amount: charge,
        bands: vec![Band { index, units: quantity, amount: charge }],
    })
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(tiers), Mode::Volume) => volume(quantity, tiers)?,
        _ => quote(quantity, price)?,
    };
    let minimum_adjustment = if quantity > 0 && minimum > charge.amount {
        minimum - charge.amount
    } else {
        0
    };
    let total = charge.amount.checked_add(minimum_adjustment)
        .ok_or(Error::CombinedOverflow)?;
    Ok(Bill { charge, minimum_adjustment, total })
}

pub fn prorate(
    fee: u64,
    days: u32,
    start: u32,
    end: u32,
    credits: &[DayRange],
    rounding: Rounding,
) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 {
        return Err(Error::Month);
    }
    if start > end || end > days {
        return Err(Error::Period);
    }

    let mut active = end - start;
    let mut previous_end = start;
    for (i, range) in credits.iter().enumerate() {
        if range.start < start || range.start >= range.end || range.end > end {
            return Err(Error::Credit { index: i });
        }
        if range.start < previous_end {
            return Err(Error::CreditOrder { index: i });
        }
        active -= range.end - range.start;
        previous_end = range.end;
    }

    let numerator = (fee as u128) * (active as u128);
    let divisor = days as u128;
    let floor = numerator / divisor;
    let remainder = numerator % divisor;
    let value = match rounding {
        Rounding::Down => floor,
        Rounding::Up if remainder != 0 => floor + 1,
        Rounding::Up => floor,
        Rounding::HalfEven => {
            let twice = remainder * 2;
            if twice > divisor || (twice == divisor && floor % 2 == 1) {
                floor + 1
            } else {
                floor
            }
        }
    };
    Ok(Proration { active_days: active, amount: value as u64 })
}

fn basis_points(value: u64, rate: u32) -> u64 {
    ((value as u128 * rate as u128) / 10_000) as u64
}

fn name_valid(name: &str) -> bool {
    let bytes = name.as_bytes();
    (1..=32).contains(&bytes.len())
        && bytes.iter().all(|b| *b > 0x1f && *b != 0x7f && *b != b'|' && *b != b':')
}

pub fn invoice(
    meters: &[Meter],
    invoice_discount_bps: u32,
    credit: u64,
) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 {
        return Err(Error::InvoiceDiscount);
    }

    for (i, meter) in meters.iter().enumerate() {
        if !name_valid(&meter.name) {
            return Err(Error::InvalidName { index: i });
        }
        if meters[..i].iter().any(|prior| prior.name == meter.name) {
            return Err(Error::DuplicateName { index: i });
        }
    }

    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;
    for (i, meter) in meters.iter().enumerate() {
        let line_result = (|| {
            if meter.discount_bps > 10_000 {
                return Err(Error::Discount);
            }
            let b = bill(meter.quantity, &meter.price, meter.mode.clone(), meter.minimum)?;
            let p = prorate(
                meter.monthly_fee,
                meter.days,
                meter.start,
                meter.end,
                &meter.credits,
                meter.rounding.clone(),
            )?;
            let base = b.total.checked_add(p.amount).ok_or(Error::CombinedOverflow)?;
            let d = basis_points(base, meter.discount_bps);
            Ok(InvoiceLine {
                name: meter.name.clone(),
                usage: b.charge.amount,
                minimum_adjustment: b.minimum_adjustment,
                monthly: p.amount,
                discount: d,
                total: base - d,
            })
        })().map_err(|source| Error::Meter { index: i, source: Box::new(source) })?;

        subtotal = subtotal.checked_add(line_result.total).ok_or(Error::InvoiceOverflow)?;
        lines.push(line_result);
    }

    lines.sort_by(|left, right| left.name.as_bytes().cmp(right.name.as_bytes()));
    let discount = basis_points(subtotal, invoice_discount_bps);
    let remaining = subtotal - discount;
    let applied_credit = credit.min(remaining);
    let total = remaining - applied_credit;

    let mut rendered = String::new();
    for line in &lines {
        rendered.push_str(&line.name.as_bytes().len().to_string());
        rendered.push(':');
        rendered.push_str(&line.name);
        rendered.push('|');
        rendered.push_str(&line.usage.to_string());
        rendered.push('|');
        rendered.push_str(&line.minimum_adjustment.to_string());
        rendered.push('|');
        rendered.push_str(&line.monthly.to_string());
        rendered.push('|');
        rendered.push_str(&line.discount.to_string());
        rendered.push('|');
        rendered.push_str(&line.total.to_string());
        rendered.push('\n');
    }
    rendered.push_str("TOTAL|");
    rendered.push_str(&subtotal.to_string());
    rendered.push('|');
    rendered.push_str(&discount.to_string());
    rendered.push('|');
    rendered.push_str(&applied_credit.to_string());
    rendered.push('|');
    rendered.push_str(&total.to_string());
    rendered.push('\n');

    Ok(Invoice {
        lines,
        subtotal,
        discount,
        credit: applied_credit,
        total,
        rendered,
    })
}
