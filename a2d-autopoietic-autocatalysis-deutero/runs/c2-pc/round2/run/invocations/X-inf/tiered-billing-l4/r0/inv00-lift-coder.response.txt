#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error { Empty, TooMany, Bound { index: usize }, Unbounded { index: usize }, MissingFinal, Overflow { index: usize }, Month, Period, Credit { index: usize }, CreditOrder { index: usize }, InvoiceDiscount, InvalidName { index: usize }, DuplicateName { index: usize }, Discount, CombinedOverflow, Meter { index: usize, source: Box<Error> }, InvoiceOverflow }

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() { return Err(Error::Empty); }
    if tiers.len() > 64 { return Err(Error::TooMany); }
    let mut previous = 0u64;
    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous { return Err(Error::Bound { index }); }
                previous = bound;
            }
            None => {
                if index + 1 != tiers.len() { return Err(Error::Unbounded { index }); }
            }
        }
    }
    if tiers.last().unwrap().up_to.is_some() { return Err(Error::MissingFinal); }
    Ok(())
}

fn band_charge(index: usize, units: u64, unit: u64, start: u64) -> Result<u64, Error> {
    let amount = units.checked_mul(unit).and_then(|v| v.checked_add(start))
        .ok_or(Error::Overflow { index })?;
    Ok(amount)
}

fn graduated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    let mut remaining = quantity;
    let mut previous = 0u64;
    let mut amount = 0u64;
    let mut bands = Vec::new();
    for (index, tier) in tiers.iter().enumerate() {
        if remaining == 0 { break; }
        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous;
                previous = bound;
                remaining.min(width)
            }
            None => remaining,
        };
        if units > 0 {
            let charge = band_charge(index, units, tier.unit, tier.start)?;
            amount = amount.checked_add(charge).ok_or(Error::Overflow { index })?;
            bands.push(Band { index, units, amount: charge });
            remaining -= units;
        }
    }
    Ok(Charge { amount, bands })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 { return Ok(Charge { amount: 0, bands: Vec::new() }); }
            let amount = band_charge(0, quantity, *unit, *start)?;
            Ok(Charge { amount, bands: vec![Band { index: 0, units: quantity, amount }] })
        }
        Price::Tiers(tiers) => graduated(quantity, tiers),
    }
}

fn volume(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;
    if quantity == 0 { return Ok(Charge { amount: 0, bands: Vec::new() }); }
    let (index, tier) = tiers.iter().enumerate().find(|(_, tier)| {
        tier.up_to.map_or(true, |bound| bound >= quantity)
    }).unwrap();
    let amount = band_charge(index, quantity, tier.unit, tier.start)?;
    Ok(Charge { amount, bands: vec![Band { index, units: quantity, amount }] })
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(tiers), Mode::Volume) => volume(quantity, tiers)?,
        _ => quote(quantity, price)?,
    };
    let minimum_adjustment = if quantity == 0 { 0 } else { minimum.saturating_sub(charge.amount) };
    let total = charge.amount.checked_add(minimum_adjustment).ok_or(Error::CombinedOverflow)?;
    Ok(Bill { charge, minimum_adjustment, total })
}

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {
    if !(28..=31).contains(&days) { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }
    let mut previous_end = start;
    let mut excluded = 0u32;
    for (index, credit) in credits.iter().enumerate() {
        if credit.start < start || credit.start >= credit.end || credit.end > end {
            return Err(Error::Credit { index });
        }
        if credit.start < previous_end {
            return Err(Error::CreditOrder { index });
        }
        excluded += credit.end - credit.start;
        previous_end = credit.end;
    }
    let active_days = end - start - excluded;
    let product = (fee as u128) * (active_days as u128);
    let divisor = days as u128;
    let quotient = product / divisor;
    let remainder = product % divisor;
    let rounded = match rounding {
        Rounding::Down => quotient,
        Rounding::Up => quotient + u128::from(remainder != 0),
        Rounding::HalfEven => {
            if remainder * 2 > divisor || (remainder * 2 == divisor && quotient % 2 == 1) {
                quotient + 1
            } else { quotient }
        }
    };
    Ok(Proration { active_days, amount: rounded as u64 })
}

fn discount_amount(base: u64, bps: u32) -> u64 {
    ((base as u128 * bps as u128) / 10_000u128) as u64
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty() && bytes.len() <= 32 &&
        !bytes.iter().any(|b| *b <= 0x1f || *b == 0x7f || *b == b'|' || *b == b':')
}

pub fn invoice(meters: &[Meter], invoice_discount_bps: u32, credit: u64) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10_000 { return Err(Error::InvoiceDiscount); }
    for (index, meter) in meters.iter().enumerate() {
        if !valid_name(&meter.name) { return Err(Error::InvalidName { index }); }
        if meters[..index].iter().any(|prior| prior.name == meter.name) {
            return Err(Error::DuplicateName { index });
        }
    }

    let mut lines = Vec::with_capacity(meters.len());
    let mut subtotal = 0u64;
    for (index, meter) in meters.iter().enumerate() {
        let result = (|| {
            if meter.discount_bps > 10_000 { return Err(Error::Discount); }
            let bill = bill(meter.quantity, &meter.price, meter.mode.clone(), meter.minimum)?;
            let monthly = prorate(meter.monthly_fee, meter.days, meter.start, meter.end, &meter.credits, meter.rounding.clone())?.amount;
            let base = bill.total.checked_add(monthly).ok_or(Error::CombinedOverflow)?;
            let discount = discount_amount(base, meter.discount_bps);
            Ok(InvoiceLine {
                name: meter.name.clone(),
                usage: bill.charge.amount,
                minimum_adjustment: bill.minimum_adjustment,
                monthly,
                discount,
                total: base - discount,
            })
        })().map_err(|source| Error::Meter { index, source: Box::new(source) })?;
        subtotal = subtotal.checked_add(result.total).ok_or(Error::InvoiceOverflow)?;
        lines.push(result);
    }
    lines.sort_by(|a, b| a.name.as_bytes().cmp(b.name.as_bytes()));
    let discount = discount_amount(subtotal, invoice_discount_bps);
    let credit = credit.min(subtotal - discount);
    let total = subtotal - discount - credit;
    let mut rendered = String::new();
    for line in &lines {
        use std::fmt::Write;
        writeln!(&mut rendered, "{}:{}|{}|{}|{}|{}|{}",
            line.name.as_bytes().len(), line.name, line.usage, line.minimum_adjustment,
            line.monthly, line.discount, line.total).unwrap();
    }
    use std::fmt::Write;
    writeln!(&mut rendered, "TOTAL|{}|{}|{}|{}", subtotal, discount, credit, total).unwrap();
    Ok(Invoice { lines, subtotal, discount, credit, total, rendered })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Invoice {
    pub lines: Vec<InvoiceLine>, pub subtotal: u64, pub discount: u64,
    pub credit: u64, pub total: u64, pub rendered: String,
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error { Empty, TooMany, Bound { index: usize }, Unbounded { index: usize }, MissingFinal, Overflow { index: usize }, Month, Period, Credit { index: usize }, CreditOrder { index: usize }, InvoiceDiscount, InvalidName { index: usize }, DuplicateName { index: usize }, Discount, CombinedOverflow, Meter { index: usize, source: Box<Error> }, InvoiceOverflow }

fn check_schedule(ts: &[Tier]) -> Result<(), Error> {
    if ts.len() == 0 { return Err(Error::Empty); }
    if ts.len() > 64 { return Err(Error::TooMany); }
    let mut bound = 0;
    for (i, t) in ts.iter().enumerate() {
        if let Some(b) = t.up_to {
            if b <= bound { return Err(Error::Bound { index: i }); }
            bound = b;
        } else if i + 1 != ts.len() {
            return Err(Error::Unbounded { index: i });
        }
    }
    if ts[ts.len() - 1].up_to.is_some() { Err(Error::MissingFinal) } else { Ok(()) }
}

fn compute(index: usize, units: u64, t: &Tier) -> Result<u64, Error> {
    units.checked_mul(t.unit).and_then(|x| x.checked_add(t.start))
        .ok_or(Error::Overflow { index })
}

fn graduated_charge(q: u64, ts: &[Tier]) -> Result<Charge, Error> {
    check_schedule(ts)?;
    let mut left = q;
    let mut low = 0u64;
    let mut sum = 0u64;
    let mut out = Vec::new();
    for (i, t) in ts.iter().enumerate() {
        let width = match t.up_to { Some(high) => { let w = high - low; low = high; w }, None => u64::MAX };
        let n = left.min(width);
        if n != 0 {
            let c = compute(i, n, t)?;
            sum = sum.checked_add(c).ok_or(Error::Overflow { index: i })?;
            out.push(Band { index: i, units: n, amount: c });
            left -= n;
        }
        if left == 0 { break; }
    }
    Ok(Charge { amount: sum, bands: out })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 { return Ok(Charge { amount: 0, bands: vec![] }); }
            let t = Tier { up_to: None, unit: *unit, start: *start };
            let amount = compute(0, quantity, &t)?;
            Ok(Charge { amount, bands: vec![Band { index: 0, units: quantity, amount }] })
        }
        Price::Tiers(ts) => graduated_charge(quantity, ts),
    }
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(ts), Mode::Volume) => {
            check_schedule(ts)?;
            if quantity == 0 { Charge { amount: 0, bands: vec![] } }
            else {
                let i = ts.iter().position(|t| t.up_to.map_or(true, |b| b >= quantity)).unwrap();
                let amount = compute(i, quantity, &ts[i])?;
                Charge { amount, bands: vec![Band { index: i, units: quantity, amount }] }
            }
        }
        _ => quote(quantity, price)?,
    };
    let adjustment = if quantity > 0 { minimum.saturating_sub(charge.amount) } else { 0 };
    let total = charge.amount.checked_add(adjustment).ok_or(Error::CombinedOverflow)?;
    Ok(Bill { charge, minimum_adjustment: adjustment, total })
}

pub fn prorate(fee: u64, days: u32, start: u32, end: u32, credits: &[DayRange], rounding: Rounding) -> Result<Proration, Error> {
    if days != 28 && days != 29 && days != 30 && days != 31 { return Err(Error::Month); }
    if start > end || end > days { return Err(Error::Period); }
    let mut cut = 0u32;
    let mut last = start;
    for (i, r) in credits.iter().enumerate() {
        if r.start < start || r.start >= r.end || r.end > end { return Err(Error::Credit { index: i }); }
        if r.start < last { return Err(Error::CreditOrder { index: i }); }
        cut += r.end - r.start;
        last = r.end;
    }
    let active = end - start - cut;
    let num = fee as u128 * active as u128;
    let den = days as u128;
    let q = num / den;
    let r = num % den;
    let value = match rounding {
        Rounding::Down => q,
        Rounding::Up => if r == 0 { q } else { q + 1 },
        Rounding::HalfEven => match (r * 2).cmp(&den) {
            std::cmp::Ordering::Less => q,
            std::cmp::Ordering::Greater => q + 1,
            std::cmp::Ordering::Equal => if q % 2 == 0 { q } else { q + 1 },
        },
    };
    Ok(Proration { active_days: active, amount: value as u64 })
}

fn pct(x: u64, bps: u32) -> u64 { ((x as u128 * bps as u128) / 10000) as u64 }

fn name_ok(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty() && b.len() <= 32 && !b.iter().any(|c| *c <= 31 || *c == 127 || *c == b'|' || *c == b':')
}

pub fn invoice(meters: &[Meter], invoice_discount_bps: u32, requested_credit: u64) -> Result<Invoice, Error> {
    if invoice_discount_bps > 10000 { return Err(Error::InvoiceDiscount); }
    for (i, m) in meters.iter().enumerate() {
        if !name_ok(&m.name) { return Err(Error::InvalidName { index: i }); }
        if meters[..i].iter().any(|p| p.name == m.name) { return Err(Error::DuplicateName { index: i }); }
    }
    let mut lines = Vec::new();
    let mut subtotal = 0u64;
    for (i, m) in meters.iter().enumerate() {
        let line = (|| -> Result<InvoiceLine, Error> {
            if m.discount_bps > 10000 { return Err(Error::Discount); }
            let b = bill(m.quantity, &m.price, m.mode.clone(), m.minimum)?;
            let p = prorate(m.monthly_fee, m.days, m.start, m.end, &m.credits, m.rounding.clone())?;
            let base = b.total.checked_add(p.amount).ok_or(Error::CombinedOverflow)?;
            let d = pct(base, m.discount_bps);
            Ok(InvoiceLine { name: m.name.clone(), usage: b.charge.amount, minimum_adjustment: b.minimum_adjustment, monthly: p.amount, discount: d, total: base - d })
        })().map_err(|source| Error::Meter { index: i, source: Box::new(source) })?;
        subtotal = subtotal.checked_add(line.total).ok_or(Error::InvoiceOverflow)?;
        lines.push(line);
    }
    lines.sort_by(|x, y| x.name.as_bytes().cmp(y.name.as_bytes()));
    let discount = pct(subtotal, invoice_discount_bps);
    let applied_credit = requested_credit.min(subtotal - discount);
    let total = subtotal - discount - applied_credit;
    let mut rendered = String::new();
    for l in &lines {
        use std::fmt::Write;
        writeln!(&mut rendered, "{}:{}|{}|{}|{}|{}|{}", l.name.len(), l.name, l.usage, l.minimum_adjustment, l.monthly, l.discount, l.total).unwrap();
    }
    use std::fmt::Write;
    writeln!(&mut rendered, "TOTAL|{}|{}|{}|{}", subtotal, discount, applied_credit, total).unwrap();
    Ok(Invoice { lines, subtotal, discount, credit: applied_credit, total, rendered })
}
