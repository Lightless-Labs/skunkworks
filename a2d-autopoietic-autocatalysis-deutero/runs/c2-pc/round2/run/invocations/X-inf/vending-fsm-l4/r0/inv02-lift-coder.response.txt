use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
    pub maintenance: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
    MaintenanceActive,
    MaintenanceRequired,
    AlreadyMode,
    ReconcileMismatch { coin: u16, expected: u16, found: u16 },
    ValueDrift { before: u64, after: u64 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
    Maintenance { enabled: bool },
    Reconciled { before: [u16; 4], after: [u16; 4] },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Audit {
    pub inserted: u64,
    pub refunded: u64,
    pub sales: u64,
    pub change: u64,
    pub adjustment: i64,
    pub cash: u64,
}

pub struct Machine {
    products: Vec<Product>,
    index: HashMap<String, usize>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    inserted: u64,
    refunded: u64,
    sales: u64,
    change: u64,
    adjustment: i64,
}

const COINS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashMap::<String, ()>::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains_key(&product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.insert(product.id.clone(), ());
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (i, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }

        let index = products
            .iter()
            .enumerate()
            .map(|(i, p)| (p.id.clone(), i))
            .collect();

        Ok(Self {
            products,
            index,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !COINS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if self.credit as u32 + coin as u32 > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        self.inserted = self.inserted.saturating_add(coin as u64);
        self.log(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(_) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        self.log(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let i = self.index[&id];
        let product = &self.products[i];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let chosen = choose_change(amount, self.tubes).ok_or(Error::NoChange { amount })?;
        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= chosen[i];
        }
        for coin in &self.escrow {
            let i = coin_index(*coin).expect("escrow only contains accepted coins");
            proposed[i] += 1;
        }
        for (i, count) in proposed.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }

        let mut change_coins = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..chosen[i] {
                change_coins.push(COINS[i]);
            }
        }

        self.tubes = proposed;
        self.products[i].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(price as u64);
        self.change = self.change.saturating_add(amount as u64);
        self.log(EventKind::Vended {
            id: id.clone(),
            price,
            change: change_coins.clone(),
        });
        Ok(Receipt {
            id,
            price,
            change: change_coins,
        })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.refunded = self.refunded.saturating_add(self.credit as u64);
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut result = self.products.clone();
        result.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        result
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&i) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[i].price;
        self.products[i].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&i) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = self.products[i].stock as u32 + count as u32;
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[i].stock = new_stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: new_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        let tube_value = self
            .tubes
            .iter()
            .zip(COINS.iter())
            .map(|(count, coin)| *count as u64 * *coin as u64)
            .sum::<u64>();
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value + self.credit as u64,
        }
    }

    pub fn reconcile(
        &mut self,
        expected: [u16; 4],
        actual: [u16; 4],
    ) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: COINS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }
        let before = tube_value(self.tubes);
        let after = tube_value(actual);
        if before.abs_diff(after) > 20 {
            return Err(Error::ValueDrift { before, after });
        }

        self.tubes = actual;
        let delta = after as i128 - before as i128;
        self.adjustment = saturating_add_i128_to_i64(self.adjustment, delta);
        self.log(EventKind::Reconciled {
            before: expected,
            after: actual,
        });
        Ok(())
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn coin_index(coin: u16) -> Option<usize> {
    COINS.iter().position(|c| *c == coin)
}

fn tube_value(tubes: [u16; 4]) -> u64 {
    tubes
        .iter()
        .zip(COINS.iter())
        .map(|(count, coin)| *count as u64 * *coin as u64)
        .sum()
}

fn saturating_add_i128_to_i64(value: i64, delta: i128) -> i64 {
    (value as i128 + delta).clamp(i64::MIN as i128, i64::MAX as i128) as i64
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / COINS[i]);
        greedy[i] = take;
        remaining -= take * COINS[i];
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<([u16; 4], u16)> = None;
    for c10 in 0..=tubes[3].min(amount / 10) {
        for c4 in 0..=tubes[2].min(amount / 4) {
            let used = c10 * 10 + c4 * 4;
            if used > amount {
                continue;
            }
            for c3 in 0..=tubes[1].min((amount - used) / 3) {
                let used2 = used + c3 * 3;
                if used2 > amount {
                    continue;
                }
                let c1 = amount - used2;
                if c1 > tubes[0] {
                    continue;
                }
                let candidate = [c1, c3, c4, c10];
                let count = c1 + c3 + c4 + c10;
                let replace = match best {
                    None => true,
                    Some((old, old_count)) => {
                        count < old_count
                            || (count == old_count
                                && (c10, c4, c3, c1)
                                    > (old[3], old[2], old[1], old[0]))
                    }
                };
                if replace {
                    best = Some((candidate, count));
                }
            }
        }
    }
    best.map(|(counts, _)| counts)
}

pub struct Machine {
    catalog: BTreeMap<String, Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    audit_inserted: u64,
    audit_refunded: u64,
    audit_sales: u64,
    audit_change: u64,
    audit_adjustment: i64,
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut catalog = BTreeMap::new();
        for product in products {
            if !(1..=8).contains(&product.id.len())
                || !product.id.bytes().all(|b| b.is_ascii_lowercase())
            {
                return Err(Error::BadId { id: product.id });
            }
            if catalog.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit { id: product.id });
            }
            catalog.insert(product.id.clone(), product);
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            catalog,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            audit_inserted: 0,
            audit_refunded: 0,
            audit_sales: 0,
            audit_change: 0,
            audit_adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !DENOMS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        self.audit_inserted = self.audit_inserted.saturating_add(u64::from(coin));
        self.add_event(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !self.catalog.contains_key(id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        self.add_event(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let product = self.catalog.get(&id).unwrap();
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        let amount = self.credit - price;
        let counts = find_change(amount, self.tubes).ok_or(Error::NoChange { amount })?;
        let mut next_tubes = self.tubes;
        for i in 0..4 {
            next_tubes[i] -= counts[i];
        }
        for &coin in &self.escrow {
            let idx = DENOMS.iter().position(|&d| d == coin).unwrap();
            next_tubes[idx] += 1;
        }
        for i in 0..4 {
            if next_tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..counts[i] {
                change.push(DENOMS[i]);
            }
        }

        let product = self.catalog.get_mut(&id).unwrap();
        product.stock -= 1;
        self.tubes = next_tubes;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.audit_sales = self.audit_sales.saturating_add(u64::from(price));
        self.audit_change = self.audit_change.saturating_add(u64::from(amount));
        self.add_event(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins = self.escrow.iter().rev().copied().collect::<Vec<_>>();
        self.audit_refunded = self.audit_refunded.saturating_add(u64::from(self.credit));
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.add_event(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        self.catalog.values().cloned().collect()

        let product = self.catalog.get(id).unwrap();
        let stock = u32::from(product.stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });

        let product = self.catalog.get_mut(id).unwrap();
        product.stock = stock as u16;
        self.add_event(EventKind::Restocked {

            added: count,
            stock: stock as u16,

        Ok(())

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {

        if self.credit != 0 {
            return Err(Error::Busy);

        self.selected = None;
        self.add_event(EventKind::Maintenance { enabled });
        Ok(())

    pub fn audit(&self) -> Audit {
        let cash = self
            .tubes
            .iter()
            .zip(DENOMS.iter())
            .map(|(&n, &d)| u64::from(n) * u64::from(d))
            .sum::<u64>()
            + u64::from(self.credit);
        Audit {
            inserted: self.audit_inserted,
            refunded: self.audit_refunded,
            sales: self.audit_sales,
            change: self.audit_change,
            adjustment: self.audit_adjustment,
            cash,
        }
    }

    pub fn reconcile(
        &mut self,
        expected: [u16; 4],
        actual: [u16; 4],
    ) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let before = value(self.tubes);
        let after = value(actual);
        if before.abs_diff(after) > 20 {
            return Err(Error::ValueDrift { before, after });
        }

        self.tubes = actual;
        let adjustment = i128::from(self.audit_adjustment) + i128::from(after) - i128::from(before);
        self.audit_adjustment = adjustment.clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        self.add_event(EventKind::Reconciled {
            before: self.tubes_before_reconcile(expected),
            after: actual,
        });
        Ok(())
    }

    fn tubes_before_reconcile(&self, previous: [u16; 4]) -> [u16; 4] {
        previous
    }

    fn add_event(&mut self, kind: EventKind) {
        self.events.push(Event {
            seq: self.events.len(),
            kind,
        });
    }
}

fn value(tubes: [u16; 4]) -> u64 {
    tubes
        .iter()
        .zip(DENOMS.iter())
        .map(|(&n, &d)| u64::from(n) * u64::from(d))
        .sum()
}

fn find_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut remaining = amount;
    let mut greedy = [0u16; 4];
    for i in (0..4).rev() {
        greedy[i] = tubes[i].min(remaining / DENOMS[i]);
        remaining -= greedy[i] * DENOMS[i];
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<([u16; 4], u16)> = None;
    for ten in 0..=tubes[3].min(amount / 10) {
        for four in 0..=tubes[2].min(amount / 4) {
            let used_a = ten * 10 + four * 4;
            if used_a > amount {
                continue;
            }
            for three in 0..=tubes[1].min((amount - used_a) / 3) {
                let used_b = used_a + three * 3;
                let one = amount - used_b;
                if one > tubes[0] {
                    continue;
                }
                let candidate = [one, three, four, ten];
                let count = one + three + four + ten;
                if let Some((old, old_count)) = best {
                    if count > old_count
                        || (count == old_count
                            && (ten, four, three, one)
                                <= (old[3], old[2], old[1], old[0]))
                    {
                        continue;
                    }
                }
                best = Some((candidate, count));
            }
        }
    }
    best.map(|(counts, _)| counts)
}
