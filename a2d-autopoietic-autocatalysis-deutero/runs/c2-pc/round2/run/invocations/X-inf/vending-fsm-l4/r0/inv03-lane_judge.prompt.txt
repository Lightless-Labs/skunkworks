pub struct Machine {
    products: Vec<Product>,
    index: HashMap<String, usize>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    inserted: u64,
    refunded: u64,
    sales: u64,
    change: u64,
    adjustment: i64,
}

const COINS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashMap::<String, ()>::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains_key(&product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.insert(product.id.clone(), ());
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
        }

        for (i, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }

        let index = products
            .iter()
            .enumerate()
            .map(|(i, p)| (p.id.clone(), i))
            .collect();

        Ok(Self {
            products,
            index,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !COINS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if self.credit as u32 + coin as u32 > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        self.inserted = self.inserted.saturating_add(coin as u64);
        self.log(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(_) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        self.log(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let i = self.index[&id];
        let product = &self.products[i];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let chosen = choose_change(amount, self.tubes).ok_or(Error::NoChange { amount })?;
        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= chosen[i];
        }
        for coin in &self.escrow {
            let i = coin_index(*coin).expect("escrow only contains accepted coins");
            proposed[i] += 1;
        }
        for (i, count) in proposed.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }

        let mut change_coins = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..chosen[i] {
                change_coins.push(COINS[i]);
            }
        }

        self.tubes = proposed;
        self.products[i].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(price as u64);
        self.change = self.change.saturating_add(amount as u64);
        self.log(EventKind::Vended {
            id: id.clone(),
            price,
            change: change_coins.clone(),
        });
        Ok(Receipt {
            id,
            price,
            change: change_coins,
        })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.refunded = self.refunded.saturating_add(self.credit as u64);
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut result = self.products.clone();
        result.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        result
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&i) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[i].price;
        self.products[i].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&i) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = self.products[i].stock as u32 + count as u32;
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[i].stock = new_stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: new_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        let tube_value = self
            .tubes
            .iter()
            .zip(COINS.iter())
            .map(|(count, coin)| *count as u64 * *coin as u64)
            .sum::<u64>();
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value + self.credit as u64,
        }
    }

    pub fn reconcile(
        &mut self,
        expected: [u16; 4],
        actual: [u16; 4],
    ) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: COINS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }
        let before = tube_value(self.tubes);
        let after = tube_value(actual);
        if before.abs_diff(after) > 20 {
            return Err(Error::ValueDrift { before, after });
        }

        self.tubes = actual;
        let delta = after as i128 - before as i128;
        self.adjustment = saturating_add_i128_to_i64(self.adjustment, delta);
        self.log(EventKind::Reconciled {
            before: expected,
            after: actual,
        });
        Ok(())
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn coin_index(coin: u16) -> Option<usize> {
    COINS.iter().position(|c| *c == coin)
}

fn tube_value(tubes: [u16; 4]) -> u64 {
    tubes
        .iter()
        .zip(COINS.iter())
        .map(|(count, coin)| *count as u64 * *coin as u64)
        .sum()
}

fn saturating_add_i128_to_i64(value: i64, delta: i128) -> i64 {
    (value as i128 + delta).clamp(i64::MIN as i128, i64::MAX as i128) as i64
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / COINS[i]);
        greedy[i] = take;
        remaining -= take * COINS[i];
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<([u16; 4], u16)> = None;
    for c10 in 0..=tubes[3].min(amount / 10) {
        for c4 in 0..=tubes[2].min(amount / 4) {
            let used = c10 * 10 + c4 * 4;
            if used > amount {
                continue;
            }
            for c3 in 0..=tubes[1].min((amount - used) / 3) {
                let used2 = used + c3 * 3;
                if used2 > amount {
                    continue;
                }
                let c1 = amount - used2;
                if c1 > tubes[0] {
                    continue;
                }
                let candidate = [c1, c3, c4, c10];
                let count = c1 + c3 + c4 + c10;
                let replace = match best {
                    None => true,
                    Some((old, old_count)) => {
                        count < old_count
                            || (count == old_count
                                && (c10, c4, c3, c1)
                                    > (old[3], old[2], old[1], old[0]))
                    }
                };
                if replace {
                    best = Some((candidate, count));
                }
            }
        }
    }
    best.map(|(counts, _)| counts)
}

    pub fn state(&self) -> State {

    }
    fn log(&mut self, kind: EventKind) { self.events.push(Event { seq: self.events.len(), kind }); }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {

        if idx(coin).is_none() { return Err(Error::BadCoin { coin }); }
        if u32::from(self.credit) + u32::from(coin) > 100 { return Err(Error::CreditLimit); }
        self.credit += coin; self.escrow.push(coin);
        self.audit.inserted = self.audit.inserted.saturating_add(u64::from(coin));
        self.log(EventKind::Inserted { coin, credit: self.credit }); Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance { return Err(Error::MaintenanceActive); }
        let Some(i) = self.products.iter().position(|p| p.id == id) else { return Err(Error::UnknownProduct { id: id.to_owned() }); };
        if self.selected == Some(i) { return Err(Error::SameSelection); }
        self.selected = Some(i); self.log(EventKind::Selected { id: id.to_owned() }); Ok(())
    }

    fn change_coins(&self, amount: u16) -> Result<Vec<u16>, Error> {
        if amount == 0 { return Ok(Vec::new()); }
        let mut g = [0u16; 4]; let mut rem = amount;
        for i in (0..4).rev() { g[i] = self.tubes[i].min(rem / D[i]); rem -= g[i] * D[i]; }
        let chosen = if rem == 0 { g } else {
            let mut best: Option<[u16;4]> = None; let mut best_n = u32::MAX;
            fn visit(i: usize, rem: u16, counts: &mut [u16;4], tubes: &[u16;4], best: &mut Option<[u16;4]>, best_n: &mut u32) {
                if i == 4 {
                    if rem != 0 { return; }
                    let n: u32 = counts.iter().map(|&x| u32::from(x)).sum();
                    let lex_better = best.map_or(true, |b| (counts[3],counts[2],counts[1],counts[0]) > (b[3],b[2],b[1],b[0]));
                    if n < *best_n || (n == *best_n && lex_better) { *best = Some(*counts); *best_n = n; }
                    return;
                }
                let max = tubes[i].min(rem / D[i]);
                for n in 0..=max { counts[i] = n; visit(i+1, rem-n*D[i], counts, tubes, best, best_n); }
                counts[i] = 0;
            }
            let mut c = [0u16;4]; visit(0, amount, &mut c, &self.tubes, &mut best, &mut best_n);
            best.ok_or(Error::NoChange { amount })?
        };
        let mut out = Vec::new();
        for i in (0..4).rev() { for _ in 0..chosen[i] { out.push(D[i]); } }
        Ok(out)
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance { return Err(Error::MaintenanceActive); }
        let Some(i) = self.selected else { return Err(Error::NoSelection); };
        let p = &self.products[i];
        if p.stock == 0 { return Err(Error::SoldOut { id: p.id.clone() }); }
        if self.credit < p.price { return Err(Error::Underpaid { price: p.price, credit: self.credit }); }
        let id = p.id.clone(); let price = p.price; let amount = self.credit-price;
        let change = self.change_coins(amount)?;
        let mut proposed = self.tubes;
        for &c in &change { proposed[idx(c).unwrap()] -= 1; }
        for &c in &self.escrow { proposed[idx(c).unwrap()] += 1; }

        self.tubes = proposed; self.products[i].stock -= 1; self.selected = None; self.credit = 0; self.escrow.clear();
        self.audit.sales = self.audit.sales.saturating_add(u64::from(price));
        self.audit.change = self.audit.change.saturating_add(u64::from(amount));
        self.log(EventKind::Vended { id: id.clone(), price, change: change.clone() });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance { return Err(Error::MaintenanceActive); }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();

        self.credit = 0; self.selected = None; self.escrow.clear();
        self.log(EventKind::Refunded { coins: coins.clone() }); Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> { let mut p=self.products.clone(); p.sort_by(|a,b| a.id.cmp(&b.id)); p }
    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance { return Err(Error::MaintenanceRequired); }
        let Some(i)=self.products.iter().position(|p|p.id==id) else { return Err(Error::UnknownProduct{id:id.to_owned()}); };
        if self.credit != 0 { return Err(Error::Busy); }

        self.log(EventKind::Reconciled{before,after:actual});Ok(())
    }
}

        if products.is_empty() {
            return Err(Error::EmptyCatalog);

            index,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),

            audit_change: 0,

            tubes: self.tubes,

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance { return Err(Error::MaintenanceActive); }
        let Some(id) = self.selected.clone() else { return Err(Error::NoSelection); };
        let i = self.index[&id];
        let p = &self.products[i];
        if p.stock == 0 { return Err(Error::SoldOut { id }); }
        let price = p.price;
        if self.credit < price {
            return Err(Error::Underpaid { price, credit: self.credit });

        let mut next_tubes = self.tubes;
        let mut change = Vec::new();
        for i in DESC {
            next_tubes[i] -= counts[i];

            let i = DENOMS.iter().position(|&d| d == coin).unwrap();
            next_tubes[i] += 1;
        }
        for i in 0..4 {
            if next_tubes[i] > 100 { return Err(Error::TubeLimit { coin: DENOMS[i] }); }
        }
        self.tubes = next_tubes;
        self.products[i].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.audit_sales = self.audit_sales.saturating_add(price as u64);
        self.audit_change = self.audit_change.saturating_add(amount as u64);
        self.push(EventKind::Vended { id: id.clone(), price, change: change.clone() });

        if self.credit != 0 { return Err(Error::Busy); }

        if new_stock > 100 { return Err(Error::StockLimit { id: id.to_owned() }); }

        Audit {
            inserted: self.audit_inserted,
            refunded: self.audit_refunded,
            sales: self.audit_sales,
            change: self.audit_change,
            adjustment: self.audit_adjustment,
            cash,
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance { return Err(Error::MaintenanceRequired); }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch { coin: DENOMS[i], expected: expected[i], found: self.tubes[i] });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 { return Err(Error::TubeLimit { coin: DENOMS[i] }); }
        }
        let before = value(self.tubes);
        let after = value(actual);
        if before.abs_diff(after) > 20 { return Err(Error::ValueDrift { before, after }); }
        self.tubes = actual;
        let delta = after as i128 - before as i128;
        self.audit_adjustment = (self.audit_adjustment as i128 + delta)
            .clamp(i64::MIN as i128, i64::MAX as i128) as i64;
        self.push(EventKind::Reconciled { before: self.tubes_before(expected), after: actual });
        Ok(())
    }

    fn tubes_before(&self, expected: [u16; 4]) -> [u16; 4] { expected }

        self.events.push(Event { seq: self.events.len(), kind });
    }

fn value(tubes: [u16; 4]) -> u64 {
    tubes[0] as u64 + tubes[1] as u64 * 3 + tubes[2] as u64 * 4 + tubes[3] as u64 * 10
}

fn exact_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut best: Option<([u16; 4], u32)> = None;
    enumerate(0, amount as i32, tubes, [0; 4], 0, &mut best);
    best.map(|x| x.0)
}

fn enumerate(pos: usize, rem: i32, tubes: [u16; 4], cur: [u16; 4], count: u32, best: &mut Option<([u16; 4], u32)>) {
    if rem == 0 {
        let better = match best {
            None => true,
            Some((bc, bn)) => count < *bn || (count == *bn && lex(cur, *bc)),
        };
        if better { *best = Some((cur, count)); }
        return;
    }
    if pos >= 4 || rem < 0 { return; }
    let i = DESC[pos];
    let max = tubes[i].min((rem as u16) / DENOMS[i]);
    for n in 0..=max {
        if let Some((_, bn)) = best {
            if count + n as u32 > *bn { continue; }
        }
        let mut next = cur;
        next[i] = n;
        enumerate(pos + 1, rem - n as i32 * DENOMS[i] as i32, tubes, next, count + n as u32, best);
    }
}

fn lex(a: [u16; 4], b: [u16; 4]) -> bool {
    for i in DESC {
        if a[i] != b[i] { return a[i] > b[i]; }
    }
    false
}

            refunded: self.audit_refunded,

        }
    }

        actual: [u16; 4],
    ) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let before = value(self.tubes);
        let after = value(actual);
        if before.abs_diff(after) > 20 {
            return Err(Error::ValueDrift { before, after });
        }

        self.tubes = actual;
        let adjustment = i128::from(self.audit_adjustment) + i128::from(after) - i128::from(before);
        self.audit_adjustment = adjustment.clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        self.add_event(EventKind::Reconciled {
            before: self.tubes_before_reconcile(expected),
            after: actual,
        });
        Ok(())
    }

    fn tubes_before_reconcile(&self, previous: [u16; 4]) -> [u16; 4] {
        previous
    }

    fn add_event(&mut self, kind: EventKind) {
        self.events.push(Event {
            seq: self.events.len(),
            kind,
        });
    }
}

}

fn find_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut remaining = amount;
    let mut greedy = [0u16; 4];
    for i in (0..4).rev() {
        greedy[i] = tubes[i].min(remaining / DENOMS[i]);
        remaining -= greedy[i] * DENOMS[i];
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<([u16; 4], u16)> = None;
    for ten in 0..=tubes[3].min(amount / 10) {
        for four in 0..=tubes[2].min(amount / 4) {
            let used_a = ten * 10 + four * 4;
            if used_a > amount {
                continue;
            }
            for three in 0..=tubes[1].min((amount - used_a) / 3) {
                let used_b = used_a + three * 3;
                let one = amount - used_b;
                if one > tubes[0] {
                    continue;
                }
                let candidate = [one, three, four, ten];
                let count = one + three + four + ten;
                if let Some((old, old_count)) = best {
                    if count > old_count
                        || (count == old_count
                            && (ten, four, three, one)
                                <= (old[3], old[2], old[1], old[0]))
                    {
                        continue;
                    }
                }
                best = Some((candidate, count));
            }
        }
    }
    best.map(|(counts, _)| counts)
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMS.iter().position(|&d| d == coin)
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn tube_value(tubes: &[u16; 4]) -> u64 {
    tubes
        .iter()
        .zip(DENOMS)
        .map(|(&count, coin)| u64::from(count) * u64::from(coin))
        .sum()
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut index = HashMap::new();
        for (i, product) in products.iter().enumerate() {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if index.contains_key(&product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
            index.insert(product.id.clone(), i);
        }

        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            products,
            index,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }

            return Err(Error::MaintenanceActive);
        }
        let Some(_) = denomination_index(coin) else {
            return Err(Error::BadCoin { coin });
        };
        let new_credit = u32::from(self.credit) + u32::from(coin);
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit = new_credit as u16;
        self.escrow.push(coin);

            coin,
            credit: self.credit,
        });
        Ok(())

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(&idx) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        let selected_id = self.products[idx].id.clone();
        self.selected = Some(selected_id.clone());
        self.log(EventKind::Selected { id: selected_id });
        Ok(())
    }

    fn choose_change(&self, amount: u16) -> Result<Vec<u16>, Error> {
        if amount == 0 {
            return Ok(Vec::new());
        }

        let mut greedy_counts = [0u16; 4];
        let mut remaining = amount;
        for i in (0..4).rev() {
            let take = self.tubes[i].min(remaining / DENOMS[i]);
            greedy_counts[i] = take;
            remaining -= take * DENOMS[i];
        }

        let counts = if remaining == 0 {
            greedy_counts
        } else {
            let mut best: Option<[u16; 4]> = None;
            let mut best_total = u32::MAX;
            for c10 in 0..=self.tubes[3].min(amount / 10) {
                let v10 = c10 * 10;
                for c4 in 0..=self.tubes[2].min((amount - v10) / 4) {
                    let v4 = v10 + c4 * 4;
                    for c3 in 0..=self.tubes[1].min((amount - v4) / 3) {
                        let v3 = v4 + c3 * 3;
                        if v3 > amount {
                            continue;
                        }
                        let rem = amount - v3;
                        if rem > self.tubes[0] {
                            continue;
                        }
                        let candidate = [rem, c3, c4, c10];
                        let total = u32::from(rem) + u32::from(c3) + u32::from(c4) + u32::from(c10);
                        let better = match best {
                            None => true,
                            Some(current) => {
                                total < best_total
                                    || (total == best_total
                                        && (candidate[3], candidate[2], candidate[1], candidate[0])
                                            > (current[3], current[2], current[1], current[0]))
                            }
                        };
                        if better {
                            best = Some(candidate);
                            best_total = total;
                        }
                    }
                }
            }
            best.ok_or(Error::NoChange { amount })?
        };

        let mut result = Vec::new();
        for i in (0..4).rev() {
            result.extend(std::iter::repeat_n(DENOMS[i], usize::from(counts[i])));
        }
        Ok(result)
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let idx = self.index[&id];

        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }

        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        let amount = self.credit - price;
        let change = self.choose_change(amount)?;

        let mut proposed = self.tubes;
        for &coin in &change {
            let i = denomination_index(coin).expect("change uses supported denominations");
            proposed[i] -= 1;
        }
        for &coin in &self.escrow {
            let i = denomination_index(coin).expect("escrow uses supported denominations");
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        self.tubes = proposed;
        self.products[idx].stock -= 1;
        self.selected = None;
        self.credit = 0;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        let change_value: u64 = change.iter().map(|&c| u64::from(c)).sum();
        self.change = self.change.saturating_add(change_value);
        self.log(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let old_credit = self.credit;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.refunded = self.refunded.saturating_add(u64::from(old_credit));
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&idx) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[idx].price;
        self.products[idx].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&idx) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let stock = u32::from(self.products[idx].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[idx].stock = stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value(&self.tubes) + u64::from(self.credit),
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        let before_value = tube_value(&self.tubes);
        let after_value = tube_value(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift {
                before: before_value,
                after: after_value,
            });
        }

        let before = self.tubes;
        self.tubes = actual;
        if after_value >= before_value {
            self.adjustment = self
                .adjustment
                .saturating_add((after_value - before_value) as i64);
        } else {
            self.adjustment = self
                .adjustment
                .saturating_sub((before_value - after_value) as i64);
        }
        self.log(EventKind::Reconciled {
            before,
            after: actual,
        });
        Ok(())
    }
}

pub struct Machine {
    products: Vec<Product>,
    indices: HashMap<String, usize>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    inserted: u64,
    refunded: u64,
    sales: u64,
    change: u64,
    adjustment: i64,
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut indices = HashMap::new();
        for (i, product) in products.iter().enumerate() {
            if !valid_id(&product.id) {
                return Err(Error::BadId { id: product.id.clone() });
            }
            if indices.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id.clone() });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id.clone() });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit { id: product.id.clone() });
            }
            indices.insert(product.id.clone(), i);
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            products,
            indices,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !DENOMS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let new_credit = self.credit as u32 + coin as u32;
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit = new_credit as u16;
        self.escrow.push(coin);
        self.inserted = self.inserted.saturating_add(coin as u64);
        self.log(EventKind::Inserted { coin, credit: self.credit });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(&index) = self.indices.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        let selected = self.products[index].id.clone();
        self.selected = Some(selected.clone());
        self.log(EventKind::Selected { id: selected });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let index = self.indices[&id];
        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid { price, credit: self.credit });
        }
        let amount = self.credit - price;
        let chosen = choose_change(amount, self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        let mut change_coins = Vec::new();
        for i in 0..4 {
            proposed[i] -= chosen[i];
            for _ in 0..chosen[i] {
                change_coins.push(DENOMS[i]);
            }
        }
        change_coins.reverse();
        for &coin in &self.escrow {
            let i = denom_index(coin).expect("escrow only contains valid denominations");
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        self.tubes = proposed;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(price as u64);
        let change_value: u64 = change_coins.iter().map(|&c| c as u64).sum();
        self.change = self.change.saturating_add(change_value);
        self.log(EventKind::Vended {
            id: id.clone(),
            price,
            change: change_coins.clone(),
        });
        Ok(Receipt { id, price, change: change_coins })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.refunded = self.refunded.saturating_add(self.credit as u64);
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut result = self.products.clone();
        result.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        result
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&index) = self.indices.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged { id: id.to_owned(), old, new: price });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&index) = self.indices.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let sum = self.products[index].stock as u32 + count as u32;
        if sum > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[index].stock = sum as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: sum as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        let tube_value: u64 = (0..4).map(|i| self.tubes[i] as u64 * DENOMS[i] as u64).sum();
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value + self.credit as u64,
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let before = tube_value(self.tubes);
        let after = tube_value(actual);
        let difference = before.abs_diff(after);
        if difference > 20 {
            return Err(Error::ValueDrift { before, after });
        }
        self.tubes = actual;
        let delta = after as i128 - before as i128;
        self.adjustment = (self.adjustment as i128 + delta)
            .clamp(i64::MIN as i128, i64::MAX as i128) as i64;
        self.log(EventKind::Reconciled { before: expected, after: actual });
        Ok(())
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }
}

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn denom_index(coin: u16) -> Option<usize> {
    DENOMS.iter().position(|&d| d == coin)
}

fn tube_value(tubes: [u16; 4]) -> u64 {
    (0..4).map(|i| tubes[i] as u64 * DENOMS[i] as u64).sum()
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let descending = [3usize, 2, 1, 0];
    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for i in descending {
        let take = tubes[i].min(remaining / DENOMS[i]);
        greedy[i] = take;
        remaining -= take * DENOMS[i];
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<([u16; 4], u32)> = None;
    search_change(0, amount as i32, tubes, [0; 4], 0, &mut best);
    best.map(|(counts, _)| counts)
}

fn search_change(
    pos: usize,
    remaining: i32,
    tubes: [u16; 4],
    current: [u16; 4],
    used: u32,
    best: &mut Option<([u16; 4], u32)>,
) {
    if remaining == 0 {
        let replace = match best {
            None => true,
            Some((best_counts, best_used)) => {
                used < *best_used
                    || (used == *best_used
                        && descending_lex_greater(current, *best_counts))
            }
        };
        if replace {
            *best = Some((current, used));
        }
        return;
    }
    if pos == 4 || remaining < 0 {
        return;
    }
    if let Some((_, best_used)) = best {
        if used >= *best_used {
            return;
        }
    }

    let i = [3usize, 2, 1, 0][pos];
    let max_take = tubes[i].min((remaining as u16) / DENOMS[i]);
    for take in (0..=max_take).rev() {
        let mut next = current;
        next[i] = take;
        search_change(
            pos + 1,
            remaining - take as i32 * DENOMS[i] as i32,
            tubes,
            next,
            used + take as u32,
            best,
        );
    }
}

fn descending_lex_greater(a: [u16; 4], b: [u16; 4]) -> bool {
    for i in [3usize, 2, 1, 0] {
        if a[i] != b[i] {
            return a[i] > b[i];
        }
    }
    false
}
