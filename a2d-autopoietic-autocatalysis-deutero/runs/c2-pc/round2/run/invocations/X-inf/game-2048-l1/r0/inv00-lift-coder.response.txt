#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && !(2..=2_147_483_648).contains(&value) || (value != 0 && !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Box<dyn Iterator<Item = usize>> = match side {
        Side::Left => Box::new(0..row.len()),
        Side::Right => Box::new((0..row.len()).rev()),
    };

    let mut tiles = Vec::new();
    for index in indices {
        if row[index] != 0 {
            tiles.push((index, row[index]));
        }
    }

    let mut outputs = Vec::new();
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut i = 0;

    while i < tiles.len() {
        if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
            let first = tiles[i];
            let second = tiles[i + 1];
            let value = first.1.checked_add(second.1).ok_or(Error::MergeOverflow {
                first: first.0,
                second: second.0,
            })?;
            let to = match side {
                Side::Left => outputs.len(),
                Side::Right => row.len() - 1 - outputs.len(),
            };
            merges.push(Merge {
                from: [first.0, second.0],
                to,
                value,
            });
            score += u64::from(value);
            outputs.push(value);
            i += 2;
        } else {
            outputs.push(tiles[i].1);
            i += 1;
        }
    }

    let mut cells = vec![0; row.len()];
    for (offset, value) in outputs.into_iter().enumerate() {
        let index = match side {
            Side::Left => offset,
            Side::Right => row.len() - 1 - offset,
        };
        cells[index] = value;
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        let valid = value == 0
            || (value >= 2 && value <= 2_147_483_648 && (value & (value - 1)) == 0);
        if !valid {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let mut cells = vec![0u32; row.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut scan = Vec::new();

    match side {
        Side::Left => {
            for index in 0..row.len() {
                if row[index] != 0 {
                    scan.push((index, row[index]));
                }
            }
        }
        Side::Right => {
            for index in (0..row.len()).rev() {
                if row[index] != 0 {
                    scan.push((index, row[index]));
                }
            }
        }
    }

    let mut input = 0usize;
    let mut output = 0usize;
    while input < scan.len() {
        let (index, value) = scan[input];
        if input + 1 < scan.len() && scan[input + 1].1 == value {
            let other_index = scan[input + 1].0;
            let doubled = value.checked_mul(2).ok_or(Error::MergeOverflow {
                first: index,
                second: other_index,
            })?;
            let destination = match side {
                Side::Left => output,
                Side::Right => row.len() - 1 - output,
            };
            cells[destination] = doubled;
            merges.push(Merge {
                from: [index, other_index],
                to: destination,
                value: doubled,
            });
            score += u64::from(doubled);
            input += 2;
        } else {
            let destination = match side {
                Side::Left => output,
                Side::Right => row.len() - 1 - output,
            };
            cells[destination] = value;
            input += 1;
        }
        output += 1;
    }

    Ok(RowResult {
        changed: cells.as_slice() != row,
        cells,
        score,
        merges,
    })
}
