#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Vec<usize> = match side {
        Side::Left => (0..row.len()).collect(),
        Side::Right => (0..row.len()).rev().collect(),
    };
    let tiles: Vec<(usize, u32)> = indices
        .into_iter()
        .filter_map(|index| (row[index] != 0).then_some((index, row[index])))
        .collect();

    let mut outputs = Vec::with_capacity(tiles.len());
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut i = 0;

    while i < tiles.len() {
        let (index, value) = tiles[i];
        if i + 1 < tiles.len() && tiles[i + 1].1 == value {
            let second_index = tiles[i + 1].0;
            let Some(merged_value) = value.checked_mul(2) else {
                return Err(Error::MergeOverflow {
                    first: index,
                    second: second_index,
                });
            };
            let to = match side {
                Side::Left => outputs.len(),
                Side::Right => row.len() - 1 - outputs.len(),
            };
            outputs.push(merged_value);
            merges.push(Merge {
                from: [index, second_index],
                to,
                value: merged_value,
            });
            score += u64::from(merged_value);
            i += 2;
        } else {
            outputs.push(value);
            i += 1;
        }
    }

    let mut cells = vec![0; row.len()];
    for (position, value) in outputs.into_iter().enumerate() {
        let destination = match side {
            Side::Left => position,
            Side::Right => row.len() - 1 - position,
        };
        cells[destination] = value;
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    let len = row.len();
    if len > 32 {
        return Err(Error::RowTooLong { len });
    }

    for index in 0..len {
        let value = row[index];
        if value != 0 && (value < 2 || value > 2_147_483_648 || value & (value - 1) != 0) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let mut cells = vec![0; len];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut read = 0usize;
    let mut write = 0usize;

    while read < len {
        let index = match side {
            Side::Left => read,
            Side::Right => len - 1 - read,
        };
        let value = row[index];
        read += 1;
        if value == 0 {
            continue;
        }

        let mut next_nonzero = None;
        let mut probe = read;
        while probe < len {
            let probe_index = match side {
                Side::Left => probe,
                Side::Right => len - 1 - probe,
            };
            if row[probe_index] != 0 {
                next_nonzero = Some((probe, probe_index, row[probe_index]));
                break;
            }
            probe += 1;
        }

        let destination = match side {
            Side::Left => write,
            Side::Right => len - 1 - write,
        };

        if let Some((next_read, next_index, next_value)) = next_nonzero {
            if next_value == value {
                let Some(merged) = value.checked_mul(2) else {
                    return Err(Error::MergeOverflow {
                        first: index,
                        second: next_index,
                    });
                };
                cells[destination] = merged;
                merges.push(Merge {
                    from: [index, next_index],
                    to: destination,
                    value: merged,
                });
                score += u64::from(merged);
                read = next_read + 1;
                write += 1;
                continue;
            }
        }

        cells[destination] = value;
        write += 1;
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
