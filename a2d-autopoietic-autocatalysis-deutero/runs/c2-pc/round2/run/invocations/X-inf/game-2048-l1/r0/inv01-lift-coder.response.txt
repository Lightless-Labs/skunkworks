#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Box<dyn Iterator<Item = usize>> = match side {
        Side::Left => Box::new(0..row.len()),
        Side::Right => Box::new((0..row.len()).rev()),
    };
    let tiles: Vec<(usize, u32)> = indices
        .filter_map(|i| (row[i] != 0).then_some((i, row[i])))
        .collect();

    let mut outputs = Vec::with_capacity(row.len());
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut i = 0;

    while i < tiles.len() {
        let (first_index, value) = tiles[i];
        if i + 1 < tiles.len() && tiles[i + 1].1 == value {
            let second_index = tiles[i + 1].0;
            let Some(merged_value) = value.checked_mul(2) else {
                return Err(Error::MergeOverflow {
                    first: first_index,
                    second: second_index,
                });
            };
            let to = match side {
                Side::Left => outputs.len(),
                Side::Right => row.len() - 1 - outputs.len(),
            };
            outputs.push(merged_value);
            merges.push(Merge {
                from: [first_index, second_index],
                to,
                value: merged_value,
            });
            score += u64::from(merged_value);
            i += 2;
        } else {
            outputs.push(value);
            i += 1;
        }
    }

    let mut cells = vec![0; row.len()];
    for (offset, value) in outputs.into_iter().enumerate() {
        let index = match side {
            Side::Left => offset,
            Side::Right => row.len() - 1 - offset,
        };
        cells[index] = value;
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    let len = row.len();
    if len > 32 {
        return Err(Error::RowTooLong { len });
    }

    for index in 0..len {
        let value = row[index];
        if value != 0 && !(2..=2_147_483_648).contains(&value) || value != 0 && !value.is_power_of_two() {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let mut packed = Vec::with_capacity(len);
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut cursor: isize = match side {
        Side::Left => 0,
        Side::Right => len as isize - 1,
    };
    let step: isize = match side {
        Side::Left => 1,
        Side::Right => -1,
    };

    while cursor >= 0 && cursor < len as isize {
        let first_index = cursor as usize;
        let first_value = row[first_index];
        cursor += step;
        if first_value == 0 {
            continue;
        }

        let mut next_nonempty = None;
        let mut probe = cursor;
        while probe >= 0 && probe < len as isize {
            let idx = probe as usize;
            if row[idx] != 0 {
                next_nonempty = Some((idx, row[idx]));
                break;
            }
            probe += step;
        }

        if let Some((second_index, second_value)) = next_nonempty {
            if first_value == second_value {
                let Some(value) = first_value.checked_add(second_value) else {
                    return Err(Error::MergeOverflow {
                        first: first_index,
                        second: second_index,
                    });
                };
                let to = match side {
                    Side::Left => packed.len(),
                    Side::Right => len - 1 - packed.len(),
                };
                packed.push(value);
                merges.push(Merge {
                    from: [first_index, second_index],
                    to,
                    value,
                });
                score += u64::from(value);
                cursor = second_index as isize + step;
                continue;
            }
        }
        packed.push(first_value);
    }

    let mut cells = vec![0; len];
    for (offset, value) in packed.into_iter().enumerate() {
        let index = match side {
            Side::Left => offset,
            Side::Right => len - 1 - offset,
        };
        cells[index] = value;
    }

    let changed = cells.as_slice() != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
