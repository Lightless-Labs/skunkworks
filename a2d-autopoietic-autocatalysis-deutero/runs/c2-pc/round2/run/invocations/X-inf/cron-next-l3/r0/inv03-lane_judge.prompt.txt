        let n = parse_number(field, bytes, offset)?;
        let (lo, hi, _) = bounds(field);
        if n < lo as u32 || n > hi as u32 {
            return Err(err(field, offset, 3));
        }
        return Ok(n);
    }
    name_value(field, bytes)
        .map(|x| x as u32)
        .ok_or_else(|| err(field, offset, 0))
}

fn parse_field(field: Field, text: &str) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max, width) = bounds(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let mut start = 0usize;
    loop {
        let end = bytes[start..].iter().position(|&b| b == b',')
            .map(|p| start + p)
            .unwrap_or(bytes.len());
        let item = &bytes[start..end];
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: start });
        }

        let slash_positions: Vec<usize> = item.iter().enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();
        if slash_positions.len() > 1 {
            return Err(err(field, start, 0));
        }
        let (base, step_bytes, step_offset) = if let Some(&p) = slash_positions.first() {
            (&item[..p], Some(&item[p + 1..]), start + p + 1)
        } else {
            (item, None, 0)
        };
        if base.is_empty() || step_bytes == Some(&[][..]) {
            return Err(err(field, start, 0));
        }

        let hyphens: Vec<usize> = base.iter().enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();
        if hyphens.len() > 1 {
            return Err(err(field, start, 0));
        }

        let star = base == b"*";
        let (first, second, range) = if star {
            (None, None, false)
        } else if let Some(&p) = hyphens.first() {
            if p == 0 || p + 1 == base.len()
                || base[..p].contains(&b'*') || base[p + 1..].contains(&b'*')
            {
                return Err(err(field, start, 0));
            }
            (Some((0, &base[..p])), Some((p + 1, &base[p + 1..])), true)
        } else {
            if base.contains(&b'*') {
                return Err(err(field, start, 0));
            }
            (Some((0, base)), None, false)
        };

        let mut step = 1u32;
        if let Some(sb) = step_bytes {
            if !sb.iter().all(u8::is_ascii_digit) {
                return Err(err(field, start, 0));
            }
            let raw = parse_number(field, sb, step_offset)?;
            if raw == 0 || raw > width as u32 {
                return Err(err(field, step_offset, 4));
            }
            step = raw;
        }

        let (from, to) = if star {
            (min as u32, max as u32)
        } else {
            let (fo, fb) = first.unwrap();
            let a = parse_endpoint(field, fb, start + fo)?;
            if range {
                let (eo, eb) = second.unwrap();
                let b = parse_endpoint(field, eb, start + eo)?;
                if a > b {
                    return Err(err(field, start + eo, 5));
                }
                (a, b)
            } else if step_bytes.is_some() {
                (a, max as u32)
            } else {
                (a, a)
            }
        };

        let mut value = from;
        while value <= to {
            selected[(value as u8 - min) as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }

    let values = selected.iter().enumerate()
        .filter_map(|(i, &yes)| if yes { Some(min + i as u8) } else { None })
        .collect();
    Ok((values, text == "*"))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }
    let mut allowed: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut stars = [false; 5];
    for i in 0..5 {
        let (values, star) = parse_field(FIELDS[i], pieces[i])?;
        allowed[i] = values;
        stars[i] = star;
    }
    Ok(Cron {
        allowed,
        day_of_month_star: stars[2],
        day_of_week_star: stars[4],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        }].clone()
    }

    fn has(&self, index: usize, value: u8) -> bool {
        self.allowed[index].binary_search(&value).is_ok()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.has(0, at.minute) || !self.has(1, at.hour) || !self.has(3, at.month) {
            return false;
        }
        let dom = self.has(2, at.day);
        let dow = self.has(4, weekday(at.year, at.month, at.day));
        match (self.day_of_month_star, self.day_of_week_star) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == month_length(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut y = after.year;
        let mut m = after.month;
        let mut d = after.day;
        let mut h = after.hour;
        let mut min = after.minute;
        loop {
            if min == 59 {
                min = 0;
                if h == 23 {
                    h = 0;
                    d += 1;
                    if d > month_length(y, m) {
                        d = 1;
                        m += 1;
                        if m > 12 {
                            m = 1;
                            y += 1;
                            if y > 2399 {
                                return Ok(None);
                            }
                        }
                    }
                } else {
                    h += 1;
                }
            } else {
                min += 1;
            }
            let candidate = CivilDateTime { year: y, month: m, day: d, hour: h, minute: min };
            if self.matches_valid(candidate) {
                return Ok(Some(candidate));
            }
            if y == 2399 && m == 12 && d == 31 && h == 23 && min == 59 {
                return Ok(None);
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn names(field: Field, s: &str) -> Option<u32> {
    let table: &[(&str, u32)] = match field {
        Field::Month => &[
            ("JAN", 1), ("FEB", 2), ("MAR", 3), ("APR", 4), ("MAY", 5), ("JUN", 6),
            ("JUL", 7), ("AUG", 8), ("SEP", 9), ("OCT", 10), ("NOV", 11), ("DEC", 12),

            ("SUN", 0), ("MON", 1), ("TUE", 2), ("WED", 3), ("THU", 4), ("FRI", 5),
            ("SAT", 6),
        ],

        .map(|(_, value)| *value)

fn syntax_error(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(field: Field, s: &str, offset: usize, step: bool) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax_error(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n = 0u32;
    for &b in bytes {
        n = n
            .checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if step {
        let (lo, hi) = bounds(field);
        let width = hi - lo + 1;
        if n == 0 || n > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (lo, hi) = bounds(field);
        if n < lo || n > hi {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(n)
}

fn parse_base(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    if s.as_bytes().iter().all(u8::is_ascii_digit) && !s.is_empty() {
        parse_number(field, s, offset, false)
    } else if let Some(n) = names(field, s) {
        Ok(n)
    } else {
        Err(syntax_error(field, offset))
    }
}

fn parse_field(field: Field, text: &str, set: &mut [bool]) -> Result<(), ParseError> {
    for (i, &b) in text.as_bytes().iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = bounds(field);
    let mut start = 0usize;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field, offset: start });
        }
        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(syntax_error(field, start));
        }
        let (base, step_text) = if let Some(pos) = item.find('/') {
            (&item[..pos], Some((&item[pos + 1..], start + pos + 1)))
        } else {
            (item, None)
        };
        if base.is_empty() || step_text.is_some_and(|(s, _)| s.is_empty()) {
            return Err(syntax_error(field, start));
        }

        let (lo, hi) = if base == "*" {
            (min, max)
        } else {
            let hyphens = base.bytes().filter(|&b| b == b'-').count();
            if hyphens > 1 || base.starts_with('-') || base.ends_with('-') {
                return Err(syntax_error(field, start));
            }
            if let Some(pos) = base.find('-') {
                let a = &base[..pos];
                let b = &base[pos + 1..];
                if a == "*" || b == "*" {
                    return Err(syntax_error(field, start));
                }
                let a_num = parse_base(field, a, start)?;
                let b_num = parse_base(field, b, start + pos + 1)?;
                if a_num > b_num {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: start + pos + 1,
                    });
                }
                (a_num, b_num)
            } else {
                let n = parse_base(field, base, start)?;
                (n, if step_text.is_some() { max } else { n })
            }
        };

        let step = if let Some((s, off)) = step_text {
            Some(parse_number(field, s, off, true)?)
        } else {
            None
        };
        let stride = step.unwrap_or(1);
        let mut n = lo;
        while n <= hi {
            set[(n - min) as usize] = true;
            match n.checked_add(stride) {
                Some(next) => n = next,
                None => break,
            }
        }
        start += item.len() + 1;
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (lo, hi) = bounds(FIELDS[i]);
        vec![false; (hi - lo + 1) as usize]
    });
    for i in 0..5 {
        parse_field(FIELDS[i], parts[i], &mut sets[i])?;
    }
    Ok(Cron {
        sets,
        unrestricted_days: [parts[2] == "*", parts[4] == "*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (lo, _) = bounds(field);
        self.sets[i]
            .iter()
            .enumerate()
            .filter_map(|(j, &yes)| yes.then_some((lo as usize + j) as u8))
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut cur = after;
        loop {
            if cur.year == 2399 && cur.month == 12 && cur.day == 31 && cur.hour == 23 && cur.minute == 59 {
                return Ok(None);
            }
            cur = increment_minute(cur);
            if self.matches_valid(cur) {
                return Ok(Some(cur));
            }
        }
    }

    fn has(&self, field: Field, value: u8) -> bool {
        let i = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (lo, _) = bounds(field);
        self.sets[i][(value as u32 - lo) as usize]
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.has(Field::Minute, at.minute)
            || !self.has(Field::Hour, at.hour)
            || !self.has(Field::Month, at.month)
        {
            return false;
        }
        let dom = self.has(Field::DayOfMonth, at.day);
        let dow = self.has(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if at.day == month_length(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {

    }

    for y in 2000..year {

    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((6 + days) % 7) as u8
}

fn increment_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
    } else {
        at.minute = 0;
        if at.hour < 23 {
            at.hour += 1;
        } else {
            at.hour = 0;
            if at.day < month_length(at.year, at.month) {
                at.day += 1;
            } else {
                at.day = 1;
                if at.month < 12 {
                    at.month += 1;
                } else {
                    at.month = 1;
                    at.year += 1;
                }
            }
        }
    }
    at
}

fn is_lexical_byte(b: u8) -> bool {
    b.is_ascii_digit() || b.is_ascii_alphabetic() || matches!(b, b'*' | b'-' | b'/' | b',')
}

fn name_value(field: Field, s: &str) -> Option<u8> {
    let upper = s.to_ascii_uppercase();
    match field {
        Field::Month => [
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN",
            "JUL", "AUG", "SEP", "OCT", "NOV", "DEC",
        ]
        .iter()
        .position(|&name| name == upper)
        .map(|i| i as u8 + 1),
        Field::DayOfWeek => [
            "SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT",
        ]
        .iter()
        .position(|&name| name == upper)
        .map(|i| i as u8),
        _ => None,
    }
}

fn parse_atom(field: Field, text: &str, offset: usize) -> Result<u8, ParseError> {
    if text.bytes().all(|b| b.is_ascii_digit()) {
        if text.len() > 1 && text.as_bytes()[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }
        let mut value: u32 = 0;
        for b in text.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        let spec = SPECS.iter().find(|s| s.field == field).unwrap();
        if value < spec.min as u32 || value > spec.max as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
        Ok(value as u8)
    } else {
        name_value(field, text).ok_or(ParseError::InvalidSyntax { field, offset })
    }
}

            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let spec = SPECS.iter().find(|s| s.field == field).unwrap();
    let width = (spec.max - spec.min + 1) as u32;
    if value == 0 || value > width {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(value)
}

fn parse_field(text: &str, spec: Spec) -> Result<Vec<u8>, ParseError> {
    for (i, &b) in text.as_bytes().iter().enumerate() {
        if !is_lexical_byte(b) {
            return Err(ParseError::InvalidCharacter { field: spec.field, offset: i });
        }
    }

    let mut selected = vec![false; (spec.max - spec.min + 1) as usize];
    let mut item_start = 0usize;

    for item in text.split(',') {
        let item_offset = item_start;
        item_start += item.len() + 1;
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field: spec.field, offset: item_offset });
        }

        let slash_count = item.bytes().filter(|&b| b == b'/').count();
        if slash_count > 1 {
            return Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset });
        }
        let (base, step_text) = if let Some(pos) = item.find('/') {
            (&item[..pos], Some((&item[pos + 1..], item_offset + pos + 1)))
        } else {
            (item, None)
        };
        if base.is_empty() {
            return Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset });
        }

        let is_star = base == "*";
        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
        if hyphen_count > 1 || (is_star && hyphen_count != 0) {
            return Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset });
        }

        let (start, end, end_offset) = if is_star {
            (spec.min, spec.max, item_offset)
        } else if let Some(pos) = base.find('-') {
            let left = &base[..pos];
            let right = &base[pos + 1..];
            if left.is_empty() || right.is_empty() || left == "*" || right == "*" {
                return Err(ParseError::InvalidSyntax { field: spec.field, offset: item_offset });
            }
            let a = parse_atom(spec.field, left, item_offset)?;
            let b = parse_atom(spec.field, right, item_offset + pos + 1)?;
            (a, b, item_offset + pos + 1)
        } else {
            let a = parse_atom(spec.field, base, item_offset)?;
            let full_end = if step_text.is_some() { spec.max } else { a };
            (a, full_end, item_offset)
        };

        if let Some((step_str, step_offset)) = step_text {
            let step = parse_step(spec.field, step_str, step_offset)?;
            if start > end {
                // Numeric range ordering is checked before step validation.
                return Err(ParseError::ReversedRange { field: spec.field, offset: end_offset });
            }
            let mut value = start as u32;
            while value <= end as u32 {
                selected[value as usize - spec.min as usize] = true;
                value += step;
            }
        } else {
            if !is_star && hyphen_count == 1 && start > end {
                return Err(ParseError::ReversedRange { field: spec.field, offset: end_offset });
            }
            let mut value = start;
            while value <= end {
                selected[value as usize - spec.min as usize] = true;
                if value == u8::MAX {
                    break;
                }
                value += 1;
            }
        }
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(i, &yes)| yes.then_some(i as u8 + spec.min))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut parsed: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    for (i, spec) in SPECS.iter().enumerate() {
        parsed[i] = parse_field(pieces[i], *spec)?;
    }

    Ok(Cron {
        allowed: parsed,
        dom_star: pieces[2] == "*",
        dow_star: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.allowed[field_index(field)].clone()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if let Some(next) = increment_minute(current) {
                current = next;
            } else {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.allowed[0].contains(&at.minute)
            || !self.allowed[1].contains(&at.hour)
            || !self.allowed[3].contains(&at.month)
        {
            return false;
        }

        let dom = self.allowed[2].contains(&at.day);
        let dow = self.allowed[4].contains(&weekday(at.year, at.month, at.day));
        if self.dom_star && self.dow_star {
            true
        } else if self.dom_star {
            dow
        } else if self.dow_star {
            dom
        } else if at.day == month_length(at.year, at.month) {
            dom && dow
        } else {
            dom || dow
        }
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_length(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_length(year: u16, month: u8) -> u8 {
    match month {
        2 if is_leap(year) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += month_length(year, m) as u32;
    }
    days += day as u32 - 1;
    ((days + 6) % 7) as u8
}

fn increment_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        Some(CivilDateTime { minute: at.minute + 1, ..at })
    } else if at.hour < 23 {
        Some(CivilDateTime { minute: 0, hour: at.hour + 1, ..at })
    } else if at.day < month_length(at.year, at.month) {
        Some(CivilDateTime { minute: 0, hour: 0, day: at.day + 1, ..at })
    } else if at.month < 12 {
        Some(CivilDateTime { minute: 0, hour: 0, day: 1, month: at.month + 1, ..at })
    } else if at.year < 2399 {
        Some(CivilDateTime { minute: 0, hour: 0, day: 1, month: 1, year: at.year + 1 })
    } else {
        None
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    bits: [u64; 5],
    dom_star: bool,
    dow_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const DOMAINS: [(u8, u8); 5] = [(0, 59), (0, 23), (1, 31), (1, 12), (0, 6)];

fn field_idx(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_at(i: usize) -> Field {
    [Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek][i]
}

fn name(field: Field, text: &str) -> Option<u8> {
    let names: &[&str] = match field {
        Field::Month => &["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    names.iter().position(|s| s.eq_ignore_ascii_case(text)).map(|n| {
        if field == Field::Month { n as u8 + 1 } else { n as u8 }
    })
}

fn atom(field: Field, s: &str, offset: usize) -> Result<u8, ParseError> {
    if s.bytes().all(|b| b.is_ascii_digit()) {
        if s.len() > 1 && s.as_bytes()[0] == b'0' {
            return Err(ParseError::LeadingZero { field, offset });
        }
        let mut n = 0u32;
        for b in s.bytes() {
            n = n.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32))
                .ok_or(ParseError::NumberOverflow { field, offset })?;
        }
        let (lo, hi) = DOMAINS[field_idx(field)];
        if n < lo as u32 || n > hi as u32 {
            return Err(ParseError::OutOfRange { field, offset });
        }
        Ok(n as u8)
    } else {
        name(field, s).ok_or(ParseError::InvalidSyntax { field, offset })
    }
}

fn step(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    if s.is_empty() || !s.bytes().all(|b| b.is_ascii_digit()) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if s.len() > 1 && s.as_bytes()[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut n = 0u32;
    for b in s.bytes() {
        n = n.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    let (lo, hi) = DOMAINS[field_idx(field)];
    if n < 1 || n > (hi - lo + 1) as u32 {
        return Err(ParseError::BadStep { field, offset });
    }
    Ok(n)
}

fn lex(field: Field, s: &str) -> Result<(), ParseError> {
    for (offset, b) in s.bytes().enumerate() {
        if !(b.is_ascii_digit() || b.is_ascii_alphabetic() || matches!(b, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }
    Ok(())
}

fn parse_one(field: Field, item: &str, offset: usize, out: &mut u64) -> Result<(), ParseError> {
    let slash_count = item.bytes().filter(|&b| b == b'/').count();
    if slash_count > 1 {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    let (base, step_part) = if let Some(i) = item.find('/') {
        (&item[..i], Some((&item[i + 1..], offset + i + 1)))
    } else {
        (item, None)
    };
    if base.is_empty() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    let (lo, hi) = DOMAINS[field_idx(field)];
    let hyphens = base.bytes().filter(|&b| b == b'-').count();
    if hyphens > 1 {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let (start, end, end_offset, range) = if base == "*" {
        (lo, hi, offset, false)
    } else if let Some(i) = base.find('-') {
        let a = &base[..i];
        let b = &base[i + 1..];
        if a.is_empty() || b.is_empty() || a == "*" || b == "*" {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
        (atom(field, a, offset)?, atom(field, b, offset + i + 1)?, offset + i + 1, true)
    } else {
        let n = atom(field, base, offset)?;
        (n, if step_part.is_some() { hi } else { n }, offset, false)
    };

    if range && start > end {
        return Err(ParseError::ReversedRange { field, offset: end_offset });
    }
    let stride = match step_part {
        Some((s, o)) => step(field, s, o)?,
        None => 1,
    };
    let mut v = start as u32;
    while v <= end as u32 {
        *out |= 1u64 << v;
        v += stride;
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount { found: fields.len() });
    }
    let mut bits = [0u64; 5];
    for i in 0..5 {
        let field = field_at(i);
        lex(field, fields[i])?;
        let mut start = 0usize;
        for item in fields[i].split(',') {
            if item.is_empty() {
                return Err(ParseError::EmptyItem { field, offset: start });
            }
            parse_one(field, item, start, &mut bits[i])?;
            start += item.len() + 1;
        }
    }
    Ok(Cron {
        bits,
        dom_star: fields[2] == "*",
        dow_star: fields[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (lo, hi) = DOMAINS[field_idx(field)];
        (lo..=hi).filter(|&v| self.bits[field_idx(field)] & (1u64 << v) != 0).collect()
    }

    fn has(&self, field: Field, v: u8) -> bool {
        self.bits[field_idx(field)] & (1u64 << v) != 0
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut x = after;
        loop {
            match advance(x) {
                Some(n) => x = n,
                None => return Ok(None),
            }
            if self.matches_valid(x) {
                return Ok(Some(x));
            }
        }
    }

    fn matches_valid(&self, x: CivilDateTime) -> bool {
        if !self.has(Field::Minute, x.minute)
            || !self.has(Field::Hour, x.hour)
            || !self.has(Field::Month, x.month)
        {
            return false;
        }
        let dom = self.has(Field::DayOfMonth, x.day);
        let dow = self.has(Field::DayOfWeek, weekday(x.year, x.month, x.day));
        match (self.dom_star, self.dow_star) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) if x.day == month_len(x.year, x.month) => dom && dow,
            (false, false) => dom || dow,
        }
    }
}

fn leap(y: u16) -> bool {
    y % 4 == 0 && (y % 100 != 0 || y % 400 == 0)
}

fn month_len(y: u16, m: u8) -> u8 {
    match m {
        2 if leap(y) => 29,
        2 => 28,
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(x: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&x.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&x.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if x.day == 0 || x.day > month_len(x.year, x.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if x.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if x.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(y: u16, m: u8, d: u8) -> u8 {
    let mut total = 6u32;
    for year in 2000..y {
        total += if leap(year) { 366 } else { 365 };
    }
    for month in 1..m {
        total += month_len(y, month) as u32;
    }
    total += d as u32 - 1;
    (total % 7) as u8
}

fn advance(mut x: CivilDateTime) -> Option<CivilDateTime> {
    if x.minute < 59 {
        x.minute += 1;
    } else {
        x.minute = 0;
        if x.hour < 23 {
            x.hour += 1;
        } else {
            x.hour = 0;
            if x.day < month_len(x.year, x.month) {
                x.day += 1;
            } else {
                x.day = 1;
                if x.month < 12 {
                    x.month += 1;
                } else if x.year < 2399 {
                    x.month = 1;
                    x.year += 1;
                } else {
                    return None;
                }
            }
        }
    }
    Some(x)
}
