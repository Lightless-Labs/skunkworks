pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line in 0..line_count {
        let indices = line_indices(board, direction, line);
        let tiles: Vec<(usize, u32)> = indices
            .iter()
            .filter_map(|&index| (board.cells[index] != 0).then_some((index, board.cells[index])))
            .collect();

        let mut outputs: Vec<(u32, Option<[usize; 2]>)> = Vec::new();
        let mut i = 0;
        while i < tiles.len() {
            if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
                if tiles[i].1 == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first: tiles[i].0, second: tiles[i + 1].0 });
                }
                let value = tiles[i].1 * 2;
                let to = indices[outputs.len()];
                merges.push(Merge { from: [tiles[i].0, tiles[i + 1].0], to, value });
                score += value as u64;
                outputs.push((value, Some([tiles[i].0, tiles[i + 1].0])));
                i += 2;
            } else {
                outputs.push((tiles[i].1, None));
                i += 1;
            }
        }
        for (offset, (value, _)) in outputs.into_iter().enumerate() {
            cells[indices[offset]] = value;
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board { width: board.width, height: board.height, cells },
        score: if changed { score } else { 0 },
        changed,
        merges: if changed { merges } else { Vec::new() },
    })
}

    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line in 0..line_count {
        let indices = line_indices(board, direction, line);
        let tiles: Vec<(usize, u32)> = indices
            .iter()
            .filter_map(|&index| {
                let value = board.cells[index];
                (value != 0).then_some((index, value))
            })
            .collect();

        let mut output_pos = 0usize;
        let mut i = 0usize;

                    return Err(Error::MergeOverflow {
                        first: first_index,
                        second: tiles[i + 1].0,
                    });

                cells[to] = value;
                score += value as u64;

                    from: [first_index, tiles[i + 1].0],
                    to,
                    value,
                });

                i += 2;

                cells[indices[output_pos]] = first_value;
                output_pos += 1;
                i += 1;
            }
        }

            width: board.width,

                        first: tiles[i].0,
                        second: tiles[i + 1].0,
                    });
                }
                let value = tiles[i].1 * 2;
                outputs.push((value, Some([tiles[i].0, tiles[i + 1].0])));
                score += u64::from(value);
                i += 2;
            } else {
                outputs.push((tiles[i].1, None));
                i += 1;
            }
        }

        for (position, (value, provenance)) in outputs.into_iter().enumerate() {
            let to = indices[position];
            cells[to] = value;
            if let Some(from) = provenance {
                merges.push(Merge { from, to, value });
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut cells = vec![0u32; board.cells.len()];
    let mut score = 0u64;
    let mut merges = Vec::new();

    for line in 0..line_count(board, direction) {
        let mut inputs = Vec::new();
        for offset in 0..line_len(board, direction) {
            let index = flat_index(board, direction, line, offset);
            let value = board.cells[index];
            if value != 0 {
                inputs.push((index, value));
            }
        }

        let mut output_offset = 0;
        let mut input_offset = 0;
        while input_offset < inputs.len() {
            let (source, value) = inputs[input_offset];
            let destination = flat_index(board, direction, line, output_offset);
            if input_offset + 1 < inputs.len() && inputs[input_offset + 1].1 == value {
                let second = inputs[input_offset + 1].0;
                let Some(merged_value) = value.checked_mul(2).filter(|v| *v <= 2_147_483_648) else {
                    return Err(Error::MergeOverflow {
                        first: source,
                        second,
                    });
                };
                cells[destination] = merged_value;
                merges.push(Merge {
                    from: [source, second],
                    to: destination,
                    value: merged_value,
                });
                score += merged_value as u64;
                input_offset += 2;
            } else {
                cells[destination] = value;
                input_offset += 1;
            }
            output_offset += 1;
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

fn move_validated(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    let mut result = board.clone();
    result.cells.fill(0);
    let mut merges = Vec::new();
    let mut score = 0u64;

    let horizontal = matches!(direction, Direction::Left | Direction::Right);
    let line_count = if horizontal { board.height } else { board.width };
    let line_len = if horizontal { board.width } else { board.height };

    for line in 0..line_count {
        let mut packed = Vec::<(u32, Option<[usize; 2]>)>::new();
        let mut pending: Option<(usize, u32)> = None;

        for offset in 0..line_len {
            let position = match direction {
                Direction::Left => line * board.width + offset,
                Direction::Right => line * board.width + (board.width - 1 - offset),
                Direction::Up => offset * board.width + line,
                Direction::Down => (board.height - 1 - offset) * board.width + line,
            };
            let value = board.cells[position];
            if value == 0 {
                continue;
            }

            if let Some((first_index, first_value)) = pending.take() {
                if first_value == value {
                    if first_value == 2_147_483_648 {
                        return Err(Error::MergeOverflow {
                            first: first_index,
                            second: position,
                        });
                    }
                    let merged_value = first_value * 2;
                    packed.push((merged_value, Some([first_index, position])));
                    score += u64::from(merged_value);
                } else {
                    packed.push((first_value, None));
                    pending = Some((position, value));
                }
            } else {
                pending = Some((position, value));
            }
        }

        if let Some((_, value)) = pending {
            packed.push((value, None));
        }

        for (offset, (value, provenance)) in packed.into_iter().enumerate() {
            let destination = match direction {
                Direction::Left => line * board.width + offset,
                Direction::Right => line * board.width + (board.width - 1 - offset),
                Direction::Up => offset * board.width + line,
                Direction::Down => (board.height - 1 - offset) * board.width + line,

                    from,
                    to: destination,
                    value,
                });
            }
        }
    }

    let changed = result.cells != board.cells;
    Ok(MoveResult {
        board: result,
        score,
        changed,
        merges,
    })

    validate(board)?;
    let mut output = vec![0u32; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let line_count = if matches!(direction, Direction::Left | Direction::Right) {
        board.height
    } else {
        board.width
    };
    let line_len = if matches!(direction, Direction::Left | Direction::Right) {
        board.width
    } else {
        board.height
    };

    for line in 0..line_count {
        let mut positions = Vec::with_capacity(line_len);
        for offset in 0..line_len {
            let index = match direction {
                Direction::Left => line * board.width + offset,
                Direction::Right => line * board.width + (board.width - 1 - offset),
                Direction::Up => offset * board.width + line,
                Direction::Down => (board.height - 1 - offset) * board.width + line,
            };
            positions.push(index);
        }

        let mut read = 0;
        let mut write = 0;
        while read < positions.len() {
            let first = positions[read];
            let value = board.cells[first];
            if value == 0 {
                read += 1;
                continue;
            }
            let mut next = read + 1;
            while next < positions.len() && board.cells[positions[next]] == 0 {
                next += 1;
            }
            if next < positions.len() && board.cells[positions[next]] == value {
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first, second: positions[next] });

                merges.push(Merge {
                    from: [first, positions[next]],
                    to: positions[write],
                    value: merged_value,
                });
                score += merged_value as u64;
                write += 1;
                read = next + 1;
            } else {
                output[positions[write]] = value;
                write += 1;
                read += 1;
            }
        }
    }

    let changed = output != board.cells;
    if !changed {
        merges.clear();
        score = 0;
    }
    Ok(MoveResult {
        board: Board { width: board.width, height: board.height, cells: output },

        changed,
        merges,
