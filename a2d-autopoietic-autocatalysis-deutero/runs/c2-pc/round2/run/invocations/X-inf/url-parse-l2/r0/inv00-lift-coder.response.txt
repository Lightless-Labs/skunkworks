```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            if let (Some(hi), Some(lo)) = (hi, lo) {
                decoded.push((hi << 4) | lo);
                origins.push(i);
                i += 3;
            } else {
                return Err(CodecError::BadEscape { at: i });
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => Err(CodecError::InvalidUtf8 {
            at: origins.get(error.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };
        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn validate_component(
    input: &str,
    start: usize,
    component: Component,
) -> Result<(), ParseError> {
    match decode_component(component, input) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: start + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: start + at }),
    }
}

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    for (offset, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + offset,
            });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: port_start })?;
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(colon) => colon,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(offset, byte)| (byte == b'@').then_some(offset))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&userinfo_end) = at_positions.first() {
        if userinfo_end == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start,
            });
        }
        (
            Some(input[authority_start..authority_start + userinfo_end].to_owned()),
            authority_start + userinfo_end + 1,
        )
    } else {
        (None, authority_start)
    };

    let hostport = &input[hostport_start..authority_end];
    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let closing = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(offset) => 1 + offset,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &hostport[1..closing];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        ipv6 = true;
        let suffix_start = closing + 1;
        let suffix = &hostport[suffix_start..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(&suffix[1..], hostport_start + suffix_start + 1)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + suffix_start,
            });
        };
        (interior.to_ascii_lowercase(), port)
    } else {
        let split = hostport.find(':');
        let (host_text, port) = match split {
            Some(offset) => (
                &hostport[..offset],
                Some(parse_port(
                    &hostport[offset + 1..],
                    hostport_start + offset + 1,
                )?),
            ),
            None => (hostport, None),
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), port)
    };

    let fragment_start = input[authority_end..]
        .find('#')
        .map(|offset| authority_end + offset);
    let before_fragment_end = fragment_start.unwrap_or(input.len());
    let query_start = input[authority_end..before_fragment_end]
        .find('?')
        .map(|offset| authority_end + offset);

    let component_end = query_start.unwrap_or(before_fragment_end);
    let path_raw = &input[authority_end..component_end];
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    let query = query_start.map(|start| input[start + 1..before_fragment_end].to_owned());
    let fragment = fragment_start.map(|start| input[start + 1..].to_owned());

    if let Some(ref userinfo_text) = userinfo {
        validate_component(
            userinfo_text,
            authority_start,
            Component::UserInfo,
        )?;
    }
    if !path_raw.is_empty() {
        validate_component(path_raw, authority_end, Component::Path)?;
    }
    if let Some(start) = query_start {
        validate_component(
            &input[start + 1..before_fragment_end],
            start + 1,
            Component::Query,
        )?;
    }
    if let Some(start) = fragment_start {
        validate_component(&input[start + 1..], start + 1, Component::Fragment)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(component: Component, input: &str) -> Result<String, CodecError> {
    let _ = component;
    let source = input.as_bytes();
    let mut output = Vec::new();
    let mut index_map = Vec::new();

    for i in 0..source.len() {
        if source[i] == b'%' {
            if i + 2 >= source.len()
                || hex_value(source[i + 1]).is_none()
                || hex_value(source[i + 2]).is_none()
            {
                return Err(CodecError::BadEscape { at: i });
            }
            let byte = (hex_value(source[i + 1]).unwrap() << 4) | hex_value(source[i + 2]).unwrap();
            output.push(byte);
            index_map.push(i);
            // Skip the two hex digits while retaining a single left-to-right syntax pass.
            // The loop's index is adjusted below by consuming a manual cursor instead.
        } else {
            output.push(source[i]);
            index_map.push(i);
        }
    }

    // The loop above intentionally did not skip escape digits, so rebuild correctly using a cursor.
    output.clear();
    index_map.clear();
    let mut i = 0;
    while i < source.len() {
        if source[i] == b'%' {
            if i + 2 >= source.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let (Some(high), Some(low)) = (hex_value(source[i + 1]), hex_value(source[i + 2])) else {
                return Err(CodecError::BadEscape { at: i });
            };
            output.push((high << 4) | low);
            index_map.push(i);
            i += 3;
        } else {
            output.push(source[i]);
            index_map.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&output) {
        Ok(value) => Ok(value.to_owned()),
        Err(error) => Err(CodecError::InvalidUtf8 {
            at: index_map
                .get(error.valid_up_to())
                .copied()
                .unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut result = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        let common = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let extra = match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => b == b'/' || b == b'?',
            Component::Fragment => b == b'/' || b == b'?' || b == b'#',
        };
        if common || extra {
            result.push(b as char);
        } else {
            result.push('%');
            result.push(HEX[(b >> 4) as usize] as char);
            result.push(HEX[(b & 15) as usize] as char);
        }
    }
    result
}

fn validate_component(input: &str, start: usize, component: Component) -> Result<(), ParseError> {
    match decode_component(component, input) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: start + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: start + at }),
    }
}

fn parse_port(text: &str, start: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (offset, byte) in text.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + offset });
        }
    }
    if text.len() > 1 && text.starts_with('0') {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value = 0u32;
    for byte in text.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((byte - b'0') as u32)) {
            Some(v) if v <= 65535 => v,
            _ => return Err(ParseError::InvalidPort { at: start }),
        };
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (index, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at: index });
        }
    }

    let bytes = input.as_bytes();
    let scheme_end = match bytes.iter().position(|&byte| byte == b':') {
        Some(index) => index,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if scheme_end == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (index, byte) in bytes[..scheme_end].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: index });
        }
    }
    let marker = scheme_end + 1;
    if marker + 1 >= bytes.len() || bytes[marker] != b'/' || bytes[marker + 1] != b'/' {
        return Err(ParseError::MissingAuthority { at: marker });
    }

    let auth_begin = marker + 2;
    let auth_end = bytes[auth_begin..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|relative| auth_begin + relative)
        .unwrap_or(bytes.len());
    let authority = &input[auth_begin..auth_end];

    let mut userinfo = None;
    let mut host_begin = auth_begin;
    let mut first_at = None;
    for (relative, byte) in authority.bytes().enumerate() {
        if byte == b'@' {
            if first_at.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: auth_begin + relative,
                });
            }
            first_at = Some(relative);
        }
    }
    if let Some(relative) = first_at {
        if relative == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_begin });
        }
        userinfo = Some(input[auth_begin..auth_begin + relative].to_owned());
        host_begin = auth_begin + relative + 1;
    }

    let hostport = &input[host_begin..auth_end];
    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let close = hostport[1..]
            .bytes()
            .position(|byte| byte == b']')
            .map(|relative| relative + 1)
            .ok_or(ParseError::InvalidIpv6 { at: host_begin })?;
        let address = &hostport[1..close];
        if address.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_begin });
        }
        ipv6 = true;
        let remainder = &hostport[close + 1..];
        let port = if remainder.is_empty() {
            None
        } else if remainder.starts_with(':') {
            Some(parse_port(
                &remainder[1..],
                host_begin + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_begin + close + 1,
            });
        };
        (address.to_ascii_lowercase(), port)
    } else {
        let separator = hostport.find(':');
        let (name, port) = if let Some(relative) = separator {
            (
                &hostport[..relative],
                Some(parse_port(
                    &hostport[relative + 1..],
                    host_begin + relative + 1,
                )?),
            )
        } else {
            (hostport, None)
        };
        if name.is_empty() {
            return Err(ParseError::InvalidHost { at: host_begin });
        }
        for (relative, byte) in name.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_begin + relative,
                });
            }
        }
        (name.to_ascii_lowercase(), port)
    };

    let fragment_begin = input[auth_end..]
        .find('#')
        .map(|relative| auth_end + relative);
    let fragment_boundary = fragment_begin.unwrap_or(input.len());
    let query_begin = input[auth_end..fragment_boundary]
        .find('?')
        .map(|relative| auth_end + relative);
    let path_boundary = query_begin.unwrap_or(fragment_boundary);
    let raw_path = &input[auth_end..path_boundary];
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };
    let query = query_begin.map(|begin| input[begin + 1..fragment_boundary].to_owned());
    let fragment = fragment_begin.map(|begin| input[begin + 1..].to_owned());

    if let Some(value) = &userinfo {
        validate_component(value, auth_begin, Component::UserInfo)?;
    }
    if !raw_path.is_empty() {
        validate_component(raw_path, auth_end, Component::Path)?;
    }
    if let Some(begin) = query_begin {
        validate_component(
            &input[begin + 1..fragment_boundary],
            begin + 1,
            Component::Query,
        )?;
    }
    if let Some(begin) = fragment_begin {
        validate_component(
            &input[begin + 1..],
            begin + 1,
            Component::Fragment,
        )?;
    }

    Ok(Url {
        scheme: input[..scheme_end].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}
```
