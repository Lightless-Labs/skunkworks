    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

pub fn decode_component(component: Component, input: &str) -> Result<String, CodecError> {
    let _ = component;
    let mut bytes = Vec::with_capacity(input.len());
    let mut source = Vec::with_capacity(input.len());
    let raw = input.as_bytes();
    let mut pos = 0;
    while pos < raw.len() {
        if raw[pos] == b'%' {
            let a = raw.get(pos + 1).and_then(|b| digit(*b));
            let b = raw.get(pos + 2).and_then(|b| digit(*b));
            let (Some(a), Some(b)) = (a, b) else {
                return Err(CodecError::BadEscape { at: pos });
            };
            bytes.push((a << 4) | b);
            source.push(pos);
            pos += 3;
        } else {
            bytes.push(raw[pos]);
            source.push(pos);
            pos += 1;
        }
    }
    let decoded = match std::str::from_utf8(&bytes) {
        Ok(s) => s,
        Err(e) => {
            let at = source.get(e.valid_up_to()).copied().unwrap_or(raw.len());
            return Err(CodecError::InvalidUtf8 { at });
        }
    };
    Ok(decoded.to_owned())
}

fn digit(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut result = String::new();
    for b in text.bytes() {
        let common = b.is_ascii_alphanumeric() || b"-._~".contains(&b);
        let extra = match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => b == b'/' || b == b'?',
            Component::Fragment => b == b'/' || b == b'?' || b == b'#',
        };
        if common || extra {
            result.push(b as char);
        } else {
            const H: &[u8] = b"0123456789ABCDEF";
            result.push('%');
            result.push(H[(b >> 4) as usize] as char);
            result.push(H[(b & 15) as usize] as char);
        }
    }
    result
}

fn port(s: &str, at: usize) -> Result<u16, ParseError> {
    if s.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    if let Some(i) = s.bytes().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: at + i });
    }
    if s.len() > 1 && s.starts_with('0') {
        return Err(ParseError::InvalidPort { at });
    }
    let mut n: u32 = 0;
    for b in s.bytes() {
        n = n.saturating_mul(10).saturating_add((b - b'0') as u32);
        if n > 65535 {
            return Err(ParseError::InvalidPort { at });
        }
    }
    if n == 0 {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(n as u16)
}

fn check(s: &str, base: usize) -> Result<(), ParseError> {
    decode_component(Component::Path, s).map(|_| ()).map_err(|e| match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    })
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (i, b) in input.bytes().enumerate() {
        if !(33..=126).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    let b = input.as_bytes();
    let c = match b.iter().position(|x| *x == b':') {
        Some(c) => c,
        None => return Err(ParseError::InvalidScheme { at: b.len() }),
    };
    if c == 0 || !b[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, x) in b[1..c].iter().enumerate() {
        if !(x.is_ascii_alphanumeric() || b"+-.".contains(x)) {
            return Err(ParseError::InvalidScheme { at: i + 1 });
        }
    }
    if b.get(c + 1) != Some(&b'/') || b.get(c + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: c + 1 });
    }
    let auth = c + 3;
    let end = b[auth..].iter().position(|x| b"/?#".contains(x)).map(|x| auth + x).unwrap_or(b.len());
    let authority = &input[auth..end];
    let ats: Vec<usize> = authority.bytes().enumerate().filter_map(|(i, x)| (x == b'@').then_some(i)).collect();
    if ats.len() > 1 {
        return Err(ParseError::InvalidUserInfo { at: auth + ats[1] });
    }
    let (userinfo, hp, hs) = match ats.first() {
        Some(&i) if i == 0 => return Err(ParseError::InvalidUserInfo { at: auth }),
        Some(&i) => (Some(authority[..i].to_owned()), &authority[i + 1..], auth + i + 1),
        None => (None, authority, auth),
    };

    let (host, ipv6, port_value) = if hp.starts_with('[') {
        let close = hp.find(']').ok_or(ParseError::InvalidIpv6 { at: hs })?;
        let inner = &hp[1..close];
        if Ipv6Addr::from_str(inner).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hs });
        }
        let rest = &hp[close + 1..];
        let p = if rest.is_empty() {
            None
        } else if rest.starts_with(':') {
            Some(port(&rest[1..], hs + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost { at: hs + close + 1 });
        };
        (inner.to_ascii_lowercase(), true, p)
    } else {
        let split = hp.find(':');
        let (h, p) = if let Some(i) = split {
            (&hp[..i], Some(port(&hp[i + 1..], hs + i + 1)?))
        } else {
            (hp, None)
        };
        if h.is_empty() {
            return Err(ParseError::InvalidHost { at: hs });
        }
        for (i, x) in h.bytes().enumerate() {
            if !(x.is_ascii_alphanumeric() || b"._-".contains(&x)) {
                return Err(ParseError::InvalidHost { at: hs + i });
            }
        }
        (h.to_ascii_lowercase(), false, p)
    };

    let hash = input[end..].find('#').map(|x| end + x);
    let before_frag = hash.unwrap_or(input.len());
    let fragment = hash.map(|x| input[x + 1..].to_owned());
    let question = input[end..before_frag].find('?').map(|x| end + x);
    let before_query = question.unwrap_or(before_frag);
    let query = question.map(|x| input[x + 1..before_frag].to_owned());
    let path = if before_query == end { "/".to_owned() } else { input[end..before_query].to_owned() };

    if let Some(ref u) = userinfo {
        check(u, auth)?;
    }
    if before_query != end {
        check(&path, end)?;
    }
    if let Some(ref q) = query {
        check(q, question.unwrap() + 1)?;
    }
    if let Some(ref f) = fragment {
        check(f, hash.unwrap() + 1)?;
    }

    Ok(Url {
        scheme: input[..c].to_ascii_lowercase(),
        userinfo,
        host,
        port: port_value,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let at = offsets
                .get(error.valid_up_to())
                .copied()
                .unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn validate_component(
    component: Component,
    text: &str,
    base: usize,
) -> Result<(), ParseError> {
    match decode_component(component, text) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: base + at }),
    }
}

fn parse_port(port: &str, at: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }

    for (offset, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + offset });
        }
    }

    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = value * 10 + (byte - b'0') as u32;
        if value > 65535 {
            return Err(ParseError::InvalidPort { at });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }

    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, byte) in bytes[..colon].iter().copied().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let mut at_signs = authority
        .bytes()
        .enumerate()
        .filter_map(|(offset, byte)| (byte == b'@').then_some(offset));
    let first_at = at_signs.next();
    if let Some(second_at) = at_signs.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second_at,
        });
    }

    let (userinfo, host_start) = if let Some(userinfo_end) = first_at {
        if userinfo_end == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start,
            });
        }
        (
            Some(authority[..userinfo_end].to_owned()),
            userinfo_end + 1,
        )
    } else {
        (None, 0)
    };

    let host_port = &authority[host_start..];
    let absolute_host_start = authority_start + host_start;
    let mut ipv6 = false;
    let (host, port) = if host_port.starts_with('[') {
        let closing = match host_port.as_bytes()[1..]
            .iter()
            .position(|&byte| byte == b']')
        {
            Some(offset) => offset + 1,
            None => {
                return Err(ParseError::InvalidIpv6 {
                    at: absolute_host_start,
                });
            }
        };
        let interior = &host_port[1..closing];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 {
                at: absolute_host_start,
            });
        }

        ipv6 = true;
        let suffix = &host_port[closing + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                absolute_host_start + closing + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: absolute_host_start + closing + 1,
            });
        };
        (interior.to_ascii_lowercase(), parsed_port)
    } else {
        let host_colon = host_port.bytes().position(|byte| byte == b':');
        let (host_text, parsed_port) = match host_colon {
            Some(offset) => (
                &host_port[..offset],
                Some(parse_port(
                    &host_port[offset + 1..],
                    absolute_host_start + offset + 1,
                )?),
            ),
            None => (host_port, None),
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost {
                at: absolute_host_start,
            });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: absolute_host_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), parsed_port)
    };

    let fragment_start = bytes[authority_end..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|offset| authority_end + offset);
    let before_fragment_end = fragment_start.unwrap_or(bytes.len());

    let query_start = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|offset| authority_end + offset);

    let path_end = query_start.unwrap_or(before_fragment_end);
    let raw_path = &input[authority_end..path_end];
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };

    let query = query_start.map(|start| input[start + 1..before_fragment_end].to_owned());
    let fragment = fragment_start.map(|start| input[start + 1..].to_owned());

    if let Some(userinfo_text) = userinfo.as_deref() {
        let userinfo_at = authority_start;
        validate_component(Component::UserInfo, userinfo_text, userinfo_at)?;
    }
    if !raw_path.is_empty() {
        validate_component(Component::Path, raw_path, authority_end)?;
    }
    if let Some(start) = query_start {
        validate_component(
            Component::Query,
            query.as_deref().unwrap_or(""),
            start + 1,
        )?;
    }
    if let Some(start) = fragment_start {
        validate_component(
            Component::Fragment,
            fragment.as_deref().unwrap_or(""),
            start + 1,
        )?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn hex_digit(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let source = input.as_bytes();
    let mut output = Vec::with_capacity(source.len());
    let mut source_at = Vec::with_capacity(source.len());
    let mut cursor = 0;

    while cursor < source.len() {
        if source[cursor] != b'%' {
            output.push(source[cursor]);
            source_at.push(cursor);
            cursor += 1;
            continue;
        }

        if cursor + 2 >= source.len() {
            return Err(CodecError::BadEscape { at: cursor });
        }
        let high = hex_digit(source[cursor + 1]);
        let low = hex_digit(source[cursor + 2]);
        if let (Some(high), Some(low)) = (high, low) {
            output.push(high * 16 + low);
            source_at.push(cursor);
            cursor += 3;
        } else {
            return Err(CodecError::BadEscape { at: cursor });
        }
    }

    match std::str::from_utf8(&output) {
        Ok(decoded) => Ok(decoded.to_owned()),
        Err(error) => Err(CodecError::InvalidUtf8 {
            at: source_at[error.valid_up_to()],
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut result = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for byte in text.bytes() {
        let allowed = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => byte == b'/' || byte == b'?',
                Component::Fragment => byte == b'/' || byte == b'?' || byte == b'#',
            };

        if allowed {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 15) as usize] as char);
        }
    }

    result
}

fn check_component(component: Component, source: &str, base: usize) -> Result<(), ParseError> {
    let error = match decode_component(component, source) {
        Ok(_) => return Ok(()),
        Err(CodecError::BadEscape { at }) => ParseError::BadEscape { at: base + at },
        Err(CodecError::InvalidUtf8 { at }) => ParseError::InvalidUtf8 { at: base + at },
    };
    Err(error)
}

fn decimal_port(digits: &str, start: usize) -> Result<u16, ParseError> {
    if digits.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (index, byte) in digits.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + index });
        }
    }
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut number: u32 = 0;
    for digit in digits.bytes() {
        number = number * 10 + u32::from(digit - b'0');
        if number > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if number == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(number as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if byte < 0x21 || byte > 0x7e {
            return Err(ParseError::InvalidByte { at: index });
        }
    }

    let source = input.as_bytes();
    let scheme_end = match source.iter().position(|byte| *byte == b':') {
        Some(index) => index,
        None => return Err(ParseError::InvalidScheme { at: source.len() }),
    };

    if scheme_end == 0 || !source[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for index in 1..scheme_end {
        let byte = source[index];
        if !(byte.is_ascii_alphanumeric() || byte == b'+' || byte == b'-' || byte == b'.') {
            return Err(ParseError::InvalidScheme { at: index });
        }
    }
    if source.get(scheme_end + 1) != Some(&b'/')
        || source.get(scheme_end + 2) != Some(&b'/')
    {
        return Err(ParseError::MissingAuthority {
            at: scheme_end + 1,
        });
    }

    let authority_start = scheme_end + 3;
    let authority_end = source[authority_start..]
        .iter()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .map_or(source.len(), |index| authority_start + index);
    let authority = &input[authority_start..authority_end];

    let ats: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(index, byte)| (byte == b'@').then_some(index))
        .collect();
    if ats.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + ats[1],
        });
    }
    let (userinfo, host_offset) = if let Some(&index) = ats.first() {
        if index == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start,
            });
        }
        (Some(authority[..index].to_owned()), index + 1)
    } else {
        (None, 0)
    };

    let host_port = &authority[host_offset..];
    let absolute_start = authority_start + host_offset;

    let (host, port, ipv6) = if host_port.starts_with('[') {
        let close = host_port.as_bytes()[1..]
            .iter()
            .position(|byte| *byte == b']')
            .map(|index| index + 1)
            .ok_or(ParseError::InvalidIpv6 { at: absolute_start })?;
        let inside = &host_port[1..close];
        Ipv6Addr::from_str(inside)
            .map_err(|_| ParseError::InvalidIpv6 { at: absolute_start })?;

        let suffix = &host_port[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(decimal_port(
                &suffix[1..],
                absolute_start + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: absolute_start + close + 1,
            });
        };
        (inside.to_ascii_lowercase(), port, true)
    } else {
        let split = host_port.bytes().position(|byte| byte == b':');
        let (hostname, port) = match split {
            Some(index) => (
                &host_port[..index],
                Some(decimal_port(
                    &host_port[index + 1..],
                    absolute_start + index + 1,
                )?),
            ),
            None => (host_port, None),
        };

        if hostname.is_empty() {
            return Err(ParseError::InvalidHost { at: absolute_start });
        }
        for (index, byte) in hostname.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: absolute_start + index,
                });
            }
        }
        (hostname.to_ascii_lowercase(), port, false)
    };

    let fragment_at = source[authority_end..]
        .iter()
        .position(|byte| *byte == b'#')
        .map(|index| authority_end + index);
    let end_before_fragment = fragment_at.unwrap_or(source.len());
    let query_at = source[authority_end..end_before_fragment]
        .iter()
        .position(|byte| *byte == b'?')
        .map(|index| authority_end + index);

    let path_end = query_at.unwrap_or(end_before_fragment);
    let raw_path = &input[authority_end..path_end];
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };
    let query = query_at.map(|index| input[index + 1..end_before_fragment].to_owned());
    let fragment = fragment_at.map(|index| input[index + 1..].to_owned());

    if let Some(value) = userinfo.as_deref() {
        check_component(Component::UserInfo, value, authority_start)?;
    }
    if !raw_path.is_empty() {
        check_component(Component::Path, raw_path, authority_end)?;
    }
    if let Some(index) = query_at {
        check_component(
            Component::Query,
            query.as_deref().unwrap_or(""),
            index + 1,
        )?;
    }
    if let Some(index) = fragment_at {
        check_component(
            Component::Fragment,
            fragment.as_deref().unwrap_or(""),
            index + 1,
        )?;
    }

    Ok(Url {
        scheme: input[..scheme_end].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let lo = hex_value(bytes[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn validate_component(input: &str, base: usize) -> Result<(), ParseError> {
    decode_component(Component::Path, input).map(|_| ()).map_err(|e| match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    })
}

fn parse_port(port: &str, start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value: u32 = 0;
    for b in port.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: start })?;
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let after_marker = colon + 3;
    let auth_end = bytes[after_marker..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| after_marker + n)
        .unwrap_or(bytes.len());
    let authority = &input[after_marker..auth_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: after_marker + at_positions[1],
        });
    }
    let (userinfo, hostpart, host_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: after_marker + at,
            });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            after_marker + at + 1,
        )
    } else {
        (None, authority, after_marker)
    };

    let (host, ipv6, port) = if hostpart.starts_with('[') {
        let close = match hostpart.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &hostpart[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let suffix = &hostpart[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                host_start + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), true, port)
    } else {
        let split = hostpart.find(':');
        let (host_text, port) = if let Some(i) = split {
            (
                &hostpart[..i],
                Some(parse_port(&hostpart[i + 1..], host_start + i + 1)?),
            )
        } else {
            (hostpart, None)
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, b) in host_text.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_start + i });
            }
        }
        (host_text.to_ascii_lowercase(), false, port)
    };

    let mut fragment = None;
    let pre_fragment_end = if let Some(hash) = input[auth_end..].find('#') {
        let start = auth_end + hash + 1;
        fragment = Some(input[start..].to_owned());
        auth_end + hash
    } else {
        input.len()
    };

    let mut query = None;
    let pre_query_end = if let Some(q) = input[auth_end..pre_fragment_end].find('?') {
        let q_abs = auth_end + q;
        query = Some(input[q_abs + 1..pre_fragment_end].to_owned());
        q_abs
    } else {
        pre_fragment_end
    };

    let path = if pre_query_end == auth_end {
        "/".to_owned()
    } else {
        input[auth_end..pre_query_end].to_owned()
    };

    if let Some(ref s) = userinfo {
        validate_component(s, after_marker)?;
    }
    if path != "/" || pre_query_end != auth_end {
        validate_component(&path, auth_end)?;
    }
    if let Some(ref s) = query {
        let q_at = auth_end + input[auth_end..pre_fragment_end].find('?').unwrap() + 1;
        validate_component(s, q_at)?;
    }
    if let Some(ref s) = fragment {
        let f_at = auth_end + input[auth_end..].find('#').unwrap() + 1;
        validate_component(s, f_at)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            if let (Some(hi), Some(lo)) = (hi, lo) {
                decoded.push((hi << 4) | lo);
                origins.push(i);
                i += 3;
            } else {
                return Err(CodecError::BadEscape { at: i });
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => Err(CodecError::InvalidUtf8 {
            at: origins.get(error.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };
        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn validate_component(
    input: &str,
    start: usize,
    component: Component,
) -> Result<(), ParseError> {
    match decode_component(component, input) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: start + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: start + at }),
    }
}

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    for (offset, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + offset,
            });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: port_start })?;
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(colon) => colon,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(offset, byte)| (byte == b'@').then_some(offset))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&userinfo_end) = at_positions.first() {
        if userinfo_end == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start,
            });
        }
        (
            Some(input[authority_start..authority_start + userinfo_end].to_owned()),
            authority_start + userinfo_end + 1,
        )
    } else {
        (None, authority_start)
    };

    let hostport = &input[hostport_start..authority_end];
    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let closing = match hostport.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(offset) => 1 + offset,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &hostport[1..closing];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        ipv6 = true;
        let suffix_start = closing + 1;
        let suffix = &hostport[suffix_start..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(parse_port(&suffix[1..], hostport_start + suffix_start + 1)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + suffix_start,
            });
        };
        (interior.to_ascii_lowercase(), port)
    } else {
        let split = hostport.find(':');
        let (host_text, port) = match split {
            Some(offset) => (
                &hostport[..offset],
                Some(parse_port(
                    &hostport[offset + 1..],
                    hostport_start + offset + 1,
                )?),
            ),
            None => (hostport, None),
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), port)
    };

    let fragment_start = input[authority_end..]
        .find('#')
        .map(|offset| authority_end + offset);
    let before_fragment_end = fragment_start.unwrap_or(input.len());
    let query_start = input[authority_end..before_fragment_end]
        .find('?')
        .map(|offset| authority_end + offset);

    let component_end = query_start.unwrap_or(before_fragment_end);
    let path_raw = &input[authority_end..component_end];
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    let query = query_start.map(|start| input[start + 1..before_fragment_end].to_owned());
    let fragment = fragment_start.map(|start| input[start + 1..].to_owned());

    if let Some(ref userinfo_text) = userinfo {
        validate_component(
            userinfo_text,
            authority_start,
            Component::UserInfo,
        )?;
    }
    if !path_raw.is_empty() {
        validate_component(path_raw, authority_end, Component::Path)?;
    }
    if let Some(start) = query_start {
        validate_component(
            &input[start + 1..before_fragment_end],
            start + 1,
            Component::Query,
        )?;
    }
    if let Some(start) = fragment_start {
        validate_component(&input[start + 1..], start + 1, Component::Fragment)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,

pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

    UserInfo,
    Path,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(component: Component, input: &str) -> Result<String, CodecError> {
    let _ = component;
    let source = input.as_bytes();
    let mut output = Vec::new();
    let mut index_map = Vec::new();

    for i in 0..source.len() {
        if source[i] == b'%' {
            if i + 2 >= source.len()
                || hex_value(source[i + 1]).is_none()
                || hex_value(source[i + 2]).is_none()
            {
                return Err(CodecError::BadEscape { at: i });
            }
            let byte = (hex_value(source[i + 1]).unwrap() << 4) | hex_value(source[i + 2]).unwrap();
            output.push(byte);
            index_map.push(i);
            // Skip the two hex digits while retaining a single left-to-right syntax pass.
            // The loop's index is adjusted below by consuming a manual cursor instead.
        } else {
            output.push(source[i]);
            index_map.push(i);
        }
    }

    // The loop above intentionally did not skip escape digits, so rebuild correctly using a cursor.
    output.clear();
    index_map.clear();
    let mut i = 0;
    while i < source.len() {
        if source[i] == b'%' {
            if i + 2 >= source.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let (Some(high), Some(low)) = (hex_value(source[i + 1]), hex_value(source[i + 2])) else {
                return Err(CodecError::BadEscape { at: i });
            };
            output.push((high << 4) | low);
            index_map.push(i);
            i += 3;
        } else {
            output.push(source[i]);
            index_map.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&output) {
        Ok(value) => Ok(value.to_owned()),
        Err(error) => Err(CodecError::InvalidUtf8 {
            at: index_map
                .get(error.valid_up_to())
                .copied()
                .unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut result = String::new();
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        let common = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let extra = match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => b == b'/' || b == b'?',
            Component::Fragment => b == b'/' || b == b'?' || b == b'#',
        };
        if common || extra {
            result.push(b as char);
        } else {
            result.push('%');
            result.push(HEX[(b >> 4) as usize] as char);
            result.push(HEX[(b & 15) as usize] as char);
        }
    }
    result
}

fn validate_component(input: &str, start: usize, component: Component) -> Result<(), ParseError> {
    match decode_component(component, input) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: start + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: start + at }),
    }
}

fn parse_port(text: &str, start: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (offset, byte) in text.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + offset });
        }
    }
    if text.len() > 1 && text.starts_with('0') {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value = 0u32;
    for byte in text.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((byte - b'0') as u32)) {
            Some(v) if v <= 65535 => v,
            _ => return Err(ParseError::InvalidPort { at: start }),
        };
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (index, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at: index });
        }
    }

    let bytes = input.as_bytes();
    let scheme_end = match bytes.iter().position(|&byte| byte == b':') {
        Some(index) => index,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if scheme_end == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (index, byte) in bytes[..scheme_end].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: index });
        }
    }
    let marker = scheme_end + 1;
    if marker + 1 >= bytes.len() || bytes[marker] != b'/' || bytes[marker + 1] != b'/' {
        return Err(ParseError::MissingAuthority { at: marker });
    }

    let auth_begin = marker + 2;
    let auth_end = bytes[auth_begin..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|relative| auth_begin + relative)
        .unwrap_or(bytes.len());
    let authority = &input[auth_begin..auth_end];

    let mut userinfo = None;
    let mut host_begin = auth_begin;
    let mut first_at = None;
    for (relative, byte) in authority.bytes().enumerate() {
        if byte == b'@' {
            if first_at.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: auth_begin + relative,
                });
            }
            first_at = Some(relative);
        }
    }
    if let Some(relative) = first_at {
        if relative == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_begin });
        }
        userinfo = Some(input[auth_begin..auth_begin + relative].to_owned());
        host_begin = auth_begin + relative + 1;
    }

    let hostport = &input[host_begin..auth_end];
    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let close = hostport[1..]
            .bytes()
            .position(|byte| byte == b']')
            .map(|relative| relative + 1)
            .ok_or(ParseError::InvalidIpv6 { at: host_begin })?;
        let address = &hostport[1..close];
        if address.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_begin });
        }
        ipv6 = true;
        let remainder = &hostport[close + 1..];
        let port = if remainder.is_empty() {
            None
        } else if remainder.starts_with(':') {
            Some(parse_port(
                &remainder[1..],
                host_begin + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_begin + close + 1,
            });
        };
        (address.to_ascii_lowercase(), port)
    } else {
        let separator = hostport.find(':');
        let (name, port) = if let Some(relative) = separator {
            (
                &hostport[..relative],
                Some(parse_port(
                    &hostport[relative + 1..],
                    host_begin + relative + 1,
                )?),
            )
        } else {
            (hostport, None)
        };
        if name.is_empty() {
            return Err(ParseError::InvalidHost { at: host_begin });
        }
        for (relative, byte) in name.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_begin + relative,
                });
            }
        }
        (name.to_ascii_lowercase(), port)
    };

    let fragment_begin = input[auth_end..]
        .find('#')
        .map(|relative| auth_end + relative);
    let fragment_boundary = fragment_begin.unwrap_or(input.len());
    let query_begin = input[auth_end..fragment_boundary]
        .find('?')
        .map(|relative| auth_end + relative);
    let path_boundary = query_begin.unwrap_or(fragment_boundary);
    let raw_path = &input[auth_end..path_boundary];
    let path = if raw_path.is_empty() {
        "/".to_owned()
    } else {
        raw_path.to_owned()
    };
    let query = query_begin.map(|begin| input[begin + 1..fragment_boundary].to_owned());
    let fragment = fragment_begin.map(|begin| input[begin + 1..].to_owned());

    if let Some(value) = &userinfo {
        validate_component(value, auth_begin, Component::UserInfo)?;
    }
    if !raw_path.is_empty() {
        validate_component(raw_path, auth_end, Component::Path)?;
    }
    if let Some(begin) = query_begin {
        validate_component(
            &input[begin + 1..fragment_boundary],
            begin + 1,
            Component::Query,
        )?;
    }
    if let Some(begin) = fragment_begin {
        validate_component(
            &input[begin + 1..],
            begin + 1,
            Component::Fragment,
        )?;
    }

    Ok(Url {
        scheme: input[..scheme_end].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}
