fn is_hex(b: u8) -> bool {
    b.is_ascii_hexdigit()
}

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'a'..=b'f' => b - b'a' + 10,
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() || !is_hex(bytes[i + 1]) || !is_hex(bytes[i + 2]) {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]));
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn map_codec_parse(e: CodecError, base: usize) -> ParseError {
    match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

fn validate_component(component: Component, text: &str, at: usize) -> Result<(), ParseError> {
    decode_component(component, text)
        .map(|_| ())
        .map_err(|e| map_codec_parse(e, at))
}

fn parse_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {
    if port_text.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    for (i, b) in port_text.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: port_start + i });
        }
    }
    if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    let mut value: u32 = 0;
    for b in port_text.bytes() {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u32);
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }
    let auth_start = colon + 3;
    let tail_rel = input[auth_start..]
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(input.len() - auth_start);
    let auth_end = auth_start + tail_rel;
    let authority = &input[auth_start..auth_end];
    let auth_bytes = authority.as_bytes();

    let ats: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if ats.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + ats[1],
        });
    }
    let (userinfo, host_start_rel) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };
    let host_start = auth_start + host_start_rel;
    let hostport = &authority[host_start_rel..];
    let (host, port, ipv6) = if hostport.starts_with('[') {
        let open = host_start;
        let close_rel = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &hostport[1..close_rel];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        let suffix = &hostport[close_rel + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], host_start + close_rel + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close_rel + 1,
            });
        };
        (interior.to_ascii_lowercase(), parsed_port, true)
    } else {
        let split = hostport.as_bytes().iter().position(|&b| b == b':');
        let (host_text, port_text) = if let Some(i) = split {
            (&hostport[..i], Some((&hostport[i + 1..], host_start + i + 1)))
        } else {
            (hostport, None)
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, b) in host_text.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_start + i });
            }
        }
        let parsed_port = match port_text {
            Some((p, at)) => Some(parse_port(p, at)?),
            None => None,
        };
        (host_text.to_ascii_lowercase(), parsed_port, false)
    };

    let suffix = &input[auth_end..];
    let fragment_rel = suffix.find('#');
    let before_fragment = fragment_rel.map(|i| &suffix[..i]).unwrap_or(suffix);
    let fragment = fragment_rel.map(|i| suffix[i + 1..].to_owned());
    let query_rel = before_fragment.find('?');
    let path_and_query = query_rel
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_rel.map(|i| before_fragment[i + 1..].to_owned());
    let path = if path_and_query.is_empty() {
        "/".to_owned()
    } else {
        path_and_query.to_owned()
    };

    if let Some(ref ui) = userinfo {
        validate_component(Component::UserInfo, ui, auth_start)?;
    }
    if path_and_query.len() > 0 {
        validate_component(
            Component::Path,
            &path,
            auth_end,
        )?;
    }
    if let Some(ref q) = query {
        let q_at = auth_end + query_rel.unwrap() + 1;
        validate_component(Component::Query, q, q_at)?;
    }
    if let Some(ref f) = fragment {
        let f_at = auth_end + fragment_rel.unwrap() + 1;
        validate_component(Component::Fragment, f, f_at)?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let parts: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_nav = false;
    for (i, raw) in parts.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 },
        })?;
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        if is_dot {
            terminal_nav = i + 1 == parts.len();
        } else if is_dotdot {
            if !stack.is_empty() {
                stack.pop();
            }
            terminal_nav = i + 1 == parts.len();
        } else {
            stack.push(raw);
            terminal_nav = false;
        }
    }
    if terminal_nav {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn global_scan_reference(reference: &str) -> Result<(), ParseError> {
    for (i, b) in reference.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn raw_url(u: &Url) -> String {
    let mut s = String::new();
    s.push_str(&u.scheme);
    s.push_str("://");
    if let Some(ref info) = u.userinfo {
        s.push_str(info);
        s.push('@');
    }
    if u.ipv6 {
        s.push('[');
        s.push_str(&u.host);
        s.push(']');
    } else {
        s.push_str(&u.host);
    }
    if let Some(port) = u.port {
        s.push(':');
        s.push_str(&port.to_string());
    }
    s
}

fn path_error(e: PathError) -> ResolveError {
    ResolveError::Path(e)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let mut base_url = parse_url(base).map_err(ResolveError::Base)?;
    base_url.path = normalise_path(&base_url.path).map_err(path_error)?;

    global_scan_reference(reference).map_err(ResolveError::Reference)?;

    let first_delim = reference
        .bytes()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference[..first_delim].contains(':') {
        let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(path_error)?;
        return Ok(u);
    }
    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(path_error)?;
        return Ok(u);
    }

    let fragment_pos = reference.find('#');
    let before_fragment = fragment_pos
        .map(|i| &reference[..i])
        .unwrap_or(reference);
    let fragment = fragment_pos.map(|i| reference[i + 1..].to_owned());
    let query_pos = before_fragment.find('?');
    let reference_path = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let reference_query = query_pos.map(|i| before_fragment[i + 1..].to_owned());

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            base_url.path.clone(),
            if query_pos.is_none() {
                base_url.query.clone()
            } else {
                reference_query
            },
        )
    } else {
        let merged = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_url.path[..=slash], reference_path)
        };
        (merged, reference_query)
    };

    let mut constructed = raw_url(&base_url);
    constructed.push_str(&selected_path);
    if let Some(ref q) = selected_query {
        constructed.push('?');
        constructed.push_str(q);
    }
    if let Some(ref f) = fragment {
        constructed.push('#');
        constructed.push_str(f);
    }
    let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
    u.path = normalise_path(&u.path).map_err(path_error)?;
    Ok(u)
}

fn is_hex(b: u8) -> bool {

fn hex_value(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'a'..=b'f' => b - b'a' + 10,
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() || !is_hex(bytes[i + 1]) || !is_hex(bytes[i + 2]) {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]));
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

fn component_safe(component: Component, b: u8) -> bool {
    b.is_ascii_alphanumeric()
        || matches!(b, b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if component_safe(component, b) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn validate_port(port: &str, port_start: usize) -> Result<Option<u16>, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: port_start + i });
        }
    }
    let bytes = port.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    let mut value: u32 = 0;
    for b in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
        {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(Some(value as u16))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Some(at) = global_byte_error(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| authority_start + n)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }
    let (userinfo, hostport, hostport_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            authority_start + at + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        let suffix = &hostport[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            validate_port(&suffix[1..], hostport_start + close + 2)?
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close + 1,
            });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), parsed_port)
    } else {
        let split = hostport.find(':');
        let (host_text, port_text, port_start) = match split {
            Some(i) => (
                &hostport[..i],
                Some(&hostport[i + 1..]),
                hostport_start + i + 1,
            ),
            None => (hostport, None, hostport_start + hostport.len()),
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in host_text.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + i,
                });
            }
        }
        let parsed_port = match port_text {
            Some(p) => validate_port(p, port_start)?,
            None => None,
        };
        (host_text.to_ascii_lowercase(), parsed_port)
    };

    let tail = &input[authority_end..];
    let (before_fragment, fragment, fragment_offset) = match tail.find('#') {
        Some(i) => (
            &tail[..i],
            Some(&tail[i + 1..]),
            authority_end + i + 1,
        ),
        None => (tail, None, input.len()),
    };
    let (path_query, query, query_offset) = match before_fragment.find('?') {
        Some(i) => (
            &before_fragment[..i],
            Some(&before_fragment[i + 1..]),
            authority_end + i + 1,
        ),
        None => (before_fragment, None, input.len()),
    };
    let (path, path_offset) = if path_query.is_empty() {
        ("/".to_owned(), input.len())
    } else {
        (path_query.to_owned(), authority_end)
    };

    if let Some(info) = userinfo.as_deref() {
        validate_component(info, authority_start)?;
    }
    if path_query.len() != 0 {
        validate_component(&path, path_offset)?;
    }
    if let Some(q) = query {
        validate_component(q, query_offset)?;
    }
    if let Some(f) = fragment {
        validate_component(f, fragment_offset)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded = decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded[1..].split('/').collect();
    let terminal_navigation = decoded_segments
        .last()
        .map(|s| *s == "." || *s == "..")
        .unwrap_or(false);

    let mut stack: Vec<&str> = Vec::new();
    for (raw, decoded_segment) in raw_segments.into_iter().zip(decoded_segments.into_iter()) {
        match decoded_segment {
            "." => {}
            ".." => {
                stack.pop();
            }
            _ => stack.push(raw),
        }
    }
    if terminal_navigation {
        stack.push("");
    }
    Ok(format!("/{}", stack.join("/")))
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path =
        normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    if let Some(at) = global_byte_error(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let bytes = reference.as_bytes();
    let first_delimiter = bytes
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());
    if bytes[..first_delimiter].contains(&b':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }
    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let (before_fragment, ref_fragment) = match reference.find('#') {
        Some(i) => (&reference[..i], Some(&reference[i + 1..])),
        None => (reference, None),
    };
    let (ref_path, ref_query) = match before_fragment.find('?') {
        Some(i) => (&before_fragment[..i], Some(&before_fragment[i + 1..])),
        None => (before_fragment, None),
    };

    let selected_path = if ref_path.is_empty() {
        base_path.clone()
    } else if ref_path.starts_with('/') {
        ref_path.to_owned()
    } else {
        let cut = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
        format!("{}{}", &base_path[..cut], ref_path)
    };
    let selected_query = if ref_path.is_empty() && ref_query.is_none() {
        base_url.query.as_deref()
    } else {
        ref_query
    };

    let mut constructed = format!("{}://", base_url.scheme);
    if let Some(info) = base_url.userinfo.as_deref() {
        constructed.push_str(info);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&selected_path);
    if let Some(query) = selected_query {
        constructed.push('?');
        constructed.push_str(query);
    }
    if let Some(fragment) = ref_fragment {
        constructed.push('#');
        constructed.push_str(fragment);
    }

    let mut url = parse_url(&constructed).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let source = input.as_bytes();
    let mut output = Vec::with_capacity(source.len());
    let mut source_offsets = Vec::with_capacity(source.len());
    let mut pos = 0;

    while pos < source.len() {
        let byte = source[pos];
        if byte != b'%' {
            output.push(byte);
            source_offsets.push(pos);
            pos += 1;
            continue;
        }
        if pos + 2 >= source.len()
            || !source[pos + 1].is_ascii_hexdigit()
            || !source[pos + 2].is_ascii_hexdigit()
        {
            return Err(CodecError::BadEscape { at: pos });
        }
        let high = source[pos + 1];
        let low = source[pos + 2];
        let nibble = |x: u8| match x {
            b'0'..=b'9' => x - b'0',
            b'a'..=b'f' => x - b'a' + 10,
            b'A'..=b'F' => x - b'A' + 10,
            _ => 0,
        };
        output.push(nibble(high) * 16 + nibble(low));
        source_offsets.push(pos);
        pos += 3;
    }

    match std::str::from_utf8(&output) {
        Ok(s) => Ok(s.to_owned()),
        Err(error) => Err(CodecError::InvalidUtf8 {
            at: source_offsets
                .get(error.valid_up_to())
                .copied()
                .unwrap_or(input.len()),
        }),
    }
}

fn safe(component: Component, byte: u8) -> bool {

            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),

pub fn encode_component(component: Component, text: &str) -> String {
    let mut result = String::with_capacity(text.len());
    for byte in text.as_bytes().iter().copied() {
        if safe(component, byte) {
            result.push(byte as char);
        } else {
            const DIGITS: &[u8; 16] = b"0123456789ABCDEF";
            result.push('%');
            result.push(DIGITS[(byte >> 4) as usize] as char);
            result.push(DIGITS[(byte & 15) as usize] as char);
        }
    }
    result

fn read_port(text: &str, offset: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: offset });
    }
    if let Some(i) = text.bytes().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: offset + i });
    }
    if text.len() > 1 && text.starts_with('0') {
        return Err(ParseError::InvalidPort { at: offset });
    }
    let mut n = 0u32;
    for b in text.bytes() {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: offset })?;
    }
    if n == 0 || n > 65535 {
        return Err(ParseError::InvalidPort { at: offset });
    }
    Ok(n as u16)
}

fn authority_parse(
    authority: &str,
    start: usize,
) -> Result<(Option<String>, String, Option<u16>, bool), ParseError> {
    let mut first_at = None;
    for (i, b) in authority.bytes().enumerate() {
        if b == b'@' {
            if first_at.is_some() {
                return Err(ParseError::InvalidUserInfo { at: start + i });
            }
            first_at = Some(i);
        }
    }
    let (userinfo, rest, rest_start) = match first_at {
        Some(0) => return Err(ParseError::InvalidUserInfo { at: start }),
        Some(i) => (
            Some(authority[..i].to_owned()),
            &authority[i + 1..],
            start + i + 1,
        ),
        None => (None, authority, start),
    };

    if rest.starts_with('[') {
        let close = match rest.bytes().position(|b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: rest_start }),
        };
        let interior = &rest[1..close];
        if interior.parse::<Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: rest_start });
        }
        let after = &rest[close + 1..];
        let port = if after.is_empty() {
            None
        } else if let Some(p) = after.strip_prefix(':') {
            Some(read_port(p, rest_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: rest_start + close + 1,
            });
        };
        return Ok((userinfo, interior.to_ascii_lowercase(), port, true));
    }

    let (host, port_text, port_start) = match rest.find(':') {
        Some(i) => (&rest[..i], Some(&rest[i + 1..]), rest_start + i + 1),
        None => (rest, None, rest_start + rest.len()),
    };
    if host.is_empty() {
        return Err(ParseError::InvalidHost { at: rest_start });
    }
    for (i, b) in host.bytes().enumerate() {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost { at: rest_start + i });
        }
    }
    let port = match port_text {
        Some(p) => Some(read_port(p, port_start)?),
        None => None,
    };
    Ok((userinfo, host.to_ascii_lowercase(), port, false))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Some(at) = invalid_raw_byte(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|b| *b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|b| matches!(*b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(input.len());
    let (userinfo, host, port, ipv6) =
        authority_parse(&input[authority_start..authority_end], authority_start)?;

    let suffix = &input[authority_end..];
    let (pre_fragment, fragment) = match suffix.find('#') {
        Some(i) => (&suffix[..i], Some((&suffix[i + 1..], authority_end + i + 1))),
        None => (suffix, None),
    };
    let (path_raw, query) = match pre_fragment.find('?') {
        Some(i) => (
            &pre_fragment[..i],
            Some((&pre_fragment[i + 1..], authority_end + i + 1)),
        ),
        None => (pre_fragment, None),
    };

    let (path, path_offset) = if path_raw.is_empty() {
        ("/".to_owned(), input.len())
    } else {
        (path_raw.to_owned(), authority_end)
    };

    if let Some(info) = userinfo.as_deref() {
        validate_raw_component(info, authority_start)?;
    }
    if !path_raw.is_empty() {
        validate_raw_component(path_raw, path_offset)?;
    }
    if let Some((q, offset)) = query {
        validate_raw_component(q, offset)?;
    }
    if let Some((f, offset)) = fragment {
        validate_raw_component(f, offset)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query: query.map(|(s, _)| s.to_owned()),
        fragment: fragment.map(|(s, _)| s.to_owned()),
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded = decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_parts: Vec<&str> = path[1..].split('/').collect();
    let decoded_parts: Vec<&str> = decoded[1..].split('/').collect();
    let trailing_navigation = matches!(decoded_parts.last(), Some(&".") | Some(&".."));

    let mut output: Vec<&str> = Vec::new();
    for (raw, part) in raw_parts.iter().zip(decoded_parts.iter()) {
        match *part {
            "." => {}
            ".." => {
                output.pop();
            }
            _ => output.push(*raw),
        }
    }
    if trailing_navigation {
        output.push("");
    }
    Ok(format!("/{}", output.join("/")))
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let normalized_base =
        normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    if let Some(at) = invalid_raw_byte(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    let reference_bytes = reference.as_bytes();
    let before_delimiter = reference_bytes
        .iter()
        .position(|b| matches!(*b, b'/' | b'?' | b'#'))
        .unwrap_or(reference_bytes.len());
    if reference_bytes[..before_delimiter].contains(&b':') {
        let mut parsed = parse_url(reference).map_err(ResolveError::Reference)?;
        parsed.path = normalise_path(&parsed.path).map_err(ResolveError::Path)?;
        return Ok(parsed);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut parsed = parse_url(&absolute).map_err(ResolveError::Reference)?;
        parsed.path = normalise_path(&parsed.path).map_err(ResolveError::Path)?;
        return Ok(parsed);
    }

    let (without_fragment, fragment) = match reference.find('#') {
        Some(i) => (&reference[..i], Some(&reference[i + 1..])),
        None => (reference, None),
    };
    let (reference_path, query) = match without_fragment.find('?') {
        Some(i) => (&without_fragment[..i], Some(&without_fragment[i + 1..])),
        None => (without_fragment, None),
    };

    let path = if reference_path.is_empty() {
        normalized_base.clone()
    } else if reference_path.starts_with('/') {
        reference_path.to_owned()
    } else {
        let slash = normalized_base.rfind('/').map(|i| i + 1).unwrap_or(0);
        format!("{}{}", &normalized_base[..slash], reference_path)
    };
    let selected_query = if reference_path.is_empty() && query.is_none() {
        base_url.query.as_deref()
    } else {
        query
    };

    let mut raw = format!("{}://", base_url.scheme);
    if let Some(info) = base_url.userinfo.as_deref() {
        raw.push_str(info);
        raw.push('@');
    }
    if base_url.ipv6 {
        raw.push('[');
        raw.push_str(&base_url.host);
        raw.push(']');
    } else {
        raw.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        raw.push(':');
        raw.push_str(&port.to_string());
    }
    raw.push_str(&path);
    if let Some(q) = selected_query {
        raw.push('?');
        raw.push_str(q);
    }
    if let Some(f) = fragment {
        raw.push('#');
        raw.push_str(f);
    }

    let mut parsed = parse_url(&raw).map_err(ResolveError::Reference)?;
    parsed.path = normalise_path(&parsed.path).map_err(ResolveError::Path)?;
    Ok(parsed)
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let mut decoded = Vec::with_capacity(input.len());
    let mut origins = Vec::with_capacity(input.len());
    let b = input.as_bytes();
    let mut i = 0;
    while i < b.len() {
        let origin = i;
        let value = if b[i] == b'%' {
            if i.checked_add(2).map_or(true, |j| j >= b.len())
                || !b[i + 1].is_ascii_hexdigit()
                || !b[i + 2].is_ascii_hexdigit()
            {
                return Err(CodecError::BadEscape { at: i });
            }
            let h = |x: u8| -> u8 {
                match x {
                    b'0'..=b'9' => x - b'0',
                    b'a'..=b'f' => x - b'a' + 10,
                    b'A'..=b'F' => x - b'A' + 10,
                    _ => 0,
                }
            };
            let v = h(b[i + 1]) * 16 + h(b[i + 2]);
            i += 3;
            v
        } else {
            i += 1;
            b[origin]
        };
        decoded.push(value);
        origins.push(origin);
    }
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::new();
    for &b in text.as_bytes() {
        let keep = b.is_ascii_alphanumeric()
            || b"-._~".contains(&b)
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => b"/?".contains(&b),
                Component::Fragment => b"/?#".contains(&b),
            };
        if keep {
            out.push(b as char);
        } else {
            let digits = b"0123456789ABCDEF";
            out.push('%');
            out.push(digits[(b >> 4) as usize] as char);
            out.push(digits[(b & 15) as usize] as char);
        }
    }
    out
}

fn codec_to_parse(e: CodecError, offset: usize) -> ParseError {
    match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn parse_port(s: &str, offset: usize) -> Result<u16, ParseError> {
    if s.is_empty() {
        return Err(ParseError::InvalidPort { at: offset });
    }
    if let Some(i) = s.bytes().position(|x| !x.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: offset + i });
    }
    if s.len() > 1 && s.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: offset });
    }
    let mut n: u32 = 0;
    for digit in s.bytes() {
        n = n.saturating_mul(10).saturating_add(u32::from(digit - b'0'));
        if n > 65535 {
            return Err(ParseError::InvalidPort { at: offset });
        }
    }
    if n == 0 {
        Err(ParseError::InvalidPort { at: offset })
    } else {
        Ok(n as u16)
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Some(i) = input
        .bytes()
        .position(|b| !(0x21..=0x7e).contains(&b))
    {
        return Err(ParseError::InvalidByte { at: i });
    }
    let b = input.as_bytes();
    let colon = match b.iter().position(|&x| x == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !b[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        if !(b[i].is_ascii_alphanumeric() || b"+-.".contains(&b[i])) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();
    if b.get(colon + 1) != Some(&b'/') || b.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let start = colon + 3;
    let end = input[start..]
        .bytes()
        .position(|x| b"/?#".contains(&x))
        .map(|i| start + i)
        .unwrap_or(input.len());
    let auth = &input[start..end];
    let ats: Vec<usize> = auth
        .bytes()
        .enumerate()
        .filter_map(|(i, x)| (x == b'@').then_some(i))
        .collect();
    if ats.len() > 1 {
        return Err(ParseError::InvalidUserInfo { at: start + ats[1] });
    }
    let (userinfo, hrel) = match ats.first() {
        Some(&i) if i == 0 => return Err(ParseError::InvalidUserInfo { at: start }),
        Some(&i) => (Some(auth[..i].to_owned()), i + 1),
        None => (None, 0),
    };
    let hp = &auth[hrel..];
    let hstart = start + hrel;

    let (host, port, ipv6) = if hp.starts_with('[') {
        let close = match hp.find(']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: hstart }),
        };
        let inner = &hp[1..close];
        if inner.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: hstart });
        }
        let rem = &hp[close + 1..];
        let port = if rem.is_empty() {
            None
        } else if rem.starts_with(':') {
            Some(parse_port(&rem[1..], hstart + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hstart + close + 1,
            });
        };
        (inner.to_ascii_lowercase(), port, true)
    } else {
        let sep = hp.find(':');
        let (name, port) = match sep {
            Some(i) => (&hp[..i], Some((&hp[i + 1..], hstart + i + 1))),
            None => (hp, None),
        };
        if name.is_empty() {
            return Err(ParseError::InvalidHost { at: hstart });
        }
        for (i, x) in name.bytes().enumerate() {
            if !(x.is_ascii_alphanumeric() || b".-_".contains(&x)) {
                return Err(ParseError::InvalidHost { at: hstart + i });
            }
        }
        let port = match port {
            Some((s, at)) => Some(parse_port(s, at)?),
            None => None,
        };
        (name.to_ascii_lowercase(), port, false)
    };

    let tail = &input[end..];
    let fp = tail.find('#');
    let prefrag = fp.map(|i| &tail[..i]).unwrap_or(tail);
    let fragment = fp.map(|i| tail[i + 1..].to_owned());
    let qp = prefrag.find('?');
    let pathpart = qp.map(|i| &prefrag[..i]).unwrap_or(prefrag);
    let query = qp.map(|i| prefrag[i + 1..].to_owned());
    let path = if pathpart.is_empty() { "/" } else { pathpart }.to_owned();

    if let Some(ref x) = userinfo {
        decode_component(Component::UserInfo, x)
            .map_err(|e| codec_to_parse(e, start))?;
    }
    if !pathpart.is_empty() {
        decode_component(Component::Path, pathpart)
            .map_err(|e| codec_to_parse(e, end))?;
    }
    if let Some(ref x) = query {
        decode_component(Component::Query, x)
            .map_err(|e| codec_to_parse(e, end + qp.unwrap() + 1))?;
    }
    if let Some(ref x) = fragment {
        decode_component(Component::Fragment, x)
            .map_err(|e| codec_to_parse(e, end + fp.unwrap() + 1))?;
    }
    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;
    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut nav_at_end = false;
    for (i, segment) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 },
        })?;
        match decoded.as_str() {
            "." => nav_at_end = i + 1 == segments.len(),
            ".." => {
                if !stack.is_empty() {
                    stack.pop();
                }
                nav_at_end = i + 1 == segments.len();
            }
            _ => {
                stack.push(segment);
                nav_at_end = false;
            }
        }
    }
    if nav_at_end {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

fn serialized_authority(url: &Url) -> String {

    if let Some(ref info) = url.userinfo {
        result.push_str(info);
        result.push('@');
    }
    if url.ipv6 {
        result.push('[');

    } else {

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let mut base_url = parse_url(base).map_err(ResolveError::Base)?;
    base_url.path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    if let Some(i) = reference
        .bytes()
        .position(|b| !(0x21..=0x7e).contains(&b))
    {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at: i }));
    }

    let boundary = reference
        .bytes()
        .position(|b| b"/?#".contains(&b))
        .unwrap_or(reference.len());
    if reference[..boundary].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let hash = reference.find('#');
    let without_fragment = hash.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = hash.map(|i| reference[i + 1..].to_owned());
    let question = without_fragment.find('?');
    let ref_path = question
        .map(|i| &without_fragment[..i])
        .unwrap_or(without_fragment);
    let ref_query = question.map(|i| without_fragment[i + 1..].to_owned());

    let query = if ref_path.is_empty() && question.is_none() {
        base_url.query.clone()
    } else {
        ref_query
    };
    let merged_path = if ref_path.is_empty() {
        base_url.path.clone()
    } else if ref_path.starts_with('/') {
        ref_path.to_owned()
    } else {
        let slash = base_url.path.rfind('/').unwrap_or(0);
        format!("{}{}", &base_url.path[..=slash], ref_path)
    };

    let mut absolute = format!("{}://{}", base_url.scheme, serialized_authority(&base_url));
    absolute.push_str(&merged_path);
    if let Some(ref q) = query {
        absolute.push('?');
        absolute.push_str(q);
    }
    if let Some(ref f) = fragment {
        absolute.push('#');
        absolute.push_str(f);
    }
    let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

                return Err(CodecError::BadEscape { at: i });
            }
            let a = hex(bs[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let b = hex(bs[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            out.push((a << 4) | b);
            source.push(i);
            i += 3;
        } else {
            out.push(bs[i]);
            source.push(i);
            i += 1;
        }
    }
    let s = std::str::from_utf8(&out).map_err(|e| CodecError::InvalidUtf8 {
        at: source.get(e.valid_up_to()).copied().unwrap_or(input.len()),
    })?;
    Ok(s.to_owned())
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut result = String::with_capacity(text.len());
    for b in text.bytes() {
        let allowed = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => b == b'/' || b == b'?',
                Component::Fragment => b == b'/' || b == b'?' || b == b'#',
            };
        if allowed {
            result.push(b as char);
        } else {
            const H: &[u8] = b"0123456789ABCDEF";
            result.push('%');
            result.push(H[(b >> 4) as usize] as char);
            result.push(H[(b & 15) as usize] as char);
        }
    }
    result
}

fn map_codec_parse(e: CodecError, offset: usize) -> ParseError {
    match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (at, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    let b = input.as_bytes();
    let scheme_end = b.iter().position(|x| *x == b':').ok_or(ParseError::InvalidScheme { at: b.len() })?;
    if scheme_end == 0 || !b[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..scheme_end {
        if !(b[i].is_ascii_alphanumeric() || matches!(b[i], b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if b.get(scheme_end + 1) != Some(&b'/') || b.get(scheme_end + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: scheme_end + 1 });
    }
    let auth_start = scheme_end + 3;
    let auth_end = input[auth_start..]
        .bytes()
        .position(|x| matches!(x, b'/' | b'?' | b'#'))
        .map_or(input.len(), |i| auth_start + i);
    let authority = &input[auth_start..auth_end];
    let at_positions: Vec<usize> = authority.bytes().enumerate().filter_map(|(i, x)| (x == b'@').then_some(i)).collect();
    if at_positions.len() > 1 {
        return Err(ParseError::InvalidUserInfo { at: auth_start + at_positions[1] });
    }
    let (userinfo, remainder, rem_start) = match at_positions.first().copied() {
        Some(0) => return Err(ParseError::InvalidUserInfo { at: auth_start }),
        Some(i) => (Some(authority[..i].to_owned()), &authority[i + 1..], auth_start + i + 1),
        None => (None, authority, auth_start),
    };

    let (host, ipv6, port_str, port_offset) = if remainder.starts_with('[') {
        let close = match remainder.bytes().position(|x| x == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: rem_start }),
        };
        let interior = &remainder[1..close];
        if interior.contains('%') || Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: rem_start });
        }
        let suffix = &remainder[close + 1..];
        let (p, off) = if suffix.is_empty() {
            (None, rem_start + close + 1)
        } else if let Some(rest) = suffix.strip_prefix(':') {
            (Some(rest), rem_start + close + 2)
        } else {
            return Err(ParseError::InvalidHost { at: rem_start + close + 1 });
        };
        (interior.to_ascii_lowercase(), true, p, off)
    } else {
        let c = remainder.find(':');
        let (h, p, off) = c.map_or((remainder, None, rem_start + remainder.len()), |i| {
            (&remainder[..i], Some(&remainder[i + 1..]), rem_start + i + 1)
        });
        if h.is_empty() {
            return Err(ParseError::InvalidHost { at: rem_start });
        }
        for (i, x) in h.bytes().enumerate() {
            if !(x.is_ascii_alphanumeric() || matches!(x, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: rem_start + i });
            }
        }
        (h.to_ascii_lowercase(), false, p, off)
    };

    let port = if let Some(p) = port_str {
        if p.is_empty() {
            return Err(ParseError::InvalidPort { at: port_offset });
        }
        for (i, x) in p.bytes().enumerate() {
            if !x.is_ascii_digit() {
                return Err(ParseError::InvalidPort { at: port_offset + i });
            }
        }
        if p.len() > 1 && p.as_bytes()[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_offset });
        }
        let mut n: u32 = 0;
        for x in p.bytes() {
            n = match n.checked_mul(10).and_then(|v| v.checked_add((x - b'0') as u32)) {
                Some(v) => v,
                None => return Err(ParseError::InvalidPort { at: port_offset }),
            };
        }
        if n == 0 || n > 65535 {
            return Err(ParseError::InvalidPort { at: port_offset });
        }
        Some(n as u16)
    } else {
        None
    };

    let tail = &input[auth_end..];
    let (prefrag, fragment, frag_off) = tail.find('#').map_or((tail, None, input.len()), |i| {
        (&tail[..i], Some(&tail[i + 1..]), auth_end + i + 1)
    });
    let (raw_path, query, query_off) = prefrag.find('?').map_or((prefrag, None, input.len()), |i| {
        (&prefrag[..i], Some(&prefrag[i + 1..]), auth_end + i + 1)
    });
    if let Some(ref info) = userinfo {
        decode_component(Component::UserInfo, info).map_err(|e| map_codec_parse(e, auth_start))?;
    }
    if !raw_path.is_empty() {
        decode_component(Component::Path, raw_path).map_err(|e| map_codec_parse(e, auth_end))?;
    }
    if let Some(ref q) = query {
        decode_component(Component::Query, q).map_err(|e| map_codec_parse(e, query_off))?;
    }
    if let Some(ref f) = fragment {
        decode_component(Component::Fragment, f).map_err(|e| map_codec_parse(e, frag_off))?;
    }
    Ok(Url {
        scheme: input[..scheme_end].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path: if raw_path.is_empty() { "/".to_owned() } else { raw_path.to_owned() },
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    })
}

fn map_codec_path(e: CodecError) -> PathError {
    match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;
    let mut stack: Vec<&str> = Vec::new();
    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut final_nav = false;
    for (i, raw) in segments.iter().enumerate() {
        let dec = decode_component(Component::Path, raw).map_err(map_codec_path)?;
        let dot = dec == ".";
        let dotdot = dec == "..";
        if i + 1 == segments.len() {
            final_nav = dot || dotdot;
        }
        if dot {
            continue;
        } else if dotdot {
            stack.pop();
        } else {
            stack.push(*raw);
        }
    }
    if final_nav {
        stack.push("");
    }
    if stack.is_empty() {
        return Ok("/".to_owned());
    }
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    Ok(result)
}

fn normalize_result_path(mut u: Url) -> Result<Url, ResolveError> {
    u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
    Ok(u)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    for (i, x) in reference.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&x) {
            return Err(ResolveError::Reference(ParseError::InvalidByte { at: i }));
        }
    }

    let bs = reference.as_bytes();
    let bound = bs.iter().position(|x| matches!(x, b'/' | b'?' | b'#')).unwrap_or(bs.len());
    if bs[..bound].contains(&b':') {
        return normalize_result_path(parse_url(reference).map_err(ResolveError::Reference)?);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        return normalize_result_path(parse_url(&absolute).map_err(ResolveError::Reference)?);
    }

    let (before_hash, frag) = reference.find('#').map_or((reference, None), |i| (&reference[..i], Some(&reference[i + 1..])));
    let (ref_path, ref_query) = before_hash.find('?').map_or((before_hash, None), |i| (&before_hash[..i], Some(&before_hash[i + 1..])));
    let query = if ref_path.is_empty() { ref_query.or(base_url.query.as_deref()) } else { ref_query };
    let path = if ref_path.is_empty() {
        base_path
    } else if ref_path.starts_with('/') {
        ref_path.to_owned()
    } else {
        let prefix = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
        format!("{prefix}{ref_path}")
    };

    let mut raw = format!("{}://", base_url.scheme);
    if let Some(info) = base_url.userinfo.as_deref() {
        raw.push_str(info);
        raw.push('@');
    }
    if base_url.ipv6 {
        raw.push('[');
        raw.push_str(&base_url.host);
        raw.push(']');
    } else {
        raw.push_str(&base_url.host);
    }
    if let Some(p) = base_url.port {
        raw.push(':');
        raw.push_str(&p.to_string());
    }
    raw.push_str(&path);
    if let Some(q) = query {
        raw.push('?');
        raw.push_str(q);
    }
    if let Some(f) = frag {
        raw.push('#');
        raw.push_str(f);
    }
    normalize_result_path(parse_url(&raw).map_err(ResolveError::Reference)?)
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let lo = hex_value(bytes[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {

        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),

        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_ascii(input)?;
    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }
    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| auth_start + i)
        .unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];

        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + ats[1],
        });
    }
    let (userinfo, hostport, hp_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            auth_start + at + 1,
        )
    } else {
        (None, authority, auth_start)
    };

    let (host, ipv6, port_text, port_at) = if hostport.starts_with('[') {
        let open = hp_start;
        let close = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let inside = &hostport[1..close];
        if inside.contains('%') || Ipv6Addr::from_str(inside).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        let after = &hostport[close + 1..];
        let (pt, pa) = if after.is_empty() {
            (None, hp_start + close + 1)
        } else if after.starts_with(':') {
            (Some(&after[1..]), hp_start + close + 2)
        } else {
            return Err(ParseError::InvalidHost {
                at: hp_start + close + 1,
            });
        };
        (inside.to_ascii_lowercase(), true, pt, pa)
    } else {
        let split = hostport.find(':');
        let (h, pt, pa) = if let Some(i) = split {
            (&hostport[..i], Some(&hostport[i + 1..]), hp_start + i + 1)
        } else {
            (hostport, None, hp_start + hostport.len())
        };
        if h.is_empty() {
            return Err(ParseError::InvalidHost { at: hp_start });
        }
        for (i, b) in h.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: hp_start + i });
            }
        }
        (h.to_ascii_lowercase(), false, pt, pa)

                Some(v) => v,
                None => return Err(ParseError::InvalidPort { at: port_at }),
            };

        if !(1..=65535).contains(&n) {
            return Err(ParseError::InvalidPort { at: port_at });
        }
        Some(n as u16)

        None
    };

    let (before_fragment, fragment, fragment_at) = if let Some(i) = tail.find('#') {
        (&tail[..i], Some(&tail[i + 1..]), auth_end + i + 1)
    } else {
        (tail, None, input.len())
    };
    let (raw_path, query, query_at) = if let Some(i) = before_fragment.find('?') {
        (
            &before_fragment[..i],
            Some(&before_fragment[i + 1..]),
            auth_end + i + 1,
        )
    } else {
        (before_fragment, None, input.len())
    };

        decode_component(Component::UserInfo, u)
            .map_err(|e| rebase_codec(e, auth_start))?;

    if !raw_path.is_empty() {
        decode_component(Component::Path, raw_path).map_err(|e| rebase_codec(e, path_at))?;
    }

        decode_component(Component::Query, q).map_err(|e| rebase_codec(e, query_at))?;
    }
    if let Some(f) = fragment {
        decode_component(Component::Fragment, f).map_err(|e| rebase_codec(e, fragment_at))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path: path.to_owned(),
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),

fn path_error(err: CodecError) -> PathError {
    match err {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(path_error)?;
    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;
    for (i, seg) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, seg).map_err(path_error)?;
        let nav = decoded == "." || decoded == "..";
        if i + 1 == raw_segments.len() {
            terminal_navigation = nav;
        }
        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(*seg);
        }
    }
    if terminal_navigation {
        stack.push("");
    }
    if stack.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", stack.join("/")))
    }
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {

        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());
    if bytes[..boundary].contains(&b':') {
        let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
        u.path = path_to_resolve(&u.path)?;
        return Ok(u);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&absolute).map_err(ResolveError::Reference)?;
        u.path = path_to_resolve(&u.path)?;
        return Ok(u);
    }

    let (pre_fragment, fragment) = if let Some(i) = reference.find('#') {
        (&reference[..i], Some(&reference[i + 1..]))
    } else {
        (reference, None)
    };
    let (ref_path, ref_query) = if let Some(i) = pre_fragment.find('?') {
        (&pre_fragment[..i], Some(&pre_fragment[i + 1..]))
    } else {
        (pre_fragment, None)
    };

    let selected_query = if ref_path.is_empty() {
        ref_query.or(base_url.query.as_deref())
    } else {
        ref_query
    };
    let selected_path = if ref_path.is_empty() {
        base_path.clone()
    } else if ref_path.starts_with('/') {
        ref_path.to_owned()
    } else {
        let slash = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
        format!("{slash}{ref_path}")
    };

    let mut constructed = String::new();
    constructed.push_str(&base_url.scheme);
    constructed.push_str("://");
    if let Some(info) = base_url.userinfo.as_deref() {
        constructed.push_str(info);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&selected_path);
    append_optional(&mut constructed, '?', selected_query);
    append_optional(&mut constructed, '#', fragment);

    let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
    u.path = path_to_resolve(&u.path)?;
    Ok(u)
}
