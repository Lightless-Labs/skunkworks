```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map_or(bytes.len(), |offset| authority_start + offset);
    let authority = &bytes[authority_start..authority_end];

    let mut at_sign = None;
    for (offset, &byte) in authority.iter().enumerate() {
        if byte == b'@' {
            if at_sign.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: authority_start + offset,
                });
            }
            at_sign = Some(offset);
        }
    }

    let (userinfo, hostport_start) = match at_sign {
        Some(offset) if offset == 0 => {
            return Err(ParseError::InvalidUserInfo { at: authority_start });
        }
        Some(offset) => (
            Some(input[authority_start..authority_start + offset].to_owned()),
            authority_start + offset + 1,
        ),
        None => (None, authority_start),
    };

    let hostport = &bytes[hostport_start..authority_end];
    let host_colon = hostport.iter().position(|&b| b == b':');
    let host_end = host_colon.map_or(authority_end, |offset| hostport_start + offset);

    if host_end == hostport_start {
        return Err(ParseError::InvalidHost { at: hostport_start });
    }
    for (offset, &byte) in bytes[hostport_start..host_end].iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: hostport_start + offset,
            });
        }
    }

    let port = match host_colon {
        None => None,
        Some(offset) => {
            let port_start = hostport_start + offset + 1;
            if port_start == authority_end {
                return Err(ParseError::InvalidPort { at: port_start });
            }

            let port_bytes = &bytes[port_start..authority_end];
            if let Some(offset) = port_bytes.iter().position(|b| !b.is_ascii_digit()) {
                return Err(ParseError::InvalidPort {
                    at: port_start + offset,
                });
            }
            if port_bytes.len() >= 2 && port_bytes[0] == b'0' {
                return Err(ParseError::InvalidPort { at: port_start });
            }

            let mut value: u32 = 0;
            for &digit in port_bytes {
                value = match value
                    .checked_mul(10)
                    .and_then(|n| n.checked_add(u32::from(digit - b'0')))
                {
                    Some(n) => n,
                    None => return Err(ParseError::InvalidPort { at: port_start }),
                };
            }
            if !(1..=65535).contains(&value) {
                return Err(ParseError::InvalidPort { at: port_start });
            }
            Some(value as u16)
        }
    };

    let fragment_start = bytes[authority_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|offset| authority_end + offset);

    let before_fragment_end = fragment_start.unwrap_or(bytes.len());
    let query_start = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|offset| authority_end + offset);

    let path_end = query_start.unwrap_or(before_fragment_end);
    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };
    let query = query_start.map(|start| {
        input[start + 1..before_fragment_end].to_owned()
    });
    let fragment = fragment_start.map(|start| input[start + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: input[hostport_start..host_end].to_ascii_lowercase(),
        port,
        path,
        query,
        fragment,
    })
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, b) in bytes.iter().copied().enumerate() {
        if b < 0x21 || b > 0x7e {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let Some(colon) = bytes.iter().position(|&b| b == b':') else {
        return Err(ParseError::InvalidScheme { at: bytes.len() });
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    if let Some(offset) = bytes[1..colon]
        .iter()
        .position(|b| !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')))
    {
        return Err(ParseError::InvalidScheme { at: offset + 1 });
    }

    let marker_ok = colon + 2 < bytes.len()
        && bytes[colon + 1] == b'/'
        && bytes[colon + 2] == b'/';
    if !marker_ok {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| authority_start + n)
        .unwrap_or(bytes.len());

    let mut first_at = None;
    for i in authority_start..authority_end {
        if bytes[i] == b'@' {
            if first_at.is_some() {
                return Err(ParseError::InvalidUserInfo { at: i });
            }
            first_at = Some(i);
        }
    }

    let (userinfo, host_start) = match first_at {
        Some(i) if i == authority_start => {
            return Err(ParseError::InvalidUserInfo { at: i });
        }
        Some(i) => (Some(input[authority_start..i].to_owned()), i + 1),
        None => (None, authority_start),
    };

    let host_colon = bytes[host_start..authority_end]
        .iter()
        .position(|&b| b == b':')
        .map(|n| host_start + n);
    let host_end = host_colon.unwrap_or(authority_end);
    if host_start == host_end {
        return Err(ParseError::InvalidHost { at: host_start });
    }
    for i in host_start..host_end {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost { at: i });
        }
    }

    let port = if let Some(separator) = host_colon {
        let start = separator + 1;
        if start == authority_end {
            return Err(ParseError::InvalidPort { at: start });
        }
        let digits = &bytes[start..authority_end];
        if let Some(offset) = digits.iter().position(|b| !b.is_ascii_digit()) {
            return Err(ParseError::InvalidPort { at: start + offset });
        }
        if digits.len() > 1 && digits[0] == b'0' {
            return Err(ParseError::InvalidPort { at: start });
        }

        let mut value = 0u32;
        for digit in digits {
            value = value
                .saturating_mul(10)
                .saturating_add(u32::from(*digit - b'0'));
            if value > 65535 {
                return Err(ParseError::InvalidPort { at: start });
            }
        }
        if value == 0 {
            return Err(ParseError::InvalidPort { at: start });
        }
        Some(value as u16)
    } else {
        None
    };

    let fragment = bytes[authority_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|n| authority_end + n);
    let component_end = fragment.unwrap_or(bytes.len());
    let query = bytes[authority_end..component_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|n| authority_end + n);
    let path_end = query.unwrap_or(component_end);

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: input[host_start..host_end].to_ascii_lowercase(),
        port,
        path: if path_end == authority_end {
            "/".to_owned()
        } else {
            input[authority_end..path_end].to_owned()
        },
        query: query.map(|i| input[i + 1..component_end].to_owned()),
        fragment: fragment.map(|i| input[i + 1..].to_owned()),
    })
}
```
