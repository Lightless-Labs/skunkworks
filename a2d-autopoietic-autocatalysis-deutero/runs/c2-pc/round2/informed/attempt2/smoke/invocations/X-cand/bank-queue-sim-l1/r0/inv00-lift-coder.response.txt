```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut ids = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut events = Vec::with_capacity(customers.len().saturating_mul(3));
    let mut waiting = VecDeque::new();
    let mut next_arrival = 0usize;
    let mut active: Option<(usize, u64)> = None;
    let mut previous_finish: Option<u64> = None;

    while next_arrival < order.len() || active.is_some() || !waiting.is_empty() {
        let arrival_time = order
            .get(next_arrival)
            .map(|&index| customers[index].arrival);
        let finish_time = active.map(|(_, finish)| finish);
        let time = match (arrival_time, finish_time) {
            (Some(a), Some(f)) => a.min(f),
            (Some(a), None) => a,
            (None, Some(f)) => f,
            (None, None) => break,
        };

        if let Some((index, finish)) = active {
            if finish == time {
                events.push(Event {
                    time,
                    customer: customers[index].id,
                    kind: Kind::Finish,
                });
                active = None;
                previous_finish = Some(time);
            }
        }

        while let Some(&index) = order.get(next_arrival) {
            if customers[index].arrival != time {
                break;
            }
            events.push(Event {
                time,
                customer: customers[index].id,
                kind: Kind::Arrive,
            });
            waiting.push_back(index);
            next_arrival += 1;
        }

        if active.is_none() {
            if let Some(index) = waiting.pop_front() {
                let customer = &customers[index];
                let wake = match previous_finish {
                    None => true,
                    Some(finish) => time > finish,
                };
                let base_finish = time.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    },
                )?;
                let finish = base_finish
                    .checked_add(if wake { wake_cost } else { 0 })
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        time,
                    })?;
                events.push(Event {
                    time,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });

                if finish == time {
                    events.push(Event {
                        time,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    previous_finish = Some(time);
                } else {
                    active = Some((index, finish));
                }
            }
        }
    }

    Ok(events)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        seen.insert(customer.id);
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by_key(|&i| (customers[i].arrival, std::cmp::Reverse(customers[i].id)));

    let mut waiting = VecDeque::new();
    let mut events = Vec::new();
    let mut arrival_cursor = 0;
    let mut busy_until = None;
    let mut finishing_customer = None;
    let mut prior_finish = None;

    while arrival_cursor < arrivals.len() || !waiting.is_empty() || busy_until.is_some() {
        let next_arrival = arrivals
            .get(arrival_cursor)
            .map(|&i| customers[i].arrival);
        let time = match (next_arrival, busy_until) {
            (Some(a), Some(f)) => a.min(f),
            (Some(a), None) => a,
            (None, Some(f)) => f,
            (None, None) => break,
        };

        if busy_until == Some(time) {
            let id = finishing_customer.take().expect("busy customer");
            events.push(Event {
                time,
                customer: id,
                kind: Kind::Finish,
            });
            busy_until = None;
            prior_finish = Some(time);
        }

        while arrival_cursor < arrivals.len()
            && customers[arrivals[arrival_cursor]].arrival == time
        {
            let i = arrivals[arrival_cursor];
            events.push(Event {
                time,
                customer: customers[i].id,
                kind: Kind::Arrive,
            });
            waiting.push_back(i);
            arrival_cursor += 1;
        }

        if busy_until.is_none() {
            if let Some(i) = waiting.pop_front() {
                let c = &customers[i];
                let wake = prior_finish.map_or(true, |f| time > f);
                let finish = time
                    .checked_add(c.service)
                    .and_then(|x| x.checked_add(if wake { wake_cost } else { 0 }))
                    .ok_or(Error::TimeOverflow {
                        customer: c.id,
                        time,
                    })?;
                events.push(Event {
                    time,
                    customer: c.id,
                    kind: Kind::Start { finish, wake },
                });
                if finish == time {
                    events.push(Event {
                        time,
                        customer: c.id,
                        kind: Kind::Finish,
                    });
                    prior_finish = Some(time);
                } else {
                    busy_until = Some(finish);
                    finishing_customer = Some(c.id);
                }
            }
        }
    }

    Ok(events)
}
```
