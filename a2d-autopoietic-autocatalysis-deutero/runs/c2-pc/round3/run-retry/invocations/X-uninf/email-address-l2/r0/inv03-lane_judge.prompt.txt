pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_close = if bytes.first() == Some(&b'"') {
        let mut i = 1;
        let mut close = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match close {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !matches!(quoted_close, Some(close) if i <= close) {
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let mut local = String::new();
    let is_quoted = quoted_close.is_some();

    if let Some(close) = quoted_close {
        if close + 1 != separator {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: i });
                }
                let escaped = bytes[i + 1];
                match escaped {
                    b'"' | b'\\' | b'@' => {
                        local.push(escaped as char);
                        origins.push(i);
                        i += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: i }),
                }
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                local.push(b as char);
                origins.push(i);
                i += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        if separator > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for i in 0..separator {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == separator || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: i });
            }
            local.push(b as char);
        }
    }

    let raw_domain_start = separator + 1;
    let mut domain_end = bytes.len();
    let had_root_dot = domain_end > raw_domain_start && bytes[domain_end - 1] == b'.';
    if had_root_dot {
        domain_end -= 1;
    }

    if domain_end == raw_domain_start {
        return Err(EmailError::EmptyDomain {
            at: raw_domain_start,
        });
    }

    if domain_end - raw_domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: raw_domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut start = raw_domain_start;

    for i in raw_domain_start..domain_end {
        if bytes[i] == b'.' {
            if i == start {
                return Err(EmailError::EmptyLabel { at: i });
            }
            label_starts.push(start);
            label_ends.push(i);
            start = i + 1;
        }
    }

    if start == domain_end {
        let at = if had_root_dot { domain_end } else { domain_end };
        return Err(EmailError::EmptyLabel { at });
    }
    label_starts.push(start);
    label_ends.push(domain_end);

    for (&label_start, &label_end) in label_starts.iter().zip(&label_ends) {
        let len = label_end - label_start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: label_start + 63,
            });
        }

        if bytes[label_start] == b'-' {
            return Err(EmailError::LabelHyphen { at: label_start });
        }
        if bytes[label_end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: label_end - 1 });
        }

        for i in label_start..label_end {
            let b = bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: i });
            }
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: raw_domain_start,
        });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = String::from_utf8(
        bytes[raw_domain_start..domain_end]
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect(),
    )
    .unwrap();

    Ok(Address {
        local,
        domain,
        quoted: is_quoted,
    })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    let len = bytes.len();

    if len == 0 {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_prefix_end = if bytes[0] == b'"' {
        let mut i = 1;
        let mut closing = None;
        while i < len {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    closing = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match closing {
            Some(at) => Some(at),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    for i in 0..len {
        if bytes[i] == b'@' {
            if quoted_prefix_end.is_some_and(|end| i <= end) {
                continue;
            }
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let (local, quoted) = if bytes[0] == b'"' {
        let closing = quoted_prefix_end.expect("quoted prefix was closed");
        if closing + 1 != separator {
            return Err(EmailError::TrailingLocal { at: closing + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < closing {
            if bytes[i] == b'\\' {
                if i + 1 >= closing
                    || !matches!(bytes[i + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape { at: i });
                }
                decoded.push(bytes[i + 1]);
                origins.push(i);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&bytes[i]) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                decoded.push(bytes[i]);
                origins.push(i);
                i += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: closing });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
        (String::from_utf8(decoded).expect("ASCII bytes"), true)
    } else {
        if separator > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for i in 0..separator {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == separator || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: i });
            }
        }

        (
            String::from_utf8(bytes[..separator].to_vec()).expect("ASCII bytes"),
            false,
        )
    };

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);
    let root_dot_at = if has_root_dot { Some(len - 1) } else { None };

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain_bytes = &raw_domain[..domain_len];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0;

    for i in 0..=domain_len {
        if i == domain_len || domain_bytes[i] == b'.' {
            if i == label_start {
                let at = if i == domain_len {
                    root_dot_at.unwrap_or(domain_start + i)
                } else {
                    domain_start + i
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }

    for &(start, end) in &labels {
        let label_len = end - start;
        if label_len > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_start + start + 63,
            });
        }
        if domain_bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + start,
            });
        }
        if domain_bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + end - 1,
            });
        }
        for i in start..end {
            let b = domain_bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: domain_start + i,
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let &(last_start, last_end) = labels.last().expect("domain has labels");
    if domain_bytes[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let domain = domain_bytes
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}

                }
                decoded.push(b);
                origins.push(j);
                j += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }
        let mut previous_dot = false;
        for (idx, &b) in bytes[..sep].iter().enumerate() {
            if b == b'.' {
                if idx == 0 || idx + 1 == sep || previous_dot {
                    return Err(EmailError::LocalDot { at: idx });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                let allowed = b.is_ascii_alphanumeric()
                    || b"!#$%&'*+-/=?^_`{|}~".contains(&b);
                if !allowed {
                    return Err(EmailError::LocalChar { at: idx });
                }
            }
        }
        (input[..sep].to_owned(), false)
    };

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let mut removed_dot = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_dot = Some(domain_end);
    }

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for pos in domain_start..domain_end {
        if bytes[pos] == b'.' {
            labels.push((label_start, pos));
            label_start = pos + 1;
        }
    }
    labels.push((label_start, domain_end));

    for &(start, end) in &labels {
        if start == end {
            let at = if start == domain_end {
                removed_dot.unwrap_or(domain_end)
            } else {
                start
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if end - start > 63 {
            return Err(EmailError::LabelTooLong { at: start + 63 });
        }

        if bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: start });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: end - 1 });
        }

        for pos in start..end {
            let b = bytes[pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar { at: pos });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end]
        .iter()
        .all(|b| b.is_ascii_digit())
    {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = input[domain_start..domain_end].to_ascii_lowercase();
    Ok(Address {
        local,
        domain,
        quoted,
    })
}
