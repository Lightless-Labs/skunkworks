#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_close = if bytes.first() == Some(&b'"') {
        let mut i = 1;
        let mut close = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match close {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !matches!(quoted_close, Some(close) if i <= close) {
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let mut local = String::new();
    let is_quoted = quoted_close.is_some();

    if let Some(close) = quoted_close {
        if close + 1 != separator {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: i });
                }
                let escaped = bytes[i + 1];
                match escaped {
                    b'"' | b'\\' | b'@' => {
                        local.push(escaped as char);
                        origins.push(i);
                        i += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: i }),
                }
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                local.push(b as char);
                origins.push(i);
                i += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        if separator > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for i in 0..separator {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == separator || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: i });
            }
            local.push(b as char);
        }
    }

    let raw_domain_start = separator + 1;
    let mut domain_end = bytes.len();
    let had_root_dot = domain_end > raw_domain_start && bytes[domain_end - 1] == b'.';
    if had_root_dot {
        domain_end -= 1;
    }

    if domain_end == raw_domain_start {
        return Err(EmailError::EmptyDomain {
            at: raw_domain_start,
        });
    }

    if domain_end - raw_domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: raw_domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut start = raw_domain_start;

    for i in raw_domain_start..domain_end {
        if bytes[i] == b'.' {
            if i == start {
                return Err(EmailError::EmptyLabel { at: i });
            }
            label_starts.push(start);
            label_ends.push(i);
            start = i + 1;
        }
    }

    if start == domain_end {
        let at = if had_root_dot { domain_end } else { domain_end };
        return Err(EmailError::EmptyLabel { at });
    }
    label_starts.push(start);
    label_ends.push(domain_end);

    for (&label_start, &label_end) in label_starts.iter().zip(&label_ends) {
        let len = label_end - label_start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: label_start + 63,
            });
        }

        if bytes[label_start] == b'-' {
            return Err(EmailError::LabelHyphen { at: label_start });
        }
        if bytes[label_end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: label_end - 1 });
        }

        for i in label_start..label_end {
            let b = bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: i });
            }
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: raw_domain_start,
        });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = String::from_utf8(
        bytes[raw_domain_start..domain_end]
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect(),
    )
    .unwrap();

    Ok(Address {
        local,
        domain,
        quoted: is_quoted,
    })
}
