#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i));

    let separator = match separators.next() {
        None => return Err(EmailError::MissingAt),
        Some(first) => {
            if let Some(second) = separators.next() {
                return Err(EmailError::MultipleAt { at: second });
            }
            first
        }
    };

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];

    if local_bytes.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (i, &b) in local_bytes.iter().enumerate() {
        let at = i;
        if b == b'.' {
            if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at });
            }
        } else if !b.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            return Err(EmailError::LocalChar { at });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_bytes = if has_root_dot {
        &raw_domain[..raw_domain.len() - 1]
    } else {
        raw_domain
    };
    let root_dot_at = has_root_dot.then_some(bytes.len() - 1);

    if domain_bytes.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut label_start = 0usize;

    for (i, &b) in domain_bytes.iter().enumerate() {
        if b == b'.' {
            if i == label_start {
                return Err(EmailError::EmptyLabel {
                    at: domain_start + i,
                });
            }
            labels.push((label_start, &domain_bytes[label_start..i]));
            label_start = i + 1;
        }
    }

    if label_start == domain_bytes.len() {
        return Err(EmailError::EmptyLabel {
            at: root_dot_at.unwrap_or(bytes.len()),
        });
    }
    labels.push((label_start, &domain_bytes[label_start..]));

    for &(offset, label) in &labels {
        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_start + offset + 63,
            });
        }

        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + offset,
            });
        }

        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + offset + label.len() - 1,
            });
        }

        for (i, &b) in label.iter().enumerate() {
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: domain_start + offset + i,
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_offset, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_offset,
        });
    }

    let local = String::from_utf8(local_bytes.to_vec()).unwrap();
    let domain = String::from_utf8(
        domain_bytes
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect(),
    )
    .unwrap();

    Ok(Address {
        local,
        domain,
        quoted: false,
    })
}
