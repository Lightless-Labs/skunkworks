use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Value {
    value: String,
    deadline: Option<u64>,
}

#[derive(Clone)]
enum Entry {
    Value(Value),
    Tombstone,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    entry: Entry,
}

struct Transaction {
    revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Entry>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Value>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn check_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn selected_entry(&self, key: &str, tx: Option<u64>) -> Option<Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(entry) = transaction.writes.get(key) {
                return Some(entry.clone());
            }
        }
        let versions = self.data.get(key)?;
        let limit = tx
            .and_then(|id| self.transactions.get(&id).map(|t| t.revision));
        versions
            .iter()
            .rev()
            .find(|v| limit.map_or(true, |r| v.revision <= r))
            .map(|v| v.entry.clone())
    }

    fn live_entry(entry: Option<Entry>, now: u64) -> Option<String> {
        match entry {
            Some(Entry::Value(v)) if v.deadline.map_or(true, |d| d > now) => {
                Some(v.value)
            }
            _ => None,
        }
    }

    fn visible(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, String> {
        let mut keys: BTreeSet<String> = self.data.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(t) = self.transactions.get(&id) {
                keys.extend(t.writes.keys().cloned());
            }
        }
        keys.into_iter()
            .filter_map(|k| {
                let value = Self::live_entry(self.selected_entry(&k, tx), now)?;
                Some((k, value))
            })
            .collect()
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let tx = self
            .transactions
            .get(&id)
            .ok_or(Error::UnknownTransaction(id))?;
        if !tx.writes.is_empty() {
            for key in tx.writes.keys() {
                if self.data.get(key).and_then(|v| v.last()).map_or(false, |v| {
                    v.revision > tx.revision
                }) {
                    return Err(Error::Conflict {
                        tx: id,
                        key: key.clone(),
                    });
                }
            }
            let writes = tx.writes.clone();
            self.revision += 1;
            let revision = self.revision;
            for (key, entry) in writes {
                self.data
                    .entry(key)
                    .or_default()
                    .push(Version { revision, entry });
            }
        }
        let revision = self.revision;
        self.transactions.remove(&id);
        Ok(revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let entry = match ttl {
            Some(0) => Entry::Tombstone,
            Some(d) => Entry::Value(Value {
                value: value.to_string(),
                deadline: Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            }),
            None => Entry::Value(Value {
                value: value.to_string(),
                deadline: None,
            }),
        };
        self.write(tx, key, entry);
        Ok(())
    }

    fn write(&mut self, tx: Option<u64>, key: &str, entry: Entry) {
        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_string(), entry);
        } else {
            self.revision += 1;
            self.data
                .entry(key.to_string())
                .or_default()
                .push(Version {
                    revision: self.revision,
                    entry,
                });
        }
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let existed = Self::live_entry(self.selected_entry(key, tx), now).is_some();
        self.write(tx, key, Entry::Tombstone);
        Ok(existed)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(Self::live_entry(self.selected_entry(key, tx), now))
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Ok(self
            .visible(tx, now)
            .values()
            .filter(|v| v.as_str() == value)
            .count())
    }

    fn validate_range(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.map_or(false, |e| e <= start) {
            Err(Error::InvalidRange)
        } else {
            Ok(())
        }
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_range(start, end)?;
        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        let visible = self.visible(tx, now);
        let keys: Vec<_> = visible
            .keys()
            .filter(|k| Self::matches(k, start, end, prefix))
            .filter(|k| cursor.map_or(true, |c| k.as_str() > c.last.as_str()))
            .cloned()
            .collect();

        let skip = if cursor.is_some() { 0 } else { offset };
        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }
        let items: Vec<(String, String)> = keys
            .iter()
            .skip(skip)
            .take(limit)
            .map(|k| (k.clone(), visible[k].clone()))
            .collect();

        let next = if items.len() > 0 && skip.saturating_add(items.len()) < keys.len() {
            Some(Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: items.last().unwrap().0.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_range(start, end)?;
        let visible = self.visible(tx, now);
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in visible {
            if !Self::matches(&key, start, end, prefix) {
                continue;
            }
            let parsed = parse_number(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum
                .checked_add(parsed)
                .ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        validate_name(name)?;
        let live: BTreeMap<String, Value> = self
            .data
            .iter()
            .filter_map(|(key, versions)| {
                let entry = versions.last()?.entry.clone();
                match entry {
                    Entry::Value(v) if v.deadline.map_or(true, |d| d > now) => {
                        Some((key.clone(), v))
                    }
                    _ => None,
                }
            })
            .collect();
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        self.snapshots.insert(name.to_string(), live);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let saved = self
            .snapshots
            .get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }
        let saved_live: BTreeMap<String, Value> = saved
            .into_iter()
            .filter(|(_, v)| v.deadline.map_or(true, |d| d > now))
            .collect();
        let current = self.visible(None, now);
        let keys: BTreeSet<String> = current.keys().cloned().chain(saved_live.keys().cloned()).collect();
        if keys.is_empty() {
            return Ok(self.revision);
        }
        self.revision += 1;
        let revision = self.revision;
        for key in keys {
            let entry = saved_live
                .get(&key)
                .cloned()
                .map(Entry::Value)
                .unwrap_or(Entry::Tombstone);
            self.data.entry(key).or_default().push(Version {
                revision,
                entry,
            });
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let begin_revisions: Vec<u64> = self.transactions.values().map(|t| t.revision).collect();
        let mut removed = 0;
        for versions in self.data.values_mut() {
            if versions.len() <= 1 {
                continue;
            }
            let mut keep = BTreeSet::new();
            keep.insert(versions.len() - 1);
            for rev in &begin_revisions {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= *rev) {
                    keep.insert(index);
                }
            }
            let mut i = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&i);
                i += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self
            .data
            .values()
            .flatten()
            .filter(|v| matches!(v.entry, Entry::Tombstone))
            .count();
        let live_keys = self.visible(None, now).len();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.values().filter(|v| !v.is_empty()).count(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self.transactions.values().filter(|t| t.read_only).count(),
            staged_writes: self.transactions.values().map(|t| t.writes.len()).sum(),
            snapshots: self.snapshots.len(),
        })
    }
}

fn validate_name(name: &str) -> Result<(), Error> {
    if name.is_empty() {
        return Err(Error::NameEmpty);
    }
    for (at, ch) in name.char_indices() {
        let valid = if at == 0 {
            ch.is_ascii_lowercase()
        } else {
            ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
        };
        if !valid {
            return Err(Error::NameInvalid { at });
        }
    }
    Ok(())
}

fn parse_number(value: &str) -> Option<i64> {
    if value == "0" {
        return Some(0);
    }
    let digits = value.strip_prefix('-').unwrap_or(value);
    if digits.is_empty()
        || !digits.as_bytes()[0].is_ascii_digit()
        || digits.as_bytes()[0] == b'0'
        || !digits.bytes().all(|b| b.is_ascii_digit())
        || value == "-0"
    {
        return None;
    }
    value.parse::<i64>().ok()
}
