#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },

impl Default for Policy {

        Self {

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize, usize)>, // cleaned insertion index, original start, original end
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn cleanup(bytes: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Vec::with_capacity(bytes.len());
    let mut positions = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut depth = 0usize;
    let mut opens = [0usize; 4];
    let mut quote = false;
    let mut first_retained = true;

    while i < bytes.len() {
        let b = bytes[i];

        if quote {
            out.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    out.push(bytes[i]);
                    positions.push(i);
                    i += 1;
                }
                continue;
            }
            if b == b'"' {
                quote = false;
            }
            i += 1;
            continue;
        }

        let run_start = i;
        let mut consumed_cfws = false;
        if depth == 0 && is_hws(b) {
            consumed_cfws = true;
            i += 1;
            while i < bytes.len() && is_hws(bytes[i]) {
                i += 1;
            }
        } else if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(bytes, i) {
                consumed_cfws = true;
                i = end;
            } else {
                return Err(EmailError::BadFold { at: i });
            }
        } else if depth == 0 && b == b'(' {
            consumed_cfws = true;
            opens[0] = i;
            depth = 1;
            i += 1;
            while i < bytes.len() && depth > 0 {
                let c = bytes[i];
                if c == b'\\' {
                    let slash = i;
                    if i + 1 >= bytes.len() || !(0x20..=0x7e).contains(&bytes[i + 1]) {
                        return Err(EmailError::CommentEscape { at: slash });
                    }
                    i += 2;
                } else if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if c == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    opens[depth] = i;
                    depth += 1;
                    i += 1;
                } else if c == b')' {
                    depth -= 1;
                    i += 1;
                } else if (0x20..=0x7e).contains(&c) || c == b'\t' {
                    i += 1;
                } else {
                    return Err(EmailError::CommentChar { at: i });
                }
            }
            if depth != 0 {
                return Err(EmailError::UnclosedComment { at: opens[0] });
            }
        } else if depth == 0 && b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        } else {
            if first_retained && b == b'"' {
                quote = true;
            }
            out.push(b);
            positions.push(i);
            first_retained = false;
            i += 1;
            continue;
        }

        if consumed_cfws {
            runs.push((out.len(), run_start, i));
        }
    }

    Ok(Cleaned {
        bytes: out,
        positions,
        runs,
    })
}

fn pos(clean: &Cleaned, index: usize, input_len: usize) -> usize {
    clean.positions.get(index).copied().unwrap_or(input_len)
}

fn atom_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn validate_unquoted(local: &[u8], start: usize, clean: &Cleaned, input_len: usize) -> Result<(), EmailError> {
    if local.len() > 64 {
        return Err(EmailError::LocalTooLong {
            at: pos(clean, start + 64, input_len),
        });
    }
    for (i, &b) in local.iter().enumerate() {
        let at = pos(clean, start + i, input_len);
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at });
            }
        } else if !atom_byte(b) {
            return Err(EmailError::LocalChar { at });
        }
    }
    Ok(())
}

fn valid_dot_atom(s: &[u8]) -> bool {
    if s.is_empty() || s.len() > 64 {
        return false;
    }
    let mut prev_dot = true;
    for (i, &b) in s.iter().enumerate() {
        if b == b'.' {
            if prev_dot || i + 1 == s.len() {
                return false;
            }
            prev_dot = true;
        } else if atom_byte(b) {
            prev_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_internal(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let clean = cleanup(raw)?;
    for &(idx, start, _) in &clean.runs {
        let at_begin = idx == 0;
        let at_end = idx == clean.bytes.len();
        let adjacent_at = clean.bytes.get(idx).copied() == Some(b'@')
            || (idx > 0 && clean.bytes.get(idx - 1).copied() == Some(b'@'));
        if !(at_begin || at_end || adjacent_at) {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }
    if clean.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let bytes = &clean.bytes;
    let mut quoted_prefix_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        let mut closing = None;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                closing = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        match closing {
            Some(end) => quoted_prefix_end = Some(end),
            None => return Err(EmailError::UnclosedQuote { at: pos(&clean, 0, raw.len()) }),
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_prefix_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            ats.push(i);
        }
        i += 1;
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt { at: pos(&clean, ats[1], raw.len()) });
    }

    let at_i = ats[0];
    if at_i == 0 {
        return Err(EmailError::EmptyLocal { at: pos(&clean, at_i, raw.len()) });
    }
    let local_bytes = &bytes[..at_i];
    let mut decoded = Vec::new();
    let mut origins = Vec::new();
    let quoted;

    if local_bytes[0] == b'"' {
        quoted = true;
        let close = quoted_prefix_end.unwrap();
        if close + 1 != at_i {
            return Err(EmailError::TrailingLocal { at: pos(&clean, close + 1, raw.len()) });
        }
        let mut j = 1;
        while j < close {
            let b = local_bytes[j];
            if b == b'\\' {
                if j + 1 >= close || !matches!(local_bytes[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: pos(&clean, j, raw.len()) });
                }
                decoded.push(local_bytes[j + 1]);
                origins.push(pos(&clean, j, raw.len()));
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: pos(&clean, j, raw.len()) });
                }
                decoded.push(b);
                origins.push(pos(&clean, j, raw.len()));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos(&clean, close, raw.len()) });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        validate_unquoted(local_bytes, 0, &clean, raw.len())?;
        decoded.extend_from_slice(local_bytes);
        origins.extend((0..local_bytes.len()).map(|j| pos(&clean, j, raw.len())));
    }

    let domain_start = at_i + 1;
    let mut domain_end = bytes.len();
    let mut root_pos = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_pos = Some(pos(&clean, domain_end, raw.len()));
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain { at: pos(&clean, domain_start, raw.len()) });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong { at: pos(&clean, domain_start + 253, raw.len()) });
    }

    let domain = &bytes[domain_start..domain_end];
    let mut label_ranges = Vec::new();
    let mut label_start = 0usize;
    for idx in 0..=domain.len() {
        if idx == domain.len() || domain[idx] == b'.' {
            let absolute_start = domain_start + label_start;
            let absolute_end = domain_start + idx;
            if label_start == idx {
                let p = if idx == domain.len() && root_pos.is_some() {
                    root_pos.unwrap()
                } else if idx > 0 {
                    pos(&clean, domain_start + idx - 1, raw.len())
                } else {
                    pos(&clean, absolute_start, raw.len())
                };
                return Err(EmailError::EmptyLabel { at: p });
            }
            let len = idx - label_start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: pos(&clean, absolute_start + 63, raw.len()) });
            }
            let label = &bytes[absolute_start..absolute_end];
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos(&clean, absolute_start, raw.len()) });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos(&clean, absolute_end - 1, raw.len()) });
            }
            for (k, &b) in label.iter().enumerate() {
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar { at: pos(&clean, absolute_start + k, raw.len()) });
                }
            }
            label_ranges.push((absolute_start, absolute_end));
            label_start = idx + 1;
        }
    }

    if !allow_single_label && label_ranges.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: pos(&clean, domain_start, raw.len()) });
    }
    let (last_start, last_end) = *label_ranges.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(&clean, last_start, raw.len()) });
    }

    let mut local = String::from_utf8(decoded).unwrap();
    let mut transformed_origins = origins;
    if false {
        transformed_origins.clear();
    }
    let domain_string = String::from_utf8(domain.iter().map(u8::to_ascii_lowercase).collect()).unwrap();

    Ok((
        Address {
            local,
            domain: domain_string,
            quoted,
        },
        transformed_origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, false).map(|v| v.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_internal(input, policy.allow_single_label)?;
    let mut local = address.local.into_bytes();
    let mut origin_map = origins;

    if policy.local_case == LocalCase::Lower {
        for b in &mut local {
            *b = b.to_ascii_lowercase();
        }
    }

    if policy.strip_subaddress {
        if let Some(plus) = local.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origin_map.get(plus).copied().unwrap_or(input.len()),
                });
            }
            local.truncate(plus);
            origin_map.truncate(plus);
        }
    }

    if local.is_empty() {
        return Err(EmailError::EmptyNormalizedLocal { at: input.len() });
    }

    let safe = valid_dot_atom(&local);
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    address.local = String::from_utf8(local).unwrap();
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => errors.push((i, e)),
            Ok(address) => {
                if let Some((_, _, first)) = keys.iter().find(|(local, domain, _)| {
                    local == &address.local && domain == &address.domain
                }) {
                    duplicate_of[i] = Some(*first);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), i));
                    source_indices.push(i);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
