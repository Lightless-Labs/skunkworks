#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;

    // The first retained byte decides whether a quoted prefix is protected.
    let mut first_retained = true;
    let mut in_quote = false;

    while i < input.len() {
        if in_quote {
            bytes.push(input[i]);
            positions.push(i);
            if input[i] == b'\\' {
                i += 1;
                if i < input.len() {
                    bytes.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
                continue;
            }
            if input[i] == b'"' {
                in_quote = false;
            }
            i += 1;
            continue;
        }

        if first_retained && input[i] == b'"' {
            first_retained = false;
            in_quote = true;
            bytes.push(input[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        if input[i] == b' ' || input[i] == b'\t' || input[i] == b'\r' || input[i] == b'\n' || input[i] == b'(' {
            let start = i;
            let mut depth = 0usize;
            let mut in_run = true;

            while i < input.len() && in_run {
                match input[i] {
                    b' ' | b'\t' => i += 1,
                    b'\r' | b'\n' => {
                        if let Some(end) = fold_end(input, i) {
                            i = end;
                        } else {
                            return Err(EmailError::BadFold { at: i });
                        }
                    }
                    b'(' => {
                        if depth == 0 {
                            depth = 1;
                        } else {
                            depth += 1;
                        }
                        if depth > 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        i += 1;
                    }
                    b')' if depth > 0 => {
                        depth -= 1;
                        i += 1;
                    }
                    b'\\' if depth > 0 => {
                        let slash = i;
                        if i + 1 >= input.len()
                            || !(0x20..=0x7e).contains(&input[i + 1])
                        {
                            return Err(EmailError::CommentEscape { at: slash });
                        }
                        i += 2;
                    }
                    b'\r' | b'\n' if depth > 0 => unreachable!(),
                    b')' => {
                        return Err(EmailError::UnexpectedClose { at: i });
                    }
                    b if depth > 0 => {
                        if !(0x20..=0x7e).contains(&b) && b != b'\t' {
                            return Err(EmailError::CommentChar { at: i });
                        }
                        i += 1;
                    }
                    _ => in_run = false,
                }

                if depth == 0 && i < input.len() {
                    match input[i] {
                        b' ' | b'\t' | b'\r' | b'\n' | b'(' => {}
                        _ => in_run = false,
                    }
                }
            }

            if depth > 0 {
                // Find the outermost opening parenthesis in this run.
                let mut d = 0usize;
                let mut outer = start;
                let mut k = start;
                while k < i {
                    if input[k] == b'(' {
                        if d == 0 {
                            outer = k;
                        }
                        d += 1;
                    } else if input[k] == b')' && d > 0 {
                        d -= 1;
                    } else if input[k] == b'\\' {
                        k += 1;
                    }
                    k += 1;
                }
                return Err(EmailError::UnclosedComment { at: outer });
            }

            runs.push(Run { start, end: i });
            continue;
        }

        if input[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        first_retained = false;
        bytes.push(input[i]);
        positions.push(i);
        i += 1;
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn original_pos(cleaned: &Cleaned, index: usize, input_len: usize) -> usize {
    cleaned.positions.get(index).copied().unwrap_or(input_len)
}

fn err_at<F>(cleaned: &Cleaned, index: usize, input_len: usize, make: F) -> EmailError
where
    F: FnOnce(usize) -> EmailError,
{
    make(original_pos(cleaned, index, input_len))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let bytes = &cleaned.bytes;

    // Boundary validation follows lexical cleanup and uses original run starts.
    for run in &cleaned.runs {
        let left_ok = run.start == 0;
        let right_ok = run.end == raw.len();
        let left_adjacent_at = cleaned
            .positions
            .iter()
            .position(|&p| p > run.end - 1)
            .and_then(|idx| idx.checked_sub(1))
            .is_some_and(|idx| bytes.get(idx) == Some(&b'@'));
        let right_adjacent_at = cleaned
            .positions
            .iter()
            .position(|&p| p >= run.end)
            .is_some_and(|idx| bytes.get(idx) == Some(&b'@'));
        if !(left_ok || right_ok || left_adjacent_at || right_adjacent_at) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quoted_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quoted_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quoted_end.is_none() {
            return Err(err_at(&cleaned, 0, raw.len(), |at| EmailError::UnclosedQuote { at }));
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            ats.push(i);
        }
        i += 1;
    }

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(err_at(&cleaned, ats[1], raw.len(), |at| EmailError::MultipleAt { at }));
    }

    let separator = ats[0];
    if separator == 0 {
        return Err(err_at(&cleaned, separator, raw.len(), |at| EmailError::EmptyLocal { at }));
    }

    let local_bytes = &bytes[..separator];
    let domain_start = separator + 1;

    let (local, is_quoted) = if local_bytes.first() == Some(&b'"') {
        let end = quoted_end.unwrap_or(0);
        if end + 1 < separator {
            return Err(err_at(&cleaned, end + 1, raw.len(), |at| EmailError::TrailingLocal { at }));
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            if local_bytes[j] == b'\\' {
                if j + 1 >= end {
                    return Err(err_at(&cleaned, j, raw.len(), |at| EmailError::BadEscape { at }));
                }
                let next = local_bytes[j + 1];
                if !matches!(next, b'"' | b'\\' | b'@') {
                    return Err(err_at(&cleaned, j, raw.len(), |at| EmailError::BadEscape { at }));
                }
                decoded.push(next);
                origins.push(original_pos(&cleaned, j, raw.len()));
                j += 2;
            } else {
                let b = local_bytes[j];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(err_at(&cleaned, j, raw.len(), |at| EmailError::QuotedChar { at }));
                }
                decoded.push(b);
                origins.push(original_pos(&cleaned, j, raw.len()));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(err_at(&cleaned, end, raw.len(), |at| EmailError::EmptyLocal { at }));
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(err_at(&cleaned, 64, raw.len(), |at| EmailError::LocalTooLong { at }));
        }
        let allowed_special = b"!#$%&'*+-/=?^_`{|}~";
        for j in 0..local_bytes.len() {
            let b = local_bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(err_at(&cleaned, j, raw.len(), |at| EmailError::LocalDot { at }));
                }
            } else if !b.is_ascii_alphanumeric() && !allowed_special.contains(&b) {
                return Err(err_at(&cleaned, j, raw.len(), |at| EmailError::LocalChar { at }));
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let mut domain_end = bytes.len();
    let mut root_dot = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        root_dot = Some(domain_end - 1);
        domain_end -= 1;
    }

    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: original_pos(&cleaned, domain_start, raw.len()),
        });
    }

    if domain_end - domain_start > 253 {
        return Err(err_at(&cleaned, domain_start + 253, raw.len(), |at| EmailError::DomainTooLong { at }));
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || bytes[j] == b'.' {
            if label_start == j {
                let pos_index = if j == domain_end {
                    root_dot.unwrap_or(j)
                } else {
                    j
                };
                return Err(EmailError::EmptyLabel {
                    at: original_pos(&cleaned, pos_index, raw.len()),
                });
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    for &(start, end) in &labels {
        if end - start > 63 {
            return Err(err_at(&cleaned, start + 63, raw.len(), |at| EmailError::LabelTooLong { at }));
        }
        if bytes[start] == b'-' {
            return Err(err_at(&cleaned, start, raw.len(), |at| EmailError::LabelHyphen { at }));
        }
        if bytes[end - 1] == b'-' {
            return Err(err_at(&cleaned, end - 1, raw.len(), |at| EmailError::LabelHyphen { at }));
        }
        for j in start..end {
            let b = bytes[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(err_at(&cleaned, j, raw.len(), |at| EmailError::DomainChar { at }));
            }
        }
    }

    if labels.len() < 2 {
        return Err(err_at(&cleaned, domain_start, raw.len(), |at| EmailError::DomainNeedsDot { at }));
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(err_at(&cleaned, last_start, raw.len(), |at| EmailError::NumericTld { at }));
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted: is_quoted,
    })
}
