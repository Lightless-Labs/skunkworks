```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn components(path: &str) -> Vec<&str> {
        path.split('/')
            .filter(|part| !part.is_empty() && *part != "." && *part != "..")
            .collect()
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts: Vec<String> = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
                .into_iter()
                .map(str::to_string)
                .collect()
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_string()),
            }
        }

        if parts.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", parts.join("/")))
        }
    }

    fn path_parts(path: &str) -> Vec<&str> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').collect()
        }
    }

    fn parent_and_name(path: &str) -> (&str, &str) {
        match path.rfind('/') {
            Some(0) => ("/", &path[1..]),
            Some(index) => (&path[..index], &path[index + 1..]),
            None => ("/", path),
        }
    }

    fn lookup(&self, path: &str) -> Result<&Entry, Error> {
        let parts = Self::path_parts(path);
        let mut prefix = String::new();

        if parts.is_empty() {
            return self.entries.get("/").ok_or_else(|| Error::Missing("/".into()));
        }

        for (index, part) in parts.iter().enumerate() {
            prefix.push('/');
            prefix.push_str(part);

            let entry = self
                .entries
                .get(&prefix)
                .ok_or_else(|| Error::Missing(prefix.clone()))?;

            if index + 1 < parts.len() && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(prefix));
            }

            if index + 1 == parts.len() {
                return Ok(entry);
            }
        }

        self.entries.get("/").ok_or_else(|| Error::Missing("/".into()))
    }

    fn check_parent(&self, path: &str) -> Result<(), Error> {
        let (parent, _) = Self::parent_and_name(path);
        match self.lookup(parent)? {
            Entry::Dir => Ok(()),
            Entry::File(_) => Err(Error::NotDir(parent.to_string())),
        }
    }

    fn exists(&self, path: &str) -> bool {
        self.entries.contains_key(path)
    }

    fn contains_path(parent: &str, child: &str) -> bool {
        if parent == "/" {
            child.starts_with('/')
        } else {
            child == parent
                || child
                    .strip_prefix(parent)
                    .is_some_and(|suffix| suffix.starts_with('/'))
        }
    }

    fn direct_children(&self, path: &str) -> Vec<(String, Kind)> {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{}/", path)
        };

        self.entries
            .iter()
            .filter_map(|(candidate, entry)| {
                let rest = candidate.strip_prefix(&prefix)?;
                if !rest.is_empty() && !rest.contains('/') {
                    Some((rest.to_string(), entry.kind()))
                } else {
                    None
                }
            })
            .collect()
    }

    fn remove_subtree(&mut self, path: &str) {
        let keys: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| Self::contains_path(path, candidate))
            .cloned()
            .collect();
        for key in keys {
            self.entries.remove(&key);
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        self.check_parent(&path)?;
        if self.exists(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        self.check_parent(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(_)) | None => {
                self.entries.insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        match self.lookup(&path)? {
            Entry::File(data) => Ok(data.clone()),
            Entry::Dir => Err(Error::IsDir(path)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        match self.lookup(&path)? {
            Entry::Dir => Ok(self.direct_children(&path)),
            Entry::File(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        match self.lookup(&path)? {
            Entry::Dir => {
                self.cwd = path;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = self.normalize(from)?;
        let to = self.normalize(to)?;

        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        self.check_parent(&from)?;
        if !self.exists(&from) {
            return Err(Error::Missing(from));
        }

        if matches!(self.entries.get(&from), Some(Entry::Dir))
            && Self::contains_path(&from, &self.cwd)
        {
            return Err(Error::Busy(from));
        }

        self.check_parent(&to)?;

        if matches!(self.entries.get(&to), Some(Entry::Dir))
            && Self::contains_path(&to, &self.cwd)
        {
            return Err(Error::Busy(to));
        }

        if from == to {
            return Ok(());
        }

        let source_is_dir = matches!(self.entries.get(&from), Some(Entry::Dir));
        if source_is_dir && Self::contains_path(&from, &to) {
            return Err(Error::Cycle(to));
        }

        let destination = self.entries.get(&to);
        if destination.is_some() && !overwrite {
            return Err(Error::Exists(to));
        }

        if let Some(destination_entry) = destination {
            match (source_is_dir, destination_entry) {
                (true, Entry::File(_)) => return Err(Error::NotDir(to)),
                (false, Entry::Dir) => return Err(Error::IsDir(to)),
                (true, Entry::Dir) if !self.direct_children(&to).is_empty() => {
                    return Err(Error::NotEmpty(to));
                }
                _ => {}
            }
        }

        let source_entries: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::contains_path(&from, path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if destination.is_some() {
            self.remove_subtree(&to);
        }
        self.remove_subtree(&from);

        for (old_path, entry) in source_entries {
            let suffix = &old_path[from.len()..];
            let new_path = format!("{to}{suffix}");
            self.entries.insert(new_path, entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        self.check_parent(&path)?;
        let entry = self
            .entries
            .get(&path)
            .ok_or_else(|| Error::Missing(path.clone()))?;

        if matches!(entry, Entry::Dir) && Self::contains_path(&path, &self.cwd) {
            return Err(Error::Busy(path));
        }

        if matches!(entry, Entry::Dir)
            && !recursive
            && !self.direct_children(&path).is_empty()
        {
            return Err(Error::NotEmpty(path));
        }

        self.remove_subtree(&path);
        Ok(())
    }
}
```
