pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|n| n == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|n| n == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: closed quote
    let mut opening_quote = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record: records.len(), byte: i });
        }

        if state != 2 && b == b'\r' {
            if i + 1 < bytes.len() && bytes[i + 1] == b'\n' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                i += 2;
                state = 0;
                record_started = false;
                continue;
            }
            return Err(CsvError::BareCr { record: records.len(), byte: i });
        }

        match state {
            2 => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = 3;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
            }
            3 => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    record_started = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    state = 0;
                    record_started = false;
                    i += 1;
                } else {
                    return Err(CsvError::TrailingByte { record: records.len(), byte: i });
                }
            }
            _ => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    record_started = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    state = 0;
                    record_started = false;
                    i += 1;
                } else if b == quote {
                    if state == 0 {
                        opening_quote = i;
                        state = 2;
                        record_started = true;
                        i += 1;
                    } else {
                        return Err(CsvError::UnexpectedQuote { record: records.len(), byte: i });
                    }
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    state = 1;
                    record_started = true;
                    i += ch.len_utf8();
                }
            }
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote { record: records.len(), byte: opening_quote });
    }
    if record_started || !record.is_empty() || !field.is_empty() || state == 3 {
        record.push(field);
        records.push(record);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        if let Some(first) = header[..second].iter().position(|n| n == &header[second]) {
            return Some(CsvError::DuplicateHeader {
                name: header[second].clone(),
                first,
                second,
            });
        }
    }
    None
}

fn build_table(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers && !records.is_empty() {
        (Some(records[0].clone()), records.into_iter().skip(1).collect())
    } else {
        (None, records)
    };

    if let Some(h) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(err) = duplicate_header(h) {
                return Err(err);
            }
        }
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), result: None })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start])
        && b[start + 1..].iter().all(|c| c.is_ascii_digit())
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.starts_with(b"-"));
    if b.len() < start + 4 { return false; }
    let dot = b.len() - 3;
    if b[dot] != b'.' || !b[dot + 1..].iter().all(|c| c.is_ascii_digit()) {
        return false;
    }
    if dot <= start { return false; }
    if b[start] == b'0' {
        if dot != start + 1 { return false; }
    } else if !(b'1'..=b'9').contains(&b[start])
        || !b[start + 1..dot].iter().all(|c| c.is_ascii_digit())
    {
        return false;
    }
    !(start == 1 && b[dot - 3..dot] == *b"0" && b[dot + 1..] == *b"00")
}

fn eval_value(s: &str, kind: ColumnType, nullable: bool) -> Result<Value, CellErrorKind> {
    if s.is_empty() {
        if nullable { return Ok(Value::Null); }
        if kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match kind {
        ColumnType::Text => Ok(Value::Text(s.to_owned())),
        ColumnType::Integer => {
            if !integer_syntax(s) { return Err(CellErrorKind::InvalidInteger); }
            s.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = s.starts_with('-');
            let body = if neg { &s[1..] } else { s };
            let dot = body.len() - 3;
            let whole: i128 = body[..dot].parse().unwrap();
            let frac: i128 = body[dot + 1..].parse().unwrap();
            let signed = (whole * 100 + frac) * if neg { -1 } else { 1 };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = vec![None; columns.len()];
        for (spec_index, spec) in columns.iter().enumerate() {
            let index = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, value) = match index {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(i) => match row.get(i) {
                    None => (Some(i), Err(CellErrorKind::MissingCell)),
                    Some(s) => (Some(i), eval_value(s, spec.kind, spec.nullable)),
                },
            };
            match value {
                Ok(v) => out[spec_index] = Some(v),
                Err(kind) => {
                    result.errors.push(CellError { row: row_index, spec: spec_index, column, kind });
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow => break,
                        ErrorPolicy::Stop => {
                            result.rows.push(out);
                            break 'rows;
                        }
                    }
                }
            }
        }
        result.rows.push(out);
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding { Lf, CrLf }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode { Preserve, Canonical }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header { original_records.push(header); }
    original_records.extend(table.rows.iter().map(Vec::as_slice));

    for (record_index, record) in original_records.iter().enumerate() {
        for (column_index, cell) in record.iter().enumerate() {
            if let Some(byte) = cell.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul { record: record_index, column: column_index, byte });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1, expected: header.len(), actual: row.len(),
                });
            }
        }
    }

    for (record, fields) in original_records.iter().enumerate() {
        if fields.is_empty() { return Err(WriteError::ZeroColumns { record }); }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(CsvError::DuplicateHeader { name, first, second }) = duplicate_header(header) {
                    return Err(WriteError::DuplicateHeader { name, first, second });
                }
            }
        }
    }

    let mut records: Vec<Vec<String>> = if let Some(header) = &table.header {
        let mut rows = Vec::new();
        match config.mode {
            WriteMode::Preserve => {
                rows.push(header.clone());
                rows.extend(table.rows.clone());
            }
            WriteMode::Canonical => {
                let mut chosen: Vec<(String, usize)> = Vec::new();
                for (index, name) in header.iter().enumerate() {
                    if let Some(pos) = chosen.iter().position(|(n, _)| n == name) {
                        if table.duplicates == DuplicatePolicy::KeepLast {
                            chosen[pos].1 = index;
                        }
                    } else {
                        chosen.push((name.clone(), index));
                    }
                }
                chosen.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
                rows.push(chosen.iter().map(|(n, _)| n.clone()).collect());
                let mut projected: Vec<Vec<String>> = table.rows.iter().map(|row| {
                    chosen.iter().map(|(_, i)| row[*i].clone()).collect()
                }).collect();
                projected.sort_by(|a, b| compare_rows(a, b));
                rows.extend(projected);
            }
        }
        rows
    } else {
        let mut rows = table.rows.clone();
        if config.mode == WriteMode::Canonical {
            rows.sort_by(|a, b| compare_rows(a, b));
        }
        rows
    };

    if records.is_empty() { return Ok(String::new()); }

    let delimiter = config.delimiter as char;
    let quote = config.quote as char;
    let mut output = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let ending = if canonical || config.line_ending == LineEnding::Lf { "\n" } else { "\r\n" };

    for (ri, record) in records.iter_mut().enumerate() {
        if ri > 0 { output.push_str(ending); }
        for (ci, field) in record.iter().enumerate() {
            if ci > 0 { output.push(delimiter); }
            let bytes = field.as_bytes();
            let must_quote = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
                || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
                || (ri == 0 && ci == 0 && field.starts_with('\u{feff}'))
                || (record.len() == 1 && field.is_empty());
            if must_quote {
                output.push(quote);
                for ch in field.chars() {
                    if ch == quote { output.push(quote); }
                    output.push(ch);
                }
                output.push(quote);
            } else {
                output.push_str(field);
            }
        }
    }
    if canonical || config.final_terminator { output.push_str(ending); }
    Ok(output)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let ord = x.as_bytes().cmp(y.as_bytes());
        if ord != std::cmp::Ordering::Equal { return ord; }
    }
    a.len().cmp(&b.len())
}
