#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut field_started = false;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0;
    let mut record_index = 0;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if !in_quotes && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: i,
                });
            }
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            field_started = false;
            after_quote = false;
            i += 2;
            record_index += 1;
            continue;
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().expect("valid UTF-8");
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                field_started = false;
                after_quote = false;
                i += 1;
                continue;
            }
            if b == b'\n' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                field_started = false;
                after_quote = false;
                i += 1;
                record_index += 1;
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            field_started = false;
            i += 1;
        } else if b == b'\n' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            field_started = false;
            i += 1;
            record_index += 1;
        } else if b == quote {
            if !field_started {
                in_quotes = true;
                quote_start = i;
                field_started = true;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
        } else {
            let ch = input[i..].chars().next().expect("valid UTF-8");
            field.push(ch);
            field_started = true;
            i += ch.len_utf8();
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if !record.is_empty() || field_started || !field.is_empty() || after_quote {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(seen_name, _)| *seen_name == name) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn table_from_records(
    mut records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(h) {
                return Err(error);
            }
        }

        for (idx, row) in records.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let digits = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return bytes[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let body = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
    let Some(dot) = body.iter().position(|b| *b == b'.') else {
        return false;
    };
    if body.len() != dot + 3 {
        return false;
    }
    let integral = &body[..dot];
    let fractional = &body[dot + 1..];
    if fractional.len() != 2 || !fractional.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if integral == b"0" {
        if bytes[0] == b'-' && fractional == b"00" {
            return false;
        }
        return true;
    }
    !integral.is_empty()
        && integral[0].is_ascii_digit()
        && integral[0] != b'0'
        && integral.iter().all(u8::is_ascii_digit)
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let digits = body.replace('.', "");
    let magnitude = digits
        .parse::<u64>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    if negative {
        if magnitude == (i64::MAX as u64) + 1 {
            Ok(i64::MIN)
        } else if magnitude <= i64::MAX as u64 {
            Ok(-(magnitude as i64))
        } else {
            Err(CellErrorKind::DecimalOverflow)
        }
    } else if magnitude <= i64::MAX as u64 {
        Ok(magnitude as i64)
    } else {
        Err(CellErrorKind::DecimalOverflow)
    }
}

fn evaluate_cell(
    table: &Table,
    row_index: usize,
    row: &[String],
    spec_index: usize,
    spec: &ColumnSpec,
) -> (Option<Value>, Option<CellError>) {
    let column = match &spec.column {
        ColumnRef::Index(index) => Some(*index),
        ColumnRef::Name(name) => match table.column_index(name) {
            Some(index) => Some(index),
            None => {
                return (
                    None,
                    Some(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    }),
                )
            }
        },
    };

    let column = column.unwrap();
    let Some(cell) = row.get(column) else {
        return (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind: CellErrorKind::MissingCell,
            }),
        );
    };

    if cell.is_empty() {
        if spec.nullable {
            return (Some(Value::Null), None);
        }
        if spec.kind != ColumnType::Text {
            return (
                None,
                Some(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column),
                    kind: CellErrorKind::Empty,
                }),
            );
        }
    }

    let parsed = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.clone())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    };

    match parsed {
        Ok(value) => (Some(value), None),
        Err(kind) => (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(column),
                kind,
            }),
        ),
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut rows = Vec::new();
    let mut errors = Vec::new();

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed {
                output.push(None);
                continue;
            }

            let (value, error) =
                evaluate_cell(table, row_index, row, spec_index, spec);
            output.push(value);
            if let Some(error) = error {
                errors.push(error);
                match policy {
                    ErrorPolicy::Collect => {}
                    ErrorPolicy::FirstPerRow => row_failed = true,
                    ErrorPolicy::Stop => {
                        output.resize(columns.len(), None);
                        rows.push(output);
                        return TypedResult { rows, errors };
                    }
                }
            }
        }

        rows.push(output);
    }

    TypedResult { rows, errors }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns { record: usize },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn scan_nul(table: &Table) -> Option<WriteError> {
    let mut scan_record = |record_index: usize, row: &[String]| {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Some(WriteError::Nul {
                    record: record_index,
                    column,
                    byte,
                });
            }
        }
        None
    };

    if let Some(header) = &table.header {
        if let Some(error) = scan_record(0, header) {
            return Some(error);
        }
        for (idx, row) in table.rows.iter().enumerate() {
            if let Some(error) = scan_record(idx + 1, row) {
                return Some(error);
            }
        }
    } else {
        for (idx, row) in table.rows.iter().enumerate() {
            if let Some(error) = scan_record(idx, row) {
                return Some(error);
            }
        }
    }
    None
}

fn quote_field(field: &str, first_field_first_record: bool, sole_empty: bool, delimiter: u8, quote: u8) -> String {
    let bytes = field.as_bytes();
    let special = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_first_record && field.starts_with('\u{feff}'))
        || (sole_empty && field.is_empty());

    if !special {
        return field.to_string();
    }

    let mut out = String::new();
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
    out
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        let ord = x.as_bytes().cmp(y.as_bytes());
        if ord != std::cmp::Ordering::Equal {
            return ord;
        }
    }
    a.len().cmp(&b.len())
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    if let Some(error) = scan_nul(table) {
        return Err(error);
    }

    if let Some(header) = &table.header {
        for (idx, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: idx + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    if let Some(header) = &table.header {
        if header.is_empty() {
            return Err(WriteError::ZeroColumns { record: 0 });
        }
    }
    for (idx, row) in table.rows.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: idx + usize::from(table.header.is_some()),
            });
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();

    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                records.push(header.clone());
            }
            records.extend(table.rows.iter().cloned());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                if table.duplicates == DuplicatePolicy::Reject {
                    if let Some(CsvError::DuplicateHeader {
                        name,
                        first,
                        second,
                    }) = duplicate_header(header)
                    {
                        return Err(WriteError::DuplicateHeader {
                            name,
                            first,
                            second,
                        });
                    }
                }

                let mut retained: Vec<(String, usize)> = Vec::new();
                for (index, name) in header.iter().enumerate() {
                    if let Some(pos) = retained.iter().position(|(n, _)| n == name) {
                        match table.duplicates {
                            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {}
                            DuplicatePolicy::KeepLast => retained[pos].1 = index,
                        }
                    } else {
                        retained.push((name.clone(), index));
                    }
                }
                retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

                let new_header: Vec<String> =
                    retained.iter().map(|(name, _)| name.clone()).collect();
                records.push(new_header);

                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| retained.iter().map(|(_, index)| row[*index].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| compare_rows(a, b));
                records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_rows(a, b));
                records.extend(rows);
            }
        }
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let (terminator, final_terminator) = match config.mode {
        WriteMode::Preserve => (
            match config.line_ending {
                LineEnding::Lf => "\n",
                LineEnding::CrLf => "\r\n",
            },
            config.final_terminator,
        ),
        WriteMode::Canonical => ("\n", true),
    };

    let mut output = String::new();
    for (record_index, record) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(terminator);
        }
        for (column, field) in record.iter().enumerate() {
            if column > 0 {
                output.push(config.delimiter as char);
            }
            output.push_str(&quote_field(
                field,
                record_index == 0 && column == 0,
                record.len() == 1,
                config.delimiter,
                config.quote,
            ));
        }
    }
    if final_terminator {
        output.push_str(terminator);
    }
    Ok(output)
}
