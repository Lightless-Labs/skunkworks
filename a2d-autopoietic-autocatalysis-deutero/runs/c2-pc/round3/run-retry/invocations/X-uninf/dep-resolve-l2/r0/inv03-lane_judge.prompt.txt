fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|&b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(constraints: &[Constraint]) -> Vec<Constraint> {
    let mut result = constraints.to_vec();
    result.sort();
    result.dedup();
    result
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical_constraints(constraints)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(text: &str) -> String {
    let mut out = String::new();
    for byte in text.as_bytes() {
        match *byte {
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'.' | b'_' | b'-' => {
                out.push(*byte as char);
            }
            b => out.push_str(&format!("%{:02X}", b)),
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches_constraint(version, *constraint))
}

fn validate_name(site: String, name: &str) -> Result<(), ResolveError> {
    if valid_name(name) {
        Ok(())
    } else {
        Err(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

fn active_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    selected: &[(String, Version)],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (package_name, version) in selected {
        if let Some(package) = catalog.iter().find(|p| p.name == *package_name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        *constraints = canonical_constraints(constraints);
    }

    active
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut Vec<(String, Version)>,
    attempts: &mut usize,
    max_attempts: usize,
    first_failure: &mut Option<ResolveError>,
) -> Result<Option<Vec<Selection>>, ResolveError> {
    let active = active_constraints(roots, catalog, selected);

    for package_name in active.keys() {
        if !catalog.iter().any(|package| package.name == *package_name) {
            return Err(ResolveError::Missing {
                package: package_name.clone(),
            });
        }
    }

    for (package_name, version) in selected.iter() {
        if let Some(constraints) = active.get(package_name) {
            if !matches_all(*version, constraints) {
                return Err(ResolveError::Conflict {
                    package: package_name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    let selected_names: std::collections::BTreeMap<&str, Version> = selected
        .iter()
        .map(|(name, version)| (name.as_str(), *version))
        .collect();

    for (package_name, constraints) in &active {
        if selected_names.contains_key(package_name.as_str()) {
            continue;
        }
        let package = catalog
            .iter()
            .find(|p| p.name == *package_name)
            .expect("active packages were checked above");
        if !package
            .releases
            .iter()
            .any(|release| matches_all(release.version, constraints))
        {
            return Err(ResolveError::NoMatch {
                package: package_name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    let next_package = active
        .keys()
        .find(|name| !selected_names.contains_key(name.as_str()))
        .cloned();

    let Some(package_name) = next_package else {
        let mut result: Vec<Selection> = selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect();
        result.sort_by(|a, b| a.package.cmp(&b.package));
        return Ok(Some(result));
    };

    let package = catalog
        .iter()
        .find(|p| p.name == package_name)
        .expect("active packages were checked above");
    let constraints = active.get(&package_name).expect("active package exists");
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|release| matches_all(release.version, constraints))
        .map(|release| release.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.push((package_name.clone(), version));

        match search(
            catalog,
            roots,
            selected,
            attempts,
            max_attempts,
            first_failure,
        ) {
            Ok(Some(solution)) => return Ok(Some(solution)),
            Ok(None) => {
                selected.pop();
            }
            Err(error) => {
                selected.pop();
                if matches!(error, ResolveError::Limit { .. }) {
                    return Err(error);
                }
                if first_failure.is_none() {
                    *first_failure = Some(error);
                }
            }
        }
    }

    Ok(None)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    for (index, root) in roots.iter().enumerate() {
        validate_name(format!("root[{}]", index), &root.package)?;
    }

    for (package_index, package) in catalog.iter().enumerate() {
        validate_name(format!("catalog[{}]", package_index), &package.name)?;
        for (release_index, release) in package.releases.iter().enumerate() {
            for (dep_index, dep) in release.deps.iter().enumerate() {
                validate_name(
                    format!(
                        "catalog[{}].release[{}].dep[{}]",
                        package_index, release_index, dep_index
                    ),
                    &dep.package,
                )?;
            }
        }
    }

    let mut seen_packages = std::collections::BTreeMap::<&str, ()>::new();
    for package in catalog {
        if seen_packages.insert(&package.name, ()).is_some() {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeMap::<Version, ()>::new();
        for release in &package.releases {
            if seen_versions.insert(release.version, ()).is_some() {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut attempts = 0;
    let mut first_failure = None;
    let mut selected = Vec::new();
    match search(
        catalog,
        roots,
        &mut selected,
        &mut attempts,
        max_attempts,
        &mut first_failure,
    )? {
        Some(solution) => Ok(solution),
        None => Err(first_failure.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
        })),
    }
}

fn validate(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut names = std::collections::BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),

                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    Ok(())
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn constraint_text(constraint: Constraint) -> String {
    let (prefix, v) = match constraint {
        Constraint::Exact(v) => ("=", v),
        Constraint::AtLeast(v) => (">=", v),
        Constraint::Before(v) => ("<", v),
        Constraint::Caret(v) => ("^", v),
    };
    format!("{prefix}{}.{}.{}", v.major, v.minor, v.patch)
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for byte in text.bytes() {
        if byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-') {
            out.push(byte as char);
        } else {
            out.push('%');
            out.push(char::from(b"0123456789ABCDEF"[(byte >> 4) as usize]));
            out.push(char::from(b"0123456789ABCDEF"[(byte & 0x0f) as usize]));
        }
    }
    out
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &mut std::collections::BTreeMap<String, Version>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in selected.iter() {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    for constraints in active.values_mut() {
        canonicalize(constraints);
    }

    if let Some(name) = active
        .keys()
        .filter(|name| !catalog.iter().any(|p| p.name == ***name))
        .next()
    {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, version) in selected.iter() {
        if let Some(constraints) = active.get(name) {
            if constraints.iter().any(|c| !matches(*version, *c)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in &active {
        if !selected.contains_key(name) {
            let package = catalog.iter().find(|p| p.name == *name).unwrap();
            if !package
                .releases
                .iter()
                .any(|r| constraints.iter().all(|c| matches(r.version, *c)))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }
    }

    let next = active.keys().find(|name| !selected.contains_key(*name));
    let Some(name) = next else {
        return Ok(selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect());
    };

    let package = catalog.iter().find(|p| p.name == *name).unwrap();
    let constraints = &active[name];
    let mut candidates: Vec<Version> = package
        .releases
        .iter()
        .filter(|r| constraints.iter().all(|c| matches(r.version, *c)))
        .map(|r| r.version)
        .collect();
    candidates.sort_by(|a, b| b.cmp(a));

    let mut first_error = None;
    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.insert(name.clone(), version);
        match search(catalog, roots, selected, attempts, max_attempts) {
            Ok(solution) => return Ok(solution),
            Err(error @ ResolveError::Limit { .. }) => return Err(error),
            Err(error) => {
                selected.remove(name);
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
        package: name.clone(),
        requirements: constraints.clone(),
    }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;
    let mut attempts = 0;
    search(
        catalog,
        roots,
        &mut std::collections::BTreeMap::new(),
        &mut attempts,
        max_attempts,
    )
}
