#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (b.x as i128) - (a.x as i128);
    let dy = (b.y as i128) - (a.y as i128);
    let scaled_squared = ((dx * dx + dy * dy) as u128) * 1_000_000;

    // Find floor(sqrt(scaled_squared)) without floating-point arithmetic.
    let mut low = 0_u128;
    let mut high = 3_000_000_001_u128;
    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid * mid <= scaled_squared {
            low = mid;
        } else {
            high = mid;
        }
    }

    // Round to nearest integer, with exact half ties rounded upward. Comparing
    // against (low + 1/2)^2 avoids taking a rounded square root.
    let boundary = 4 * scaled_squared;
    let midpoint_squared = (2 * low + 1) * (2 * low + 1);
    if boundary >= midpoint_squared {
        (low + 1) as u64
    } else {
        low as u64
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    if points.len() > 4096 {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, point) in points.iter().enumerate() {
        if !(-1_000_000_000..=1_000_000_000).contains(&point.x)
            || !(-1_000_000_000..=1_000_000_000).contains(&point.y)
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = points.len()
        - usize::from(points.len() >= 2 && points.first() == points.last());
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let vertices = &points[..effective_len];
    let mut signed_area2 = 0_i128;
    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0_u64;
    let mut longest_edge: Option<usize> = None;

    for index in 0..effective_len {
        let a = vertices[index];
        let b = vertices[(index + 1) % effective_len];

        signed_area2 += (a.x as i128) * (b.y as i128)
            - (a.y as i128) * (b.x as i128);

        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace_longest = match longest_edge {
            None => true,
            Some(best_index) => {
                if length > edge_milli[best_index] {
                    true
                } else if length < edge_milli[best_index] {
                    false
                } else {
                    let best_a = vertices[best_index];
                    let best_b = vertices[(best_index + 1) % effective_len];
                    let pair = (a.min(b), a.max(b));
                    let best_pair = (best_a.min(best_b), best_a.max(best_b));
                    pair < best_pair || (pair == best_pair && index < best_index)
                }
            }
        };
        if replace_longest {
            longest_edge = Some(index);
        }
    }

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
