#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (b.x as i128 - a.x as i128).unsigned_abs();
    let dy = (b.y as i128 - a.y as i128).unsigned_abs();
    let scaled_squared = (dx * dx + dy * dy) * 1_000_000;

    // Find floor(sqrt(scaled_squared)) with integer arithmetic.
    let mut low = 0u64;
    let mut high = 2_000_000_000_000u64;
    while low < high {
        let mid = low + (high - low + 1) / 2;
        let square = (mid as u128) * (mid as u128);
        if square <= scaled_squared {
            low = mid;
        } else {
            high = mid - 1;
        }
    }

    let floor = low as u128;
    let remainder = scaled_squared - floor * floor;
    // The half-integer threshold is floor + 0.5, so compare 4*remainder
    // with 4*floor + 1 without evaluating a floating-point square root.
    if 4 * remainder >= 4 * floor + 1 {
        low + 1
    } else {
        low
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let raw_len = points.len();
    if raw_len > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: raw_len });
    }

    for (index, point) in points.iter().enumerate() {
        if point.x < -COORD_LIMIT
            || point.x > COORD_LIMIT
            || point.y < -COORD_LIMIT
            || point.y > COORD_LIMIT
        {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if raw_len >= 2 && points[raw_len - 1] == points[0] {
        raw_len - 1
    } else {
        raw_len
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let polygon = &points[..effective_len];
    let mut signed_area2 = 0i128;
    for i in 0..effective_len {
        let current = polygon[i];
        let next = polygon[(i + 1) % effective_len];
        signed_area2 += current.x as i128 * next.y as i128
            - current.y as i128 * next.x as i128;
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..effective_len {
        let a = polygon[i];
        let b = polygon[(i + 1) % effective_len];
        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(previous_index) => {
                let previous_length = edge_milli[previous_index];
                if length != previous_length {
                    length > previous_length
                } else {
                    let pair = if a <= b { (a, b) } else { (b, a) };
                    let previous_a = polygon[previous_index];
                    let previous_b = polygon[(previous_index + 1) % effective_len];
                    let previous_pair = if previous_a <= previous_b {
                        (previous_a, previous_b)
                    } else {
                        (previous_b, previous_a)
                    };
                    pair < previous_pair || (pair == previous_pair && i < previous_index)
                }
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
