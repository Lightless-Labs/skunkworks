#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependency entries in the prescribed order.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }

    // Kahn traversal, always taking the smallest available task ID.
    let mut available = std::collections::BTreeSet::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(id);
        }
    }

    let mut topo = Vec::with_capacity(n);
    while let Some(&id) = available.iter().next() {
        available.remove(&id);
        topo.push(id);
        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if topo.len() != n {
        let mut removed = vec![false; n];
        for &id in &topo {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom-level tails in reverse topological order.
    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let max_successor_tail = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut completed = vec![false; n];
    let mut started = vec![false; n];

    // A worker's availability time includes cooldown. Its current task is
    // tracked separately so zero-duration completion can be processed now.
    let mut worker_available_at = vec![0u64; workers];
    let mut worker_task: Vec<Option<usize>> = vec![None; workers];

    // Running positive-duration tasks, represented by (finish, task, worker).
    let mut running: Vec<(u64, usize, usize)> = Vec::new();

    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete every task finishing now in ascending task ID order.
        let mut finishing: Vec<(usize, usize)> = running
            .iter()
            .filter_map(|&(finish, task, worker)| {
                if finish == now {
                    Some((task, worker))
                } else {
                    None
                }
            })
            .collect();
        finishing.sort_unstable_by_key(|&(task, _)| task);

        for (task, worker) in finishing {
            running.retain(|&(_, running_task, _)| running_task != task);
            worker_task[worker] = None;
            completed[task] = true;
            completed_count += 1;
            events.push(Event {
                time: now,
                task,
                worker,
                kind: EventKind::Finish,
            });
        }

        // Admit tasks one at a time, recalculating eligibility and ranking
        // after every start and every immediate zero-duration completion.
        loop {
            let worker = (0..workers).find(|&w| {
                worker_task[w].is_none() && worker_available_at[w] <= now
            });
            let Some(worker) = worker else {
                break;
            };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if tasks[id].deps.iter().any(|&dep| !completed[dep]) {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }

            let Some(task) = best else {
                break;
            };

            let finish = now.checked_add(tasks[task].duration).ok_or(Error::Overflow {
                task,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task,
                stage: Stage::Turnaround,
            })?;

            started[task] = true;
            worker_task[worker] = Some(task);
            worker_available_at[worker] = available_at;
            runs[task] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task,
                worker,
                kind: EventKind::Start,
            });

            if tasks[task].duration == 0 {
                // The completion is immediate and precedes the next admission.
                worker_task[worker] = None;
                completed[task] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: finish,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((finish, task, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest event that can change eligibility or
        // worker availability.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                let release = tasks[id].release;
                next_time = Some(next_time.map_or(release, |t| t.min(release)));
            }
        }
        for worker in 0..workers {
            if worker_task[worker].is_none() && worker_available_at[worker] > now {
                let expiry = worker_available_at[worker];
                next_time = Some(next_time.map_or(expiry, |t| t.min(expiry)));
            }
        }

        // With an acyclic graph, unfinished work always has a future enabling
        // event. This fallback also avoids any possibility of spinning.
        if let Some(next) = next_time {
            now = next;
        } else {
            break;
        }
    }

    // The loop can only end successfully after every task was completed.
    let runs = runs
        .into_iter()
        .map(|run| run.expect("all tasks completed"))
        .collect();

    // Derive makespan from actual finishes, excluding cooldown.
    for run in &runs {
        makespan = makespan.max(run.finish);
    }

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
