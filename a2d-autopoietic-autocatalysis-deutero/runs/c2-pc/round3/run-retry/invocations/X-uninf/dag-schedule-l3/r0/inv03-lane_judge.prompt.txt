pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependency entries in the specified order.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest currently available ID.
    let mut available = Vec::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(id);
        }
    }
    let mut topo = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        topo.push(id);
        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if topo.len() != n {
        let removed = {
            let mut done = vec![false; n];
            for &id in &topo {
                done[id] = true;
            }
            done
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });

    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {

            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]

            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;

    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut running: Vec<Running> = Vec::new();
    let mut worker_available_at = vec![0u64; workers];
    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now in task ID order.
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|item| item.finish == time)
            .collect();
        finishing.sort_by_key(|item| item.task);

        if !finishing.is_empty() {
            let finishing_ids: Vec<usize> = finishing.iter().map(|item| item.task).collect();
            running.retain(|item| !finishing_ids.contains(&item.task));

            for item in finishing {
                completed[item.task] = true;
                completed_count += 1;
                makespan = makespan.max(item.finish);
                events.push(Event {
                    time,
                    task: item.task,
                    worker: item.worker,
                    kind: EventKind::Finish,
                });
                for &child in &successors[item.task] {
                    remaining_predecessors[child] -= 1;
                }
            }
        }

        loop {
            let worker = (0..workers).find(|&id| worker_available_at[id] <= time);
            let Some(worker) = worker else {
                break;
            };

            let selected = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= time
                        && remaining_predecessors[id] == 0
                })
                .max_by(|&a, &b| {
                    tails[a]
                        .cmp(&tails[b])
                        .then_with(|| tasks[a].priority.cmp(&tasks[b].priority))
                        .then_with(|| b.cmp(&a))
                });

            let Some(task_id) = selected else {
                break;

                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available_at[worker] = available_at;
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });
            events.push(Event {
                time,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if finish == time {
                // Zero-duration tasks complete immediately before another admission.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        for item in &running {
            consider(item.finish);
        }
        for id in 0..n {
            if !started[id] && remaining_predecessors[id] == 0 {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        // A task may become startable when its last predecessor completes; such a
        // completion is already represented by a running task's finish event.
        time = next_time.expect("unfinished tasks must have a future event");
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();
    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

    // scheduling work.
    for (task_id, task) in tasks.iter().enumerate() {

        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,

            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }

        return Err(Error::Workers { count: workers });

    // affect Kahn's traversal, so successor lists are sorted by task ID.

    let mut indegree = vec![0usize; n];

        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }
    for list in &mut successors {
        list.sort_unstable();
    }

    let mut available = std::collections::BTreeSet::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.insert(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while let Some(id) = available.pop_first() {
        order.push(id);
        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {

            .iter()
            .map(|&child| tails[child])

    let mut now = 0u64;
    let mut makespan = 0u64;

                            || (tails[id] == tails[old]
                                && (tasks[id].priority > tasks[old].priority
                                    || (tasks[id].priority == tasks[old].priority && id < old)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }
            let Some(task_id) = best else {
                break;
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at =
                finish
                    .checked_add(turnaround)
                    .ok_or(Error::Overflow {
                        task: task_id,
                        stage: Stage::Turnaround,
                    })?;

            started[task_id] = true;
            worker_available_at[worker] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if finish == now {
                // Immediate completion for a zero-duration task.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the next strictly later event that can change eligibility
        // or worker availability.
        let mut next = u64::MAX;
        let mut found = false;
        for r in &running {
            if r.finish > now {
                next = next.min(r.finish);
                found = true;
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = next.min(tasks[id].release);
                found = true;
            }
        }
        for &available_at in &worker_available_at {
            if available_at > now {
                next = next.min(available_at);
                found = true;
            }
        }

        if !found {
            // With an acyclic graph and valid event progression this cannot
            // occur, but avoid panicking if an invariant is ever violated.
            break;
        }
        now = next;
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task completes"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
