#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const MAX_COORD: i64 = 1_000_000_000;
const MAX_ITEMS: usize = 4096;

fn in_range(p: Point) -> bool {
    (-MAX_COORD..=MAX_COORD).contains(&p.x) && (-MAX_COORD..=MAX_COORD).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_ITEMS {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }
    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective[0] == *effective.last().unwrap() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    let abx = b.x as i128 - a.x as i128;
    let aby = b.y as i128 - a.y as i128;
    let acx = c.x as i128 - a.x as i128;
    let acy = c.y as i128 - a.y as i128;
    abx * acy - aby * acx
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn edge_mill(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128) as u128;
    let dy = (a.y as i128 - b.y as i128) as u128;
    let scaled_sq = (dx * dx + dy * dy) * 1_000_000u128;

    let mut lo = 0u128;
    let mut hi = 3_000_000_000_000u128;
    while lo < hi {
        let mid = lo + (hi - lo + 1) / 2;
        if mid * mid <= scaled_sq {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }
    let remainder = scaled_sq - lo * lo;
    let rounded = if remainder >= lo * 2 + 1 { lo + 1 } else { lo };
    rounded as u64
}

fn perimeter_milli(points: &[Point]) -> u64 {
    if points.is_empty() {
        return 0;
    }
    let mut sum = 0u64;
    for i in 0..points.len() {
        sum += edge_mill(points[i], points[(i + 1) % points.len()]);
    }
    sum
}

fn calculate_measures(points: &[Point]) -> Measures {
    let signed = signed_area2(points);
    let orientation = if signed > 0 {
        Orientation::CounterClockwise
    } else if signed < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    let edge_milli: Vec<u64> = (0..points.len())
        .map(|i| edge_mill(points[i], points[(i + 1) % points.len()]))
        .collect();
    let perimeter_milli = edge_milli.iter().copied().sum();

    let longest_edge = (0..points.len()).max_by(|&i, &j| {
        edge_milli[i].cmp(&edge_milli[j]).then_with(|| {
            let endpoints = |index: usize| {
                let a = points[index];
                let b = points[(index + 1) % points.len()];
                if a <= b { (a, b) } else { (b, a) }
            };
            endpoints(j)
                .cmp(&endpoints(i))
                .then_with(|| j.cmp(&i))
        })
    });

    Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let effective = prepare_polygon(points)?;
    Ok(calculate_measures(&effective))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_ITEMS {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    Ok(queries
        .iter()
        .map(|&q| {
            for i in 0..polygon.len() {
                if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], q) {
                    return Location::Boundary { edge: i };
                }
            }
            let mut inside = false;
            for i in 0..polygon.len() {
                let a = polygon[i];
                let b = polygon[(i + 1) % polygon.len()];
                if (a.y > q.y) != (b.y > q.y) {
                    let side = cross(a, b, q);
                    if (b.y > a.y && side > 0) || (b.y < a.y && side < 0) {
                        inside = !inside;
                    }
                }
            }
            if inside {
                Location::Inside
            } else {
                Location::Outside
            }
        })
        .collect())
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_ITEMS {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|(p, _)| *p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    let mut vertices_with_sources: Vec<(Point, usize)> = match unique.len() {
        0 => Vec::new(),
        1 => unique.clone(),
        2 => unique.clone(),
        _ => {
            let all_collinear = unique
                .iter()
                .skip(2)
                .all(|&(point, _)| cross(unique[0].0, unique[1].0, point) == 0);
            if all_collinear {
                if mode == Collinear::Include {
                    unique.clone()
                } else {
                    vec![unique[0], *unique.last().unwrap()]
                }
            } else {
                let mut lower: Vec<(Point, usize)> = Vec::new();
                for &item in &unique {
                    while lower.len() >= 2 {
                        let n = lower.len();
                        let turn = cross(lower[n - 2].0, lower[n - 1].0, item.0);
                        if if mode == Collinear::Include {
                            turn < 0
                        } else {
                            turn <= 0
                        } {
                            lower.pop();
                        } else {
                            break;
                        }
                    }
                    lower.push(item);
                }
                let mut upper: Vec<(Point, usize)> = Vec::new();
                for &item in unique.iter().rev() {
                    while upper.len() >= 2 {
                        let n = upper.len();
                        let turn = cross(upper[n - 2].0, upper[n - 1].0, item.0);
                        if if mode == Collinear::Include {
                            turn < 0
                        } else {
                            turn <= 0
                        } {
                            upper.pop();
                        } else {
                            break;
                        }
                    }
                    upper.push(item);
                }
                lower.pop();
                upper.pop();
                lower.extend(upper);
                lower
            }
        }
    };

    let vertices: Vec<Point> = vertices_with_sources.iter().map(|x| x.0).collect();
    let source_indices: Vec<usize> = vertices_with_sources.drain(..).map(|x| x.1).collect();
    let area = if vertices.len() >= 3 {
        signed_area2(&vertices).unsigned_abs()
    } else {
        0
    };
    let perimeter = perimeter_milli(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2: area,
        perimeter_milli: perimeter,
    })
}
