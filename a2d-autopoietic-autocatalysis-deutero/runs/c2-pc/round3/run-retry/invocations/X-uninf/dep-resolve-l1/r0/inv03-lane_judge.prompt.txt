    constraints.sort();

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(x) => version == x,
        Constraint::AtLeast(x) => version >= x,
        Constraint::Before(x) => version < x,
        Constraint::Caret(x) => {
            version >= x

                } else if x.minor > 0 {
                    version.major == 0 && version.minor == x.minor
                } else {
                    version == x

        }
    }

fn validate(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut names = std::collections::BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate(catalog, roots)?;

    let catalog_by_name: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();

    loop {
        for constraints in active.values_mut() {
            *constraints = canonical(std::mem::take(constraints));
        }

        if let Some(package) = active
            .keys()
            .filter(|name| !catalog_by_name.contains_key(name.as_str()))
            .next()
        {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }

        for (package, version) in &selected {
            let constraints = active.get(package).cloned().unwrap_or_default();
            if !constraints.iter().all(|c| matches(*version, *c)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: canonical(constraints),
                });
            }
        }

        for (package, constraints) in &active {
            if selected.contains_key(package) {
                continue;
            }
            let data = catalog_by_name[package.as_str()];
            if !data
                .releases
                .iter()
                .any(|r| constraints.iter().all(|c| matches(r.version, *c)))
            {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical(constraints.clone()),
                });
            }
        }

        let next = active
            .keys()
            .find(|package| !selected.contains_key(*package))
            .cloned();

        let Some(package) = next else {
            return Ok(selected
                .into_iter()
                .map(|(package, version)| Selection { package, version })
                .collect());
        };

        let constraints = active.get(&package).cloned().unwrap_or_default();
        let release = catalog_by_name[package.as_str()]
            .releases
            .iter()
            .filter(|r| constraints.iter().all(|c| matches(r.version, *c)))
            .max_by_key(|r| r.version)
            .expect("candidate existence checked");

        selected.insert(package, release.version);
        for dep in &release.deps {
            active
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort_by(|a, b| {
        constraint_rank(a)
            .cmp(&constraint_rank(b))
            .then_with(|| match (a, b) {
                (Constraint::Exact(x), Constraint::Exact(y))
                | (Constraint::AtLeast(x), Constraint::AtLeast(y))
                | (Constraint::Before(x), Constraint::Before(y))
                | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(y),
                _ => std::cmp::Ordering::Equal,
            })
    });
    constraints.dedup();
}

    }

    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();

    loop {
        for constraints in active.values_mut() {
            canonicalize(constraints);
        }

        if let Some(package) = active
            .keys()
            .filter(|name| !index.contains_key(name.as_str()))
            .next()
        {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }

        for (package, version) in &selected {
            if let Some(constraints) = active.get(package) {
                if constraints.iter().any(|c| !matches(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (package, constraints) in &active {
            if selected.contains_key(package) {
                continue;
            }
            let releases = &index[package.as_str()].releases;
            if !releases
                .iter()
                .any(|release| constraints.iter().all(|c| matches(release.version, *c)))
            {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active
            .keys()
            .find(|package| !selected.contains_key(*package))
            .cloned();
        let Some(package) = next else {
            return Ok(selected
                .into_iter()
                .map(|(package, version)| Selection { package, version })
                .collect());
        };

        let release = index[package.as_str()]
            .releases
            .iter()
            .filter(|release| {
                active[&package]
                    .iter()
                    .all(|c| matches(release.version, *c))
            })
            .max_by_key(|release| release.version)
            .expect("candidate existence was checked");

        selected.insert(package, release.version);
        for dep in &release.deps {
            active
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes[bytes.len() - 1] == b'-' {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(constraints: &[Constraint]) -> Vec<Constraint> {
    let mut result = constraints.to_vec();
    result.sort();
    result.dedup();
    result
}

fn permits(constraint: Constraint, version: Version) -> bool {
    match constraint {
        Constraint::Exact(expected) => version == expected,
        Constraint::AtLeast(minimum) => version >= minimum,
        Constraint::Before(bound) => version < bound,
        Constraint::Caret(base) => {
            version >= base
                && if base.major > 0 {
                    version.major == base.major
                } else if base.minor > 0 {
                    version.major == 0 && version.minor == base.minor
                } else {
                    version == base
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .copied()
        .all(|constraint| permits(constraint, version))
}

fn encode_text(text: &str) -> String {
    let mut encoded = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {
            encoded.push(*byte as char);
        } else {
            encoded.push('%');
            encoded.push(char::from_digit((byte >> 4) as u32, 16).unwrap().to_ascii_uppercase());
            encoded.push(char::from_digit((byte & 0x0f) as u32, 16).unwrap().to_ascii_uppercase());
        }
    }
    encoded
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => {
            let constraints = canonical_constraints(requirements)
                .iter()
                .copied()
                .map(constraint_text)
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "no-match|package={}|constraints={}",
                encode_text(package),
                constraints
            )
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => {
            let constraints = canonical_constraints(requirements)
                .iter()
                .copied()
                .map(constraint_text)
                .collect::<Vec<_>>()
                .join(",");
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode_text(package),
                version_text(*selected),
                constraints
            )
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let catalog_by_name: std::collections::BTreeMap<&str, &Package> = catalog
        .iter()
        .map(|package| (package.name.as_str(), package))
        .collect();

    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();

    loop {
        if let Some(package) = active
            .keys()
            .filter(|name| !catalog_by_name.contains_key(name.as_str()))
            .next()
        {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }

        for (package, version) in &selected {
            if let Some(constraints) = active.get(package) {
                if !matches_all(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: canonical_constraints(constraints),
                    });
                }
            }
        }

        for (package, constraints) in &active {
            if selected.contains_key(package) {
                continue;
            }
            let data = catalog_by_name[package.as_str()];
            let has_match = data
                .releases
                .iter()
                .any(|release| matches_all(release.version, constraints));
            if !has_match {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: canonical_constraints(constraints),
                });
            }
        }

        let next_package = active
            .keys()
            .find(|package| !selected.contains_key(*package))
            .cloned();

        let Some(package) = next_package else {
            return Ok(selected
                .into_iter()
                .map(|(package, version)| Selection { package, version })
                .collect());
        };

        let constraints = &active[&package];
        let data = catalog_by_name[package.as_str()];
        let version = data
            .releases
            .iter()
            .filter(|release| matches_all(release.version, constraints))
            .map(|release| release.version)
            .max()
            .unwrap();

        selected.insert(package.clone(), version);

        let release = data
            .releases
            .iter()
            .find(|release| release.version == version)
            .unwrap();
        for dep in &release.deps {
            active
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }
}
