#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    paused_at: Option<u64>,
}

struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(at), Some(bt)) => at.cmp(&bt).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut finished_count = 0usize;

    while finished_count < customers.len() {
        let mut next_time = arrivals.get(next_arrival).map(|c| c.arrival);

        if let Some(active) = &running {
            for t in [Some(active.finish), Some(active.work_at)] .into_iter().flatten() {
                if next_time.map_or(true, |n| t < n) {
                    next_time = Some(t);
                }
            }
        }

        let time = next_time.expect("unfinished customers imply a pending event");

        if running.as_ref().is_some_and(|active| active.finish == time) {
            let active = running.take().unwrap();
            events.push(Event {
                time,
                customer: active.customer.id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                paused_at: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = &running {
            if time >= active.work_at {
                waiting.sort_by(waiting_cmp);
                if waiting
                    .first()
                    .is_some_and(|waiter| waiter.customer.priority < active.customer.priority)
                {
                    let by = waiting[0].customer.id;
                    let active = running.take().unwrap();
                    events.push(Event {
                        time,
                        customer: active.customer.id,
                        kind: Kind::Preempt {
                            remaining: active.finish - time,
                            by,
                        },
                    });
                    waiting.push(Waiting {
                        customer: active.customer,
                        remaining: active.finish - time,
                        started: true,
                        paused_at: Some(time),
                    });
                }
            }
        }

        if running.is_none() {
            while !waiting.is_empty() {
                waiting.sort_by(waiting_cmp);
                let selected = waiting.remove(0);
                let cost = if last_dispatched == Some(selected.customer.id) {
                    0
                } else if last_dispatched.is_none() {
                    0
                } else {
                    switch_cost
                };

                let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: selected.customer.id,
                    time,
                })?;
                let finish = work_at.checked_add(selected.remaining).ok_or(Error::TimeOverflow {
                    customer: selected.customer.id,
                    time,
                })?;

                let kind = if selected.started {
                    Kind::Resume {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                };
                events.push(Event {
                    time,
                    customer: selected.customer.id,
                    kind,
                });
                last_dispatched = Some(selected.customer.id);
                running = Some(Running {
                    customer: selected.customer,
                    remaining: selected.remaining,
                    work_at,
                    finish,
                });

                if finish == time {
                    let active = running.take().unwrap();
                    events.push(Event {
                        time,
                        customer: active.customer.id,
                        kind: Kind::Finish,
                    });
                    finished_count += 1;
                    continue;
                }
                break;
            }
        }
    }

    Ok(events)
}
