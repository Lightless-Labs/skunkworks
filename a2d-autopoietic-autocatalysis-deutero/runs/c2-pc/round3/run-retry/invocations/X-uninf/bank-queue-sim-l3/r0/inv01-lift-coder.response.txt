#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: Option<u64>,
}

struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(ta), Some(tb)) => ta.cmp(&tb).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut now = arrivals[0].arrival;

    loop {
        if let Some(active) = running.as_ref() {
            if active.finish == now {
                let finished = running.take().unwrap();
                trace.push(Event {
                    time: now,
                    customer: finished.customer.id,
                    kind: Kind::Finish,
                });
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            trace.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if now >= active.work_at {
                waiting.sort_by(waiting_cmp);
                if let Some(best) = waiting.first() {
                    if best.customer.priority < active.customer.priority {
                        let by = best.customer.id;
                        let active = running.take().unwrap();
                        let remaining = active.finish - now;
                        trace.push(Event {
                            time: now,
                            customer: active.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        waiting.push(Waiting {
                            customer: active.customer,
                            remaining,
                            started: true,
                            pause_time: Some(now),
                        });
                    }
                }
            }
        }

        while running.is_none() && !waiting.is_empty() {
            waiting.sort_by(waiting_cmp);
            let selected = waiting.remove(0);
            let customer = selected.customer;
            let remaining = selected.remaining;
            let cost = if last_dispatched == Some(customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };

            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: customer.id,
                time: now,
            })?;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: customer.id,
                time: now,
            })?;

            last_dispatched = Some(customer.id);
            let kind = if selected.started {
                Kind::Resume {
                    remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining,
                    work_at,
                    finish,
                }
            };
            trace.push(Event {
                time: now,
                customer: customer.id,
                kind,
            });
            running = Some(Running {
                customer: customer.clone(),
                remaining,
                work_at,
                finish,
            });

            if finish == now {
                running = None;
                trace.push(Event {
                    time: now,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                continue;
            }
        }

        if next_arrival == arrivals.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time = arrivals.get(next_arrival).map(|c| c.arrival);
        if let Some(active) = running.as_ref() {
            for candidate in [Some(active.finish), Some(active.work_at)] .into_iter().flatten() {
                if candidate > now && next_time.map_or(true, |t| candidate < t) {
                    next_time = Some(candidate);
                }
            }
        }

        now = match next_time {
            Some(time) => time,
            None => break,
        };
    }

    Ok(trace)
}
