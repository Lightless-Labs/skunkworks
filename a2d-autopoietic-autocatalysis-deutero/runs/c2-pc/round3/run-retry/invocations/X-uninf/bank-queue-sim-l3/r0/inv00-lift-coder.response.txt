#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: Option<u64>,
}

struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn waiter_is_better(a: &Job, b: &Job) -> bool {
    let pa = a.customer.priority;
    let pb = b.customer.priority;
    if pa != pb {
        return pa < pb;
    }

    match (a.pause_time, b.pause_time) {
        (Some(ta), Some(tb)) => {
            ta < tb || (ta == tb && a.customer.id < b.customer.id)
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            let aa = a.customer.arrival;
            let ab = b.customer.arrival;
            aa < ab || (aa == ab && a.customer.id > b.customer.id)
        }
    }
}

fn best_waiter_index(waiting: &[Job]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.is_none_or(|j| waiter_is_better(&waiting[i], &waiting[j])) {
            best = Some(i);
        }
    }
    best
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut seen = std::collections::HashSet::new();

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut arrival_index = 0usize;
    let mut time = arrivals[0].arrival;

    loop {
        if let Some(active) = running.as_ref() {
            if active.finish == time {
                let finished = running.take().unwrap();
                trace.push(Event {
                    time,
                    customer: finished.job.customer.id,
                    kind: Kind::Finish,
                });
            }
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == time {
            let customer = arrivals[arrival_index].clone();
            trace.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            arrival_index += 1;
        }

        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                if let Some(best_index) = best_waiter_index(&waiting) {
                    let best_priority = waiting[best_index].customer.priority;
                    if best_priority < active.job.customer.priority {
                        let by = waiting[best_index].customer.id;
                        let mut paused = running.take().unwrap().job;
                        paused.remaining = active.finish - time;
                        paused.pause_time = Some(time);
                        trace.push(Event {
                            time,
                            customer: paused.customer.id,
                            kind: Kind::Preempt {
                                remaining: paused.remaining,
                                by,
                            },
                        });
                        waiting.push(paused);
                    }
                }
            }
        }

        if running.is_none() {
            while let Some(index) = best_waiter_index(&waiting) {
                let mut job = waiting.remove(index);
                let cost = if last_dispatched.is_some_and(|id| id != job.customer.id) {
                    switch_cost
                } else {
                    0
                };

                let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: job.customer.id,
                    time,
                })?;
                let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                    customer: job.customer.id,
                    time,
                })?;

                let kind = if job.started {
                    Kind::Resume {
                        remaining: job.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    job.started = true;
                    Kind::Start {
                        remaining: job.remaining,
                        work_at,
                        finish,
                    }
                };
                last_dispatched = Some(job.customer.id);
                trace.push(Event {
                    time,
                    customer: job.customer.id,
                    kind,
                });

                if finish == time {
                    trace.push(Event {
                        time,
                        customer: job.customer.id,
                        kind: Kind::Finish,
                    });
                    continue;
                }

                running = Some(Running {
                    job,
                    work_at,
                    finish,
                });
                break;
            }
        }

        if arrival_index == arrivals.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut next = None;
        if arrival_index < arrivals.len() {
            next = Some(arrivals[arrival_index].arrival);
        }
        if let Some(active) = running.as_ref() {
            if active.finish > time {
                next = Some(next.map_or(active.finish, |n| n.min(active.finish)));
            }
            if active.work_at > time {
                next = Some(next.map_or(active.work_at, |n| n.min(active.work_at)));
            }
        }

        match next {
            Some(next_time) => time = next_time,
            None => break,
        }
    }

    Ok(trace)
}
