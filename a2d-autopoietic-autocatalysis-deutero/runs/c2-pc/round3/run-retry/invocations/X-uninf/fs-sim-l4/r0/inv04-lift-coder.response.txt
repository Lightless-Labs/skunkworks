```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|c| !c.is_empty() && *c != ".")
        .fold(Vec::new(), |mut out, c| {
            if c == ".." {
                out.pop();
            } else {
                out.push(c.to_string());
            }
            out
        })
}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn path_parts(path: &str) -> Vec<String> {
    components(path)
}

fn parent(path: &str) -> String {
    let mut p = path_parts(path);
    p.pop();
    canonical(&p)
}

fn is_within(path: &str, dir: &str) -> bool {
    path == dir || dir == "/" || path.strip_prefix(dir).is_some_and(|s| s.starts_with('/'))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        validate(path)?;
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            path_parts(&self.cwd)
        };
        for c in path.split('/') {
            if c.is_empty() || c == "." {
                continue;
            }
            if c == ".." {
                parts.pop();
            } else {
                parts.push(c.to_string());
            }
        }
        Ok(canonical(&parts))
    }

    fn entry_kind(entry: &Entry) -> Kind {
        match entry {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let mut pending = path_parts(path);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(canonical(&resolved));
            }

            let part = pending.remove(0);
            resolved.push(part);
            let current = canonical(&resolved);
            let is_final = pending.is_empty();
            let entry = self.entries.get(&current);

            if entry.is_none() {
                if is_final && create_final {
                    let p = canonical(&resolved[..resolved.len() - 1]);
                    match self.entries.get(&p) {
                        None => return Err(Error::Missing(p)),
                        Some(Entry::Dir) => return Ok(current),
                        Some(_) => return Err(Error::NotDir(p)),
                    }
                }
                return Err(Error::Missing(current));
            }

            let entry = entry.unwrap();
            if let Entry::Link(target) = entry {
                if !is_final || follow_final {
                    if !visited.insert(current.clone()) {
                        return Err(Error::Loop(current));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(current));
                    }
                    expansions += 1;

                    let mut base = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        resolved[..resolved.len() - 1].to_vec()
                    };
                    for c in target.split('/') {
                        if c.is_empty() || c == "." {
                            continue;
                        }
                        if c == ".." {
                            base.pop();
                        } else {
                            base.push(c.to_string());
                        }
                    }
                    let mut combined = base;
                    combined.extend(pending.into_iter());
                    pending = combined;
                    resolved.clear();
                    continue;
                }
            }

            if !is_final && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(current));
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parts = path_parts(path);
        if parts.is_empty() {
            return Ok(("/".to_string(), "/".to_string()));
        }
        let name = parts.last().unwrap().clone();
        let p = canonical(&parts[..parts.len() - 1]);
        let rp = self.resolve(&p, true, false)?;
        match self.entries.get(&rp) {
            Some(Entry::Dir) => Ok((rp, name)),
            Some(_) => Err(Error::NotDir(rp)),
            None => Err(Error::Missing(rp)),
        }
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries.keys().any(|p| p != path && is_within(p, path))
    }

    fn descendants(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|p| is_within(p, path))
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let p = self.normalize(path)?;
        if p == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (parent, name) = self.resolve_parent(&p)?;
        let dest = if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        };
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let p = self.normalize(path)?;
        let resolved = self.resolve(&p, true, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let par = parent(&resolved);
                match self.entries.get(&par) {
                    Some(Entry::Dir) => {
                        self.entries.insert(resolved, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(par)),
                    None => Err(Error::Missing(par)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let p = self.normalize(path)?;
        let r = self.resolve(&p, true, false)?;
        match self.entries.get(&r) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(r)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(r)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let p = self.normalize(path)?;
        let r = self.resolve(&p, true, false)?;
        match self.entries.get(&r) {
            Some(Entry::Dir) => {
                let prefix = if r == "/" {
                    "/".to_string()
                } else {
                    format!("{}/", r)
                };
                let mut out = self
                    .entries
                    .iter()
                    .filter_map(|(k, v)| {
                        if k == &r || !k.starts_with(&prefix) {
                            return None;
                        }
                        let rest = &k[prefix.len()..];
                        if rest.contains('/') {
                            return None;
                        }
                        Some((rest.to_string(), Self::entry_kind(v)))
                    })
                    .collect::<Vec<_>>();
                out.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(out)
            }
            _ => Err(Error::NotDir(r)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let p = self.normalize(path)?;
        let r = self.resolve(&p, true, false)?;
        match self.entries.get(&r) {
            Some(Entry::Dir) => {
                self.cwd = r;
                Ok(())
            }
            _ => Err(Error::NotDir(r)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let f = self.normalize(from)?;
        let t = self.normalize(to)?;
        if f == "/" || t == "/" {
            return Err(Error::Root);
        }
        let (fp, fnm) = self.resolve_parent(&f)?;
        let src = if fp == "/" {
            format!("/{}", fnm)
        } else {
            format!("{}/{}", fp, fnm)
        };
        let source = self
            .entries
            .get(&src)
            .cloned()
            .ok_or_else(|| Error::Missing(src.clone()))?;

        if matches!(source, Entry::Dir) && is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (tp, tnm) = self.resolve_parent(&t)?;
        let dst = if tp == "/" {
            format!("/{}", tnm)
        } else {
            format!("{}/{}", tp, tnm)
        };

        if self
            .entries
            .get(&dst)
            .is_some_and(|e| matches!(e, Entry::Dir))
            && is_within(&self.cwd, &dst)
        {
            return Err(Error::Busy(dst));
        }
        if src == dst {
            return Ok(());
        }
        if matches!(source, Entry::Dir) && is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        if self.entries.contains_key(&dst) {
            if !overwrite {
                return Err(Error::Exists(dst));
            }
            let dest_entry = self.entries.get(&dst).unwrap();
            match (&source, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&dst) {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let subtree = self.descendants(&src);
        let mut moved = Vec::new();
        for old in &subtree {
            let suffix = &old[src.len()..];
            moved.push((
                old.clone(),
                format!("{}{}", dst, suffix),
                self.entries[old].clone(),
            ));
        }
        if self.entries.contains_key(&dst) {
            for old in self.descendants(&dst) {
                self.entries.remove(&old);
            }
        }
        for (old, _, _) in &moved {
            self.entries.remove(old);
        }
        for (_, new, entry) in moved {
            self.entries.insert(new, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let p = self.normalize(path)?;
        if p == "/" {
            return Err(Error::Root);
        }
        let (par, name) = self.resolve_parent(&p)?;
        let target = if par == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", par, name)
        };
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;
        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && self.has_children(&target) && !recursive {
            return Err(Error::NotEmpty(target));
        }
        for p in self.descendants(&target) {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let l = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        if l == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (par, name) = self.resolve_parent(&l)?;
        let dest = if par == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", par, name)
        };
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let p = self.normalize(path)?;
        if p == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (par, name) = self.resolve_parent(&p)?;
        let dest = if par == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", par, name)
        };
        match self.entries.get(&dest) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(dest)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let p = self.normalize(path)?;
        let r = self.resolve(&p, true, false)?;
        let entry = self
            .entries
            .get(&r)
            .ok_or_else(|| Error::Missing(r.clone()))?;
        let paths = if matches!(entry, Entry::Dir) {
            self.descendants(&r)
        } else {
            vec![r.clone()]
        };
        let mut rows = Vec::new();
        for name in paths {
            let size = match self.entries.get(&name).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.as_bytes().len() as u64,
                Entry::Dir => self
                    .descendants(&name)
                    .iter()
                    .map(|p| match self.entries.get(p).unwrap() {
                        Entry::File(data) => data.len() as u64,
                        Entry::Link(target) => target.as_bytes().len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
            };
            rows.push((name, size));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}
```
