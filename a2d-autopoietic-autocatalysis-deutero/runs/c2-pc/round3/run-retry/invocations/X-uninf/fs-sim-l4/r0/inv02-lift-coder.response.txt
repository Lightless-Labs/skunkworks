```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

#[derive(Debug, Clone)]
struct Lookup {
    path: String,
    entry: Entry,
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_string).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize_from(path: &str, base: &[String]) -> Vec<String> {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            base.to_vec()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_string()),
            }
        }
        parts
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let base = Self::components(&self.cwd);
        Ok(Self::canonical(&Self::normalize_from(path, &base)))
    }

    fn parent(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_string(), String::new());
        }
        let parts = Self::components(path);
        let name = parts.last().cloned().unwrap_or_default();
        let parent = Self::canonical(&parts[..parts.len() - 1]);
        (parent, name)
    }

    fn join_child(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            path.starts_with('/')
        } else {
            path == dir || path.strip_prefix(dir).is_some_and(|s| s.starts_with('/'))
        }
    }

    fn direct_children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                if *candidate == path || !Self::is_within(candidate, path) {
                    return false;
                }
                let (parent, _) = Self::parent(candidate);
                parent == path
            })
            .cloned()
            .collect()
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        create_missing_final: bool,
    ) -> Result<Lookup, Error> {
        let mut pending = Self::components(path);
        let mut physical = Vec::<String>::new();
        let mut visited = HashSet::<String>::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                let canonical = Self::canonical(&physical);
                let entry = self
                    .entries
                    .get(&canonical)
                    .cloned()
                    .ok_or_else(|| Error::Missing(canonical.clone()))?;
                return Ok(Lookup {
                    path: canonical,
                    entry,
                });
            }

            let component = pending.remove(0);
            let candidate_parts = {
                let mut p = physical.clone();
                p.push(component);
                p
            };
            let candidate = Self::canonical(&candidate_parts);
            let is_final = pending.is_empty();

            let parent_path = Self::canonical(&physical);
            match self.entries.get(&parent_path) {
                None => return Err(Error::Missing(parent_path)),
                Some(Entry::Dir) => {}
                Some(_) => return Err(Error::NotDir(parent_path)),
            }

            let entry = match self.entries.get(&candidate) {
                Some(entry) => entry.clone(),
                None if is_final && create_missing_final => {
                    return Ok(Lookup {
                        path: candidate,
                        entry: Entry::File(Vec::new()),
                    });
                }
                None => return Err(Error::Missing(candidate)),
            };

            if let Entry::Link(target) = &entry {
                if !is_final || follow_final {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let mut target_parts = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        physical.clone()
                    };
                    for part in target.split('/') {
                        match part {
                            "" | "." => {}
                            ".." => {
                                target_parts.pop();
                            }
                            other => target_parts.push(other.to_string()),
                        }
                    }

                    let mut rest = Vec::with_capacity(target_parts.len() + pending.len());
                    rest.extend(target_parts);
                    rest.extend(pending.drain(..));
                    let normalized = Self::normalize_from(
                        &format!("/{}", rest.join("/")),
                        &[],
                    );
                    pending = normalized;
                    physical.clear();
                    continue;
                }
            }

            if is_final {
                return Ok(Lookup {
                    path: candidate,
                    entry,
                });
            }

            match entry {
                Entry::Dir => physical.push(component),
                _ => return Err(Error::NotDir(candidate)),
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent(path);
        let resolved = self.resolve(&parent, true, false)?;
        match resolved.entry {
            Entry::Dir => Ok((resolved.path, name)),
            _ => Err(Error::NotDir(resolved.path)),
        }
    }

    fn remove_subtree(&mut self, path: &str) {
        let keys: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| Self::is_within(candidate, path))
            .cloned()
            .collect();
        for key in keys {
            self.entries.remove(&key);
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = Self::parent(&canonical);
        let resolved_parent = self.resolve(&parent, true, false)?;
        if !matches!(resolved_parent.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved_parent.path));
        }
        let destination = Self::join_child(&resolved_parent.path, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = match self.resolve(&canonical, true, true) {
            Ok(found) => found,
            Err(Error::Missing(missing)) => {
                let (parent, name) = Self::parent(&missing);
                let resolved_parent = self.resolve(&parent, true, false)?;
                if !matches!(resolved_parent.entry, Entry::Dir) {
                    return Err(Error::NotDir(resolved_parent.path));
                }
                let destination = Self::join_child(&resolved_parent.path, &name);
                self.entries
                    .insert(destination.clone(), Entry::File(data.to_vec()));
                return Ok(());
            }
            Err(err) => return Err(err),
        };

        match resolved.entry {
            Entry::Dir => Err(Error::IsDir(resolved.path)),
            Entry::File(_) => {
                self.entries
                    .insert(resolved.path, Entry::File(data.to_vec()));
                Ok(())
            }
            Entry::Link(_) => {
                let (parent, name) = Self::parent(&resolved.path);
                let resolved_parent = self.resolve(&parent, true, false)?;
                if !matches!(resolved_parent.entry, Entry::Dir) {
                    return Err(Error::NotDir(resolved_parent.path));
                }
                let destination = Self::join_child(&resolved_parent.path, &name);
                self.entries
                    .insert(destination, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        match resolved.entry {
            Entry::File(bytes) => Ok(bytes),
            Entry::Dir => Err(Error::IsDir(resolved.path)),
            Entry::Link(_) => Err(Error::Missing(resolved.path)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        if !matches!(resolved.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved.path));
        }

        let mut result = Vec::new();
        for child in self.direct_children(&resolved.path) {
            let (_, name) = Self::parent_and_name(&child);
            let kind = match self.entries.get(&child) {
                Some(Entry::Dir) => Kind::Dir,
                Some(Entry::File(_)) => Kind::File,
                Some(Entry::Link(_)) => Kind::Link,
                None => continue,
            };
            result.push((name, kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    fn parent_and_name(path: &str) -> (String, String) {
        Self::parent(path)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        match resolved.entry {
            Entry::Dir => {
                self.cwd = resolved.path;
                Ok(())
            }
            _ => Err(Error::NotDir(resolved.path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = Self::parent(&source);
        let resolved_source_parent = self.resolve(&source_parent, true, false)?;
        if !matches!(resolved_source_parent.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved_source_parent.path));
        }
        let source_path = Self::join_child(&resolved_source_parent.path, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = Self::parent(&destination);
        let resolved_destination_parent = self.resolve(&destination_parent, true, false)?;
        if !matches!(resolved_destination_parent.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved_destination_parent.path));
        }
        let destination_path =
            Self::join_child(&resolved_destination_parent.path, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(ref dest_entry) = destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.direct_children(&destination_path).len() != 0 {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_within(path, &source_path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if destination_entry.is_some() {
            self.remove_subtree(&destination_path);
        }
        self.remove_subtree(&source_path);

        for (old_path, entry) in moved {
            let suffix = if old_path == source_path {
                String::new()
            } else {
                old_path[source_path.len()..].to_string()
            };
            let new_path = format!("{destination_path}{suffix}");
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = Self::parent(&canonical);
        let resolved_parent = self.resolve(&parent, true, false)?;
        if !matches!(resolved_parent.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved_parent.path));
        }
        let target = Self::join_child(&resolved_parent.path, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }

        if matches!(entry, Entry::Dir) && !recursive && !self.direct_children(&target).is_empty() {
            return Err(Error::NotEmpty(target));
        }

        self.remove_subtree(&target);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical = self.normalize(link)?;
        Self::validate_target(target)?;

        let (parent, name) = Self::parent(&canonical);
        let resolved_parent = self.resolve(&parent, true, false)?;
        if !matches!(resolved_parent.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved_parent.path));
        }
        let destination = Self::join_child(&resolved_parent.path, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_string()));
        Ok(())
    }

    fn validate_target(target: &str) -> Result<(), Error> {
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, false, false)?;
        match resolved.entry {
            Entry::Link(target) => Ok(target),
            _ => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        let mut paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &resolved.path))
            .cloned()
            .collect();
        paths.sort();

        let mut sizes = BTreeMap::<String, u64>::new();
        for entry_path in paths.iter().rev() {
            let own_size = match self.entries.get(entry_path) {
                Some(Entry::File(bytes)) => bytes.len() as u64,
                Some(Entry::Link(target)) => target.len() as u64,
                _ => 0,
            };
            let descendant_sum: u64 = paths
                .iter()
                .filter(|p| {
                    *p != entry_path
                        && Self::is_within(p, entry_path)
                        && Self::parent(p).0 == *entry_path
                })
                .filter_map(|p| sizes.get(p))
                .copied()
                .sum();
            sizes.insert(entry_path.clone(), own_size + descendant_sum);
        }

        let mut result: Vec<(String, u64)> = paths
            .into_iter()
            .map(|entry_path| {
                let size = sizes.get(&entry_path).copied().unwrap_or(0);
                (entry_path, size)
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }
}
```
