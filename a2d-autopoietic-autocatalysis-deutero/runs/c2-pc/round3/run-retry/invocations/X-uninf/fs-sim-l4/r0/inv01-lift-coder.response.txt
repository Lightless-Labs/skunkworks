use std::collections::{BTreeMap, BTreeSet, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

#[derive(Clone)]
struct Resolver {
    visited: HashSet<String>,
    expansions: usize,
}

impl Resolver {
    fn new() -> Self {
        Self {
            visited: HashSet::new(),
            expansions: 0,
        }
    }
}

fn validate(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty() && *part != ".")
        .fold(Vec::new(), |mut acc, part| {
            if part == ".." {
                acc.pop();
            } else {
                acc.push(part.to_owned());
            }
            acc
        })
}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn path_components(path: &str) -> Vec<String> {
    if path == "/" {
        Vec::new()
    } else {
        path[1..].split('/').map(str::to_owned).collect()
    }
}

fn join_child(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn is_within(path: &str, dir: &str) -> bool {
    path == dir || if dir == "/" {
        path.starts_with('/')
    } else {
        path.strip_prefix(dir)
            .map(|rest| rest.starts_with('/'))
            .unwrap_or(false)
    }
}

fn parent_and_name(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_owned(), String::new());
    }
    let i = path.rfind('/').unwrap_or(0);
    let parent = if i == 0 { "/" } else { &path[..i] };
    (parent.to_owned(), path[i + 1..].to_owned())
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        validate(path)?;
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            path_components(&self.cwd)
        };
        for part in path.split('/') {
            if part.is_empty() || part == "." {
                continue;
            }
            if part == ".." {
                parts.pop();
            } else {
                parts.push(part.to_owned());
            }
        }
        Ok(canonical(&parts))
    }

    fn resolve_path(
        &self,
        path: &str,
        follow_final: bool,
        resolver: &mut Resolver,
    ) -> Result<String, Error> {
        let mut pending = path_components(path);
        let mut resolved: Vec<String> = Vec::new();

        while !pending.is_empty() {
            let part = pending.remove(0);
            resolved.push(part);
            let current = canonical(&resolved);
            let final_component = pending.is_empty();

            let entry = self
                .entries
                .get(&current)
                .ok_or_else(|| Error::Missing(current.clone()))?;

            if matches!(entry, Entry::Link(_)) && (!final_component || follow_final) {
                if !resolver.visited.insert(current.clone()) {
                    return Err(Error::Loop(current));
                }
                if resolver.expansions >= 8 {
                    return Err(Error::Depth(current));
                }
                resolver.expansions += 1;

                let target = match entry {
                    Entry::Link(target) => target,
                    _ => unreachable!(),
                };
                let mut target_parts = if target.starts_with('/') {
                    Vec::new()
                } else {
                    let link_parent = parent_and_name(&current).0;
                    path_components(&link_parent)
                };
                for c in target.split('/') {
                    if c.is_empty() || c == "." {
                        continue;
                    }
                    if c == ".." {
                        target_parts.pop();
                    } else {
                        target_parts.push(c.to_owned());
                    }
                }
                target_parts.extend(pending.into_iter());
                let expanded = canonical(&target_parts);
                return self.resolve_path(&expanded, follow_final, resolver);
            }

            if !final_component && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(current));
            }
        }

        let result = canonical(&resolved);
        if !self.entries.contains_key(&result) {
            return Err(Error::Missing(result));
        }
        Ok(result)
    }

    fn resolve_parent(
        &self,
        path: &str,
        resolver: &mut Resolver,
    ) -> Result<(String, String), Error> {
        let (parent, name) = parent_and_name(path);
        let physical_parent = self.resolve_path(&parent, true, resolver)?;
        if !matches!(self.entries.get(&physical_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(physical_parent));
        }
        Ok((physical_parent, name))
    }

    fn children(&self, dir: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter_map(|path| {
                if path == dir {
                    return None;
                }
                let (parent, _) = parent_and_name(path);
                (parent == dir).then(|| path.clone())
            })
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Exists("/".to_owned()));
        }
        let mut resolver = Resolver::new();
        let (parent, name) = self.resolve_parent(&path, &mut resolver)?;
        let destination = join_child(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir("/".to_owned()));
        }

        let (parent, name) = parent_and_name(&path);
        let mut resolver = Resolver::new();
        let physical_parent = self.resolve_path(&parent, true, &mut resolver)?;
        if !matches!(self.entries.get(&physical_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(physical_parent));
        }
        let candidate = join_child(&physical_parent, &name);

        let mut final_resolver = Resolver::new();
        let destination = match self.resolve_path(&candidate, true, &mut final_resolver) {
            Ok(resolved) => resolved,
            Err(Error::Missing(missing)) if missing == candidate => candidate,
            Err(err) => return Err(err),
        };

        if matches!(self.entries.get(&destination), Some(Entry::Dir)) {
            return Err(Error::IsDir(destination));
        }
        self.entries
            .insert(destination, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        let mut resolver = Resolver::new();
        let resolved = self.resolve_path(&path, true, &mut resolver)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        let mut resolver = Resolver::new();
        let resolved = self.resolve_path(&path, true, &mut resolver)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        let mut rows: Vec<_> = self
            .children(&resolved)
            .into_iter()
            .filter_map(|child| {
                let (_, name) = parent_and_name(&child);
                self.entries
                    .get(&child)
                    .map(|entry| (name, entry.kind()))
            })
            .collect();
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        let mut resolver = Resolver::new();
        let resolved = self.resolve_path(&path, true, &mut resolver)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = self.normalize(from)?;
        let to = self.normalize(to)?;
        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let mut source_parent_resolver = Resolver::new();
        let (source_parent, source_name) =
            self.resolve_parent(&from, &mut source_parent_resolver)?;
        let source = join_child(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let mut destination_parent_resolver = Resolver::new();
        let (destination_parent, destination_name) =
            self.resolve_parent(&to, &mut destination_parent_resolver)?;
        let destination = join_child(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref dest_entry) = destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| is_within(path, &source))
            .map(|(path, entry)| {
                let suffix = &path[source.len()..];
                (format!("{destination}{suffix}"), entry.clone())
            })
            .collect();

        if destination_entry.is_some() {
            let remove_dest: Vec<String> = self
                .entries
                .keys()
                .filter(|path| is_within(path, &destination))
                .cloned()
                .collect();
            for path in remove_dest {
                self.entries.remove(&path);
            }
        }

        let remove_source: Vec<String> = self
            .entries
            .keys()
            .filter(|path| is_within(path, &source))
            .cloned()
            .collect();
        for path in remove_source {
            self.entries.remove(&path);
        }

        for (path, entry) in moved {
            self.entries.insert(path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }
        let mut resolver = Resolver::new();
        let (parent, name) = self.resolve_parent(&path, &mut resolver)?;
        let target = join_child(&parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir)
            && !recursive
            && !self.children(&target).is_empty()
        {
            return Err(Error::NotEmpty(target));
        }

        let removals: Vec<String> = self
            .entries
            .keys()
            .filter(|p| is_within(p, &target))
            .cloned()
            .collect();
        for p in removals {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link = self.normalize(link)?;
        validate_target(target)?;
        if link == "/" {
            return Err(Error::Exists("/".to_owned()));
        }
        let mut resolver = Resolver::new();
        let (parent, name) = self.resolve_parent(&link, &mut resolver)?;
        let destination = join_child(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries
            .insert(destination, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let path = self.normalize(path)?;
        let mut resolver = Resolver::new();
        let (parent, name) = self.resolve_parent(&path, &mut resolver)?;
        let target = join_child(&parent, &name);
        match self.entries.get(&target) {
            Some(Entry::Link(value)) => Ok(value.clone()),
            Some(Entry::Dir) | Some(Entry::File(_)) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(target)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let path = self.normalize(path)?;
        let mut resolver = Resolver::new();
        let resolved = self.resolve_path(&path, true, &mut resolver)?;
        let mut rows = Vec::new();
        let mut sizes = BTreeMap::<String, u64>::new();

        let descendants: Vec<String> = self
            .entries
            .keys()
            .filter(|entry_path| is_within(entry_path, &resolved))
            .cloned()
            .collect();

        for entry_path in &descendants {
            let size = match self.entries.get(entry_path) {
                Some(Entry::File(data)) => data.len() as u64,
                Some(Entry::Link(target)) => target.len() as u64,
                _ => 0,
            };
            sizes.insert(entry_path.clone(), size);
        }

        for entry_path in descendants.iter().rev() {
            if matches!(self.entries.get(entry_path), Some(Entry::Dir)) {
                let total = descendants
                    .iter()
                    .filter(|candidate| {
                        if *candidate == entry_path {
                            return false;
                        }
                        let (parent, _) = parent_and_name(candidate);
                        is_within(&parent, entry_path)
                    })
                    .filter_map(|candidate| sizes.get(candidate))
                    .copied()
                    .sum();
                sizes.insert(entry_path.clone(), total);
            }
        }

        for entry_path in descendants {
            rows.push((entry_path.clone(), *sizes.get(&entry_path).unwrap_or(&0)));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}

fn validate_target(target: &str) -> Result<(), Error> {
    if target.is_empty() {
        return Err(Error::InvalidTarget(0));
    }
    if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidTarget(i));
    }
    Ok(())
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
