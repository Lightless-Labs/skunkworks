#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let complement = denominator - r;

    Some(if r > complement || (r == complement && q % 2 == 1) {
        q + 1
    } else {
        q
    })
}

#[derive(Clone)]
struct Waiter {
    index: usize,
    entry: u64,
    deadline: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiter_better(a: &Waiter, b: &Waiter, customers: &[Customer]) -> bool {
    let ca = &customers[a.index];
    let cb = &customers[b.index];
    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }
    if a.paused != b.paused {
        return a.paused;
    }
    if a.paused {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        ca.id < cb.id
    } else if ca.arrival != cb.arrival {
        ca.arrival < cb.arrival
    } else {
        ca.id > cb.id
    }
}

fn best_waiter(waiting: &[Waiter], customers: &[Customer]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.map_or(true, |b| waiter_better(&waiting[i], &waiting[b], customers)) {
            best = Some(i);
        }
    }
    best
}

fn emit(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn renege_waiter(
    waiter: Waiter,
    now: u64,
    customers: &[Customer],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let segment = now - waiter.entry;
    stats.wait_ticks += segment as u128;
    let waited = waiter.waited + segment;
    stats.reneged += 1;
    emit(events, now, customers[waiter.index].id, Kind::Renege { waited });
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    if customers.is_empty() {
        return Ok(Report { events, stats });
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a].arrival.cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut next_arrival = 0usize;
    let mut waiting: Vec<Waiter> = Vec::new();
    let mut active: Option<Active> = None;
    let mut dispatched_before = vec![false; customers.len()];
    let mut last_dispatched: Option<u32> = None;
    let mut now = customers[order[0]].arrival;

    loop {
        let mut did_phase_e = false;

        // Phase A: finish the active customer at this boundary.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let c = &customers[a.index];
            emit(&mut events, now, c.id, Kind::Finish);
            stats.completed += 1;
            stats.turnaround_ticks += (now - c.arrival) as u128;
            stats.completion_order.push(c.id);
        }

        // Phase B: expire those waiting at the start of the tick.
        let mut expired: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| (i, d)))
            .map(|(i, _)| i)
            .collect();
        expired.sort_by_key(|&i| (waiting[i].deadline.unwrap(), customers[waiting[i].index].id));
        for i in expired.into_iter().rev() {
            let w = waiting.remove(i);
            renege_waiter(w, now, customers, &mut events, &mut stats);
        }

        // Phase C: process the complete cohort without dispatching between arrivals.
        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let idx = order[next_arrival];
            next_arrival += 1;
            let c = &customers[idx];
            emit(&mut events, now, c.id, Kind::Arrive);
            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= limit {
                emit(&mut events, now, c.id, Kind::Balk);
                stats.balked += 1;
            } else {
                stats.accepted += 1;
                waiting.push(Waiter {
                    index: idx,
                    entry: now,
                    deadline: c.patience.map(|p| now + p),
                    waited: 0,
                    paused: false,
                    pause_time: 0,
                });
            }
        }

        // Phase D: an unprotected active customer may be preempted.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(wi) = best_waiter(&waiting, customers) {
                    let running_priority = customers[a.index].priority;
                    let waiting_priority = customers[waiting[wi].index].priority;
                    if waiting_priority < running_priority {
                        let a = active.take().unwrap();
                        let by = customers[waiting[wi].index].id;
                        let remaining = a.finish - now;
                        emit(&mut events, now, customers[a.index].id,
                             Kind::Preempt { remaining, by });
                        stats.preemptions += 1;
                        let c = &customers[a.index];
                        let deadline = c.patience.map(|budget| {
                            now.checked_add(budget).ok_or(Error::TimeOverflow {
                                customer: c.id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })
                        }).transpose()?;
                        waiting.push(Waiter {
                            index: a.index,
                            entry: now,
                            deadline,
                            waited: 0,
                            paused: true,
                            pause_time: now,
                        });
                    }
                }
            }
        }

        // Phase E: dispatch until occupied, including inline zero-time finishes.
        loop {
            if active.is_some() {
                break;
            }
            let Some(wi) = best_waiter(&waiting, customers) else { break };
            let w = waiting.remove(wi);
            let c = &customers[w.index];
            let segment = now - w.entry;
            stats.wait_ticks += segment as u128;
            let waited = w.waited + segment;
            let remaining = if dispatched_before[w.index] {
                // Remaining service is stored separately for paused customers below.
                // This branch is replaced by the retained value in service_left.
                service_left[w.index]
            } else {
                c.service
            };

            let switch = if last_dispatched.is_some_and(|id| id != c.id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: c.id, time: now, stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: c.id, time: now, stage: OverflowStage::Finish,
            })?;
            let kind = if dispatched_before[w.index] {
                Kind::Resume { remaining, work_at, finish }
            } else {
                Kind::Start { remaining, work_at, finish }
            };
            emit(&mut events, now, c.id, kind);
            if !dispatched_before[w.index] {
                dispatched_before[w.index] = true;
                stats.started += 1;
            }
            last_dispatched = Some(c.id);
            stats.busy_ticks += (finish - now) as u128;
            service_left[w.index] = remaining;
            active = Some(Active { index: w.index, remaining, work_at, finish });
            did_phase_e = true;

            if finish == now {
                active = None;
                emit(&mut events, now, c.id, Kind::Finish);
                stats.completed += 1;
                stats.turnaround_ticks += (now - c.arrival) as u128;
                stats.completion_order.push(c.id);
                continue;
            }
        }

        // Phase F: expire waiters newly queued in this tick.
        let mut expired: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, w)| w.deadline.filter(|&d| d <= now).map(|d| i))
            .collect();
        expired.sort_by_key(|&i| (waiting[i].deadline.unwrap(), customers[waiting[i].index].id));
        for i in expired.into_iter().rev() {
            let w = waiting.remove(i);
            renege_waiter(w, now, customers, &mut events, &mut stats);
        }

        if next_arrival == order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next = None;
        let mut consider = |t: u64| {
            if t > now && next.map_or(true, |n| t < n) {
                next = Some(t);
            }
        };
        if next_arrival < order.len() {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            if a.work_at > now {
                consider(a.work_at);
            }
        }
        for w in &waiting {
            if let Some(d) = w.deadline {
                consider(d);
            }
        }

        // If a zero-time event altered state but no future boundary exists, finish.
        let Some(next_time) = next else {
            if active.is_none() && waiting.is_empty() && next_arrival == order.len() {
                break;
            }
            // All live entities at u64::MAX are handled in the current tick.
            break;
        };
        now = next_time;
        let _ = did_phase_e;
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}
