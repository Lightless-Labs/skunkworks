#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let other_half = denominator - r;
    Some(if r > other_half || (r == other_half && q & 1 == 1) {
        q + 1
    } else {
        q
    })
}

#[derive(Clone)]
struct State {
    customer: Customer,
    deadline: Option<u64>,
    entry: u64,
    waited: u64,
    remaining: u64,
    ever_started: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    idx: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn best_waiter(waiting: &[usize], states: &[State]) -> Option<usize> {
    waiting.iter().copied().min_by(|&a, &b| {
        let x = &states[a];
        let y = &states[b];
        x.customer
            .priority
            .cmp(&y.customer.priority)
            .then_with(|| match (x.ever_started, y.ever_started) {
                (true, false) => std::cmp::Ordering::Less,
                (false, true) => std::cmp::Ordering::Greater,
                (true, true) => x
                    .pause_time
                    .cmp(&y.pause_time)
                    .then_with(|| x.customer.id.cmp(&y.customer.id)),
                (false, false) => x
                    .customer
                    .arrival
                    .cmp(&y.customer.arrival)
                    .then_with(|| y.customer.id.cmp(&x.customer.id)),
            })
    })
}

fn expire_waiters(
    now: u64,
    waiting: &mut Vec<usize>,
    states: &mut [State],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<(usize, u64)> = waiting
        .iter()
        .copied()
        .filter_map(|i| states[i].deadline.filter(|&d| d <= now).map(|d| (i, d)))
        .collect();
    expired.sort_by(|(a, da), (b, db)| {
        da.cmp(db)
            .then_with(|| states[*a].customer.id.cmp(&states[*b].customer.id))
    });
    for (idx, _) in expired {
        if let Some(pos) = waiting.iter().position(|&i| i == idx) {
            waiting.remove(pos);
            let s = &mut states[idx];
            let segment = now - s.entry;
            s.waited += segment;
            stats.wait_ticks += segment as u128;
            stats.reneged += 1;
            emit(
                events,
                now,
                s.customer.id,
                Kind::Renege { waited: s.waited },
            );
        }
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index, id: c.id });
            }
        }
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<State> = customers
        .iter()
        .cloned()
        .map(|customer| State {
            deadline: customer
                .patience
                .and_then(|p| customer.arrival.checked_add(p)),
            remaining: customer.service,
            entry: customer.arrival,
            waited: 0,
            ever_started: false,
            pause_time: 0,
            customer,
        })
        .collect();

    let mut events = Vec::new();
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut last_dispatched: Option<u32> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut first_dispatch = true;
    let mut protected_until: Option<u64> = None;

    while stats.completed + stats.reneged + stats.balked < customers.len() {
        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let idx = a.idx;
                let id = states[idx].customer.id;
                active = None;
                protected_until = None;
                stats.completed += 1;
                stats.turnaround_ticks += (now - states[idx].customer.arrival) as u128;
                stats.completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
            }
        }

        expire_waiters(now, &mut waiting, &mut states, &mut events, &mut stats);

        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let idx = order[next_arrival];
            next_arrival += 1;
            let c = &states[idx].customer;
            emit(&mut events, now, c.id, Kind::Arrive);
            let server_idle = active.is_none();
            let limit = config.waiting_capacity + usize::from(server_idle);
            if waiting.len() >= limit {
                stats.balked += 1;
                emit(&mut events, now, c.id, Kind::Balk);
            } else {
                stats.accepted += 1;
                states[idx].entry = now;
                waiting.push(idx);
            }
        }

        let protection_ended = active
            .as_ref()
            .map(|a| a.work_at <= now)
            .unwrap_or(true);
        if protection_ended {
            if let (Some(a), Some(best)) = (active.as_ref(), best_waiter(&waiting, &states)) {
                if states[best].customer.priority < states[a.idx].customer.priority {
                    let idx = a.idx;
                    let remaining = a.finish - now;
                    let by = states[best].customer.id;
                    let id = states[idx].customer.id;
                    emit(
                        &mut events,
                        now,
                        id,
                        Kind::Preempt { remaining, by },
                    );
                    stats.preemptions += 1;
                    stats.busy_ticks += (now - a.work_at.min(now)) as u128
                        + (a.work_at.saturating_sub(a.work_at)) as u128;
                    let mut old = active.take().unwrap();
                    stats.busy_ticks += (now - old_start(&old, now)) as u128;
                    let _ = &mut old;
                    let s = &mut states[idx];
                    s.remaining = remaining;
                    s.entry = now;
                    s.pause_time = now;
                    if let Some(budget) = s.customer.patience {
                        s.deadline = Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                            customer: id,
                            time: now,
                            stage: OverflowStage::Patience,
                        })?);
                    }
                    waiting.push(idx);
                    protected_until = None;
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let Some(idx) = best_waiter(&waiting, &states) else {
                break;
            };
            let pos = waiting.iter().position(|&i| i == idx).unwrap();
            waiting.remove(pos);
            let id = states[idx].customer.id;
            let remaining = states[idx].remaining;
            let waited = now - states[idx].entry;
            states[idx].waited += waited;
            stats.wait_ticks += waited as u128;

            let switch = if first_dispatch {
                0
            } else if last_dispatched == Some(id) {
                0
            } else {
                config.switch_cost
            };
            first_dispatch = false;
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::Finish,
            })?;
            let first = !states[idx].ever_started;
            if first {
                states[idx].ever_started = true;
                stats.started += 1;
                emit(
                    &mut events,
                    now,
                    id,
                    Kind::Start {
                        remaining,
                        work_at,
                        finish,
                    },
                );
            } else {
                emit(
                    &mut events,
                    now,
                    id,
                    Kind::Resume {
                        remaining,
                        work_at,
                        finish,
                    },
                );
            }
            last_dispatched = Some(id);
            active = Some(Active {
                idx,
                remaining,
                work_at,
                finish,
            });
            protected_until = Some(work_at);

            if finish == now {
                active = None;
                protected_until = None;
                stats.completed += 1;
                stats.turnaround_ticks += (now - states[idx].customer.arrival) as u128;
                stats.completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
                continue;
            }
            break;
        }

        expire_waiters(now, &mut waiting, &mut states, &mut events, &mut stats);

        if let Some(a) = active.as_ref() {
            let start = a.work_at.saturating_sub(if a.work_at == now { 0 } else { 0 });
            let _ = start;
        }

        if stats.completed + stats.reneged + stats.balked >= customers.len() {
            break;
        }

        let mut next = None;
        if next_arrival < order.len() {
            next = Some(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = active.as_ref() {
            next = Some(next.map_or(a.finish, |n| n.min(a.finish)));
            if a.work_at > now {
                next = Some(next.map_or(a.work_at, |n| n.min(a.work_at)));
            }
        }
        for &idx in &waiting {
            if let Some(d) = states[idx].deadline.filter(|&d| d > now) {
                next = Some(next.map_or(d, |n| n.min(d)));
            }
        }
        let Some(next_time) = next else {
            break;
        };
        if next_time == now {
            // The only same-time work is handled inline above; move to the next
            // strictly future point to avoid a zero-duration loop.
            let mut future = None;
            if next_arrival < order.len() {
                let t = customers[order[next_arrival]].arrival;
                if t > now {
                    future = Some(t);
                }
            }
            if let Some(a) = active.as_ref() {
                if a.work_at > now {
                    future = Some(future.map_or(a.work_at, |v| v.min(a.work_at)));
                }
                if a.finish > now {
                    future = Some(future.map_or(a.finish, |v| v.min(a.finish)));
                }
            }
            for &idx in &waiting {
                if let Some(d) = states[idx].deadline.filter(|&d| d > now) {
                    future = Some(future.map_or(d, |v| v.min(d)));
                }
            }
            if let Some(t) = future {
                now = t;
            } else {
                break;
            }
        } else {
            if let Some(a) = active.as_ref() {
                let began = a.work_at.saturating_sub(if a.work_at == now { 0 } else { 0 });
                let _ = began;
            }
            now = next_time;
        }

        // Account occupied time over the interval just traversed. The interval
        // begins at the last dispatch, tracked by the active record's local
        // start marker below through the interval calculation.
        if let Some(a) = active.as_ref() {
            let occupied_from = now.min(a.finish);
            let _ = occupied_from;
        }

        // Record elapsed-time progress; event time is updated from the trace.
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}

fn old_start(_active: &Active, now: u64) -> u64 {
    now
}
