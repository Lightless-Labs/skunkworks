```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Tier {
    pub up_to: Option<u64>,
    pub unit: u64,
    pub start: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Price {
    Flat { unit: u64, start: u64 },
    Tiers(Vec<Tier>),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Band {
    pub index: usize,
    pub units: u64,
    pub amount: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Charge {
    pub amount: u64,
    pub bands: Vec<Band>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Mode {
    Graduated,
    Volume,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Bill {
    pub charge: Charge,
    pub minimum_adjustment: u64,
    pub total: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    TooMany,
    Bound { index: usize },
    Unbounded { index: usize },
    MissingFinal,
    Overflow { index: usize },
}

fn validate_tiers(tiers: &[Tier]) -> Result<(), Error> {
    if tiers.is_empty() {
        return Err(Error::Empty);
    }
    if tiers.len() > 64 {
        return Err(Error::TooMany);
    }

    let mut previous = 0u64;
    let last_index = tiers.len() - 1;

    for (index, tier) in tiers.iter().enumerate() {
        match tier.up_to {
            Some(bound) => {
                if bound <= previous {
                    return Err(Error::Bound { index });
                }
                previous = bound;
            }
            None => {
                if index != last_index {
                    return Err(Error::Unbounded { index });
                }
            }
        }
    }

    if tiers[last_index].up_to.is_some() {
        return Err(Error::MissingFinal);
    }

    Ok(())
}

fn band_amount(index: usize, units: u64, unit: u64, start: u64) -> Result<u64, Error> {
    units
        .checked_mul(unit)
        .and_then(|product| product.checked_add(start))
        .ok_or(Error::Overflow { index })
}

fn add_to_total(total: u64, amount: u64, index: usize) -> Result<u64, Error> {
    total
        .checked_add(amount)
        .ok_or(Error::Overflow { index })
}

fn graduated(quantity: u64, tiers: &[Tier]) -> Result<Charge, Error> {
    validate_tiers(tiers)?;

    let mut bands = Vec::new();
    let mut total = 0u64;
    let mut previous = 0u64;

    for (index, tier) in tiers.iter().enumerate() {
        let units = match tier.up_to {
            Some(bound) => {
                let width = bound - previous;
                previous = bound;
                quantity.saturating_sub(bound.saturating_sub(width)).min(width)
            }
            None => quantity.saturating_sub(previous),
        };

        if units > 0 {
            let amount = band_amount(index, units, tier.unit, tier.start)?;
            total = add_to_total(total, amount, index)?;
            bands.push(Band {
                index,
                units,
                amount,
            });
        }
    }

    Ok(Charge {
        amount: total,
        bands,
    })
}

pub fn quote(quantity: u64, price: &Price) -> Result<Charge, Error> {
    match price {
        Price::Flat { unit, start } => {
            if quantity == 0 {
                return Ok(Charge {
                    amount: 0,
                    bands: Vec::new(),
                });
            }
            let amount = band_amount(0, quantity, *unit, *start)?;
            Ok(Charge {
                amount,
                bands: vec![Band {
                    index: 0,
                    units: quantity,
                    amount,
                }],
            })
        }
        Price::Tiers(tiers) => graduated(quantity, tiers),
    }
}

pub fn bill(quantity: u64, price: &Price, mode: Mode, minimum: u64) -> Result<Bill, Error> {
    let charge = match (price, mode) {
        (Price::Tiers(tiers), Mode::Volume) => {
            validate_tiers(tiers)?;
            if quantity == 0 {
                Charge {
                    amount: 0,
                    bands: Vec::new(),
                }
            } else {
                let index = tiers
                    .iter()
                    .position(|tier| tier.up_to.is_none_or(|bound| quantity <= bound))
                    .expect("validated tier schedule has a final unbounded tier");
                let tier = &tiers[index];
                let amount = band_amount(index, quantity, tier.unit, tier.start)?;
                Charge {
                    amount,
                    bands: vec![Band {
                        index,
                        units: quantity,
                        amount,
                    }],
                }
            }
        }
        (_, Mode::Graduated) | (Price::Flat { .. }, Mode::Volume) => quote(quantity, price)?,
    };

    let minimum_adjustment = if quantity > 0 {
        minimum.saturating_sub(charge.amount)
    } else {
        0
    };
    let total = charge
        .amount
        .checked_add(minimum_adjustment)
        .ok_or(Error::Overflow { index: 0 })?;

    Ok(Bill {
        charge,
        minimum_adjustment,
        total,
    })
}
```
