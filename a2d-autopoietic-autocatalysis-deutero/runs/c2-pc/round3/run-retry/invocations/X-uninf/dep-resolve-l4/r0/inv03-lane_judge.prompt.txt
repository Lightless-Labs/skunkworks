fn valid_name(name: &str) -> bool {

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(values: &[Constraint]) -> Vec<Constraint> {
    let mut out = values.to_vec();
    out.sort_by(|a, b| {
        let tag = |c: &Constraint| match c {
            Constraint::Exact(_) => 0u8,
            Constraint::AtLeast(_) => 1,
            Constraint::Before(_) => 2,
            Constraint::Caret(_) => 3,
        };
        tag(a).cmp(&tag(b)).then_with(|| match (*a, *b) {
            (Constraint::Exact(x), Constraint::Exact(y))
            | (Constraint::AtLeast(x), Constraint::AtLeast(y))
            | (Constraint::Before(x), Constraint::Before(y))
            | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(&y),
            _ => std::cmp::Ordering::Equal,
        })
    });
    out.dedup();
    out
}

fn constraint_list_text(values: &[Constraint]) -> String {
    canonical_constraints(values)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn accepts(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {

fn constraint_order(a: Constraint, b: Constraint) -> std::cmp::Ordering {
    fn tag(c: Constraint) -> u8 {
        match c {
            Constraint::Exact(_) => 0,
            Constraint::AtLeast(_) => 1,
            Constraint::Before(_) => 2,
            Constraint::Caret(_) => 3,
        }

        | (Constraint::AtLeast(x), Constraint::AtLeast(y))
        | (Constraint::Before(x), Constraint::Before(y))
        | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(&y),
        _ => std::cmp::Ordering::Equal,
    })
}

fn evidence_lines(evidence: &[Evidence]) -> String {
    let mut out = String::new();
    for ev in sorted_evidence(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(ev.constraint));
        out.push_str(" via ");
        out.push_str(&ev.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraint_list_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraint_list_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\
             \n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        )
        .replace("\\\n             ", ""),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: Vec<Origin>,
    features: BTreeSet<String>,
}

use std::collections::{BTreeMap, BTreeSet};

struct Solver<'a> {
    catalog: &'a [Package],
    package_indices: BTreeMap<String, usize>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    budget: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Solver<'a> {
    fn release(&self, package: &str, version: Version) -> Option<&'a Release> {
        let idx = *self.package_indices.get(package)?;
        self.catalog[idx]
            .releases
            .iter()
            .find(|r| r.version == version)
    }

    fn target_matches(&self, platform: Platform) -> bool {
        matches!(platform, Platform::Any)
            || matches!(
                (platform, self.target),
                (Platform::Linux, Target::Linux)
                    | (Platform::Mac, Target::Mac)
                    | (Platform::Windows, Target::Windows)
            )
    }

    fn build_active(
        &self,
        selected: &BTreeMap<String, Version>,
    ) -> (BTreeMap<String, Active>, BTreeMap<String, Vec<String>>) {
        let mut active: BTreeMap<String, Active> = BTreeMap::new();
        for (i, root) in self.roots.iter().enumerate() {
            let mut features = root.features.clone();
            features.sort();
            features.dedup();
            let token = format!(
                "root[{}]:{}@{}+{{{}}}",
                i,
                root.package,
                constraint_text(root.constraint),
                features.join(",")
            );
            let entry = active.entry(root.package.clone()).or_default();
            entry.origins.push(Origin {
                constraint: root.constraint,
                path: vec![token],
            });
            entry.features.extend(root.features.iter().cloned());
        }

        let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
        loop {
            let mut changed = false;

            // Compute least feature closure and dependencies. The loop monotonically adds
            // requested packages, features, and origins; duplicate dependency origins are
            // guarded by their source and original dependency index.
            let mut known_edges = BTreeSet::<(String, Version, usize)>::new();
            let names: Vec<String> = active.keys().cloned().collect();

            for package in names {
                let Some(&version) = selected.get(&package) else {
                    continue;
                };
                let Some(release) = self.release(&package, version) else {
                    continue;
                };

                let requested: Vec<String> = active[&package].features.iter().cloned().collect();
                for feature_name in requested {
                    if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                        for enabled in &feature.enables {
                            if active.get_mut(&package).unwrap().features.insert(enabled.clone()) {
                                changed = true;
                            }
                        }
                    }
                }

                let activated = active[&package].features.clone();
                for (j, dep) in release.deps.iter().enumerate() {
                    if !self.target_matches(dep.platform)
                        || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                    {
                        continue;
                    }
                    if !known_edges.insert((package.clone(), version, j)) {
                        continue;
                    }
                    let source_path = paths
                        .get(&package)
                        .cloned()
                        .or_else(|| {
                            active[&package]
                                .origins
                                .iter()
                                .map(|o| o.path.clone())
                                .min_by(|a, b| a.len().cmp(&b.len()).then(a.cmp(b)))
                        })
                        .unwrap_or_default();
                    let mut features = dep.features.clone();
                    features.sort();
                    features.dedup();
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        package,
                        version_text(version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        features.join(","),
                        dep.gate.as_deref().unwrap_or("*"),
                        match dep.platform {
                            Platform::Any => "any",
                            Platform::Linux => "linux",
                            Platform::Mac => "mac",
                            Platform::Windows => "windows",
                        }
                    );
                    let mut path = source_path;
                    path.push(token);
                    let entry = active.entry(dep.package.clone()).or_default();
                    entry.origins.push(Origin {
                        constraint: dep.constraint,
                        path,
                    });
                    entry.features.extend(dep.features.iter().cloned());
                    changed = true;
                }
            }
            if !changed {
                break;
            }

            // Keep the closure finite when traversing cycles. Origins are handled as
            // individual graph edges, while package/feature closure is monotone.
            let mut next_paths = paths.clone();
            for (name, state) in &active {
                for origin in &state.origins {
                    let replace = next_paths.get(name).is_none_or(|old| {
                        origin.path.len() < old.len()
                            || (origin.path.len() == old.len() && origin.path < *old)
                    });
                    if replace {
                        next_paths.insert(name.clone(), origin.path.clone());
                    }
                }
            }
            paths = next_paths;
        }

        // Canonical shortest incoming paths are derived from all discovered origins.
        for (name, state) in &active {
            let best = state
                .origins
                .iter()
                .map(|o| o.path.clone())
                .min_by(|a, b| a.len().cmp(&b.len()).then(a.cmp(b)));
            if let Some(path) = best {
                paths.insert(name.clone(), path);
            }
        }

        (active, paths)
    }

    fn canonical_origins(active: &Active) -> Vec<Constraint> {
        canonical_constraints(
            &active
                .origins
                .iter()
                .map(|o| o.constraint)
                .collect::<Vec<_>>(),
        )
    }

    fn evidence(active: &Active) -> Vec<Evidence> {
        let mut result = active
            .origins
            .iter()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect::<Vec<_>>();
        result.sort_by(|a, b| {
            constraint_order(a.constraint, b.constraint).then_with(|| a.path.cmp(&b.path))
        });
        result
    }

    fn inspect(
        &mut self,
        selected: &BTreeMap<String, Version>,
    ) -> Result<Option<(String, Vec<Version>)>, ResolveError> {
        let (active, paths) = self.build_active(selected);

        if let Some(name) = active
            .keys()
            .find(|name| !self.package_indices.contains_key(*name))
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
                evidence: Self::evidence(&active[name]),
            });
        }

        for (name, version) in selected {
            if let Some(state) = active.get(name) {
                if !state
                    .origins
                    .iter()
                    .all(|o| accepts(o.constraint, *version))
                {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: Self::canonical_origins(state),
                        evidence: Self::evidence(state),
                    });
                }
            }
        }

        for (name, version) in selected {
            let Some(release) = self.release(name, *version) else {
                continue;
            };
            let state = &active[name];
            for feature in &state.features {
                if !release.features.iter().any(|f| &f.name == feature) {
                    return Err(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: paths.get(name).cloned().unwrap_or_default(),
                    });
                }
            }
        }

        for (name, state) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let idx = self.package_indices[name];
            let mut candidates: Vec<Version> = self.catalog[idx]
                .releases
                .iter()
                .map(|r| r.version)
                .filter(|v| state.origins.iter().all(|o| accepts(o.constraint, *v)))
                .collect();
            candidates.sort_by(|a, b| b.cmp(a));
            candidates.dedup();
            if candidates.is_empty() {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: Self::canonical_origins(state),
                    evidence: Self::evidence(state),
                });
            }
        }

        let Some(name) = active.keys().find(|name| !selected.contains_key(*name)) else {
            return Ok(None);
        };
        let idx = self.package_indices[name];
        let state = &active[name];
        let mut candidates: Vec<Version> = self.catalog[idx]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| state.origins.iter().all(|o| accepts(o.constraint, *v)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(lock_version) = self.locks.get(name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock_version) {
                let locked = candidates.remove(pos);
                candidates.insert(0, locked);
            }
        }
        Ok(Some((name.clone(), candidates)))
    }

    fn lock_cost(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.locks
            .iter()
            .filter(|(package, version)| selected.get(*package) != Some(*version))
            .count()
    }

    fn better(
        &self,
        cost: usize,
        candidate: &BTreeMap<String, Version>,
        current: &(usize, BTreeMap<String, Version>),
    ) -> bool {
        if cost != current.0 {
            return cost < current.0;
        }
        for name in self.package_indices.keys() {
            let a = candidate.get(name);
            let b = current.1.get(name);
            match (a, b) {
                (Some(x), Some(y)) if x != y => return x > y,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }

    fn search(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        match self.inspect(selected) {
            Err(error) => {
                if let ResolveError::Limit { .. } = error {
                    return Err(error);
                }
                if self.first_error.is_none() {
                    self.first_error = Some(error);
                }
                Ok(())
            }
            Ok(None) => {
                let cost = self.lock_cost(selected);
                if self
                    .best
                    .as_ref()
                    .is_none_or(|current| self.better(cost, selected, current))
                {
                    self.best = Some((cost, selected.clone()));
                }
                Ok(())
            }
            Ok(Some((package, candidates))) => {
                for version in candidates {
                    if self.attempts == self.budget {
                        return Err(ResolveError::Limit {
                            attempts: self.budget,
                        });
                    }
                    self.attempts += 1;
                    selected.insert(package.clone(), version);
                    let result = self.search(selected);
                    selected.remove(&package);
                    result?;
                }
                Ok(())
            }
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", base),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", base, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
        lock_map.insert(lock.package.clone(), lock.version);
    }

    let package_indices = catalog
        .iter()
        .enumerate()
        .map(|(i, p)| (p.name.clone(), i))
        .collect::<BTreeMap<_, _>>();

    let mut solver = Solver {
        catalog,
        package_indices,
        roots,
        locks: lock_map.clone(),
        target,
        budget: max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut selected = BTreeMap::new();
    solver.search(&mut selected)?;

    let Some((_, chosen)) = solver.best else {
        return Err(solver.first_error.unwrap_or_else(|| ResolveError::Missing {
            package: String::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_out = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let (final_active, _) = solver.build_active(&chosen);
    let mut names = BTreeSet::new();
    names.extend(lock_map.keys().cloned());
    names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for package in names {
        let before = lock_map.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let available = package_indices
                    .get(&package)
                    .is_some_and(|idx| catalog[*idx].releases.iter().any(|r| r.version == old));
                if !available {
                    ChangeReason::Unavailable
                } else {
                    let rejected = final_active.get(&package).is_some_and(|active| {
                        !active
                            .origins
                            .iter()
                            .all(|origin| accepts(origin.constraint, old))
                    });
                    if rejected {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}

        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by_key(|c| constraint_rank(*c));
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn satisfies(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sorted_evidence(e: &[Evidence]) -> Vec<Evidence> {
    let mut v = e.to_vec();
    v.sort_by(|a, b| {
        constraint_rank(a.constraint)
            .cmp(&constraint_rank(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    v
}

fn path_text(path: &[String]) -> String {
    path.join(" -> ")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => evidence_text(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => evidence_text(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => evidence_text(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path_text(path)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn evidence_text(mut first: String, evidence: &[Evidence]) -> String {
    for e in sorted_evidence(evidence) {
        first.push_str("\n  C:");
        first.push_str(&constraint_text(e.constraint));
        first.push_str(" via ");
        first.push_str(&path_text(&e.path));
    }
    first
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: r.package.clone(),
            });
        }
        for (f, name) in r.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: name.clone(),
                });
            }
        }
    }

    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: p.name.clone(),
            });
        }
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: d.package.clone(),
                    });
                }
                if let Some(g) = &d.gate {
                    if !valid_name(g) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].gate"),
                            name: g.clone(),
                        });
                    }
                }
                for (f, name) in d.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].feature[{f}]"),
                            name: name.clone(),
                        });
                    }
                }
            }
            for (f, feature) in r.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].feature[{f}]"),
                        name: feature.name.clone(),
                    });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].feature[{f}].enable[{e}]"),
                            name: name.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut names = std::collections::BTreeSet::new();
    for p in catalog {
        if !names.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }

    for p in catalog {
        let mut seen = std::collections::BTreeSet::new();
        for r in &p.releases {
            if !seen.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for p in catalog {
        for r in &p.releases {
            let mut seen = std::collections::BTreeSet::new();
            for f in &r.features {
                if !seen.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = std::collections::BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

        Platform::Linux => target == Target::Linux,
        Platform::Mac => target == Target::Mac,
        Platform::Windows => target == Target::Windows,

        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }

fn closure(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let catalog_map: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut result = Closure::default();

    for (i, root) in roots.iter().enumerate() {
        let features: std::collections::BTreeSet<String> =
            root.features.iter().cloned().collect();
        result
            .packages
            .entry(root.package.clone())
            .or_default()
            .origins
            .push(Origin {
                constraint: root.constraint,
                path: vec![format!(
                    "root[{}]:{}@{}+{{{}}}",
                    i,
                    root.package,
                    constraint_text(root.constraint),
                    sorted_names(&features)
                )],
            });
        result
            .packages
            .entry(root.package.clone())
            .or_default()
            .features
            .extend(features);
    }

    loop {
        let mut changed = false;

        let names: Vec<String> = result.packages.keys().cloned().collect();
        for name in names {
            let Some(version) = selected.get(&name).copied() else {
                continue;
            };
            let Some(package) = catalog_map.get(name.as_str()).copied() else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let activated: Vec<String> = result.packages[&name]
                .features
                .iter()
                .cloned()
                .collect();
            for fname in activated {
                if let Some(feature) = release.features.iter().find(|f| f.name == fname) {
                    for enabled in &feature.enables {
                        if result
                            .packages
                            .get_mut(&name)
                            .unwrap()
                            .features
                            .insert(enabled.clone())
                        {
                            changed = true;
                        }
                    }
                }
            }

            let active_features = result.packages[&name].features.clone();
            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !active_features.contains(g))
                {
                    continue;
                }
                let feature_set: std::collections::BTreeSet<String> =
                    dep.features.iter().cloned().collect();
                let source_path = result
                    .paths
                    .get(&name)
                    .cloned()
                    .unwrap_or_else(|| vec![String::new()]);
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    name,
                    version_text(version),
                    j,
                    dep.package,
                    constraint_text(dep.constraint),
                    sorted_names(&feature_set),
                    dep.gate.as_deref().unwrap_or("*"),
                    platform_text(dep.platform)
                );
                let path = if source_path.first().is_some_and(|s| s.is_empty()) {
                    vec![token]
                } else {
                    let mut p = source_path;
                    p.push(token);
                    p
                };
                let info = result.packages.entry(dep.package.clone()).or_default();
                let old_len = info.origins.len();
                info.origins.push(Origin {
                    constraint: dep.constraint,
                    path,
                });
                info.features.extend(feature_set);
                if info.origins.len() != old_len {
                    changed = true;
                }
            }
        }

        let new_paths = canonical_paths(&result.packages);
        if new_paths != result.paths {
            result.paths = new_paths;
            changed = true;
        }

        // Dependencies are generated repeatedly during closure; keep one origin per
        // selected source release and dependency index, while retaining root origins.
        for (pkg_name, info) in result.packages.iter_mut() {
            let mut seen = std::collections::BTreeSet::new();
            info.origins.retain(|o| {
                if o.path.first().is_some_and(|p| p.starts_with("root[")) {
                    return true;
                }
                seen.insert((o.constraint, o.path.clone()))
            });
            let _ = pkg_name;
        }

        if !changed {
            break;
        }
    }

    result.paths = canonical_paths(&result.packages);
    result
}

fn canonical_paths(
    packages: &std::collections::BTreeMap<String, PackageInfo>,
) -> std::collections::BTreeMap<String, Vec<String>> {
    let mut paths = std::collections::BTreeMap::new();
    for (name, info) in packages {
        let mut candidates: Vec<Vec<String>> = info
            .origins
            .iter()
            .map(|o| o.path.clone())
            .filter(|p| !p.is_empty())
            .collect();
        if let Some(best) = candidates.drain(..).min_by(|a, b| {
            a.len().cmp(&b.len()).then_with(|| a.cmp(b))
        }) {
            paths.insert(name.clone(), best);
        }
    }
    paths
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    budget: usize,
    attempts: usize,
    best: Option<(usize, std::collections::BTreeMap<String, Version>, Closure)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut std::collections::BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let c = closure(self.catalog, self.roots, selected, self.target);
        let catalog_map: std::collections::BTreeMap<&str, &Package> =
            self.catalog.iter().map(|p| (p.name.as_str(), p)).collect();

        for name in c.packages.keys() {
            if !catalog_map.contains_key(name.as_str()) {
                let evidence = c.packages[name]
                    .origins
                    .iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path.clone(),
                    })
                    .collect();
                self.record_error(ResolveError::Missing {
                    package: name.clone(),
                    evidence,
                });
                return Ok(());
            }
        }

        for (name, version) in selected.iter() {
            let info = &c.packages[name];
            let constraints: Vec<Constraint> =
                info.origins.iter().map(|o| o.constraint).collect();
            if constraints.iter().any(|x| !satisfies(*version, *x)) {
                let evidence = info
                    .origins
                    .iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path.clone(),
                    })
                    .collect();
                self.record_error(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints,
                    evidence,
                });
                return Ok(());
            }
        }

        for (name, info) in &c.packages {
            if selected.contains_key(name) {
                continue;
            }
            let constraints: Vec<Constraint> =
                info.origins.iter().map(|o| o.constraint).collect();
            let package = catalog_map[name.as_str()];
            if !package.releases.iter().any(|r| {
                constraints.iter().all(|x| satisfies(r.version, *x))
            }) {
                let evidence = info
                    .origins
                    .iter()
                    .map(|o| Evidence {
                        constraint: o.constraint,
                        path: o.path.clone(),
                    })
                    .collect();
                self.record_error(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints,
                    evidence,
                });
                return Ok(());
            }
        }

        for (name, version) in selected.iter() {
            let package = catalog_map[name.as_str()];
            let release = package.releases.iter().find(|r| r.version == *version).unwrap();
            let features = &c.packages[name].features;
            for feature in features {
                if !release.features.iter().any(|f| f.name == *feature) {
                    self.record_error(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: c.paths.get(name).cloned().unwrap_or_default(),
                    });
                    return Ok(());
                }
            }
        }

        let next = c
            .packages
            .keys()
            .find(|n| !selected.contains_key(*n))
            .cloned();
        let Some(name) = next else {
            self.record_solution(selected, c);
            return Ok(());
        };

        let info = &c.packages[&name];
        let constraints: Vec<Constraint> = info.origins.iter().map(|o| o.constraint).collect();
        let mut candidates: Vec<Version> = catalog_map[name.as_str()]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| constraints.iter().all(|c| satisfies(*v, *c)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            self.visit(selected)?;
            selected.remove(&name);
        }
        Ok(())
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn record_solution(
        &mut self,
        selected: &std::collections::BTreeMap<String, Version>,
        closure: Closure,
    ) {
        let lock_map: std::collections::BTreeMap<&str, Version> = self
            .locks
            .iter()
            .map(|l| (l.package.as_str(), l.version))
            .collect();
        let cost = lock_map
            .iter()
            .filter(|(p, v)| selected.get(**p) != Some(**v))
            .count();

        let better = match &self.best {
            None => true,
            Some((old_cost, old, _)) => {
                if cost != *old_cost {
                    cost < *old_cost
                } else {
                    let all: std::collections::BTreeSet<String> = self
                        .catalog
                        .iter()
                        .map(|p| p.name.clone())
                        .collect();
                    let mut cmp = std::cmp::Ordering::Equal;
                    for name in all {
                        let a = selected.get(&name);
                        let b = old.get(&name);
                        cmp = match (a, b) {
                            (None, None) => std::cmp::Ordering::Equal,
                            (None, Some(_)) => std::cmp::Ordering::Less,
                            (Some(_), None) => std::cmp::Ordering::Greater,
                            (Some(x), Some(y)) => x.cmp(y),
                        };
                        if cmp != std::cmp::Ordering::Equal {
                            break;
                        }
                    }
                    cmp == std::cmp::Ordering::Greater
                }
            }
        };
        if better {
            self.best = Some((cost, selected.clone(), closure));
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        budget: max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    search.visit(&mut std::collections::BTreeMap::new())?;

    if let Some((_, selected_map, final_closure)) = search.best {
        let selected: Vec<Selection> = selected_map
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect();

        let before: std::collections::BTreeMap<String, Version> = locks
            .iter()
            .map(|l| (l.package.clone(), l.version))
            .collect();
        let after = &selected_map;
        let all: std::collections::BTreeSet<String> =
            before.keys().chain(after.keys()).cloned().collect();
        let mut changes = Vec::new();

        for package in all {
            let b = before.get(&package).copied();
            let a = after.get(&package).copied();
            if b == a {
                continue;
            }
            let reason = match (b, a) {
                (None, Some(_)) => ChangeReason::Added,
                (Some(_), None) => ChangeReason::Removed,
                (Some(old), Some(_)) => {
                    let exists = catalog.iter().find(|p| p.name == package).is_some_and(|p| {
                        p.releases.iter().any(|r| r.version == old)
                    });
                    if !exists {
                        ChangeReason::Unavailable
                    } else {
                        let rejected = final_closure
                            .packages
                            .get(&package)
                            .is_some_and(|info| {
                                info.origins
                                    .iter()
                                    .any(|o| !satisfies(old, o.constraint))
                            });
                        if rejected {
                            ChangeReason::Rejected
                        } else {
                            ChangeReason::Rebalanced
                        }
                    }
                }
                (None, None) => unreachable!(),
            };
            changes.push(Change {
                package,
                before: b,
                after: a,
                reason,
            });
        }

        Ok(Resolution { selected, changes })
    } else {
        Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }))
    }
}

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn enc(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> std::cmp::Ordering {
    a.constraint
        .cmp(&b.constraint)
        .then_with(|| a.path.cmp(&b.path))
}

fn append_evidence(line: &mut String, evidence: &[Evidence]) {
    let mut sorted = evidence.to_vec();
    sorted.sort_by(evidence_cmp);
    for e in sorted {
        line.push_str("\n  C:");
        line.push_str(&constraint_text(e.constraint));
        line.push_str(" via ");
        line.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", enc(site), enc(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", enc(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            enc(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            enc(package),
            version_text(*version),
            enc(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", enc(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", enc(package));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!("no-match|package={}|constraints={}", enc(package), cs);
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let cs = canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(",");
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                enc(package),
                version_text(*selected),
                cs
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            enc(package),
            version_text(*selected),
            enc(feature),
            enc(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: Vec<Origin>,
    features: BTreeSet<String>,
    paths: Vec<Vec<String>>,
}

use std::collections::{BTreeMap, BTreeSet};

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, BTreeMap<String, Version>, BTreeMap<String, Vec<Constraint>>)>,
    first_error: Option<ResolveError>,
    limit: bool,
}

impl<'a> Search<'a> {
    fn package(&self, name: &str) -> Option<&'a Package> {
        self.catalog.iter().find(|p| p.name == name)
    }

    fn release(&self, name: &str, version: Version) -> Option<&'a Release> {
        self.package(name)?.releases.iter().find(|r| r.version == version)
    }

    fn target_matches(&self, p: Platform) -> bool {
        matches!(
            (p, self.target),
            (Platform::Any, _)
                | (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
    }

    fn active(&self, selected: &BTreeMap<String, Version>) -> BTreeMap<String, Active> {
        let mut active = BTreeMap::<String, Active>::new();
        for (i, r) in self.roots.iter().enumerate() {
            let mut fs = r.features.clone();
            fs.sort();
            fs.dedup();
            let token = format!(
                "root[{}]:{}@{}+{{{}}}",
                i,
                r.package,
                constraint_text(r.constraint),
                fs.join(",")
            );
            let a = active.entry(r.package.clone()).or_default();
            a.origins.push(Origin {
                constraint: r.constraint,
                path: vec![token.clone()],
            });
            a.features.extend(r.features.iter().cloned());
            a.paths.push(vec![token]);
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = active.keys().cloned().collect();
            for name in names {
                let Some(&version) = selected.get(&name) else {
                    continue;
                };
                let Some(release) = self.release(&name, version) else {
                    continue;
                };
                let feature_names: Vec<String> =
                    active[&name].features.iter().cloned().collect();
                for feature_name in feature_names {
                    if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                        for enabled in &feature.enables {
                            if active.get_mut(&name).unwrap().features.insert(enabled.clone()) {
                                changed = true;
                            }
                        }
                    }
                }

                let activated = active[&name].features.clone();
                for (j, dep) in release.deps.iter().enumerate() {
                    if !self.target_matches(dep.platform)
                        || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                    {
                        continue;
                    }
                    let mut fs = dep.features.clone();
                    fs.sort();
                    fs.dedup();
                    let gate = dep.gate.as_deref().unwrap_or("*");
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        name,
                        version_text(version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        fs.join(","),
                        gate,
                        match dep.platform {
                            Platform::Any => "any",
                            Platform::Linux => "linux",
                            Platform::Mac => "mac",
                            Platform::Windows => "windows",
                        }
                    );
                    let src_paths = active[&name].paths.clone();
                    for src in src_paths {
                        let mut path = src;
                        path.push(token.clone());
                        let dest = active.entry(dep.package.clone()).or_default();
                        let origin = Origin {
                            constraint: dep.constraint,
                            path: path.clone(),
                        };
                        if !dest.origins.iter().any(|o| o.constraint == origin.constraint && o.path == origin.path) {
                            dest.origins.push(origin);
                            changed = true;
                        }
                        if dest.features.extend(dep.features.iter().cloned()) {
                            changed = true;
                        }
                        if !dest.paths.contains(&path) {
                            dest.paths.push(path);
                            changed = true;
                        }
                    }
                }
            }
            if !changed {
                break;
            }
        }

        for a in active.values_mut() {
            a.paths.sort_by(|x, y| x.len().cmp(&y.len()).then_with(|| x.cmp(y)));
            a.paths.dedup();
            a.origins.sort_by(|x, y| {
                x.constraint
                    .cmp(&y.constraint)
                    .then_with(|| x.path.cmp(&y.path))
            });
        }
        active
    }

    fn canonical_path(a: &Active) -> Vec<String> {
        a.paths.first().cloned().unwrap_or_default()
    }

    fn evidence(a: &Active) -> Vec<Evidence> {
        let mut e: Vec<Evidence> = a
            .origins
            .iter()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect();
        e.sort_by(evidence_cmp);
        e
    }

    fn requirements(a: &Active) -> Vec<Constraint> {
        canonical_constraints(&a.origins.iter().map(|o| o.constraint).collect::<Vec<_>>())
    }

    fn record_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn vector_better(&self, a: &BTreeMap<String, Version>, b: &BTreeMap<String, Version>) -> bool {
        let names: BTreeSet<String> = self.catalog.iter().map(|p| p.name.clone()).collect();
        for name in names {
            match (a.get(&name), b.get(&name)) {
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(x), Some(y)) if x != y => return x > y,
                _ => {}
            }
        }
        false
    }

    fn complete(&mut self, selected: &BTreeMap<String, Version>, active: &BTreeMap<String, Active>) {
        let cost = self
            .locks
            .iter()
            .filter(|l| selected.get(&l.package) != Some(&l.version))
            .count();
        let constraints: BTreeMap<String, Vec<Constraint>> = active
            .iter()
            .map(|(n, a)| (n.clone(), Self::requirements(a)))
            .collect();
        let replace = match &self.best {
            None => true,
            Some((best_cost, best_sel, _)) => {
                cost < *best_cost
                    || (cost == *best_cost && self.vector_better(selected, best_sel))
            }
        };
        if replace {
            self.best = Some((cost, selected.clone(), constraints));
        }
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) {
        if self.limit {
            return;
        }
        let active = self.active(selected);

        if let Some(name) = active.keys().find(|n| self.package(n).is_none()).cloned() {
            self.record_error(ResolveError::Missing {
                package: name,
                evidence: Self::evidence(&active.values().next().cloned().unwrap_or_default()),
            });
            // Replace evidence with the implicated package's evidence.
            let name = match self.first_error.as_ref() {
                Some(ResolveError::Missing { package, .. }) => package.clone(),
                _ => String::new(),
            };
            if let Some(a) = active.get(&name) {
                self.first_error = Some(ResolveError::Missing {
                    package: name,
                    evidence: Self::evidence(a),
                });
            }
            return;
        }

        for (name, version) in selected.iter() {
            let Some(a) = active.get(name) else { continue };
            if !a.origins.iter().all(|o| matches(*version, o.constraint)) {
                self.record_error(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: Self::requirements(a),
                    evidence: Self::evidence(a),
                });
                return;
            }
        }

        for (name, a) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let package = self.package(name).unwrap();
            if !package
                .releases
                .iter()
                .any(|r| a.origins.iter().all(|o| matches(r.version, o.constraint)))
            {
                self.record_error(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: Self::requirements(a),
                    evidence: Self::evidence(a),
                });
                return;
            }
        }

        for (name, version) in selected.iter() {
            let Some(a) = active.get(name) else { continue };
            let Some(release) = self.release(name, *version) else { continue };
            for feature in &a.features {
                if !release.features.iter().any(|f| f.name == *feature) {
                    self.record_error(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: Self::canonical_path(a),
                    });
                    return;
                }
            }
        }

        let next = active.keys().find(|n| !selected.contains_key(*n)).cloned();
        let Some(name) = next else {
            self.complete(selected, &active);
            return;
        };

        let a = &active[&name];
        let mut candidates: Vec<Version> = self
            .package(&name)
            .unwrap()
            .releases
            .iter()
            .filter(|r| a.origins.iter().all(|o| matches(r.version, o.constraint)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|x, y| y.cmp(x));
        candidates.dedup();
        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                self.limit = true;
                return;
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            self.visit(selected);
            selected.remove(&name);
            if self.limit {
                return;
            }
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen = BTreeSet::new();
    for p in catalog {
        if !seen.insert(p.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: p.name.clone(),
            });
        }
    }
    for p in catalog {
        let mut versions = BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut features = BTreeSet::new();
            for f in &r.features {
                if !features.insert(f.name.as_str()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_seen = BTreeSet::new();
    for lock in locks {
        if !lock_seen.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        limit: false,
    };
    let mut selected = BTreeMap::new();
    search.visit(&mut selected);
    if search.limit {
        return Err(ResolveError::Limit {
            attempts: max_attempts,
        });
    }
    let Some((_, chosen, constraints)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_out: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    let mut before = BTreeMap::<String, Version>::new();
    for lock in locks {
        before.insert(lock.package.clone(), lock.version);
    }
    let mut names: BTreeSet<String> = before.keys().cloned().collect();
    names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for name in names {
        let b = before.get(&name).copied();
        let a = chosen.get(&name).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                if search
                    .package(&name)
                    .is_none_or(|p| !p.releases.iter().any(|r| r.version == old))
                {
                    ChangeReason::Unavailable
                } else if !constraints
                    .get(&name)
                    .cloned()
                    .unwrap_or_default()
                    .iter()
                    .all(|c| matches(old, *c))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before: b,
            after: a,
            reason,
        });
    }
    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}
