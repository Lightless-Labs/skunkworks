pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before checking any other input.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal with the smallest available task ID first.
    let mut remaining_indegree = indegree.clone();
    let mut available: Vec<usize> = (0..n)
        .filter(|&task_id| remaining_indegree[task_id] == 0)
        .collect();
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, task_id)| *task_id)
            .map(|(pos, _)| pos)
            .unwrap();
        let task_id = available.swap_remove(smallest_pos);
        order.push(task_id);
        for &child in &children[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = vec![false; n];
        for &task_id in &order {
            blocked[task_id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !blocked[task_id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut running: Vec<Running> = Vec::new();
    let mut worker_available = vec![0u64; worker_count];
    let mut used: Vec<u64> = vec![0; capacity.len()];
    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completion_count = 0usize;

    fn append_resources(trace: &mut String, used: &[u64]) {
        if used.is_empty() {
            trace.push('-');
        } else {
            for (i, value) in used.iter().enumerate() {
                if i > 0 {
                    trace.push(',');
                }
                trace.push_str(&value.to_string());
            }
        }
    }

    while completion_count < n {
        // Complete all tasks finishing now, in ascending task ID order.
        let mut done_now: Vec<Running> = running
            .iter()
            .filter(|entry| entry.finish == now)
            .cloned()
            .collect();
        done_now.sort_by_key(|entry| entry.task);

        if !done_now.is_empty() {
            running.retain(|entry| entry.finish != now);
            for entry in done_now {
                completed[entry.task] = true;
                completion_count += 1;
                for (resource, &need) in tasks[entry.task].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                trace.push_str(&format!(
                    "{now}|end|{}|{}|",
                    entry.task, entry.worker
                ));
                append_resources(&mut trace, &used);
                trace.push('\n');
            }
        }

        // Keep admitting one task at a time, reevaluating after each start.
        let mut admitted = false;
        loop {
            let mut eligible: Vec<usize> = (0..n)
                .filter(|&task_id| {
                    if started[task_id] || tasks[task_id].release > now {
                        return false;
                    }
                    tasks[task_id]
                        .deps
                        .iter()
                        .all(|&pred| completed[pred])
                })
                .filter(|&task_id| {
                    tasks[task_id]
                        .needs
                        .iter()
                        .enumerate()
                        .all(|(resource, &need)| {
                            u64::from(need) <= u64::from(capacity[resource]) - used[resource]
                        })
                })
                .collect();

            eligible.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let task_id = match eligible.first().copied() {
                Some(task_id) => task_id,
                None => break,
            };

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= now);

            let worker = match worker {
                Some(worker) => worker,
                None => break,
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }

            trace.push_str(&format!("{now}|start|{task_id}|{worker}|"));
            append_resources(&mut trace, &used);
            trace.push('\n');

            if tasks[task_id].duration == 0 {
                completed[task_id] = true;
                completion_count += 1;
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                trace.push_str(&format!("{now}|end|{task_id}|{worker}|"));
                append_resources(&mut trace, &used);
                trace.push('\n');
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }

            makespan = makespan.max(finish);
            admitted = true;
        }

        if completion_count == n {
            break;
        }

        // If no more task can start now, advance to the next relevant event.
        let mut next = u64::MAX;
        for entry in &running {
            if entry.finish > now {
                next = next.min(entry.finish);
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                next = next.min(tasks[task_id].release);
            }
        }
        for worker in 0..worker_count {
            if worker_available[worker] > now {
                next = next.min(worker_available[worker]);
            }
        }

        if next == u64::MAX {
            // A valid acyclic graph always has a future event unless all tasks
            // have completed. Keep this guard to ensure malformed states do not
            // loop forever.
            if !admitted {
                break;
            }
        } else {
            now = next;
        }
    }

    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

                    position,
                    value: dep,
                });
            }
            if dep == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dep) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dep,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resource shapes and demands task by task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build graph edges from predecessors to children.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Deterministic Kahn traversal, always removing the smallest available ID.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut present = vec![false; n];
            for &id in &order {
                present[id] = true;
            }
            present
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let mut max_child_tail = 0u64;
        for &child in &children[id] {
            max_child_tail = max_child_tail.max(tails[child]);
        }
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_ready = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut used = vec![0u64; capacity.len()];
    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;
    let mut trace = format!("workers={};resources={}\n", worker_count, capacity.len());

    while completed_count < n {
        // Process all completions at this instant in ascending task ID.
        let mut completing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(i, entry)| (entry.finish == now).then_some(i))
            .collect();
        completing.sort_by_key(|&i| running[i].task);

        for index in completing.into_iter().rev() {
            let entry = running.remove(index);
            let task_id = entry.task;
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(entry.finish);

            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] -= u64::from(need);
            }

            trace.push_str(&format!(
                "{}|end|{}|{}|{}\n",
                now,
                task_id,
                entry.worker,
                format_resources(&used)
            ));
        }

        // Start tasks one at a time, reevaluating after every admission.
        loop {
            let mut candidates = Vec::<usize>::new();
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || !tasks[id].deps.iter().all(|&dep| completed[dep])
                {
                    continue;
                }

                let fits = tasks[id].needs.iter().enumerate().all(|(r, &need)| {
                    u64::from(need) <= u64::from(capacity[r]) - used[r]
                });
                if fits {
                    candidates.push(id);
                }
            }

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            // Select a currently available worker using the task's cyclic preference.
            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&w| worker_ready[w] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let ready_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_ready[worker] = ready_at;
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] += u64::from(need);
            }

            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                format_resources(&used)
            ));

            if finish == now {
                // Zero-duration completion is immediate, before selecting another start.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    format_resources(&used)
                ));
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest event that can change eligibility or availability.
        let mut next = None::<u64>;
        for entry in &running {
            if entry.finish > now {
                next = Some(next.map_or(entry.finish, |v| v.min(entry.finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |v| v.min(tasks[id].release)));
            }
        }
        for &ready in &worker_ready {
            if ready > now {
                next = Some(next.map_or(ready, |v| v.min(ready)));
            }
        }

        // If nothing can change, an unstarted task is necessarily blocked by a cycle,
        // which was already ruled out; this is only a defensive no-panic fallback.
        let Some(next_time) = next else {
            break;
        };
        now = next_time;
    }

    trace.push_str(&format!("makespan={}\n", makespan));

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every validated task must run"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn format_resources(used: &[u64]) -> String {
    if used.is_empty() {
        return "-".to_owned();
    }
    let mut result = String::new();
    for (index, value) in used.iter().enumerate() {
        if index > 0 {
            result.push(',');
        }
        result.push_str(&value.to_string());
    }
    result
}

fn append_event(

    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],

    let _ = write!(trace, "{time}|{kind}|{task}|{worker}|");

    trace.push('\n');

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Resource validation is task-major, then resource-major.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &cap)) in task.needs.iter().zip(capacity).enumerate() {
            if need > cap {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: cap,
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }

    // Kahn traversal always removes the smallest currently available ID.
    let mut ready = BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            ready.push(Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree.clone();
    while let Some(Reverse(task_id)) = ready.pop() {
        order.push(task_id);
        for &child in &successors[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                ready.push(Reverse(child));
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut flags = vec![false; n];
            for &task_id in &order {
                flags[task_id] = true;
            }
            flags
        };
        let blocked = (0..n).filter(|&task_id| !removed[task_id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let successor_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(successor_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = String::new();
    let _ = writeln!(
        trace,
        "workers={worker_count};resources={}",
        capacity.len()
    );

    let mut used = vec![0u64; capacity.len()];
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut worker_available = vec![0u64; worker_count];

    // Running entries are (finish time, task ID, worker ID).
    let mut running: Vec<(u64, usize, usize)> = Vec::new();
    let mut current_time = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete all tasks finishing now, in ascending task ID.
        let mut finishing: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|&(finish, _, _)| finish == current_time)
            .collect();
        finishing.sort_by_key(|&(_, task_id, _)| task_id);

        if !finishing.is_empty() {
            running.retain(|&(finish, _, _)| finish != current_time);
            for (finish, task_id, worker) in finishing {
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                append_event(
                    &mut trace,
                    current_time,
                    "end",
                    task_id,
                    worker,
                    &used,
                );
            }
        }

        // Repeatedly admit one task, reevaluating eligibility and ranking.
        let mut admitted_any = false;
        loop {
            let mut chosen: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id] || tasks[task_id].release > current_time {
                    continue;
                }
                if !tasks[task_id]
                    .deps
                    .iter()
                    .all(|&dep| completed[dep])
                {
                    continue;
                }
                let fits = tasks[task_id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(resource, &need)| {
                        u64::from(need) <= u64::from(capacity[resource]) - used[resource]
                    });
                if !fits {
                    continue;
                }
                let better = match chosen {
                    None => true,
                    Some(best) => {
                        tails[task_id] > tails[best]
                            || (tails[task_id] == tails[best]
                                && (tasks[task_id].priority > tasks[best].priority
                                    || (tasks[task_id].priority == tasks[best].priority
                                        && task_id < best)))
                    }
                };
                if better {
                    chosen = Some(task_id);
                }
            }

            let task_id = match chosen {
                Some(task_id) => task_id,
                None => break,
            };

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= current_time);

            let worker = match worker {
                Some(worker) => worker,
                None => break,
            };

            let finish = current_time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let worker_free = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            worker_available[worker] = worker_free;
            started[task_id] = true;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: current_time,
                finish,
                worker,
            });
            append_event(
                &mut trace,
                current_time,
                "start",
                task_id,
                worker,
                &used,
            );

            if finish == current_time {
                // A zero-duration completion is immediate, before another start.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                append_event(
                    &mut trace,
                    current_time,
                    "end",
                    task_id,
                    worker,
                    &used,
                );
            } else {
                running.push((finish, task_id, worker));
            }
            admitted_any = true;
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest future event that can change admission state.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > current_time {
                next_time = Some(next_time.map_or(finish, |old| old.min(finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > current_time {
                let release = tasks[task_id].release;
                next_time = Some(next_time.map_or(release, |old| old.min(release)));
            }
        }
        for &available in &worker_available {
            if available > current_time {
                next_time = Some(next_time.map_or(available, |old| old.min(available)));
            }
        }

        if let Some(next) = next_time {
            current_time = next;
        } else if !admitted_any {
            // Acyclic tasks with valid dependencies always have a future event
            // unless all tasks are complete; this guard avoids an accidental loop.
            break;
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task in an acyclic schedule must run"))
        .collect();

    let _ = writeln!(trace, "makespan={makespan}");

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}
