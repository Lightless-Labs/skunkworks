#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut record = Vec::new();
    let mut field = String::new();
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if !in_quotes && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().unwrap();
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                record.push(std::mem::take(&mut field));
                after_quote = false;
                i += 1;
            } else if b == b'\n' || b == b'\r' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                after_quote = false;
                if b == b'\r' {
                    i += 2;
                } else {
                    i += 1;
                }
            } else {
                return Err(CsvError::TrailingByte {
                    record: record_index,
                    byte: i,
                });
            }
            continue;
        }

        if b == delimiter {
            record.push(std::mem::take(&mut field));
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            record_index += 1;
            if b == b'\r' {
                i += 2;
            } else {
                i += 1;
            }
        } else if b == quote {
            if !field.is_empty() {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
            quote_start = i;
            in_quotes = true;
            i += 1;
        } else {
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if after_quote || !record.is_empty() || !field.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        if let Some(first) = header[..second]
            .iter()
            .position(|name| name == &header[second])
        {
            return Some(CsvError::DuplicateHeader {
                name: header[second].clone(),
                first,
                second,
            });
        }
    }
    None
}

fn table_from_records(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(ref header) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(err) = duplicate_header(header) {
                return Err(err);
            }
        }
        for (index, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote).map_err(|component| {
            CsvError::InvalidConfig { component }
        })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    (b'1'..=b'9').contains(&b[start]) && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.starts_with(b"-"));
    if b.len() < start + 4 {
        return false;
    }
    let dot = b.len() - 3;
    if b[dot] != b'.'
        || !b[dot + 1..].iter().all(u8::is_ascii_digit)
        || dot <= start
    {
        return false;
    }
    let integer = &b[start..dot];
    let integral_ok = if integer == b"0" {
        true
    } else {
        !integer.is_empty()
            && (b'1'..=b'9').contains(&integer[0])
            && integer[1..].iter().all(u8::is_ascii_digit)
    };
    integral_ok && !(start == 1 && integer == b"0" && b[dot + 1..] == *b"00")
}

fn evaluate_cell(
    text: Option<&String>,
    column: Option<usize>,
    spec: &ColumnSpec,
    row_index: usize,
    spec_index: usize,
) -> Result<Value, CellError> {
    let fail = |kind| CellError {
        row: row_index,
        spec: spec_index,
        column,
        kind,
    };
    let Some(text) = text else {
        return Err(fail(CellErrorKind::MissingCell));
    };
    if spec.nullable && text.is_empty() {
        return Ok(Value::Null);
    }
    if text.is_empty() && spec.kind != ColumnType::Text {
        return Err(fail(CellErrorKind::Empty));
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(text.clone())),
        ColumnType::Integer => {
            if !integer_syntax(text) {
                return Err(fail(CellErrorKind::InvalidInteger));
            }
            text.parse::<i64>()
                .map(Value::Integer)
                .map_err(|_| fail(CellErrorKind::IntegerOverflow))
        }
        ColumnType::Decimal => {
            if !decimal_syntax(text) {
                return Err(fail(CellErrorKind::InvalidDecimal));
            }
            let negative = text.starts_with('-');
            let unsigned = if negative { &text[1..] } else { text.as_str() };
            let dot = unsigned.len() - 3;
            let whole = unsigned[..dot].parse::<i128>().unwrap();
            let fraction = unsigned[dot + 1..].parse::<i128>().unwrap();
            let magnitude = whole * 100 + fraction;
            let signed = if negative { -magnitude } else { magnitude };
            i64::try_from(signed)
                .map(Value::Decimal)
                .map_err(|_| fail(CellErrorKind::DecimalOverflow))
        }
        ColumnType::Boolean => match text.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(fail(CellErrorKind::InvalidBoolean)),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };
    for (row_index, row) in table.rows.iter().enumerate() {
        let mut values = vec![None; columns.len()];
        let mut failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if failed {
                break;
            }
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let Some(column) = resolved else {
                output.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                failed = true;
                continue;
            };
            match evaluate_cell(
                row.get(column),
                Some(column),
                spec,
                row_index,
                spec_index,
            ) {
                Ok(value) => values[spec_index] = Some(value),
                Err(error) => {
                    output.errors.push(error);
                    failed = true;
                }
            }
            if failed && policy == ErrorPolicy::Stop {
                output.rows.push(values);
                return output;
            }
            if failed && policy == ErrorPolicy::FirstPerRow {
                break;
            }
        }
        output.rows.push(values);
    }
    output
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {

    Nul {

        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,

fn write_records(table: &Table) -> Vec<Vec<String>> {
    let mut records = Vec::new();
    if let Some(header) = &table.header {
        records.push(header.clone());
    }
    records.extend(table.rows.iter().cloned());
    records
}

fn field_needs_quotes(
    field: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole_empty: bool,
) -> bool {
    let bytes = field.as_bytes();
    bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_first_record && field.starts_with('\u{feff}'))
        || sole_empty
}

fn encode_field(
    output: &mut String,
    field: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole_empty: bool,
) {
    if field_needs_quotes(field, delimiter, quote, first_field_first_record, sole_empty) {
        output.push(quote as char);
        for ch in field.chars() {
            if ch as u32 == quote as u32 {
                output.push(quote as char);
            }
            output.push(ch);
        }
        output.push(quote as char);
    } else {
        output.push_str(field);
    }
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let original = write_records(table);
    for (record_index, record) in original.iter().enumerate() {
        for (column_index, field) in record.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, record) in original.iter().enumerate() {
        if record.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    if config.mode == WriteMode::Canonical
        && table.header.is_some()
        && table.duplicates == DuplicatePolicy::Reject
    {
        if let Some(CsvError::DuplicateHeader {
            name,
            first,
            second,
        }) = duplicate_header(table.header.as_ref().unwrap())
        {
            return Err(WriteError::DuplicateHeader {
                name,
                first,
                second,
            });
        }
    }

    let records = match (config.mode, table.header.as_ref()) {
        (WriteMode::Canonical, Some(header)) => {
            let mut chosen = Vec::<(String, usize)>::new();
            for (index, name) in header.iter().enumerate() {
                if let Some(pos) = chosen.iter().position(|(existing, _)| existing == name) {
                    if table.duplicates == DuplicatePolicy::KeepLast {
                        chosen[pos].1 = index;
                    }
                } else {
                    chosen.push((name.clone(), index));
                }
            }
            chosen.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

            let new_header: Vec<String> = chosen.iter().map(|(name, _)| name.clone()).collect();
            let projected_rows: Vec<Vec<String>> = table
                .rows
                .iter()
                .map(|row| chosen.iter().map(|(_, index)| row[*index].clone()).collect())
                .collect();
            let mut rows = vec![new_header];
            let mut sorted = projected_rows;
            sorted.sort_by(|a, b| compare_rows(a, b));
            rows.extend(sorted);
            rows
        }
        (WriteMode::Canonical, None) => {
            let mut rows = table.rows.clone();
            rows.sort_by(|a, b| compare_rows(a, b));
            rows
        }
        (WriteMode::Preserve, _) => original,
    };

    if records.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let terminator: &str = if canonical || config.line_ending == LineEnding::Lf {
        "\n"
    } else {
        "\r\n"
    };

    for (record_index, record) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(terminator);
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            encode_field(
                &mut output,
                field,
                config.delimiter,
                config.quote,
                record_index == 0 && column_index == 0,
                record.len() == 1 && field.is_empty(),
            );
        }
    }

    if canonical || config.final_terminator {
        output.push_str(terminator);
    }
    Ok(output)
}

fn compare_rows(left: &[String], right: &[String]) -> std::cmp::Ordering {
    for (a, b) in left.iter().zip(right.iter()) {
        let ordering = a.as_bytes().cmp(b.as_bytes());
        if ordering != std::cmp::Ordering::Equal {
            return ordering;
        }
    }
    left.len().cmp(&right.len())
}
