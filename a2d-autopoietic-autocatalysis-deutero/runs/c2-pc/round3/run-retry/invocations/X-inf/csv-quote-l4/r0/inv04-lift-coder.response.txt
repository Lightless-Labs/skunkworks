#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut record_started = false;
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut quote_start = 0usize;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        // Outside quotes, malformed CR takes precedence over state checks.
        if state != 2 && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        if state == 2 {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                } else {
                    state = 3;
                    i += 1;
                }
            } else {
                let start = i;
                i += 1;
                while i < bytes.len() && bytes[i] != quote && bytes[i] != 0 {
                    i += 1;
                }
                field.push_str(&input[start..i]);
            }
            record_started = true;
            continue;
        }

        if state == 3 {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                record_started = true;
                state = 0;
                i += 1;
            } else if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record_started = false;
                state = 0;
                record_index += 1;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte {
                    record: record_index,
                    byte: i,
                });
            }
            continue;
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            record_started = true;
            state = 0;
            i += 1;
        } else if b == b'\n' || (b == b'\r' && bytes.get(i + 1) == Some(&b'\n')) {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record_started = false;
            state = 0;
            record_index += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if state == 0 {
                quote_start = i;
                state = 2;
                record_started = true;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
        } else {
            let start = i;
            i += 1;
            while i < bytes.len()
                && bytes[i] != delimiter
                && bytes[i] != quote
                && bytes[i] != b'\n'
                && bytes[i] != b'\r'
                && bytes[i] != 0
            {
                i += 1;
            }
            field.push_str(&input[start..i]);
            state = 1;
            record_started = true;
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if record_started || !row.is_empty() || !field.is_empty() || state == 3 {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|n| n == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|n| n == name)
            }
        }
    }
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        if let Some(first) = header[..second].iter().position(|v| v == &header[second]) {
            return Some(CsvError::DuplicateHeader {
                name: header[second].clone(),
                first,
                second,
            });
        }
    }
    None
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers && !records.is_empty() {
        (Some(records[0].clone()), records[1..].to_vec())
    } else {
        (None, records)
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(err) = duplicate_header(h) {
                return Err(err);
            }
        }
        for (index, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return b.len() == start + 1 && start == 0;
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if b.len() < start + 4 {
        return false;
    }
    let dot = b.len() - 3;
    if b[dot] != b'.' || !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    let integer = &b[start..dot];
    if integer.is_empty() {
        return false;
    }
    if integer[0] == b'0' {
        if integer.len() != 1 {
            return false;
        }
    } else if !(b'1'..=b'9').contains(&integer[0])
        || !integer[1..].iter().all(u8::is_ascii_digit)
    {
        return false;
    }
    !(start == 1 && integer == b"0" && &b[dot + 1..] == b"00")
}

fn evaluate_cell(s: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if s.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(s.to_string())),
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
        ColumnType::Integer => {
            if !integer_syntax(s) {
                return Err(CellErrorKind::InvalidInteger);
            }
            match s.parse::<i64>() {
                Ok(v) => Ok(Value::Integer(v)),
                Err(_) => Err(CellErrorKind::IntegerOverflow),
            }
        }
        ColumnType::Decimal => {
            if !decimal_syntax(s) {
                return Err(CellErrorKind::InvalidDecimal);
            }
            let negative = s.starts_with('-');
            let unsigned = if negative { &s[1..] } else { s };
            let dot = unsigned.len() - 3;
            let whole = unsigned[..dot]
                .parse::<i128>()
                .map_err(|_| CellErrorKind::DecimalOverflow)?;
            let frac = unsigned[dot + 1..]
                .parse::<i128>()
                .map_err(|_| CellErrorKind::DecimalOverflow)?;
            let magnitude = whole
                .checked_mul(100)
                .and_then(|v| v.checked_add(frac))
                .ok_or(CellErrorKind::DecimalOverflow)?;
            let signed = if negative { -magnitude } else { magnitude };
            i64::try_from(signed)
                .map(Value::Decimal)
                .map_err(|_| CellErrorKind::DecimalOverflow)
        }
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };
    let mut stopped = false;

    for (row_index, row) in table.rows.iter().enumerate() {
        if stopped {
            break;
        }
        let mut output = vec![None; columns.len()];

        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let (column, value) = match resolved {
                None => (None, Err(CellErrorKind::UnknownColumn)),
                Some(index) => match row.get(index) {
                    None => (Some(index), Err(CellErrorKind::MissingCell)),
                    Some(cell) => (Some(index), evaluate_cell(cell, spec)),
                },
            };

            match value {
                Ok(value) => output[spec_index] = Some(value),
                Err(kind) => {
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column,
                        kind,
                    });
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow => break,
                        ErrorPolicy::Stop => {
                            stopped = true;
                            break;
                        }
                    }
                }
            }
        }
        result.rows.push(output);
    }
    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,

        second: usize,

fn quote_field(field: &str, delimiter: u8, quote: u8, first: bool, sole: bool) -> String {
    let bytes = field.as_bytes();
    let needs_quote = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first && field.starts_with('\u{feff}'))
        || (sole && field.is_empty());

    if !needs_quote {
        return field.to_string();
    }

    let mut output = String::with_capacity(field.len() + 2);
    output.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            output.push(quote as char);
        }
        output.push(ch);
    }
    output.push(quote as char);
    output
}

fn row_has_nul(row: &[String], record: usize) -> Option<WriteError> {
    for (column, field) in row.iter().enumerate() {
        if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
            return Some(WriteError::Nul {
                record,
                column,
                byte,
            });
        }
    }
    None
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    if let Some(header) = &table.header {
        if let Some(error) = row_has_nul(header, 0) {
            return Err(error);
        }
    }
    let data_record_start = usize::from(table.header.is_some());
    for (index, row) in table.rows.iter().enumerate() {
        if let Some(error) = row_has_nul(row, index + data_record_start) {
            return Err(error);
        }
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    if let Some(header) = &table.header {
        if header.is_empty() {
            return Err(WriteError::ZeroColumns { record: 0 });
        }
    }
    for (index, row) in table.rows.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: index + data_record_start,
            });
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        let mut selected: Vec<usize> = Vec::new();
        if config.mode == WriteMode::Preserve {
            selected.extend(0..header.len());
        } else {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(CsvError::DuplicateHeader { name, first, second }) =
                    duplicate_header(header)
                {
                    return Err(WriteError::DuplicateHeader {
                        name,
                        first,
                        second,
                    });
                }
            }
            for index in 0..header.len() {
                let retain = match table.duplicates {
                    DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                        header[..index].iter().all(|name| name != &header[index])
                    }
                    DuplicatePolicy::KeepLast => {
                        header[index + 1..].iter().all(|name| name != &header[index])
                    }
                };
                if retain {
                    selected.push(index);
                }
            }
            selected.sort_by(|a, b| header[*a].as_bytes().cmp(header[*b].as_bytes()));
        }

        records.push(selected.iter().map(|i| header[*i].clone()).collect());
        if config.mode == WriteMode::Preserve {
            records.extend(table.rows.iter().cloned());
        } else {
            let mut projected: Vec<Vec<String>> = table
                .rows
                .iter()
                .map(|row| selected.iter().map(|i| row[*i].clone()).collect())
                .collect();
            projected.sort_by(|a, b| compare_rows(a, b));
            records.extend(projected);
        }
    } else {
        let mut rows = table.rows.clone();
        if config.mode == WriteMode::Canonical {
            rows.sort_by(|a, b| compare_rows(a, b));
        }
        records = rows;
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let ending = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };
    let mut output = String::new();

    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(ending);
        }
        let sole = row.len() == 1;
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            let first_output_field = record_index == 0 && column_index == 0;
            output.push_str(&quote_field(
                field,
                config.delimiter,
                config.quote,
                first_output_field,
                sole,
            ));
        }
    }

    if canonical || config.final_terminator {
        output.push_str(ending);
    }

    Ok(output)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let order = left.as_bytes().cmp(right.as_bytes());
        if order != std::cmp::Ordering::Equal {
            return order;
        }
    }
    a.len().cmp(&b.len())
}
