    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {

        task: usize,
        position: usize,
        value: usize,

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate all dependency entries before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Kahn traversal with the smallest available ID first.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_index = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(index, _)| index)
            .unwrap();
        let id = available.swap_remove(smallest_index);
        order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for id in 0..n {
            if indegree[id] != 0 {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        finish: u64,
        worker: usize,
    }

    let workers = cooldowns.len();
    let mut worker_available = vec![0u64; workers];
    let mut running = Vec::<Running>::new();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut used = vec![0u64; capacity.len()];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut makespan = 0u64;
    let mut trace = format!("workers={};resources={}\n", workers, capacity.len());

    fn append_usage(trace: &mut String, used: &[u64]) {
        if used.is_empty() {
            trace.push('-');
        } else {
            for (index, value) in used.iter().enumerate() {
                if index != 0 {
                    trace.push(',');
                }
                trace.push_str(&value.to_string());
            }
        }
    }

    let mut time = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete all tasks finishing now in ascending task ID.
        let mut finishing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, item)| (item.finish == time).then_some(index))
            .collect();
        finishing.sort_by_key(|&index| running[index].task);

        for index in finishing.into_iter().rev() {
            let item = running.remove(index);
            for (resource, &need) in tasks[item.task].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }
            completed[item.task] = true;
            completed_count += 1;
            makespan = makespan.max(item.finish);

            trace.push_str(&format!(
                "{}|end|{}|{}|",
                time, item.task, item.worker
            ));
            append_usage(&mut trace, &used);
            trace.push('\n');
        }

        loop {
            let mut candidates = Vec::<usize>::new();

            for id in 0..n {
                if started[id] || tasks[id].release > time {
                    continue;
                }
                if tasks[id].deps.iter().any(|&dep| !completed[dep]) {
                    continue;
                }

                let fits = tasks[id].needs.iter().enumerate().all(|(resource, &need)| {
                    u64::from(need) <= u64::from(capacity[resource]) - used[resource]
                });
                if fits {
                    candidates.push(id);
                }
            }

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let mut chosen_worker = None;
            for offset in 0..workers {
                let worker = (task_id % workers + offset) % workers;
                if worker_available[worker] <= time {
                    chosen_worker = Some(worker);
                    break;
                }
            }
            let Some(worker) = chosen_worker else {
                break;
            };

            let finish = time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: time,
                finish,
                worker,
            });

            trace.push_str(&format!("{}|start|{}|{}|", time, task_id, worker));
            append_usage(&mut trace, &used);
            trace.push('\n');

            if finish == time {
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                trace.push_str(&format!("{}|end|{}|{}|", time, task_id, worker));
                append_usage(&mut trace, &used);
                trace.push('\n');
            } else {
                running.push(Running {
                    task: task_id,
                    finish,
                    worker,
                });
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        for item in &running {
            consider(item.finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        time = next_time.expect("unfinished work always has a future event");
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task was started"))
        .collect();

    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest available ID chosen each time.
    let mut available = Vec::<usize>::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(task_id);
        }
    }
    let mut order = Vec::with_capacity(n);
    let mut degrees = indegree;
    while !available.is_empty() {
        let smallest_index = available
            .iter()
            .enumerate()
            .min_by_key(|(_, task_id)| **task_id)
            .map(|(index, _)| index)
            .unwrap();
        let task_id = available.swap_remove(smallest_index);
        order.push(task_id);
        for &child in &successors[task_id] {
            degrees[child] -= 1;
            if degrees[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut removed = vec![false; n];
            for &task_id in &order {
                removed[task_id] = true;
            }
            removed
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let successor_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(successor_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = String::new();
    trace.push_str(&format!(
        "workers={worker_count};resources={}\n",
        capacity.len()
    ));

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors = tasks.iter().map(|task| task.deps.len()).collect::<Vec<_>>();
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut used = vec![0u128; capacity.len()];
    let mut now = 0u64;
    let mut completion_count = 0usize;
    let mut makespan = 0u64;

    while completion_count < n {
        // Complete every task finishing now, in ascending task ID.
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|item| item.finish == now)
            .collect();
        finishing.sort_by_key(|item| item.task);

        if !finishing.is_empty() {
            running.retain(|item| item.finish != now);
            for item in finishing {
                let task_id = item.task;
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u128::from(need);
                }
                completed[task_id] = true;
                completion_count += 1;
                makespan = makespan.max(now);

                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }

                trace.push_str(&format!("{now}|end|{task_id}|{}|", item.worker));
                append_usage(&mut trace, &used);
                trace.push('\n');
            }
        }

        // Admit one task at a time, re-ranking after each admission and
        // immediate completion of zero-duration tasks.
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > now
                    || remaining_predecessors[task_id] != 0
                    || !task_fits(&tasks[task_id], &used, capacity)
                {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(current) => {
                        tails[task_id] > tails[current]
                            || (tails[task_id] == tails[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                    }
                };
                if better {
                    best = Some(task_id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            // Choose an available worker by cyclic preference.
            let chosen_worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= now);

            let Some(worker) = chosen_worker else {
                break;
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });

            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] += u128::from(need);
            }

            trace.push_str(&format!("{now}|start|{task_id}|{worker}|"));
            append_usage(&mut trace, &used);
            trace.push('\n');

            running.push(Running {
                task: task_id,
                worker,
                finish,
            });

            if finish == now {
                // Zero-duration completions happen immediately before the
                // next task is selected.
                let position = running
                    .iter()
                    .position(|item| item.task == task_id)
                    .unwrap();
                running.remove(position);

                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u128::from(need);
                }
                completed[task_id] = true;
                completion_count += 1;
                makespan = makespan.max(now);

                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }

                trace.push_str(&format!("{now}|end|{task_id}|{worker}|"));
                append_usage(&mut trace, &used);
                trace.push('\n');
            }
        }

        if completion_count == n {
            break;
        }

        // Advance to the earliest relevant future event.
        let mut next = None;
        for item in &running {
            if item.finish > now {
                next = Some(next.map_or(item.finish, |value: u64| value.min(item.finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                next = Some(next.map_or(tasks[task_id].release, |value| {
                    value.min(tasks[task_id].release)
                }));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next = Some(next.map_or(available_at, |value| value.min(available_at)));
            }
        }

        // For a valid acyclic graph, an unstarted task always has a future
        // release, predecessor completion, or worker availability event.
        now = next.expect("unfinished tasks must have a future event");
    }

    trace.push_str(&format!("makespan={makespan}\n"));

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

        position: usize,

fn resource_snapshot(used: &[u128]) -> String {
    if used.is_empty() {
        return "-".to_string();
    }

    let mut result = String::new();
    for (i, value) in used.iter().enumerate() {
        if i != 0 {
            result.push(',');
        }
        result.push_str(&value.to_string());
    }
    result
}

fn append_event(
    trace: &mut String,
    time: u64,
    event: &str,
    task: usize,
    worker: usize,
    used: &[u128],

    use std::fmt::Write;
    let _ = writeln!(
        trace,
        "{}|{}|{}|{}|{}",
        time,
        event,
        task,
        worker,
        resource_snapshot(used)
    );
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry in the required order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    // Validate resource shapes and demands task by task.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Build graph edges from predecessors to their dependents.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available = std::collections::BTreeSet::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree;
    while let Some(id) = available.pop_first() {
        order.push(id);
        for &child in &children[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut flags = vec![false; n];
            for &id in &order {
                flags[id] = true;
            }
            flags
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut running = vec![false; n];
    let mut worker_available_at = vec![0u64; worker_count];
    let mut used = vec![0u128; capacity.len()];
    let mut completed_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    let mut trace = String::new();
    use std::fmt::Write;
    let _ = writeln!(
        trace,
        "workers={};resources={}",
        worker_count,
        capacity.len()
    );

    while completed_count < n {
        // Complete all tasks ending now, in task ID order.
        for id in 0..n {
            if running[id]
                && runs[id]
                    .as_ref()
                    .is_some_and(|run| run.finish == now)
            {
                running[id] = false;
                completed[id] = true;
                completed_count += 1;

                let run = runs[id].as_ref().unwrap();
                for (resource, &need) in tasks[id].needs.iter().enumerate() {
                    used[resource] -= u128::from(need);
                }
                append_event(&mut trace, now, "end", id, run.worker, &used);
            }
        }

        loop {
            let mut chosen_task: Option<usize> = None;

            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if !tasks[id].deps.iter().all(|&pred| completed[pred]) {
                    continue;
                }
                if tasks[id]
                    .needs
                    .iter()
                    .enumerate()
                    .any(|(r, &need)| u128::from(need) > u128::from(capacity[r]) - used[r])
                {
                    continue;
                }

                let better = match chosen_task {
                    None => true,
                    Some(current) => {
                        tail[id] > tail[current]
                            || (tail[id] == tail[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if better {
                    chosen_task = Some(id);
                }
            }

            let Some(id) = chosen_task else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (id % worker_count + offset) % worker_count)
                .find(|&w| worker_available_at[w] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[id].duration)
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Turnaround,
                })?;

            worker_available_at[worker] = available_at;
            started[id] = true;
            running[id] = true;

            for (resource, &need) in tasks[id].needs.iter().enumerate() {
                used[resource] += u128::from(need);
            }

            runs[id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            append_event(&mut trace, now, "start", id, worker, &used);

            // A zero-duration task completes immediately before another admission.
            if finish == now {
                running[id] = false;
                completed[id] = true;
                completed_count += 1;
                for (resource, &need) in tasks[id].needs.iter().enumerate() {
                    used[resource] -= u128::from(need);
                }
                append_event(&mut trace, now, "end", id, worker, &used);
            } else {
                makespan = makespan.max(finish);
            }
        }

        if completed_count == n {
            break;
        }

        // Move to the earliest future event that can change eligibility.
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now && next_time.is_none_or(|current| time < current) {
                next_time = Some(time);
            }
        };

        for id in 0..n {
            if running[id] {
                consider(runs[id].as_ref().unwrap().finish);
            } else if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &available_at in &worker_available_at {
            consider(available_at);
        }

        // If a task is eligible but workers are busy, their cooldown expiries
        // are already considered. Future releases and finishes are also covered.
        if let Some(next) = next_time {
            now = next;
        } else {
            // With a DAG and finite task releases, this is only reachable for
            // an empty schedule, handled by the loop condition.
            break;
        }
    }

    for run in runs.iter().flatten() {
        makespan = makespan.max(run.finish);
    }

    let _ = writeln!(trace, "makespan={}", makespan);

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}
