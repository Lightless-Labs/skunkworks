fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_version(c: Constraint) -> Version {
    match c {
        Constraint::Exact(v)
        | Constraint::AtLeast(v)
        | Constraint::Before(v)
        | Constraint::Caret(v) => v,
    }
}

fn constraint_cmp(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    constraint_rank(*a)
        .cmp(&constraint_rank(*b))
        .then_with(|| constraint_version(*a).cmp(&constraint_version(*b)))
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort_by(constraint_cmp);
    out.dedup();
    out
}

fn constraint_text(c: Constraint) -> String {
    let prefix = match c {
        Constraint::Exact(_) => "=",
        Constraint::AtLeast(_) => ">=",
        Constraint::Before(_) => "<",
        Constraint::Caret(_) => "^",
    };
    format!("{prefix}{}", version_text(constraint_version(c)))
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn all_match(cs: &[Constraint], v: Version) -> bool {
    cs.iter().all(|c| matches(*c, v))
}

fn enc(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> std::cmp::Ordering {
    constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut e = evidence.to_vec();
    e.sort_by(evidence_cmp);
    e
}

fn evidence_lines(evidence: &[Evidence]) -> String {
    let mut out = String::new();
    for e in sorted_evidence(evidence) {
        out.push_str("\n  C:");
        out.push_str(&constraint_text(e.constraint));
        out.push_str(" via ");
        out.push_str(&e.path.join(" -> "));
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", enc(site), enc(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", enc(package))
        }
        ResolveError::DuplicateVersion { package, version } => {
            format!(
                "duplicate-version|package={}|version={}",
                enc(package),
                version_text(*version)
            )
        }
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            enc(package),
            version_text(*version),
            enc(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", enc(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            enc(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            enc(package),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            enc(package),
            version_text(*selected),
            constraints_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            enc(package),
            version_text(*selected),
            enc(feature),
            enc(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn platform_text(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn platform_matches(p: Platform, t: Target) -> bool {
    match p {
        Platform::Any => true,
        Platform::Linux => t == Target::Linux,
        Platform::Mac => t == Target::Mac,
        Platform::Windows => t == Target::Windows,
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Closure {
    constraints: std::collections::BTreeMap<String, Vec<Constraint>>,
    requests: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
    origins: std::collections::BTreeMap<String, Vec<Origin>>,
    paths: std::collections::BTreeMap<String, Vec<String>>,
    activated: std::collections::BTreeMap<String, std::collections::BTreeSet<String>>,
}

fn shortest_path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn make_root_token(index: usize, r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{index}]:{}@{}+{{{}}}",
        r.package,
        constraint_text(r.constraint),
        fs.join(",")
    )
}

fn make_dep_token(source: &str, version: Version, index: usize, d: &Dependency) -> String {
    let mut fs = d.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        d.package,
        constraint_text(d.constraint),
        fs.join(","),
        d.gate.as_deref().unwrap_or("*"),
        platform_text(d.platform)
    )
}

fn feature_decl<'a>(release: &'a Release, name: &str) -> Option<&'a Feature> {
    release.features.iter().find(|f| f.name == name)
}

fn compute_closure(
    catalog: &[Package],
    roots: &[Requirement],
    selected: &std::collections::BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let packages: std::collections::BTreeMap<&str, &Package> =
        catalog.iter().map(|p| (p.name.as_str(), p)).collect();
    let mut c = Closure::default();

    for (i, root) in roots.iter().enumerate() {
        c.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        c.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        c.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![make_root_token(i, root)],
            });
        let path = vec![make_root_token(i, root)];
        match c.paths.get_mut(&root.package) {
            Some(old) if shortest_path_better(&path, old) => *old = path,
            None => {
                c.paths.insert(root.package.clone(), path);
            }
            _ => {}
        }
    }

    loop {
        let mut changed = false;
        let names: Vec<String> = selected.keys().cloned().collect();

        for name in names {
            let Some(&version) = selected.get(&name) else {
                continue;
            };
            let Some(package) = packages.get(name.as_str()) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let reqs = c.requests.get(&name).cloned().unwrap_or_default();
            let mut activated = c.activated.get(&name).cloned().unwrap_or_default();

            loop {
                let next = reqs.iter().find(|f| !activated.contains(*f)).cloned();
                let Some(feature_name) = next else { break };
                activated.insert(feature_name.clone());
                if let Some(feature) = feature_decl(release, &feature_name) {
                    for enabled in &feature.enables {
                        if c.requests
                            .entry(name.clone())
                            .or_default()
                            .insert(enabled.clone())
                        {
                            changed = true;
                        }
                    }
                }
                changed = true;
            }
            c.activated.insert(name.clone(), activated.clone());

            for (index, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep.gate.as_ref().is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }

                let token = make_dep_token(&name, version, index, dep);
                let mut path = c.paths.get(&name).cloned().unwrap_or_default();
                path.push(token);

                c.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                c.requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());

                match c.paths.get_mut(&dep.package) {
                    Some(old) if shortest_path_better(&path, old) => {
                        *old = path;
                        changed = true;
                    }
                    None => {
                        c.paths.insert(dep.package.clone(), path);
                        changed = true;
                    }
                    _ => {}
                }
            }
        }

        if !changed {
            break;
        }
    }

    c.origins.clear();
    for (i, root) in roots.iter().enumerate() {
        c.origins
            .entry(root.package.clone())
            .or_default()
            .push(Origin {
                constraint: root.constraint,
                path: vec![make_root_token(i, root)],
            });
    }
    for (name, version) in selected {
        let Some(package) = packages.get(name.as_str()) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        let Some(source_path) = c.paths.get(name) else {
            continue;
        };
        let activated = c.activated.get(name);
        for (index, dep) in release.deps.iter().enumerate() {
            if !platform_matches(dep.platform, target)
                || dep
                    .gate
                    .as_ref()
                    .is_some_and(|g| !activated.is_some_and(|fs| fs.contains(g)))
            {
                continue;
            }
            let mut path = source_path.clone();
            path.push(make_dep_token(name, *version, index, dep));
            c.origins
                .entry(dep.package.clone())
                .or_default()
                .push(Origin {
                    constraint: dep.constraint,
                    path,
                });
        }
    }

    for values in c.constraints.values_mut() {
        *values = canonical_constraints(values);
    }
    c
}

fn catalog_map<'a>(catalog: &'a [Package]) -> std::collections::BTreeMap<&'a str, &'a Package> {
    catalog.iter().map(|p| (p.name.as_str(), p)).collect()
}

fn evidence_for(c: &Closure, package: &str) -> Vec<Evidence> {
    let mut out = c
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    out.sort_by(evidence_cmp);
    out
}

struct SearchResult {
    best: Option<std::collections::BTreeMap<String, Version>>,
    cost: usize,
    first_error: Option<ResolveError>,
}

fn lock_cost(locks: &[Lock], selected: &std::collections::BTreeMap<String, Version>) -> usize {
    locks
        .iter()
        .filter(|l| selected.get(&l.package) != Some(&l.version))
        .count()
}

fn vector_better(
    a: &std::collections::BTreeMap<String, Version>,
    b: &std::collections::BTreeMap<String, Version>,
    names: &[String],
) -> bool {
    for name in names {
        match (a.get(name), b.get(name)) {
            (None, None) => {}
            (None, Some(_)) => return false,
            (Some(_), None) => return true,
            (Some(av), Some(bv)) if av != bv => return av > bv,
            _ => {}
        }
    }
    false
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    selected: &mut std::collections::BTreeMap<String, Version>,
    attempts: &mut usize,
    max_attempts: usize,
    package_map: &std::collections::BTreeMap<&str, &Package>,
    all_names: &[String],
    result: &mut SearchResult,
) -> Result<(), ResolveError> {
    let closure = compute_closure(catalog, roots, selected, target);
    let active: Vec<String> = closure.constraints.keys().cloned().collect();

    for name in &active {
        if !package_map.contains_key(name.as_str()) {
            let e = ResolveError::Missing {
                package: name.clone(),
                evidence: evidence_for(&closure, name),
            };
            if result.first_error.is_none() {
                result.first_error = Some(e);
            }
            return Ok(());
        }
    }

    for name in selected.keys() {
        let Some(&version) = selected.get(name) else {
            continue;
        };
        let Some(package) = package_map.get(name.as_str()) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == version) else {
            continue;
        };
        let fs = closure.activated.get(name).cloned().unwrap_or_default();
        for feature in fs {
            if feature_decl(release, &feature).is_none() {
                let e = ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: version,
                    feature,
                    path: closure.paths.get(name).cloned().unwrap_or_default(),
                };
                if result.first_error.is_none() {
                    result.first_error = Some(e);
                }
                return Ok(());
            }
        }
    }

    for (name, version) in selected.iter() {
        if let Some(cs) = closure.constraints.get(name) {
            if !all_match(cs, *version) {
                let e = ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: cs.clone(),
                    evidence: evidence_for(&closure, name),
                };
                if result.first_error.is_none() {
                    result.first_error = Some(e);
                }
                return Ok(());
            }
        }
    }

    for name in &active {
        if !selected.contains_key(name) {
            let package = package_map[name.as_str()];
            let cs = closure.constraints.get(name).map(Vec::as_slice).unwrap_or(&[]);
            let has_match = package
                .releases
                .iter()
                .any(|r| all_match(cs, r.version));
            if !has_match {
                let e = ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs.to_vec(),
                    evidence: evidence_for(&closure, name),
                };
                if result.first_error.is_none() {
                    result.first_error = Some(e);
                }
                return Ok(());
            }
        }
    }

    let next = active.iter().find(|name| !selected.contains_key(*name)).cloned();
    let Some(name) = next else {
        let cost = lock_cost(locks, selected);
        let replace = match &result.best {
            None => true,
            Some(best) => {
                cost < result.cost || (cost == result.cost && vector_better(selected, best, all_names))
            }
        };
        if replace {
            result.best = Some(selected.clone());
            result.cost = cost;
        }
        return Ok(());
    };

    let package = package_map[name.as_str()];
    let cs = closure.constraints.get(&name).map(Vec::as_slice).unwrap_or(&[]);
    let mut candidates = package
        .releases
        .iter()
        .filter(|r| all_match(cs, r.version))
        .map(|r| r.version)
        .collect::<Vec<_>>();
    candidates.sort_by(|a, b| b.cmp(a));
    candidates.dedup();

    if let Some(lock) = locks.iter().find(|l| l.package == name) {
        if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
            let v = candidates.remove(pos);
            candidates.insert(0, v);
        }
    }

    for version in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        selected.insert(name.clone(), version);
        let child = search(
            catalog,
            roots,
            locks,
            target,
            selected,
            attempts,
            max_attempts,
            package_map,
            all_names,
            result,
        );
        selected.remove(&name);
        child?;
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].gate"),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].feature[{f}]"),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].feature[{f}]"),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].feature[{f}].enable[{e}]"),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = std::collections::BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let package_map = catalog_map(catalog);
    let mut all_names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    all_names.sort();

    let mut selected = std::collections::BTreeMap::new();
    let mut attempts = 0usize;
    let mut result = SearchResult {
        best: None,
        cost: usize::MAX,
        first_error: None,
    };

    search(
        catalog,
        roots,
        locks,
        target,
        &mut selected,
        &mut attempts,
        max_attempts,
        &package_map,
        &all_names,
        &mut result,
    )?;

    let Some(best) = result.best else {
        return Err(result.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let selected_out = best
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let final_closure = compute_closure(catalog, roots, &best, target);
    let mut before: std::collections::BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let mut after = best.clone();
    let mut names = std::collections::BTreeSet::new();
    names.extend(before.keys().cloned());
    names.extend(after.keys().cloned());
    let mut changes = Vec::new();

    for name in names {
        let b = before.remove(&name);
        let a = after.remove(&name);
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = package_map
                    .get(name.as_str())
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_closure
                    .constraints
                    .get(&name)
                    .is_some_and(|cs| all_match(cs, old))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}
