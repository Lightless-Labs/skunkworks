fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn enc(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn evidence_sorted(ev: &[Evidence]) -> Vec<Evidence> {
    let mut out = ev.to_vec();
    out.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(mut s: String, ev: &[Evidence]) -> String {
    for e in evidence_sorted(ev) {
        s.push_str("\n  C:");
        s.push_str(&constraint_text(e.constraint));
        s.push_str(" via ");
        s.push_str(&e.path.join(" -> "));
    }
    s
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", enc(site), enc(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", enc(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            enc(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            enc(package),
            version_text(*version),
            enc(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", enc(package))
        }
        ResolveError::Missing { package, evidence } => append_evidence(
            format!("missing|package={}", enc(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "no-match|package={}|constraints={}",
                enc(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                enc(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            enc(package),
            version_text(*selected),
            enc(feature),
            enc(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn matches_all(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| matches_constraint(v, *c))
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: Vec<Origin>,
    requests: BTreeSet<String>,
    features: BTreeSet<String>,
    paths: Vec<Vec<String>>,
}

use std::collections::{BTreeMap, BTreeSet, HashMap};

struct Solver<'a> {
    catalog: &'a [Package],
    package_index: HashMap<String, usize>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    budget: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, BTreeMap<String, Version>)>,
}

impl<'a> Solver<'a> {
    fn release(&self, package: &str, version: Version) -> Option<&'a Release> {
        let i = *self.package_index.get(package)?;
        self.catalog[i]
            .releases
            .iter()
            .find(|r| r.version == version)
    }

    fn active(&self, selected: &BTreeMap<String, Version>) -> BTreeMap<String, Active> {
        let mut active: BTreeMap<String, Active> = BTreeMap::new();
        for (i, root) in self.roots.iter().enumerate() {
            let feature_names = sorted_names(&root.features);
            let token = format!(
                "root[{}]:{}@{}+{{{}}}",
                i,
                root.package,
                constraint_text(root.constraint),
                feature_names.join(",")
            );
            let item = active.entry(root.package.clone()).or_default();
            item.origins.push(Origin {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
            item.requests.extend(root.features.iter().cloned());
            item.paths.push(vec![token]);
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = selected.keys().cloned().collect();
            for source in names {
                let version = selected[&source];
                let Some(release) = self.release(&source, version) else {
                    continue;
                };
                let current_requests = active
                    .get(&source)
                    .map(|a| a.requests.clone())
                    .unwrap_or_default();

                let mut closure = current_requests.clone();
                let mut queue: Vec<String> = current_requests.into_iter().collect();
                let mut cursor = 0;
                while cursor < queue.len() {
                    let f = queue[cursor].clone();
                    cursor += 1;
                    if let Some(decl) = release.features.iter().find(|x| x.name == f) {
                        for en in &decl.enables {
                            if closure.insert(en.clone()) {
                                queue.push(en.clone());
                            }
                        }
                    }
                }

                {
                    let item = active.entry(source.clone()).or_default();
                    let old = item.features.len();
                    item.features.extend(closure.iter().cloned());
                    changed |= item.features.len() != old;
                }

                for (j, dep) in release.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, self.target)
                        || dep.gate.as_ref().is_some_and(|g| !closure.contains(g))
                    {
                        continue;
                    }
                    let feature_names = sorted_names(&dep.features);
                    let gate = dep.gate.as_deref().unwrap_or("*");
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        source,
                        version_text(version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        feature_names.join(","),
                        gate,
                        platform_text(dep.platform)
                    );
                    let source_paths = active
                        .get(&source)
                        .map(|a| a.paths.clone())
                        .unwrap_or_default();
                    let item = active.entry(dep.package.clone()).or_default();
                    let old_origins = item.origins.len();
                    for path in source_paths {
                        let mut p = path;
                        p.push(token.clone());
                        item.origins.push(Origin {
                            constraint: dep.constraint,
                            path: p.clone(),
                        });
                        item.paths.push(p);
                    }
                    changed |= item.origins.len() != old_origins;
                    let old_requests = item.requests.len();
                    item.requests.extend(dep.features.iter().cloned());
                    changed |= item.requests.len() != old_requests;
                }
            }
            if !changed {
                break;
            }
        }

        for item in active.values_mut() {
            item.paths.sort_by(|a, b| {
                a.len().cmp(&b.len()).then_with(|| a.cmp(b))
            });
            item.paths.dedup();
        }
        active
    }

    fn inspect(
        &self,
        selected: &BTreeMap<String, Version>,
        active: &BTreeMap<String, Active>,
    ) -> Result<(), ResolveError> {
        for package in active.keys() {
            if !self.package_index.contains_key(package) {
                return Err(ResolveError::Missing {
                    package: package.clone(),
                    evidence: evidence(active.get(package)),
                });
            }
        }

        for (package, item) in active {
            if let Some(&version) = selected.get(package) {
                if let Some(release) = self.release(package, version) {
                    let activated = &item.features;
                    for feature in activated {
                        if !release.features.iter().any(|f| &f.name == feature) {
                            return Err(ResolveError::MissingFeature {
                                package: package.clone(),
                                selected: version,
                                feature: feature.clone(),
                                path: canonical_path(item),
                            });
                        }
                    }
                }
            }
        }

        for (package, version) in selected {
            if let Some(item) = active.get(package) {
                let cs = canonical_constraints(
                    &item.origins.iter().map(|o| o.constraint).collect::<Vec<_>>(),
                );
                if !matches_all(*version, &cs) {
                    return Err(ResolveError::Conflict {
                        package: package.clone(),
                        selected: *version,
                        requirements: cs,
                        evidence: evidence(Some(item)),
                    });
                }
            }
        }

        for (package, item) in active {
            if selected.contains_key(package) {
                continue;
            }
            let cs = canonical_constraints(
                &item.origins.iter().map(|o| o.constraint).collect::<Vec<_>>(),
            );
            let idx = self.package_index[package];
            if !self.catalog[idx]
                .releases
                .iter()
                .any(|r| matches_all(r.version, &cs))
            {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: cs,
                    evidence: evidence(Some(item)),
                });
            }
        }
        Ok(())
    }

    fn search(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = self.active(selected);
        if let Err(err) = self.inspect(selected, &active) {
            if self.first_error.is_none() {
                self.first_error = Some(err);
            }
            return Ok(());
        }

        let next = active.keys().find(|p| !selected.contains_key(*p)).cloned();
        let Some(package) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|l| selected.get(&l.package) != Some(&l.version))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_versions)) => {
                    cost < *best_cost
                        || (cost == *best_cost
                            && lex_vector_cmp(selected, best_versions, &self.package_names())
                                == std::cmp::Ordering::Greater)
                }
            };
            if replace {
                self.best = Some((cost, selected.clone()));
            }
            return Ok(());
        };

        let item = &active[&package];
        let constraints = canonical_constraints(
            &item.origins.iter().map(|o| o.constraint).collect::<Vec<_>>(),
        );
        let idx = self.package_index[&package];
        let mut candidates: Vec<Version> = self.catalog[idx]
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| matches_all(*v, &constraints))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.iter().find(|l| l.package == package) {
            if let Some(pos) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            self.search(selected)?;
            selected.remove(&package);
        }
        Ok(())
    }

    fn package_names(&self) -> Vec<String> {
        let mut names: Vec<String> = self.catalog.iter().map(|p| p.name.clone()).collect();
        names.sort();
        names.dedup();
        names
    }
}

fn sorted_names(names: &[String]) -> Vec<String> {
    let mut v = names.to_vec();
    v.sort();
    v.dedup();
    v
}

fn platform_matches(p: Platform, target: Target) -> bool {
    p == Platform::Any
        || matches!(
            (p, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn platform_text(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn evidence(item: Option<&Active>) -> Vec<Evidence> {
    item.map(|a| {
        let mut out: Vec<Evidence> = a
            .origins
            .iter()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect();
        out.sort_by(|x, y| {
            x.constraint
                .cmp(&y.constraint)
                .then_with(|| x.path.cmp(&y.path))
        });
        out
    })
    .unwrap_or_default()
}

fn canonical_path(item: &Active) -> Vec<String> {
    item.paths
        .iter()
        .min_by(|a, b| a.len().cmp(&b.len()).then_with(|| a.cmp(b)))
        .cloned()
        .unwrap_or_default()
}

fn lex_vector_cmp(
    a: &BTreeMap<String, Version>,
    b: &BTreeMap<String, Version>,
    names: &[String],
) -> std::cmp::Ordering {
    for name in names {
        let av = a.get(name);
        let bv = b.get(name);
        let ord = match (av, bv) {
            (None, None) => std::cmp::Ordering::Equal,
            (None, Some(_)) => std::cmp::Ordering::Less,
            (Some(_), None) => std::cmp::Ordering::Greater,
            (Some(x), Some(y)) => x.cmp(y),
        };
        if ord != std::cmp::Ordering::Equal {
            return ord;
        }
    }
    std::cmp::Ordering::Equal
}

fn invalid(site: String, name: &str) -> ResolveError {
    ResolveError::InvalidName {
        site,
        name: name.to_string(),
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(invalid(format!("root[{}]", i), &root.package));
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(invalid(format!("root[{}].feature[{}]", i, f), feature));
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(invalid(format!("catalog[{}]", i), &package.name));
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(invalid(site.clone(), &dep.package));
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(invalid(format!("{}.gate", site), gate));
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(invalid(format!("{}.feature[{}]", site, f), feature));
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let site = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if !valid_name(&feature.name) {
                    return Err(invalid(site.clone(), &feature.name));
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(invalid(format!("{}.enable[{}]", site, e), enabled));
                    }
                }
            }
        }
    }

    let mut seen = BTreeSet::new();
    for package in catalog {
        if !seen.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(invalid(format!("lock[{}]", i), &lock.package));
        }
    }
    let mut seen_locks = BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut package_index = HashMap::new();
    for (i, package) in catalog.iter().enumerate() {
        package_index.insert(package.name.clone(), i);
    }

    let mut solver = Solver {
        catalog,
        package_index,
        roots,
        locks,
        target,
        budget: max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
    };

    solver.search(&mut BTreeMap::new())?;
    let Some((_, chosen)) = solver.best else {
        return Err(solver.first_error.unwrap_or_else(|| ResolveError::Missing {
            package: String::new(),
            evidence: Vec::new(),
        }));
    };

    let final_active = solver.active(&chosen);
    let selected: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut all_names = BTreeSet::new();
    for lock in locks {
        all_names.insert(lock.package.clone());
    }
    for selection in &selected {
        all_names.insert(selection.package.clone());
    }

    let mut changes = Vec::new();
    for package in all_names {
        let before = locks
            .iter()
            .find(|lock| lock.package == package)
            .map(|lock| lock.version);
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = solver
                    .package_index
                    .get(&package)
                    .is_some_and(|idx| {
                        solver.catalog[*idx]
                            .releases
                            .iter()
                            .any(|r| r.version == old)
                    });
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let cs = final_active
                        .get(&package)
                        .map(|a| {
                            canonical_constraints(
                                &a.origins.iter().map(|o| o.constraint).collect::<Vec<_>>(),
                            )
                        })
                        .unwrap_or_default();
                    if !matches_all(old, &cs) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
