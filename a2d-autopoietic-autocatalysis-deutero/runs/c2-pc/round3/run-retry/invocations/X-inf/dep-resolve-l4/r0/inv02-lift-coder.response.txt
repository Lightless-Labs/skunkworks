fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_rank(c: Constraint) -> u8 {
    match c {
        Constraint::Exact(_) => 0,
        Constraint::AtLeast(_) => 1,
        Constraint::Before(_) => 2,
        Constraint::Caret(_) => 3,
    }
}

fn constraint_cmp(a: &Constraint, b: &Constraint) -> std::cmp::Ordering {
    constraint_rank(*a)
        .cmp(&constraint_rank(*b))
        .then_with(|| match (*a, *b) {
            (Constraint::Exact(x), Constraint::Exact(y))
            | (Constraint::AtLeast(x), Constraint::AtLeast(y))
            | (Constraint::Before(x), Constraint::Before(y))
            | (Constraint::Caret(x), Constraint::Caret(y)) => x.cmp(&y),
            _ => std::cmp::Ordering::Equal,
        })
}

fn canonical_constraints(items: &[Constraint]) -> Vec<Constraint> {
    let mut out = items.to_vec();
    out.sort_by(constraint_cmp);
    out.dedup();
    out
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraint_list_text(items: &[Constraint]) -> String {
    canonical_constraints(items)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
        && *b.last().unwrap() != b'-'
}

fn invalid(name: &str, site: String) -> Option<ResolveError> {
    if valid_name(name) {
        None
    } else {
        Some(ResolveError::InvalidName {
            site,
            name: name.to_string(),
        })
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_cmp(a: &Evidence, b: &Evidence) -> std::cmp::Ordering {
    constraint_cmp(&a.constraint, &b.constraint).then_with(|| a.path.cmp(&b.path))
}

fn evidence_lines(items: &[Evidence]) -> String {
    let mut e = items.to_vec();
    e.sort_by(evidence_cmp);
    e.iter()
        .map(|x| {
            format!(
                "\n  C:{} via {}",
                constraint_text(x.constraint),
                x.path.join(" -> ")
            )
        })
        .collect::<String>()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_lines(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            constraint_list_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            constraint_list_text(requirements),
            evidence_lines(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn validate_names(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if let Some(e) = invalid(&root.package, format!("root[{}]", i)) {
            return Err(e);
        }
        for (f, name) in root.features.iter().enumerate() {
            if let Some(e) = invalid(name, format!("root[{}].feature[{}]", i, f)) {
                return Err(e);
            }
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if let Some(e) = invalid(&package.name, format!("catalog[{}]", i)) {
            return Err(e);
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if let Some(e) = invalid(&dep.package, base.clone()) {
                    return Err(e);
                }
                if let Some(gate) = &dep.gate {
                    if let Some(e) = invalid(gate, format!("{}.gate", base)) {
                        return Err(e);
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if let Some(e) = invalid(name, format!("{}.feature[{}]", base, f)) {
                        return Err(e);
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].feature[{}]", i, j, f);
                if let Some(e) = invalid(&feature.name, base.clone()) {
                    return Err(e);
                }
                for (eidx, name) in feature.enables.iter().enumerate() {
                    if let Some(e) = invalid(name, format!("{}.enable[{}]", base, eidx)) {
                        return Err(e);
                    }
                }
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if let Some(e) = invalid(&lock.package, format!("lock[{}]", i)) {
            return Err(e);
        }
    }
    Ok(())
}

fn canonical_path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

#[derive(Clone)]
struct Active {
    constraints: BTreeMap<String, Vec<Constraint>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    incoming: BTreeMap<String, Vec<Evidence>>,
    active_deps: BTreeMap<String, Vec<usize>>,
}

fn compute_active(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Active {
    let mut a = Active {
        constraints: BTreeMap::new(),
        requests: BTreeMap::new(),
        paths: BTreeMap::new(),
        incoming: BTreeMap::new(),
        active_deps: BTreeMap::new(),
    };

    for (i, root) in roots.iter().enumerate() {
        let fs: BTreeSet<String> = root.features.iter().cloned().collect();
        let token = format!(
            "root[{}]:{}@{}+{{{}}}",
            i,
            root.package,
            constraint_text(root.constraint),
            fs.into_iter().collect::<Vec<_>>().join(",")
        );
        a.constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
        a.requests
            .entry(root.package.clone())
            .or_default()
            .extend(root.features.iter().cloned());
        a.incoming
            .entry(root.package.clone())
            .or_default()
            .push(Evidence {
                constraint: root.constraint,
                path: vec![token.clone()],
            });
        let path = vec![token];
        match a.paths.get(&root.package) {
            Some(old) if !canonical_path_better(&path, old) => {}
            _ => {
                a.paths.insert(root.package.clone(), path);
            }
        }
    }

    loop {
        let mut changed = false;
        let names: Vec<String> = selected.keys().cloned().collect();
        for source in names {
            let Some(version) = selected.get(&source).copied() else {
                continue;
            };
            let Some(package) = catalog.get(&source) else {
                continue;
            };
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let mut activated = a.requests.get(&source).cloned().unwrap_or_default();
            loop {
                let before = activated.len();
                let declarations: BTreeMap<&str, &Feature> = release
                    .features
                    .iter()
                    .map(|f| (f.name.as_str(), f))
                    .collect();
                let current: Vec<String> = activated.iter().cloned().collect();
                for name in current {
                    if let Some(feature) = declarations.get(name.as_str()) {
                        for enabled in &feature.enables {
                            if activated.insert(enabled.clone()) {
                                changed = true;
                            }
                        }
                    }
                }
                if activated.len() == before {
                    break;
                }
            }
            if a.requests.entry(source.clone()).or_default() != &activated {
                a.requests.insert(source.clone(), activated.clone());
                changed = true;
            }

            let path_to_source = a.paths.get(&source).cloned().unwrap_or_default();
            let mut deps = Vec::new();
            for (j, dep) in release.deps.iter().enumerate() {
                let platform_ok = match dep.platform {
                    Platform::Any => true,
                    Platform::Linux => target == Target::Linux,
                    Platform::Mac => target == Target::Mac,
                    Platform::Windows => target == Target::Windows,
                };
                let gate_ok = dep
                    .gate
                    .as_ref()
                    .map(|g| activated.contains(g))
                    .unwrap_or(true);
                if !platform_ok || !gate_ok {
                    continue;
                }
                deps.push(j);
                let fs: BTreeSet<String> = dep.features.iter().cloned().collect();
                let gate = dep.gate.as_deref().unwrap_or("*");
                let platform = match dep.platform {
                    Platform::Any => "any",
                    Platform::Linux => "linux",
                    Platform::Mac => "mac",
                    Platform::Windows => "windows",
                };
                let token = format!(
                    "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    source,
                    version_text(version),
                    j,
                    dep.package,
                    constraint_text(dep.constraint),
                    fs.into_iter().collect::<Vec<_>>().join(","),
                    gate,
                    platform
                );
                let mut path = path_to_source.clone();
                path.push(token);
                a.constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                a.requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
                a.incoming
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Evidence {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                match a.paths.get(&dep.package) {
                    Some(old) if !canonical_path_better(&path, old) => {}
                    _ => {
                        a.paths.insert(dep.package.clone(), path);
                        changed = true;
                    }
                }
            }
            if a.active_deps.get(&source) != Some(&deps) {
                a.active_deps.insert(source, deps);
                changed = true;
            }
        }
        if !changed {
            break;
        }
    }

    for cs in a.constraints.values_mut() {
        *cs = canonical_constraints(cs);
    }
    for ev in a.incoming.values_mut() {
        ev.sort_by(evidence_cmp);
    }
    a
}

fn constraints_match(version: Version, constraints: &[Constraint]) -> bool {
    constraints.iter().all(|c| matches(version, *c))
}

fn dynamic_error(
    catalog: &BTreeMap<String, &Package>,
    selected: &BTreeMap<String, Version>,
    active: &Active,
) -> Option<ResolveError> {
    for name in active.constraints.keys() {
        if !catalog.contains_key(name) {
            return Some(ResolveError::Missing {
                package: name.clone(),
                evidence: active.incoming.get(name).cloned().unwrap_or_default(),
            });
        }
    }

    for (name, version) in selected {
        let Some(cs) = active.constraints.get(name) else {
            continue;
        };
        if !constraints_match(*version, cs) {
            return Some(ResolveError::Conflict {
                package: name.clone(),
                selected: *version,
                requirements: cs.clone(),
                evidence: active.incoming.get(name).cloned().unwrap_or_default(),
            });
        }
    }

    for (name, version) in selected {
        let Some(package) = catalog.get(name) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        let requests = active.requests.get(name).cloned().unwrap_or_default();
        for feature in requests {
            if !release.features.iter().any(|f| f.name == feature) {
                return Some(ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: *version,
                    feature,
                    path: active.paths.get(name).cloned().unwrap_or_default(),
                });
            }
        }
    }

    for (name, cs) in &active.constraints {
        if selected.contains_key(name) {
            continue;
        }
        let package = catalog.get(name).unwrap();
        let any = package
            .releases
            .iter()
            .any(|r| constraints_match(r.version, cs));
        if !any {
            return Some(ResolveError::NoMatch {
                package: name.clone(),
                requirements: cs.clone(),
                evidence: active.incoming.get(name).cloned().unwrap_or_default(),
            });
        }
    }
    None
}

#[derive(Clone)]
struct SearchResult {
    best: Option<BTreeMap<String, Version>>,
    best_cost: usize,
    first_error: Option<ResolveError>,
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a [Lock],
    target: Target,
    max_attempts: usize,
    attempts: usize,
    lock_map: BTreeMap<String, Version>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn cost(&self, selected: &BTreeMap<String, Version>) -> usize {
        self.lock_map
            .iter()
            .filter(|(p, v)| selected.get(*p) != Some(*v))
            .count()
    }

    fn better_tie(
        &self,
        candidate: &BTreeMap<String, Version>,
        incumbent: &BTreeMap<String, Version>,
    ) -> bool {
        for name in &self.catalog_names {
            match (candidate.get(name), incumbent.get(name)) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(a), Some(b)) if a != b => return a > b,
                _ => {}
            }
        }
        false
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = compute_active(self.catalog, self.roots, selected, self.target);
        if let Some(error) = dynamic_error(self.catalog, selected, &active) {
            return Err(error);
        }

        let next = active
            .constraints
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();
        let Some(name) = next else {
            let cost = self.cost(selected);
            let replace = match &self.best_result().best {
                None => true,
                Some(old) => {
                    cost < self.best_result().best_cost
                        || (cost == self.best_result().best_cost && self.better_tie(selected, old))
                }
            };
            if replace {
                self.current_best = Some((selected.clone(), cost));
            }
            return Ok(());
        };

        let cs = active.constraints.get(&name).cloned().unwrap_or_default();
        let package = self.catalog.get(&name).unwrap();
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| constraints_match(*v, &cs))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(lock) = self.lock_map.get(&name).copied() {
            if let Some(i) = candidates.iter().position(|v| *v == lock) {
                candidates.remove(i);
                candidates.insert(0, lock);
            }
        }

        let mut first_error = None;
        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), candidate);
            match self.visit(selected) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    selected.remove(&name);
                    return Err(ResolveError::Limit { attempts });
                }
                Err(error) => {
                    if first_error.is_none() {
                        first_error = Some(error);
                    }
                }
            }
            selected.remove(&name);
        }

        if let Some(error) = first_error {
            Err(error)
        } else {
            Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: cs,
                evidence: active.incoming.get(&name).cloned().unwrap_or_default(),
            })
        }
    }

    // Scratch best solution is stored separately to keep recursive borrowing simple.
    fn best_result(&self) -> SearchResult {
        match &self.current_best {
            Some((s, c)) => SearchResult {
                best: Some(s.clone()),
                best_cost: *c,
                first_error: None,
            },
            None => SearchResult {
                best: None,
                best_cost: usize::MAX,
                first_error: None,
            },
        }
    }

    current_best: Option<(BTreeMap<String, Version>, usize)>,
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate_names(catalog, roots, locks)?;

    let mut packages = BTreeMap::new();
    for package in catalog {
        if packages.contains_key(&package.name) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
        packages.insert(package.name.clone(), package);
    }

    for package in catalog {
        let mut seen = BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if lock_map.contains_key(&lock.package) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
        lock_map.insert(lock.package.clone(), lock.version);
    }

    let mut names: Vec<String> = packages.keys().cloned().collect();
    names.sort();
    let mut search = Search {
        catalog: &packages,
        roots,
        locks,
        target,
        max_attempts,
        attempts: 0,
        lock_map: lock_map.clone(),
        catalog_names: names,
        current_best: None,
    };
    let mut selected = BTreeMap::new();
    let result = search.visit(&mut selected);

    let solution = match result {
        Err(ResolveError::Limit { attempts }) => return Err(ResolveError::Limit { attempts }),
        _ => match search.current_best {
            Some((solution, _)) => solution,
            None => {
                return Err(result.err().unwrap_or(ResolveError::NoMatch {
                    package: String::new(),
                    requirements: Vec::new(),
                    evidence: Vec::new(),
                }))
            }
        },
    };

    let final_active = compute_active(&packages, roots, &solution, target);
    let selections: Vec<Selection> = solution
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut change_names = BTreeSet::new();
    change_names.extend(lock_map.keys().cloned());
    change_names.extend(solution.keys().cloned());
    let mut changes = Vec::new();
    for package in change_names {
        let before = lock_map.get(&package).copied();
        let after = solution.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = packages
                    .get(&package)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .constraints
                    .get(&package)
                    .map(|cs| !constraints_match(old, cs))
                    .unwrap_or(false)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selections,
        changes,
    })
}
