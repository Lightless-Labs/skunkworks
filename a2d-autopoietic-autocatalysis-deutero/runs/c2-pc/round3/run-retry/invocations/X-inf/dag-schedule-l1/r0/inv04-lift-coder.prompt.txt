pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let count = tasks.len();

    // Validate every dependency entry before graph construction or cycle detection.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();

        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    // Build predecessor-to-child edges and indegrees.
    let mut children = vec![Vec::new(); count];
    let mut indegree = vec![0usize; count];

    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest available task ID selected each time.
    let mut available = std::collections::BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::with_capacity(count);
    let mut remaining_indegree = indegree;

    while let Some(&task_id) = available.iter().next() {
        available.remove(&task_id);
        order.push(task_id);

        for &child in &children[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != count {
        let mut blocked = Vec::new();
        for (task_id, &degree) in remaining_indegree.iter().enumerate() {
            if degree > 0 {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut times = vec![Timing { start: 0, finish: 0 }; count];
    let mut makespan = 0u64;

    for &task_id in &order {
        let mut start = 0u64;
        for &predecessor in &tasks[task_id].deps {
            start = start.max(times[predecessor].finish);
        }

        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;

        times[task_id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })
}
