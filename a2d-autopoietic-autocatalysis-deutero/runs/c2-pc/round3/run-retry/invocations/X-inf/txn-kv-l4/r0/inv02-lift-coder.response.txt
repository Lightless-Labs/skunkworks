```rust
use std::collections::{BTreeMap, BTreeSet, HashMap};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Entry {
    value: Option<String>,
    deadline: Option<u64>,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    entry: Entry,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Entry>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: HashMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: HashMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn validate_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn visible_entry(&self, key: &str, tx: Option<u64>) -> Option<Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(entry) = transaction.writes.get(key) {
                return Some(entry.clone());
            }
            self.data.get(key)?.iter()
                .rev()
                .find(|v| v.revision <= transaction.begin_revision)
                .map(|v| v.entry.clone())
        } else {
            self.data.get(key)?.last().map(|v| v.entry.clone())
        }
    }

    fn live(entry: &Entry, now: u64) -> Option<String> {
        if entry.value.is_none() || entry.deadline.is_some_and(|d| d <= now) {
            None
        } else {
            entry.value.clone()
        }
    }

    fn visible_live(&self, key: &str, tx: Option<u64>, now: u64) -> Option<String> {
        Self::live(&self.visible_entry(key, tx)?, now)
    }

    fn visible_map(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, String> {
        let mut result = BTreeMap::new();
        for key in self.data.keys() {
            if let Some(value) = self.visible_live(key, tx, now) {
                result.insert(key.clone(), value);
            }
        }
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                for (key, entry) in &transaction.writes {
                    if !self.data.contains_key(key) {
                        if let Some(value) = Self::live(entry, now) {
                            result.insert(key.clone(), value);
                        }
                    }
                }
            }
        }
        result
    }

    fn append(&mut self, key: String, entry: Entry) {
        self.revision += 1;
        self.data.entry(key).or_default().push(Version {
            revision: self.revision,
            entry,
        });
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(id, Transaction {
            begin_revision: self.revision,
            read_only,
            writes: BTreeMap::new(),
        });
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let transaction = self.transactions.get(&id).ok_or(Error::UnknownTransaction(id))?;
        let begin_revision = transaction.begin_revision;
        let writes = transaction.writes.clone();

        for key in writes.keys() {
            if self.data.get(key)
                .and_then(|versions| versions.last())
                .is_some_and(|version| version.revision > begin_revision)
            {
                return Err(Error::Conflict { tx: id, key: key.clone() });
            }
        }

        if writes.is_empty() {
            self.transactions.remove(&id);
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for (key, entry) in writes {
            self.data.entry(key).or_default().push(Version { revision, entry });
        }
        self.transactions.remove(&id);
        Ok(revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let entry = if ttl == Some(0) {
            Entry { value: None, deadline: None }
        } else {
            let deadline = match ttl {
                Some(duration) => Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
                None => None,
            };
            Entry { value: Some(value.to_string()), deadline }
        };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), entry);
        } else {
            self.append(key.to_string(), entry);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let existed = self.visible_live(key, tx, now).is_some();
        let tombstone = Entry { value: None, deadline: None };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_string(), tombstone);
        } else {
            self.append(key.to_string(), tombstone);
        }
        Ok(existed)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.visible_live(key, tx, now))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        Ok(self.visible_map(tx, now).values().filter(|v| v.as_str() == value).count())
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }
        if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || c.last.as_str() < start
                || end.is_some_and(|e| c.last.as_str() >= e)
                || prefix.is_some_and(|p| !c.last.starts_with(p))
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page { items: Vec::new(), next: None });
        }

        let after = cursor.map(|c| c.last.as_str());
        let candidates: Vec<(String, String)> = self.visible_map(tx, now).into_iter()
            .filter(|(k, _)| {
                k.as_str() >= start
                    && end.is_none_or(|e| k.as_str() < e)
                    && prefix.is_none_or(|p| k.starts_with(p))
                    && after.is_none_or(|last| k.as_str() > last)
            })
            .collect();

        let skip = if cursor.is_none() { offset } else { 0 };
        let mut iter = candidates.into_iter().skip(skip);
        let items: Vec<_> = iter.by_ref().take(limit).collect();
        let next = if iter.next().is_some() {
            items.last().map(|(last, _)| Cursor {
                start: start.to_string(),
                end: end.map(str::to_string),
                prefix: prefix.map(str::to_string),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.validate_tx(tx)?;
        if end.is_some_and(|e| e <= start) {
            return Err(Error::InvalidRange);
        }

        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in self.visible_map(tx, now) {
            if key.as_str() < start
                || end.is_some_and(|e| key.as_str() >= e)
                || prefix.is_some_and(|p| !key.starts_with(p))
            {
                continue;
            }
            let parsed = parse_canonical_i64(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum.checked_add(parsed).ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        validate_name(name)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let saved = self.visible_map(None, now).into_iter().filter_map(|(key, value)| {
            let entry = self.data.get(&key)?.last()?.entry.clone();
            if Self::live(&entry, now).as_deref() == Some(value.as_str()) {
                Some((key, entry))
            } else {
                None
            }
        }).collect();
        self.snapshots.insert(name.to_string(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let saved = self.snapshots.get(name)
            .cloned()
            .ok_or_else(|| Error::SnapshotMissing(name.to_string()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let copied: BTreeMap<String, Entry> = saved.into_iter()
            .filter(|(_, entry)| Self::live(entry, now).is_some())
            .collect();
        let mut union = BTreeSet::new();
        for (key, versions) in &self.data {
            if versions.last().is_some_and(|v| Self::live(&v.entry, now).is_some()) {
                union.insert(key.clone());
            }
        }
        union.extend(copied.keys().cloned());
        if union.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in union {
            let entry = copied.get(&key).cloned().unwrap_or(Entry { value: None, deadline: None });
            self.data.entry(key).or_default().push(Version { revision, entry });
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_string()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;
        for versions in self.data.values_mut() {
            if versions.is_empty() {
                continue;
            }
            let mut keep = BTreeSet::new();
            keep.insert(versions.len() - 1);
            for transaction in self.transactions.values() {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= transaction.begin_revision) {
                    keep.insert(index);
                }
            }
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let keys = self.data.len();
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self.data.values().flat_map(|vs| vs.iter())
            .filter(|v| v.entry.value.is_none()).count();
        let live_keys = self.data.values().filter(|vs| {
            vs.last().is_some_and(|v| Self::live(&v.entry, now).is_some())
        }).count();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys,
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self.transactions.values().filter(|t| t.read_only).count(),
            staged_writes: self.transactions.values().map(|t| t.writes.len()).sum(),
            snapshots: self.snapshots.len(),
        })
    }
}

fn validate_name(name: &str) -> Result<(), Error> {
    if name.is_empty() {
        return Err(Error::NameEmpty);
    }
    for (at, ch) in name.char_indices() {
        let valid = if at == 0 {
            ch.is_ascii_lowercase()
        } else {
            ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
        };
        if !valid {
            return Err(Error::NameInvalid { at });
        }
    }
    Ok(())
}

fn parse_canonical_i64(value: &str) -> Option<i64> {
    if value.is_empty() {
        return None;
    }
    let bytes = value.as_bytes();
    let digits = if bytes[0] == b'-' { &bytes[1..] } else { bytes };
    if digits.is_empty() || !digits.iter().all(u8::is_ascii_digit) {
        return None;
    }
    if digits.len() > 1 && digits[0] == b'0' {
        return None;
    }
    if bytes[0] == b'-' && digits == b"0" {
        return None;
    }
    if bytes[0] == b'0' || (bytes[0] == b'-' && digits[0] == b'0') {
        return None;
    }
    value.parse::<i64>().ok()
}
```
