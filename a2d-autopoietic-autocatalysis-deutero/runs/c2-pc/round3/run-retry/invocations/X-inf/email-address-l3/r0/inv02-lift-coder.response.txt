#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Gap {
    start: usize,
    end: usize,
    left: Option<u8>,
    right: Option<u8>,
}

fn horizontal(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let mut clean = Vec::<u8>::new();
    let mut map = Vec::<usize>::new();
    let mut gaps = Vec::<Gap>::new();
    let mut i = 0usize;
    let mut quote_prefix = false;
    let mut quote_closed = false;

    while i < bytes.len() {
        let b = bytes[i];

        if quote_prefix && !quote_closed {
            clean.push(b);
            map.push(i);
            if b == b'\\' {
                if i + 1 < bytes.len() {
                    clean.push(bytes[i + 1]);
                    map.push(i + 1);
                    i += 2;
                    continue;
                }
            } else if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if !quote_prefix && clean.is_empty() && b == b'"' {
            quote_prefix = true;
            clean.push(b);
            map.push(i);
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'(' || b == b')' || b == b'\r' || b == b'\n' {
            let start = i;
            let left = clean.last().copied();
            let mut depth = 0usize;
            let mut outer_open = None;

            loop {
                if i >= bytes.len() {
                    break;
                }
                let c = bytes[i];

                if c == b'\r' || c == b'\n' {
                    if c == b'\r'
                        && i + 2 < bytes.len()
                        && bytes[i + 1] == b'\n'
                        && horizontal(bytes[i + 2])
                    {
                        i += 2;
                        while i < bytes.len() && horizontal(bytes[i]) {
                            i += 1;
                        }
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }

                if depth == 0 {
                    if horizontal(c) {
                        i += 1;
                        continue;
                    }
                    if c == b'(' {
                        depth = 1;
                        outer_open = Some(i);
                        i += 1;
                        continue;
                    }
                    if c == b')' {
                        return Err(EmailError::UnexpectedClose { at: i });
                    }
                    break;
                }

                if c == b'\\' {
                    if i + 1 >= bytes.len()
                        || !printable(bytes[i + 1])
                    {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                    continue;
                }
                if !printable(c) && !horizontal(c) {
                    return Err(EmailError::CommentChar { at: i });
                }
                if c == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                } else if c == b')' {
                    depth -= 1;
                    i += 1;
                } else {
                    i += 1;
                }
            }

            if depth != 0 {
                return Err(EmailError::UnclosedComment {
                    at: outer_open.unwrap_or(start),
                });
            }

            gaps.push(Gap {
                start,
                end: i,
                left,
                right: bytes.get(i).copied(),
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        map.push(i);
        i += 1;
    }

    for gap in &gaps {
        let legal = clean.is_empty()
            || gap.left.is_none()
            || gap.right.is_none()
            || gap.left == Some(b'@')
            || gap.right == Some(b'@');
        if !legal {
            return Err(EmailError::CfwsBoundary { at: gap.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = None;
    if clean[0] == b'"' {
        let mut j = 1usize;
        let mut found = None;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j += 2;
            } else if clean[j] == b'"' {
                found = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        let Some(end) = found else {
            return Err(EmailError::UnclosedQuote { at: map[0] });
        };
        protected_end = Some(end);
    }

    let mut ats = Vec::new();
    for (j, &b) in clean.iter().enumerate() {
        if b == b'@' && !protected_end.is_some_and(|end| j <= end) {
            ats.push(j);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt { at: map[ats[1]] });
    }
    let at_idx = ats[0];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal { at: map[at_idx] });
    }

    let local_bytes = &clean[..at_idx];
    let local;
    let quoted;

    if local_bytes[0] == b'"' {
        quoted = true;
        let end = protected_end.unwrap();
        if end + 1 != at_idx {
            return Err(EmailError::TrailingLocal {
                at: map[end + 1],
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < end {
            let b = local_bytes[j];
            if b == b'\\' {
                if j + 1 >= end
                    || !matches!(local_bytes[j + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape { at: map[j] });
                }
                decoded.push(local_bytes[j + 1]);
                origins.push(map[j]);
                j += 2;
            } else {
                if !printable(b) {
                    return Err(EmailError::QuotedChar { at: map[j] });
                }
                decoded.push(b);
                origins.push(map[j]);
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: map[end] });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        local = String::from_utf8(decoded).unwrap();
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: map[64] });
        }
        for j in 0..local_bytes.len() {
            let b = local_bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: map[j] });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar { at: map[j] });
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).unwrap();
    }

    let domain_start = at_idx + 1;
    let mut domain_end = clean.len();
    let mut root_dot = None;
    if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(map[domain_end]);
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: map.get(domain_start).copied().unwrap_or(input.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong { at: map[domain_start + 253] });
    }

    let mut labels = Vec::<(usize, usize)>::new();
    let mut start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || clean[j] == b'.' {
            if start == j {
                return Err(EmailError::EmptyLabel {
                    at: if j == domain_end {
                        root_dot.unwrap_or_else(|| map.get(j).copied().unwrap_or(input.len()))
                    } else {
                        map[j]
                    },
                });
            }
            let len = j - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: map[start + 63] });
            }
            if clean[start] == b'-' {
                return Err(EmailError::LabelHyphen { at: map[start] });
            }
            if clean[j - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: map[j - 1] });
            }
            for k in start..j {
                if !(clean[k].is_ascii_alphanumeric() || clean[k] == b'-') {
                    return Err(EmailError::DomainChar { at: map[k] });
                }
            }
            labels.push((start, j));
            start = j + 1;
        }
    }
    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: map[domain_start] });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: map[last_start] });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address { local, domain, quoted })
}
