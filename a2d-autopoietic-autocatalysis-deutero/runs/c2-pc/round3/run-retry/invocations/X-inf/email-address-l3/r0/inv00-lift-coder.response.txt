#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn parse_cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let mut clean = Vec::with_capacity(input.len());
    let mut map = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut protected = false;
    let mut escaped_quote = false;

    while i < input.len() {
        let b = input[i];

        if protected {
            clean.push(b);
            map.push(i);
            if escaped_quote {
                escaped_quote = false;
                i += 1;
            } else if b == b'\\' {
                escaped_quote = true;
            } else if b == b'"' {
                protected = false;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            if b == b'\\' {
                let Some(&next) = input.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !is_printable(next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !(is_printable(b) || b == b'\t') {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            let run_start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if c == b' ' || c == b'\t' {
                    i += 1;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    outer_open = i;
                    depth = 1;
                    i += 1;
                    while i < input.len() && depth > 0 {
                        let x = input[i];
                        if x == b'\\' {
                            let Some(&next) = input.get(i + 1) else {
                                return Err(EmailError::CommentEscape { at: i });
                            };
                            if !is_printable(next) {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                            continue;
                        }
                        if x == b'\r' || x == b'\n' {
                            if let Some(end) = fold_end(input, i) {
                                i = end;
                                continue;
                            }
                            return Err(EmailError::BadFold { at: i });
                        }
                        if x == b'(' {
                            if depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            depth += 1;
                            i += 1;
                            continue;
                        }
                        if x == b')' {
                            depth -= 1;
                            i += 1;
                            continue;
                        }
                        if !(is_printable(x) || x == b'\t') {
                            return Err(EmailError::CommentChar { at: i });
                        }
                        i += 1;
                    }
                    if depth > 0 {
                        return Err(EmailError::UnclosedComment { at: outer_open });
                    }
                    continue;
                }
                break;
            }
            runs.push(Run { start: run_start, end: i });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if b == b'"' && clean.is_empty() {
            protected = true;
            clean.push(b);
            map.push(i);
            i += 1;
            continue;
        }

        clean.push(b);
        map.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }

    for run in &runs {
        let left_at = clean.iter().rposition(|&b| b == b'@' && map.iter().position(|&p| p == map[clean.len().saturating_sub(1)]) .is_some()).is_some();
        let _ = left_at;
        let before = map.iter().rposition(|&p| p < run.start).and_then(|idx| clean.get(idx));
        let after = map.iter().position(|&p| p >= run.end).and_then(|idx| clean.get(idx));
        let at_boundary = before.is_none()
            || after.is_none()
            || before == Some(&b'@')
            || after == Some(&b'@');
        if !at_boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    Ok((clean, map, runs))
}

fn pos(map: &[usize], idx: usize, input_len: usize) -> usize {
    map.get(idx).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let bytes = input.as_bytes();
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, map, _) = parse_cleanup(bytes)?;
    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean.first() == Some(&b'"') {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: pos(&map, 0, bytes.len()) });
        }
    }

    let mut ats = Vec::new();
    let protected_end = quote_end.unwrap_or(0);
    for i in 0..clean.len() {
        if i <= protected_end && quote_end.is_some() {
            continue;
        }
        if clean[i] == b'@' {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt { at: pos(&map, ats[1], bytes.len()) });
    }

    let at_idx = ats[0];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal { at: pos(&map, at_idx, bytes.len()) });
    }

    let local_bytes = &clean[..at_idx];
    let local;
    let quoted;
    if local_bytes.first() == Some(&b'"') {
        let qend = quote_end.unwrap();
        if qend + 1 != at_idx {
            return Err(EmailError::TrailingLocal { at: pos(&map, qend + 1, bytes.len()) });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < qend {
            let b = clean[i];
            if b == b'\\' {
                let Some(&next) = clean.get(i + 1) else {
                    return Err(EmailError::BadEscape { at: pos(&map, i, bytes.len()) });
                };
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: pos(&map, i, bytes.len()) });
                }
                decoded.push(next);
                origins.push(pos(&map, i, bytes.len()));
                i += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar { at: pos(&map, i, bytes.len()) });
                }
                decoded.push(b);
                origins.push(pos(&map, i, bytes.len()));
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos(&map, qend, bytes.len()) });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        local = String::from_utf8(decoded).unwrap();
        quoted = true;
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: pos(&map, 64, bytes.len()) });
        }
        let mut prev_dot = false;
        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || prev_dot {
                    return Err(EmailError::LocalDot { at: pos(&map, i, bytes.len()) });
                }
                prev_dot = true;
            } else {
                prev_dot = false;
                if !(b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)) {
                    return Err(EmailError::LocalChar { at: pos(&map, i, bytes.len()) });
                }
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).unwrap();
        quoted = false;
    }

    let domain_start = at_idx + 1;
    let raw_domain = &clean[domain_start..];
    let mut domain_end = clean.len();
    let mut removed_dot = None;
    if raw_domain.last() == Some(&b'.') {
        domain_end -= 1;
        removed_dot = Some(domain_end);
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain { at: pos(&map, domain_start, bytes.len()) });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong { at: pos(&map, domain_start + 253, bytes.len()) });
    }

    let domain_bytes = &clean[domain_start..domain_end];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0;
    for i in 0..=domain_bytes.len() {
        if i == domain_bytes.len() || domain_bytes[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    for &(s, e) in &labels {
        if s == e {
            let idx = if e == domain_bytes.len() {
                removed_dot.unwrap_or(domain_end)
            } else {
                domain_start + e
            };
            return Err(EmailError::EmptyLabel { at: pos(&map, idx, bytes.len()) });
        }
        if e - s > 63 {
            return Err(EmailError::LabelTooLong { at: pos(&map, domain_start + s + 63, bytes.len()) });
        }
        let label = &domain_bytes[s..e];
        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(&map, domain_start + s, bytes.len()) });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(&map, domain_start + e - 1, bytes.len()) });
        }
        for (j, &b) in label.iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar { at: pos(&map, domain_start + s + j, bytes.len()) });
            }
        }
    }
    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: pos(&map, domain_start, bytes.len()) });
    }
    let (last_s, last_e) = *labels.last().unwrap();
    if domain_bytes[last_s..last_e].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(&map, domain_start + last_s, bytes.len()) });
    }

    let domain = domain_bytes.iter().map(|b| b.to_ascii_lowercase() as char).collect();
    Ok(Address { local, domain, quoted })
}
