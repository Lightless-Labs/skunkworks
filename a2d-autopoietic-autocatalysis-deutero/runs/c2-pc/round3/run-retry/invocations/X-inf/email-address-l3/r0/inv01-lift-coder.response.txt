#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn is_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let mut clean = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_comment = 0usize;
    let mut protected = false;
    let mut escaped_quote = false;
    let mut run_start: Option<usize> = None;
    let mut run_end = 0;

    while i < input.len() {
        let b = input[i];

        if protected {
            clean.push(b);
            positions.push(i);
            if escaped_quote {
                escaped_quote = false;
            } else if b == b'\\' {
                escaped_quote = true;
            } else if b == b'"' {
                protected = false;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            if b == b'\r' || b == b'\n' {
                if let Some(end) = is_fold(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'\\' {
                let Some(&next) = input.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !is_printable(next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !(is_printable(b) || b == b'\t') {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'(' || b == b'\r' || b == b'\n' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            run_end = i + 1;
            if b == b'(' {
                depth = 1;
                outer_comment = i;
                i += 1;
            } else if b == b'\r' || b == b'\n' {
                if let Some(end) = is_fold(input, i) {
                    i = end;
                    run_end = i;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            } else {
                i += 1;
            }
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if run_start.is_some() {
            runs.push(Run {
                start: run_start.take().unwrap(),
                end: run_end,
            });
        }

        if b == b'"' && clean.is_empty() {
            protected = true;
        }
        clean.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }
    if let Some(start) = run_start {
        runs.push(Run { start, end: run_end });
    }

    Ok((clean, positions, runs))
}

fn pos(positions: &[usize], index: usize, input_len: usize) -> usize {
    positions.get(index).copied().unwrap_or(input_len)
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, positions, runs) = cleanup(bytes)?;
    for run in runs {
        let left = positions.iter().position(|&p| p >= run.end);
        let right = positions.iter().rposition(|&p| p < run.start);
        let left_is_at = left.is_some_and(|i| clean[i] == b'@');
        let right_is_at = right.is_some_and(|i| clean[i] == b'@');
        if right.is_none() || left.is_none() || left_is_at || right_is_at {
            continue;
        }
        return Err(EmailError::CfwsBoundary { at: run.start });
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: positions[0],
            });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < clean.len() {
        if let Some(end) = quote_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if clean[i] == b'@' {
            ats.push(i);
        }
        i += 1;
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: positions[ats[1]],
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: positions[sep],
        });
    }

    let mut local = String::new();
    let quoted = clean[0] == b'"';
    if quoted {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&positions, end + 1, bytes.len()),
            });
        }
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            let b = clean[j];
            if b == b'\\' {
                let Some(&next) = clean.get(j + 1) else {
                    return Err(EmailError::BadEscape {
                        at: positions[j],
                    });
                };
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: positions[j],
                    });
                }
                local.push(next as char);
                origins.push(positions[j]);
                j += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: positions[j],
                    });
                }
                local.push(b as char);
                origins.push(positions[j]);
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: positions[end],
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        let length = sep;
        if length > 64 {
            return Err(EmailError::LocalTooLong {
                at: positions[64],
            });
        }
        for j in 0..sep {
            let b = clean[j];
            if b == b'.' {
                if j == 0 || j + 1 == sep || clean[j - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: positions[j],
                    });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar {
                    at: positions[j],
                });
            }
        }
        local = String::from_utf8(clean[..sep].to_vec()).unwrap();
    }

    let domain_start = sep + 1;
    let raw_len = clean.len() - domain_start;
    let mut domain_end = clean.len();
    let root_dot = if raw_len > 0 && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: pos(&positions, domain_start, bytes.len()),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start + 253],
        });
    }

    let mut labels = Vec::new();
    let mut start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || clean[j] == b'.' {
            if start == j {
                let at = if j == domain_end {
                    root_dot.map(|k| positions[k]).unwrap_or_else(|| pos(&positions, j, bytes.len()))
                } else {
                    positions[j]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let len = j - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: positions[start + 63],
                });
            }
            if clean[start] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: positions[start],
                });
            }
            if clean[j - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: positions[j - 1],
                });
            }
            for k in start..j {
                if !(clean[k].is_ascii_alphanumeric() || clean[k] == b'-') {
                    return Err(EmailError::DomainChar {
                        at: positions[k],
                    });
                }
            }
            labels.push((start, j));
            start = j + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: positions[domain_start],
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();
    Ok(Address {
        local,
        domain,
        quoted,
    })
}
