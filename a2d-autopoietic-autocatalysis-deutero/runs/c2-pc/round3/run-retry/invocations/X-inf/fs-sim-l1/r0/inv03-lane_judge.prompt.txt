use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

fn parse_path(path: &str) -> Result<Vec<&str>, Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() || !path.starts_with('/') {
        return Err(Error::InvalidPath(0));
    }
    if path == "/" {
        return Ok(Vec::new());
    }

    let bytes = path.as_bytes();
    let mut components = Vec::new();
    let mut start = 1;

    for i in 1..=bytes.len() {
        if i == bytes.len() || bytes[i] == b'/' {
            if i == start {
                return Err(Error::InvalidPath(if i == bytes.len() {
                    bytes.len() - 1
                } else {
                    i
                }));
            }
            let component = &path[start..i];
            if component == "." || component == ".." {
                return Err(Error::InvalidPath(start));
            }
            components.push(component);
            start = i + 1;
        }
    }
    Ok(components)
}

fn canonical(components: &[&str]) -> String {
    if components.is_empty() {
        "/".to_owned()
    } else {
        let mut result = String::new();
        for component in components {
            result.push('/');
            result.push_str(component);
        }
        result
    }
}

fn get_entry<'a>(root: &'a Entry, components: &[&str]) -> Result<&'a Entry, Error> {
    let mut current = root;
    let mut prefix: Vec<&str> = Vec::new();

    for (i, component) in components.iter().enumerate() {
        match current {
            Entry::Dir(children) => {
                prefix.push(component);
                current = children
                    .get(*component)
                    .ok_or_else(|| Error::Missing(canonical(&prefix)))?;
            }
            Entry::File(_) => return Err(Error::NotDir(canonical(&prefix))),
        }
        if i + 1 < components.len() && matches!(current, Entry::File(_)) {
            return Err(Error::NotDir(canonical(&prefix)));
        }
    }
    Ok(current)
}

fn get_parent_mut<'a>(
    root: &'a mut Entry,
    components: &[&str],
) -> Result<&'a mut Entry, Error> {
    let mut current = root;
    let mut prefix: Vec<&str> = Vec::new();

    for component in components {
        match current {
            Entry::Dir(children) => {
                prefix.push(component);
                current = children
                    .get_mut(*component)
                    .ok_or_else(|| Error::Missing(canonical(&prefix)))?;
            }
            Entry::File(_) => return Err(Error::NotDir(canonical(&prefix))),
        }
    }
    match current {
        Entry::Dir(_) => Ok(current),
        Entry::File(_) => Err(Error::NotDir(canonical(&prefix))),
    }
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = parse_path(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let parent_components = &components[..components.len() - 1];
        let name = components[components.len() - 1];
        let full_path = canonical(&components);
        let parent = get_parent_mut(&mut self.root, parent_components)?;

        match parent {
            Entry::Dir(children) => {
                if children.contains_key(name) {
                    return Err(Error::Exists(full_path));
                }
                children.insert(name.to_owned(), Entry::Dir(BTreeMap::new()));
                Ok(())
            }
            Entry::File(_) => unreachable!(),
        }
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = parse_path(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let parent_components = &components[..components.len() - 1];
        let name = components[components.len() - 1];
        let full_path = canonical(&components);
        let parent = get_parent_mut(&mut self.root, parent_components)?;

        match parent {
            Entry::Dir(children) => match children.get_mut(name) {
                Some(Entry::Dir(_)) => Err(Error::IsDir(full_path)),
                Some(Entry::File(contents)) => {
                    contents.clear();
                    contents.extend_from_slice(data);
                    Ok(())
                }
                None => {
                    children.insert(name.to_owned(), Entry::File(data.to_vec()));
                    Ok(())
                }
            },
            Entry::File(_) => unreachable!(),
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = parse_path(path)?;
        match get_entry(&self.root, &components)? {
            Entry::Dir(_) => Err(Error::IsDir(canonical(&components))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = parse_path(path)?;
        let entry = get_entry(&self.root, &components)?;
        match entry {
            Entry::File(_) => Err(Error::NotDir(canonical(&components))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: std::collections::HashMap<String, Entry>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = std::collections::HashMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self { entries }
    }

    fn validate(path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut start = 1;
        for (i, b) in path.bytes().enumerate().skip(1) {
            if b == b'/' {
                let component = &path[start..i];
                if component.is_empty() {
                    return Err(Error::InvalidPath(i));
                }
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component.to_string());
                start = i + 1;
            }
        }

        let component = &path[start..];
        if component.is_empty() {
            return Err(Error::InvalidPath(path.len() - 1));
        }
        if component == "." || component == ".." {
            return Err(Error::InvalidPath(start));
        }
        components.push(component.to_string());
        Ok(components)
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            let mut path = String::new();
            for component in components {
                path.push('/');
                path.push_str(component);
            }
            path
        }
    }

    fn check_directory(&self, components: &[String]) -> Result<(), Error> {
        let mut prefix = Vec::new();
        for component in components {
            let parent_path = Self::canonical(&prefix);
            match self.entries.get(&parent_path) {
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(parent_path)),
                None => return Err(Error::Missing(parent_path)),
            }
            prefix.push(component.clone());
        }
        let path = Self::canonical(components);
        match self.entries.get(&path) {
            Some(Entry::Dir) => Ok(()),
            Some(Entry::File(_)) => Err(Error::NotDir(path)),
            None => Err(Error::Missing(path)),
        }
    }

    fn check_parent(&self, components: &[String]) -> Result<String, Error> {
        let parent_components = &components[..components.len() - 1];
        self.check_directory(parent_components)?;
        Ok(Self::canonical(parent_components))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }

        let parent = self.check_parent(&components)?;
        let canonical = Self::canonical(&components);
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }

        let _ = parent;
        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        self.check_parent(&components)?;
        let canonical = Self::canonical(&components);
        match self.entries.get(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(_)) | None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        let canonical = Self::canonical(&components);
        let parent_components = &components[..components.len() - 1];
        self.check_directory(parent_components)?;
        match self.entries.get(&canonical) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            None => Err(Error::Missing(canonical)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        let canonical = Self::canonical(&components);
        self.check_directory(&components)?;

        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == "/" {
                continue;
            }

            let parent = match entry_path.rsplit_once('/') {
                Some((parent, _)) if parent.is_empty() => "/",
                Some((parent, _)) => parent,
                None => continue,
            };

            if parent == canonical {
                let basename = entry_path
                    .rsplit_once('/')
                    .map(|(_, name)| name.to_string())
                    .unwrap_or_default();
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                children.push((basename, kind));
            }
        }

        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }
}

use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn validate(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut components = Vec::new();
        let mut start = 1;

        for (i, byte) in bytes.iter().enumerate().skip(1) {
            if *byte == b'/' {
                if i == start {
                    return Err(Error::InvalidPath(i));
                }
                let component = &path[start..i];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component);
                start = i + 1;
            }
        }

        if start == bytes.len() {
            return Err(Error::InvalidPath(bytes.len() - 1));
        }

        let component = &path[start..];
        if component == "." || component == ".." {
            return Err(Error::InvalidPath(start));
        }
        components.push(component);
        Ok(components)
    }

    fn prefix(components: &[&str], end: usize) -> String {
        if end == 0 {
            return "/".to_owned();
        }
        let mut path = String::new();
        for component in &components[..end] {
            path.push('/');
            path.push_str(component);
        }
        path
    }

    fn lookup<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        for (i, component) in components.iter().enumerate() {
            match current {
                Entry::Dir(children) => {
                    current = children
                        .get(*component)
                        .ok_or_else(|| Error::Missing(Self::prefix(components, i + 1)))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::prefix(components, i)));
                }
            }
        }
        Ok(current)
    }

    fn lookup_parent_mut<'a>(
        &'a mut self,
        components: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let mut current = &mut self.root;
        for (i, component) in components[..components.len() - 1].iter().enumerate() {
            match current {
                Entry::Dir(children) => {
                    current = children
                        .get_mut(*component)
                        .ok_or_else(|| Error::Missing(Self::prefix(components, i + 1)))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::prefix(components, i)));
                }
            }
        }

        match current {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::prefix(
                components,
                components.len() - 1,
            ))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let final_path = Self::prefix(&components, components.len());
        let parent = self.lookup_parent_mut(&components)?;
        if parent.contains_key(components[components.len() - 1]) {
            return Err(Error::Exists(final_path));
        }
        parent.insert(
            components[components.len() - 1].to_owned(),
            Entry::Dir(BTreeMap::new()),
        );
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let final_path = Self::prefix(&components, components.len());
        let parent = self.lookup_parent_mut(&components)?;
        match parent.get_mut(components[components.len() - 1]) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(final_path)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(
                    components[components.len() - 1].to_owned(),
                    Entry::File(data.to_vec()),
                );
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        let final_path = Self::prefix(&components, components.len());
        match self.lookup(&components)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir(_) => Err(Error::IsDir(final_path)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        let final_path = Self::prefix(&components, components.len());
        match self.lookup(&components)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(final_path)),
        }
    }
}
