#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    occupied_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut customer_order: Vec<usize> = (0..customers.len()).collect();
    customer_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut server_states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            occupied_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0;
    let mut finished_customers = 0;
    let mut now: Option<u64> = None;

    while finished_customers < customers.len() || server_states.iter().any(|s| !s.opened) {
        let next_arrival = customer_order
            .get(next_customer)
            .map(|&i| customers[i].arrival);
        let next_open = server_states
            .iter()
            .filter(|s| !s.opened)
            .map(|s| s.server.opens)
            .min();
        let next_finish = server_states
            .iter()
            .filter_map(|s| s.occupied_until)
            .min();

        let tick = match (next_arrival, next_open, next_finish) {
            (None, None, None) => break,
            (a, o, f) => [a, o, f].into_iter().flatten().min().unwrap(),
        };

        if now.is_some_and(|previous| tick < previous) {
            unreachable!();
        }
        now = Some(tick);

        let mut completing: Vec<usize> = server_states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (s.occupied_until == Some(tick)).then_some(i))
            .collect();
        completing.sort_by_key(|&i| server_states[i].server.id);
        for i in completing {
            let customer_id = match server_states[i].occupied_until {
                Some(_) => {
                    // The currently served customer is tracked by the scheduled completion.
                    // It is stored separately below in the completion records.
                    0
                }
                None => unreachable!(),
            };
            let _ = customer_id;
        }

        // Finish records are tracked by the active assignments.
        let mut finish_indices: Vec<usize> = active
            .iter()
            .enumerate()
            .filter_map(|(i, assignment)| {
                (assignment.finish == tick).then_some(i)
            })
            .collect();
        finish_indices.sort_by_key(|&i| server_states[active[i].server_index].server.id);
        for index in finish_indices.into_iter().rev() {
            let assignment = active.remove(index);
            let server = &mut server_states[assignment.server_index];
            events.push(Event::Finish {
                time: tick,
                customer: customers[assignment.customer_index].id,
                server: server.server.id,
            });
            server.occupied_until = None;
            server.idle_since = tick;
            finished_customers += 1;
        }

        let mut opening_indices: Vec<usize> = server_states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.server.opens == tick).then_some(i))
            .collect();
        opening_indices.sort_by_key(|&i| std::cmp::Reverse(server_states[i].server.id));
        for i in opening_indices {
            server_states[i].opened = true;
            events.push(Event::Open {
                time: tick,
                server: server_states[i].server.id,
            });
        }

        while next_customer < customer_order.len()
            && customers[customer_order[next_customer]].arrival == tick
        {
            let customer_index = customer_order[next_customer];
            events.push(Event::Arrive {
                time: tick,
                customer: customers[customer_index].id,
            });
            waiting.push(customer_index);
            next_customer += 1;
        }

        while !waiting.is_empty() {
            let selected = server_states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && s.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &server_states[server_index].server;
            let after_service = tick.checked_add(customer.service).ok_or(Error::TimeOverflow {
                customer: customer.id,
                server: server.id,
                time: tick,
            })?;
            let finish = after_service
                .checked_add(server.overhead)
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: tick,
                })?;

            events.push(Event::Start {
                time: tick,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == tick {
                events.push(Event::Finish {
                    time: tick,
                    customer: customer.id,
                    server: server.id,
                });
                server_states[server_index].idle_since = tick;
                finished_customers += 1;
            } else {
                server_states[server_index].occupied_until = Some(finish);
                active.push(Assignment {
                    customer_index,
                    server_index,
                    finish,
                });
            }
        }
    }

    Ok(events)
}

#[derive(Clone)]
struct Assignment {
    customer_index: usize,
    server_index: usize,
    finish: u64,
}
