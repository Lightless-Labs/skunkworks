```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components: Vec<&str> = Vec::new();
        if !path.starts_with('/') {
            components.extend(self.cwd.split('/').filter(|part| !part.is_empty()));
        }

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                _ => components.push(part),
            }
        }

        if components.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", components.join("/")))
        }
    }

    fn parent_path(path: &str) -> &str {
        match path.rfind('/') {
            Some(0) | None => "/",
            Some(i) => &path[..i],
        }
    }

    fn basename(path: &str) -> &str {
        path.rsplit('/').next().unwrap_or("")
    }

    fn lookup(&self, path: &str, allow_missing_final: bool) -> Result<Option<&Entry>, Error> {
        if path == "/" {
            return Ok(self.entries.get("/"));
        }

        let mut prefix = String::new();
        let components: Vec<&str> = path.split('/').filter(|part| !part.is_empty()).collect();

        for (i, component) in components.iter().enumerate() {
            prefix.push('/');
            prefix.push_str(component);
            let is_final = i + 1 == components.len();

            match self.entries.get(&prefix) {
                Some(Entry::Dir) => {
                    if is_final {
                        return Ok(self.entries.get(&prefix));
                    }
                }
                Some(Entry::File(_)) => {
                    if is_final {
                        return Ok(self.entries.get(&prefix));
                    }
                    return Err(Error::NotDir(prefix));
                }
                None => {
                    if is_final && allow_missing_final {
                        return Ok(None);
                    }
                    return Err(Error::Missing(prefix));
                }
            }
        }

        Ok(self.entries.get(path))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Exists(canonical));
        }

        let parent = Self::parent_path(&canonical);
        match self.lookup(parent, false)? {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent.to_string())),
            None => return Err(Error::Missing(parent.to_string())),
        }

        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }

        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::IsDir(canonical));
        }

        let parent = Self::parent_path(&canonical);
        match self.lookup(parent, false)? {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent.to_string())),
            None => return Err(Error::Missing(parent.to_string())),
        }

        match self.entries.get(&canonical) {
            Some(Entry::Dir) => return Err(Error::IsDir(canonical)),
            Some(Entry::File(_)) | None => {}
        }

        self.entries
            .insert(canonical, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        match self.lookup(&canonical, false)? {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            None => Err(Error::Missing(canonical)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        match self.lookup(&canonical, false)? {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(canonical)),
            None => return Err(Error::Missing(canonical)),
        }

        let mut result = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == "/" {
                continue;
            }
            if Self::parent_path(entry_path) == canonical {
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                result.push((Self::basename(entry_path).to_string(), kind));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        match self.lookup(&canonical, false)? {
            Some(Entry::Dir) => {
                self.cwd = canonical;
                Ok(())
            }
            Some(Entry::File(_)) => Err(Error::NotDir(canonical)),
            None => Err(Error::Missing(canonical)),
        }
    }
}```
