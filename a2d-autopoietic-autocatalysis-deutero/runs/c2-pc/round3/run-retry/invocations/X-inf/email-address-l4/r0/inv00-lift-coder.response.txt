use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while input.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;

    let mut quote_protected = false;
    let mut quote_closed = false;

    while i < input.len() {
        if quote_protected && !quote_closed {
            let b = input[i];
            out.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    out.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    quote_closed = true;
                }
                i += 1;
            }
            continue;
        }

        if !quote_protected && out.is_empty() && input[i] == b'"' {
            quote_protected = true;
            quote_closed = false;
            out.push(input[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        if input[i] == b'\r' || input[i] == b'\n' {
            let Some(end) = fold_end(input, i) else {
                return Err(EmailError::BadFold { at: i });
            };
            let start = i;
            i = end;
            runs.push(Run { start, end });
            continue;
        }

        if is_hws(input[i]) {
            let start = i;
            while i < input.len() && is_hws(input[i]) {
                i += 1;
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if input[i] == b'(' {
            let start = i;
            let mut stack = vec![i];
            i += 1;
            while i < input.len() {
                let b = input[i];
                if b == b'\\' {
                    let slash = i;
                    if i + 1 >= input.len()
                        || !(0x20..=0x7e).contains(&input[i + 1])
                    {
                        return Err(EmailError::CommentEscape { at: slash });
                    }
                    i += 2;
                    continue;
                }
                if b == b'\r' || b == b'\n' {
                    let Some(end) = fold_end(input, i) else {
                        return Err(EmailError::BadFold { at: i });
                    };
                    i = end;
                    continue;
                }
                if b == b'(' {
                    if stack.len() == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    stack.push(i);
                    i += 1;
                    continue;
                }
                if b == b')' {
                    stack.pop();
                    i += 1;
                    if stack.is_empty() {
                        break;
                    }
                    continue;
                }
                if !(0x20..=0x7e).contains(&b) && b != b'\t' {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
            }
            if !stack.is_empty() {
                return Err(EmailError::UnclosedComment { at: start });
            }
            runs.push(Run { start, end: i });
            continue;
        }

        if input[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        out.push(input[i]);
        positions.push(i);
        i += 1;
    }

    Ok(Cleaned {
        bytes: out,
        positions,
        runs,
    })
}

fn pos_at(clean: &Cleaned, index: usize, input_len: usize) -> usize {
    clean.positions.get(index).copied().unwrap_or(input_len)
}

fn local_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(s: &[u8]) -> bool {
    if s.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in s.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == s.len() {
                return false;
            }
            previous_dot = true;
        } else if local_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_internal(
    input: &str,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let clean = cleanup(raw)?;
    let retained = clean.bytes.len();

    for run in &clean.runs {
        let before = clean.positions.iter().any(|&p| p < run.start);
        let after = clean.positions.iter().any(|&p| p >= run.end);
        let left_at = before
            && clean.bytes[..]
                .iter()
                .zip(clean.positions.iter())
                .any(|(&b, &p)| p < run.start && b == b'@');
        let right_at = after
            && clean.bytes[..]
                .iter()
                .zip(clean.positions.iter())
                .any(|(&b, &p)| p >= run.end && b == b'@');
        if (before && after) && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if retained == 0 {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = None;
    if clean.bytes.first() == Some(&b'"') {
        let mut i = 1;
        let mut closing = None;
        while i < retained {
            if clean.bytes[i] == b'\\' {
                i += 2;
            } else if clean.bytes[i] == b'"' {
                closing = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        match closing {
            Some(end) => protected_end = Some(end),
            None => return Err(EmailError::UnclosedQuote { at: pos_at(&clean, 0, raw.len()) }),
        }
    }

    let mut ats = Vec::new();
    for i in 0..retained {
        if clean.bytes[i] == b'@' && !protected_end.map(|e| i <= e).unwrap_or(false) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: pos_at(&clean, ats[1], raw.len()),
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos_at(&clean, sep, raw.len()),
        });
    }

    let local_raw = &clean.bytes[..sep];
    let local_positions = &clean.positions[..sep];
    let (mut local, mut local_origins, mut quoted) = if local_raw[0] == b'"' {
        let close = protected_end.unwrap_or(0);
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos_at(&clean, close + 1, raw.len()),
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            let b = local_raw[i];
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape {
                        at: local_positions[i],
                    });
                }
                let escaped = local_raw[i + 1];
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape {
                        at: local_positions[i],
                    });
                }
                decoded.push(escaped);
                origins.push(local_positions[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: local_positions[i],
                    });
                }
                decoded.push(b);
                origins.push(local_positions[i]);
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: local_positions[close],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (decoded, origins, true)
    } else {
        if local_raw.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_positions[64],
            });
        }
        for i in 0..local_raw.len() {
            let b = local_raw[i];
            if b == b'.' {
                if i == 0 || i + 1 == local_raw.len() || local_raw[i - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: local_positions[i],
                    });
                }
            } else if !local_char(b) {
                return Err(EmailError::LocalChar {
                    at: local_positions[i],
                });
            }
        }
        (local_raw.to_vec(), local_positions.to_vec(), false)
    };

    let domain_start = sep + 1;
    if domain_start >= retained {
        return Err(EmailError::EmptyDomain {
            at: pos_at(&clean, domain_start, raw.len()),
        });
    }
    let mut domain_end = retained;
    let root_dot = if clean.bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(clean.positions[domain_end])
    } else {
        None
    };
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos_at(&clean, domain_start, raw.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos_at(&clean, domain_start + 253, raw.len()),
        });
    }

    let mut labels = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || clean.bytes[i] == b'.' {
            if start == i {
                let at = if i == domain_end {
                    root_dot.unwrap_or_else(|| pos_at(&clean, start, raw.len()))
                } else {
                    pos_at(&clean, i, raw.len())
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let len = i - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: pos_at(&clean, start + 63, raw.len()),
                });
            }
            if clean.bytes[start] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: pos_at(&clean, start, raw.len()),
                });
            }
            if clean.bytes[i - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: pos_at(&clean, i - 1, raw.len()),
                });
            }
            for j in start..i {
                let b = clean.bytes[j];
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar {
                        at: pos_at(&clean, j, raw.len()),
                    });
                }
            }
            labels.push((start, i));
            start = i + 1;
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: pos_at(&clean, domain_start, raw.len()),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if clean.bytes[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: pos_at(&clean, last_start, raw.len()),
        });
    }

    let mut domain = clean.bytes[domain_start..domain_end].to_vec();
    domain.make_ascii_lowercase();

    Ok((
        Address {
            local: String::from_utf8(local).unwrap(),
            domain: String::from_utf8(domain).unwrap(),
            quoted,
        },
        local_origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_internal(input, policy.allow_single_label)?;
    let mut local = address.local.into_bytes();

    if policy.local_case == LocalCase::Lower {
        local.make_ascii_lowercase();
    }

    let mut origin_map = origins;
    if policy.strip_subaddress {
        if let Some(plus) = local.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origin_map.first().copied().unwrap_or(input.len()),
                });
            }
            local.truncate(plus);
            origin_map.truncate(plus);
        }
    }

    if !valid_dot_atom(&local) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }

    address.local = String::from_utf8(local).unwrap();
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut output = String::new();
    if address.quoted {
        output.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                output.push('\\');
            }
            output.push(ch);
        }
        output.push('"');
    } else {
        output.push_str(&address.local);
    }
    output.push('@');
    output.push_str(&address.domain);
    output
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    seen.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
