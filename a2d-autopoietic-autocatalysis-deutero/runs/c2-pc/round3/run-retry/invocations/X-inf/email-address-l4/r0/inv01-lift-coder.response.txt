#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn scan_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &str) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut map = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut in_quote = false;
    let mut quote_escape = false;
    let mut comment_depth = 0usize;
    let mut outer_open = 0usize;
    let mut run_start: Option<usize> = None;

    while i < bytes.len() {
        let b = bytes[i];

        if in_quote {
            out.push(b);
            map.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                in_quote = false;
            }
            i += 1;
            continue;
        }

        if comment_depth > 0 {
            if let Some(start) = run_start.take() {
                runs.push(Run { start, end: i });
            }
            if b == b'\\' {
                let Some(&next) = bytes.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !is_printable(next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = scan_fold(bytes, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
            } else if b == b')' {
                comment_depth -= 1;
            } else if !is_printable(b) && b != b'\t' {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if !out.is_empty() && out[0] == b'"' {
            in_quote = true;
            out.push(b);
            map.push(i);
            if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                in_quote = false;
            }
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            if b == b'(' {
                comment_depth = 1;
                outer_open = i;
            }
            i += 1;
            continue;
        }
        if b == b'\r' || b == b'\n' {
            if let Some(end) = scan_fold(bytes, i) {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                i = end;
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if let Some(start) = run_start.take() {
            runs.push(Run { start, end: i });
        }
        out.push(b);
        map.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = run_start {
        runs.push(Run {
            start,
            end: bytes.len(),
        });
    }

    // Runs spanning comments may have been interrupted above. Merge adjacent
    // removed ranges so each maximal CFWS run has its first source position.
    let mut merged: Vec<Run> = Vec::new();
    for r in runs {
        if let Some(last) = merged.last_mut() {
            if r.start <= last.end {
                last.end = last.end.max(r.end);
                continue;
            }
        }
        merged.push(r);
    }
    Ok((out, map, merged))
}

fn pos(map: &[usize], idx: usize, input_len: usize) -> usize {
    map.get(idx).copied().unwrap_or(input_len)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if b.is_ascii_alphanumeric()
            || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_internal(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let bytes = input.as_bytes();
    if let Some(i) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let (clean, map, runs) = cleanup(input)?;
    for r in &runs {
        let left = map.iter().position(|&p| p >= r.end);
        let right = map.iter().position(|&p| p >= r.end);
        let begins = r.start == 0;
        let ends = r.end == bytes.len();
        let adjacent_at = left
            .and_then(|i| clean.get(i))
            .is_some_and(|&b| b == b'@')
            || right
                .and_then(|i| clean.get(i))
                .is_some_and(|&b| b == b'@')
            || map
                .iter()
                .position(|&p| p >= r.start)
                .and_then(|i| i.checked_sub(1))
                .and_then(|i| clean.get(i))
                .is_some_and(|&b| b == b'@');
        if !(begins || ends || adjacent_at) {
            return Err(EmailError::CfwsBoundary { at: r.start });
        }
    }
    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: map[0] });
        }
    }

    let protected_end = quote_end.unwrap_or(0);
    let ats: Vec<usize> = clean
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| {
            if b == b'@' && (quote_end.is_none() || i > protected_end) {
                Some(i)
            } else {
                None
            }
        })
        .collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt { at: map[ats[1]] });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: map[sep] });
    }

    let local_bytes = &clean[..sep];
    let (local, quoted, origins) = if local_bytes[0] == b'"' {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&map, end + 1, bytes.len()),
            });
        }
        let mut decoded = Vec::new();
        let mut org = Vec::new();
        let mut i = 1;
        while i < end {
            if local_bytes[i] == b'\\' {
                let next = local_bytes.get(i + 1).copied().unwrap_or(0);
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: map[i] });
                }
                decoded.push(next);
                org.push(map[i]);
                i += 2;
            } else {
                let b = local_bytes[i];
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar { at: map[i] });
                }
                decoded.push(b);
                org.push(map[i]);
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: map[end] });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: org[64] });
        }
        (String::from_utf8(decoded).unwrap(), true, org)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: map[64] });
        }
        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: map[i] });
                }
            } else if !b.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&b) {
                return Err(EmailError::LocalChar { at: map[i] });
            }
        }
        (
            String::from_utf8(local_bytes.to_vec()).unwrap(),
            false,
            map[..sep].to_vec(),
        )
    };

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let mut root_dot = None;
    if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(map[domain_end]);
    }
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: pos(&map, domain_start, bytes.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong { at: map[domain_start + 253] });
    }

    let domain = &clean[domain_start..domain_end];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0;
    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            let label_start = start;
            let label_end = i;
            if label_start == label_end {
                let at = if label_start == domain.len() {
                    root_dot.unwrap_or_else(|| pos(&map, domain_start + label_start, bytes.len()))
                } else {
                    map[domain_start + label_start]
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let len = label_end - label_start;
            if len > 63 {
                return Err(EmailError::LabelTooLong { at: map[domain_start + label_start + 63] });
            }
            if domain[label_start] == b'-' {
                return Err(EmailError::LabelHyphen { at: map[domain_start + label_start] });
            }
            if domain[label_end - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: map[domain_start + label_end - 1] });
            }
            for j in label_start..label_end {
                let b = domain[j];
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar { at: map[domain_start + j] });
                }
            }
            labels.push((label_start, label_end));
            start = i + 1;
        }
    }
    if !allow_single_label && labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: map[domain_start] });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if domain[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: map[domain_start + last_start] });
    }

    let domain_string = domain.iter().map(|b| b.to_ascii_lowercase() as char).collect();
    Ok((
        Address {
            local,
            domain: domain_string,
            quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, false).map(|(a, _)| a)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_internal(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.get(plus).copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(plus);
        }
    }

    if !valid_dot_atom(address.local.as_bytes()) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(err) => result.errors.push((i, err)),
            Ok(address) => {
                if let Some((_, _, first)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    result.duplicate_of[i] = Some(*first);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), i));
                    result.source_indices.push(i);
                    result.addresses.push(address);
                }
            }
        }
    }
    result
}
