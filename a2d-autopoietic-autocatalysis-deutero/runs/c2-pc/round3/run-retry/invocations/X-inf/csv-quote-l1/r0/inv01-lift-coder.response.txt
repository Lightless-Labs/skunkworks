#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) { 3 } else { 0 };
    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;

    let mut quoted = false;
    let mut after_quote = false;
    let mut field_started = false;
    let mut quote_start = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record: record_index, byte: i });
        }

        if !quoted && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record: record_index, byte: i });
        }

        if quoted {
            if b == b'"' {
                if bytes.get(i + 1) == Some(&b'"') {
                    field.push('"');
                    i += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let ch = input[i..].chars().next().expect("valid UTF-8");
                field.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            match b {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    after_quote = false;
                    field_started = false;
                    record_started = true;
                    i += 1;
                }
                b'\n' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    after_quote = false;
                    field_started = false;
                    record_started = false;
                    i += 1;
                }
                b'\r' => {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    after_quote = false;
                    field_started = false;
                    record_started = false;
                    i += 2;
                }
                _ => return Err(CsvError::TrailingByte { record: record_index, byte: i }),
            }
            continue;
        }

        match b {
            b'"' if !field_started => {
                quoted = true;
                quote_start = i;
                field_started = true;
                record_started = true;
                i += 1;
            }
            b'"' => {
                return Err(CsvError::UnexpectedQuote { record: record_index, byte: i });
            }
            b',' => {
                record.push(std::mem::take(&mut field));
                field_started = false;
                record_started = true;
                i += 1;
            }
            b'\n' => {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                field_started = false;
                record_started = false;
                i += 1;
            }
            b'\r' => {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                record_index += 1;
                field_started = false;
                record_started = false;
                i += 2;
            }
            _ => {
                let ch = input[i..].chars().next().expect("valid UTF-8");
                field.push(ch);
                field_started = true;
                record_started = true;
                i += ch.len_utf8();
            }
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote { record: record_index, byte: quote_start });
    }

    if after_quote || record_started || !record.is_empty() || !field.is_empty() {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}
