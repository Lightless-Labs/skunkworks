use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate_and_normalize(path: &str, cwd: &str) -> Result<String, Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }

    let mut components: Vec<&str> = Vec::new();
    if !path.starts_with('/') {
        components.extend(cwd.split('/').filter(|c| !c.is_empty()));
    }

    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                components.pop();
            }
            name => components.push(name),
        }
    }

    if components.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", components.join("/")))
    }
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|c| !c.is_empty()).collect()
}

fn parent_and_name(path: &str) -> (String, String) {
    let mut parts = components(path);
    let name = parts.pop().unwrap_or("").to_owned();
    let parent = if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    };
    (parent, name)
}

fn join_child(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{}", name)
    } else {
        format!("{}/{}", parent, name)
    }
}

fn is_at_or_below(path: &str, ancestor: &str) -> bool {
    path == ancestor
        || (ancestor == "/" && path.starts_with('/'))
        || (path.starts_with(ancestor)
            && path.as_bytes().get(ancestor.len()) == Some(&b'/'))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn resolve_parent(&self, parent: &str) -> Result<(), Error> {
        let mut prefix = String::new();
        for component in components(parent) {
            prefix = join_child(&prefix_or_root(&prefix), component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn resolve_existing(&self, path: &str) -> Result<&Entry, Error> {
        if path == "/" {
            return self
                .entries
                .get("/")
                .ok_or_else(|| Error::Missing("/".to_owned()));
        }

        let parts = components(path);
        let mut prefix = String::new();
        for (i, component) in parts.iter().enumerate() {
            prefix = join_child(&prefix_or_root(&prefix), component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(entry) if i + 1 < parts.len() && !matches!(entry, Entry::Dir) => {
                    return Err(Error::NotDir(prefix));
                }
                Some(entry) => {
                    if i + 1 == parts.len() {
                        return Ok(entry);
                    }
                }
            }
        }
        self.entries
            .get("/")
            .ok_or_else(|| Error::Missing("/".to_owned()))
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && is_at_or_below(candidate, path))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = validate_and_normalize(path, &self.cwd)?;
        if path == "/" {
            return Err(Error::Exists(path));
        }
        let (parent, _) = parent_and_name(&path);
        self.resolve_parent(&parent)?;
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = validate_and_normalize(path, &self.cwd)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }
        let (parent, _) = parent_and_name(&path);
        self.resolve_parent(&parent)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(_)) | None => {
                self.entries.insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = validate_and_normalize(path, &self.cwd)?;
        match self.resolve_existing(&path)? {
            Entry::Dir => Err(Error::IsDir(path)),
            Entry::File(data) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = validate_and_normalize(path, &self.cwd)?;
        match self.resolve_existing(&path)? {
            Entry::File(_) => return Err(Error::NotDir(path)),
            Entry::Dir => {}
        }

        let mut result = Vec::new();
        for (candidate, entry) in &self.entries {
            if candidate == "/" || candidate == &path {
                continue;
            }
            let (parent, name) = parent_and_name(candidate);
            if parent == path {
                result.push((name, entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = validate_and_normalize(path, &self.cwd)?;
        match self.resolve_existing(&path)? {
            Entry::Dir => {
                self.cwd = path;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = validate_and_normalize(from, &self.cwd)?;
        let to = validate_and_normalize(to, &self.cwd)?;

        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (from_parent, _) = parent_and_name(&from);
        self.resolve_parent(&from_parent)?;
        let source = self
            .entries
            .get(&from)
            .cloned()
            .ok_or_else(|| Error::Missing(from.clone()))?;

        if matches!(source, Entry::Dir) && is_at_or_below(&self.cwd, &from) {
            return Err(Error::Busy(from));
        }

        let (to_parent, _) = parent_and_name(&to);
        self.resolve_parent(&to_parent)?;

        if matches!(self.entries.get(&to), Some(Entry::Dir))
            && is_at_or_below(&self.cwd, &to)
        {
            return Err(Error::Busy(to));
        }

        if from == to {
            return Ok(());
        }

        if matches!(source, Entry::Dir) && is_at_or_below(&to, &from) {
            return Err(Error::Cycle(to));
        }

        if let Some(destination) = self.entries.get(&to) {
            if !overwrite {
                return Err(Error::Exists(to));
            }
            match (&source, destination) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(to)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(to)),
                (Entry::Dir, Entry::Dir) if self.has_children(&to) => {
                    return Err(Error::NotEmpty(to));
                }
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| is_at_or_below(path, &from))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if self.entries.contains_key(&to) {
            let destination_paths: Vec<String> = self
                .entries
                .keys()
                .filter(|path| is_at_or_below(path, &to))
                .cloned()
                .collect();
            for path in destination_paths {
                self.entries.remove(&path);
            }
        }

        for (path, _) in &moving {
            self.entries.remove(path);
        }

        for (old_path, entry) in moving {
            let suffix = &old_path[from.len()..];
            let new_path = format!("{}{}", to, suffix);
            self.entries.insert(new_path, entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = validate_and_normalize(path, &self.cwd)?;
        if path == "/" {
            return Err(Error::Root);
        }

        let (parent, _) = parent_and_name(&path);
        self.resolve_parent(&parent)?;
        let entry = self
            .entries
            .get(&path)
            .ok_or_else(|| Error::Missing(path.clone()))?;

        if matches!(entry, Entry::Dir) && is_at_or_below(&self.cwd, &path) {
            return Err(Error::Busy(path));
        }

        if matches!(entry, Entry::Dir) && self.has_children(&path) && !recursive {
            return Err(Error::NotEmpty(path));
        }

        let remove_paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| is_at_or_below(candidate, &path))
            .cloned()
            .collect();
        for candidate in remove_paths {
            self.entries.remove(&candidate);
        }
        Ok(())
    }
}

fn prefix_or_root(prefix: &str) -> &str {
    if prefix.is_empty() {
        "/"
    } else {
        prefix
    }
}
