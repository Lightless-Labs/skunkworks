use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts: Vec<&str> = Vec::new();
        if !path.starts_with('/') {
            parts.extend(self.cwd.split('/').filter(|s| !s.is_empty()));
        }

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part),
            }
        }

        if parts.is_empty() {
            Ok("/".to_owned())
        } else {
            Ok(format!("/{}", parts.join("/")))
        }
    }

    fn parent_and_name(path: &str) -> (String, String) {
        let i = path.rfind('/').unwrap_or(0);
        if i == 0 {
            ("/".to_owned(), path[1..].to_owned())
        } else {
            (path[..i].to_owned(), path[i + 1..].to_owned())
        }
    }

    fn resolve_existing(&self, path: &str) -> Result<&Entry, Error> {
        if path == "/" {
            return self
                .entries
                .get("/")
                .ok_or_else(|| Error::Missing("/".to_owned()));
        }

        let mut prefix = String::new();
        let components: Vec<&str> = path[1..].split('/').collect();
        for (i, component) in components.iter().enumerate() {
            prefix.push('/');
            prefix.push_str(component);

            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(entry) if i + 1 < components.len() && !matches!(entry, Entry::Dir) => {
                    return Err(Error::NotDir(prefix));
                }
                Some(entry) if i + 1 == components.len() => return Ok(entry),
                Some(_) => {}
            }
        }

        self.entries
            .get("/")
            .ok_or_else(|| Error::Missing("/".to_owned()))
    }

    fn check_parent(&self, parent: &str) -> Result<(), Error> {
        match self.resolve_existing(parent)? {
            Entry::Dir => Ok(()),
            Entry::File(_) => Err(Error::NotDir(parent.to_owned())),
        }
    }

    fn contains_path(directory: &str, path: &str) -> bool {
        directory == "/"
            || path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn children(&self, path: &str) -> Vec<String> {
        let prefix = if path == "/" {
            "/".to_owned()
        } else {
            format!("{}/", path)
        };

        self.entries
            .keys()
            .filter_map(|candidate| {
                let remainder = candidate.strip_prefix(&prefix)?;
                if !remainder.is_empty() && !remainder.contains('/') {
                    Some(candidate.clone())
                } else {
                    None
                }
            })
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Exists(path));
        }

        let (parent, _) = Self::parent_and_name(&path);
        self.check_parent(&parent)?;
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }

        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }

        let (parent, _) = Self::parent_and_name(&path);
        self.check_parent(&parent)?;

        match self.entries.get(&path) {
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(_)) | None => {
                self.entries
                    .insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        match self.resolve_existing(&path)? {
            Entry::File(data) => Ok(data.clone()),
            Entry::Dir => Err(Error::IsDir(path)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        match self.resolve_existing(&path)? {
            Entry::File(_) => return Err(Error::NotDir(path)),
            Entry::Dir => {}
        }

        let mut result: Vec<(String, Kind)> = self
            .children(&path)
            .into_iter()
            .filter_map(|child| {
                let name = child.rsplit('/').next()?.to_owned();
                let kind = self.entries.get(&child)?.kind();
                Some((name, kind))
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        match self.resolve_existing(&path)? {
            Entry::Dir => {
                self.cwd = path;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = self.normalize(from)?;
        let to = self.normalize(to)?;

        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (from_parent, _) = Self::parent_and_name(&from);
        self.check_parent(&from_parent)?;
        let source = self
            .entries
            .get(&from)
            .cloned()
            .ok_or_else(|| Error::Missing(from.clone()))?;

        if matches!(source, Entry::Dir) && Self::contains_path(&from, &self.cwd) {
            return Err(Error::Busy(from));
        }

        let (to_parent, _) = Self::parent_and_name(&to);
        self.check_parent(&to_parent)?;

        if let Some(Entry::Dir) = self.entries.get(&to) {
            if Self::contains_path(&to, &self.cwd) {
                return Err(Error::Busy(to));
            }
        }

        if from == to {
            return Ok(());
        }

        if matches!(source, Entry::Dir) && Self::contains_path(&from, &to) {
            return Err(Error::Cycle(to));
        }

        let destination = self.entries.get(&to).cloned();
        if destination.is_some() && !overwrite {
            return Err(Error::Exists(to));
        }

        if overwrite {
            if let Some(ref target) = destination {
                match (&source, target) {
                    (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(to)),
                    (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(to)),
                    (Entry::Dir, Entry::Dir) if !self.children(&to).is_empty() => {
                        return Err(Error::NotEmpty(to));
                    }
                    _ => {}
                }
            }
        }

        let source_paths: Vec<(String, String)> = self
            .entries
            .keys()
            .filter(|candidate| Self::contains_path(&from, candidate))
            .map(|candidate| {
                let suffix = &candidate[from.len()..];
                (candidate.clone(), format!("{to}{suffix}"))
            })
            .collect();

        if destination.is_some() {
            self.entries.remove(&to);
        }

        let mut moved = Vec::with_capacity(source_paths.len());
        for (old, new) in &source_paths {
            if let Some(entry) = self.entries.remove(old) {
                moved.push((new.clone(), entry));
            }
        }
        for (new, entry) in moved {
            self.entries.insert(new, entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        let (parent, _) = Self::parent_and_name(&path);
        self.check_parent(&parent)?;
        let entry = self
            .entries
            .get(&path)
            .cloned()
            .ok_or_else(|| Error::Missing(path.clone()))?;

        if matches!(entry, Entry::Dir) && Self::contains_path(&path, &self.cwd) {
            return Err(Error::Busy(path));
        }

        let children = if matches!(entry, Entry::Dir) {
            self.children(&path)
        } else {
            Vec::new()
        };
        if !recursive && !children.is_empty() {
            return Err(Error::NotEmpty(path));
        }

        if recursive {
            let to_remove: Vec<String> = self
                .entries
                .keys()
                .filter(|candidate| Self::contains_path(&path, candidate))
                .cloned()
                .collect();
            for candidate in to_remove {
                self.entries.remove(&candidate);
            }
        } else {
            self.entries.remove(&path);
        }

        Ok(())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
