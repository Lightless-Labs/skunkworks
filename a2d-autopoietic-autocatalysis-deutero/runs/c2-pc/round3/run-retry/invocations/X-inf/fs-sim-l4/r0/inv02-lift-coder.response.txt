use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    if path == "/" {
        Vec::new()
    } else {
        path[1..].split('/').map(str::to_owned).collect()
    }
}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn normalize(path: &str, cwd: &str) -> String {
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(cwd)
    };
    for item in path.split('/') {
        match item {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_owned()),
        }
    }
    canonical(&parts)
}

fn parent_and_name(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_owned(), String::new());
    }
    let i = path.rfind('/').unwrap();
    let parent = if i == 0 { "/" } else { &path[..i] };
    (parent.to_owned(), path[i + 1..].to_owned())
}

fn join(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{}", name)
    } else {
        format!("{}/{}", parent, name)
    }
}

fn within(path: &str, dir: &str) -> bool {
    dir == "/" || path == dir || path.strip_prefix(dir).is_some_and(|s| s.starts_with('/'))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let mut todo = components(path);
        let mut physical: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if todo.is_empty() {
                return Ok(canonical(&physical));
            }
            let name = todo.remove(0);
            let candidate_parts = {
                let mut p = physical.clone();
                p.push(name);
                p
            };
            let candidate = canonical(&candidate_parts);
            let is_final = todo.is_empty();

            match self.entries.get(&candidate) {
                None if is_final && create_final => return Ok(candidate),
                None => return Err(Error::Missing(candidate)),
                Some(Entry::Link(target)) if follow_final || !is_final => {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;

                    let mut base = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        physical.clone()
                    };
                    for part in target.split('/') {
                        match part {
                            "" | "." => {}
                            ".." => {
                                base.pop();
                            }
                            other => base.push(other.to_owned()),
                        }
                    }
                    let mut expanded = base;
                    expanded.extend(todo);
                    todo = expanded;
                    physical.clear();
                }
                Some(entry) => {
                    if !is_final && !matches!(entry, Entry::Dir) {
                        return Err(Error::NotDir(candidate));
                    }
                    physical.push(name);
                }
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = parent_and_name(path);
        let resolved = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Ok((resolved, name)),
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        validate(path)?;
        let normalized = normalize(path, &self.cwd);
        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = join(&parent, &name);
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        validate(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, true)?;
        if resolved == "/" {
            return Err(Error::IsDir("/".to_owned()));
        }
        if let Some(entry) = self.entries.get(&resolved) {
            if matches!(entry, Entry::Dir) {
                return Err(Error::IsDir(resolved));
            }
        } else {
            let (parent, _) = parent_and_name(&resolved);
            match self.entries.get(&parent) {
                Some(Entry::Dir) => {}
                Some(_) => return Err(Error::NotDir(parent)),
                None => return Err(Error::Missing(parent)),
            }
        }
        self.entries.insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        validate(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        validate(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut result = Vec::new();
                for (p, e) in &self.entries {
                    if p == "/" {
                        continue;
                    }
                    let (parent, name) = parent_and_name(p);
                    if parent == resolved {
                        result.push((name, e.kind()));
                    }
                }
                result.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(result)
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        validate(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        validate(from)?;
        let source_path = normalize(from, &self.cwd);
        validate(to)?;
        let dest_path = normalize(to, &self.cwd);
        if source_path == "/" || dest_path == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source_path)?;
        let source = join(&source_parent, &source_name);
        let source_entry = self.entries.get(&source).cloned().ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (dest_parent, dest_name) = self.resolve_parent(&dest_path)?;
        let destination = join(&dest_parent, &dest_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir)) && within(&self.cwd, &destination) {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let dest_entry = self.entries.get(&destination).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }
        if let Some(ref dest) = dest_entry {
            match (&source_entry, dest) {
                (Entry::Dir, Entry::Dir) => {
                    if self.entries.keys().any(|p| p != &destination && within(p, &destination)) {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        if dest_entry.is_some() {
            self.entries.remove(&destination);
        }

        let moved: Vec<(String, Entry)> = self.entries.iter()
            .filter(|(p, _)| within(p, &source))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();
        for (p, _) in &moved {
            self.entries.remove(p);
        }
        for (p, e) in moved {
            let suffix = if p == source { String::new() } else { p[source.len()..].to_owned() };
            self.entries.insert(format!("{}{}", destination, suffix), e);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        validate(path)?;
        let normalized = normalize(path, &self.cwd);
        if normalized == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = join(&parent, &name);
        let entry = self.entries.get(&target).ok_or_else(|| Error::Missing(target.clone()))?;
        if matches!(entry, Entry::Dir) && within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && !recursive && self.entries.keys().any(|p| p != &target && within(p, &target)) {
            return Err(Error::NotEmpty(target));
        }
        let removed: Vec<String> = self.entries.keys().filter(|p| within(p, &target)).cloned().collect();
        for p in removed {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        validate(link)?;
        let normalized = normalize(link, &self.cwd);
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent, name) = self.resolve_parent(&normalized)?;
        let resolved = join(&parent, &name);
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        validate(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        validate(path)?;
        let normalized = normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        let entry = self.entries.get(&resolved).ok_or_else(|| Error::Missing(resolved.clone()))?;
        if !matches!(entry, Entry::Dir | Entry::File(_)) {
            return Err(Error::NotDir(resolved));
        }

        let mut rows = Vec::new();
        for (p, e) in &self.entries {
            if within(p, &resolved) {
                let size = match e {
                    Entry::File(data) => data.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                    Entry::Dir => self.entries.iter()
                        .filter(|(q, child)| within(q, p) && *q != p)
                        .map(|(_, child)| match child {
                            Entry::File(data) => data.len() as u64,
                            Entry::Link(target) => target.len() as u64,
                            Entry::Dir => 0,
                        })
                        .sum(),
                };
                rows.push((p.clone(), size));
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
