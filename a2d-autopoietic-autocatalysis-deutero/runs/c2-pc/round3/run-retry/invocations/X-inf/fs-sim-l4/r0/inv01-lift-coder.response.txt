use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_owned).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for c in path.split('/') {
            match c {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(c.to_string()),
            }
        }
        Self::canonical(&parts)
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        match path.rfind('/') {
            Some(0) => "/".to_string(),
            Some(i) => path[..i].to_string(),
            _ => "/".to_string(),
        }
    }

    fn basename(path: &str) -> String {
        if path == "/" {
            String::new()
        } else {
            path.rsplit('/').next().unwrap_or("").to_string()
        }
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            return path.starts_with('/');
        }
        path == dir
            || path
                .strip_prefix(dir)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn normalize_target(target: &str, link_parent: &str, suffix: &[String]) -> String {
        let mut parts = if target.starts_with('/') {
            Vec::new()
        } else {
            Self::components(link_parent)
        };
        for c in target.split('/') {
            match c {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(c.to_string()),
            }
        }
        parts.extend(suffix.iter().cloned());
        Self::canonical(&parts)
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        create_missing_final: bool,
    ) -> Result<String, Error> {
        let mut pending = Self::components(path);
        let mut current = Vec::<String>::new();
        let mut visited = HashSet::<String>::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(Self::canonical(&current));
            }

            let name = pending.remove(0);
            let candidate_parts = {
                let mut p = current.clone();
                p.push(name);
                p
            };
            let candidate = Self::canonical(&candidate_parts);
            let is_final = pending.is_empty();

            let Some(entry) = self.entries.get(&candidate) else {
                if create_missing_final && is_final {
                    return Ok(candidate);
                }
                return Err(Error::Missing(candidate));
            };

            let should_follow = matches!(entry, Entry::Link(_)) && (!is_final || follow_final);
            if should_follow {
                if !visited.insert(candidate.clone()) {
                    return Err(Error::Loop(candidate));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(candidate));
                }
                expansions += 1;

                let Entry::Link(target) = entry else {
                    unreachable!();
                };
                let link_parent = Self::parent(&candidate);
                let expanded = Self::normalize_target(target, &link_parent, &pending);
                pending = Self::components(&expanded);
                current.clear();
                continue;
            }

            if !is_final && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(candidate));
            }
            current.push(name);
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((
                resolved_parent.clone(),
                Self::child_path(&resolved_parent, &Self::basename(path)),
            )),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let (_, final_path) = self.resolve_parent(&normalized)?;
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, true)?;
        match self.entries.get_mut(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let parent = Self::parent(&resolved);
                match self.entries.get(&parent) {
                    Some(Entry::Dir) => {
                        self.entries
                            .insert(resolved, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(parent)),
                    None => Err(Error::Missing(parent)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let mut children = Vec::new();
        for (child, entry) in &self.entries {
            if child == &resolved {
                continue;
            }
            if Self::parent(child) == resolved {
                children.push((Self::basename(child), entry.kind()));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        let from_norm = Self::normalize(from, &self.cwd);
        Self::validate(to)?;
        let to_norm = Self::normalize(to, &self.cwd);

        if from_norm == "/" || to_norm == "/" {
            return Err(Error::Root);
        }

        let (_, source) = self.resolve_parent(&from_norm)?;
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (_, destination) = self.resolve_parent(&to_norm)?;

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let dest_entry = self.entries.get(&destination).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref dest) = dest_entry {
            match (&source_entry, dest) {
                (Entry::Dir, Entry::Dir) => {
                    if self
                        .entries
                        .keys()
                        .any(|p| p != &destination && Self::is_within(p, &destination))
                    {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        let mut moved = Vec::<(String, Entry)>::new();
        for (old_path, entry) in &self.entries {
            if Self::is_within(old_path, &source) {
                let suffix = &old_path[source.len()..];
                moved.push((format!("{}{}", destination, suffix), entry.clone()));
            }
        }

        if dest_entry.is_some() {
            self.entries.remove(&destination);
        }
        let old_paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &source))
            .cloned()
            .collect();
        for p in old_paths {
            self.entries.remove(&p);
        }
        for (new_path, entry) in moved {
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        if normalized == "/" {
            return Err(Error::Root);
        }

        let (_, target) = self.resolve_parent(&normalized)?;
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }

        if matches!(entry, Entry::Dir) {
            let nonempty = self
                .entries
                .keys()
                .any(|p| p != &target && Self::is_within(p, &target));
            if nonempty && !recursive {
                return Err(Error::NotEmpty(target));
            }
        }

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &target))
            .cloned()
            .collect();
        for p in paths {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let link_norm = Self::normalize(link, &self.cwd);
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let (_, link_path) = self.resolve_parent(&link_norm)?;
        if self.entries.contains_key(&link_path) {
            return Err(Error::Exists(link_path));
        }
        self.entries
            .insert(link_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let normalized = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&normalized, true, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &resolved))
            .cloned()
            .collect();
        let mut rows = Vec::new();
        for p in paths {
            let size = match self.entries.get(&p).expect("entry key collected") {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => self
                    .entries
                    .iter()
                    .filter(|(child, entry)| {
                        matches!(entry, Entry::File(_) | Entry::Link(_))
                            && Self::is_within(child, &p)
                    })
                    .map(|(_, entry)| match entry {
                        Entry::File(data) => data.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
            };
            rows.push((p, size));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}
