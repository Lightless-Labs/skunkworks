#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(if round_up { q + 1 } else { q })
}

#[derive(Clone)]
struct CustomerState {
    customer: Customer,
    initial_deadline: Option<u64>,
    accepted: bool,
    terminal: bool,
    started: bool,
    remaining: u64,
    entry_time: u64,
    deadline: Option<u64>,
    waited: u64,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Active {
    index: usize,
    start_time: u64,
    work_at: u64,
    finish: u64,
}

fn event(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn better(a: usize, b: usize, states: &[CustomerState]) -> bool {
    let x = &states[a];
    let y = &states[b];
    let xp = x.customer.priority;
    let yp = y.customer.priority;
    if xp != yp {
        return xp < yp;
    }
    match (x.pause_time, y.pause_time) {
        (Some(px), Some(py)) => (px, x.customer.id) < (py, y.customer.id),
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            (x.customer.arrival, std::cmp::Reverse(x.customer.id))
                < (y.customer.arrival, std::cmp::Reverse(y.customer.id))
        }
    }
}

fn best_waiter(waiting: &[usize], states: &[CustomerState]) -> Option<usize> {
    waiting.iter().copied().reduce(|a, b| if better(a, b, states) { a } else { b })
}

fn consume_wait(index: usize, time: u64, states: &mut [CustomerState], wait_ticks: &mut u128) {
    let elapsed = time - states[index].entry_time;
    states[index].waited += elapsed;
    *wait_ticks += elapsed as u128;
}

fn simulate_inner(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (i, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index: i });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index: i, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index: i, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow { index: i, id: c.id })?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut states: Vec<CustomerState> = customers.iter().enumerate().map(|(i, c)| CustomerState {
        customer: c.clone(),
        initial_deadline: initial_deadlines[i],
        accepted: false,
        terminal: false,
        started: false,
        remaining: c.service,
        entry_time: c.arrival,
        deadline: None,
        waited: 0,
        pause_time: None,
    }).collect();

    let mut cohort: Vec<usize> = (0..customers.len()).collect();
    cohort.sort_by_key(|&i| (states[i].customer.arrival, states[i].customer.priority, std::cmp::Reverse(states[i].customer.id)));

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();

    while next_arrival < cohort.len() || active.is_some() || !waiting.is_empty() {
        let mut candidate = u64::MAX;
        let mut has_candidate = false;
        if next_arrival < cohort.len() {
            candidate = states[cohort[next_arrival]].customer.arrival;
            has_candidate = true;
        }
        if let Some(a) = &active {
            for t in [a.finish, a.work_at] {
                if t > now && (!has_candidate || t < candidate) {
                    candidate = t;
                    has_candidate = true;
                }
            }
        }
        for &i in &waiting {
            if let Some(d) = states[i].deadline {
                if d > now && (!has_candidate || d < candidate) {
                    candidate = d;
                    has_candidate = true;
                }
            }
        }
        if !has_candidate {
            // All remaining work is handled inline at the current tick.
            candidate = now;
        }
        now = candidate;

        // Phase A.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let s = &mut states[a.index];
            busy_ticks += (now - a.start_time) as u128;
            s.terminal = true;
            completed += 1;
            turnaround_ticks += (now - s.customer.arrival) as u128;
            completion_order.push(s.customer.id);
            event(&mut events, now, s.customer.id, Kind::Finish);
        }

        // Phase B.
        let mut expired: Vec<usize> = waiting.iter().copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by_key(|&i| (states[i].deadline.unwrap(), states[i].customer.id));
        for i in expired {
            waiting.retain(|&x| x != i);
            consume_wait(i, now, &mut states, &mut wait_ticks);
            let s = &mut states[i];
            s.terminal = true;
            reneged += 1;
            event(&mut events, now, s.customer.id, Kind::Renege { waited: s.waited });
        }

        // Phase C.
        while next_arrival < cohort.len() && states[cohort[next_arrival]].customer.arrival == now {
            let i = cohort[next_arrival];
            next_arrival += 1;
            let id = states[i].customer.id;
            event(&mut events, now, id, Kind::Arrive);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                balked += 1;
                states[i].terminal = true;
                event(&mut events, now, id, Kind::Balk);
            } else {
                accepted += 1;
                states[i].accepted = true;
                states[i].entry_time = now;
                states[i].deadline = states[i].initial_deadline;
                waiting.push(i);
            }
        }

        // Phase D.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&waiting, &states) {
                    if states[w].customer.priority < states[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let remaining = a.finish - now;
                        let id = states[a.index].customer.id;
                        let by = states[w].customer.id;
                        busy_ticks += (now - a.start_time) as u128;
                        event(&mut events, now, id, Kind::Preempt { remaining, by });
                        preemptions += 1;
                        let deadline = match states[a.index].customer.patience {
                            Some(b) => Some(now.checked_add(b).ok_or(Error::TimeOverflow {
                                customer: id, time: now, stage: OverflowStage::Patience
                            })?),
                            None => None,
                        };
                        states[a.index].remaining = remaining;
                        states[a.index].entry_time = now;
                        states[a.index].deadline = deadline;
                        states[a.index].pause_time = Some(now);
                        waiting.push(a.index);
                    }
                }
            }
        }

        // Phase E.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&waiting, &states) else { break };
            waiting.retain(|&x| x != i);
            consume_wait(i, now, &mut states, &mut wait_ticks);
            let id = states[i].customer.id;
            let remaining = states[i].remaining;
            let switch = if last_dispatched.is_some_and(|old| old != id) { config.switch_cost } else { 0 };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::WorkAt
            })?;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::Finish
            })?;
            let kind = if states[i].started {
                Kind::Resume { remaining, work_at, finish }
            } else {
                states[i].started = true;
                started += 1;
                Kind::Start { remaining, work_at, finish }
            };
            last_dispatched = Some(id);
            event(&mut events, now, id, kind);
            active = Some(Active { index: i, start_time: now, work_at, finish });
            if finish == now {
                active = None;
                states[i].terminal = true;
                completed += 1;
                turnaround_ticks += (now - states[i].customer.arrival) as u128;
                completion_order.push(id);
                event(&mut events, now, id, Kind::Finish);
                continue;
            }
            break;
        }

        // Phase F.
        let mut expired: Vec<usize> = waiting.iter().copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by_key(|&i| (states[i].deadline.unwrap(), states[i].customer.id));
        for i in expired {
            waiting.retain(|&x| x != i);
            consume_wait(i, now, &mut states, &mut wait_ticks);
            let s = &mut states[i];
            s.terminal = true;
            reneged += 1;
            event(&mut events, now, s.customer.id, Kind::Renege { waited: s.waited });
        }

        if next_arrival >= cohort.len() && active.is_none() && waiting.is_empty() {
            break;
        }
    }

    let elapsed = events.last().map_or(0, |e| e.time);
    let mean_wait_milli = if accepted == 0 { 0 } else { rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap() };
    let mean_turnaround_milli = if completed == 0 { 0 } else { rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap() };
    let utilization_bps = if elapsed == 0 { 0 } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128).unwrap().min(u32::MAX as u128) as u32
    };
    Ok(Report {
        events,
        stats: Stats {
            accepted, balked, reneged, completed, started, preemptions, elapsed,
            busy_ticks, wait_ticks, turnaround_ticks, mean_wait_milli,
            mean_turnaround_milli, utilization_bps, completion_order,
        },
    })
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    simulate_inner(customers, config)
}
