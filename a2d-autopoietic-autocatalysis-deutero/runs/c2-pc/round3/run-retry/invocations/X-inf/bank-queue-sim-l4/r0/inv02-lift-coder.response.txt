#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };
    Some(quotient + u128::from(round_up))
}

#[derive(Clone)]
struct State {
    accepted: bool,
    started: bool,
    terminal: bool,
    remaining: u64,
    waited: u64,
    queue_since: u64,
    deadline: Option<u64>,
    paused_at: Option<u64>,
}

#[derive(Clone, Copy)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiting_order(a: usize, b: usize, customers: &[Customer], states: &[State]) -> std::cmp::Ordering {
    let ca = &customers[a];
    let cb = &customers[b];
    ca.priority
        .cmp(&cb.priority)
        .then_with(|| match (states[a].paused_at, states[b].paused_at) {
            (Some(pa), Some(pb)) => pa.cmp(&pb).then_with(|| ca.id.cmp(&cb.id)),
            (Some(_), None) => std::cmp::Ordering::Less,
            (None, Some(_)) => std::cmp::Ordering::Greater,
            (None, None) => ca
                .arrival
                .cmp(&cb.arrival)
                .then_with(|| cb.id.cmp(&ca.id)),
        })
}

fn best_waiter(waiting: &[usize], customers: &[Customer], states: &[State]) -> Option<usize> {
    waiting
        .iter()
        .copied()
        .min_by(|&a, &b| waiting_order(a, b, customers, states))
}

fn remove_waiter(waiting: &mut Vec<usize>, index: usize) {
    if let Some(pos) = waiting.iter().position(|&x| x == index) {
        waiting.remove(pos);
    }
}

fn renege_waiters(
    now: u64,
    waiting: &mut Vec<usize>,
    customers: &[Customer],
    states: &mut [State],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<usize> = waiting
        .iter()
        .copied()
        .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
        .collect();
    expired.sort_by_key(|&i| (states[i].deadline.unwrap(), customers[i].id));

    for i in expired {
        remove_waiter(waiting, i);
        let segment = now - states[i].queue_since;
        states[i].waited = states[i].waited.saturating_add(segment);
        stats.wait_ticks += segment as u128;
        states[i].terminal = true;
        stats.reneged += 1;
        emit(
            events,
            now,
            customers[i].id,
            Kind::Renege {
                waited: states[i].waited,
            },
        );
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        ids.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        if let Some(patience) = customer.patience {
            if customer.arrival.checked_add(patience).is_none() {
                return Err(Error::DeadlineOverflow {
                    index,
                    id: customer.id,
                });
            }
        }
    }

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    if customers.is_empty() {
        return Ok(Report { events, stats });
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<State> = customers
        .iter()
        .map(|c| State {
            accepted: false,
            started: false,
            terminal: false,
            remaining: c.service,
            waited: 0,
            queue_since: c.arrival,
            deadline: c.patience.map(|p| c.arrival + p),
            paused_at: None,
        })
        .collect();

    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0;
    let mut now = customers[order[0]].arrival;
    let mut last_dispatched: Option<usize> = None;

    loop {
        // Phase A: finish active work before any other event at this tick.
        if let Some(a) = active {
            if a.finish == now {
                active = None;
                states[a.index].terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - customers[a.index].arrival) as u128;
                stats.completion_order.push(customers[a.index].id);
                emit(&mut events, now, customers[a.index].id, Kind::Finish);
            }
        }

        // Phase B: expire customers that were already waiting.
        renege_waiters(
            now,
            &mut waiting,
            customers,
            &mut states,
            &mut events,
            &mut stats,
        );

        // Phase C: admit the full arrival cohort without dispatching between arrivals.
        while next_arrival < order.len() && customers[order[next_arrival]].arrival == now {
            let i = order[next_arrival];
            next_arrival += 1;
            emit(&mut events, now, customers[i].id, Kind::Arrive);
            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= limit {
                stats.balked += 1;
                states[i].terminal = true;
                emit(&mut events, now, customers[i].id, Kind::Balk);
            } else {
                states[i].accepted = true;
                stats.accepted += 1;
                states[i].queue_since = now;
                waiting.push(i);
            }
        }

        // Phase D: preempt only after the active switching protection ends.
        if let Some(a) = active {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&waiting, customers, &states) {
                    if customers[w].priority < customers[a.index].priority {
                        let remaining = a.finish - now;
                        emit(
                            &mut events,
                            now,
                            customers[a.index].id,
                            Kind::Preempt {
                                remaining,
                                by: customers[w].id,
                            },
                        );
                        active = None;
                        stats.preemptions += 1;
                        states[a.index].remaining = remaining;
                        states[a.index].queue_since = now;
                        states[a.index].paused_at = Some(now);
                        if let Some(patience) = customers[a.index].patience {
                            states[a.index].deadline = Some(now.checked_add(patience).ok_or(
                                Error::TimeOverflow {
                                    customer: customers[a.index].id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                },
                            )?);
                        }
                        waiting.push(a.index);
                    }
                }
            }
        }

        // Phase E: dispatch until occupied, or immediately finish zero-duration work.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&waiting, customers, &states) else {
                break;
            };
            remove_waiter(&mut waiting, i);

            let segment = now - states[i].queue_since;
            states[i].waited = states[i].waited.saturating_add(segment);
            stats.wait_ticks += segment as u128;

            let switch = if last_dispatched.is_some_and(|last| last != i) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: customers[i].id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(states[i].remaining)
                .ok_or(Error::TimeOverflow {
                    customer: customers[i].id,
                    time: now,
                    stage: OverflowStage::Finish,
                })?;

            let resumed = states[i].started;
            if resumed {
                emit(
                    &mut events,
                    now,
                    customers[i].id,
                    Kind::Resume {
                        remaining: states[i].remaining,
                        work_at,
                        finish,
                    },
                );
            } else {
                states[i].started = true;
                stats.started += 1;
                emit(
                    &mut events,
                    now,
                    customers[i].id,
                    Kind::Start {
                        remaining: states[i].remaining,
                        work_at,
                        finish,
                    },
                );
            }
            last_dispatched = Some(i);
            active = Some(Active {
                index: i,
                work_at,
                finish,
            });

            if finish == now {
                active = None;
                states[i].terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - customers[i].arrival) as u128;
                stats.completion_order.push(customers[i].id);
                emit(&mut events, now, customers[i].id, Kind::Finish);
                continue;
            }
            break;
        }

        // Phase F: expire waiters created or paused during this tick.
        renege_waiters(
            now,
            &mut waiting,
            customers,
            &mut states,
            &mut events,
            &mut stats,
        );

        if next_arrival == order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now && next_time.map_or(true, |old| t < old) {
                next_time = Some(t);
            }
        };

        if next_arrival < order.len() {
            consider(customers[order[next_arrival]].arrival);
        }
        if let Some(a) = active {
            consider(a.finish);
            if a.work_at > now {
                consider(a.work_at);
            }
        }
        for &i in &waiting {
            if let Some(deadline) = states[i].deadline {
                consider(deadline);
            }
        }

        if let Some(t) = next_time {
            now = t;
        } else {
            break;
        }
    }

    stats.elapsed = events.last().map_or(0, |event| event.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(1000 * stats.wait_ticks, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(1000 * stats.turnaround_ticks, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(10000 * stats.busy_ticks, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    // Reconstruct occupied intervals from dispatch and terminal/preemption events.
    let mut dispatch_time: Option<u64> = None;
    let mut busy = 0u128;
    for event in &events {
        match event.kind {
            Kind::Start { .. } | Kind::Resume { .. } => dispatch_time = Some(event.time),
            Kind::Preempt { .. } | Kind::Finish => {
                if let Some(start) = dispatch_time.take() {
                    busy += (event.time - start) as u128;
                }
            }
            _ => {}
        }
    }
    stats.busy_ticks = busy;
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(10000 * busy, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
