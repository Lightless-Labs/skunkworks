#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

pub fn schedule(
    tasks: &[Task],
    workers: usize,
    turnaround: u64,
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency in task and list order before other checks.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Deterministic Kahn order: always remove the smallest currently available ID.
    let mut available = std::collections::BTreeSet::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.insert(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(id) = available.pop_first() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut bits = vec![false; n];
            for &id in &order {
                bits[id] = true;
            }
            bits
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();

    // A worker's availability time includes its cooldown. A worker assigned at
    // the current time is reserved even when the task completes immediately.
    let mut worker_available = vec![0u64; workers];
    let mut worker_busy = vec![false; workers];

    // Running tasks are represented by (finish time, task ID, worker ID).
    let mut running: Vec<(u64, usize, usize)> = Vec::new();
    let mut events = Vec::new();
    let mut now = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete every task finishing now, ordered by task ID.
        running.sort_unstable_by_key(|&(finish, task, _)| (finish, task));
        let mut idx = 0;
        while idx < running.len() {
            if running[idx].0 == now {
                let (_, task, worker) = running.remove(idx);
                completed[task] = true;
                completed_count += 1;
                events.push(Event {
                    time: now,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
                for &child in &children[task] {
                    remaining_deps[child] -= 1;
                }
                worker_busy[worker] = false;
            } else {
                idx += 1;
            }
        }

        loop {
            // The lowest ID worker whose cooldown has expired is selected.
            let worker = (0..workers)
                .find(|&w| !worker_busy[w] && worker_available[w] <= now);

            let Some(worker) = worker else {
                break;
            };

            // Choose the highest-ranked currently eligible task.
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || remaining_deps[id] != 0
                {
                    continue;
                }
                let better = match best {
                    None => true,
                    Some(current) => {
                        tail[id] > tail[current]
                            || (tail[id] == tail[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }

            let Some(task) = best else {
                break;
            };

            let finish = now.checked_add(tasks[task].duration).ok_or(Error::Overflow {
                task,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task,
                stage: Stage::Turnaround,
            })?;

            started[task] = true;
            worker_busy[worker] = true;
            worker_available[worker] = available_at;
            runs[task] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task,
                worker,
                kind: EventKind::Start,
            });

            if finish == now {
                // Immediate completion occurs before selecting another task.
                completed[task] = true;
                completed_count += 1;
                events.push(Event {
                    time: now,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
                for &child in &children[task] {
                    remaining_deps[child] -= 1;
                }
                if worker_available[worker] <= now {
                    worker_busy[worker] = false;
                }
            } else {
                running.push((finish, task, worker));
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest future event that can change eligibility or
        // worker availability.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                let deps_ready = remaining_deps[id] == 0;
                if deps_ready {
                    let release = tasks[id].release;
                    next_time = Some(next_time.map_or(release, |t| t.min(release)));
                }
            }
        }
        for w in 0..workers {
            if worker_busy[w] && worker_available[w] > now {
                let available_at = worker_available[w];
                next_time = Some(next_time.map_or(available_at, |t| t.min(available_at)));
            }
        }

        if let Some(next) = next_time {
            now = next;
            for w in 0..workers {
                if worker_busy[w] && worker_available[w] <= now {
                    worker_busy[w] = false;
                }
            }
        } else {
            // Tasks may be waiting on dependencies whose running predecessor
            // was completed at a later time; that finish should have been an
            // event above. This guard keeps all indexing and arithmetic safe.
            break;
        }
    }

    let runs: Vec<Run> = runs
        .into_iter()
        .map(|run| run.expect("acyclic tasks are all schedulable"))
        .collect();
    let makespan = runs.iter().map(|run| run.finish).max().unwrap_or(0);

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
