```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn normalize_from(base: &str, path: &str) -> String {
        let mut parts: Vec<&str> = Vec::new();
        if !path.starts_with('/') && base != "/" {
            parts.extend(base[1..].split('/').filter(|p| !p.is_empty()));
        }
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other),
            }
        }
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        Ok(Self::normalize_from(&self.cwd, path))
    }

    fn parent(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_string(), String::new());
        }
        match path.rfind('/') {
            Some(0) => ("/".to_string(), path[1..].to_string()),
            Some(i) => (path[..i].to_string(), path[i + 1..].to_string()),
            None => ("/".to_string(), path.to_string()),
        }
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn prefixes(path: &str) -> Vec<String> {
        if path == "/" {
            return vec!["/".to_string()];
        }
        let mut result = vec!["/".to_string()];
        let mut acc = String::new();
        for component in path[1..].split('/') {
            acc.push('/');
            acc.push_str(component);
            result.push(acc.clone());
        }
        result
    }

    fn resolve(&self, path: &str) -> Result<String, Error> {
        let mut pending: Vec<String> = if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_string).collect()
        };
        let mut current = String::from("/");
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        while !pending.is_empty() {
            let component = pending.remove(0);
            let candidate = Self::join(&current, &component);
            let entry = self
                .entries
                .get(&candidate)
                .ok_or_else(|| Error::Missing(candidate.clone()))?;

            match entry {
                Entry::Dir => current = candidate,
                Entry::File(_) => {
                    if pending.is_empty() {
                        current = candidate;
                    } else {
                        return Err(Error::NotDir(candidate));
                    }
                }
                Entry::Link(target) => {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;

                    let physical_parent = Self::parent(&candidate).0;
                    let mut expanded =
                        Self::normalize_from(&physical_parent, target);
                    if !pending.is_empty() {
                        let suffix = pending.join("/");
                        expanded = Self::normalize_from(&expanded, &suffix);
                    }
                    pending = if expanded == "/" {
                        Vec::new()
                    } else {
                        expanded[1..].split('/').map(str::to_string).collect()
                    };
                    current = "/".to_string();
                }
            }
        }
        Ok(current)
    }

    fn resolve_parent(&self, canonical: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent(canonical);
        let physical_parent = self.resolve(&parent)?;
        match self.entries.get(&physical_parent) {
            Some(Entry::Dir) => Ok((physical_parent, name)),
            Some(_) => Err(Error::NotDir(physical_parent)),
            None => Err(Error::Missing(physical_parent)),
        }
    }

    fn contains_path(parent: &str, child: &str) -> bool {
        parent == child
            || (parent == "/" && child.starts_with('/'))
            || (child.starts_with(parent)
                && child.as_bytes().get(parent.len()) == Some(&b'/'))
    }

    fn descendants(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|p| *p != path && Self::contains_path(path, p))
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let destination = Self::join(&parent, &name);
        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }
        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = match self.resolve(&canonical) {
            Ok(p) => p,
            Err(Error::Missing(missing)) if missing == canonical => {
                let (parent, name) = self.resolve_parent(&canonical)?;
                Self::join(&parent, &name)
            }
            Err(e) => return Err(e),
        };

        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, _) = Self::parent(&resolved);
                match self.entries.get(&parent) {
                    Some(Entry::Dir) => {
                        self.entries
                            .insert(resolved, Entry::File(data.to_vec()));
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(parent)),
                    None => Err(Error::Missing(parent)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }

        let mut result = Vec::new();
        for (child, entry) in &self.entries {
            if child == "/" {
                continue;
            }
            let (parent, name) = Self::parent(child);
            if parent == resolved {
                result.push((name, entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(
        &mut self,
        from: &str,
        to: &str,
        overwrite: bool,
    ) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = Self::join(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir)
            && Self::contains_path(&source_path, &self.cwd)
        {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) =
            self.resolve_parent(&destination)?;
        let destination_path =
            Self::join(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && Self::contains_path(&destination_path, &self.cwd)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir)
            && Self::contains_path(&source_path, &destination_path)
        {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if overwrite {
            if let Some(dest) = &destination_entry {
                match (&source_entry, dest) {
                    (Entry::Dir, Entry::Dir) => {
                        if !self.descendants(&destination_path).is_empty() {
                            return Err(Error::NotEmpty(destination_path));
                        }
                    }
                    (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                    (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                    _ => {}
                }
            }
        }

        let mut moved = Vec::new();
        for (path, entry) in &self.entries {
            if path == &source_path || Self::contains_path(&source_path, path) {
                let suffix = &path[source_path.len()..];
                moved.push((format!("{}{}", destination_path, suffix), entry.clone()));
            }
        }

        if destination_entry.is_some() {
            let remove: Vec<String> = self
                .entries
                .keys()
                .filter(|p| {
                    *p == &destination_path
                        || Self::contains_path(&destination_path, p)
                })
                .cloned()
                .collect();
            for path in remove {
                self.entries.remove(&path);
            }
        }

        let remove: Vec<String> = self
            .entries
            .keys()
            .filter(|p| *p == &source_path || Self::contains_path(&source_path, p))
            .cloned()
            .collect();
        for path in remove {
            self.entries.remove(&path);
        }

        for (path, entry) in moved {
            self.entries.insert(path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = self.resolve_parent(&canonical)?;
        let resolved = Self::join(&parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && Self::contains_path(&resolved, &self.cwd) {
            return Err(Error::Busy(resolved));
        }

        let descendants = self.descendants(&resolved);
        if matches!(entry, Entry::Dir) && !recursive && !descendants.is_empty() {
            return Err(Error::NotEmpty(resolved));
        }

        for child in descendants {
            self.entries.remove(&child);
        }
        self.entries.remove(&resolved);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        if canonical_link == "/" {
            return Err(Error::Exists("/".to_string()));
        }

        let (parent, name) = self.resolve_parent(&canonical_link)?;
        let resolved = Self::join(&parent, &name);
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries
            .insert(resolved, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::InvalidTarget(0));
        }

        let (parent, name) = Self::parent(&canonical);
        let physical_parent = self.resolve(&parent)?;
        let final_path = Self::join(&physical_parent, &name);
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let mut paths = vec![resolved.clone()];
        paths.extend(
            self.entries
                .keys()
                .filter(|p| *p != &resolved && Self::contains_path(&resolved, p))
                .cloned(),
        );
        paths.sort();

        let mut sizes = BTreeMap::new();
        for p in &paths {
            let size = match self.entries.get(p).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(p.clone(), size);
        }

        let mut totals = sizes.clone();
        for p in paths.iter().rev() {
            let size = *sizes.get(p).unwrap();
            let parent = Self::parent(p).0;
            if p != &resolved && totals.contains_key(&parent) {
                let total = totals.get(&parent).copied().unwrap_or(0);
                totals.insert(parent, total.saturating_add(size));
            }
        }

        let mut result = Vec::new();
        for p in paths {
            let value = match self.entries.get(&p).unwrap() {
                Entry::Dir => totals.get(&p).copied().unwrap_or(0),
                _ => sizes.get(&p).copied().unwrap_or(0),
            };
            result.push((p, value));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }
}
```
