#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let other_half = denominator - r;
    if r > other_half || (r == other_half && q & 1 == 1) {
        Some(q + 1)
    } else {
        Some(q)
    }
}

#[derive(Clone)]
struct State {
    customer: Customer,
    deadline: Option<u64>,
    queue_since: u64,
    waited: u64,
    remaining: u64,
    ever_started: bool,
    paused: bool,
    pause_time: u64,
}

struct Running {
    index: usize,
    work_at: u64,
    finish: u64,
    interval_start: u64,
}

fn add_wait(state: &mut State, now: u64, wait_total: &mut u128) {
    let segment = now - state.queue_since;
    state.waited += segment;
    *wait_total += segment as u128;
}

fn better(a: usize, b: usize, states: &[State]) -> bool {
    let x = &states[a];
    let y = &states[b];
    if x.customer.priority != y.customer.priority {
        return x.customer.priority < y.customer.priority;
    }
    if x.paused != y.paused {
        return x.paused;
    }
    if x.paused {
        if x.pause_time != y.pause_time {
            return x.pause_time < y.pause_time;
        }
        x.customer.id < y.customer.id
    } else if x.customer.arrival != y.customer.arrival {
        x.customer.arrival < y.customer.arrival
    } else {
        x.customer.id > y.customer.id
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut seen = std::collections::BTreeSet::new();
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (i, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index: i });
        }
        if !seen.insert(c.id) {
            return Err(Error::DuplicateId { index: i, id: c.id });
        }
        if c.priority > 3 {
            return Err(Error::BadPriority { index: i, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow { index: i, id: c.id })?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a].arrival.cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<State> = customers.iter().enumerate().map(|(i, c)| State {
        customer: c.clone(),
        deadline: initial_deadlines[i],
        queue_since: c.arrival,
        waited: 0,
        remaining: c.service,
        ever_started: false,
        paused: false,
        pause_time: 0,
    }).collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut running: Option<Running> = None;
    let mut cursor = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;
    let mut completion_order = Vec::new();
    let mut now = if customers.is_empty() { 0 } else { customers[order[0]].arrival };
    let mut last_event_time = 0u64;

    let emit = |events: &mut Vec<Event>, t: u64, id: u32, kind: Kind, last: &mut u64| {
        events.push(Event { time: t, customer: id, kind });
        *last = t;
    };

    loop {
        if let Some(r) = running.as_ref() {
            if r.finish == now {
                let idx = r.index;
                busy_ticks += (now - r.interval_start) as u128;
                emit(&mut events, now, states[idx].customer.id, Kind::Finish, &mut last_event_time);
                completed += 1;
                turnaround_ticks += (now - states[idx].customer.arrival) as u128;
                completion_order.push(states[idx].customer.id);
                running = None;
            }
        }

        let mut expiring: Vec<usize> = waiting.iter().copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expiring.sort_by(|&a, &b| {
            states[a].deadline.cmp(&states[b].deadline)
                .then(states[a].customer.id.cmp(&states[b].customer.id))
        });
        for idx in expiring {
            if let Some(pos) = waiting.iter().position(|&x| x == idx) {
                waiting.remove(pos);
                add_wait(&mut states[idx], now, &mut wait_ticks);
                emit(&mut events, now, states[idx].customer.id, Kind::Renege { waited: states[idx].waited }, &mut last_event_time);
                reneged += 1;
            }
        }

        while cursor < order.len() && customers[order[cursor]].arrival == now {
            let idx = order[cursor];
            cursor += 1;
            let id = states[idx].customer.id;
            emit(&mut events, now, id, Kind::Arrive, &mut last_event_time);
            let limit = config.waiting_capacity + usize::from(running.is_none());
            if waiting.len() >= limit {
                emit(&mut events, now, id, Kind::Balk, &mut last_event_time);
                balked += 1;
            } else {
                accepted += 1;
                waiting.push(idx);
            }
        }

        if let Some(r) = running.as_ref() {
            if now >= r.work_at {
                if let Some(&best) = waiting.iter().min_by(|&&a, &&b| {
                    if better(a, b, &states) { std::cmp::Ordering::Less }
                    else if better(b, a, &states) { std::cmp::Ordering::Greater }
                    else { std::cmp::Ordering::Equal }
                }) {
                    if states[best].customer.priority < states[r.index].customer.priority {
                        let old = r.index;
                        let remaining = r.finish - now;
                        busy_ticks += (now - r.interval_start) as u128;
                        emit(&mut events, now, states[old].customer.id,
                             Kind::Preempt { remaining, by: states[best].customer.id },
                             &mut last_event_time);
                        preemptions += 1;
                        running = None;
                        states[old].remaining = remaining;
                        states[old].paused = true;
                        states[old].pause_time = now;
                        states[old].queue_since = now;
                        states[old].deadline = match states[old].customer.patience {
                            Some(budget) => Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                                customer: states[old].customer.id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })?),
                            None => None,
                        };
                        waiting.push(old);
                    }
                }
            }
        }

        loop {
            if running.is_some() {
                break;
            }
            let best_pos = waiting.iter().enumerate().min_by(|(_, &a), (_, &b)| {
                if better(a, b, &states) { std::cmp::Ordering::Less }
                else if better(b, a, &states) { std::cmp::Ordering::Greater }
                else { std::cmp::Ordering::Equal }
            }).map(|(p, _)| p);
            let Some(pos) = best_pos else { break };
            let idx = waiting.remove(pos);
            add_wait(&mut states[idx], now, &mut wait_ticks);
            let id = states[idx].customer.id;
            let cost = if last_dispatched.is_some_and(|x| x != id) { config.switch_cost } else { 0 };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::WorkAt
            })?;
            let finish = work_at.checked_add(states[idx].remaining).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::Finish
            })?;
            let first = !states[idx].ever_started;
            let kind = if first {
                started += 1;
                states[idx].ever_started = true;
                Kind::Start { remaining: states[idx].remaining, work_at, finish }
            } else {
                Kind::Resume { remaining: states[idx].remaining, work_at, finish }
            };
            emit(&mut events, now, id, kind, &mut last_event_time);
            last_dispatched = Some(id);
            running = Some(Running { index: idx, work_at, finish, interval_start: now });
            if finish == now {
                busy_ticks += 0;
                emit(&mut events, now, id, Kind::Finish, &mut last_event_time);
                completed += 1;
                turnaround_ticks += (now - states[idx].customer.arrival) as u128;
                completion_order.push(id);
                running = None;
                continue;
            }
            break;
        }

        let mut expiring: Vec<usize> = waiting.iter().copied()
            .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
            .collect();
        expiring.sort_by(|&a, &b| {
            states[a].deadline.cmp(&states[b].deadline)
                .then(states[a].customer.id.cmp(&states[b].customer.id))
        });
        for idx in expiring {
            if let Some(pos) = waiting.iter().position(|&x| x == idx) {
                waiting.remove(pos);
                add_wait(&mut states[idx], now, &mut wait_ticks);
                emit(&mut events, now, states[idx].customer.id, Kind::Renege { waited: states[idx].waited }, &mut last_event_time);
                reneged += 1;
            }
        }

        if cursor == order.len() && waiting.is_empty() && running.is_none() {
            break;
        }

        let mut next = None;
        if cursor < order.len() {
            next = Some(customers[order[cursor]].arrival);
        }
        if let Some(r) = running.as_ref() {
            for t in [r.work_at, r.finish] {
                if t > now && next.is_none_or(|n| t < n) {
                    next = Some(t);
                }
            }
        }
        for &idx in &waiting {
            if let Some(d) = states[idx].deadline {
                if d > now && next.is_none_or(|n| d < n) {
                    next = Some(d);
                }
            }
        }
        let Some(t) = next else { break };
        now = t;
    }

    let elapsed = last_event_time;
    let mean_wait_milli = if accepted == 0 { 0 } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 { 0 } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 { 0 } else {
        rounded_ratio(busy_ticks * 10000, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted, balked, reneged, completed, started, preemptions,
            elapsed, busy_ticks, wait_ticks, turnaround_ticks,
            mean_wait_milli, mean_turnaround_milli, utilization_bps,
            completion_order,
        },
    })
}
