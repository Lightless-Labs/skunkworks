#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };

    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    queued_at: u64,
    waited: u64,
    remaining: u64,
    ever_started: bool,
    pause_time: Option<u64>,
    accepted: bool,
    terminal: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch_at: u64,
    protected: bool,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn best_waiter(queue: &[usize], jobs: &[Job]) -> Option<usize> {
    queue.iter().copied().min_by(|&a, &b| {
        let ja = &jobs[a];
        let jb = &jobs[b];

        ja.customer.priority.cmp(&jb.customer.priority)
            .then_with(|| match (ja.pause_time, jb.pause_time) {
                (Some(pa), Some(pb)) => pa.cmp(&pb)
                    .then_with(|| ja.customer.id.cmp(&jb.customer.id)),
                (Some(_), None) => std::cmp::Ordering::Less,
                (None, Some(_)) => std::cmp::Ordering::Greater,
                (None, None) => ja.customer.arrival.cmp(&jb.customer.arrival)
                    .then_with(|| jb.customer.id.cmp(&ja.customer.id)),
            })
    })
}

fn expire_waiters(
    now: u64,
    queue: &mut Vec<usize>,
    jobs: &mut [Job],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<usize> = queue.iter().copied()
        .filter(|&i| jobs[i].deadline.map_or(false, |d| d <= now))
        .collect();

    expired.sort_by_key(|&i| (jobs[i].deadline.unwrap(), jobs[i].customer.id));

    for i in expired {
        if let Some(pos) = queue.iter().position(|&x| x == i) {
            queue.remove(pos);
        }
        let job = &mut jobs[i];
        let segment = now.saturating_sub(job.queued_at);
        job.waited = job.waited.saturating_add(segment);
        stats.wait_ticks += segment as u128;
        stats.reneged += 1;
        job.terminal = true;
        emit(events, now, job.customer.id, Kind::Renege { waited: job.waited });
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut deadlines = Vec::with_capacity(customers.len());

    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p)
                .ok_or(Error::DeadlineOverflow { index, id: c.id })?),
            None => None,
        };
        deadlines.push(deadline);
    }

    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a].arrival.cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers.iter().enumerate().map(|(i, c)| Job {
        customer: c.clone(),
        deadline: deadlines[i],
        queued_at: c.arrival,
        waited: 0,
        remaining: c.service,
        ever_started: false,
        pause_time: None,
        accepted: false,
        terminal: false,
    }).collect();

    let mut events = Vec::new();
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut queue: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut now = 0u64;
    let mut arrival_pos = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut first_dispatch = true;

    while stats.completed + stats.reneged + stats.balked < customers.len() {
        let next_arrival = order.get(arrival_pos).map(|&i| customers[i].arrival);
        let next_finish = active.as_ref().map(|a| a.finish);
        let next_work = active.as_ref().and_then(|a| {
            if a.protected && a.work_at > now { Some(a.work_at) } else { None }
        });
        let next_deadline = queue.iter().filter_map(|&i| jobs[i].deadline)
            .filter(|&d| d > now).min();

        let next_time = [next_arrival, next_finish, next_work, next_deadline]
            .into_iter().flatten().min();

        if let Some(t) = next_time {
            now = t;
        } else if active.is_none() && queue.is_empty() {
            break;
        }

        // Phase A: finish active work.
        if active.as_ref().map_or(false, |a| a.finish == now) {
            let a = active.take().unwrap();
            let id = jobs[a.index].customer.id;
            stats.busy_ticks += (now - a.dispatch_at) as u128;
            stats.completed += 1;
            stats.turnaround_ticks += (now - jobs[a.index].customer.arrival) as u128;
            stats.completion_order.push(id);
            jobs[a.index].terminal = true;
            emit(&mut events, now, id, Kind::Finish);
        }

        // Phase B: expire existing waiters.
        expire_waiters(now, &mut queue, &mut jobs, &mut events, &mut stats);

        // Phase C: process the entire arrival cohort.
        while arrival_pos < order.len() && customers[order[arrival_pos]].arrival == now {
            let i = order[arrival_pos];
            arrival_pos += 1;
            let id = jobs[i].customer.id;
            emit(&mut events, now, id, Kind::Arrive);

            let server_idle = active.is_none();
            let limit = config.waiting_capacity + usize::from(server_idle);
            if queue.len() >= limit {
                stats.balked += 1;
                jobs[i].terminal = true;
                emit(&mut events, now, id, Kind::Balk);
            } else {
                stats.accepted += 1;
                jobs[i].accepted = true;
                jobs[i].queued_at = now;
                jobs[i].deadline = deadlines[i];
                queue.push(i);
            }
        }

        // Phase D: consider preemption after protection.
        if let Some(a) = active.as_ref() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&queue, &jobs) {
                    if jobs[w].customer.priority < jobs[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let id = jobs[a.index].customer.id;
                        let remaining = a.finish - now;
                        emit(&mut events, now, id, Kind::Preempt {
                            remaining,
                            by: jobs[w].customer.id,
                        });
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch_at) as u128;

                        let job = &mut jobs[a.index];
                        job.remaining = remaining;
                        job.queued_at = now;
                        job.pause_time = Some(now);
                        if let Some(budget) = job.customer.patience {
                            job.deadline = Some(now.checked_add(budget).ok_or(
                                Error::TimeOverflow {
                                    customer: id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                }
                            )?);
                        } else {
                            job.deadline = None;
                        }
                        queue.push(a.index);
                    }
                }
            }
        }

        // Phase E: dispatch until an active interval remains or no work waits.
        while active.is_none() {
            let selected = match best_waiter(&queue, &jobs) {
                Some(i) => i,
                None => break,
            };
            queue.retain(|&i| i != selected);

            let id = jobs[selected].customer.id;
            let segment = now - jobs[selected].queued_at;
            jobs[selected].waited = jobs[selected].waited.saturating_add(segment);
            stats.wait_ticks += segment as u128;

            let switch = if first_dispatch || last_dispatched == Some(id) {
                0
            } else {
                config.switch_cost
            };
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(jobs[selected].remaining).ok_or(
                Error::TimeOverflow {
                    customer: id,
                    time: now,
                    stage: OverflowStage::Finish,
                }
            )?;

            let remaining = jobs[selected].remaining;
            let was_started = jobs[selected].ever_started;
            if was_started {
                emit(&mut events, now, id, Kind::Resume { remaining, work_at, finish });
            } else {
                emit(&mut events, now, id, Kind::Start { remaining, work_at, finish });
                jobs[selected].ever_started = true;
                stats.started += 1;
            }
            first_dispatch = false;
            last_dispatched = Some(id);

            active = Some(Active {
                index: selected,
                work_at,
                finish,
                dispatch_at: now,
                protected: work_at > now,
            });

            if finish == now {
                active = None;
                stats.busy_ticks += 0;
                stats.completed += 1;
                stats.turnaround_ticks += (now - jobs[selected].customer.arrival) as u128;
                stats.completion_order.push(id);
                jobs[selected].terminal = true;
                emit(&mut events, now, id, Kind::Finish);
                continue;
            }
        }

        // Phase F: expire newly queued waiters.
        expire_waiters(now, &mut queue, &mut jobs, &mut events, &mut stats);

        if let Some(last) = events.last() {
            stats.elapsed = last.time;
        }
    }

    if let Some(last) = events.last() {
        stats.elapsed = last.time;
    }
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio((stats.wait_ticks as u128).saturating_mul(1000), stats.accepted as u128).unwrap_or(0)
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks.saturating_mul(1000), stats.completed as u128).unwrap_or(0)
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks.saturating_mul(10000), stats.elapsed as u128)
            .unwrap_or(0).min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
