#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: std::collections::BTreeMap<String, Entry>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = std::collections::BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self { entries }
    }

    fn validate(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.bytes().position(|b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut start = 1;
        for (i, b) in path.bytes().enumerate().skip(1) {
            if b == b'/' {
                if i == start {
                    return Err(Error::InvalidPath(i));
                }
                let component = &path[start..i];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component);
                start = i + 1;
            }
        }

        if start == path.len() {
            return Err(Error::InvalidPath(path.len() - 1));
        }

        let component = &path[start..];
        if component == "." || component == ".." {
            return Err(Error::InvalidPath(start));
        }
        components.push(component);
        Ok(components)
    }

    fn prefix(components: &[&str], count: usize) -> String {
        if count == 0 {
            "/".to_string()
        } else {
            format!("/{}", components[..count].join("/"))
        }
    }

    fn resolve_parent(&self, components: &[&str]) -> Result<String, Error> {
        let mut current = "/".to_string();
        for (i, component) in components.iter().take(components.len().saturating_sub(1)).enumerate() {
            match self.entries.get(&current) {
                None => return Err(Error::Missing(current)),
                Some(Entry::File(_)) => return Err(Error::NotDir(current)),
                Some(Entry::Dir) => {}
            }
            current = Self::prefix(components, i + 1);
            match self.entries.get(&current) {
                None => return Err(Error::Missing(current)),
                Some(Entry::File(_)) if i + 1 < components.len() - 1 => {
                    return Err(Error::NotDir(current));
                }
                Some(_) => {}
            }
        }
        match self.entries.get(&current) {
            None => Err(Error::Missing(current)),
            Some(Entry::File(_)) => Err(Error::NotDir(current)),
            Some(Entry::Dir) => Ok(current),
        }
    }

    fn canonical(components: &[&str]) -> String {
        Self::prefix(components, components.len())
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }
        let parent = self.resolve_parent(&components)?;
        let target = Self::canonical(&components);
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        let _ = parent;
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }
        self.resolve_parent(&components)?;
        let target = Self::canonical(&components);
        match self.entries.get_mut(&target) {
            Some(Entry::Dir) => Err(Error::IsDir(target)),
            Some(Entry::File(bytes)) => {
                bytes.clear();
                bytes.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries.insert(target, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }
        self.resolve_parent(&components)?;
        let target = Self::canonical(&components);
        match self.entries.get(&target) {
            None => Err(Error::Missing(target)),
            Some(Entry::Dir) => Err(Error::IsDir(target)),
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate(path)?;
        let target = Self::canonical(&components);
        if !components.is_empty() {
            self.resolve_parent(&components)?;
        }
        match self.entries.get(&target) {
            None => Err(Error::Missing(target)),
            Some(Entry::File(_)) => Err(Error::NotDir(target)),
            Some(Entry::Dir) => {}
        }

        let prefix = if target == "/" {
            "/".to_string()
        } else {
            format!("{}/", target)
        };
        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if let Some(rest) = entry_path.strip_prefix(&prefix) {
                if !rest.is_empty() && !rest.contains('/') {
                    let kind = match entry {
                        Entry::Dir => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    children.push((rest.to_string(), kind));
                }
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
