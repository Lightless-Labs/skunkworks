#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    let separator = match separators.as_slice() {
        [] => return Err(EmailError::MissingAt),
        [separator] => *separator,
        [_, second, ..] => return Err(EmailError::MultipleAt { at: *second }),
    };

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];
    if local_bytes.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (index, &byte) in local_bytes.iter().enumerate() {
        let absolute = index;
        if byte == b'.' {
            if index == 0
                || index + 1 == local_bytes.len()
                || local_bytes[index - 1] == b'.'
            {
                return Err(EmailError::LocalDot { at: absolute });
            }
        } else if !(byte.is_ascii_alphanumeric()
            || b"!#$%&'*+-/=?^_`{|}~".contains(&byte))
        {
            return Err(EmailError::LocalChar { at: absolute });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain = if has_root_dot {
        &raw_domain[..raw_domain.len() - 1]
    } else {
        raw_domain
    };

    if domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let root_dot_position = input.len() - 1;
    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut label_start = 0;

    for (index, &byte) in domain.iter().enumerate() {
        if byte == b'.' {
            if index == label_start {
                return Err(EmailError::EmptyLabel {
                    at: domain_start + index,
                });
            }
            labels.push((label_start, &domain[label_start..index]));
            label_start = index + 1;
        }
    }

    if label_start == domain.len() {
        return Err(EmailError::EmptyLabel {
            at: if has_root_dot {
                root_dot_position
            } else {
                input.len()
            },
        });
    }
    labels.push((label_start, &domain[label_start..]));

    for &(offset, label) in &labels {
        let start = domain_start + offset;

        if label.len() > 63 {
            return Err(EmailError::LabelTooLong { at: start + 63 });
        }

        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen { at: start });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: start + label.len() - 1,
            });
        }

        for (index, &byte) in label.iter().enumerate() {
            if !(byte.is_ascii_alphanumeric() || byte == b'-') {
                return Err(EmailError::DomainChar { at: start + index });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_offset, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_offset,
        });
    }

    let local = String::from_utf8(local_bytes.to_vec()).expect("input was checked as ASCII");
    let lowered_domain = domain
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect::<Vec<_>>();
    let domain =
        String::from_utf8(lowered_domain).expect("input was checked as ASCII");

    Ok(Address {
        local,
        domain,
        quoted: false,
    })
}
