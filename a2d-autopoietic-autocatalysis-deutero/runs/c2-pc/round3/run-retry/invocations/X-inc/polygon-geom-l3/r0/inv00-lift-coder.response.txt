#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;

fn coordinate_valid(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, &point) in points.iter().enumerate() {
        if !coordinate_valid(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = if points.len() >= 2 && points[0] == points[points.len() - 1] {
        points.len() - 1
    } else {
        points.len()
    };

    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    Ok(&points[..effective_len])
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (q.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (q.x as i128 - a.x as i128)
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_sqrt(value: u128) -> u64 {
    let mut low = 0u128;
    let mut high = 1u128 << 64;

    while low + 1 < high {
        let mid = (low + high) / 2;
        if mid <= value / mid {
            low = mid;
        } else {
            high = mid;
        }
    }

    let lower_square = low * low;
    let upper = low + 1;
    let upper_square = upper * upper;

    let rounded = if value - lower_square >= upper_square - value {
        upper
    } else {
        low
    };

    rounded as u64
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    rounded_sqrt(squared_distance(a, b) * 1_000_000)
}

fn signed_area2(points: &[Point]) -> i128 {
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn polygon_perimeter_milli(points: &[Point]) -> u64 {
    if points.is_empty() {
        return 0;
    }
    (0..points.len())
        .map(|i| edge_length_milli(points[i], points[(i + 1) % points.len()]))
        .sum()
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let polygon = prepare_polygon(points)?;
    let signed_area2 = signed_area2(polygon);

    let mut edge_milli = Vec::with_capacity(polygon.len());
    let mut perimeter_milli = 0u64;
    let mut longest_edge: Option<usize> = None;

    for i in 0..polygon.len() {
        let a = polygon[i];
        let b = polygon[(i + 1) % polygon.len()];
        let length = edge_length_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let should_replace = match longest_edge {
            None => true,
            Some(best) => {
                if length > edge_milli[best] {
                    true
                } else if length < edge_milli[best] {
                    false
                } else {
                    let pair = if a <= b { (a, b) } else { (b, a) };
                    let best_a = polygon[best];
                    let best_b = polygon[(best + 1) % polygon.len()];
                    let best_pair = if best_a <= best_b {
                        (best_a, best_b)
                    } else {
                        (best_b, best_a)
                    };
                    pair < best_pair || (pair == best_pair && i < best)
                }
            }
        };
        if should_replace {
            longest_edge = Some(i);
        }
    }

    let orientation = if signed_area2 > 0 {
        Orientation::CounterClockwise
    } else if signed_area2 < 0 {
        Orientation::Clockwise
    } else {
        Orientation::Degenerate
    };

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;

    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !coordinate_valid(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &query in queries {
        let mut boundary_edge = None;
        let mut inside = false;

        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];

            if cross(a, b, query) == 0
                && query.x >= a.x.min(b.x)
                && query.x <= a.x.max(b.x)
                && query.y >= a.y.min(b.y)
                && query.y <= a.y.max(b.y)
            {
                boundary_edge = Some(i);
                break;
            }

            if (a.y > query.y) != (b.y > query.y) {
                let cross_value = cross(a, b, query);
                if (b.y > a.y && cross_value > 0) || (b.y < a.y && cross_value < 0) {
                    inside = !inside;
                }
            }
        }

        results.push(match boundary_edge {
            Some(edge) => Location::Boundary { edge },
            None if inside => Location::Inside,
            None => Location::Outside,
        });
    }

    Ok(results)
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

fn turn(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !coordinate_valid(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut ordered: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if let Some(existing) = ordered.iter_mut().find(|entry| entry.0 == point) {
            existing.1 = existing.1.min(index);
        } else {
            ordered.push((point, index));
        }
    }
    ordered.sort_by_key(|entry| entry.0);

    if ordered.is_empty() {
        return Ok(Hull {
            vertices: Vec::new(),
            source_indices: Vec::new(),
            area2: 0,
            perimeter_milli: 0,
        });
    }

    if ordered.len() == 1 {
        return Ok(Hull {
            vertices: vec![ordered[0].0],
            source_indices: vec![ordered[0].1],
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = ordered
        .iter()
        .skip(2)
        .all(|entry| turn(ordered[0].0, ordered[1].0, entry.0) == 0);

    let hull_points: Vec<Point> = if all_collinear {
        match mode {
            Collinear::Include => ordered.iter().map(|entry| entry.0).collect(),
            Collinear::Exclude => vec![ordered[0].0, ordered[ordered.len() - 1].0],
        }
    } else {
        let mut lower = Vec::new();
        for &entry in &ordered {
            while lower.len() >= 2 {
                let t = turn(lower[lower.len() - 2].0, lower[lower.len() - 1].0, entry.0);
                let remove = match mode {
                    Collinear::Exclude => t <= 0,
                    Collinear::Include => t < 0,
                };
                if remove {
                    lower.pop();
                } else {
                    break;
                }
            }
            lower.push(entry);
        }

        let mut upper = Vec::new();
        for &entry in ordered.iter().rev() {
            while upper.len() >= 2 {
                let t = turn(upper[upper.len() - 2].0, upper[upper.len() - 1].0, entry.0);
                let remove = match mode {
                    Collinear::Exclude => t <= 0,
                    Collinear::Include => t < 0,
                };
                if remove {
                    upper.pop();
                } else {
                    break;
                }
            }
            upper.push(entry);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower.iter().map(|entry| entry.0).collect()
    };

    let source_indices: Vec<usize> = hull_points
        .iter()
        .map(|point| ordered.iter().find(|entry| entry.0 == *point).unwrap().1)
        .collect();

    let signed = if hull_points.len() >= 3 {
        signed_area2(&hull_points)
    } else {
        0
    };

    Ok(Hull {
        area2: signed.unsigned_abs(),
        perimeter_milli: polygon_perimeter_milli(&hull_points),
        vertices: hull_points,
        source_indices,
    })
}
