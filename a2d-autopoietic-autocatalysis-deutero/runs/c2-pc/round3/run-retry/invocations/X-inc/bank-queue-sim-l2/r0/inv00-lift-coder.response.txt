#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|s| s.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index].iter().any(|c| c.id == customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();

    let mut pending_customers = customers.to_vec();
    pending_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0;
    let mut events = Vec::new();

    loop {
        let next_finish = states.iter().filter_map(|s| s.busy_until).min();
        let next_open = states
            .iter()
            .filter(|s| !s.opened)
            .map(|s| s.server.opens)
            .min();
        let next_arrival = pending_customers
            .get(next_customer)
            .map(|c| c.arrival);

        let time = match [next_finish, next_open, next_arrival]
            .into_iter()
            .flatten()
            .min()
        {
            Some(time) => time,
            None => break,
        };

        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (s.busy_until == Some(time)).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| states[i].server.id);

        for i in finishing {
            let customer = match events.iter().rev().find_map(|event| match event {
                Event::Start {
                    customer,
                    server,
                    finish,
                    ..
                } if *server == states[i].server.id && *finish == time => Some(*customer),
                _ => None,
            }) {
                Some(id) => id,
                None => unreachable!(),
            };
            events.push(Event::Finish {
                time,
                customer,
                server: states[i].server.id,
            });
            states[i].busy_until = None;
            states[i].idle_since = time;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.server.opens == time).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].server.id));

        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time,
                server: states[i].server.id,
            });
        }

        while next_customer < pending_customers.len()
            && pending_customers[next_customer].arrival == time
        {
            let customer_index = next_customer;
            events.push(Event::Arrive {
                time,
                customer: pending_customers[customer_index].id,
            });
            waiting.push(customer_index);
            next_customer += 1;
        }

        while !waiting.is_empty() {
            let selected_server = states
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && s.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected_server else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &pending_customers[customer_index];
            let server_id = states[server_index].server.id;

            let finish = time
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(states[server_index].server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = time;
                states[server_index].busy_until = None;
            } else {
                states[server_index].busy_until = Some(finish);
            }
        }
    }

    Ok(events)
}
