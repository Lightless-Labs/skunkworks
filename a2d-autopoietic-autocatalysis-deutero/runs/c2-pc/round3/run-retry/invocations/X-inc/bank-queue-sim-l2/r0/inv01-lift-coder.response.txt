#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    occupied: Option<(u64, u32)>, // (finish time, customer ID)
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|previous| previous.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            occupied: None,
        })
        .collect();

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut arrival_index = 0;
    let mut waiting: Vec<Customer> = Vec::new();
    let mut events = Vec::new();

    loop {
        let next_finish = states
            .iter()
            .filter_map(|state| state.occupied.map(|(time, _)| time))
            .min();
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.server.opens)
            .min();
        let next_arrival = arrivals.get(arrival_index).map(|customer| customer.arrival);

        let Some(time) = [next_finish, next_open, next_arrival]
            .into_iter()
            .flatten()
            .min()
        else {
            break;
        };

        // Finish events are ordered by ascending server ID.
        let mut finishing: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                state
                    .occupied
                    .filter(|(finish, _)| *finish == time)
                    .map(|_| index)
            })
            .collect();
        finishing.sort_by_key(|&index| states[index].server.id);
        for index in finishing {
            let (finish, customer) = states[index].occupied.take().unwrap();
            events.push(Event::Finish {
                time: finish,
                customer,
                server: states[index].server.id,
            });
            states[index].idle_since = time;
        }

        // Open events are ordered by descending server ID.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.server.opens == time).then_some(index)
            })
            .collect();
        opening.sort_by_key(|&index| std::cmp::Reverse(states[index].server.id));
        for index in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time,
                server: states[index].server.id,
            });
        }

        // Arrivals at this tick are already ordered by descending customer ID.
        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == time {
            let customer = arrivals[arrival_index].clone();
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(customer);
            arrival_index += 1;
        }

        loop {
            if waiting.is_empty() {
                break;
            }

            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.occupied.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };
            let customer = waiting.remove(0);
            let server_id = states[server_index].server.id;
            let overhead = states[server_index].server.overhead;

            let finish = customer
                .service
                .checked_add(time)
                .and_then(|value| value.checked_add(overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = time;
            } else {
                states[server_index].occupied = Some((finish, customer.id));
            }
        }
    }

    Ok(events)
}
