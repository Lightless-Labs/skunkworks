#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|previous| previous.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<Customer> = customers.to_vec();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            idle_since: server.opens,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut customer_index = 0usize;
    let mut waiting: Vec<usize> = Vec::new();
    let mut time = 0u64;

    loop {
        let mut finished: Vec<(usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (state.busy_until == Some(time)).then_some((index, state.id))
            })
            .collect();
        finished.sort_by_key(|&(_, id)| id);
        for (index, _) in finished {
            // The occupied customer is encoded by its matching scheduled event below.
            let customer = events.iter().rev().find_map(|event| match event {
                Event::Start {
                    time: start_time,
                    customer,
                    server,
                    finish,
                } if *server == states[index].id && *finish == time && *start_time <= time => {
                    let already_finished = events.iter().any(|candidate| {
                        matches!(
                            candidate,
                            Event::Finish {
                                time: finish_time,
                                customer: finished_customer,
                                server: finished_server
                            } if *finish_time == time
                                && *finished_customer == *customer
                                && *finished_server == *server
                        )
                    });
                    (!already_finished).then_some(*customer)
                }
                _ => None,
            });
            if let Some(customer) = customer {
                events.push(Event::Finish {
                    time,
                    customer,
                    server: states[index].id,
                });
            }
            states[index].busy_until = None;
            states[index].idle_since = time;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.opens == time).then_some(index)
            })
            .collect();
        opening.sort_by_key(|&index| std::cmp::Reverse(states[index].id));
        for index in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time,
                server: states[index].id,
            });
        }

        let arrival_start = customer_index;
        while customer_index < ordered_customers.len()
            && ordered_customers[customer_index].arrival == time
        {
            customer_index += 1;
        }
        for customer in &ordered_customers[arrival_start..customer_index] {
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(
                customers
                    .iter()
                    .position(|original| original.id == customer.id)
                    .expect("validated customer"),
            );
        }

        loop {
            if waiting.is_empty() {
                break;
            }
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);
            let Some(server_index) = selected else {
                break;
            };

            let customer_index_in_input = waiting.remove(0);
            let customer = &customers[customer_index_in_input];
            let server = &states[server_index];
            let finish = customer
                .service
                .checked_add(time)
                .and_then(|partial| partial.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server.id,
                finish,
            });
            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = time;
            } else {
                states[server_index].busy_until = Some(finish);
            }
        }

        let all_customers_done = customer_index == ordered_customers.len() && waiting.is_empty();
        let all_servers_opened = states.iter().all(|state| state.opened);
        let all_servers_idle = states.iter().all(|state| state.busy_until.is_none());

        if all_customers_done && all_servers_opened && all_servers_idle {
            break;
        }

        let next_arrival = ordered_customers
            .get(customer_index)
            .map(|customer| customer.arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = states.iter().filter_map(|state| state.busy_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&candidate| candidate > time)
            .min();

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    Ok(events)
}
