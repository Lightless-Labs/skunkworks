#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_table(&self.input, self.config);
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config)?;
    parse_table(input, config)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_records(input, b',', b'"')
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_records(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut i = start;
    let mut record = 0usize;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;
    let mut record_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !in_quotes && b == b'\r' && (i + 1 >= bytes.len() || bytes[i + 1] != b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if in_quotes {
            if b == quote {
                if i + 1 < bytes.len() && bytes[i + 1] == quote {
                    field.push(quote);
                    i += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                field.push(b);
                i += 1;
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(String::from_utf8(field).expect("input fields are valid UTF-8"));
                field = Vec::new();
                after_quote = false;
                record_started = true;
                i += 1;
            } else if b == b'\n' || b == b'\r' {
                row.push(String::from_utf8(field).expect("input fields are valid UTF-8"));
                field = Vec::new();
                records.push(row);
                row = Vec::new();
                after_quote = false;
                record_started = false;
                record += 1;
                i += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte { record, byte: i });
            }
            continue;
        }

        if b == quote {
            if field.is_empty() {
                in_quotes = true;
                quote_start = i;
                record_started = true;
                i += 1;
            } else {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
        } else if b == delimiter {
            row.push(String::from_utf8(field).expect("input fields are valid UTF-8"));
            field = Vec::new();
            record_started = true;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            row.push(String::from_utf8(field).expect("input fields are valid UTF-8"));
            field = Vec::new();
            records.push(row);
            row = Vec::new();
            record_started = false;
            record += 1;
            i += if b == b'\r' { 2 } else { 1 };
        } else {
            field.push(b);
            record_started = true;
            i += 1;
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_start,
        });
    }

    if record_started || after_quote || !row.is_empty() || !field.is_empty() {
        row.push(String::from_utf8(field).expect("input fields are valid UTF-8"));
        records.push(row);
    }

    Ok(records)
}

fn parse_table(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut records = parse_records(input, config.delimiter, config.quote)?;
    let header = if config.headers && !records.is_empty() {
        Some(records.remove(0))
    } else {
        None
    };

    if let Some(names) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..names.len() {
                if let Some(first) = names[..second].iter().position(|name| name == &names[second]) {
                    return Err(CsvError::DuplicateHeader {
                        name: names[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }

        for (index, row) in records.iter().enumerate() {
            if row.len() != names.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: names.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows: records,
        duplicates: config.duplicates,
    })
}

    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut stopped = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if stopped {
                output.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let (value, error) = match resolved {
                None => (
                    None,
                    Some(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    }),
                ),
                Some(column_index) => match row.get(column_index) {
                    None => (
                        None,
                        Some(CellError {
                            row: row_index,
                            spec: spec_index,
                            column: Some(column_index),
                            kind: CellErrorKind::MissingCell,
                        }),
                    ),
                    Some(cell) => evaluate_cell(cell, spec, row_index, spec_index, column_index),
                },
            };

            output.push(value);
            if let Some(error) = error {
                result.errors.push(error);
                if policy != ErrorPolicy::Collect {
                    stopped = true;
                }
                if policy == ErrorPolicy::Stop {
                    result.rows.push(output);
                    return result;
                }
            }
        }

        result.rows.push(output);
    }

    result
}

fn evaluate_cell(
    cell: &str,
    spec: &ColumnSpec,
    row: usize,
    spec_index: usize,
    column: usize,
) -> (Option<Value>, Option<CellError>) {
    let fail = |kind| {
        (
            None,
            Some(CellError {
                row,
                spec: spec_index,
                column: Some(column),
                kind,
            }),
        )
    };

    if cell.is_empty() {
        if spec.nullable {
            return (Some(Value::Null), None);
        }
        if spec.kind != ColumnType::Text {
            return fail(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => (Some(Value::Text(cell.to_string())), None),
        ColumnType::Integer => match parse_integer(cell) {
            NumericParse::Invalid => fail(CellErrorKind::InvalidInteger),
            NumericParse::Overflow => fail(CellErrorKind::IntegerOverflow),
            NumericParse::Value(value) => (Some(Value::Integer(value)), None),
        },
        ColumnType::Decimal => match parse_decimal(cell) {
            NumericParse::Invalid => fail(CellErrorKind::InvalidDecimal),
            NumericParse::Overflow => fail(CellErrorKind::DecimalOverflow),
            NumericParse::Value(value) => (Some(Value::Decimal(value)), None),
        },
        ColumnType::Boolean => match cell {
            "yes" | "1" => (Some(Value::Boolean(true)), None),
            "no" | "0" => (Some(Value::Boolean(false)), None),
            _ => fail(CellErrorKind::InvalidBoolean),
        },
    }
}

enum NumericParse {
    Invalid,
    Overflow,
    Value(i64),
}

fn parse_integer(s: &str) -> NumericParse {
    let bytes = s.as_bytes();
    let (negative, digits) = if bytes.first() == Some(&b'-') {
        (true, &bytes[1..])
    } else {
        (false, bytes)
    };

    if digits.is_empty()
        || (digits.len() > 1 && digits[0] == b'0')
        || !digits.iter().all(u8::is_ascii_digit)
        || (negative && digits == b"0")
    {
        return NumericParse::Invalid;
    }

    let mut value: i128 = 0;
    for &digit in digits {
        value = value * 10 + i128::from(digit - b'0');
    }
    if negative {
        value = -value;
    }

    if value < i64::MIN as i128 || value > i64::MAX as i128 {
        NumericParse::Overflow
    } else {
        NumericParse::Value(value as i64)
    }
}

fn parse_decimal(s: &str) -> NumericParse {
    let bytes = s.as_bytes();
    let (negative, body) = if bytes.first() == Some(&b'-') {
        (true, &bytes[1..])
    } else {
        (false, bytes)
    };

    let Some(dot) = body.iter().position(|&b| b == b'.') else {
        return NumericParse::Invalid;
    };
    let integral = &body[..dot];
    let fractional = &body[dot + 1..];

    if integral.is_empty()
        || fractional.len() != 2
        || !fractional.iter().all(u8::is_ascii_digit)
        || !integral.iter().all(u8::is_ascii_digit)
        || (integral.len() > 1 && integral[0] == b'0')
        || (negative && integral == b"0" && fractional == b"00")
    {
        return NumericParse::Invalid;
    }

    let mut magnitude: i128 = 0;
    for &digit in integral.iter().chain(fractional.iter()) {
        magnitude = magnitude * 10 + i128::from(digit - b'0');
    }
    if negative {
        magnitude = -magnitude;
    }

    if magnitude < i64::MIN as i128 || magnitude > i64::MAX as i128 {
        NumericParse::Overflow
    } else {
        NumericParse::Value(magnitude as i64)
    }
}
