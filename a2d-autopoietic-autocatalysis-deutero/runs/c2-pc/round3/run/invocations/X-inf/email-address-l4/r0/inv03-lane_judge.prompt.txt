pub struct DedupResult {

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn at_position(positions: &[usize], index: usize, input_len: usize) -> usize {
    positions.get(index).copied().unwrap_or(input_len)
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if i + 2 >= input.len() || input[i] != b'\r' || input[i + 1] != b'\n' {
        return None;
    }
    let mut end = i + 2;
    if !is_hws(input[end]) {
        return None;
    }
    while end < input.len() && is_hws(input[end]) {
        end += 1;
    }
    Some(end)
}

fn lexical_error(input: &[u8], i: usize, err: impl FnOnce(usize) -> EmailError) -> EmailError {
    err(i.min(input.len()))
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();

    let mut i = 0usize;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut comment_stack: Vec<usize> = Vec::new();
    let mut quoted_prefix = false;
    let mut quote_closed = false;
    let mut quote_escape = false;
    let mut run_start: Option<usize> = None;

    while i < input.len() {
        let b = input[i];

        if quoted_prefix && !quote_closed {
            bytes.push(b);
            positions.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            if b == b'\\' {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                if i + 1 >= input.len()
                    || !(0x20..=0x7e).contains(&input[i + 1])
                {
                    return Err(lexical_error(input, i, |at| EmailError::CommentEscape { at }));
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(lexical_error(input, i, |at| EmailError::BadFold { at }));
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                comment_stack.push(i);
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                comment_stack.pop();
                i += 1;
                if depth == 0 {
                    if let Some(start) = run_start.take() {
                        runs.push(Run { start, end: i });
                    }
                }
                continue;
            }
            if !(0x20..=0x7e).contains(&b) && !is_hws(b) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                i = end;
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }

        if b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            depth = 1;
            outer_open = i;
            comment_stack.push(i);
            i += 1;
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if is_hws(b) {
            if run_start.is_none() {
                run_start = Some(i);
            }
            i += 1;
            continue;
        }

        if let Some(start) = run_start.take() {
            runs.push(Run { start, end: i });
        }

        if bytes.is_empty() && b == b'"' {
            quoted_prefix = true;
            quote_closed = false;
            quote_escape = false;
        }
        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = run_start.take() {
        runs.push(Run {
            start,
            end: input.len(),
        });
    }

    // A quote may have started while scanning, but cleanup deliberately does
    // not report an unclosed quote; that is the address parser's responsibility.
    let mut cleaned = Cleaned {
        bytes,
        positions,
        runs,
    };

    for run in &cleaned.runs {
        let left = cleaned
            .positions
            .iter()
            .position(|&p| p >= run.end)
            .and_then(|idx| cleaned.bytes.get(idx))
            .copied();
        let right = cleaned
            .positions
            .iter()
            .rposition(|&p| p < run.start)
            .and_then(|idx| cleaned.bytes.get(idx))
            .copied();

        // Runs at either edge are legal. Internal runs must touch a retained @.
        let before = cleaned.positions.iter().any(|&p| p < run.start);
        let after = cleaned.positions.iter().any(|&p| p >= run.end);
        if before && after && left != Some(b'@') && right != Some(b'@') {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    // Keep the local mutable for clarity if this structure is extended.
    let _ = &mut cleaned;
    Ok(cleaned)
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if is_atom_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_core(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let cleaned = cleanup(raw)?;
    let bytes = &cleaned.bytes;
    let pos = &cleaned.positions;
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_close = None;
    if bytes[0] == b'"' {
        let mut i = 1usize;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i = i.saturating_add(2);
            } else if bytes[i] == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: at_position(pos, 0, raw.len()),
            });
        }
    }

    let protected_end = quote_close.map(|i| i + 1).unwrap_or(0);
    let ats: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@' && i >= protected_end).then_some(i))
        .collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: at_position(pos, ats[1], raw.len()),
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: at_position(pos, sep, raw.len()),
        });
    }

    let local_bytes = &bytes[..sep];
    let mut local = Vec::new();
    let mut local_origins = Vec::new();
    let quoted;

    if local_bytes[0] == b'"' {
        let close = quote_close.unwrap_or(local_bytes.len());
        if close + 1 < local_bytes.len() {
            return Err(EmailError::TrailingLocal {
                at: at_position(pos, close + 1, raw.len()),
            });
        }
        quoted = true;
        let mut i = 1usize;
        while i < close {
            let b = local_bytes[i];
            if b == b'\\' {
                if i + 1 >= close
                    || !matches!(local_bytes[i + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape {
                        at: at_position(pos, i, raw.len()),
                    });
                }
                local.push(local_bytes[i + 1]);
                local_origins.push(pos[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: at_position(pos, i, raw.len()),
                    });
                }
                local.push(b);
                local_origins.push(pos[i]);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: at_position(pos, close, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: at_position(pos, 64, raw.len()),
            });
        }
        for i in 0..local_bytes.len() {
            let b = local_bytes[i];
            if b == b'.' {
                if i == 0
                    || i + 1 == local_bytes.len()
                    || local_bytes[i - 1] == b'.'
                {
                    return Err(EmailError::LocalDot {
                        at: at_position(pos, i, raw.len()),
                    });
                }
                local.push(b);
                local_origins.push(pos[i]);
            } else if is_atom_char(b) {
                local.push(b);
                local_origins.push(pos[i]);
            } else {
                return Err(EmailError::LocalChar {
                    at: at_position(pos, i, raw.len()),
                });
            }
        }
    }

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let mut root_pos = None;
    if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_pos = Some(pos[domain_end]);
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: at_position(pos, domain_start, raw.len()),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: at_position(pos, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || bytes[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    for (label_start, label_end) in &labels {
        if label_start == label_end {
            let at = if *label_start == domain_start {
                at_position(pos, *label_start, raw.len())
            } else if *label_start == domain_end && root_pos.is_some() {
                root_pos.unwrap()
            } else {
                at_position(pos, label_start.saturating_sub(1), raw.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if label_end - label_start > 63 {
            return Err(EmailError::LabelTooLong {
                at: at_position(pos, label_start + 63, raw.len()),
            });
        }
        if bytes[*label_start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: at_position(pos, *label_start, raw.len()),
            });
        }
        if bytes[*label_end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: at_position(pos, label_end - 1, raw.len()),
            });
        }
        for i in *label_start..*label_end {
            let b = bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: at_position(pos, i, raw.len()),
                });
            }
        }
    }
    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: at_position(pos, domain_start, raw.len()),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: at_position(pos, last_start, raw.len()),
        });
    }

    let local_string = String::from_utf8(local).expect("ASCII local");
    let domain_string = String::from_utf8(
        bytes[domain_start..domain_end]
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect(),
    )
    .expect("ASCII domain");

    Ok((
        Address {
            local: local_string,
            domain: domain_string,
            quoted,
        },
        local_origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(index) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if index == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(index);
        }
    }

    let safe = valid_dot_atom(address.local.as_bytes());
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    use std::collections::HashMap;

    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut first: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = first.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    first.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}

fn bad_fold_at(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes[i] == b'\r' {
        if i + 2 < bytes.len()
            && bytes[i + 1] == b'\n'
            && matches!(bytes[i + 2], b' ' | b'\t')
        {
            None
        } else {
            Some(i)
        }
    } else if bytes[i] == b'\n' {
        Some(i)
    } else {
        None
    }
}

fn cleanup(input: &str) -> Result<(Vec<Byte>, Vec<Run>), EmailError> {
    let b = input.as_bytes();
    let mut kept = Vec::with_capacity(b.len());
    let mut runs = Vec::new();
    let mut i = 0;

    // Only the initial retained quote prefix protects content during cleanup.
    let mut quote_prefix = false;

    while i < b.len() {
        if quote_prefix {
            let start = i;
            while i < b.len() {
                let c = b[i];
                kept.push(Byte { value: c, pos: i });
                i += 1;
                if c == b'\\' {
                    if i < b.len() {
                        kept.push(Byte {
                            value: b[i],
                            pos: i,
                        });
                        i += 1;
                    }
                } else if c == b'"' {
                    quote_prefix = false;
                    break;
                }
            }
            let _ = start;
            continue;
        }

        let c = b[i];

        if c == b'"' && kept.is_empty() {
            quote_prefix = true;
            kept.push(Byte { value: c, pos: i });
            i += 1;
            continue;
        }

        if matches!(c, b' ' | b'\t' | b'\r' | b'\n' | b'(') {
            let run_start = i;
            loop {
                if i >= b.len() {
                    break;
                }
                match b[i] {
                    b' ' | b'\t' => i += 1,
                    b'\r' | b'\n' => {
                        if let Some(at) = bad_fold_at(b, i) {
                            return Err(EmailError::BadFold { at });
                        }
                        i += 2;
                        while i < b.len() && matches!(b[i], b' ' | b'\t') {
                            i += 1;
                        }
                    }
                    b'(' => {
                        let outer = i;
                        let mut depth = 0usize;
                        loop {
                            if i >= b.len() {
                                return Err(EmailError::UnclosedComment { at: outer });
                            }
                            let x = b[i];
                            match x {
                                b'\\' => {
                                    let slash = i;
                                    if i + 1 >= b.len()
                                        || !(0x20..=0x7e).contains(&b[i + 1])
                                    {
                                        return Err(EmailError::CommentEscape { at: slash });
                                    }
                                    i += 2;
                                }
                                b'(' => {
                                    depth += 1;
                                    if depth > 3 {
                                        return Err(EmailError::CommentDepth { at: i });
                                    }
                                    i += 1;
                                }
                                b')' => {
                                    depth -= 1;
                                    i += 1;
                                    if depth == 0 {
                                        break;
                                    }
                                }
                                b'\r' | b'\n' => {
                                    if let Some(at) = bad_fold_at(b, i) {
                                        return Err(EmailError::BadFold { at });
                                    }
                                    i += 2;
                                    while i < b.len() && matches!(b[i], b' ' | b'\t') {
                                        i += 1;
                                    }
                                }
                                b' ' | b'\t' => i += 1,
                                0x20..=0x7e => i += 1,
                                _ => return Err(EmailError::CommentChar { at: i }),
                            }
                        }
                    }
                    _ => break,
                }
            }
            runs.push(Run {
                start: run_start,
                end: i,
            });
            continue;
        }

        if c == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if let Some(at) = bad_fold_at(b, i) {
            return Err(EmailError::BadFold { at });
        }
        kept.push(Byte { value: c, pos: i });
        i += 1;
    }

    // Validate run boundaries only after all cleanup lexical errors are known.
    for r in &runs {
        let left = kept.iter().position(|x| x.pos >= r.end);
        let right = kept.iter().rposition(|x| x.pos < r.start);
        let left_ok = match left {
            None => true,
            Some(ix) => kept[ix].value == b'@',
        };
        let right_ok = match right {
            None => true,
            Some(ix) => kept[ix].value == b'@',
        };
        // A run is boundary legal if at sequence edge or adjacent to @.
        if !(left_ok || right_ok) {
            return Err(EmailError::CfwsBoundary { at: r.start });
        }
    }

    Ok((kept, runs))
}

fn pos(bytes: &[Byte], i: usize, input_len: usize) -> usize {
    bytes.get(i).map_or(input_len, |x| x.pos)
}

fn dot_atom_valid(s: &[u8]) -> bool {
    if s.is_empty() {
        return false;
    }
    let mut prev_dot = true;
    for (i, &c) in s.iter().enumerate() {
        if c == b'.' {
            if prev_dot || i + 1 == s.len() {
                return false;
            }
            prev_dot = true;
        } else if c.is_ascii_alphanumeric()
            || b"!#$%&'*+-/=?^_`{|}~".contains(&c)
        {
            prev_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_core(
    input: &str,
    allow_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = input.as_bytes().iter().position(|&c| c >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (bytes, _) = cleanup(input)?;
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let input_len = input.len();
    let quoted_prefix = bytes[0].value == b'"';
    let closing = if quoted_prefix {
        let mut i = 1;
        let mut found = None;
        while i < bytes.len() {
            match bytes[i].value {
                b'\\' => i += 2,
                b'"' => {
                    found = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match found {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: bytes[0].pos }),
        }
    } else {
        None
    };

    let mut ats = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i].value == b'@'
            && !(quoted_prefix && closing.map_or(false, |c| i <= c))
        {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: bytes[ats[1]].pos,
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: bytes[sep].pos,
        });
    }

    let mut local = Vec::new();
    let mut origins = Vec::new();
    let mut is_quoted = false;

    if quoted_prefix {
        let close = closing.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&bytes, close + 1, input_len),
            });
        }
        is_quoted = true;
        let mut i = 1;
        while i < close {
            let c = bytes[i].value;
            if c == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: bytes[i].pos });
                }
                let next = bytes[i + 1].value;
                if !matches!(next, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: bytes[i].pos });
                }
                local.push(next);
                origins.push(bytes[i].pos);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&c) {
                    return Err(EmailError::QuotedChar { at: bytes[i].pos });
                }
                local.push(c);
                origins.push(bytes[i].pos);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: bytes[close].pos,
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: bytes[64].pos });
        }
        let mut prev_dot = false;
        for i in 0..sep {
            let c = bytes[i].value;
            if c == b'.' {
                if i == 0 || i + 1 == sep || prev_dot {
                    return Err(EmailError::LocalDot { at: bytes[i].pos });
                }
                prev_dot = true;
            } else {
                if !c.is_ascii_alphanumeric()
                    && !b"!#$%&'*+-/=?^_`{|}~".contains(&c)
                {
                    return Err(EmailError::LocalChar { at: bytes[i].pos });
                }
                prev_dot = false;
            }
            local.push(c);
            origins.push(bytes[i].pos);
        }
    }

    let domain_start = sep + 1;
    let domain_end = bytes.len();
    let rooted = domain_end > domain_start && bytes[domain_end - 1].value == b'.';
    let effective_end = if rooted { domain_end - 1 } else { domain_end };
    if effective_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: pos(&bytes, domain_start, input_len),
        });
    }
    if effective_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: bytes[domain_start + 253].pos,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..effective_end {
        if bytes[i].value == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    labels.push((start, effective_end));

    for &(a, z) in &labels {
        if a == z {
            let at = if a == effective_end && rooted {
                bytes[domain_end - 1].pos
            } else if a == domain_start {
                pos(&bytes, a, input_len)
            } else {
                bytes[a - 1].pos
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if z - a > 63 {
            return Err(EmailError::LabelTooLong { at: bytes[a + 63].pos });
        }
        if bytes[a].value == b'-' {
            return Err(EmailError::LabelHyphen { at: bytes[a].pos });
        }
        if bytes[z - 1].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: bytes[z - 1].pos,
            });
        }
        for i in a..z {
            let c = bytes[i].value;
            if !c.is_ascii_alphanumeric() && c != b'-' {
                return Err(EmailError::DomainChar { at: bytes[i].pos });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&bytes, domain_start, input_len),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end]
        .iter()
        .all(|x| x.value.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: bytes[last_start].pos,
        });
    }

    let local_string = String::from_utf8(local).unwrap();
    let domain_string = bytes[domain_start..effective_end]
        .iter()
        .map(|x| x.value as char)
        .collect::<String>()
        .to_ascii_lowercase();

    Ok((
        Address {
            local: local_string,
            domain: domain_string,
            quoted: is_quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(ix) = address.local.as_bytes().iter().position(|&c| c == b'+') {
            if ix == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(ix);
        }
    }

    let safe = dot_atom_valid(address.local.as_bytes());
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }

    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = keys
                    .iter()
                    .find(|(local, domain, _)| {
                        local == &address.local && domain == &address.domain
                    })
                {
                    duplicate_of[index] = Some(*first_index);
                } else {
                    keys.push((
                        address.local.clone(),
                        address.domain.clone(),
                        index,
                    ));
                    addresses.push(address);
                    source_indices.push(index);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}

    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn pos(positions: &[usize], index: usize, input_len: usize) -> usize {
    positions.get(index).copied().unwrap_or(input_len)
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut first_retained = true;
    let mut quote_open = false;
    let mut quote_escape = false;

    while i < input.len() {
        let b = input[i];

        if quote_open {
            bytes.push(b);
            positions.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                quote_open = false;
            }
            i += 1;
            continue;
        }

        if first_retained && b == b'"' {
            first_retained = false;
            quote_open = true;
            bytes.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        if is_hws(b) || b == b'(' || b == b'\r' || b == b'\n' {
            let run_start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if is_hws(c) {
                    i += 1;
                } else if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if c == b'(' {
                    let outer = i;
                    let mut depth = 1usize;
                    i += 1;
                    while i < input.len() && depth > 0 {
                        let x = input[i];
                        if x == b'\\' {
                            let slash = i;
                            if i + 1 >= input.len()
                                || !(0x20..=0x7e).contains(&input[i + 1])
                            {
                                return Err(EmailError::CommentEscape { at: slash });
                            }
                            i += 2;
                        } else if x == b'\r' || x == b'\n' {
                            if let Some(end) = fold_end(input, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if x == b'(' {
                            if depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            depth += 1;
                            i += 1;
                        } else if x == b')' {
                            depth -= 1;
                            i += 1;
                        } else if x == b'\t' || (0x20..=0x7e).contains(&x) {
                            i += 1;
                        } else {
                            return Err(EmailError::CommentChar { at: i });
                        }
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                } else {
                    break;
                }
            }
            runs.push(Run {
                start: run_start,
                end: i,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if b == b'\r' || b == b'\n' {
            return Err(EmailError::BadFold { at: i });
        }

        bytes.push(b);
        positions.push(i);
        first_retained = false;
        i += 1;
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for &b in bytes {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else if is_atom_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    !previous_dot
}

fn parse_internal(
    input: &str,
    policy: &Policy,
    allow_single_label: bool,
) -> Result<(Address, Option<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let bytes = &cleaned.bytes;
    let positions = &cleaned.positions;

    for run in &cleaned.runs {
        let left_at = run.start > 0 && bytes.get(run.start - 1) == Some(&b'@');
        let right_at = bytes.get(run.start) == Some(&b'@');
        if run.start != 0 && run.start != bytes.len() && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary {
                at: run.start.min(raw.len()),
            });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i = i.saturating_add(2);
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(positions, 0, raw.len()),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quote_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos(positions, separators[1], raw.len()),
        });
    }
    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(positions, separator, raw.len()),
        });
    }

    let local_bytes = &bytes[..separator];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;

    if local_bytes.first() == Some(&b'"') {
        quoted = true;
        let close = quote_end.expect("opening quote has a closing quote");
        if close + 1 != separator {
            return Err(EmailError::TrailingLocal {
                at: pos(positions, close + 1, raw.len()),
            });
        }

        let mut j = 1;
        while j < close {
            let b = local_bytes[j];
            if b == b'\\' {
                if j + 1 >= close
                    || !matches!(local_bytes[j + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape {
                        at: pos(positions, j, raw.len()),
                    });
                }
                local.push(local_bytes[j + 1]);
                origins.push(pos(positions, j, raw.len()));
                j += 2;
            } else if (0x20..=0x7e).contains(&b) {
                local.push(b);
                origins.push(pos(positions, j, raw.len()));
                j += 1;
            } else {
                return Err(EmailError::QuotedChar {
                    at: pos(positions, j, raw.len()),
                });
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(positions, close, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(positions, 64, raw.len()),
            });
        }
        for (j, &b) in local_bytes.iter().enumerate() {
            let at = pos(positions, j, raw.len());
            if b == b'.' {
                if j == 0
                    || j + 1 == local_bytes.len()
                    || local_bytes.get(j.wrapping_sub(1)) == Some(&b'.')
                {
                    return Err(EmailError::LocalDot { at });
                }
            } else if !is_atom_char(b) {
                return Err(EmailError::LocalChar { at });
            }
            local.push(b);
            origins.push(at);
        }
    }

    let domain_start_index = separator + 1;
    let domain_start = pos(positions, domain_start_index, raw.len());
    let mut domain_end = bytes.len();
    if domain_end > domain_start_index && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
    }
    if domain_end == domain_start_index {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_bytes = &bytes[domain_start_index..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(positions, domain_start_index + 253, raw.len()),
        });
    }

    let mut labels = Vec::new();
    let mut label_start = 0;
    for idx in 0..=domain_bytes.len() {
        if idx == domain_bytes.len() || domain_bytes[idx] == b'.' {
            let label = &domain_bytes[label_start..idx];
            let original_label_start = domain_start_index + label_start;
            if label.is_empty() {
                let at = if idx == domain_bytes.len() && domain_end < bytes.len() {
                    pos(positions, domain_end, raw.len())
                } else if idx < domain_bytes.len() {
                    pos(positions, domain_start_index + idx, raw.len())
                } else {
                    domain_start
                };
                return Err(EmailError::EmptyLabel { at });
            }
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: pos(positions, original_label_start + 63, raw.len()),
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: pos(positions, original_label_start, raw.len()),
                });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: pos(
                        positions,
                        original_label_start + label.len() - 1,
                        raw.len(),
                    ),
                });
            }
            for (offset, &b) in label.iter().enumerate() {
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar {
                        at: pos(positions, original_label_start + offset, raw.len()),
                    });
                }
            }
            labels.push((label, original_label_start));
            label_start = idx + 1;
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let (last_label, last_start) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: pos(positions, last_start, raw.len()),
        });
    }

    let normalized_local = match policy.local_case {
        LocalCase::Preserve => local.clone(),
        LocalCase::Lower => local
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect::<Vec<_>>(),
    };

    let mut transformed = normalized_local;
    let mut transformed_origins = origins;
    if policy.strip_subaddress {
        if let Some(plus) = transformed.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: transformed_origins[plus],
                });
            }
            transformed.truncate(plus);
            transformed_origins.truncate(plus);
        }
    }

    let is_safe = valid_dot_atom(&transformed);
    let final_quoted = if is_safe {
        if policy.unquote_safe {
            false
        } else {
            quoted
        }
    } else {
        true
    };

    let local = String::from_utf8(transformed).expect("validated ASCII local");
    let domain = domain_bytes
        .iter()
        .map(u8::to_ascii_lowercase)
        .map(char::from)
        .collect::<String>();

    Ok((
        Address {
            local,
            domain,
            quoted: final_quoted,
        },
        None,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, &Policy::default(), false).map(|pair| pair.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_internal(input, policy, policy.allow_single_label).map(|pair| pair.0)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    seen.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
