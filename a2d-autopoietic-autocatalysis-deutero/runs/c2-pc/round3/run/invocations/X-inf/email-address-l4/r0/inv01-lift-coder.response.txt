use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<Run>,
}

fn pos(positions: &[usize], index: usize, input_len: usize) -> usize {
    positions.get(index).copied().unwrap_or(input_len)
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut first_retained = true;
    let mut quote_open = false;
    let mut quote_escape = false;

    while i < input.len() {
        let b = input[i];

        if quote_open {
            bytes.push(b);
            positions.push(i);
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                quote_open = false;
            }
            i += 1;
            continue;
        }

        if first_retained && b == b'"' {
            first_retained = false;
            quote_open = true;
            bytes.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        if is_hws(b) || b == b'(' || b == b'\r' || b == b'\n' {
            let run_start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if is_hws(c) {
                    i += 1;
                } else if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if c == b'(' {
                    let outer = i;
                    let mut depth = 1usize;
                    i += 1;
                    while i < input.len() && depth > 0 {
                        let x = input[i];
                        if x == b'\\' {
                            let slash = i;
                            if i + 1 >= input.len()
                                || !(0x20..=0x7e).contains(&input[i + 1])
                            {
                                return Err(EmailError::CommentEscape { at: slash });
                            }
                            i += 2;
                        } else if x == b'\r' || x == b'\n' {
                            if let Some(end) = fold_end(input, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if x == b'(' {
                            if depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            depth += 1;
                            i += 1;
                        } else if x == b')' {
                            depth -= 1;
                            i += 1;
                        } else if x == b'\t' || (0x20..=0x7e).contains(&x) {
                            i += 1;
                        } else {
                            return Err(EmailError::CommentChar { at: i });
                        }
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                } else {
                    break;
                }
            }
            runs.push(Run {
                start: run_start,
                end: i,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if b == b'\r' || b == b'\n' {
            return Err(EmailError::BadFold { at: i });
        }

        bytes.push(b);
        positions.push(i);
        first_retained = false;
        i += 1;
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for &b in bytes {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else if is_atom_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    !previous_dot
}

fn parse_internal(
    input: &str,
    policy: &Policy,
    allow_single_label: bool,
) -> Result<(Address, Option<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = cleanup(raw)?;
    let bytes = &cleaned.bytes;
    let positions = &cleaned.positions;

    for run in &cleaned.runs {
        let left_at = run.start > 0 && bytes.get(run.start - 1) == Some(&b'@');
        let right_at = bytes.get(run.start) == Some(&b'@');
        if run.start != 0 && run.start != bytes.len() && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary {
                at: run.start.min(raw.len()),
            });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i = i.saturating_add(2);
            } else if bytes[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(positions, 0, raw.len()),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quote_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos(positions, separators[1], raw.len()),
        });
    }
    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(positions, separator, raw.len()),
        });
    }

    let local_bytes = &bytes[..separator];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;

    if local_bytes.first() == Some(&b'"') {
        quoted = true;
        let close = quote_end.expect("opening quote has a closing quote");
        if close + 1 != separator {
            return Err(EmailError::TrailingLocal {
                at: pos(positions, close + 1, raw.len()),
            });
        }

        let mut j = 1;
        while j < close {
            let b = local_bytes[j];
            if b == b'\\' {
                if j + 1 >= close
                    || !matches!(local_bytes[j + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape {
                        at: pos(positions, j, raw.len()),
                    });
                }
                local.push(local_bytes[j + 1]);
                origins.push(pos(positions, j, raw.len()));
                j += 2;
            } else if (0x20..=0x7e).contains(&b) {
                local.push(b);
                origins.push(pos(positions, j, raw.len()));
                j += 1;
            } else {
                return Err(EmailError::QuotedChar {
                    at: pos(positions, j, raw.len()),
                });
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(positions, close, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(positions, 64, raw.len()),
            });
        }
        for (j, &b) in local_bytes.iter().enumerate() {
            let at = pos(positions, j, raw.len());
            if b == b'.' {
                if j == 0
                    || j + 1 == local_bytes.len()
                    || local_bytes.get(j.wrapping_sub(1)) == Some(&b'.')
                {
                    return Err(EmailError::LocalDot { at });
                }
            } else if !is_atom_char(b) {
                return Err(EmailError::LocalChar { at });
            }
            local.push(b);
            origins.push(at);
        }
    }

    let domain_start_index = separator + 1;
    let domain_start = pos(positions, domain_start_index, raw.len());
    let mut domain_end = bytes.len();
    if domain_end > domain_start_index && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
    }
    if domain_end == domain_start_index {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_bytes = &bytes[domain_start_index..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(positions, domain_start_index + 253, raw.len()),
        });
    }

    let mut labels = Vec::new();
    let mut label_start = 0;
    for idx in 0..=domain_bytes.len() {
        if idx == domain_bytes.len() || domain_bytes[idx] == b'.' {
            let label = &domain_bytes[label_start..idx];
            let original_label_start = domain_start_index + label_start;
            if label.is_empty() {
                let at = if idx == domain_bytes.len() && domain_end < bytes.len() {
                    pos(positions, domain_end, raw.len())
                } else if idx < domain_bytes.len() {
                    pos(positions, domain_start_index + idx, raw.len())
                } else {
                    domain_start
                };
                return Err(EmailError::EmptyLabel { at });
            }
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: pos(positions, original_label_start + 63, raw.len()),
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: pos(positions, original_label_start, raw.len()),
                });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: pos(
                        positions,
                        original_label_start + label.len() - 1,
                        raw.len(),
                    ),
                });
            }
            for (offset, &b) in label.iter().enumerate() {
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar {
                        at: pos(positions, original_label_start + offset, raw.len()),
                    });
                }
            }
            labels.push((label, original_label_start));
            label_start = idx + 1;
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }
    let (last_label, last_start) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: pos(positions, last_start, raw.len()),
        });
    }

    let normalized_local = match policy.local_case {
        LocalCase::Preserve => local.clone(),
        LocalCase::Lower => local
            .iter()
            .map(u8::to_ascii_lowercase)
            .collect::<Vec<_>>(),
    };

    let mut transformed = normalized_local;
    let mut transformed_origins = origins;
    if policy.strip_subaddress {
        if let Some(plus) = transformed.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: transformed_origins[plus],
                });
            }
            transformed.truncate(plus);
            transformed_origins.truncate(plus);
        }
    }

    let is_safe = valid_dot_atom(&transformed);
    let final_quoted = if is_safe {
        if policy.unquote_safe {
            false
        } else {
            quoted
        }
    } else {
        true
    };

    let local = String::from_utf8(transformed).expect("validated ASCII local");
    let domain = domain_bytes
        .iter()
        .map(u8::to_ascii_lowercase)
        .map(char::from)
        .collect::<String>();

    Ok((
        Address {
            local,
            domain,
            quoted: final_quoted,
        },
        None,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, &Policy::default(), false).map(|pair| pair.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_internal(input, policy, policy.allow_single_label).map(|pair| pair.0)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    seen.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
