pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resources in task order, completing each task's checks first.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in task.needs.iter().zip(capacity).enumerate() {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build successor lists and indegrees. Dependency list order cannot affect
    // the min-ID Kahn traversal.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    let mut available = Vec::<usize>::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(task_id);
        }
    }

    let mut kahn_order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, task_id)| *task_id)
            .map(|(pos, _)| pos)
            .unwrap();
        let task_id = available.swap_remove(smallest_pos);
        kahn_order.push(task_id);

        for &child in &successors[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if kahn_order.len() != n {
        let removed = {
            let mut bits = vec![false; n];
            for &task_id in &kahn_order {
                bits[task_id] = true;
            }
            bits
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&task_id| !removed[task_id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in kahn_order.iter().rev() {
        let max_successor_tail = successors[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = String::new();
    use std::fmt::Write as _;
    writeln!(
        &mut trace,
        "workers={};resources={}",
        worker_count,
        capacity.len()
    )
    .unwrap();

    let mut runs = vec![
        Run {
            start: 0,
            finish: 0,
            worker: 0,
        };
        n
    ];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut used = vec![0u64; capacity.len()];
    let mut current_time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now in ascending task ID.
        let mut due: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, item)| (item.finish == current_time).then_some(index))
            .collect();
        due.sort_by_key(|&index| running[index].task);

        for index in due.into_iter().rev() {
            let item = running.swap_remove(index);
            let task_id = item.task;

            for resource in 0..capacity.len() {
                used[resource] -= u64::from(tasks[task_id].needs[resource]);
            }

            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(item.finish);

            write!(
                &mut trace,
                "{}|end|{}|{}|",
                current_time, task_id, item.worker
            )
            .unwrap();
            write_resource_vector(&mut trace, &used);
            trace.push('\n');

            for &child in &successors[task_id] {
                remaining_predecessors[child] -= 1;
            }
        }

        loop {
            let mut best: Option<usize> = None;

            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > current_time
                    || remaining_predecessors[task_id] != 0
                    || !fits(&tasks[task_id].needs, &used, capacity)
                {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(old) => {
                        tails[task_id] > tails[old]
                            || (tails[task_id] == tails[old]
                                && (tasks[task_id].priority > tasks[old].priority
                                    || (tasks[task_id].priority == tasks[old].priority
                                        && task_id < old)))
                    }
                };
                if better {
                    best = Some(task_id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            let mut chosen_worker = None;
            for offset in 0..worker_count {
                let worker = (task_id % worker_count + offset) % worker_count;
                if worker_available[worker] <= current_time {
                    chosen_worker = Some(worker);
                    break;
                }
            }
            let Some(worker) = chosen_worker else {
                break;
            };

            let finish = current_time
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Run {
                start: current_time,
                finish,
                worker,
            };

            for resource in 0..capacity.len() {
                used[resource] += u64::from(tasks[task_id].needs[resource]);
            }

            write!(
                &mut trace,
                "{}|start|{}|{}|",
                current_time, task_id, worker
            )
            .unwrap();
            write_resource_vector(&mut trace, &used);
            trace.push('\n');

            if tasks[task_id].duration == 0 {
                // Zero-duration completion is immediate, before another start.
                for resource in 0..capacity.len() {
                    used[resource] -= u64::from(tasks[task_id].needs[resource]);
                }

                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);

                write!(
                    &mut trace,
                    "{}|end|{}|{}|",
                    current_time, task_id, worker
                )
                .unwrap();
                write_resource_vector(&mut trace, &used);
                trace.push('\n');

                for &child in &successors[task_id] {
                    remaining_predecessors[child] -= 1;
                }
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > current_time {
                next_time = Some(next_time.map_or(candidate, |old| old.min(candidate)));
            }
        };

        for item in &running {
            consider(item.finish);
        }
        for task_id in 0..n {
            if !started[task_id] {
                consider(tasks[task_id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        // There must be a future event if unfinished tasks remain: all releases
        // are finite, and a cycle has already been ruled out.
        current_time = next_time.expect("unfinished acyclic schedule has a future event");
    }

    writeln!(&mut trace, "makespan={}", makespan).unwrap();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn fits(needs: &[u32], used: &[u64], capacity: &[u32]) -> bool {
    needs.iter().enumerate().all(|(resource, &need)| {
        u64::from(need) <= u64::from(capacity[resource]) - used[resource]
    })
}

fn write_resource_vector(output: &mut String, used: &[u64]) {
    use std::fmt::Write as _;
    if used.is_empty() {
        output.push('-');
        return;
    }
    for (index, count) in used.iter().enumerate() {
        if index != 0 {
            output.push(',');
        }
        write!(output, "{}", count).unwrap();
    }
}

        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    let mut children = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    let mut available = BinaryMinHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(id) = available.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        for id in 0..n {
            if !removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut used = vec![0u64; capacity.len()];
    let mut trace = format!("workers={worker_count};resources={}\n", capacity.len());
    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        let mut finishing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(i, r)| (r.finish == now).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| running[i].task);

        for index in finishing.into_iter().rev() {
            let entry = running.remove(index);
            let task_id = entry.task;
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] -= u64::from(need);
            }
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(now);
            write_event(&mut trace, now, "end", task_id, entry.worker, &used);
            for &child in &children[task_id] {
                remaining_deps[child] -= 1;
            }
        }

        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || remaining_deps[id] != 0
                    || worker_available.iter().all(|&t| t > now)
                    || !fits(&tasks[id].needs, &used, capacity)
                {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= now)
                .expect("available worker was checked");

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let turn_available = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = turn_available;
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            write_event(&mut trace, now, "start", task_id, worker, &used);

            if finish == now {
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                write_event(&mut trace, now, "end", task_id, worker, &used);
                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push(Running {
                    task: task_id,
                    finish,
                    worker,
                });
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now && next_time.map_or(true, |current| time < current) {
                next_time = Some(time);
            }
        };

        for entry in &running {
            consider(entry.finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &time in &worker_available {
            consider(time);
        }

        now = next_time.expect("unfinished tasks always have a future event");
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();
    trace.push_str(&format!("makespan={makespan}\n"));
    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

#[derive(Clone, Copy)]
struct Running {
    task: usize,
    finish: u64,
    worker: usize,
}

fn fits(needs: &[u32], used: &[u64], capacity: &[u32]) -> bool {
    needs
        .iter()
        .zip(used.iter().zip(capacity.iter()))
        .all(|(&need, (&in_use, &cap))| u64::from(need) <= u64::from(cap) - in_use)
}

fn write_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    use std::fmt::Write;

    let _ = write!(trace, "{time}|{kind}|{task}|{worker}|");
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            let _ = write!(trace, "{value}");
        }
    }
    trace.push('\n');
}

struct BinaryMinHeap {
    values: Vec<usize>,
}

impl BinaryMinHeap {
    fn new() -> Self {
        Self { values: Vec::new() }
    }

    fn push(&mut self, value: usize) {
        self.values.push(value);
        let mut index = self.values.len() - 1;
        while index > 0 {
            let parent = (index - 1) / 2;
            if self.values[parent] <= self.values[index] {
                break;
            }
            self.values.swap(parent, index);
            index = parent;
        }
    }

    fn pop(&mut self) -> Option<usize> {
        let result = *self.values.first()?;
        let last = self.values.pop().unwrap();
        if !self.values.is_empty() {
            self.values[0] = last;
            let mut index = 0;
            loop {
                let left = index * 2 + 1;
                let right = left + 1;
                if left >= self.values.len() {
                    break;
                }
                let child = if right < self.values.len()
                    && self.values[right] < self.values[left]
                {
                    right
                } else {
                    left
                };
                if self.values[index] <= self.values[child] {
                    break;
                }
                self.values.swap(index, child);
                index = child;
            }
        }
        Some(result)
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency { task: usize, position: usize, value: usize },
    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },
    Cycle { blocked: Vec<usize> },
    Workers { count: usize },
    Overflow { task: usize, stage: Stage },
    ResourceShape { task: usize, expected: usize, actual: usize },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry in the specified order.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal, always taking the smallest available ID.
    let mut remaining_indegree = indegree.clone();
    let mut available = std::collections::BTreeSet::new();
    for id in 0..n {
        if remaining_indegree[id] == 0 {
            available.insert(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while let Some(id) = available.iter().next().copied() {
        available.remove(&id);
        order.push(id);
        for &child in &children[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let blocked = (0..n)
            .filter(|&id| remaining_indegree[id] != 0)
            .collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let longest_child = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(longest_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut used = vec![0u64; resource_count];
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut remaining_predecessors = indegree;

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        finish: u64,
        worker: usize,
    }

    let mut running: Vec<Running> = Vec::new();
    let mut worker_available = vec![0u64; worker_count];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut makespan = 0u64;
    let mut trace = String::new();
    use std::fmt::Write;
    writeln!(
        &mut trace,
        "workers={};resources={}",
        worker_count, resource_count
    )
    .unwrap();

    let format_used = |used: &[u64]| -> String {
        if used.is_empty() {
            "-".to_string()
        } else {
            used.iter()
                .map(u64::to_string)
                .collect::<Vec<_>>()
                .join(",")
        }
    };

    let mut now = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete all tasks finishing now, in ascending task ID.
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|entry| entry.finish == now)
            .collect();
        finishing.sort_by_key(|entry| entry.task);
        for entry in finishing {
            running.retain(|other| other.task != entry.task);
            completed[entry.task] = true;
            completed_count += 1;
            for (resource, &need) in tasks[entry.task].needs.iter().enumerate() {
                used[resource] -= u64::from(need);
            }
            for &child in &children[entry.task] {
                remaining_predecessors[child] -= 1;
            }
            writeln!(
                &mut trace,
                "{}|end|{}|{}|{}",
                now,
                entry.task,
                entry.worker,
                format_used(&used)
            )
            .unwrap();
        }

        // Admit tasks one at a time, reevaluating after every start.
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_predecessors[id] == 0
                        && tasks[id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r])
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let mut chosen = None;
            for task_id in candidates {
                for offset in 0..worker_count {
                    let worker = (task_id % worker_count + offset) % worker_count;
                    if worker_available[worker] <= now {
                        chosen = Some((task_id, worker));
                        break;
                    }
                }
                if chosen.is_some() {
                    break;
                }
            }

            let Some((task_id, worker)) = chosen else {
                break;
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            running.push(Running {
                task: task_id,
                finish,
                worker,
            });

            writeln!(
                &mut trace,
                "{}|start|{}|{}|{}",
                now,
                task_id,
                worker,
                format_used(&used)
            )
            .unwrap();

            if tasks[task_id].duration == 0 {
                running.retain(|entry| entry.task != task_id);
                completed[task_id] = true;
                completed_count += 1;
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                for &child in &children[task_id] {
                    remaining_predecessors[child] -= 1;
                }
                writeln!(
                    &mut trace,
                    "{}|end|{}|{}|{}",
                    now,
                    task_id,
                    worker,
                    format_used(&used)
                )
                .unwrap();
            } else {
                makespan = makespan.max(finish);
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for entry in &running {
            if entry.finish > now {
                next_time = Some(next_time.map_or(entry.finish, |t| t.min(entry.finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next_time = Some(next_time.map_or(tasks[id].release, |t| t.min(tasks[id].release)));
            }
        }
        for worker in 0..worker_count {
            if worker_available[worker] > now {
                next_time =
                    Some(next_time.map_or(worker_available[worker], |t| t.min(worker_available[worker])));
            }
        }

        // A task's finish is also its resource and dependency release event.
        // If no future event exists, no unfinished task can be schedulable.
        now = next_time.expect("unfinished tasks must have a future event");
    }

    // Include zero-duration finishes in makespan (always their start time).
    for run in runs.iter().flatten() {
        makespan = makespan.max(run.finish);
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();
    writeln!(&mut trace, "makespan={}", makespan).unwrap();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}
