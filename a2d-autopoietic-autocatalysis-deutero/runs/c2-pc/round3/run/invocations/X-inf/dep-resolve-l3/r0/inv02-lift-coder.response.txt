#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn encode(text: &str) -> String {
    let mut out = String::new();
    for b in text.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", b));
        }
    }
    out
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches(version: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

struct Search<'a> {
    catalog: &'a [Package],
    by_name: std::collections::BTreeMap<&'a str, &'a Package>,
    locks: std::collections::BTreeMap<&'a str, Version>,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<Selection>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn inspect(
        &mut self,
        assigned: &mut std::collections::BTreeMap<String, Version>,
        active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    ) -> Result<(), ResolveError> {
        *active = std::collections::BTreeMap::new();
        for (name, constraints) in active.iter_mut() {
            let _ = (name, constraints);
        }
        // Roots are stored separately by the caller through `root_constraints`.
        unreachable!()
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }
    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }
    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut by_name = std::collections::BTreeMap::new();
    for package in catalog {
        by_name.insert(package.name.as_str(), package);
    }
    let mut lock_map = std::collections::BTreeMap::new();
    for lock in locks {
        lock_map.insert(lock.package.as_str(), lock.version);
    }

    let mut search = Search {
        catalog,
        by_name,
        locks: lock_map,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut assigned = std::collections::BTreeMap::new();
    let mut active = std::collections::BTreeMap::new();
    search_exhaustive(&mut search, roots, &mut assigned, &mut active)?;

    let selected = search
        .best
        .map(|(_, selections)| selections)
        .unwrap_or_default();
    let final_constraints = constraints_for(roots, &selected);

    let before: std::collections::BTreeMap<&str, Version> =
        locks.iter().map(|l| (l.package.as_str(), l.version)).collect();
    let after: std::collections::BTreeMap<&str, Version> = selected
        .iter()
        .map(|s| (s.package.as_str(), s.version))
        .collect();
    let mut names = std::collections::BTreeSet::new();
    names.extend(before.keys().copied());
    names.extend(after.keys().copied());

    let mut changes = Vec::new();
    for name in names {
        let b = before.get(name).copied();
        let a = after.get(name).copied();
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .by_name
                    .get(name)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_constraints
                    .get(name)
                    .map(|cs| !cs.iter().all(|c| matches(old, *c)))
                    .unwrap_or(false)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name.to_string(),
            before: b,
            after: a,
            reason,
        });
    }
    Ok(Resolution { selected, changes })
}

fn constraints_for(
    roots: &[Requirement],
    selected: &[Selection],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    let selected_map: std::collections::BTreeMap<&str, Version> = selected
        .iter()
        .map(|s| (s.package.as_str(), s.version))
        .collect();
    for selection in selected {
        // Caller selected a release; dependency expansion is handled in search.
        let _ = selected_map.get(selection.package.as_str());
    }
    active
}

fn search_exhaustive(
    search: &mut Search<'_>,
    roots: &[Requirement],
    assigned: &mut std::collections::BTreeMap<String, Version>,
    _active: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
) -> Result<(), ResolveError> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    loop {
        for (name, version) in assigned.iter() {
            if let Some(package) = search.by_name.get(name.as_str()) {
                if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        let mut changed = false;
        for name in active.keys() {
            if !assigned.contains_key(name) {
                continue;
            }
            if let Some(package) = search.by_name.get(name.as_str()) {
                if let Some(release) = package
                    .releases
                    .iter()
                    .find(|r| Some(&r.version) == assigned.get(name))
                {
                    for dep in &release.deps {
                        let entry = active.entry(dep.package.clone()).or_default();
                        if !entry.contains(&dep.constraint) {
                            entry.push(dep.constraint);
                            changed = true;
                        }
                    }
                }
            }
        }
        if !changed {
            break;
        }
    }

    if let Some(name) = active
        .keys()
        .filter(|n| !search.by_name.contains_key(n.as_str()))
        .min()
    {
        remember_error(search, ResolveError::Missing { package: name.clone() });
        return Ok(());
    }

    for (name, version) in assigned.iter() {
        if let Some(constraints) = active.get(name) {
            if !constraints.iter().all(|c| matches(*version, *c)) {
                remember_error(
                    search,
                    ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: canonical(constraints.clone()),
                    },
                );
                return Ok(());
            }
        }
    }

    for (name, constraints) in &active {
        if assigned.contains_key(name) {
            continue;
        }
        let package = search.by_name[name.as_str()];
        if !package
            .releases
            .iter()
            .any(|r| constraints.iter().all(|c| matches(r.version, *c)))
        {
            remember_error(
                search,
                ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical(constraints.clone()),
                },
            );
            return Ok(());
        }
    }

    let next = active.keys().find(|name| !assigned.contains_key(*name)).cloned();
    if let Some(name) = next {
        let constraints = canonical(active[&name].clone());
        let package = search.by_name[name.as_str()];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| constraints.iter().all(|c| matches(*v, *c)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(locked) = search.locks.get(name.as_str()).copied() {
            if let Some(pos) = candidates.iter().position(|v| *v == locked) {
                let version = candidates.remove(pos);
                candidates.insert(0, version);
            }
        }
        for version in candidates {
            if search.attempts == search.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: search.max_attempts,
                });
            }
            search.attempts += 1;
            assigned.insert(name.clone(), version);
            search_exhaustive(search, roots, assigned, &mut active)?;
            assigned.remove(&name);
        }
        return Ok(());
    }

    let mut selections: Vec<Selection> = assigned
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    selections.sort_by(|a, b| a.package.cmp(&b.package));
    let cost = search
        .locks
        .iter()
        .filter(|(package, version)| {
            selections
                .iter()
                .find(|s| s.package == **package)
                .map(|s| s.version != **version)
                .unwrap_or(true)
        })
        .count();
    let replace = match &search.best {
        None => true,
        Some((best_cost, best_selections)) => {
            if cost != *best_cost {
                cost < *best_cost
            } else {
                compare_vectors(&selections, best_selections, search.catalog)
                    == std::cmp::Ordering::Greater
            }
        }
    };
    if replace {
        search.best = Some((cost, selections));
    }
    Ok(())
}

fn remember_error(search: &mut Search<'_>, error: ResolveError) {
    if search.first_error.is_none() {
        search.first_error = Some(error);
    }
}

fn compare_vectors(
    left: &[Selection],
    right: &[Selection],
    catalog: &[Package],
) -> std::cmp::Ordering {
    let names: std::collections::BTreeSet<&str> =
        catalog.iter().map(|p| p.name.as_str()).collect();
    for name in names {
        let l = left.iter().find(|s| s.package == name).map(|s| s.version);
        let r = right.iter().find(|s| s.package == name).map(|s| s.version);
        let ord = match (l, r) {
            (None, None) => std::cmp::Ordering::Equal,
            (None, Some(_)) => std::cmp::Ordering::Less,
            (Some(_), None) => std::cmp::Ordering::Greater,
            (Some(a), Some(b)) => a.cmp(&b),
        };
        if ord != std::cmp::Ordering::Equal {
            return ord;
        }
    }
    std::cmp::Ordering::Equal
}
