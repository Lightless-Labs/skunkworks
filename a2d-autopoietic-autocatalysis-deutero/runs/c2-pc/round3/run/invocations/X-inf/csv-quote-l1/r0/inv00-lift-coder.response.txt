#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut index = if input.starts_with('\u{FEFF}') { 3 } else { 0 };

    if index == bytes.len() {
        return Ok(Vec::new());
    }

    #[derive(Clone, Copy, PartialEq, Eq)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted,
        AfterQuote,
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = State::FieldStart;
    let mut quote_open = 0;
    let mut record_index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: index,
            });
        }

        if state != State::Quoted && byte == b'\r' {
            if bytes.get(index + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr {
                    record: record_index,
                    byte: index,
                });
            }

            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            state = State::FieldStart;
            index += 2;
            record_index += 1;
            continue;
        }

        if state != State::Quoted && byte == b'\n' {
            record.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut record));
            state = State::FieldStart;
            index += 1;
            record_index += 1;
            continue;
        }

        match state {
            State::FieldStart => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    index += 1;
                }
                b'"' => {
                    state = State::Quoted;
                    quote_open = index;
                    index += 1;
                }
                _ => {
                    field.push(byte as char);
                    state = State::Unquoted;
                    index += 1;
                }
            },
            State::Unquoted => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    index += 1;
                }
                b'"' => {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: index,
                    });
                }
                _ => {
                    field.push(byte as char);
                    index += 1;
                }
            },
            State::Quoted => {
                if byte == b'"' {
                    if bytes.get(index + 1) == Some(&b'"') {
                        field.push('"');
                        index += 2;
                    } else {
                        state = State::AfterQuote;
                        index += 1;
                    }
                } else {
                    field.push(byte as char);
                    index += 1;
                }
            }
            State::AfterQuote => match byte {
                b',' => {
                    record.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    index += 1;
                }
                _ => {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: index,
                    });
                }
            },
        }
    }

    if state == State::Quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open,
        });
    }

    record.push(field);
    records.push(record);
    Ok(records)
}
