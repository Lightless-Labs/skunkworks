#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateFeature {
        package: String,
        version: Version,
        feature: String,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
        evidence: Vec<Evidence>,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },

        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(v: &[Constraint]) -> Vec<Constraint> {
    let mut out = v.to_vec();
    out.sort();
    out.dedup();
    out
}

fn constraints_text(v: &[Constraint]) -> String {
    canonical_constraints(v)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn sort_evidence(v: &[Evidence]) -> Vec<Evidence> {
    let mut out = v.to_vec();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    out
}

fn append_evidence(s: &mut String, evidence: &[Evidence]) {
    for e in sort_evidence(evidence) {
        s.push_str("\n  C:");
        s.push_str(&constraint_text(e.constraint));
        s.push_str(" via ");
        s.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn satisfies(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn platform_matches(p: Platform, t: Target) -> bool {
    matches!(
        (p, t),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Info {
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    active_features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

struct Solver<'a> {
    catalog: &'a [Package],
    by_name: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_leaf: Option<ResolveError>,
    best: Option<(usize, BTreeMap<String, Version>)>,
}

impl<'a> Solver<'a> {
    fn selected_release(&self, name: &str, selected: &BTreeMap<String, Version>) -> Option<&'a Release> {
        let ver = selected.get(name)?;
        self.by_name
            .get(name)?
            .releases
            .iter()
            .find(|r| r.version == *ver)
    }

    fn closure(&self, selected: &BTreeMap<String, Version>) -> Info {
        let mut info = Info::default();

        for (i, root) in self.roots.iter().enumerate() {
            let mut fs = root.features.clone();
            fs.sort();
            fs.dedup();
            let token = format!(
                "root[{}]:{}@{}+{{{}}}",
                i,
                root.package,
                constraint_text(root.constraint),
                fs.join(",")
            );
            info.origins
                .entry(root.package.clone())
                .or_default()
                .push(Origin {
                    constraint: root.constraint,
                    path: vec![token],
                });
            info.requests
                .entry(root.package.clone())
                .or_default()
                .extend(root.features.iter().cloned());
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = selected.keys().cloned().collect();
            for source in names {
                let Some(release) = self.selected_release(&source, selected) else {
                    continue;
                };
                let reqs = info.requests.get(&source).cloned().unwrap_or_default();
                let mut active = info.active_features.get(&source).cloned().unwrap_or_default();

                loop {
                    let before = active.len();
                    let known: BTreeMap<String, &Feature> = release
                        .features
                        .iter()
                        .map(|f| (f.name.clone(), f))
                        .collect();
                    let current: Vec<String> = active.iter().cloned().collect();
                    for f in current {
                        if let Some(decl) = known.get(&f) {
                            active.extend(decl.enables.iter().cloned());
                        }
                    }
                    if active.len() == before {
                        break;
                    }
                }

                if !reqs.is_subset(&active) {
                    active.extend(reqs);
                    loop {
                        let before = active.len();
                        let known: BTreeMap<String, &Feature> = release
                            .features
                            .iter()
                            .map(|f| (f.name.clone(), f))
                            .collect();
                        let current: Vec<String> = active.iter().cloned().collect();
                        for f in current {
                            if let Some(decl) = known.get(&f) {
                                active.extend(decl.enables.iter().cloned());
                            }
                        }
                        if active.len() == before {
                            break;
                        }
                    }
                }

                if info.active_features.get(&source) != Some(&active) {
                    info.active_features.insert(source.clone(), active.clone());
                    changed = true;
                }

                for (j, dep) in release.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, self.target)
                        || dep.gate.as_ref().is_some_and(|g| !active.contains(g))
                    {
                        continue;
                    }
                    let mut fs = dep.features.clone();
                    fs.sort();
                    fs.dedup();
                    let gate = dep.gate.as_deref().unwrap_or("*");
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        source,
                        version_text(release.version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        fs.join(","),
                        gate,
                        match dep.platform {
                            Platform::Any => "any",
                            Platform::Linux => "linux",
                            Platform::Mac => "mac",
                            Platform::Windows => "windows",
                        }
                    );

                    let source_path = info.paths.get(&source).cloned().unwrap_or_default();
                    let mut path = source_path;
                    path.push(token);

                    let origin = Origin {
                        constraint: dep.constraint,
                        path,
                    };
                    let origins = info.origins.entry(dep.package.clone()).or_default();
                    if !origins.iter().any(|x| x.constraint == origin.constraint && x.path == origin.path) {
                        origins.push(origin);
                        changed = true;
                    }
                    let dest = info.requests.entry(dep.package.clone()).or_default();
                    let old_len = dest.len();
                    dest.extend(dep.features.iter().cloned());
                    if dest.len() != old_len {
                        changed = true;
                    }
                }
            }

            let mut best_paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
            for os in info.origins.values() {
                for origin in os {
                    if origin.path.len() == 1 {
                        let package = info
                            .origins
                            .iter()
                            .find(|(_, origins)| origins.iter().any(|o| o.path == origin.path && o.constraint == origin.constraint))
                            .map(|(name, _)| name.clone());
                        if let Some(name) = package {
                            choose_path(&mut best_paths, name, origin.path.clone());
                        }
                    }
                }
            }
            for (source, release) in selected.iter().filter_map(|(n, v)| {
                self.selected_release(n, selected).map(|r| (n.clone(), r))
            }) {
                let active = info.active_features.get(&source).cloned().unwrap_or_default();
                let path = info.paths.get(&source).cloned().unwrap_or_default();
                for (j, dep) in release.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, self.target)
                        || dep.gate.as_ref().is_some_and(|g| !active.contains(g))
                    {
                        continue;
                    }
                    if path.is_empty() {
                        continue;
                    }
                    let mut fs = dep.features.clone();
                    fs.sort();
                    fs.dedup();
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        source,
                        version_text(release.version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        fs.join(","),
                        dep.gate.as_deref().unwrap_or("*"),
                        match dep.platform {
                            Platform::Any => "any",
                            Platform::Linux => "linux",
                            Platform::Mac => "mac",
                            Platform::Windows => "windows",
                        }
                    );
                    let mut p = path.clone();
                    p.push(token);
                    choose_path(&mut best_paths, dep.package.clone(), p);
                }
            }
            if info.paths != best_paths {
                info.paths = best_paths;
                changed = true;
            }
            if !changed {
                break;
            }
        }
        info
    }

    fn dynamic_error(
        &self,
        selected: &BTreeMap<String, Version>,
        info: &Info,
    ) -> Option<ResolveError> {
        let active: BTreeSet<String> = info.origins.keys().cloned().collect();

        for name in &active {
            if !self.by_name.contains_key(name) {
                return Some(ResolveError::Missing {
                    package: name.clone(),
                    evidence: make_evidence(info, name),
                });
            }
        }

        for (name, version) in selected {
            let constraints: Vec<Constraint> = info
                .origins
                .get(name)
                .into_iter()
                .flatten()
                .map(|x| x.constraint)
                .collect();
            if !constraints.iter().all(|c| satisfies(*version, *c)) {
                return Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: canonical_constraints(&constraints),
                    evidence: make_evidence(info, name),
                });
            }
        }

        for name in active.iter().filter(|n| !selected.contains_key(*n)) {
            let constraints: Vec<Constraint> = info
                .origins
                .get(name)
                .into_iter()
                .flatten()
                .map(|x| x.constraint)
                .collect();
            let has_candidate = self.by_name[*name].releases.iter().any(|r| {
                constraints.iter().all(|c| satisfies(r.version, *c))
            });
            if !has_candidate {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical_constraints(&constraints),
                    evidence: make_evidence(info, name),
                });
            }
        }

        for name in selected.keys() {
            let Some(release) = self.selected_release(name, selected) else {
                continue;
            };
            let active_features = info.active_features.get(name).cloned().unwrap_or_default();
            for feature in active_features {
                if !release.features.iter().any(|f| f.name == feature) {
                    return Some(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: release.version,
                        feature,
                        path: info.paths.get(name).cloned().unwrap_or_default(),
                    });
                }
            }
        }
        None
    }

    fn search(&mut self, selected: BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let info = self.closure(&selected);
        if let Some(error) = self.dynamic_error(&selected, &info) {
            if matches!(error, ResolveError::Limit { .. }) {
                return Err(error);
            }
            if self.first_leaf.is_none() {
                self.first_leaf = Some(error);
            }
            return Ok(());
        }

        let active: BTreeSet<String> = info.origins.keys().cloned().collect();
        let next = active.iter().find(|n| !selected.contains_key(*n)).cloned();
        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_sel)) => {
                    if cost != *best_cost {
                        cost < *best_cost
                    } else {
                        self.vector_cmp(&selected, best_sel).is_gt()
                    }
                }
            };
            if replace {
                self.best = Some((cost, selected));
            }
            return Ok(());
        };

        let constraints: Vec<Constraint> = info
            .origins
            .get(&name)
            .into_iter()
            .flatten()
            .map(|x| x.constraint)
            .collect();
        let mut candidates: Vec<Version> = self.by_name[&name]
            .releases
            .iter()
            .filter(|r| constraints.iter().all(|c| satisfies(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            let mut child = selected.clone();
            child.insert(name.clone(), candidate);
            self.search(child)?;
        }
        Ok(())
    }

    fn vector_cmp(
        &self,
        a: &BTreeMap<String, Version>,
        b: &BTreeMap<String, Version>,
    ) -> std::cmp::Ordering {
        let names: BTreeSet<String> = self
            .catalog
            .iter()
            .map(|p| p.name.clone())
            .collect();
        for n in names {
            match (a.get(&n), b.get(&n)) {
                (None, None) => {}
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (Some(x), Some(y)) => {
                    let c = x.cmp(y);
                    if !c.is_eq() {
                        return c;
                    }
                }
            }
        }
        std::cmp::Ordering::Equal
    }
}

fn choose_path(paths: &mut BTreeMap<String, Vec<String>>, name: String, path: Vec<String>) {
    match paths.get(&name) {
        None => {
            paths.insert(name, path);
        }
        Some(old) if (path.len(), &path) < (old.len(), old) => {
            paths.insert(name, path);
        }
        _ => {}
    }
}

fn make_evidence(info: &Info, package: &str) -> Vec<Evidence> {
    let mut out = info
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    out
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f),
                            name: name.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: name.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if lock_map.insert(lock.package.clone(), lock.version).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let by_name: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let mut solver = Solver {
        catalog,
        by_name,
        roots,
        locks: lock_map.clone(),
        target,
        max_attempts,
        attempts: 0,
        first_leaf: None,
        best: None,
    };

    solver.search(BTreeMap::new())?;

    let Some((_, chosen)) = solver.best else {
        return Err(solver.first_leaf.unwrap_or(ResolveError::Missing {
            package: String::new(),
            evidence: Vec::new(),
        }));
    };

    let selected = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let final_info = solver.closure(&chosen);
    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = by_name
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints: Vec<Constraint> = final_info
                        .origins
                        .get(&package)
                        .into_iter()
                        .flatten()
                        .map(|o| o.constraint)
                        .collect();
                    if !constraints.iter().all(|c| satisfies(old, *c)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
