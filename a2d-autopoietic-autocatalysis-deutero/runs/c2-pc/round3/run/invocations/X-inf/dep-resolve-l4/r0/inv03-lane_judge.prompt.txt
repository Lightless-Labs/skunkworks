fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn constraint_key(c: &Constraint) -> (u8, Version) {
    match *c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort_by_key(|c| constraint_key(c));
    cs.dedup();
    cs
}

fn constraint_text(c: Constraint) -> String {
    let (s, v) = match c {
        Constraint::Exact(v) => ("=", v),
        Constraint::AtLeast(v) => (">=", v),
        Constraint::Before(v) => ("<", v),
        Constraint::Caret(v) => ("^", v),
    };
    format!("{s}{}.{}.{}", v.major, v.minor, v.patch)
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x && if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {

        ResolveError::DuplicateVersion { package, version } =>
            (format!("duplicate-version|package={}|version={}", encode(package), version_text(*version)), None),
        ResolveError::DuplicateFeature { package, version, feature } =>
            (format!("duplicate-feature|package={}|version={}|feature={}", encode(package), version_text(*version), encode(feature)), None),
        ResolveError::DuplicateLock { package } =>
            (format!("duplicate-lock|package={}", encode(package)), None),
        ResolveError::Missing { package, evidence } =>
            (format!("missing|package={}", encode(package)), Some(evidence.as_slice())),
        ResolveError::NoMatch { package, requirements, evidence } =>

            (format!("conflict|package={}|selected={}|constraints={}", encode(package), version_text(*selected), canonical(requirements.clone()).iter().map(|c| constraint_text(*c)).collect::<Vec<_>>().join(",")), Some(evidence.as_slice())),
        ResolveError::MissingFeature { package, selected, feature, path } =>
            (format!("missing-feature|package={}|selected={}|feature={}", encode(package), version_text(*selected), encode(feature)), None),
        ResolveError::Limit { attempts } => (format!("limit|attempts={attempts}"), None),
    };

    }
}

    first_error: Option<ResolveError>,
    catalog_names: Vec<String>,
}

impl<'a> Solver<'a> {
    fn platform_matches(&self, p: Platform) -> bool {
        p == Platform::Any || matches!((p, self.target),
            (Platform::Linux, Target::Linux) |
            (Platform::Mac, Target::Mac) |
            (Platform::Windows, Target::Windows))
    }

    fn release(&self, name: &str, version: Version) -> Option<&'a Release> {
        self.packages.get(name)?.releases.iter().find(|r| r.version == version)
    }

    fn closure(&self, assigned: &BTreeMap<String, Version>) -> Closure {
        let mut c = Closure::default();
        let mut queue: Vec<(String, String)> = Vec::new();
        for (i, root) in self.roots.iter().enumerate() {
            let mut fs = root.features.clone();
            fs.sort();
            fs.dedup();
            let token = format!("root[{i}]:{}@{}+{{{}}}", root.package, constraint_text(root.constraint), fs.join(","));
            c.constraints.entry(root.package.clone()).or_default().push(root.constraint);
            c.origins.entry(root.package.clone()).or_default().push(Origin {
                constraint: root.constraint, path: vec![token.clone()],
            });
            c.paths.entry(root.package.clone()).and_modify(|p| {
                let candidate = vec![token.clone()];
                if candidate.len() < p.len() || (candidate.len() == p.len() && candidate < *p) { *p = candidate; }
            }).or_insert_with(|| vec![token]);
            for f in fs { queue.push((root.package.clone(), f)); }
        }

        let mut processed_features: BTreeSet<(String, Version, String)> = BTreeSet::new();
        let mut processed_deps: BTreeSet<(String, Version, usize)> = BTreeSet::new();

        loop {
            let mut changed = false;

            while let Some((pkg, feature)) = queue.pop() {
                if c.features.entry(pkg.clone()).or_default().insert(feature.clone()) {
                    changed = true;
                }
                let Some(&version) = assigned.get(&pkg) else { continue };
                if !processed_features.insert((pkg.clone(), version, feature.clone())) { continue; }
                let Some(release) = self.release(&pkg, version) else { continue };
                if let Some(decl) = release.features.iter().find(|f| f.name == feature) {
                    for enabled in &decl.enables { queue.push((pkg.clone(), enabled.clone())); }
                }
            }

            let selected: Vec<(String, Version)> = assigned.iter().map(|(n, v)| (n.clone(), *v)).collect();
            for (pkg, version) in selected {
                let Some(release) = self.release(&pkg, version) else { continue };
                for (j, dep) in release.deps.iter().enumerate() {
                    if !self.platform_matches(dep.platform) { continue; }
                    if dep.gate.as_ref().is_some_and(|g| !c.features.get(&pkg).is_some_and(|fs| fs.contains(g))) { continue; }
                    if !processed_deps.insert((pkg.clone(), version, j)) { continue; }
                    changed = true;
                    let mut fs = dep.features.clone();
                    fs.sort(); fs.dedup();
                    let gate = dep.gate.as_deref().unwrap_or("*");
                    let plat = match dep.platform { Platform::Any => "any", Platform::Linux => "linux", Platform::Mac => "mac", Platform::Windows => "windows" };
                    let token = format!("{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        pkg, version_text(version), j, dep.package, constraint_text(dep.constraint),
                        fs.join(","), gate, plat);

                    let mut path = c.paths.get(&pkg).cloned().unwrap_or_default();
                    path.push(token);
                    c.constraints.entry(dep.package.clone()).or_default().push(dep.constraint);
                    c.origins.entry(dep.package.clone()).or_default().push(Origin { constraint: dep.constraint, path: path.clone() });
                    c.paths.entry(dep.package.clone()).and_modify(|old| {
                        if path.len() < old.len() || (path.len() == old.len() && path < *old) { *old = path.clone(); }
                    }).or_insert(path);
                    for f in fs { queue.push((dep.package.clone(), f)); }
                }
            }

            if queue.is_empty() && !changed { break; }
            if queue.is_empty() {
                // A newly found path or edge can affect downstream canonical provenance.
                // Revisit all active dependency edges until the closure stabilizes.
                let mut any = false;
                for (pkg, version) in assigned.iter() {
                    let Some(release) = self.release(pkg, *version) else { continue };
                    for (j, dep) in release.deps.iter().enumerate() {
                        if !self.platform_matches(dep.platform) { continue; }
                        if dep.gate.as_ref().is_some_and(|g| !c.features.get(pkg).is_some_and(|fs| fs.contains(g))) { continue; }
                        let gate = dep.gate.as_deref().unwrap_or("*");
                        let plat = match dep.platform { Platform::Any => "any", Platform::Linux => "linux", Platform::Mac => "mac", Platform::Windows => "windows" };
                        let mut path = c.paths.get(pkg).cloned().unwrap_or_default();
                        let mut fs = dep.features.clone(); fs.sort(); fs.dedup();
                        path.push(format!("{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                            pkg, version_text(*version), j, dep.package, constraint_text(dep.constraint), fs.join(","), gate, plat));
                        let old = c.paths.get(&dep.package).cloned();
                        if old.as_ref().is_none_or(|o| path.len() < o.len() || (path.len() == o.len() && path < *o)) {
                            c.paths.insert(dep.package.clone(), path);
                            any = true;
                        }
                    }
                }
                if !any { break; }
            }
        }

        for cs in c.constraints.values_mut() { *cs = canonical(std::mem::take(cs)); }
        c
    }

    fn active_origins(&self, assigned: &BTreeMap<String, Version>, c: &Closure) -> BTreeMap<String, Vec<Origin>> {
        let mut out: BTreeMap<String, Vec<Origin>> = BTreeMap::new();
        for (i, root) in self.roots.iter().enumerate() {
            let mut fs = root.features.clone(); fs.sort(); fs.dedup();
            let token = format!("root[{i}]:{}@{}+{{{}}}", root.package, constraint_text(root.constraint), fs.join(","));
            out.entry(root.package.clone()).or_default().push(Origin { constraint: root.constraint, path: vec![token] });
        }
        for (pkg, version) in assigned {
            let Some(release) = self.release(pkg, *version) else { continue };
            for (j, dep) in release.deps.iter().enumerate() {
                if !self.platform_matches(dep.platform) { continue; }
                if dep.gate.as_ref().is_some_and(|g| !c.features.get(pkg).is_some_and(|fs| fs.contains(g))) { continue; }
                let mut fs = dep.features.clone(); fs.sort(); fs.dedup();
                let gate = dep.gate.as_deref().unwrap_or("*");
                let plat = match dep.platform { Platform::Any => "any", Platform::Linux => "linux", Platform::Mac => "mac", Platform::Windows => "windows" };
                let mut path = c.paths.get(pkg).cloned().unwrap_or_default();
                path.push(format!("{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                    pkg, version_text(*version), j, dep.package, constraint_text(dep.constraint), fs.join(","), gate, plat));
                out.entry(dep.package.clone()).or_default().push(Origin { constraint: dep.constraint, path });
            }
        }
        out
    }

    fn dynamic_error(&self, assigned: &BTreeMap<String, Version>, c: &Closure) -> Option<ResolveError> {
        for name in c.constraints.keys() {
            if !self.packages.contains_key(name) {
                return Some(ResolveError::Missing { package: name.clone(), evidence: self.evidence(name, c, assigned) });
            }
        }

        for (pkg, version) in assigned {
            let Some(features) = c.features.get(pkg) else { continue };
            for feature in features {
                let declared = self.release(pkg, *version).is_some_and(|r| r.features.iter().any(|f| f.name == *feature));
                if !declared {
                    return Some(ResolveError::MissingFeature {
                        package: pkg.clone(), selected: *version, feature: feature.clone(),
                        path: c.paths.get(pkg).cloned().unwrap_or_default(),
                    });
                }
            }
        }

        for (pkg, version) in assigned {
            let reqs = c.constraints.get(pkg).cloned().unwrap_or_default();
            if reqs.iter().any(|r| !matches(*version, *r)) {
                return Some(ResolveError::Conflict {
                    package: pkg.clone(), selected: *version, requirements: reqs,
                    evidence: self.evidence(pkg, c, assigned),
                });
            }
        }

        for (pkg, reqs) in &c.constraints {
            if assigned.contains_key(pkg) { continue; }
            let has = self.packages[*pkg].releases.iter().any(|r| reqs.iter().all(|c| matches(r.version, *c)));
            if !has {
                return Some(ResolveError::NoMatch {
                    package: pkg.clone(), requirements: reqs.clone(),
                    evidence: self.evidence(pkg, c, assigned),
                });
            }
        }
        None
    }

    fn evidence(&self, pkg: &str, c: &Closure, assigned: &BTreeMap<String, Version>) -> Vec<Evidence> {
        self.active_origins(assigned, c).remove(pkg).unwrap_or_default().into_iter()
            .map(|o| Evidence { constraint: o.constraint, path: o.path }).collect()
    }

    fn score(&self, assigned: &BTreeMap<String, Version>) -> usize {
        self.locks.iter().filter(|(p, v)| assigned.get(*p) != Some(*v)).count()
    }

    fn better(&self, score: usize, assigned: &BTreeMap<String, Version>) -> bool {
        let Some((best_score, best, _)) = &self.best else { return true };
        if score != *best_score { return score < *best_score; }
        for name in &self.catalog_names {
            let a = assigned.get(name);
            let b = best.iter().find(|s| &s.package == name).map(|s| &s.version);
            match (a, b) {
                (None, None) => {}
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(x), Some(y)) if x != y => return x > y,
                _ => {}
            }
        }
        false
    }

    fn search(&mut self, assigned: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = self.closure(assigned);
        if let Some(err) = self.dynamic_error(assigned, &closure) {
            if self.first_error.is_none() { self.first_error = Some(err); }
            return Ok(());
        }

        let next = closure.constraints.keys().find(|p| !assigned.contains_key(*p)).cloned();
        let Some(pkg) = next else {
            let score = self.score(assigned);
            if self.better(score, assigned) {
                let selected = assigned.iter().map(|(package, version)| Selection { package: package.clone(), version: *version }).collect();
                self.best = Some((score, selected, closure));
            }
            return Ok(());
        };

        let mut candidates: Vec<Version> = self.packages[&pkg].releases.iter()
            .filter(|r| closure.constraints[&pkg].iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version).collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(lock) = self.locks.get(&pkg) {
            if let Some(i) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit { attempts: self.max_attempts });
            }
            self.attempts += 1;
            assigned.insert(pkg.clone(), candidate);
            let result = self.search(assigned);
            assigned.remove(&pkg);
            result?;
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName { site: format!("root[{i}]"), name: r.package.clone() });
        }
        for (f, name) in r.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName { site: format!("root[{i}].feature[{f}]"), name: name.clone() });
            }
        }
    }
    for (i, p) in catalog.iter().enumerate() {
        if !valid_name(&p.name) {
            return Err(ResolveError::InvalidName { site: format!("catalog[{i}]"), name: p.name.clone() });
        }
        for (j, r) in p.releases.iter().enumerate() {
            for (k, d) in r.deps.iter().enumerate() {
                if !valid_name(&d.package) {
                    return Err(ResolveError::InvalidName { site: format!("catalog[{i}].release[{j}].dep[{k}]"), name: d.package.clone() });
                }
                if let Some(g) = &d.gate {
                    if !valid_name(g) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{i}].release[{j}].dep[{k}].gate"), name: g.clone() });
                    }
                }
                for (f, name) in d.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{i}].release[{j}].dep[{k}].feature[{f}]"), name: name.clone() });
                    }
                }
            }
            for (f, feature) in r.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName { site: format!("catalog[{i}].release[{j}].feature[{f}]"), name: feature.name.clone() });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName { site: format!("catalog[{i}].release[{j}].feature[{f}].enable[{e}]"), name: name.clone() });
                    }
                }
            }
        }
    }

    let mut seen = BTreeSet::new();
    for p in catalog {
        if !seen.insert(p.name.clone()) { return Err(ResolveError::DuplicatePackage { package: p.name.clone() }); }
    }
    for p in catalog {
        let mut versions = BTreeSet::new();
        for r in &p.releases {
            if !versions.insert(r.version) {
                return Err(ResolveError::DuplicateVersion { package: p.name.clone(), version: r.version });
            }
        }
    }
    for p in catalog {
        for r in &p.releases {
            let mut features = BTreeSet::new();
            for f in &r.features {
                if !features.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature { package: p.name.clone(), version: r.version, feature: f.name.clone() });
                }
            }
        }
    }
    for (i, l) in locks.iter().enumerate() {
        if !valid_name(&l.package) {
            return Err(ResolveError::InvalidName { site: format!("lock[{i}]"), name: l.package.clone() });
        }
    }
    let mut lock_seen = BTreeSet::new();
    for l in locks {
        if !lock_seen.insert(l.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: l.package.clone() });
        }
    }

    let package_map: BTreeMap<String, &Package> = catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let locks_map: BTreeMap<String, Version> = locks.iter().map(|l| (l.package.clone(), l.version)).collect();
    let mut solver = Solver {
        packages: package_map,
        roots,
        locks: locks_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        catalog_names: catalog.iter().map(|p| p.name.clone()).collect(),
    };
    solver.catalog_names.sort();
    let mut assigned = BTreeMap::new();
    solver.search(&mut assigned)?;

    let Some((_, selected, closure)) = solver.best else {
        return Err(solver.first_error.unwrap_or(ResolveError::Missing { package: String::new(), evidence: vec![] }));
    };

    let after: BTreeMap<String, Version> = selected.iter().map(|s| (s.package.clone(), s.version)).collect();
    let mut names: BTreeSet<String> = solver.locks.keys().cloned().collect();
    names.extend(after.keys().cloned());
    let mut changes = Vec::new();
    for name in names {
        let before = solver.locks.get(&name).copied();
        let now = after.get(&name).copied();
        if before == now { continue; }
        let reason = match (before, now) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = solver.packages.get(&name).is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists { ChangeReason::Unavailable }
                else if closure.constraints.get(&name).is_some_and(|cs| cs.iter().any(|c| !matches(old, *c))) {
                    ChangeReason::Rejected
                } else { ChangeReason::Rebalanced }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change { package: name, before, after: now, reason });
    }
    Ok(Resolution { selected, changes })
}

            evidence,
        } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn satisfies(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn platform_matches(p: Platform, t: Target) -> bool {
    matches!(
        (p, t),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Info {
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    active_features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

struct Solver<'a> {
    catalog: &'a [Package],
    by_name: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    first_leaf: Option<ResolveError>,
    best: Option<(usize, BTreeMap<String, Version>)>,
}

impl<'a> Solver<'a> {
    fn selected_release(&self, name: &str, selected: &BTreeMap<String, Version>) -> Option<&'a Release> {
        let ver = selected.get(name)?;
        self.by_name
            .get(name)?
            .releases
            .iter()
            .find(|r| r.version == *ver)
    }

    fn closure(&self, selected: &BTreeMap<String, Version>) -> Info {
        let mut info = Info::default();

        for (i, root) in self.roots.iter().enumerate() {
            let mut fs = root.features.clone();
            fs.sort();
            fs.dedup();
            let token = format!(
                "root[{}]:{}@{}+{{{}}}",
                i,
                root.package,
                constraint_text(root.constraint),
                fs.join(",")
            );
            info.origins
                .entry(root.package.clone())
                .or_default()
                .push(Origin {
                    constraint: root.constraint,
                    path: vec![token],
                });
            info.requests
                .entry(root.package.clone())
                .or_default()
                .extend(root.features.iter().cloned());
        }

        loop {
            let mut changed = false;
            let names: Vec<String> = selected.keys().cloned().collect();
            for source in names {
                let Some(release) = self.selected_release(&source, selected) else {
                    continue;
                };
                let reqs = info.requests.get(&source).cloned().unwrap_or_default();
                let mut active = info.active_features.get(&source).cloned().unwrap_or_default();

                loop {
                    let before = active.len();
                    let known: BTreeMap<String, &Feature> = release
                        .features
                        .iter()
                        .map(|f| (f.name.clone(), f))
                        .collect();
                    let current: Vec<String> = active.iter().cloned().collect();
                    for f in current {
                        if let Some(decl) = known.get(&f) {
                            active.extend(decl.enables.iter().cloned());
                        }
                    }
                    if active.len() == before {
                        break;
                    }
                }

                if !reqs.is_subset(&active) {
                    active.extend(reqs);
                    loop {
                        let before = active.len();
                        let known: BTreeMap<String, &Feature> = release
                            .features
                            .iter()
                            .map(|f| (f.name.clone(), f))
                            .collect();
                        let current: Vec<String> = active.iter().cloned().collect();
                        for f in current {
                            if let Some(decl) = known.get(&f) {
                                active.extend(decl.enables.iter().cloned());
                            }
                        }
                        if active.len() == before {
                            break;
                        }
                    }
                }

                if info.active_features.get(&source) != Some(&active) {
                    info.active_features.insert(source.clone(), active.clone());
                    changed = true;
                }

                for (j, dep) in release.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, self.target)
                        || dep.gate.as_ref().is_some_and(|g| !active.contains(g))
                    {
                        continue;
                    }
                    let mut fs = dep.features.clone();
                    fs.sort();
                    fs.dedup();
                    let gate = dep.gate.as_deref().unwrap_or("*");
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        source,
                        version_text(release.version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        fs.join(","),
                        gate,
                        match dep.platform {
                            Platform::Any => "any",
                            Platform::Linux => "linux",
                            Platform::Mac => "mac",
                            Platform::Windows => "windows",
                        }
                    );

                    let source_path = info.paths.get(&source).cloned().unwrap_or_default();
                    let mut path = source_path;
                    path.push(token);

                    let origin = Origin {
                        constraint: dep.constraint,
                        path,
                    };
                    let origins = info.origins.entry(dep.package.clone()).or_default();
                    if !origins.iter().any(|x| x.constraint == origin.constraint && x.path == origin.path) {
                        origins.push(origin);
                        changed = true;
                    }
                    let dest = info.requests.entry(dep.package.clone()).or_default();
                    let old_len = dest.len();
                    dest.extend(dep.features.iter().cloned());
                    if dest.len() != old_len {
                        changed = true;
                    }
                }
            }

            let mut best_paths: BTreeMap<String, Vec<String>> = BTreeMap::new();
            for os in info.origins.values() {
                for origin in os {
                    if origin.path.len() == 1 {
                        let package = info
                            .origins
                            .iter()
                            .find(|(_, origins)| origins.iter().any(|o| o.path == origin.path && o.constraint == origin.constraint))
                            .map(|(name, _)| name.clone());
                        if let Some(name) = package {
                            choose_path(&mut best_paths, name, origin.path.clone());
                        }
                    }
                }
            }
            for (source, release) in selected.iter().filter_map(|(n, v)| {
                self.selected_release(n, selected).map(|r| (n.clone(), r))
            }) {
                let active = info.active_features.get(&source).cloned().unwrap_or_default();
                let path = info.paths.get(&source).cloned().unwrap_or_default();
                for (j, dep) in release.deps.iter().enumerate() {
                    if !platform_matches(dep.platform, self.target)
                        || dep.gate.as_ref().is_some_and(|g| !active.contains(g))
                    {
                        continue;
                    }
                    if path.is_empty() {
                        continue;
                    }
                    let mut fs = dep.features.clone();
                    fs.sort();
                    fs.dedup();
                    let token = format!(
                        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
                        source,
                        version_text(release.version),
                        j,
                        dep.package,
                        constraint_text(dep.constraint),
                        fs.join(","),
                        dep.gate.as_deref().unwrap_or("*"),
                        match dep.platform {
                            Platform::Any => "any",
                            Platform::Linux => "linux",
                            Platform::Mac => "mac",
                            Platform::Windows => "windows",
                        }
                    );
                    let mut p = path.clone();
                    p.push(token);
                    choose_path(&mut best_paths, dep.package.clone(), p);
                }
            }
            if info.paths != best_paths {
                info.paths = best_paths;
                changed = true;
            }
            if !changed {
                break;
            }
        }
        info
    }

    fn dynamic_error(
        &self,
        selected: &BTreeMap<String, Version>,
        info: &Info,
    ) -> Option<ResolveError> {
        let active: BTreeSet<String> = info.origins.keys().cloned().collect();

        for name in &active {
            if !self.by_name.contains_key(name) {
                return Some(ResolveError::Missing {
                    package: name.clone(),
                    evidence: make_evidence(info, name),
                });
            }
        }

        for (name, version) in selected {
            let constraints: Vec<Constraint> = info
                .origins
                .get(name)
                .into_iter()
                .flatten()
                .map(|x| x.constraint)
                .collect();
            if !constraints.iter().all(|c| satisfies(*version, *c)) {
                return Some(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: canonical_constraints(&constraints),
                    evidence: make_evidence(info, name),
                });
            }
        }

        for name in active.iter().filter(|n| !selected.contains_key(*n)) {
            let constraints: Vec<Constraint> = info
                .origins
                .get(name)
                .into_iter()
                .flatten()
                .map(|x| x.constraint)
                .collect();
            let has_candidate = self.by_name[*name].releases.iter().any(|r| {
                constraints.iter().all(|c| satisfies(r.version, *c))
            });
            if !has_candidate {
                return Some(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: canonical_constraints(&constraints),
                    evidence: make_evidence(info, name),
                });
            }
        }

        for name in selected.keys() {
            let Some(release) = self.selected_release(name, selected) else {
                continue;
            };
            let active_features = info.active_features.get(name).cloned().unwrap_or_default();
            for feature in active_features {
                if !release.features.iter().any(|f| f.name == feature) {
                    return Some(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: release.version,
                        feature,
                        path: info.paths.get(name).cloned().unwrap_or_default(),
                    });
                }
            }
        }
        None
    }

    fn search(&mut self, selected: BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let info = self.closure(&selected);
        if let Some(error) = self.dynamic_error(&selected, &info) {
            if matches!(error, ResolveError::Limit { .. }) {
                return Err(error);
            }
            if self.first_leaf.is_none() {
                self.first_leaf = Some(error);
            }
            return Ok(());
        }

        let active: BTreeSet<String> = info.origins.keys().cloned().collect();
        let next = active.iter().find(|n| !selected.contains_key(*n)).cloned();
        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_sel)) => {
                    if cost != *best_cost {
                        cost < *best_cost
                    } else {
                        self.vector_cmp(&selected, best_sel).is_gt()
                    }
                }
            };
            if replace {
                self.best = Some((cost, selected));
            }
            return Ok(());
        };

        let constraints: Vec<Constraint> = info
            .origins
            .get(&name)
            .into_iter()
            .flatten()
            .map(|x| x.constraint)
            .collect();
        let mut candidates: Vec<Version> = self.by_name[&name]
            .releases
            .iter()
            .filter(|r| constraints.iter().all(|c| satisfies(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();

        if let Some(lock) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            let mut child = selected.clone();
            child.insert(name.clone(), candidate);
            self.search(child)?;
        }
        Ok(())
    }

    fn vector_cmp(
        &self,
        a: &BTreeMap<String, Version>,
        b: &BTreeMap<String, Version>,
    ) -> std::cmp::Ordering {
        let names: BTreeSet<String> = self
            .catalog
            .iter()
            .map(|p| p.name.clone())
            .collect();
        for n in names {
            match (a.get(&n), b.get(&n)) {
                (None, None) => {}
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (Some(x), Some(y)) => {
                    let c = x.cmp(y);
                    if !c.is_eq() {
                        return c;
                    }
                }
            }
        }
        std::cmp::Ordering::Equal
    }
}

fn choose_path(paths: &mut BTreeMap<String, Vec<String>>, name: String, path: Vec<String>) {
    match paths.get(&name) {
        None => {
            paths.insert(name, path);
        }
        Some(old) if (path.len(), &path) < (old.len(), old) => {
            paths.insert(name, path);
        }
        _ => {}
    }
}

fn make_evidence(info: &Info, package: &str) -> Vec<Evidence> {
    let mut out = info
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    out
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, name) in root.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f),
                            name: name.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, name) in feature.enables.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: name.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen_features = BTreeSet::new();
            for feature in &release.features {
                if !seen_features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if lock_map.insert(lock.package.clone(), lock.version).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let by_name: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let mut solver = Solver {
        catalog,
        by_name,
        roots,
        locks: lock_map.clone(),
        target,
        max_attempts,
        attempts: 0,
        first_leaf: None,
        best: None,
    };

    solver.search(BTreeMap::new())?;

    let Some((_, chosen)) = solver.best else {
        return Err(solver.first_leaf.unwrap_or(ResolveError::Missing {
            package: String::new(),
            evidence: Vec::new(),
        }));
    };

    let selected = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let final_info = solver.closure(&chosen);
    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = by_name
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints: Vec<Constraint> = final_info
                        .origins
                        .get(&package)
                        .into_iter()
                        .flatten()
                        .map(|o| o.constraint)
                        .collect();
                    if !constraints.iter().all(|c| satisfies(old, *c)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter().all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(ev: &[Evidence]) -> Vec<Evidence> {
    let mut out = ev.to_vec();
    out.sort_by(|a, b| a.constraint.cmp(&b.constraint).then_with(|| a.path.cmp(&b.path)));
    out
}

fn append_evidence(text: &mut String, ev: &[Evidence]) {
    for e in evidence_sorted(ev) {
        text.push_str("\n  C:");
        text.push_str(&constraint_text(e.constraint));
        text.push_str(" via ");
        text.push_str(&e.path.join(" -> "));
    }
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature { package, version, feature } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => {
            let mut s = format!("missing|package={}", encode(package));
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::NoMatch { package, requirements, evidence } => {
            let mut s = format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::Conflict { package, selected, requirements, evidence } => {
            let mut s = format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            );
            append_evidence(&mut s, evidence);
            s
        }
        ResolveError::MissingFeature { package, selected, feature, path } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone, Default)]
struct NodeData {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
    active_deps: BTreeMap<String, Vec<usize>>,
    paths: BTreeMap<String, Vec<String>>,
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut fs = r.features.clone();
    fs.sort();
    fs.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        fs.join(",")
    )
}

fn platform_text(p: Platform) -> &'static str {
    match p {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    }
}

fn target_matches(p: Platform, t: Target) -> bool {
    matches!(p, Platform::Any)
        || matches!(
            (p, t),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        {
            let mut fs = dep.features.clone();
            fs.sort();
            fs.dedup();
            fs.join(",")
        },
        dep.gate.as_deref().unwrap_or("*"),
        platform_text(dep.platform)
    )
}

fn append_path(source: &[String], token: String) -> Vec<String> {
    let mut p = source.to_vec();
    p.push(token);
    p
}

fn compute_node(
    catalog: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> NodeData {
    let mut data = NodeData::default();
    let mut pending: Vec<(String, String)> = Vec::new();

    for (i, r) in roots.iter().enumerate() {
        data.constraints.entry(r.package.clone()).or_default().push(r.constraint);
        data.origins.entry(r.package.clone()).or_default().push(Origin {
            constraint: r.constraint,
            path: vec![root_token(i, r)],
        });
        data.requests
            .entry(r.package.clone())
            .or_default()
            .extend(r.features.iter().cloned());
        data.paths
            .entry(r.package.clone())
            .or_insert_with(|| vec![root_token(i, r)]);
        for f in &r.features {
            pending.push((r.package.clone(), f.clone()));
        }
    }

    let mut processed_features: BTreeSet<(String, String)> = BTreeSet::new();
    let mut processed_deps: BTreeSet<(String, usize)> = BTreeSet::new();

    loop {
        let mut changed = false;

        let packages: Vec<(String, Version)> = selected
            .iter()
            .map(|(p, v)| (p.clone(), *v))
            .collect();

        for (source, version) in packages {
            let Some(pkg) = catalog.get(&source) else { continue };
            let Some(release) = pkg.releases.iter().find(|r| r.version == version) else { continue };

            let requested: Vec<String> = data
                .requests
                .get(&source)
                .map(|s| s.iter().cloned().collect())
                .unwrap_or_default();

            for f in requested {
                if processed_features.insert((source.clone(), f.clone())) {
                    data.activated.entry(source.clone()).or_default().insert(f.clone());
                    if let Some(feature) = release.features.iter().find(|x| x.name == f) {
                        for enabled in &feature.enables {
                            if data
                                .requests
                                .entry(source.clone())
                                .or_default()
                                .insert(enabled.clone())
                            {
                                changed = true;
                            }
                        }
                    }
                    changed = true;
                }
            }

            for (j, dep) in release.deps.iter().enumerate() {
                if !target_matches(dep.platform, target) {
                    continue;
                }
                if let Some(gate) = &dep.gate {
                    if !data
                        .activated
                        .get(&source)
                        .map(|s| s.contains(gate))
                        .unwrap_or(false)
                    {
                        continue;
                    }
                }
                if !processed_deps.insert((source.clone(), j)) {
                    continue;
                }

                data.active_deps.entry(source.clone()).or_default().push(j);
                let token = dep_token(&source, version, j, dep);
                let path = data.paths.get(&source).cloned().unwrap_or_default();
                let origin_path = append_path(&path, token);
                data.constraints.entry(dep.package.clone()).or_default().push(dep.constraint);
                data.origins.entry(dep.package.clone()).or_default().push(Origin {
                    constraint: dep.constraint,
                    path: origin_path.clone(),
                });

                let new_path = match data.paths.get(&dep.package) {
                    None => Some(origin_path.clone()),
                    Some(old) if (origin_path.len(), &origin_path) < (old.len(), old) => {
                        Some(origin_path.clone())
                    }
                    _ => None,
                };
                if let Some(p) = new_path {
                    data.paths.insert(dep.package.clone(), p);
                    changed = true;
                }

                for f in &dep.features {
                    if data
                        .requests
                        .entry(dep.package.clone())
                        .or_default()
                        .insert(f.clone())
                    {
                        changed = true;
                    }
                }
                changed = true;
            }
        }
        if !changed {
            break;
        }
    }

    for origins in data.origins.values_mut() {
        origins.sort_by(|a, b| {
            a.constraint
                .cmp(&b.constraint)
                .then_with(|| a.path.cmp(&b.path))
        });
    }
    data
}

fn canonical_from_origins(origins: &[Origin]) -> Vec<Evidence> {
    let mut cs: Vec<Constraint> = origins.iter().map(|o| o.constraint).collect();
    cs.sort();
    cs.dedup();
    cs
}

fn error_for_node(
    catalog: &BTreeMap<String, &Package>,
    selected: &BTreeMap<String, Version>,
    data: &NodeData,
) -> Option<ResolveError> {
    let active: BTreeSet<String> = data.constraints.keys().cloned().collect();

    for name in &active {
        if !catalog.contains_key(name) {
            let evidence = data.origins.get(name).cloned().unwrap_or_default();
            return Some(ResolveError::Missing {
                package: name.clone(),
                evidence: evidence
                    .iter()
                    .map(|o| Evidence { constraint: o.constraint, path: o.path.clone() })
                    .collect(),
            });
        }
    }

    for (name, version) in selected {
        if !active.contains(name) {
            continue;
        }
        let cs = canonical_constraints(data.constraints.get(name).map(Vec::as_slice).unwrap_or(&[]));
        if !cs.iter().all(|c| matches(*c, *version)) {
            let evidence = data.origins.get(name).cloned().unwrap_or_default();
            return Some(ResolveError::Conflict {
                package: name.clone(),
                selected: *version,
                requirements: cs,
                evidence: evidence
                    .iter()
                    .map(|o| Evidence { constraint: o.constraint, path: o.path.clone() })
                    .collect(),
            });
        }
    }

    for name in &active {
        if selected.contains_key(name) {
            continue;
        }
        let pkg = catalog.get(name).unwrap();
        let cs = canonical_constraints(data.constraints.get(name).map(Vec::as_slice).unwrap_or(&[]));
        if !pkg.releases.iter().any(|r| cs.iter().all(|c| matches(*c, r.version))) {
            let evidence = data.origins.get(name).cloned().unwrap_or_default();
            return Some(ResolveError::NoMatch {
                package: name.clone(),
                requirements: cs,
                evidence: evidence
                    .iter()
                    .map(|o| Evidence { constraint: o.constraint, path: o.path.clone() })
                    .collect(),
            });
        }
    }

    for (name, version) in selected {
        let Some(reqs) = data.activated.get(name) else { continue };
        let Some(pkg) = catalog.get(name) else { continue };
        let Some(release) = pkg.releases.iter().find(|r| r.version == *version) else { continue };
        for feature in reqs {
            if !release.features.iter().any(|f| f.name == *feature) {
                return Some(ResolveError::MissingFeature {
                    package: name.clone(),
                    selected: *version,
                    feature: feature.clone(),
                    path: data.paths.get(name).cloned().unwrap_or_default(),
                });
            }
        }
    }
    None
}

struct Search<'a> {
    catalog: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<BTreeMap<String, Version>>,
    best_cost: usize,
    first_failure: Option<ResolveError>,
}

impl Search<'_> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let data = compute_node(self.catalog, self.roots, selected, self.target);
        if let Some(err) = error_for_node(self.catalog, selected, &data) {
            if self.first_failure.is_none() {
                self.first_failure = Some(err);
            }
            return Ok(());
        }

        let active: BTreeSet<String> = data.constraints.keys().cloned().collect();
        if active.iter().all(|p| selected.contains_key(p)) {
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let replace = match &self.best {
                None => true,
                Some(best) if cost < self.best_cost => true,
                Some(best) if cost == self.best_cost => {
                    let names: Vec<String> = self.catalog.keys().cloned().collect();
                    names.into_iter().find_map(|p| {
                        match (selected.get(&p), best.get(&p)) {
                            (None, None) => None,
                            (Some(_), None) => Some(true),
                            (None, Some(_)) => Some(false),
                            (Some(a), Some(b)) if a != b => Some(a > b),
                            _ => None,
                        }
                    }).unwrap_or(false)
                }
                _ => false,
            };
            if replace {
                self.best = Some(selected.clone());
                self.best_cost = cost;
            }
            return Ok(());
        }

        let package = active.iter().find(|p| !selected.contains_key(*p)).unwrap().clone();
        let pkg = self.catalog.get(&package).unwrap();
        let cs = data.constraints.get(&package).map(Vec::as_slice).unwrap_or(&[]);
        let mut candidates: Vec<Version> = pkg
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| cs.iter().all(|c| matches(*c, *v)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.get(&package) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit { attempts: self.max_attempts });
            }
            self.attempts += 1;
            selected.insert(package.clone(), candidate);
            self.visit(selected)?;
            selected.remove(&package);
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, r) in roots.iter().enumerate() {
        if !valid_name(&r.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: r.package.clone(),
            });
        }
        for (f, name) in r.features.iter().enumerate() {
            if !valid_name(name) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: name.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName { site: base, name: dep.package.clone() });
                }
                if let Some(g) = &dep.gate {
                    if !valid_name(g) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: g.clone(),
                        });
                    }
                }
                for (f, name) in dep.features.iter().enumerate() {
                    if !valid_name(name) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].feature[{}]", i, j, k, f),
                            name: name.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for p in catalog {
        if !seen_packages.insert(p.name.clone()) {
            return Err(ResolveError::DuplicatePackage { package: p.name.clone() });
        }
    }

    for p in catalog {
        let mut seen = BTreeSet::new();
        for r in &p.releases {
            if !seen.insert(r.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: p.name.clone(),
                    version: r.version,
                });
            }
        }
    }

    for p in catalog {
        for r in &p.releases {
            let mut seen = BTreeSet::new();
            for f in &r.features {
                if !seen.insert(f.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: p.name.clone(),
                        version: r.version,
                        feature: f.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut lock_names = BTreeSet::new();
    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock { package: lock.package.clone() });
        }
        lock_map.insert(lock.package.clone(), lock.version);
    }

    let catalog_map: BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();

    let mut search = Search {
        catalog: &catalog_map,
        roots,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        best_cost: usize::MAX,
        first_failure: None,
    };
    search.visit(&mut BTreeMap::new())?;

    let selected = if let Some(best) = search.best {
        best
    } else {
        return Err(search.first_failure.unwrap_or_else(|| ResolveError::NoMatch {
            package: String::new(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let final_data = compute_node(&catalog_map, roots, &selected, target);
    let selections: Vec<Selection> = selected
        .iter()
        .map(|(package, version)| Selection { package: package.clone(), version: *version })
        .collect();

    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(selected.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = selected.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&package)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let cs = final_data.constraints.get(&package).map(Vec::as_slice).unwrap_or(&[]);
                    if !cs.iter().all(|c| matches(*c, old)) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change { package, before, after, reason });
    }

    Ok(Resolution { selected: selections, changes })
}
