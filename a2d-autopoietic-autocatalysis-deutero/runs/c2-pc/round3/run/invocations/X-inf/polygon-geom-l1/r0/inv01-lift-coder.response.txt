#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
}

const MAX_VERTICES: usize = 4096;
const COORD_LIMIT: i64 = 1_000_000_000;
const SCALE2: u128 = 1_000_000;

fn is_valid_point(point: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&point.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&point.y)
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = (i128::from(a.x) - i128::from(b.x)) as u128;
    let dy = (i128::from(a.y) - i128::from(b.y)) as u128;
    let squared_scaled = (dx * dx + dy * dy) * SCALE2;

    // Integer square root, rounded to the nearest integer with half ties upward.
    let mut low = 0_u128;
    let mut high = squared_scaled + 1;
    while low + 1 < high {
        let mid = low + (high - low) / 2;
        if mid <= squared_scaled / mid {
            low = mid;
        } else {
            high = mid;
        }
    }

    let lower_square = low * low;
    let remainder = squared_scaled - lower_square;
    if remainder >= low + 1 {
        (low + 1) as u64
    } else {
        low as u64
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }

    for (index, &point) in points.iter().enumerate() {
        if !is_valid_point(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let effective_len = points.len()
        - usize::from(points.len() >= 2 && points[points.len() - 1] == points[0]);
    if effective_len < 3 {
        return Err(GeomError::TooFewVertices { len: effective_len });
    }

    let polygon = &points[..effective_len];
    let mut signed_area2 = 0_i128;
    let mut edge_milli = Vec::with_capacity(effective_len);
    let mut perimeter_milli = 0_u64;
    let mut longest_edge: Option<usize> = None;

    for index in 0..effective_len {
        let a = polygon[index];
        let b = polygon[(index + 1) % effective_len];

        signed_area2 += i128::from(a.x) * i128::from(b.y)
            - i128::from(a.y) * i128::from(b.x);

        let length = rounded_edge_milli(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace_longest = match longest_edge {
            None => true,
            Some(current) if length > edge_milli[current] => true,
            Some(current) if length < edge_milli[current] => false,
            Some(current) => {
                let endpoints = if a <= b { (a, b) } else { (b, a) };
                let current_a = polygon[current];
                let current_b = polygon[(current + 1) % effective_len];
                let current_endpoints = if current_a <= current_b {
                    (current_a, current_b)
                } else {
                    (current_b, current_a)
                };
                endpoints < current_endpoints
                    || (endpoints == current_endpoints && index < current)
            }
        };
        if replace_longest {
            longest_edge = Some(index);
        }
    }

    let orientation = match signed_area2 {
        value if value > 0 => Orientation::CounterClockwise,
        value if value < 0 => Orientation::Clockwise,
        _ => Orientation::Degenerate,
    };
    let area2 = signed_area2.unsigned_abs();

    Ok(Measures {
        signed_area2,
        area2,
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}
