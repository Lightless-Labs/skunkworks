```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(component.to_owned()),
            }
        }
        Ok(parts)
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_owned).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn parent_path(parts: &[String]) -> String {
        Self::canonical(&parts[..parts.len() - 1])
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn is_within(path: &str, directory: &str) -> bool {
        if directory == "/" {
            return true;
        }
        path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn lookup_parent(&self, parts: &[String]) -> Result<String, Error> {
        let mut prefix = Vec::new();
        for component in &parts[..parts.len().saturating_sub(1)] {
            prefix.push(component.clone());
            let path = Self::canonical(&prefix);
            match self.entries.get(&path) {
                None => return Err(Error::Missing(path)),
                Some(Entry::File(_)) => return Err(Error::NotDir(path)),
                Some(Entry::Dir) => {}
            }
        }
        Ok(Self::canonical(&prefix))
    }

    fn resolve_existing(&self, parts: &[String]) -> Result<(String, Entry), Error> {
        if parts.is_empty() {
            return Ok(("/".to_owned(), Entry::Dir));
        }

        let parent = self.lookup_parent(parts)?;
        let path = Self::child_path(&parent, parts.last().unwrap());
        match self.entries.get(&path) {
            Some(entry) => Ok((path, entry.clone())),
            None => Err(Error::Missing(path)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        if parts.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }
        let parent = self.lookup_parent(&parts)?;
        let target = Self::child_path(&parent, parts.last().unwrap());
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        if parts.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }
        let parent = self.lookup_parent(&parts)?;
        let target = Self::child_path(&parent, parts.last().unwrap());
        match self.entries.get(&target) {
            Some(Entry::Dir) => Err(Error::IsDir(target)),
            Some(Entry::File(_)) | None => {
                self.entries
                    .insert(target, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = self.normalize(path)?;
        let (target, entry) = self.resolve_existing(&parts)?;
        match entry {
            Entry::Dir => Err(Error::IsDir(target)),
            Entry::File(data) => Ok(data),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = self.normalize(path)?;
        let (target, entry) = self.resolve_existing(&parts)?;
        if !matches!(entry, Entry::Dir) {
            return Err(Error::NotDir(target));
        }

        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == "/" {
                continue;
            }
            let entry_parts = Self::components(entry_path);
            if entry_parts.len() == parts.len() + 1 && entry_parts[..parts.len()] == parts {
                children.push((entry_parts.last().unwrap().clone(), entry.kind()));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        let (target, entry) = self.resolve_existing(&parts)?;
        if !matches!(entry, Entry::Dir) {
            return Err(Error::NotDir(target));
        }
        self.cwd = target;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source_parts = self.normalize(from)?;
        let destination_parts = self.normalize(to)?;
        let source = Self::canonical(&source_parts);
        let destination = Self::canonical(&destination_parts);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let source_parent = self.lookup_parent(&source_parts)?;
        let source = Self::child_path(&source_parent, source_parts.last().unwrap());
        let source_entry = match self.entries.get(&source) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(source)),
        };

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let destination_parent = self.lookup_parent(&destination_parts)?;
        let destination = Self::child_path(
            &destination_parent,
            destination_parts.last().unwrap(),
        );

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref dest_entry) = destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(destination)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(destination)),
                (Entry::Dir, Entry::Dir) => {
                    let has_children = self.entries.keys().any(|path| {
                        path != &destination && Self::is_within(path, &destination)
                    });
                    if has_children {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| *path == &source || Self::is_within(path, &source))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if let Some(ref dest_entry) = destination_entry {
            if matches!(dest_entry, Entry::Dir) {
                self.entries.retain(|path, _| {
                    path != &destination && !Self::is_within(path, &destination)
                });
            } else {
                self.entries.remove(&destination);
            }
        }

        for (path, _) in &moving {
            self.entries.remove(path);
        }
        for (old_path, entry) in moving {
            let suffix = &old_path[source.len()..];
            self.entries.insert(format!("{destination}{suffix}"), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        if parts.is_empty() {
            return Err(Error::Root);
        }

        let parent = self.lookup_parent(&parts)?;
        let target = Self::child_path(&parent, parts.last().unwrap());
        let entry = match self.entries.get(&target) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(target)),
        };

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }

        let has_children = matches!(entry, Entry::Dir)
            && self
                .entries
                .keys()
                .any(|path| path != &target && Self::is_within(path, &target));
        if has_children && !recursive {
            return Err(Error::NotEmpty(target));
        }

        self.entries.retain(|path, _| {
            path != &target && !(recursive && Self::is_within(path, &target))
        });
        Ok(())
    }
}
```
