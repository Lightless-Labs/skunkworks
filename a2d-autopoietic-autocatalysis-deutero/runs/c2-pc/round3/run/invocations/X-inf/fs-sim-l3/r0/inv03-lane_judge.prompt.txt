use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(component.to_owned()),
            }
        }
        Ok(parts)
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_owned).collect()
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn parent_path(parts: &[String]) -> String {
        Self::canonical(&parts[..parts.len() - 1])
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn is_within(path: &str, directory: &str) -> bool {
        if directory == "/" {
            return true;
        }
        path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn lookup_parent(&self, parts: &[String]) -> Result<String, Error> {
        let mut prefix = Vec::new();
        for component in &parts[..parts.len().saturating_sub(1)] {
            prefix.push(component.clone());
            let path = Self::canonical(&prefix);
            match self.entries.get(&path) {
                None => return Err(Error::Missing(path)),
                Some(Entry::File(_)) => return Err(Error::NotDir(path)),
                Some(Entry::Dir) => {}
            }
        }
        Ok(Self::canonical(&prefix))
    }

    fn resolve_existing(&self, parts: &[String]) -> Result<(String, Entry), Error> {
        if parts.is_empty() {
            return Ok(("/".to_owned(), Entry::Dir));
        }

        let parent = self.lookup_parent(parts)?;
        let path = Self::child_path(&parent, parts.last().unwrap());
        match self.entries.get(&path) {
            Some(entry) => Ok((path, entry.clone())),
            None => Err(Error::Missing(path)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        if parts.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }
        let parent = self.lookup_parent(&parts)?;
        let target = Self::child_path(&parent, parts.last().unwrap());
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        if parts.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }
        let parent = self.lookup_parent(&parts)?;
        let target = Self::child_path(&parent, parts.last().unwrap());
        match self.entries.get(&target) {
            Some(Entry::Dir) => Err(Error::IsDir(target)),
            Some(Entry::File(_)) | None => {
                self.entries
                    .insert(target, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = self.normalize(path)?;
        let (target, entry) = self.resolve_existing(&parts)?;
        match entry {
            Entry::Dir => Err(Error::IsDir(target)),
            Entry::File(data) => Ok(data),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = self.normalize(path)?;
        let (target, entry) = self.resolve_existing(&parts)?;
        if !matches!(entry, Entry::Dir) {
            return Err(Error::NotDir(target));
        }

        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == "/" {
                continue;
            }
            let entry_parts = Self::components(entry_path);
            if entry_parts.len() == parts.len() + 1 && entry_parts[..parts.len()] == parts {
                children.push((entry_parts.last().unwrap().clone(), entry.kind()));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        let (target, entry) = self.resolve_existing(&parts)?;
        if !matches!(entry, Entry::Dir) {
            return Err(Error::NotDir(target));
        }
        self.cwd = target;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source_parts = self.normalize(from)?;
        let destination_parts = self.normalize(to)?;
        let source = Self::canonical(&source_parts);
        let destination = Self::canonical(&destination_parts);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let source_parent = self.lookup_parent(&source_parts)?;
        let source = Self::child_path(&source_parent, source_parts.last().unwrap());
        let source_entry = match self.entries.get(&source) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(source)),
        };

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let destination_parent = self.lookup_parent(&destination_parts)?;
        let destination = Self::child_path(
            &destination_parent,
            destination_parts.last().unwrap(),
        );

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref dest_entry) = destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(destination)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(destination)),
                (Entry::Dir, Entry::Dir) => {
                    let has_children = self.entries.keys().any(|path| {
                        path != &destination && Self::is_within(path, &destination)
                    });
                    if has_children {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| *path == &source || Self::is_within(path, &source))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if let Some(ref dest_entry) = destination_entry {
            if matches!(dest_entry, Entry::Dir) {
                self.entries.retain(|path, _| {
                    path != &destination && !Self::is_within(path, &destination)
                });
            } else {
                self.entries.remove(&destination);
            }
        }

        for (path, _) in &moving {
            self.entries.remove(path);
        }
        for (old_path, entry) in moving {
            let suffix = &old_path[source.len()..];
            self.entries.insert(format!("{destination}{suffix}"), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let parts = self.normalize(path)?;
        if parts.is_empty() {
            return Err(Error::Root);
        }

        let parent = self.lookup_parent(&parts)?;
        let target = Self::child_path(&parent, parts.last().unwrap());
        let entry = match self.entries.get(&target) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(target)),
        };

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }

        let has_children = matches!(entry, Entry::Dir)
            && self
                .entries
                .keys()
                .any(|path| path != &target && Self::is_within(path, &target));
        if has_children && !recursive {
            return Err(Error::NotEmpty(target));
        }

        self.entries.retain(|path, _| {
            path != &target && !(recursive && Self::is_within(path, &target))
        });
        Ok(())
    }
}

use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn components(path: &str, cwd: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else if cwd == "/" {
            Vec::new()
        } else {
            cwd[1..].split('/').map(str::to_string).collect()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_string()),
            }
        }
        Ok(parts)
    }

    fn resolve_path(&self, path: &str) -> Result<String, Error> {
        Self::components(path, &self.cwd).map(|parts| Self::canonical(&parts))
    }

    fn parent(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_string(), String::new());
        }
        match path.rfind('/') {
            Some(0) => ("/".to_string(), path[1..].to_string()),
            Some(i) => (path[..i].to_string(), path[i + 1..].to_string()),
            None => ("/".to_string(), path.to_string()),
        }
    }

    fn prefix(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    // Checks every component needed to traverse to the requested path.
    fn lookup(&self, path: &str) -> Result<Option<&Entry>, Error> {
        if path == "/" {
            return Ok(self.entries.get("/"));
        }

        let mut current = String::new();
        let components: Vec<&str> = path[1..].split('/').collect();
        for (i, component) in components.iter().enumerate() {
            current.push('/');
            current.push_str(component);
            match self.entries.get(&current) {
                None => return Err(Error::Missing(current)),
                Some(entry) if i + 1 < components.len() && !matches!(entry, Entry::Dir) => {
                    return Err(Error::NotDir(current));
                }
                Some(entry) if i + 1 == components.len() => return Ok(Some(entry)),
                Some(_) => {}
            }
        }
        Ok(None)
    }

    fn require_parent(&self, parent: &str) -> Result<(), Error> {
        match self.lookup(parent)? {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(parent.to_string())),
            None => Err(Error::Missing(parent.to_string())),
        }
    }

    fn contains_path(directory: &str, path: &str) -> bool {
        directory == path
            || (directory == "/" && path.starts_with('/'))
            || (path.len() > directory.len()
                && path.starts_with(directory)
                && path.as_bytes().get(directory.len()) == Some(&b'/'))
    }

    fn children_of(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                if path == "/" {
                    candidate
                        .strip_prefix('/')
                        .is_some_and(|rest| !rest.is_empty() && !rest.contains('/'))
                } else {
                    candidate
                        .strip_prefix(path)
                        .and_then(|rest| rest.strip_prefix('/'))
                        .is_some_and(|rest| !rest.is_empty() && !rest.contains('/'))
                }
            })
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.resolve_path(path)?;
        if resolved == "/" {
            return Err(Error::Exists(resolved));
        }
        let (parent, _) = Self::parent(&resolved);
        self.require_parent(&parent)?;
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let resolved = self.resolve_path(path)?;
        if resolved == "/" {
            return Err(Error::IsDir(resolved));
        }

        let (parent, _) = Self::parent(&resolved);
        self.require_parent(&parent)?;

        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            None => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let resolved = self.resolve_path(path)?;
        match self.lookup(&resolved)? {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let resolved = self.resolve_path(path)?;
        match self.lookup(&resolved)? {
            Some(Entry::Dir) => {
                let mut result = Vec::new();
                for child in self.children_of(&resolved) {
                    let name = if resolved == "/" {
                        child[1..].to_string()
                    } else {
                        child[resolved.len() + 1..].to_string()
                    };
                    if let Some(entry) = self.entries.get(&child) {
                        result.push((name, entry.kind()));
                    }
                }
                result.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(result)
            }
            Some(_) | None => Err(Error::NotDir(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.resolve_path(path)?;
        match self.lookup(&resolved)? {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) | None => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.resolve_path(from)?;
        let destination = self.resolve_path(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, _) = Self::parent(&source);
        self.require_parent(&source_parent)?;

        let source_entry = match self.entries.get(&source) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(source.clone())),
        };

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        let (destination_parent, _) = Self::parent(&destination);
        self.require_parent(&destination_parent)?;

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(existing) = &destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(destination)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(destination)),
                (Entry::Dir, Entry::Dir) => {
                    if !self.children_of(&destination).is_empty() {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::contains_path(&source, path))
            .map(|(path, entry)| {
                let suffix = &path[source.len()..];
                (format!("{}{}", destination, suffix), entry.clone())
            })
            .collect();

        if destination_entry.is_some() {
            self.entries.retain(|path, _| !Self::contains_path(&destination, path));
        }
        self.entries.retain(|path, _| !Self::contains_path(&source, path));
        for (path, entry) in moved {
            self.entries.insert(path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let resolved = self.resolve_path(path)?;
        if resolved == "/" {
            return Err(Error::Root);
        }

        let (parent, _) = Self::parent(&resolved);
        self.require_parent(&parent)?;

        let entry = match self.entries.get(&resolved) {
            Some(entry) => entry,
            None => return Err(Error::Missing(resolved)),
        };

        if matches!(entry, Entry::Dir) && Self::contains_path(&resolved, &self.cwd) {
            return Err(Error::Busy(resolved));
        }

        let children = self.children_of(&resolved);
        if matches!(entry, Entry::Dir) && !children.is_empty() && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        self.entries
            .retain(|candidate, _| !Self::contains_path(&resolved, candidate));
        Ok(())
    }
}

use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components: Vec<String> = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                other => components.push(other.to_string()),
            }
        }

        Ok(Self::path_from_components(&components))
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()
    }

    fn path_from_components(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn parent_and_name(path: &str) -> Option<(String, String)> {
        if path == "/" {
            return None;
        }
        let index = path.rfind('/')?;
        let parent = if index == 0 {
            "/".to_string()
        } else {
            path[..index].to_string()
        };
        Some((parent, path[index + 1..].to_string()))
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) =
            Self::parent_and_name(path).expect("non-root path has a parent and basename");
        self.resolve_directory(&parent)?;
        Ok((parent, name))
    }

    fn resolve_directory(&self, path: &str) -> Result<(), Error> {
        let components = Self::components(path);
        let mut current = String::from("/");

        if !matches!(self.entries.get("/"), Some(Entry::Dir)) {
            return Err(Error::Missing("/".to_string()));
        }

        for (index, component) in components.iter().enumerate() {
            let next = Self::child_path(&current, component);
            match self.entries.get(&next) {
                None => return Err(Error::Missing(next)),
                Some(Entry::Dir) => current = next,
                Some(Entry::File(_)) => {
                    if index + 1 < components.len() {
                        return Err(Error::NotDir(next));
                    }
                    return Err(Error::NotDir(next));
                }
            }
        }
        Ok(())
    }

    fn entry_kind(&self, path: &str) -> Option<Kind> {
        self.entries.get(path).map(Entry::kind)
    }

    fn is_within(path: &str, directory: &str) -> bool {
        if directory == "/" {
            return path.starts_with('/');
        }
        path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries.keys().any(|candidate| {
            candidate != path && Self::is_within(candidate, path)
        })
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.normalize(path)?;
        if resolved == "/" {
            return Err(Error::Exists(resolved));
        }
        let (parent, name) = self.resolve_parent(&resolved)?;
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries
            .insert(Self::child_path(&parent, &name), Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let resolved = self.normalize(path)?;
        if resolved == "/" {
            return Err(Error::IsDir(resolved));
        }
        let (parent, name) = self.resolve_parent(&resolved)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => return Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) | None => {}
        }
        self.entries
            .insert(Self::child_path(&parent, &name), Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let resolved = self.normalize(path)?;
        if resolved == "/" {
            return Err(Error::IsDir(resolved));
        }
        let (parent, _) = self.resolve_parent(&resolved)?;
        match self.entries.get(&resolved) {
            None => Err(Error::Missing(resolved)),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(data)) => {
                let _ = parent;
                Ok(data.clone())
            }
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let resolved = self.normalize(path)?;
        self.resolve_directory(&resolved)?;
        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == "/" {
                continue;
            }
            if let Some((parent, name)) = Self::parent_and_name(entry_path) {
                if parent == resolved {
                    children.push((name, entry.kind()));
                }
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.normalize(path)?;
        match self.entries.get(&resolved) {
            None => {
                self.resolve_directory(&resolved)?;
                Err(Error::Missing(resolved))
            }
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(Entry::File(_)) => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_entry = match self.entries.get(&source) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(source)),
        };

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        if let Some(destination_entry) = self.entries.get(&destination) {
            if !overwrite {
                return Err(Error::Exists(destination));
            }
            match (&source_entry, destination_entry) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(destination)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(destination)),
                (Entry::Dir, Entry::Dir) if self.has_children(&destination) => {
                    return Err(Error::NotEmpty(destination))
                }
                _ => {}
            }
        }

        let destination_path = Self::child_path(&destination_parent, &destination_name);
        let old_prefix = Self::child_path(&source_parent, &source_name);
        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| **path == old_prefix || Self::is_within(path, &old_prefix))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if self.entries.contains_key(&destination_path) {
            self.entries.remove(&destination_path);
        }
        for (old_path, _) in &moved {
            self.entries.remove(old_path);
        }
        for (old_path, entry) in moved {
            let suffix = &old_path[old_prefix.len()..];
            self.entries.insert(format!("{destination_path}{suffix}"), entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let resolved = self.normalize(path)?;
        if resolved == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&resolved)?;
        let entry = match self.entries.get(&resolved) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(resolved)),
        };

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }

        if matches!(entry, Entry::Dir) && self.has_children(&resolved) && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| {
                **candidate == resolved
                    || (recursive && matches!(entry, Entry::Dir) && Self::is_within(candidate, &resolved))
            })
            .cloned()
            .collect();
        for entry_path in paths {
            self.entries.remove(&entry_path);
        }
        Ok(())
    }
}
