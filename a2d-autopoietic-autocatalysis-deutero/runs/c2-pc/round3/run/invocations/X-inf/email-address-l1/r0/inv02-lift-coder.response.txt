#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i));

    let Some(separator) = separators.next() else {
        return Err(EmailError::MissingAt);
    };
    if let Some(second) = separators.next() {
        return Err(EmailError::MultipleAt { at: second });
    }

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];
    if local_bytes.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (i, &b) in local_bytes.iter().enumerate() {
        let position = i;
        if b == b'.' {
            if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at: position });
            }
        } else if !b.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            return Err(EmailError::LocalChar { at: position });
        }
    }

    let raw_domain_start = separator + 1;
    let mut domain_end = bytes.len();
    let mut removed_root_dot = None;
    if domain_end > raw_domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        removed_root_dot = Some(domain_end);
    }

    if domain_end == raw_domain_start {
        return Err(EmailError::EmptyDomain {
            at: raw_domain_start,
        });
    }

    let domain_bytes = &bytes[raw_domain_start..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: raw_domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut start = 0;
    loop {
        let end = domain_bytes[start..]
            .iter()
            .position(|&b| b == b'.')
            .map(|offset| start + offset)
            .unwrap_or(domain_bytes.len());

        if end == start {
            let at = if start == domain_bytes.len() {
                removed_root_dot.unwrap_or(raw_domain_start + start)
            } else {
                raw_domain_start + start
            };
            return Err(EmailError::EmptyLabel { at });
        }

        let label = &domain_bytes[start..end];
        let label_start = raw_domain_start + start;

        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: label_start + 63,
            });
        }

        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen { at: label_start });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: label_start + label.len() - 1,
            });
        }

        for (i, &b) in label.iter().enumerate() {
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: label_start + i,
                });
            }
        }

        label_starts.push((start, end));
        if end == domain_bytes.len() {
            break;
        }
        start = end + 1;
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: raw_domain_start,
        });
    }

    let (last_start, last_end) = *label_starts.last().unwrap();
    let last_label = &domain_bytes[last_start..last_end];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: raw_domain_start + last_start,
        });
    }

    let domain = domain_bytes
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local: String::from_utf8(local_bytes.to_vec()).unwrap(),
        domain,
        quoted: false,
    })
}
