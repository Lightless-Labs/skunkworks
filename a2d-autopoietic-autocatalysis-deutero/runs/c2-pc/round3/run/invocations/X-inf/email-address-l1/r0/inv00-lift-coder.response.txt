#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    let separator = match separators.as_slice() {
        [] => return Err(EmailError::MissingAt),
        [separator] => *separator,
        [_, second, ..] => return Err(EmailError::MultipleAt { at: *second }),
    };

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];
    if local_bytes.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (index, &byte) in local_bytes.iter().enumerate() {
        if byte == b'.' {
            if index == 0 || index + 1 == local_bytes.len() || local_bytes[index - 1] == b'.' {
                return Err(EmailError::LocalDot { at: index });
            }
        } else if !byte.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
        {
            return Err(EmailError::LocalChar { at: index });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain_bytes = &raw_domain[..domain_len];
    let root_dot_position = domain_start + domain_len;

    let mut label_starts = Vec::new();
    let mut label_ranges = Vec::new();
    let mut start = 0;

    for index in 0..=domain_len {
        if index == domain_len || domain_bytes[index] == b'.' {
            if start == index {
                let at = if index == domain_len && has_root_dot {
                    root_dot_position
                } else {
                    domain_start + index
                };
                return Err(EmailError::EmptyLabel { at });
            }

            label_starts.push(domain_start + start);
            label_ranges.push((start, index));
            start = index + 1;
        }
    }

    for &(start, end) in &label_ranges {
        let label = &domain_bytes[start..end];
        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_start + start + 63,
            });
        }

        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + start,
            });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + end - 1,
            });
        }

        for (offset, &byte) in label.iter().enumerate() {
            if !byte.is_ascii_alphanumeric() && byte != b'-' {
                return Err(EmailError::DomainChar {
                    at: domain_start + start + offset,
                });
            }
        }
    }

    if label_ranges.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *label_ranges.last().unwrap();
    let last_label = &domain_bytes[last_start..last_end];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: label_starts[label_starts.len() - 1],
        });
    }

    let local = String::from_utf8(local_bytes.to_vec()).unwrap();
    let domain = String::from_utf8(domain_bytes.to_vec())
        .unwrap()
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted: false,
    })
}
