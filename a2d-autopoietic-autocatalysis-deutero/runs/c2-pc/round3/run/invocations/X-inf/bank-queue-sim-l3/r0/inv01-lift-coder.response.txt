#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    match a.customer.priority.cmp(&b.customer.priority) {
        Ordering::Equal => {}
        other => return other,
    }
    match (a.pause_time, b.pause_time) {
        (Some(at), Some(bt)) => at
            .cmp(&bt)
            .then_with(|| a.customer.id.cmp(&b.customer.id)),
        (Some(_), None) => Ordering::Less,
        (None, Some(_)) => Ordering::Greater,
        (None, None) => a
            .customer
            .arrival
            .cmp(&b.customer.arrival)
            .then_with(|| b.customer.id.cmp(&a.customer.id)),
    }
}

fn best_waiter_index(waiting: &[Job]) -> Option<usize> {
    waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiter_cmp(a, b))
        .map(|(i, _)| i)
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut completed = 0usize;
    let mut time = arrivals[0].arrival;

    while completed < customers.len() {
        // Finish takes precedence over arrivals and all other work at this tick.
        if running.as_ref().is_some_and(|r| r.finish == time) {
            let finished = running.take().unwrap();
            trace.push(Event {
                time,
                customer: finished.job.customer.id,
                kind: Kind::Finish,
            });
            completed += 1;
        }

        // Process the whole arrival cohort before making scheduling decisions.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            trace.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                if let Some(best_index) = best_waiter_index(&waiting) {
                    let best_priority = waiting[best_index].customer.priority;
                    if best_priority < active.job.customer.priority {
                        let by = waiting[best_index].customer.id;
                        let mut paused = running.take().unwrap().job;
                        paused.remaining = active.finish - time;
                        paused.pause_time = Some(time);
                        trace.push(Event {
                            time,
                            customer: paused.customer.id,
                            kind: Kind::Preempt {
                                remaining: paused.remaining,
                                by,
                            },
                        });
                        waiting.push(paused);
                    }
                }
            }
        }

        // Dispatch repeatedly at this tick, including inline zero-duration finishes.
        if running.is_none() {
            loop {
                let Some(index) = best_waiter_index(&waiting) else {
                    break;
                };
                let job = waiting.remove(index);
                let id = job.customer.id;
                let cost = if last_dispatched.is_some_and(|previous| previous != id) {
                    switch_cost
                } else {
                    0
                };
                let work_at = time.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: id,
                    time,
                })?;
                let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                    customer: id,
                    time,
                })?;
                last_dispatched = Some(id);

                let kind = if job.started {
                    Kind::Resume {
                        remaining: job.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining: job.remaining,
                        work_at,
                        finish,
                    }
                };
                trace.push(Event {
                    time,
                    customer: id,
                    kind,
                });

                if finish == time {
                    trace.push(Event {
                        time,
                        customer: id,
                        kind: Kind::Finish,
                    });
                    completed += 1;
                    continue;
                }

                let mut job = job;
                job.started = true;
                running = Some(Running {
                    job,
                    work_at,
                    finish,
                });
                break;
            }
        }

        if completed == customers.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_time = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            for candidate in [Some(active.finish), Some(active.work_at)] {
                if let Some(candidate) = candidate.filter(|&t| t > time) {
                    next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
                }
            }
        }
        time = next_time.expect("unfinished simulation must have a future event");
    }

    Ok(trace)
}
