```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize_from(path: &str, base: &[String]) -> String {
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        base.to_vec()
    };
    for c in path.split('/') {
        match c {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            _ => parts.push(c.to_owned()),
        }
    }
    canonical(&parts)
}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn components(path: &str) -> Vec<String> {
    if path == "/" {
        Vec::new()
    } else {
        path[1..].split('/').map(str::to_owned).collect()
    }
}

fn parent_and_name(path: &str) -> (String, String) {
    let mut c = components(path);
    let name = c.pop().unwrap();
    (canonical(&c), name)
}

fn join_child(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn is_within(path: &str, dir: &str) -> bool {
    path == dir || dir == "/" || path.starts_with(&format!("{dir}/"))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        validate_path(path)?;
        Ok(normalize_from(path, &components(&self.cwd)))
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let mut pending = components(path);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(canonical(&resolved));
            }

            let name = pending.remove(0);
            let candidate = join_child(&canonical(&resolved), &name);
            let final_component = pending.is_empty();
            let entry = match self.entries.get(&candidate) {
                Some(e) => e.clone(),
                None if create_final && final_component => return Ok(candidate),
                None => return Err(Error::Missing(candidate)),
            };

            if let Entry::Link(target) = entry {
                if follow_final || !final_component {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let target_base = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        resolved.clone()
                    };
                    let expanded = normalize_from(&target, &target_base);
                    let mut next = components(&expanded);
                    next.extend(pending);
                    pending = next;
                    resolved.clear();
                    continue;
                }
            }

            if !final_component && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(candidate));
            }
            resolved.push(name);
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        if path == "/" {
            return Ok(("/".to_owned(), String::new()));
        }
        let (parent, name) = parent_and_name(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn stored_child_paths(&self, dir: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|p| *p != dir && is_within(p, dir))
            .cloned()
            .collect()
    }

    fn direct_children(&self, dir: &str) -> Vec<(String, String)> {
        self.entries
            .iter()
            .filter_map(|(p, e)| {
                if p == dir || !is_within(p, dir) {
                    return None;
                }
                let (parent, name) = parent_and_name(p);
                (parent == dir).then(|| {
                    let kind = match e {
                        Entry::Dir => "Dir",
                        Entry::File(_) => "File",
                        Entry::Link(_) => "Link",
                    };
                    (name, kind.to_owned())
                })
            })
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&path)?;
        let dest = if name.is_empty() { "/".to_owned() } else { join_child(&parent, &name) };
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, true)?;
        if resolved == "/" {
            return Err(Error::IsDir(resolved));
        }
        let (parent, _) = parent_and_name(&resolved);
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let mut out: Vec<_> = self
            .entries
            .iter()
            .filter_map(|(p, e)| {
                if p == &resolved || !is_within(p, &resolved) {
                    return None;
                }
                let (par, name) = parent_and_name(p);
                if par != resolved {
                    return None;
                }
                let kind = match e {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                    Entry::Link(_) => Kind::Link,
                };
                Some((name, kind))
            })
            .collect();
        out.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(out)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = self.normalize(from)?;
        let to = self.normalize(to)?;
        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&from)?;
        let source = join_child(&src_parent, &src_name);
        let source_entry = self.entries.get(&source).cloned().ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&to)?;
        let dest = join_child(&dst_parent, &dst_name);
        if matches!(self.entries.get(&dest), Some(Entry::Dir)) && is_within(&self.cwd, &dest) {
            return Err(Error::Busy(dest));
        }

        if source == dest {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_within(&dest, &source) {
            return Err(Error::Cycle(dest));
        }

        let dest_entry = self.entries.get(&dest).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(dest));
        }
        if let Some(ref de) = dest_entry {
            match (&source_entry, de) {
                (Entry::Dir, Entry::Dir) => {
                    if self.stored_child_paths(&dest).len() > 0 {
                        return Err(Error::NotEmpty(dest));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dest)),
                (_, Entry::Dir) => return Err(Error::IsDir(dest)),
                _ => {}
            }
        }

        if dest_entry.is_some() {
            let paths = self.stored_child_paths(&dest);
            for p in paths {
                self.entries.remove(&p);
            }
            self.entries.remove(&dest);
        }

        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| **p == source || is_within(p, &source))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();
        for (p, _) in &moved {
            self.entries.remove(p);
        }
        for (p, e) in moved {
            let suffix = if p == source { String::new() } else { p[source.len()..].to_owned() };
            self.entries.insert(format!("{dest}{suffix}"), e);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&path)?;
        let target = join_child(&parent, &name);
        let entry = self.entries.get(&target).cloned().ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        let descendants = self.stored_child_paths(&target);
        if matches!(entry, Entry::Dir) && !recursive && !descendants.is_empty() {
            return Err(Error::NotEmpty(target));
        }

        for p in descendants {
            self.entries.remove(&p);
        }
        self.entries.remove(&target);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link_path = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent, name) = self.resolve_parent(&link_path)?;
        let dest = if name.is_empty() { "/".to_owned() } else { join_child(&parent, &name) };
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, false)?;
        let root_entry = self.entries.get(&resolved).ok_or_else(|| Error::Missing(resolved.clone()))?;
        if let Entry::File(data) = root_entry {
            return Ok(vec![(resolved, data.len() as u64)]);
        }
        if !matches!(root_entry, Entry::Dir) {
            return Err(Error::NotDir(resolved));
        }

        let mut paths = vec![resolved.clone()];
        paths.extend(self.stored_child_paths(&resolved));
        paths.sort();

        let mut sizes = BTreeMap::<String, u64>::new();
        for p in paths.iter().rev() {
            let size = match self.entries.get(p).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => {
                    let mut sum = 0u64;
                    for child in self.stored_child_paths(p) {
                        if let Some(s) = sizes.get(&child) {
                            sum = sum.saturating_add(*s);
                        }
                    }
                    sum
                }
            };
            sizes.insert(p.clone(), size);
        }
        Ok(paths.into_iter().map(|p| {
            let size = *sizes.get(&p).unwrap_or(&0);
            (p, size)
        }).collect())
    }
}
```
