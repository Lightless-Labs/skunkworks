use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|s| !s.is_empty() && *s != ".")
            .fold(Vec::new(), |mut acc, part| {
                if part == ".." {
                    acc.pop();
                } else {
                    acc.push(part.to_owned());
                }
                acc
            })
    }

    fn from_components(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };
        for part in path.split('/') {
            if part.is_empty() || part == "." {
                continue;
            }
            if part == ".." {
                parts.pop();
            } else {
                parts.push(part.to_owned());
            }
        }
        Self::from_components(&parts)
    }

    fn parent(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_owned(), String::new());
        }
        let idx = path.rfind('/').unwrap_or(0);
        let name = path[idx + 1..].to_owned();
        let parent = if idx == 0 {
            "/".to_owned()
        } else {
            path[..idx].to_owned()
        };
        (parent, name)
    }

    fn append(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        dir == "/" || path == dir || path.strip_prefix(dir).is_some_and(|s| s.starts_with('/'))
    }

    fn children(&self, dir: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|path| **path != dir && Self::parent(path).0 == dir)
            .cloned()
            .collect()
    }

    fn resolve(&self, path: &str, follow_final: bool, create_missing_final: bool) -> Result<String, Error> {
        let initial = Self::components(path);
        let mut pending = initial;
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;
        let mut index = 0usize;

        loop {
            if index >= pending.len() {
                return Ok(Self::from_components(&resolved));
            }

            let candidate = {
                let mut c = resolved.clone();
                c.push(pending[index].clone());
                Self::from_components(&c)
            };
            let is_final = index + 1 == pending.len();

            match self.entries.get(&candidate) {
                None if is_final && create_missing_final => {
                    return Ok(candidate);
                }
                None => return Err(Error::Missing(candidate)),
                Some(entry) => {
                    if let Entry::Link(target) = entry {
                        if follow_final || !is_final {
                            if !visited.insert(candidate.clone()) {
                                return Err(Error::Loop(candidate));
                            }
                            if expansions >= 8 {
                                return Err(Error::Depth(candidate));
                            }
                            expansions += 1;

                            let mut base = if target.starts_with('/') {
                                Vec::new()
                            } else {
                                resolved.clone()
                            };
                            for p in target.split('/') {
                                if p.is_empty() || p == "." {
                                    continue;
                                }
                                if p == ".." {
                                    base.pop();
                                } else {
                                    base.push(p.to_owned());
                                }
                            }
                            let mut replacement = base;
                            replacement.extend(pending[index + 1..].iter().cloned());
                            pending = replacement;
                            resolved.clear();
                            index = 0;
                            continue;
                        }
                    }

                    if !is_final && !matches!(entry, Entry::Dir) {
                        return Err(Error::NotDir(candidate));
                    }
                    resolved.push(pending[index].clone());
                    index += 1;
                }
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn entry_path(parent: &str, name: &str) -> String {
        Self::append(parent, name)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = self.normalize(path);
        if canonical == "/" {
            return Err(Error::Exists("/".to_owned()));
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let final_path = Self::entry_path(&parent, &name);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = self.normalize(path);
        let resolved = self.resolve(&canonical, true, true)?;
        if resolved == "/" {
            return Err(Error::IsDir("/".to_owned()));
        }
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, name) = Self::parent(&resolved);
                let resolved_parent = self.resolve(&parent, true, false)?;
                match self.entries.get(&resolved_parent) {
                    Some(Entry::Dir) => {
                        self.entries.insert(
                            Self::entry_path(&resolved_parent, &name),
                            Entry::File(data.to_vec()),
                        );
                        Ok(())
                    }
                    Some(_) => Err(Error::NotDir(resolved_parent)),
                    None => Err(Error::Missing(resolved_parent)),
                }
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let canonical = self.normalize(path);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let canonical = self.normalize(path);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                let mut result: Vec<_> = self
                    .children(&resolved)
                    .into_iter()
                    .map(|child| {
                        let name = Self::parent(&child).1;
                        (name, self.entries[&child].kind())
                    })
                    .collect();
                result.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(result)
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = self.normalize(path);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        let source = self.normalize(from);
        Self::validate(to)?;
        let destination = self.normalize(to);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&source)?;
        let src_path = Self::entry_path(&src_parent, &src_name);
        let src_entry = self
            .entries
            .get(&src_path)
            .cloned()
            .ok_or_else(|| Error::Missing(src_path.clone()))?;

        if matches!(src_entry, Entry::Dir) && Self::is_within(&self.cwd, &src_path) {
            return Err(Error::Busy(src_path));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&destination)?;
        let dst_path = Self::entry_path(&dst_parent, &dst_name);

        if matches!(self.entries.get(&dst_path), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &dst_path)
        {
            return Err(Error::Busy(dst_path));
        }

        if src_path == dst_path {
            return Ok(());
        }

        if matches!(src_entry, Entry::Dir) && Self::is_within(&dst_path, &src_path) {
            return Err(Error::Cycle(dst_path));
        }

        let dst_entry = self.entries.get(&dst_path).cloned();
        if dst_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst_path));
        }

        if let Some(ref dst) = dst_entry {
            match (&src_entry, dst) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&dst_path).is_empty() {
                        return Err(Error::NotEmpty(dst_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst_path)),
                _ => {}
            }
        }

        let source_paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &src_path))
            .cloned()
            .collect();

        if dst_entry.is_some() {
            let to_remove: Vec<String> = self
                .entries
                .keys()
                .filter(|p| Self::is_within(p, &dst_path))
                .cloned()
                .collect();
            for p in to_remove {
                self.entries.remove(&p);
            }
        }

        let moved: Vec<(String, Entry)> = source_paths
            .iter()
            .map(|p| {
                let suffix = &p[src_path.len()..];
                (format!("{}{}", dst_path, suffix), self.entries[p].clone())
            })
            .collect();

        for p in source_paths {
            self.entries.remove(&p);
        }
        for (p, entry) in moved {
            self.entries.insert(p, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = self.normalize(path);
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let resolved = Self::entry_path(&parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }

        if matches!(entry, Entry::Dir) && !recursive && !self.children(&resolved).is_empty() {
            return Err(Error::NotEmpty(resolved));
        }

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &resolved))
            .cloned()
            .collect();
        for p in paths {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let canonical_link = self.normalize(link);
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        if canonical_link == "/" {
            return Err(Error::Exists("/".to_owned()));
        }
        let (parent, name) = self.resolve_parent(&canonical_link)?;
        let final_path = Self::entry_path(&parent, &name);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries
            .insert(final_path, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let canonical = self.normalize(path);
        if canonical == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (parent, name) = Self::parent(&canonical);
        let resolved_parent = self.resolve(&parent, true, false)?;
        let final_path = Self::entry_path(&resolved_parent, &name);
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let canonical = self.normalize(path);
        let resolved = self.resolve(&canonical, true, false)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|p| Self::is_within(p, &resolved))
            .cloned()
            .collect();

        let mut sizes = BTreeMap::new();
        for p in &paths {
            let size = match &self.entries[p] {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(p.clone(), size);
        }

        for p in paths.iter().rev() {
            if matches!(self.entries[p], Entry::Dir) {
                let subtotal: u64 = paths
                    .iter()
                    .filter(|child| {
                        **child != *p && Self::is_within(child, p)
                    })
                    .filter(|child| {
                        let (parent, _) = Self::parent(child);
                        parent == *p || Self::is_within(&parent, p)
                    })
                    .map(|child| {
                        if matches!(self.entries[child], Entry::Dir) {
                            0
                        } else {
                            sizes[child]
                        }
                    })
                    .sum();
                sizes.insert(p.clone(), subtotal);
            }
        }

        let _ = entry;
        let mut result: Vec<_> = sizes.into_iter().collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
