use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn normalize_components(mut components: Vec<String>, path: &str) -> Vec<String> {
    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                components.pop();
            }
            name => components.push(name.to_owned()),
        }
    }
    components
}

fn canonical(components: &[String]) -> String {
    if components.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", components.join("/"))
    }
}

fn components(path: &str) -> Vec<String> {
    if path == "/" {
        Vec::new()
    } else {
        path[1..].split('/').map(str::to_owned).collect()
    }
}

fn parent_path(path: &str) -> String {
    if path == "/" {
        return "/".to_owned();
    }
    let mut c = components(path);
    c.pop();
    canonical(&c)
}

fn basename(path: &str) -> String {
    components(path).last().cloned().unwrap_or_default()
}

fn is_within(path: &str, directory: &str) -> bool {
    if directory == "/" {
        return path.starts_with('/');
    }
    path == directory
        || path
            .strip_prefix(directory)
            .is_some_and(|suffix| suffix.starts_with('/'))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        validate_path(path)?;
        let initial = if path.starts_with('/') {
            Vec::new()
        } else {
            components(&self.cwd)
        };
        Ok(canonical(&normalize_components(initial, path)))
    }

    fn resolve(&self, path: &str, follow_final: bool, create_final: bool) -> Result<String, Error> {
        let requested = components(path);
        let mut physical: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;
        let mut index = 0usize;

        loop {
            if index == requested.len() {
                return Ok(canonical(&physical));
            }
            let name = requested[index].clone();
            physical.push(name);
            let candidate = canonical(&physical);
            let final_component = index + 1 == requested.len();

            match self.entries.get(&candidate) {
                None => {
                    if create_final && final_component {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
                Some(Entry::Dir) => {
                    index += 1;
                    continue;
                }
                Some(entry) => {
                    if !final_component {
                        if let Entry::Link(target) = entry {
                            expand_link(
                                target,
                                &candidate,
                                &mut physical,
                                &mut visited,
                                &mut expansions,
                                &requested[index + 1..],
                            )?;
                            let expanded = canonical(&physical);
                            let normalized = normalize_components(Vec::new(), &expanded);
                            physical.clear();
                            requested_from_components_placeholder();
                            let _ = normalized;
                        } else {
                            return Err(Error::NotDir(candidate));
                        }
                    } else if follow_final {
                        if let Entry::Link(target) = entry {
                            let target_components = target_base(target, &physical);
                            if visited.contains(&candidate) {
                                return Err(Error::Loop(candidate));
                            }
                            if expansions >= 8 {
                                return Err(Error::Depth(candidate));
                            }
                            visited.insert(candidate);
                            expansions += 1;

                            let mut combined = target_components;
                            combined.extend(requested[index + 1..].iter().cloned());
                            let target_path = canonical(&combined);
                            let target_components = components(&target_path);
                            return self.resolve_components_with_state(
                                target_components,
                                visited,
                                expansions,
                                create_final,
                            );
                        }
                    } else {
                        index += 1;
                        continue;
                    }
                }
            }
        }
    }

    fn resolve_components_with_state(
        &self,
        requested: Vec<String>,
        mut visited: BTreeSet<String>,
        mut expansions: usize,
        create_final: bool,
    ) -> Result<String, Error> {
        let mut physical: Vec<String> = Vec::new();
        let mut index = 0;
        loop {
            if index == requested.len() {
                return Ok(canonical(&physical));
            }
            physical.push(requested[index].clone());
            let candidate = canonical(&physical);
            let final_component = index + 1 == requested.len();

            match self.entries.get(&candidate) {
                None => {
                    if create_final && final_component {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
                Some(Entry::Dir) => index += 1,
                Some(Entry::File(_)) => {
                    if final_component {
                        return Ok(candidate);
                    }
                    return Err(Error::NotDir(candidate));
                }
                Some(Entry::Link(target)) => {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;
                    let mut expanded = target_base(target, &physical);
                    expanded.extend(requested[index + 1..].iter().cloned());
                    return self.resolve_components_with_state(
                        expanded,
                        visited,
                        expansions,
                        create_final,
                    );
                }
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = parent_path(path);
        let name = basename(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => Ok((resolved_parent, name)),
            Some(_) => Err(Error::NotDir(resolved_parent)),
            None => Err(Error::Missing(resolved_parent)),
        }
    }

    fn resolved_child(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                *candidate != path
                    && parent_path(candidate) == path
                    && self.entries.contains_key(*candidate)
            })
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&path)?;
        let target = Self::resolved_child(&parent, &name);
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, true)?;
        let parent = parent_path(&resolved);
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => return Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) | None => {}
            Some(Entry::Link(_)) => {
                return Err(Error::Loop(resolved));
            }
        }
        self.entries
            .insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => Err(Error::Loop(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let mut rows: Vec<_> = self
            .entries
            .iter()
            .filter_map(|(candidate, entry)| {
                if candidate != &resolved && parent_path(candidate) == resolved {
                    Some((basename(candidate), entry.kind()))
                } else {
                    None
                }
            })
            .collect();
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = self.normalize(from)?;
        let to = self.normalize(to)?;
        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&from)?;
        let src = Self::resolved_child(&src_parent, &src_name);
        let source_entry = self
            .entries
            .get(&src)
            .cloned()
            .ok_or_else(|| Error::Missing(src.clone()))?;
        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&to)?;
        let dst = Self::resolved_child(&dst_parent, &dst_name);
        if matches!(self.entries.get(&dst), Some(Entry::Dir)) && is_within(&self.cwd, &dst) {
            return Err(Error::Busy(dst));
        }

        if src == dst {
            return Ok(());
        }
        if matches!(source_entry, Entry::Dir) && is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        let destination_entry = self.entries.get(&dst).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }
        if let Some(ref dest_entry) = destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.children(&dst).len() > 0 {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| is_within(p, &src))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if destination_entry.is_some() {
            self.entries.retain(|p, _| !is_within(p, &dst));
        }
        self.entries.retain(|p, _| !is_within(p, &src));
        for (old, entry) in moving {
            let suffix = &old[src.len()..];
            self.entries.insert(format!("{dst}{suffix}"), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&path)?;
        let resolved = Self::resolved_child(&parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }
        if matches!(entry, Entry::Dir) && !recursive && !self.children(&resolved).is_empty() {
            return Err(Error::NotEmpty(resolved));
        }
        self.entries.retain(|p, _| !is_within(p, &resolved));
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link = self.normalize(link)?;
        validate_path(link.as_str())?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent, name) = self.resolve_parent(&link)?;
        let resolved = Self::resolved_child(&parent, &name);
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (parent, name) = self.resolve_parent(&path)?;
        let resolved = Self::resolved_child(&parent, &name);
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let path = self.normalize(path)?;
        let resolved = self.resolve(&path, true, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let mut rows = Vec::new();
        for (candidate, entry) in &self.entries {
            if is_within(candidate, &resolved) {
                let size = match entry {
                    Entry::File(bytes) => bytes.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                    Entry::Dir => self
                        .entries
                        .iter()
                        .filter(|(child, e)| {
                            is_within(child, candidate)
                                && child.as_str() != candidate.as_str()
                                && matches!(e, Entry::File(_) | Entry::Link(_))
                        })
                        .map(|(_, e)| match e {
                            Entry::File(bytes) => bytes.len() as u64,
                            Entry::Link(target) => target.len() as u64,
                            Entry::Dir => 0,
                        })
                        .sum(),
                };
                rows.push((candidate.clone(), size));
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}

fn target_base(target: &str, physical_link_components: &[String]) -> Vec<String> {
    let initial = if target.starts_with('/') {
        Vec::new()
    } else {
        let mut parent = physical_link_components.to_vec();
        parent.pop();
        parent
    };
    normalize_components(initial, target)
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
