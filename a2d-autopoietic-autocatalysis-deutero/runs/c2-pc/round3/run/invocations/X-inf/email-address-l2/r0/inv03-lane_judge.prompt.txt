    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

        let mut i = 1;
        let mut closing = None;

            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    closing = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match closing {
            Some(end) => Some(end),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_prefix_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
            if separators.len() == 2 {
                return Err(EmailError::MultipleAt { at: i });
            }
        }
        i += 1;
    }

    let separator = match separators.first() {
        Some(&at) => at,
        None => return Err(EmailError::MissingAt),
    };

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let (local, quoted) = if bytes[0] == b'"' {
        let end = quoted_prefix_end.expect("quoted prefix has a closing quote");
        if end + 1 != separator {
            return Err(EmailError::TrailingLocal { at: end + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            let b = bytes[j];
            if b == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape { at: j });
                }
                let escaped = bytes[j + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: j });
                }
                decoded.push(escaped);
                origins.push(j);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: j });
                }
                decoded.push(b);
                origins.push(j);
                j += 1;
            }
        }

            return Err(EmailError::EmptyLocal { at: end });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).expect("ASCII bytes are valid UTF-8"), true)
    } else {
        if separator > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for j in 0..separator {
            let b = bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == separator || bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: j });
                }
            } else if !is_atom_byte(b) {
                return Err(EmailError::LocalChar { at: j });
            }
        }

        (
            String::from_utf8(bytes[..separator].to_vec()).expect("ASCII bytes are valid UTF-8"),
            false,
        )
    };

    let domain_start = separator + 1;
    let raw_domain_end = bytes.len();
    let has_root_dot = raw_domain_end > domain_start && bytes[raw_domain_end - 1] == b'.';
    let domain_end = if has_root_dot {
        raw_domain_end - 1
    } else {
        raw_domain_end
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || bytes[j] == b'.' {
            if label_start == j {
                let at = if j == domain_end && has_root_dot {
                    raw_domain_end - 1
                } else {
                    label_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label_len = j - label_start;
            if label_len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: label_start + 63,
                });
            }
            if bytes[label_start] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_start });
            }
            if bytes[j - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: j - 1 });
            }
            for k in label_start..j {
                if !is_domain_byte(bytes[k]) {
                    return Err(EmailError::DomainChar { at: k });
                }
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}

fn is_atom_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn is_domain_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

                continue;

            separators.push(i);
        }
        i += 1;

        return Err(EmailError::MissingAt);

        return Err(EmailError::MultipleAt { at: separators[1] });

        if closing + 1 != sep {
            return Err(EmailError::TrailingLocal { at: closing + 1 });

        let mut origins = Vec::new();

            let b = bytes[j];

                if j + 1 >= closing {
                    return Err(EmailError::BadEscape { at: j });

                        decoded.push(escaped);
                        origins.push(j);

                    return Err(EmailError::QuotedChar { at: j });

                origins.push(j);

        .collect::<Vec<_>>();

        local,
        domain: String::from_utf8(lower_domain).expect("validated ASCII"),
        quoted: is_quoted,
    })
}

    } else {
        if sep > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for i in 0..sep {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || i + 1 == sep || bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: i });
            }
            local.push(b as char);
        }
    }

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let root_dot = raw_domain.last() == Some(&b'.');
    let domain_end = if root_dot {
        bytes.len() - 1
    } else {
        bytes.len()
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for i in domain_start..domain_end {
        if bytes[i] == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }
    labels.push((label_start, domain_end));

    for &(start, end) in &labels {
        if start == end {
            let at = if root_dot && start == domain_end {
                domain_end
            } else {
                start
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if end - start > 63 {
            return Err(EmailError::LabelTooLong { at: start + 63 });
        }

        if bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: start });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: end - 1 });
        }

        for i in start..end {
            let b = bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: i });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if bytes[last_start..last_end]
        .iter()
        .all(|b| b.is_ascii_digit())
    {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = String::from_utf8(bytes[domain_start..domain_end].to_vec())
        .unwrap()
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted: is_quoted,
    })
}
