#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0;
    let mut now = 0u64;
    let mut current: Option<(u32, u64)> = None; // customer ID and finish time
    let mut has_dispatched = false;
    let mut last_finish = 0u64;

    loop {
        // Advance to the next scheduled finish or arrival.
        if let Some((_, finish)) = current {
            now = finish;
        } else if next_arrival < arrivals.len() {
            now = now.max(arrivals[next_arrival].arrival);
        } else {
            break;
        }

        // A completion at this tick precedes all arrivals.
        if let Some((id, finish)) = current {
            if finish == now {
                events.push(Event {
                    time: now,
                    customer: id,
                    kind: Kind::Finish,
                });
                current = None;
                last_finish = now;
            }
        }

        // Emit the entire arrival cohort before dispatching.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        // Dispatch one waiting customer whenever the server is idle.
        if current.is_none() {
            if let Some(customer) = waiting.pop_front() {
                let wake = !has_dispatched || now > last_finish;
                let base_finish = now.checked_add(customer.service).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    },
                )?;
                let finish = base_finish.checked_add(if wake { wake_cost } else { 0 }).ok_or(
                    Error::TimeOverflow {
                        customer: customer.id,
                        time: now,
                    },
                )?;

                events.push(Event {
                    time: now,
                    customer: customer.id,
                    kind: Kind::Start { finish, wake },
                });
                has_dispatched = true;
                current = Some((customer.id, finish));

                // Zero-duration occupation finishes inline before the next start.
                if finish == now {
                    events.push(Event {
                        time: now,
                        customer: customer.id,
                        kind: Kind::Finish,
                    });
                    current = None;
                    last_finish = now;

                    while let Some(customer) = waiting.pop_front() {
                        let wake = false;
                        let base_finish = now.checked_add(customer.service).ok_or(
                            Error::TimeOverflow {
                                customer: customer.id,
                                time: now,
                            },
                        )?;
                        let finish = base_finish.checked_add(0).ok_or(
                            Error::TimeOverflow {
                                customer: customer.id,
                                time: now,
                            },
                        )?;

                        events.push(Event {
                            time: now,
                            customer: customer.id,
                            kind: Kind::Start { finish, wake },
                        });
                        current = Some((customer.id, finish));

                        if finish != now {
                            break;
                        }

                        events.push(Event {
                            time: now,
                            customer: customer.id,
                            kind: Kind::Finish,
                        });
                        current = None;
                        last_finish = now;
                    }
                }
            }
        }
    }

    Ok(events)
}
