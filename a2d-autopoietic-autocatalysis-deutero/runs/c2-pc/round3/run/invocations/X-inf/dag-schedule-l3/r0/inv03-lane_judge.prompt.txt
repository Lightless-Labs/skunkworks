    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest available ID.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }
    let mut topo = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, &id)| id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        topo.push(id);

        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if topo.len() != n {
        let removed = {
            let mut seen = vec![false; n];
            for &id in &topo {
                seen[id] = true;
            }
            seen

        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });

    for &id in topo.iter().rev() {
        let max_child_tail = successors[id]
            .iter()

                stage: Stage::Tail,

        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut running = Vec::<Running>::new();
    let mut worker_available = vec![0u64; workers];
    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

        // Complete every task due now, in ascending task ID order.
        let mut finishing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(i, item)| (item.finish == now).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| running[i].task);

        for index in finishing.into_iter().rev() {
            let item = running.swap_remove(index);
            completed[item.task] = true;
            completed_count += 1;
            makespan = makespan.max(item.finish);
            events.push(Event {
                time: now,
                task: item.task,
                worker: item.worker,
                kind: EventKind::Finish,
            });
            for &child in &successors[item.task] {
                remaining_deps[child] -= 1;
            }
        }

        // Start one task at a time, reconsidering eligibility and ranking after
        // each start (including immediate zero-duration completions).
        loop {
            let worker = (0..workers).find(|&w| worker_available[w] <= now);
            let Some(worker) = worker else {
                break;
            };

            let selected = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_deps[id] == 0
                })
                .min_by(|&a, &b| {
                    tail[b]
                        .cmp(&tail[a])
                        .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                        .then_with(|| a.cmp(&b))
                });

            let Some(task_id) = selected else {
                break;
            };

                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,

            worker_available[worker] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if finish == now {
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
                for &child in &successors[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest future event that can change eligibility.
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |current| current.min(time)));
            }
        };

        for item in &running {
            consider(item.finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &time in &worker_available {
            consider(time);
        }

        now = next_time.expect("unfinished tasks must have a future event");
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();
    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

                    task: task_id,
                    position,

                    task: task_id,
                    position,
                    value,

                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
                for &child in &children[task_id] {
                    remaining_preds[child] -= 1;
                }
            } else {
                running.push((finish, task_id, worker_id));
            }
            admitted = true;
        }

        if completed_count == n {
            break;
        }

        // If work is running, releases and cooldowns can be skipped when a finish
        // already occurs earlier; otherwise advance to the earliest relevant event.
        let next_finish = running.iter().map(|entry| entry.0).min();
        let next_release = (0..n)
            .filter(|&id| !started[id] && tasks[id].release > now)
            .map(|id| tasks[id].release)
            .min();
        let next_worker = worker_available
            .iter()
            .copied()
            .filter(|&t| t > now)
            .min();

        let next = [next_finish, next_release, next_worker]
            .into_iter()
            .flatten()
            .min();

        if let Some(time) = next {
            now = time;
        } else if !admitted {
            // All tasks are either already completed or would have been schedulable;
            // this guard prevents an accidental non-progress loop.
            break;
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("acyclic tasks are all scheduled"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

use std::cmp::Reverse;
use std::collections::BinaryHeap;

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate all dependency entries before other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Smallest available ID first Kahn traversal.
    let mut available = BinaryHeap::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(Reverse(child));
            }
        }
    }

    if order.len() != n {
        let removed: Vec<bool> = {
            let mut v = vec![false; n];
            for &id in &order {
                v[id] = true;
            }
            v
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_preds: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();

    // Running tasks keyed by (finish time, task ID).
    let mut running: BinaryHeap<Reverse<(u64, usize, usize)>> = BinaryHeap::new();
    let mut worker_free = vec![0u64; workers];
    let mut now = 0u64;
    let mut done_count = 0usize;
    let mut makespan = 0u64;

    while done_count < n {
        // Complete all positive-duration tasks finishing now, ordered by task ID.
        while let Some(Reverse((finish, task_id, worker_id))) = running.peek().copied() {
            if finish != now {
                break;
            }
            running.pop();
            completed[task_id] = true;
            done_count += 1;
            makespan = makespan.max(finish);
            events.push(Event {
                time: finish,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });
            for &child in &children[task_id] {
                remaining_preds[child] -= 1;
            }
        }

        loop {
            let worker_id = match (0..workers).find(|&w| worker_free[w] <= now) {
                Some(w) => w,
                None => break,
            };

            let best = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && remaining_preds[id] == 0
                })
                .max_by(|&a, &b| {
                    tails[a]
                        .cmp(&tails[b])
                        .then_with(|| tasks[a].priority.cmp(&tasks[b].priority))
                        .then_with(|| b.cmp(&a))
                });

            let task_id = match best {
                Some(id) => id,
                None => break,
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let free_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_free[worker_id] = free_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker: worker_id,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });

            if finish == now {
                // Immediate completion of a zero-duration task.
                completed[task_id] = true;
                done_count += 1;
                makespan = makespan.max(finish);
                events.push(Event {
                    time: finish,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
                for &child in &children[task_id] {
                    remaining_preds[child] -= 1;
                }
            } else {
                running.push(Reverse((finish, task_id, worker_id)));
            }
        }

        if done_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        if let Some(Reverse((finish, _, _))) = running.peek().copied() {
            if finish > now {
                next_time = Some(finish);
            }
        }

        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next_time = Some(next_time.map_or(tasks[id].release, |t| t.min(tasks[id].release)));
            }
        }

        for &free_at in &worker_free {
            if free_at > now {
                next_time = Some(next_time.map_or(free_at, |t| t.min(free_at)));
            }
        }

        // A blocked task may become eligible only when a running predecessor finishes.
        if let Some(next) = next_time {
            now = next;
        } else {
            // With an acyclic graph, this can only occur when no tasks remain.
            break;
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("every task is started exactly once"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
