#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: std::collections::HashMap<String, Entry>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = std::collections::HashMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self { entries }
    }

    fn validate_path(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut components = Vec::new();
        let mut start = 1;
        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                if start == i {
                    return Err(Error::InvalidPath(i.saturating_sub(1)));
                }
                let component = &path[start..i];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component);
                start = i + 1;
            }
        }
        Ok(components)
    }

    fn canonical(components: &[&str]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            let mut path = String::new();
            for component in components {
                path.push('/');
                path.push_str(component);
            }
            path
        }
    }

    fn parent_path(components: &[&str]) -> String {
        Self::canonical(&components[..components.len() - 1])
    }

    fn check_parent(&self, components: &[&str]) -> Result<String, Error> {
        let parent_components = &components[..components.len() - 1];
        let mut current = String::from("/");
        for (i, component) in parent_components.iter().enumerate() {
            if current != "/" {
                current.push('/');
            }
            current.push_str(component);

            match self.entries.get(&current) {
                None => return Err(Error::Missing(current)),
                Some(Entry::File(_)) if i + 1 < parent_components.len() => {
                    return Err(Error::NotDir(current));
                }
                Some(Entry::File(_)) => return Err(Error::NotDir(current)),
                Some(Entry::Dir) => {}
            }
        }
        Ok(Self::parent_path(components))
    }

    fn resolve_final(&self, components: &[&str]) -> Result<String, Error> {
        let mut current = String::from("/");
        for (i, component) in components.iter().enumerate() {
            if current != "/" {
                current.push('/');
            }
            current.push_str(component);

            match self.entries.get(&current) {
                None => return Err(Error::Missing(current)),
                Some(Entry::File(_)) if i + 1 < components.len() => {
                    return Err(Error::NotDir(current));
                }
                Some(_) => {}
            }
        }
        Ok(current)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::validate_path(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }
        let parent = self.check_parent(&components)?;
        let canonical = Self::canonical(&components);
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::validate_path(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }
        let parent = self.check_parent(&components)?;
        let canonical = Self::canonical(&components);
        match self.entries.get(&canonical) {
            Some(Entry::Dir) => return Err(Error::IsDir(canonical)),
            Some(Entry::File(_)) => {
                self.entries.insert(canonical, Entry::File(data.to_vec()));
                return Ok(());
            }
            None => {}
        }
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        self.entries.insert(canonical, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::validate_path(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }
        let canonical = self.resolve_final(&components)?;
        match self.entries.get(&canonical) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            None => Err(Error::Missing(canonical)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::validate_path(path)?;
        let canonical = self.resolve_final(&components)?;
        match self.entries.get(&canonical) {
            Some(Entry::File(_)) => return Err(Error::NotDir(canonical)),
            Some(Entry::Dir) => {}
            None => return Err(Error::Missing(canonical)),
        }

        let prefix = if canonical == "/" {
            "/".to_string()
        } else {
            format!("{canonical}/")
        };
        let mut children = Vec::new();
        for (entry_path, entry) in &self.entries {
            if let Some(rest) = entry_path.strip_prefix(&prefix) {
                if !rest.is_empty() && !rest.contains('/') {
                    let kind = match entry {
                        Entry::Dir => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    children.push((rest.to_string(), kind));
                }
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }
}
