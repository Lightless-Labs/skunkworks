#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = remainder > half
        || (denominator % 2 == 0 && remainder == half && quotient % 2 == 1);
    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct CustomerState {
    customer: Customer,
    deadline: Option<u64>,
    queue_entry: u64,
    waited: u64,
    remaining: u64,
    ever_started: bool,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    dispatch_time: u64,
    protected_until: u64,
}

fn push_event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn waiting_better(a: usize, b: usize, states: &[CustomerState]) -> bool {
    let x = &states[a];
    let y = &states[b];
    if x.customer.priority != y.customer.priority {
        return x.customer.priority < y.customer.priority;
    }
    match (x.paused, y.paused) {
        (true, false) => true,
        (false, true) => false,
        (true, true) => {
            if x.pause_time != y.pause_time {
                x.pause_time < y.pause_time
            } else {
                x.customer.id < y.customer.id
            }
        }
        (false, false) => {
            if x.customer.arrival != y.customer.arrival {
                x.customer.arrival < y.customer.arrival
            } else {
                x.customer.id > y.customer.id
            }
        }
    }
}

fn best_waiter(waiting: &[usize], states: &[CustomerState]) -> Option<usize> {
    let mut best = None;
    for &i in waiting {
        if best.map_or(true, |b| waiting_better(i, b, states)) {
            best = Some(i);
        }
    }
    best
}

fn expire_waiters(
    now: u64,
    waiting: &mut Vec<usize>,
    states: &mut [CustomerState],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<(u64, u32, usize)> = waiting
        .iter()
        .filter_map(|&i| states[i].deadline.filter(|&d| d <= now).map(|d| (d, states[i].customer.id, i)))
        .collect();
    expired.sort_by_key(|&(deadline, id, _)| (deadline, id));
    for (_, _, i) in expired {
        if let Some(pos) = waiting.iter().position(|&x| x == i) {
            waiting.remove(pos);
            let state = &mut states[i];
            let segment = now - state.queue_entry;
            state.waited += segment;
            stats.wait_ticks += segment as u128;
            stats.reneged += 1;
            push_event(events, now, state.customer.id, Kind::Renege { waited: state.waited });
        }
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut seen = Vec::new();
    let mut deadlines = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&customer.id) {
            return Err(Error::DuplicateId { index, id: customer.id });
        }
        seen.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority { index, value: customer.priority });
        }
        let deadline = match customer.patience {
            Some(p) => Some(customer.arrival.checked_add(p).ok_or(
                Error::DeadlineOverflow { index, id: customer.id },
            )?),
            None => None,
        };
        deadlines.push(deadline);
    }

    let mut states: Vec<CustomerState> = customers
        .iter()
        .zip(deadlines)
        .map(|(customer, deadline)| CustomerState {
            customer: customer.clone(),
            deadline,
            queue_entry: customer.arrival,
            waited: 0,
            remaining: customer.service,
            ever_started: false,
            paused: false,
            pause_time: 0,
        })
        .collect();

    let mut arrival_order: Vec<usize> = (0..states.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        let x = &states[a].customer;
        let y = &states[b].customer;
        (x.arrival, x.priority, std::cmp::Reverse(x.id))
            .cmp(&(y.arrival, y.priority, std::cmp::Reverse(y.id)))
    });

    let mut events = Vec::new();
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut arrival_pos = 0usize;
    let mut last_dispatched: Option<usize> = None;
    let mut now = states
        .iter()
        .map(|s| s.customer.arrival)
        .min()
        .unwrap_or(0);

    loop {
        // A: finish at the current time.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let state = &states[a.index];
            push_event(&mut events, now, state.customer.id, Kind::Finish);
            stats.completed += 1;
            stats.completion_order.push(state.customer.id);
            stats.turnaround_ticks += (now - state.customer.arrival) as u128;
        }

        // B: waiting deadlines already due.
        expire_waiters(now, &mut waiting, &mut states, &mut events, &mut stats);

        // C: process the whole arrival cohort. Admission uses the server state
        // before dispatch and reserves the idle virtual slot through the cohort.
        while arrival_pos < arrival_order.len()
            && states[arrival_order[arrival_pos]].customer.arrival == now
        {
            let i = arrival_order[arrival_pos];
            arrival_pos += 1;
            let id = states[i].customer.id;
            push_event(&mut events, now, id, Kind::Arrive);
            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= limit {
                stats.balked += 1;
                push_event(&mut events, now, id, Kind::Balk);
            } else {
                stats.accepted += 1;
                states[i].queue_entry = now;
                waiting.push(i);
            }
        }

        // D: preempt only after switching protection ends.
        if let Some(a) = active.as_ref() {
            if now >= a.protected_until {
                if let Some(candidate) = best_waiter(&waiting, &states) {
                    if states[candidate].customer.priority < states[a.index].customer.priority {
                        let a = active.take().unwrap();
                        let remaining = a.finish - now;
                        let by = states[candidate].customer.id;
                        let paused_id = states[a.index].customer.id;
                        push_event(&mut events, now, paused_id, Kind::Preempt { remaining, by });
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch_time) as u128;

                        let patience = states[a.index].customer.patience;
                        let deadline = match patience {
                            Some(_) => {
                                let budget = states[a.index]
                                    .deadline
                                    .map(|d| d.saturating_sub(now))
                                    .unwrap_or(0);
                                Some(now.checked_add(budget).ok_or(Error::TimeOverflow {
                                    customer: paused_id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                })?)
                            }
                            None => None,
                        };
                        let state = &mut states[a.index];
                        state.remaining = remaining;
                        state.queue_entry = now;
                        state.pause_time = now;
                        state.paused = true;
                        state.deadline = deadline;
                        waiting.push(a.index);
                    }
                }
            }
        }

        // E: inline dispatches and zero-time finishes.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&waiting, &states) else {
                break;
            };
            waiting.retain(|&x| x != i);
            let state = &mut states[i];
            let segment = now - state.queue_entry;
            state.waited += segment;
            stats.wait_ticks += segment as u128;
            let remaining = state.remaining;
            let first = !state.ever_started;
            let cost = if last_dispatched.is_some_and(|last| last != i) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: state.customer.id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: state.customer.id,
                time: now,
                stage: OverflowStage::Finish,
            })?;
            let id = state.customer.id;
            if first {
                stats.started += 1;
                state.ever_started = true;
                push_event(&mut events, now, id, Kind::Start { remaining, work_at, finish });
            } else {
                push_event(&mut events, now, id, Kind::Resume { remaining, work_at, finish });
            }
            state.paused = false;
            last_dispatched = Some(i);
            active = Some(Active {
                index: i,
                work_at,
                finish,
                dispatch_time: now,
                protected_until: work_at,
            });

            if finish == now {
                let a = active.take().unwrap();
                stats.busy_ticks += (now - a.dispatch_time) as u128;
                push_event(&mut events, now, id, Kind::Finish);
                stats.completed += 1;
                stats.completion_order.push(id);
                stats.turnaround_ticks += (now - states[i].customer.arrival) as u128;
                continue;
            }
        }

        // F: expire newly queued waiters.
        expire_waiters(now, &mut waiting, &mut states, &mut events, &mut stats);

        if arrival_pos == arrival_order.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next: Option<u64> = None;
        let mut consider = |t: u64| {
            if t > now && next.map_or(true, |old| t < old) {
                next = Some(t);
            }
        };
        if arrival_pos < arrival_order.len() {
            consider(states[arrival_order[arrival_pos]].customer.arrival);
        }
        if let Some(a) = &active {
            consider(a.finish);
            if a.work_at > now {
                consider(a.work_at);
            }
        }
        for &i in &waiting {
            if let Some(deadline) = states[i].deadline {
                consider(deadline);
            }
        }
        let Some(next_time) = next else {
            break;
        };
        now = next_time;
    }

    stats.elapsed = events.last().map(|e| e.time).unwrap_or(0);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}
