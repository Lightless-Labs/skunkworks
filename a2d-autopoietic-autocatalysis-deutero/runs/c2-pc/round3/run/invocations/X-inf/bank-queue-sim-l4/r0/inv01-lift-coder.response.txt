use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        r > half || (r == half && q % 2 == 1)
    } else {
        r > half
    };
    Some(q + u128::from(round_up))
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    entry: u64,
    waited: u64,
    ever_started: bool,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Active {
    job: Job,
    dispatch: u64,
    work_at: u64,
    finish: u64,
}

fn event(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn waiting_cmp(a: &Job, b: &Job) -> Ordering {
    a.customer.priority.cmp(&b.customer.priority)
        .then_with(|| match (a.paused, b.paused) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (true, true) => a.pause_time.cmp(&b.pause_time)
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (false, false) => a.customer.arrival.cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn consume_wait(job: &mut Job, now: u64, wait_ticks: &mut u128) {
    let segment = now - job.entry;
    job.waited += segment;
    *wait_ticks += segment as u128;
    if let Some(deadline) = job.deadline {
        let budget = deadline - job.entry;
        job.deadline = Some(now + (budget - segment));
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut seen = Vec::with_capacity(customers.len());
    let mut deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        seen.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow { index, id: c.id })?),
            None => None,
        };
        deadlines.push(deadline);
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| customers[a].arrival.cmp(&customers[b].arrival)
        .then_with(|| customers[a].priority.cmp(&customers[b].priority))
        .then_with(|| customers[b].id.cmp(&customers[a].id)));

    let mut stats = Stats {
        accepted: 0, balked: 0, reneged: 0, completed: 0, started: 0,
        preemptions: 0, elapsed: 0, busy_ticks: 0, wait_ticks: 0,
        turnaround_ticks: 0, mean_wait_milli: 0, mean_turnaround_milli: 0,
        utilization_bps: 0, completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut active: Option<Active> = None;
    let mut arrival_pos = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut now = 0u64;

    loop {
        if let Some(a) = &active {
            if a.finish == now {
                let a = active.take().unwrap();
                stats.busy_ticks += (now - a.dispatch) as u128;
                stats.completed += 1;
                stats.turnaround_ticks += (now - a.job.customer.arrival) as u128;
                stats.completion_order.push(a.job.customer.id);
                event(&mut events, now, a.job.customer.id, Kind::Finish);
            }
        }

        let mut expired: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, j)| j.deadline.filter(|&d| d <= now).map(|d| (i, d)))
            .map(|(i, _)| i).collect();
        expired.sort_by(|&a, &b| waiting[a].deadline.unwrap().cmp(&waiting[b].deadline.unwrap())
            .then_with(|| waiting[a].customer.id.cmp(&waiting[b].customer.id)));
        for i in expired.into_iter().rev() {
            let mut j = waiting.remove(i);
            consume_wait(&mut j, now, &mut stats.wait_ticks);
            stats.reneged += 1;
            event(&mut events, now, j.customer.id, Kind::Renege { waited: j.waited });
        }

        while arrival_pos < arrivals.len() && customers[arrivals[arrival_pos]].arrival == now {
            let idx = arrivals[arrival_pos];
            arrival_pos += 1;
            let c = &customers[idx];
            event(&mut events, now, c.id, Kind::Arrive);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                stats.balked += 1;
                event(&mut events, now, c.id, Kind::Balk);
            } else {
                stats.accepted += 1;
                waiting.push(Job {
                    customer: c.clone(),
                    deadline: deadlines[idx],
                    remaining: c.service,
                    entry: now,
                    waited: 0,
                    ever_started: false,
                    paused: false,
                    pause_time: 0,
                });
            }
        }

        if let Some(a) = &active {
            if now >= a.work_at {
                if let Some((best_i, _)) = waiting.iter().enumerate()
                    .min_by(|(_, x), (_, y)| waiting_cmp(x, y))
                {
                    if waiting[best_i].customer.priority < a.job.customer.priority {
                        let waiter = waiting[best_i].customer.id;
                        let mut paused = active.take().unwrap();
                        stats.busy_ticks += (now - paused.dispatch) as u128;
                        let remaining = paused.finish - now;
                        event(&mut events, now, paused.job.customer.id, Kind::Preempt { remaining, by: waiter });
                        stats.preemptions += 1;
                        paused.job.remaining = remaining;
                        paused.job.entry = now;
                        paused.job.paused = true;
                        paused.job.pause_time = now;
                        if let Some(budget) = paused.job.customer.patience {
                            let consumed = paused.job.deadline.unwrap() - paused.job.entry;
                            let remaining_budget = budget.saturating_sub(consumed);
                            paused.job.deadline = Some(now.checked_add(remaining_budget).ok_or(
                                Error::TimeOverflow { customer: paused.job.customer.id, time: now, stage: OverflowStage::Patience }
                            )?);
                        }
                        waiting.push(paused.job);
                    }
                }
            }
        }

        loop {
            if active.is_some() || waiting.is_empty() {
                break;
            }
            let best_i = (0..waiting.len()).min_by(|&a, &b| waiting_cmp(&waiting[a], &waiting[b])).unwrap();
            let mut job = waiting.remove(best_i);
            consume_wait(&mut job, now, &mut stats.wait_ticks);
            let switching = match last_dispatched {
                None => 0,
                Some(id) if id == job.customer.id => 0,
                Some(_) => config.switch_cost,
            };
            let work_at = now.checked_add(switching).ok_or(Error::TimeOverflow {
                customer: job.customer.id, time: now, stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: job.customer.id, time: now, stage: OverflowStage::Finish,
            })?;
            let kind = if job.ever_started {
                Kind::Resume { remaining: job.remaining, work_at, finish }
            } else {
                job.ever_started = true;
                stats.started += 1;
                Kind::Start { remaining: job.remaining, work_at, finish }
            };
            last_dispatched = Some(job.customer.id);
            event(&mut events, now, job.customer.id, kind);
            active = Some(Active { job, dispatch: now, work_at, finish });
            if finish == now {
                let a = active.take().unwrap();
                stats.completed += 1;
                stats.turnaround_ticks += (now - a.job.customer.arrival) as u128;
                stats.completion_order.push(a.job.customer.id);
                event(&mut events, now, a.job.customer.id, Kind::Finish);
                continue;
            }
        }

        let mut expired: Vec<usize> = waiting.iter().enumerate()
            .filter_map(|(i, j)| j.deadline.filter(|&d| d <= now).map(|d| (i, d)))
            .map(|(i, _)| i).collect();
        expired.sort_by(|&a, &b| waiting[a].deadline.unwrap().cmp(&waiting[b].deadline.unwrap())
            .then_with(|| waiting[a].customer.id.cmp(&waiting[b].customer.id)));
        for i in expired.into_iter().rev() {
            let mut j = waiting.remove(i);
            consume_wait(&mut j, now, &mut stats.wait_ticks);
            stats.reneged += 1;
            event(&mut events, now, j.customer.id, Kind::Renege { waited: j.waited });
        }

        if arrival_pos == arrivals.len() && active.is_none() && waiting.is_empty() {
            break;
        }

        let mut next = None;
        let mut offer = |t: u64| {
            if t > now && next.map_or(true, |n| t < n) {
                next = Some(t);
            }
        };
        if arrival_pos < arrivals.len() {
            offer(customers[arrivals[arrival_pos]].arrival);
        }
        if let Some(a) = &active {
            offer(a.finish);
            if a.work_at > now { offer(a.work_at); }
        }
        for j in &waiting {
            if let Some(d) = j.deadline { offer(d); }
        }
        match next {
            Some(t) => now = t,
            None => break,
        }
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 { 0 } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 { 0 } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 { 0 } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}
