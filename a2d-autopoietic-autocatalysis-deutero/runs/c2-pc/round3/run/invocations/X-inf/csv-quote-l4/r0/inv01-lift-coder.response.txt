use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record = 0usize;
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut opening_quote = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if state != 2 && b == b'\r' {
            if i + 1 >= bytes.len() || bytes[i + 1] != b'\n' {
                return Err(CsvError::BareCr { record, byte: i });
            }
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            state = 0;
            record += 1;
            i += 2;
            continue;
        }

        match state {
            0 => {
                if b == delimiter {
                    row.push(String::new());
                    i += 1;
                } else if b == quote {
                    state = 2;
                    opening_quote = i;
                    i += 1;
                } else if b == b'\n' {
                    row.push(String::new());
                    records.push(std::mem::take(&mut row));
                    record += 1;
                    i += 1;
                } else {
                    field.push(b as char);
                    state = 1;
                    i += 1;
                }
            }
            1 => {
                if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = 0;
                    i += 1;
                } else if b == b'\n' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    state = 0;
                    record += 1;
                    i += 1;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote { record, byte: i });
                } else {
                    field.push(b as char);
                    i += 1;
                }
            }
            2 => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = 3;
                        i += 1;
                    }
                } else {
                    field.push(b as char);
                    i += 1;
                }
            }
            _ => {
                if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = 0;
                    i += 1;
                } else if b == b'\n' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    state = 0;
                    record += 1;
                    i += 1;
                } else {
                    return Err(CsvError::TrailingByte { record, byte: i });
                }
            }
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: opening_quote,
        });
    }

    if !row.is_empty() || state != 0 || !field.is_empty() {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name.as_str(), second);
    }
    None
}

fn table_from_records(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut iter = records.into_iter();
            (iter.next(), iter.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header(h) {
                return Err(error);
            }
        }
        for (index, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote).map_err(|component| {
            CsvError::InvalidConfig { component }
        })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| table_from_records(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(|c| c.is_ascii_digit())
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let negative = b[0] == b'-';
    let start = usize::from(negative);
    if start >= b.len() {
        return false;
    }
    let dot = match b[start..].iter().position(|&c| c == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    if dot + 3 != b.len() || !b[dot + 1..].iter().all(|c| c.is_ascii_digit()) {
        return false;
    }
    let integral = &b[start..dot];
    if integral.is_empty() {
        return false;
    }
    let valid_integral = if integral[0] == b'0' {
        integral.len() == 1
    } else {
        (b'1'..=b'9').contains(&integral[0])
            && integral[1..].iter().all(|c| c.is_ascii_digit())
    };
    if !valid_integral {
        return false;
    }
    !(negative && integral == b"0" && b[dot + 1..] == *b"00")
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = s.starts_with('-');
    let unsigned = if negative { &s[1..] } else { s };
    let (whole, frac) = unsigned.split_once('.').unwrap();
    let whole_value = whole
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let frac_value = (frac.as_bytes()[0] - b'0') as i128 * 10
        + (frac.as_bytes()[1] - b'0') as i128;
    let magnitude = whole_value
        .checked_mul(100)
        .and_then(|v| v.checked_add(frac_value))
        .ok_or(CellErrorKind::DecimalOverflow)?;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn convert_cell(
    text: &str,
    kind: ColumnType,
    nullable: bool,
) -> Result<Value, CellErrorKind> {
    if text.is_empty() {
        if nullable {
            return Ok(Value::Null);
        }
        if kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match kind {
        ColumnType::Text => Ok(Value::Text(text.to_string())),
        ColumnType::Integer => parse_integer(text).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(text).map(Value::Decimal),
        ColumnType::Boolean => match text {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut output = Vec::with_capacity(table.rows.len());
    let mut errors = Vec::new();

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut values = vec![None; columns.len()];
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let column_index = match resolved {
                Some(index) => index,
                None => {
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    });
                    row_failed = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                    continue;
                }
            };

            let cell = match row.get(column_index) {
                Some(cell) => cell,
                None => {
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind: CellErrorKind::MissingCell,
                    });
                    row_failed = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                    continue;
                }
            };

            match convert_cell(cell, spec.kind, spec.nullable) {
                Ok(value) => values[spec_index] = Some(value),
                Err(kind) => {
                    errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column_index),
                        kind,
                    });
                    row_failed = true;
                    if policy != ErrorPolicy::Collect {
                        break;
                    }
                }
            }
        }

        output.push(values);
        if policy == ErrorPolicy::Stop && row_failed {
            break;
        }
    }

    TypedResult {
        rows: output,
        errors,
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding {
    Lf,
    CrLf,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode {
    Preserve,
    Canonical,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns { record: usize },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

fn validate_write_table(table: &Table) -> Result<(), WriteError> {
    let mut record_index = 0usize;
    if let Some(header) = &table.header {
        for (column, field) in header.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column,
                    byte,
                });
            }
        }
        record_index = 1;
    }
    for row in &table.rows {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column,
                    byte,
                });
            }
        }
        record_index += 1;
    }

    if let Some(header) = &table.header {
        for (index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    let mut index = 0usize;
    if let Some(header) = &table.header {
        if header.is_empty() {
            return Err(WriteError::ZeroColumns { record: 0 });
        }
        index = 1;
    }
    for row in &table.rows {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record: index });
        }
        index += 1;
    }

    Ok(())
}

fn duplicate_header_write(header: &[String]) -> Option<WriteError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(WriteError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name.as_str(), second);
    }
    None
}

fn canonical_records(table: &Table) -> Result<Vec<Vec<String>>, WriteError> {
    if let Some(header) = &table.header {
        if table.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header_write(header) {
                return Err(error);
            }
        }

        let mut retained = HashMap::<&str, usize>::new();
        for (index, name) in header.iter().enumerate() {
            match table.duplicates {
                DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                    retained.entry(name.as_str()).or_insert(index);
                }
                DuplicatePolicy::KeepLast => {
                    retained.insert(name.as_str(), index);
                }
            }
        }
        let mut columns: Vec<(&str, usize)> = retained.into_iter().collect();
        columns.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

        let new_header = columns.iter().map(|(name, _)| (*name).to_string()).collect();
        let mut rows = Vec::with_capacity(table.rows.len());
        for row in &table.rows {
            rows.push(columns.iter().map(|(_, index)| row[*index].clone()).collect());
        }
        rows.sort_by(|a: &Vec<String>, b: &Vec<String>| {
            a.iter()
                .map(String::as_bytes)
                .cmp(b.iter().map(String::as_bytes))
        });

        let mut records = Vec::with_capacity(rows.len() + 1);
        records.push(new_header);
        records.extend(rows);
        Ok(records)
    } else {
        let mut rows = table.rows.clone();
        rows.sort_by(|a, b| {
            a.iter()
                .map(String::as_bytes)
                .cmp(b.iter().map(String::as_bytes))
        });
        Ok(rows)
    }
}

fn needs_quote(
    field: &str,
    delimiter: u8,
    quote: u8,
    first_field_of_first_record: bool,
    sole_empty_field: bool,
) -> bool {
    let bytes = field.as_bytes();
    bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
        || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
        || (first_field_of_first_record && field.starts_with('\u{feff}'))
        || (sole_empty_field && field.is_empty())
}

fn emit_records(
    records: &[Vec<String>],
    delimiter: u8,
    quote: u8,
    ending: &[u8],
    final_terminator: bool,
) -> String {
    let mut output = String::new();
    for (record_index, row) in records.iter().enumerate() {
        if record_index != 0 {
            output.push_str(std::str::from_utf8(ending).unwrap());
        }
        let sole_empty = row.len() == 1 && row[0].is_empty();
        for (column_index, field) in row.iter().enumerate() {
            if column_index != 0 {
                output.push(delimiter as char);
            }
            let quote_it = needs_quote(
                field,
                delimiter,
                quote,
                record_index == 0 && column_index == 0,
                sole_empty,
            );
            if quote_it {
                output.push(quote as char);
                for ch in field.chars() {
                    if ch as u32 == quote as u32 {
                        output.push(quote as char);
                    }
                    output.push(ch);
                }
                output.push(quote as char);
            } else {
                output.push_str(field);
            }
        }
    }
    if final_terminator && !records.is_empty() {
        output.push_str(std::str::from_utf8(ending).unwrap());
    }
    output
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    validate_write_table(table)?;

    let records = match config.mode {
        WriteMode::Preserve => {
            let mut records = Vec::with_capacity(table.rows.len() + usize::from(table.header.is_some()));
            if let Some(header) = &table.header {
                records.push(header.clone());
            }
            records.extend(table.rows.clone());
            records
        }
        WriteMode::Canonical => canonical_records(table)?,
    };

    let (ending, final_terminator) = match config.mode {
        WriteMode::Preserve => (
            match config.line_ending {
                LineEnding::Lf => b"\n".as_slice(),
                LineEnding::CrLf => b"\r\n".as_slice(),
            },
            config.final_terminator,
        ),
        WriteMode::Canonical => (b"\n".as_slice(), true),
    };

    Ok(emit_records(
        &records,
        config.delimiter,
        config.quote,
        ending,
        final_terminator,
    ))
}
