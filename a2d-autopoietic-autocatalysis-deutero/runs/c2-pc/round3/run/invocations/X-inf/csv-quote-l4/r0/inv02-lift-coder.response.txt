use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart { Delimiter, Quote }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy { Reject, KeepFirst, KeepLast }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}
impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name),
            _ => header.iter().position(|v| v == name),
        }
    }
}

fn valid_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0 unquoted, 1 quoted, 2 closed quote
    let mut opening = 0usize;
    let mut record = 0usize;
    let mut started = false;

    while i < bytes.len() {
        let b = bytes[i];
        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }
        if state != 1 && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if state == 1 {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                    continue;
                }
                state = 2;
                i += 1;
                continue;
            }
            field.push(b as char);
            i += 1;
            continue;
        }

        if state == 2 {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                state = 0;
                started = true;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record += 1;
                state = 0;
                started = false;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte { record, byte: i });
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            started = true;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record += 1;
            started = false;
            i += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if !field.is_empty() || started {
                return Err(CsvError::UnexpectedQuote { record, byte: i });
            }
            opening = i;
            state = 1;
            started = true;
            i += 1;
        } else {
            field.push(b as char);
            started = true;
            i += 1;
        }
    }

    if state == 1 {
        return Err(CsvError::UnclosedQuote { record, byte: opening });
    }
    if started || !row.is_empty() || !field.is_empty() || state == 2 {
        row.push(field);
        records.push(row);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn table_from_records(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            (Some(records[0].clone()), records[1..].to_vec())
        }
    } else {
        (None, records)
    };

    if let Some(h) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: HashMap<&str, usize> = HashMap::new();
            for (second, name) in h.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(), first, second,
                    });
                }
                seen.insert(name, second);
            }
        }
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1, expected: h.len(), actual: row.len(),
                });
            }
        }
    }
    Ok(Table { header, rows, duplicates: config.duplicates })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}
impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        valid_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self { config, input: String::new(), cached: None })
    }
    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }
    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|r| table_from_records(r, self.config));
        self.cached = Some(result.clone());
        result
    }
    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}
pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ColumnRef { Index(usize), Name(String) }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType { Text, Integer, Decimal, Boolean }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value { Null, Text(String), Integer(i64), Decimal(i64), Boolean(bool) }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn, MissingCell, Empty, InvalidInteger, IntegerOverflow,
    InvalidDecimal, DecimalOverflow, InvalidBoolean,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy { Collect, FirstPerRow, Stop }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if start == b.len() { return false; }
    if b[start] == b'0' { return start == 0 && b.len() == 1; }
    (b'1'..=b'9').contains(&b[start]) && b[start+1..].iter().all(u8::is_ascii_digit)
}
fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() { return false; }
    let start = usize::from(b[0] == b'-');
    if b.len() < start + 4 { return false; }
    let dot = b.len() - 3;
    if b[dot] != b'.' || !b[dot+1..].iter().all(u8::is_ascii_digit) { return false; }
    if dot == start + 1 && b[start] == b'0' { true }
    else if dot > start + 1 && (b'1'..=b'9').contains(&b[start]) {
        b[start+1..dot].iter().all(u8::is_ascii_digit)
    } else { false }
}
fn evaluate(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable { return Ok(Value::Null); }
        if spec.kind != ColumnType::Text { return Err(CellErrorKind::Empty); }
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
        ColumnType::Integer => {
            if !integer_syntax(cell) { return Err(CellErrorKind::InvalidInteger); }
            cell.parse::<i64>().map(Value::Integer).map_err(|_| CellErrorKind::IntegerOverflow)
        }
        ColumnType::Decimal => {
            if !decimal_syntax(cell) { return Err(CellErrorKind::InvalidDecimal); }
            let neg = cell.starts_with('-');
            let digits = if neg { &cell[1..] } else { cell };
            let (whole, frac) = digits.split_once('.').unwrap();
            let magnitude = whole.parse::<i128>().unwrap() * 100 + frac.parse::<i128>().unwrap();
            let signed = if neg { -magnitude } else { magnitude };
            i64::try_from(signed).map(Value::Decimal).map_err(|_| CellErrorKind::DecimalOverflow)
        }
    }
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut result = TypedResult { rows: Vec::new(), errors: Vec::new() };
    for (ri, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut stopped = false;
        for (si, spec) in columns.iter().enumerate() {
            if stopped { out.push(None); continue; }
            let index = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(n) => table.column_index(n),
            };
            let evaluated = match index {
                None => Err((None, CellErrorKind::UnknownColumn)),
                Some(i) => match row.get(i) {
                    None => Err((Some(i), CellErrorKind::MissingCell)),
                    Some(cell) => evaluate(cell, spec).map_err(|e| (Some(i), e)),
                }
            };
            match evaluated {
                Ok(v) => out.push(Some(v)),
                Err((column, kind)) => {
                    out.push(None);
                    result.errors.push(CellError { row: ri, spec: si, column, kind });
                    if policy != ErrorPolicy::Collect { stopped = true; }
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(out);
                        return result;
                    }
                }
            }
        }
        result.rows.push(out);
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LineEnding { Lf, CrLf }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WriteMode { Preserve, Canonical }
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {
    pub delimiter: u8,
    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,
    pub mode: WriteMode,
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn bytewise_vec_cmp(a: &Vec<String>, b: &Vec<String>) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b) {
        let ord = x.as_bytes().cmp(y.as_bytes());
        if ord != std::cmp::Ordering::Equal { return ord; }
    }
    a.len().cmp(&b.len())
}
fn encode_field(s: &str, delimiter: u8, quote: u8, first: bool, sole_empty: bool) -> String {
    let b = s.as_bytes();
    let edge_space = b.first().is_some_and(|x| *x == b' ' || *x == b'\t')
        || b.last().is_some_and(|x| *x == b' ' || *x == b'\t');
    let bom = first && s.starts_with('\u{feff}');
    let must = s.contains(delimiter as char) || s.contains(quote as char)
        || s.contains('\r') || s.contains('\n') || edge_space || bom || sole_empty;
    if !must { return s.to_string(); }
    let mut out = String::new();
    out.push(quote as char);
    for ch in s.chars() {
        if ch as u32 == quote as u32 { out.push(quote as char); }
        out.push(ch);
    }
    out.push(quote as char);
    out
}
pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    valid_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original: Vec<&Vec<String>> = Vec::new();
    if let Some(h) = &table.header { original.push(h); }
    original.extend(table.rows.iter());
    for (record, row) in original.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|b| *b == 0) {
                return Err(WriteError::Nul { record, column, byte });
            }
        }
    }
    if let Some(h) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(WriteError::Width { record: i + 1, expected: h.len(), actual: row.len() });
            }
        }
    }
    for (record, row) in original.iter().enumerate() {
        if row.is_empty() { return Err(WriteError::ZeroColumns { record }); }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    let mut rows = table.rows.clone();
    let mut header = table.header.clone();
    if config.mode == WriteMode::Canonical {
        if let Some(h) = &header {
            if table.duplicates == DuplicatePolicy::Reject {
                let mut seen: HashMap<&str, usize> = HashMap::new();
                for (second, name) in h.iter().enumerate() {
                    if let Some(&first) = seen.get(name.as_str()) {
                        return Err(WriteError::DuplicateHeader { name: name.clone(), first, second });
                    }
                    seen.insert(name, second);
                }
            }
            let mut chosen: HashMap<String, usize> = HashMap::new();
            for (i, name) in h.iter().enumerate() {
                match table.duplicates {
                    DuplicatePolicy::KeepLast => { chosen.insert(name.clone(), i); }
                    _ => { chosen.entry(name.clone()).or_insert(i); }
                }
            }
            let mut cols: Vec<(String, usize)> = chosen.into_iter().collect();
            cols.sort_by(|a,b| a.0.as_bytes().cmp(b.0.as_bytes()));
            let indexes: Vec<usize> = cols.iter().map(|x| x.1).collect();
            header = Some(cols.into_iter().map(|x| x.0).collect());
            rows = rows.iter().map(|r| indexes.iter().map(|i| r[*i].clone()).collect()).collect();
        }
        rows.sort_by(bytewise_vec_cmp);
    }
    if let Some(h) = header { records.push(h); }
    records.extend(rows);

    if records.is_empty() { return Ok(String::new()); }
    let mut out = String::new();
    let sep = match (config.mode, config.line_ending) {
        (WriteMode::Canonical, _) | (_, LineEnding::Lf) => "\n",
        (_, LineEnding::CrLf) => "\r\n",
    };
    for (ri, row) in records.iter().enumerate() {
        if ri > 0 { out.push_str(sep); }
        for (ci, field) in row.iter().enumerate() {
            if ci > 0 { out.push(config.delimiter as char); }
            out.push_str(&encode_field(
                field, config.delimiter, config.quote,
                ri == 0 && ci == 0, row.len() == 1 && field.is_empty(),
            ));
        }
    }
    if config.mode == WriteMode::Canonical || config.final_terminator {
        out.push_str(sep);
    }
    Ok(out)
}
