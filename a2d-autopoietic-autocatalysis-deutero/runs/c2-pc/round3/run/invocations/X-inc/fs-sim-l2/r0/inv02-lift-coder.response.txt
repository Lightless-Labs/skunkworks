```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
    cwd: Vec<String>,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
            cwd: Vec::new(),
        }
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                name => components.push(name.to_owned()),
            }
        }
        Ok(components)
    }

    fn lookup<'a>(&'a self, components: &[String]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        let mut prefix = Vec::new();

        for (i, component) in components.iter().enumerate() {
            match current {
                Entry::Dir(children) => {
                    prefix.push(component.clone());
                    current = children
                        .get(component)
                        .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
                    if i + 1 < components.len() && !matches!(current, Entry::Dir(_)) {
                        return Err(Error::NotDir(Self::canonical(&prefix)));
                    }
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical(&prefix)));
                }
            }
        }
        Ok(current)
    }

    fn lookup_parent_mut<'a>(
        &'a mut self,
        components: &[String],
    ) -> Result<(&'a mut BTreeMap<String, Entry>, String), Error> {
        let name = components
            .last()
            .expect("non-root path has a final component")
            .clone();
        let mut current = &mut self.root;
        let mut prefix = Vec::new();

        for component in &components[..components.len() - 1] {
            match current {
                Entry::Dir(children) => {
                    prefix.push(component.clone());
                    current = children
                        .get_mut(component)
                        .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
                    if !matches!(current, Entry::Dir(_)) {
                        return Err(Error::NotDir(Self::canonical(&prefix)));
                    }
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical(&prefix)));
                }
            }
        }

        match current {
            Entry::Dir(children) => Ok((children, name)),
            Entry::File(_) => Err(Error::NotDir(Self::canonical(
                &components[..components.len() - 1],
            ))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let full_path = Self::canonical(&components);
        let (parent, name) = self.lookup_parent_mut(&components)?;
        if parent.contains_key(&name) {
            return Err(Error::Exists(full_path));
        }
        parent.insert(name, Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let full_path = Self::canonical(&components);
        let (parent, name) = self.lookup_parent_mut(&components)?;
        match parent.get_mut(&name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(full_path)),
            Some(Entry::File(contents)) => {
                *contents = data.to_vec();
                Ok(())
            }
            None => {
                parent.insert(name, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = self.normalize(path)?;
        let full_path = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir(_) => Err(Error::IsDir(full_path)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = self.normalize(path)?;
        let full_path = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(full_path)),
        }
    }

    pub fn cwd(&self) -> String {
        Self::canonical(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        let full_path = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::Dir(_) => {
                self.cwd = components;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(full_path)),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
