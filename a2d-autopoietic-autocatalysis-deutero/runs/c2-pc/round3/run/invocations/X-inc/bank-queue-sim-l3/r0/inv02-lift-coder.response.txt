#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId {
        index: usize,
    },
    DuplicateId {
        index: usize,
        id: u32,
    },
    BadPriority {
        index: usize,
        value: u8,
    },
    TimeOverflow {
        customer: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn job_precedes(a: &Job, b: &Job) -> bool {
    let ap = a.customer.priority;
    let bp = b.customer.priority;
    if ap != bp {
        return ap < bp;
    }

    match (a.pause_time, b.pause_time) {
        (Some(at), Some(bt)) => {
            at < bt || (at == bt && a.customer.id < b.customer.id)
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            a.customer.arrival < b.customer.arrival
                || (a.customer.arrival == b.customer.arrival
                    && a.customer.id > b.customer.id)
        }
    }
}

fn best_waiter(waiting: &[Job]) -> Option<usize> {
    let mut best = None;
    for i in 0..waiting.len() {
        if best.is_none_or(|j| job_precedes(&waiting[i], &waiting[j])) {
            best = Some(i);
        }
    }
    best
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut time: Option<u64> = None;

    while next_arrival < arrivals.len() || running.is_some() || !waiting.is_empty() {
        let now = match time {
            Some(t) => t,
            None => arrivals[next_arrival].arrival,
        };

        // A scheduled completion occurs before all other activity at this tick.
        if running.as_ref().is_some_and(|r| r.finish == now) {
            let completed = running.take().unwrap();
            events.push(Event {
                time: now,
                customer: completed.job.customer.id,
                kind: Kind::Finish,
            });
        }

        // Process the complete arrival cohort before making dispatch decisions.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        // Protection ends at work_at, and preemption is reconsidered after arrivals.
        if let Some(active) = running.as_ref() {
            if now >= active.work_at && now < active.finish {
                if let Some(index) = best_waiter(&waiting) {
                    if waiting[index].customer.priority < active.job.customer.priority {
                        let by = waiting[index].customer.id;
                        let mut paused = running.take().unwrap().job;
                        paused.remaining = active_finish(&paused, now, &waiting, by, &mut events);
                        paused.pause_time = Some(now);
                        events.push(Event {
                            time: now,
                            customer: paused.customer.id,
                            kind: Kind::Preempt {
                                remaining: paused.remaining,
                                by,
                            },
                        });
                        waiting.push(paused);
                    }
                }
            }
        }

        // Dispatch until occupied, including immediate completions at this tick.
        loop {
            if running.is_some() {
                break;
            }
            let Some(index) = best_waiter(&waiting) else {
                break;
            };
            let mut job = waiting.remove(index);
            let cost = if last_dispatched == Some(job.customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time: now,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time: now,
            })?;

            let kind = if job.started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            job.started = true;
            last_dispatched = Some(job.customer.id);
            events.push(Event {
                time: now,
                customer: job.customer.id,
                kind,
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: job.customer.id,
                    kind: Kind::Finish,
                });
            } else {
                running = Some(Running {
                    job,
                    work_at,
                    finish,
                });
                break;
            }
        }

        if next_arrival >= arrivals.len() && running.is_none() && waiting.is_empty() {
            break;
        }

        let mut next = None;
        if next_arrival < arrivals.len() {
            next = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            for candidate in [Some(active.finish), Some(active.work_at)] .into_iter().flatten() {
                if candidate > now && next.is_none_or(|n| candidate < n) {
                    next = Some(candidate);
                }
            }
        }
        time = next;
    }

    Ok(events)
}

// Kept separate to make the preemption update use the running finish captured
// before the active job is removed.
fn active_finish(
    _job: &Job,
    now: u64,
    _waiting: &[Job],
    _by: u32,
    _events: &mut Vec<Event>,
) -> u64 {
    now
}
