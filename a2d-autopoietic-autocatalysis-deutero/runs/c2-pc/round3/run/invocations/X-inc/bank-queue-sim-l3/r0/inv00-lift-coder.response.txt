#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Running {
    customer: Customer,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiter_precedes(a: &Waiting, b: &Waiting) -> bool {
    if a.customer.priority != b.customer.priority {
        return a.customer.priority < b.customer.priority;
    }
    if a.started != b.started {
        return a.started;
    }
    if a.started {
        if a.pause_time != b.pause_time {
            return a.pause_time < b.pause_time;
        }
        a.customer.id < b.customer.id
    } else if a.customer.arrival != b.customer.arrival {
        a.customer.arrival < b.customer.arrival
    } else {
        a.customer.id > b.customer.id
    }
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    // Validate the whole input in the specified precedence order.
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if customers[..index].iter().any(|prior| prior.id == customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut now = arrivals[0].arrival;

    loop {
        // Scheduled completion is emitted before anything else at this tick.
        if running.as_ref().is_some_and(|r| r.finish == now) {
            let completed = running.take().unwrap();
            events.push(Event {
                time: now,
                customer: completed.customer.id,
                kind: Kind::Finish,
            });
        }

        // Add the entire arrival cohort before considering dispatch or preemption.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: 0,
            });
            next_arrival += 1;
        }

        // At the work boundary protection has ended. A strictly superior
        // waiter may preempt the running customer.
        let may_preempt = running.as_ref().is_some_and(|r| now >= r.work_at && now < r.finish);
        if may_preempt && !waiting.is_empty() {
            let best_index = (0..waiting.len())
                .min_by(|&a, &b| {
                    if waiter_precedes(&waiting[a], &waiting[b]) {
                        std::cmp::Ordering::Less
                    } else if waiter_precedes(&waiting[b], &waiting[a]) {
                        std::cmp::Ordering::Greater
                    } else {
                        std::cmp::Ordering::Equal
                    }
                })
                .unwrap();
            let best_priority = waiting[best_index].customer.priority;
            if best_priority < running.as_ref().unwrap().customer.priority {
                let by = waiting[best_index].customer.id;
                let mut paused = running.take().unwrap();
                let remaining = paused.finish - now;
                events.push(Event {
                    time: now,
                    customer: paused.customer.id,
                    kind: Kind::Preempt { remaining, by },
                });
                waiting.push(Waiting {
                    customer: paused.customer,
                    remaining,
                    started: true,
                    pause_time: now,
                });
            }
        }

        // Fill an idle server. Inline completions repeat at the same tick.
        if running.is_none() {
            loop {
                if waiting.is_empty() {
                    break;
                }
                let best_index = (0..waiting.len())
                    .min_by(|&a, &b| {
                        if waiter_precedes(&waiting[a], &waiting[b]) {
                            std::cmp::Ordering::Less
                        } else if waiter_precedes(&waiting[b], &waiting[a]) {
                            std::cmp::Ordering::Greater
                        } else {
                            std::cmp::Ordering::Equal
                        }
                    })
                    .unwrap();
                let selected = waiting.remove(best_index);
                let cost = if last_dispatched == Some(selected.customer.id) {
                    0
                } else if last_dispatched.is_none() {
                    0
                } else {
                    switch_cost
                };
                let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                    customer: selected.customer.id,
                    time: now,
                })?;
                let finish = work_at.checked_add(selected.remaining).ok_or(Error::TimeOverflow {
                    customer: selected.customer.id,
                    time: now,
                })?;

                let kind = if selected.started {
                    Kind::Resume {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining: selected.remaining,
                        work_at,
                        finish,
                    }
                };
                events.push(Event {
                    time: now,
                    customer: selected.customer.id,
                    kind,
                });
                last_dispatched = Some(selected.customer.id);

                if finish == now {
                    events.push(Event {
                        time: now,
                        customer: selected.customer.id,
                        kind: Kind::Finish,
                    });
                    continue;
                }
                running = Some(Running {
                    customer: selected.customer,
                    remaining: selected.remaining,
                    work_at,
                    finish,
                });
                break;
            }
        }

        if next_arrival == arrivals.len() && waiting.is_empty() && running.is_none() {
            break;
        }

        let mut next_tick = arrivals.get(next_arrival).map(|c| c.arrival);
        if let Some(active) = &running {
            for candidate in [Some(active.finish), Some(active.work_at)] {
                if let Some(t) = candidate.filter(|&t| t > now) {
                    next_tick = Some(next_tick.map_or(t, |n| n.min(t)));
                }
            }
        }

        match next_tick {
            Some(t) => now = t,
            None => break,
        }
    }

    Ok(events)
}
