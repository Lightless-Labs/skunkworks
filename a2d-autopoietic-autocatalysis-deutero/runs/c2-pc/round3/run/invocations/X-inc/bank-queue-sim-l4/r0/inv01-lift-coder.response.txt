#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let q = numerator / denominator;
    let r = numerator % denominator;
    let complement = denominator - r;
    if r > complement || (r == complement && q % 2 == 1) {
        Some(q + 1)
    } else {
        Some(q)
    }
}

#[derive(Clone)]
struct Person {
    input: Customer,
    deadline: Option<u64>,
    remaining: u64,
    entry: u64,
    waited: u64,
    ever_started: bool,
    pause_time: Option<u64>,
    accepted: bool,
    terminal: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    dispatch: u64,
    work_at: u64,
    finish: u64,
}

fn emit(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn better(a: usize, b: usize, people: &[Person]) -> bool {
    let x = &people[a];
    let y = &people[b];
    if x.input.priority != y.input.priority {
        return x.input.priority < y.input.priority;
    }
    match (x.pause_time, y.pause_time) {
        (Some(px), Some(py)) => {
            if px != py { px < py } else { x.input.id < y.input.id }
        }
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            if x.input.arrival != y.input.arrival {
                x.input.arrival < y.input.arrival
            } else {
                x.input.id > y.input.id
            }
        }
    }
}

fn best_waiter(waiting: &[usize], people: &[Person]) -> Option<usize> {
    waiting.iter().copied().reduce(|a, b| if better(a, b, people) { a } else { b })
}

fn expiration_order(waiting: &[usize], people: &[Person], now: u64) -> Vec<usize> {
    let mut expired: Vec<usize> = waiting.iter().copied()
        .filter(|&i| people[i].deadline.map_or(false, |d| d <= now))
        .collect();
    expired.sort_by_key(|&i| (people[i].deadline.unwrap(), people[i].input.id));
    expired
}

fn consume_wait(i: usize, now: u64, people: &mut [Person], wait_ticks: &mut u128) {
    let segment = now - people[i].entry;
    people[i].waited += segment;
    if let Some(d) = people[i].deadline {
        people[i].deadline = Some(d.saturating_sub(segment));
    }
    *wait_ticks += segment as u128;
}

fn renege(i: usize, now: u64, people: &mut [Person], waiting: &mut Vec<usize>,
          events: &mut Vec<Event>, stats: &mut Stats, wait_ticks: &mut u128) {
    consume_wait(i, now, people, wait_ticks);
    let waited = people[i].waited;
    people[i].terminal = true;
    stats.reneged += 1;
    emit(events, now, people[i].input.id, Kind::Renege { waited });
    waiting.retain(|&j| j != i);
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    for (i, c) in customers.iter().enumerate() {
        if c.id == 0 { return Err(Error::ZeroId { index: i }); }
        if customers[..i].iter().any(|x| x.id == c.id) {
            return Err(Error::DuplicateId { index: i, id: c.id });
        }
        if c.priority > 3 { return Err(Error::BadPriority { index: i, value: c.priority }); }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow { index: i, id: c.id });
            }
        }
    }

    let mut people: Vec<Person> = customers.iter().cloned().map(|c| Person {
        deadline: c.patience.map(|p| c.arrival + p),
        remaining: c.service,
        entry: c.arrival,
        waited: 0,
        ever_started: false,
        pause_time: None,
        accepted: false,
        terminal: false,
        input: c,
    }).collect();

    let mut arrivals: Vec<usize> = (0..people.len()).collect();
    arrivals.sort_by_key(|&i| (people[i].input.arrival, people[i].input.priority,
                               std::cmp::Reverse(people[i].input.id)));

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut arrival_pos = 0;
    let mut last_dispatched: Option<u32> = None;
    let mut first_dispatch = true;
    let mut now = 0u64;
    let mut stats = Stats {
        accepted: 0, balked: 0, reneged: 0, completed: 0, started: 0,
        preemptions: 0, elapsed: 0, busy_ticks: 0, wait_ticks: 0,
        turnaround_ticks: 0, mean_wait_milli: 0, mean_turnaround_milli: 0,
        utilization_bps: 0, completion_order: Vec::new(),
    };

    while stats.completed + stats.reneged + stats.balked < people.len() {
        if active.is_none() && waiting.is_empty() && arrival_pos < arrivals.len() {
            now = now.max(people[arrivals[arrival_pos]].input.arrival);
        }

        // Phase A: completion at this tick.
        if active.as_ref().map_or(false, |a| a.finish == now) {
            let a = active.take().unwrap();
            let id = people[a.index].input.id;
            people[a.index].terminal = true;
            stats.completed += 1;
            stats.turnaround_ticks += (now - people[a.index].input.arrival) as u128;
            stats.completion_order.push(id);
            emit(&mut events, now, id, Kind::Finish);
        }

        // Phase B: expirations present at the start of the tick.
        let expired = expiration_order(&waiting, &people, now);
        for i in expired {
            renege(i, now, &mut people, &mut waiting, &mut events, &mut stats, &mut stats.wait_ticks);
        }

        // Phase C: process the entire cohort.
        while arrival_pos < arrivals.len() && people[arrivals[arrival_pos]].input.arrival == now {
            let i = arrivals[arrival_pos];
            arrival_pos += 1;
            let id = people[i].input.id;
            emit(&mut events, now, id, Kind::Arrive);
            let gate = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= gate {
                stats.balked += 1;
                people[i].terminal = true;
                emit(&mut events, now, id, Kind::Balk);
            } else {
                people[i].accepted = true;
                people[i].entry = now;
                waiting.push(i);
                stats.accepted += 1;
            }
        }

        // Phase D: reconsider preemption after protection ends.
        if let Some(a) = active.clone() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&waiting, &people) {
                    if people[w].input.priority < people[a.index].input.priority {
                        let remaining = a.finish - now;
                        let by = people[w].input.id;
                        emit(&mut events, now, people[a.index].input.id,
                             Kind::Preempt { remaining, by });
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - a.dispatch) as u128;
                        active = None;
                        people[a.index].remaining = remaining;
                        people[a.index].entry = now;
                        people[a.index].pause_time = Some(now);
                        if let Some(budget) = people[a.index].deadline {
                            let spent = people[a.index].input.patience.unwrap()
                                .saturating_sub(budget.saturating_sub(people[a.index].input.arrival));
                            let _ = spent;
                        }
                        if let Some(p) = people[a.index].input.patience {
                            let used = people[a.index].waited;
                            let budget = p.saturating_sub(used);
                            let deadline = now.checked_add(budget).ok_or(Error::TimeOverflow {
                                customer: people[a.index].input.id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })?;
                            people[a.index].deadline = Some(deadline);
                        }
                        waiting.push(a.index);
                    }
                }
            }
        }

        // Phase E: dispatch while inline finishes permit further dispatches.
        loop {
            if active.is_some() { break; }
            let Some(i) = best_waiter(&waiting, &people) else { break; };
            waiting.retain(|&j| j != i);
            consume_wait(i, now, &mut people, &mut stats.wait_ticks);
            let id = people[i].input.id;
            let switch = if first_dispatch || last_dispatched == Some(id) { 0 } else { config.switch_cost };
            first_dispatch = false;
            last_dispatched = Some(id);
            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(people[i].remaining).ok_or(Error::TimeOverflow {
                customer: id, time: now, stage: OverflowStage::Finish,
            })?;
            if people[i].ever_started {
                emit(&mut events, now, id, Kind::Resume {
                    remaining: people[i].remaining, work_at, finish,
                });
            } else {
                people[i].ever_started = true;
                stats.started += 1;
                emit(&mut events, now, id, Kind::Start {
                    remaining: people[i].remaining, work_at, finish,
                });
            }
            active = Some(Active { index: i, dispatch: now, work_at, finish });
            if finish == now {
                let a = active.take().unwrap();
                people[a.index].terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - people[a.index].input.arrival) as u128;
                stats.completion_order.push(id);
                emit(&mut events, now, id, Kind::Finish);
                continue;
            }
            break;
        }

        // Phase F: expire customers queued during this tick.
        let expired = expiration_order(&waiting, &people, now);
        for i in expired {
            renege(i, now, &mut people, &mut waiting, &mut events, &mut stats, &mut stats.wait_ticks);
        }

        if stats.completed + stats.reneged + stats.balked >= people.len() {
            break;
        }

        let mut next = None;
        if arrival_pos < arrivals.len() {
            next = Some(people[arrivals[arrival_pos]].input.arrival);
        }
        if let Some(a) = &active {
            next = Some(next.map_or(a.finish, |v| v.min(a.finish)));
            if a.work_at > now {
                next = Some(next.map_or(a.work_at, |v| v.min(a.work_at)));
            }
        }
        for &i in &waiting {
            if let Some(d) = people[i].deadline {
                if d > now { next = Some(next.map_or(d, |v| v.min(d))); }
            }
        }
        if let Some(t) = next {
            now = t;
        }
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    let accepted = stats.accepted as u128;
    let completed = stats.completed as u128;
    stats.mean_wait_milli = if accepted == 0 { 0 } else {
        rounded_ratio(stats.wait_ticks * 1000, accepted).unwrap()
    };
    stats.mean_turnaround_milli = if completed == 0 { 0 } else {
        rounded_ratio(stats.turnaround_ticks * 1000, completed).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 { 0 } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}
