#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|v| v == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|v| v == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config.delimiter, config.quote)?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with_config(&self.input, self.config)
            .and_then(|records| records_to_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    validate_config(config.delimiter, config.quote)?;
    parse_with_config(input, config).and_then(|records| records_to_table(records, config))
}

fn validate_config(delimiter: u8, quote: u8) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if delimiter == quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn parse_with_config(input: &str, config: Config) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, config.delimiter, config.quote)
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = Vec::new();
    let mut record_started = false;
    let mut quoted = false;
    let mut closed_quote = false;
    let mut quote_start = 0usize;
    let mut record_index = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if !quoted && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        if quoted {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote);
                    i += 2;
                    continue;
                }
                quoted = false;
                closed_quote = true;
                i += 1;
                continue;
            }
            field.push(b);
            i += 1;
            continue;
        }

        if closed_quote {
            if b == delimiter {
                row.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                closed_quote = false;
                record_started = true;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                row.push(String::from_utf8(field).unwrap());
                field = Vec::new();
                records.push(row);
                row = Vec::new();
                closed_quote = false;
                record_started = false;
                record_index += 1;
                i += if b == b'\r' { 2 } else { 1 };
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if b == delimiter {
            row.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            record_started = true;
            i += 1;
            continue;
        }

        if b == b'\n' || b == b'\r' {
            row.push(String::from_utf8(field).unwrap());
            field = Vec::new();
            records.push(row);
            row = Vec::new();
            record_started = false;
            record_index += 1;
            i += if b == b'\r' { 2 } else { 1 };
            continue;
        }

        if b == quote {
            if field.is_empty() && !record_started {
                quoted = true;
                quote_start = i;
                record_started = true;
                i += 1;
                continue;
            }
            return Err(CsvError::UnexpectedQuote {
                record: record_index,
                byte: i,
            });
        }

        field.push(b);
        record_started = true;
        i += 1;
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if record_started || !field.is_empty() || !row.is_empty() || closed_quote {
        row.push(String::from_utf8(field).unwrap());
        records.push(row);
    }

    Ok(records)
}

fn records_to_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    if records.is_empty() {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    }

    let mut iter = records.into_iter();
    let header = iter.next().unwrap();

    if config.duplicates == DuplicatePolicy::Reject {
        let mut seen = std::collections::HashMap::<String, usize>::new();
        for (second, name) in header.iter().enumerate() {
            if let Some(&first) = seen.get(name) {
                return Err(CsvError::DuplicateHeader {
                    name: name.clone(),
                    first,
                    second,
                });
            }
            seen.insert(name.clone(), second);
        }
    }

    let mut rows = Vec::new();
    for (record, row) in iter.enumerate() {
        let input_record = record + 1;
        if row.len() != header.len() {
            return Err(CsvError::HeaderWidth {
                record: input_record,
                expected: header.len(),
                actual: row.len(),
            });
        }
        rows.push(row);
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

pub fn extract_columns(table: &Table, columns: &[ColumnSpec], policy: ErrorPolicy) -> TypedResult {
    let mut output = Vec::new();
    let mut errors = Vec::new();

    'rows: for (row_index, row) in table.rows.iter().enumerate() {
        let mut values = vec![None; columns.len()];
        for (spec_index, spec) in columns.iter().enumerate() {
            let column = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let failure = if column.is_none() {
                Some((None, CellErrorKind::UnknownColumn))
            } else {
                let index = column.unwrap();
                match row.get(index) {
                    None => Some((Some(index), CellErrorKind::MissingCell)),
                    Some(cell) => match convert_cell(cell, spec) {
                        Ok(value) => {
                            values[spec_index] = Some(value);
                            None
                        }
                        Err(kind) => Some((Some(index), kind)),
                    },
                }
            };

            if let Some((column, kind)) = failure {
                errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column,
                    kind,
                });
                match policy {
                    ErrorPolicy::Collect => {}
                    ErrorPolicy::FirstPerRow => break,
                    ErrorPolicy::Stop => {
                        output.push(values);
                        break 'rows;
                    }
                }
            }
        }
        output.push(values);
    }

    TypedResult {
        rows: output,
        errors,
    }
}

fn convert_cell(cell: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if cell.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(CellErrorKind::Empty);
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.to_string())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let mut i = 0;
    if bytes[0] == b'-' {
        i += 1;
    }
    if i >= bytes.len() {
        return false;
    }
    if bytes[i] == b'0' {
        return i + 1 == bytes.len() && i == 0;
    }
    if !(b'1'..=b'9').contains(&bytes[i]) {
        return false;
    }
    i += 1;
    bytes[i..].iter().all(u8::is_ascii_digit)
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>().map_err(|_| CellErrorKind::IntegerOverflow)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    let mut i = 0;
    if bytes.first() == Some(&b'-') {
        i += 1;
    }
    if i >= bytes.len() {
        return false;
    }
    let int_start = i;
    if bytes[i] == b'0' {
        i += 1;
        if i < bytes.len() && bytes[i].is_ascii_digit() {
            return false;
        }
    } else if (b'1'..=b'9').contains(&bytes[i]) {
        i += 1;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
    } else {
        return false;
    }
    if i == int_start || bytes.get(i) != Some(&b'.') {
        return false;
    }
    i += 1;
    if bytes.len() != i + 2 || !bytes[i].is_ascii_digit() || !bytes[i + 1].is_ascii_digit() {
        return false;
    }
    !(bytes.first() == Some(&b'-') && &bytes[int_start..i - 1] == b"0" && &bytes[i..] == b"00")
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let (whole, frac) = body.split_once('.').unwrap();
    let magnitude = whole
        .parse::<i128>()
        .unwrap()
        .checked_mul(100)
        .and_then(|v| v.checked_add(frac.parse::<i128>().unwrap()))
        .unwrap();
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

    pub line_ending: LineEnding,
    pub final_terminator: bool,

}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_config(config.delimiter, config.quote).map_err(|e| match e {
        CsvError::InvalidConfig { component } => WriteError::InvalidConfig { component },
        _ => unreachable!(),
    })?;

    let mut records: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header {
        records.push(header);
    }
    records.extend(table.rows.iter().map(Vec::as_slice));

    for (record_index, record) in records.iter().enumerate() {
        for (column_index, cell) in record.iter().enumerate() {
            if let Some(byte) = cell.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (row_index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: row_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, record) in records.iter().enumerate() {
        if record.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                let mut seen = std::collections::HashMap::<String, usize>::new();
                for (second, name) in header.iter().enumerate() {
                    if let Some(&first) = seen.get(name) {
                        return Err(WriteError::DuplicateHeader {
                            name: name.clone(),
                            first,
                            second,
                        });
                    }
                    seen.insert(name.clone(), second);
                }
            }
        }
    }

    let mut owned_records: Vec<Vec<String>> = match config.mode {
        WriteMode::Preserve => records.iter().map(|r| r.to_vec()).collect(),
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained = Vec::<(String, usize)>::new();
                for (index, name) in header.iter().enumerate() {
                    let keep = match table.duplicates {
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            !header[..index].iter().any(|prev| prev == name)
                        }
                        DuplicatePolicy::KeepLast => {
                            !header[index + 1..].iter().any(|next| next == name)
                        }
                    };
                    if keep {
                        retained.push((name.clone(), index));
                    }
                }
                retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));
                let new_header: Vec<String> = retained.iter().map(|(n, _)| n.clone()).collect();
                let mut projected_rows: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| retained.iter().map(|(_, i)| row[*i].clone()).collect())
                    .collect();
                projected_rows.sort_by(|a, b| compare_rows(a, b));
                let mut out = vec![new_header];
                out.extend(projected_rows);
                out
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_rows(a, b));
                rows
            }
        }
    };

    if owned_records.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    let canonical = config.mode == WriteMode::Canonical;
    let ending: &[u8] = if canonical || config.line_ending == LineEnding::Lf {
        b"\n"
    } else {
        b"\r\n"
    };

    for (record_index, record) in owned_records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(std::str::from_utf8(ending).unwrap());
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter as char);
            }
            let bytes = field.as_bytes();
            let sole_empty = record.len() == 1 && field.is_empty();
            let boundary_space = bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
                || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t');
            let initial_bom = record_index == 0 && column_index == 0 && field.starts_with('\u{feff}');
            let needs_quote = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || boundary_space
                || initial_bom
                || sole_empty;
            if needs_quote {
                output.push(config.quote as char);
                for &b in bytes {
                    if b == config.quote {
                        output.push(config.quote as char);
                    }
                    output.push(b as char);
                }
                output.push(config.quote as char);
            } else {
                output.push_str(field);
            }
        }
        if canonical || config.final_terminator {
            output.push_str(std::str::from_utf8(ending).unwrap());
        }
    }

    Ok(output)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        match left.as_bytes().cmp(right.as_bytes()) {
            std::cmp::Ordering::Equal => {}
            other => return other,
        }
    }
    a.len().cmp(&b.len())
}
