#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut quoted_close = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    quoted_close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        if quoted_close.is_none() {
            return Err(EmailError::UnclosedQuote { at: 0 });
        }
    }

    let mut separators = Vec::new();
    let protected_end = quoted_close.map(|i| i + 1).unwrap_or(0);
    for i in 0..bytes.len() {
        if i >= protected_end && bytes[i] == b'@' {
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let mut local = String::new();
    let quoted;

    if bytes[0] == b'"' {
        let close = quoted_close.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            if bytes[i] == b'\\' {
                let origin = i;
                i += 1;
                if i >= close {
                    return Err(EmailError::BadEscape { at: origin });
                }
                match bytes[i] {
                    b'"' | b'\\' | b'@' => {
                        local.push(bytes[i] as char);
                        origins.push(origin);
                    }
                    _ => return Err(EmailError::BadEscape { at: origin }),
                }
                i += 1;
            } else {
                let b = bytes[i];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                local.push(b as char);
                origins.push(i);
                i += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        quoted = true;
    } else {
        let local_bytes = &bytes[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for i in 0..local_bytes.len() {
            let b = local_bytes[i];
            if b == b'.' {
                if i == 0
                    || i + 1 == local_bytes.len()
                    || local_bytes[i - 1] == b'.'
                {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: i });
            }
        }

        local = String::from_utf8(local_bytes.to_vec()).unwrap();
        quoted = false;
    }

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let removed_dot_at = if raw_domain.last() == Some(&b'.') {
        Some(bytes.len() - 1)
    } else {
        None
    };
    let domain_end = removed_dot_at.unwrap_or(bytes.len());

    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut start = domain_start;

    for i in domain_start..domain_end {
        if bytes[i] == b'.' {
            if i == start {
                return Err(EmailError::EmptyLabel { at: i });
            }
            label_starts.push(start);
            label_ends.push(i);
            start = i + 1;
        }
    }

    if start == domain_end {
        return Err(EmailError::EmptyLabel {
            at: removed_dot_at.unwrap_or(domain_end),
        });
    }
    label_starts.push(start);
    label_ends.push(domain_end);

    for (&label_start, &label_end) in label_starts.iter().zip(label_ends.iter()) {
        let len = label_end - label_start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: label_start + 63,
            });
        }

        if bytes[label_start] == b'-' {
            return Err(EmailError::LabelHyphen { at: label_start });
        }
        if bytes[label_end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: label_end - 1 });
        }

        for i in label_start..label_end {
            let b = bytes[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar { at: i });
            }
        }
    }

    if label_starts.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(|&b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
