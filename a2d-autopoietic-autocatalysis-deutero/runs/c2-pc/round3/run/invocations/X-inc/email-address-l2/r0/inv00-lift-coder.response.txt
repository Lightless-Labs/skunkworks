#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut quoted_close = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    quoted_close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        if quoted_close.is_none() {
            return Err(EmailError::UnclosedQuote { at: 0 });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if Some(i) == quoted_close {
            i += 1;
            continue;
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let (local, quoted) = if bytes[0] == b'"' {
        let close = quoted_close.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut p = 1;
        while p < close {
            let b = bytes[p];
            if b == b'\\' {
                if p + 1 >= close || !matches!(bytes[p + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: p });
                }
                decoded.push(bytes[p + 1]);
                origins.push(p);
                p += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: p });
                }
                decoded.push(b);
                origins.push(p);
                p += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }

        (String::from_utf8(decoded).unwrap(), true)
    } else {
        let local_bytes = &bytes[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (j, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: j });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: j });
            }
        }

        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let root_dot = if raw_domain.last() == Some(&b'.') {
        Some(input.len() - 1)
    } else {
        None
    };
    let domain_end = if root_dot.is_some() {
        input.len() - 1
    } else {
        input.len()
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_bytes = &bytes[domain_start..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for pos in domain_start..=domain_end {
        if pos == domain_end || bytes[pos] == b'.' {
            if label_start == pos {
                let at = if pos == domain_end {
                    root_dot.unwrap_or(pos)
                } else {
                    pos
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let len = pos - label_start;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: label_start + 63,
                });
            }

            if bytes[label_start] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_start });
            }
            if bytes[pos - 1] == b'-' {
                return Err(EmailError::LabelHyphen { at: pos - 1 });
            }
            for p in label_start..pos {
                if !bytes[p].is_ascii_alphanumeric() && bytes[p] != b'-' {
                    return Err(EmailError::DomainChar { at: p });
                }
            }

            labels.push((label_start, pos));
            label_start = pos + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(|&b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
