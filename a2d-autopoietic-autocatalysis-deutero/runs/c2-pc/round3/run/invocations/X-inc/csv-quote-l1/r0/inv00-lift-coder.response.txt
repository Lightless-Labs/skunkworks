#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut record = 0usize;
    let mut quoted = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if !quoted && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr { record, byte: i });
        }

        if quoted {
            if b == b'"' {
                if bytes.get(i + 1) == Some(&b'"') {
                    field.push('"');
                    i += 2;
                } else {
                    quoted = false;
                    after_quote = true;
                    i += 1;
                }
            } else {
                let start = i;
                i += 1;
                while i < bytes.len() && (bytes[i] & 0xC0) == 0x80 {
                    i += 1;
                }
                field.push_str(&input[start..i]);
            }
            continue;
        }

        if after_quote {
            match b {
                b',' => {
                    fields.push(std::mem::take(&mut field));
                    after_quote = false;
                    i += 1;
                }
                b'\n' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    after_quote = false;
                    i += 1;
                }
                b'\r' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    after_quote = false;
                    i += 2;
                }
                _ => return Err(CsvError::TrailingByte { record, byte: i }),
            }
            continue;
        }

        match b {
            b',' => {
                fields.push(std::mem::take(&mut field));
                i += 1;
            }
            b'\n' => {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record += 1;
                i += 1;
            }
            b'\r' => {
                fields.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut fields));
                record += 1;
                i += 2;
            }
            b'"' => {
                if field.is_empty() {
                    quoted = true;
                    quote_open = i;
                    i += 1;
                } else {
                    return Err(CsvError::UnexpectedQuote { record, byte: i });
                }
            }
            _ => {
                let start = i;
                i += 1;
                while i < bytes.len() && (bytes[i] & 0xC0) == 0x80 {
                    i += 1;
                }
                field.push_str(&input[start..i]);
            }
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }

    if after_quote || !field.is_empty() || !fields.is_empty() || input.as_bytes().last() == Some(&b',') {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}
