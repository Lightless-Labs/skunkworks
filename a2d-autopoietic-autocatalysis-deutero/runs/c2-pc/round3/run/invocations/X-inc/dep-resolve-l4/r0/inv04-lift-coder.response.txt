fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    if b.is_empty() || b.len() > 24 || !b[0].is_ascii_lowercase() || b[b.len() - 1] == b'-' {
        return false;
    }
    b.iter()
        .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(items: &[Constraint]) -> Vec<Constraint> {
    let mut out = items.to_vec();
    out.sort_by_key(|c| constraint_key(*c));
    out.dedup();
    out
}

fn constraints_text(items: &[Constraint]) -> String {
    canonical_constraints(items)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn permits(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_sorted(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(mut s: String, evidence: &[Evidence]) -> String {
    for item in evidence_sorted(evidence) {
        s.push_str("\n  C:");
        s.push_str(&constraint_text(item.constraint));
        s.push_str(" via ");
        s.push_str(&item.path.join(" -> "));
    }
    s
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => append_evidence(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone)]
struct Closure {
    constraints: BTreeMap<String, Vec<Constraint>>,
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    active_features: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
}

use std::collections::{BTreeMap, BTreeSet};

fn platform_matches(platform: Platform, target: Target) -> bool {
    matches!(platform, Platform::Any)
        || matches!(
            (platform, target),
            (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
}

fn feature_declaration<'a>(release: &'a Release, name: &str) -> Option<&'a Feature> {
    release.features.iter().find(|f| f.name == name)
}

fn root_token(index: usize, req: &Requirement) -> String {
    let mut features = req.features.clone();
    features.sort();
    features.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        index,
        req.package,
        constraint_text(req.constraint),
        features.join(",")
    )
}

fn dep_token(source: &str, version: Version, index: usize, dep: &Dependency) -> String {
    let mut features = dep.features.clone();
    features.sort();
    features.dedup();
    let gate = dep.gate.as_deref().unwrap_or("*");
    let platform = match dep.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        index,
        dep.package,
        constraint_text(dep.constraint),
        features.join(","),
        gate,
        platform
    )
}

fn path_better(a: &[String], b: &[String]) -> bool {
    a.len() < b.len() || (a.len() == b.len() && a < b)
}

fn compute_closure(
    packages: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Closure {
    let mut closure = Closure {
        constraints: BTreeMap::new(),
        origins: BTreeMap::new(),
        requests: BTreeMap::new(),
        active_features: BTreeMap::new(),
        paths: BTreeMap::new(),
    };

    for (i, req) in roots.iter().enumerate() {
        closure
            .constraints
            .entry(req.package.clone())
            .or_default()
            .push(req.constraint);
        closure
            .requests
            .entry(req.package.clone())
            .or_default()
            .extend(req.features.iter().cloned());
        let path = vec![root_token(i, req)];
        match closure.paths.get(&req.package) {
            None => {
                closure.paths.insert(req.package.clone(), path.clone());
            }
            Some(existing) if path_better(&path, existing) => {
                closure.paths.insert(req.package.clone(), path.clone());
            }
            _ => {}
        }
        closure
            .origins
            .entry(req.package.clone())
            .or_default()
            .push(Origin {
                constraint: req.constraint,
                path,
            });
    }

    // Iterate to a joint fixed point. Edges are generated after feature requests
    // and canonical paths stabilize, so each dependency origin uses its final path.
    loop {
        let previous_constraints = closure.constraints.clone();
        let previous_requests = closure.requests.clone();
        let previous_features = closure.active_features.clone();
        let previous_paths = closure.paths.clone();
        let previous_origins = closure.origins.clone();

        let selected_names: Vec<String> = selected.keys().cloned().collect();
        for source in selected_names {
            let Some(package) = packages.get(&source) else {
                continue;
            };
            let version = selected[&source];
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let mut activated = closure.requests.get(&source).cloned().unwrap_or_default();
            let mut queue: Vec<String> = activated.iter().cloned().collect();
            while let Some(feature_name) = queue.pop() {
                if let Some(feature) = feature_declaration(release, &feature_name) {
                    for enabled in &feature.enables {
                        if activated.insert(enabled.clone()) {
                            queue.push(enabled.clone());
                        }
                    }
                }
            }
            closure
                .active_features
                .entry(source.clone())
                .or_default()
                .extend(activated.iter().cloned());

            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target) {
                    continue;
                }
                if dep
                    .gate
                    .as_ref()
                    .is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }
                closure
                    .constraints
                    .entry(dep.package.clone())
                    .or_default()
                    .push(dep.constraint);
                closure
                    .requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());

                if let Some(base) = closure.paths.get(&source).cloned() {
                    let mut path = base;
                    path.push(dep_token(&source, version, j, dep));
                    match closure.paths.get(&dep.package) {
                        None => {
                            closure.paths.insert(dep.package.clone(), path);
                        }
                        Some(existing) if path_better(&path, existing) => {
                            closure.paths.insert(dep.package.clone(), path);
                        }
                        _ => {}
                    }
                }
            }
        }

        // Rebuild dependency origins from scratch using the canonical source paths.
        let mut origins: BTreeMap<String, Vec<Origin>> = BTreeMap::new();
        for (i, req) in roots.iter().enumerate() {
            origins
                .entry(req.package.clone())
                .or_default()
                .push(Origin {
                    constraint: req.constraint,
                    path: vec![root_token(i, req)],
                });
        }
        for source in selected.keys() {
            let Some(package) = packages.get(source) else {
                continue;
            };
            let version = selected[source];
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };
            let Some(base) = closure.paths.get(source) else {
                continue;
            };
            let active = closure.active_features.get(source);
            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !active.is_some_and(|features| features.contains(g)))
                {
                    continue;
                }
                let mut path = base.clone();
                path.push(dep_token(source, version, j, dep));
                origins.entry(dep.package.clone()).or_default().push(Origin {
                    constraint: dep.constraint,
                    path,
                });
            }
        }
        closure.origins = origins;

        if closure.constraints == previous_constraints
            && closure.requests == previous_requests
            && closure.active_features == previous_features
            && closure.paths == previous_paths
            && closure.origins.len() == previous_origins.len()
        {
            break;
        }
    }

    for constraints in closure.constraints.values_mut() {
        *constraints = canonical_constraints(constraints);
    }
    closure
}

fn make_evidence(origins: Option<&Vec<Origin>>) -> Vec<Evidence> {
    let mut out = origins
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn dynamic_error(
    packages: &BTreeMap<String, &Package>,
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    target: Target,
) -> Result<Closure, ResolveError> {
    let closure = compute_closure(packages, roots, selected, target);

    for name in closure.constraints.keys() {
        if !packages.contains_key(name) {
            return Err(ResolveError::Missing {
                package: name.clone(),
                evidence: make_evidence(closure.origins.get(name)),
            });
        }
    }

    for (name, version) in selected {
        let package = packages[name];
        let release = package.releases.iter().find(|r| r.version == *version).unwrap();
        if let Some(features) = closure.active_features.get(name) {
            for feature in features {
                if feature_declaration(release, feature).is_none() {
                    return Err(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: closure.paths.get(name).cloned().unwrap_or_default(),
                    });
                }
            }
        }
    }

    for (name, version) in selected {
        if let Some(constraints) = closure.constraints.get(name) {
            if !constraints.iter().all(|c| permits(*c, *version)) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                    evidence: make_evidence(closure.origins.get(name)),
                });
            }
        }
    }

    for name in closure.constraints.keys() {
        if selected.contains_key(name) {
            continue;
        }
        let package = packages[name];
        let constraints = &closure.constraints[name];
        if !package
            .releases
            .iter()
            .any(|r| constraints.iter().all(|c| permits(*c, r.version)))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
                evidence: make_evidence(closure.origins.get(name)),
            });
        }
    }

    Ok(closure)
}

struct Search<'a> {
    packages: &'a BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_error: Option<ResolveError>,
    catalog_names: Vec<String>,
}

impl Search<'_> {
    fn record_leaf_error(&mut self, error: ResolveError) {
        if self.first_error.is_none() {
            self.first_error = Some(error);
        }
    }

    fn solution_better(&self, cost: usize, selected: &BTreeMap<String, Version>) -> bool {
        let Some((best_cost, best_vector)) = &self.best else {
            return true;
        };
        if cost != *best_cost {
            return cost < *best_cost;
        }
        for name in &self.catalog_names {
            let a = selected.get(name);
            let b = best_vector
                .iter()
                .find(|(n, _)| n == name)
                .map(|(_, v)| v);
            match (a, b) {
                (None, Some(_)) => return false,
                (Some(_), None) => return true,
                (Some(x), Some(y)) if x != y => return x > y,
                _ => {}
            }
        }
        false
    }

    fn search(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let closure = match dynamic_error(self.packages, self.roots, selected, self.target) {
            Ok(c) => c,
            Err(e) => {
                self.record_leaf_error(e);
                return Ok(());
            }
        };

        let next = closure
            .constraints
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();

        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(name, version)| selected.get(*name) != Some(*version))
                .count();
            if self.solution_better(cost, selected) {
                self.best = Some((cost, selected.iter().map(|(n, v)| (n.clone(), *v)).collect()));
            }
            return Ok(());
        };

        let package = self.packages[&name];
        let constraints = &closure.constraints[&name];
        let mut candidates: Vec<Version> = package
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| constraints.iter().all(|c| permits(*c, *v)))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        candidates.dedup();
        if let Some(lock) = self.locks.get(&name) {
            if let Some(pos) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for candidate in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), candidate);
            self.search(selected)?;
            selected.remove(&name);
        }
        Ok(())
    }
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{}].release[{}].dep[{}].gate", i, j, k),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].dep[{}].feature[{}]",
                                i, j, k, f
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut names = BTreeSet::new();
    for package in catalog {
        if !names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut features = BTreeSet::new();
            for feature in &release.features {
                if !features.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_names = BTreeSet::new();
    for lock in locks {
        if !lock_names.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let mut packages = BTreeMap::new();
    for package in catalog {
        packages.insert(package.name.clone(), package);
    }
    let lock_map: BTreeMap<String, Version> = locks
        .iter()
        .map(|l| (l.package.clone(), l.version))
        .collect();
    let catalog_names = packages.keys().cloned().collect::<Vec<_>>();
    let mut search = Search {
        packages: &packages,
        roots,
        locks: &lock_map,
        target,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        catalog_names,
    };

    let mut selected = BTreeMap::new();
    search.search(&mut selected)?;
    let Some((_, selected_pairs)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
            evidence: Vec::new(),
        }));
    };
    let selected_map: BTreeMap<String, Version> = selected_pairs.iter().cloned().collect();
    let final_closure = compute_closure(&packages, roots, &selected_map, target);
    let selected_output = selected_map
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();

    let mut changes = Vec::new();
    let all_names: BTreeSet<String> = locks
        .iter()
        .map(|l| l.package.clone())
        .chain(selected_map.keys().cloned())
        .collect();
    for name in all_names {
        let before = lock_map.get(&name).copied();
        let after = selected_map.get(&name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = packages
                    .get(&name)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if final_closure
                    .constraints
                    .get(&name)
                    .is_some_and(|cs| !cs.iter().all(|c| permits(*c, old)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_output,
        changes,
    })
}
