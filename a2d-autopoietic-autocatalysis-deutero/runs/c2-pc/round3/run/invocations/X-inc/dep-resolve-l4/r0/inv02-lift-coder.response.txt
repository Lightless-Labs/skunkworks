fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b.last() != Some(&b'-')
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut out = cs.to_vec();
    out.sort();
    out.dedup();
    out
}

fn matches(c: Constraint, v: Version) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn satisfies(cs: &[Constraint], v: Version) -> bool {
    cs.iter().all(|c| matches(*c, v))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

fn evidence_text(evidence: &[Evidence]) -> String {
    let mut items = evidence.to_vec();
    items.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    items
        .iter()
        .map(|e| {
            format!(
                "\n  C:{} via {}",
                constraint_text(e.constraint),
                e.path.join(" -> ")
            )
        })
        .collect::<String>()
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => format!(
            "missing|package={}{}",
            encode(package),
            evidence_text(evidence)
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => format!(
            "no-match|package={}|constraints={}{}",
            encode(package),
            canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(","),
            evidence_text(evidence)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => format!(
            "conflict|package={}|selected={}|constraints={}{}",
            encode(package),
            version_text(*selected),
            canonical_constraints(requirements)
                .iter()
                .map(|c| constraint_text(*c))
                .collect::<Vec<_>>()
                .join(","),
            evidence_text(evidence)
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: BTreeMap<String, Vec<Origin>>,
    requests: BTreeMap<String, BTreeSet<String>>,
    paths: BTreeMap<String, Vec<String>>,
    activated: BTreeMap<String, BTreeSet<String>>,
    active_deps: BTreeMap<String, BTreeSet<usize>>,
}

fn platform_matches(p: Platform, t: Target) -> bool {
    matches!(
        (p, t),
        (Platform::Any, _)
            | (Platform::Linux, Target::Linux)
            | (Platform::Mac, Target::Mac)
            | (Platform::Windows, Target::Windows)
    )
}

fn dep_token(source: &str, version: Version, j: usize, d: &Dependency) -> String {
    let fs = {
        let mut f = d.features.clone();
        f.sort();
        f.dedup();
        f.join(",")
    };
    let gate = d.gate.as_deref().unwrap_or("*");
    let platform = match d.platform {
        Platform::Any => "any",
        Platform::Linux => "linux",
        Platform::Mac => "mac",
        Platform::Windows => "windows",
    };
    format!(
        "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
        source,
        version_text(version),
        j,
        d.package,
        constraint_text(d.constraint),
        fs,
        gate,
        platform
    )
}

fn root_token(i: usize, r: &Requirement) -> String {
    let mut f = r.features.clone();
    f.sort();
    f.dedup();
    format!(
        "root[{}]:{}@{}+{{{}}}",
        i,
        r.package,
        constraint_text(r.constraint),
        f.join(",")
    )
}

fn compute_active(
    roots: &[Requirement],
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
    target: Target,
) -> Active {
    let mut a = Active::default();

    for (i, r) in roots.iter().enumerate() {
        let path = vec![root_token(i, r)];
        a.origins
            .entry(r.package.clone())
            .or_default()
            .push(Origin {
                constraint: r.constraint,
                path: path.clone(),
            });
        a.requests
            .entry(r.package.clone())
            .or_default()
            .extend(r.features.iter().cloned());
        let current = a.paths.entry(r.package.clone()).or_insert(path.clone());
        if path_cmp(&path, current).is_lt() {
            *current = path;
        }
    }

    loop {
        let before = (
            a.origins.clone(),
            a.requests.clone(),
            a.paths.clone(),
            a.activated.clone(),
            a.active_deps.clone(),
        );

        let names: Vec<String> = selected.keys().cloned().collect();
        for name in names {
            let Some(package) = catalog.get(&name) else {
                continue;
            };
            let version = selected[&name];
            let Some(release) = package.releases.iter().find(|r| r.version == version) else {
                continue;
            };

            let mut activated = a.requests.get(&name).cloned().unwrap_or_default();
            let mut queue: Vec<String> = activated.iter().cloned().collect();
            while let Some(feature_name) = queue.pop() {
                if let Some(feature) = release.features.iter().find(|f| f.name == feature_name) {
                    for enabled in &feature.enables {
                        if activated.insert(enabled.clone()) {
                            queue.push(enabled.clone());
                        }
                    }
                }
            }
            a.activated.insert(name.clone(), activated.clone());

            for (j, dep) in release.deps.iter().enumerate() {
                if !platform_matches(dep.platform, target)
                    || dep
                        .gate
                        .as_ref()
                        .is_some_and(|g| !activated.contains(g))
                {
                    continue;
                }
                a.active_deps.entry(name.clone()).or_default().insert(j);
                let source_path = a.paths.get(&name).cloned().unwrap_or_default();
                let mut path = source_path;
                path.push(dep_token(&name, version, j, dep));
                a.origins
                    .entry(dep.package.clone())
                    .or_default()
                    .push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                a.requests
                    .entry(dep.package.clone())
                    .or_default()
                    .extend(dep.features.iter().cloned());
                let current = a.paths.entry(dep.package.clone()).or_insert(path.clone());
                if path_cmp(&path, current).is_lt() {
                    *current = path;
                }
            }
        }

        if before.0 == a.origins
            && before.1 == a.requests
            && before.2 == a.paths
            && before.3 == a.activated
            && before.4 == a.active_deps
        {
            break;
        }

        // Rebuild origins and dependency-derived state on the next pass without
        // retaining duplicate edges generated by prior passes.
        let roots_only = root_origins(roots);
        a.origins = roots_only.0;
        a.requests = roots_only.1;
        a.paths = roots_only.2;
        a.active_deps.clear();
    }
    a
}

fn root_origins(
    roots: &[Requirement],
) -> (
    BTreeMap<String, Vec<Origin>>,
    BTreeMap<String, BTreeSet<String>>,
    BTreeMap<String, Vec<String>>,
) {
    let mut origins = BTreeMap::new();
    let mut requests: BTreeMap<String, BTreeSet<String>> = BTreeMap::new();
    let mut paths = BTreeMap::new();
    for (i, r) in roots.iter().enumerate() {
        let path = vec![root_token(i, r)];
        origins
            .entry(r.package.clone())
            .or_insert_with(Vec::new)
            .push(Origin {
                constraint: r.constraint,
                path: path.clone(),
            });
        requests
            .entry(r.package.clone())
            .or_default()
            .extend(r.features.iter().cloned());
        let current = paths.entry(r.package.clone()).or_insert(path.clone());
        if path_cmp(&path, current).is_lt() {
            *current = path;
        }
    }
    (origins, requests, paths)
}

fn path_cmp(a: &[String], b: &[String]) -> std::cmp::Ordering {
    a.len().cmp(&b.len()).then_with(|| a.cmp(b))
}

fn active_error(
    active: &Active,
    selected: &BTreeMap<String, Version>,
    catalog: &BTreeMap<String, &Package>,
) -> Option<ResolveError> {
    for name in active.origins.keys() {
        if !catalog.contains_key(name) {
            return Some(ResolveError::Missing {
                package: name.clone(),
                evidence: make_evidence(active, name),
            });
        }
    }

    for (name, version) in selected {
        let constraints = active
            .origins
            .get(name)
            .map(|os| os.iter().map(|o| o.constraint).collect::<Vec<_>>())
            .unwrap_or_default();
        if !satisfies(&constraints, *version) {
            return Some(ResolveError::Conflict {
                package: name.clone(),
                selected: *version,
                requirements: canonical_constraints(&constraints),
                evidence: make_evidence(active, name),
            });
        }
    }

    for name in active.origins.keys() {
        if selected.contains_key(name) {
            continue;
        }
        let constraints = active.origins[name]
            .iter()
            .map(|o| o.constraint)
            .collect::<Vec<_>>();
        let package = catalog[name];
        if !package
            .releases
            .iter()
            .any(|r| satisfies(&constraints, r.version))
        {
            return Some(ResolveError::NoMatch {
                package: name.clone(),
                requirements: canonical_constraints(&constraints),
                evidence: make_evidence(active, name),
            });
        }
    }

    for (name, version) in selected {
        let Some(package) = catalog.get(name) else {
            continue;
        };
        let Some(release) = package.releases.iter().find(|r| r.version == *version) else {
            continue;
        };
        if let Some(features) = active.activated.get(name) {
            for feature in features {
                if !release.features.iter().any(|f| f.name == *feature) {
                    return Some(ResolveError::MissingFeature {
                        package: name.clone(),
                        selected: *version,
                        feature: feature.clone(),
                        path: active.paths.get(name).cloned().unwrap_or_default(),
                    });
                }
            }
        }
    }
    None
}

fn make_evidence(active: &Active, package: &str) -> Vec<Evidence> {
    let mut out = active
        .origins
        .get(package)
        .into_iter()
        .flatten()
        .map(|o| Evidence {
            constraint: o.constraint,
            path: o.path.clone(),
        })
        .collect::<Vec<_>>();
    out.sort_by(|a, b| {
        a.constraint
            .cmp(&b.constraint)
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

struct Search<'a> {
    roots: &'a [Requirement],
    catalog: &'a BTreeMap<String, &'a Package>,
    locks: &'a BTreeMap<String, Version>,
    target: Target,
    budget: usize,
    attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, Vec<Selection>, BTreeMap<String, Version>, Active)>,
    catalog_names: Vec<String>,
}

impl<'a> Search<'a> {
    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = compute_active(self.roots, selected, self.catalog, self.target);
        if let Some(error) = active_error(&active, selected, self.catalog) {
            if self.first_error.is_none() {
                self.first_error = Some(error);
            }
            return Ok(());
        }

        let next = active
            .origins
            .keys()
            .find(|name| !selected.contains_key(*name))
            .cloned();

        let Some(package_name) = next else {
            self.record_solution(selected, active);
            return Ok(());
        };

        let constraints = active.origins[&package_name]
            .iter()
            .map(|o| o.constraint)
            .collect::<Vec<_>>();
        let package = self.catalog[&package_name];
        let mut candidates = package
            .releases
            .iter()
            .filter(|r| satisfies(&constraints, r.version))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(locked) = self.locks.get(&package_name) {
            if let Some(pos) = candidates.iter().position(|v| v == locked) {
                let v = candidates.remove(pos);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            selected.insert(package_name.clone(), version);
            self.visit(selected)?;
            selected.remove(&package_name);
        }
        Ok(())
    }

    fn record_solution(&mut self, selected: &BTreeMap<String, Version>, active: Active) {
        let cost = self
            .locks
            .iter()
            .filter(|(name, version)| selected.get(*name) != Some(*version))
            .count();
        let selection = selected
            .iter()
            .map(|(package, version)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect::<Vec<_>>();

        let better = match &self.best {
            None => true,
            Some((best_cost, _, best_map, _)) => {
                if cost != *best_cost {
                    cost < *best_cost
                } else {
                    self.catalog_names.iter().find_map(|name| {
                        let a = selected.get(name);
                        let b = best_map.get(name);
                        if a == b {
                            None
                        } else {
                            Some(match (a, b) {
                                (None, Some(_)) => false,
                                (Some(_), None) => true,
                                (Some(a), Some(b)) => a > b,
                                (None, None) => false,
                            })
                        }
                    }) == Some(true)
                }
            }
        };

        if better {
            self.best = Some((cost, selection, selected.clone(), active));
        }
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{}].feature[{}]", i, f),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                let base = format!("catalog[{}].release[{}].dep[{}]", i, j, k);
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: base.clone(),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.gate", base),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!("{}.feature[{}]", base, f),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].feature[{}]", i, j, f),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{}].release[{}].feature[{}].enable[{}]",
                                i, j, f, e
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen = BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut seen = BTreeSet::new();
            for feature in &release.features {
                if !seen.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if lock_map.insert(lock.package.clone(), lock.version).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let catalog_map = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect::<BTreeMap<_, _>>();
    if roots.is_empty() {
        return Ok(Resolution {
            selected: Vec::new(),
            changes: locks
                .iter()
                .map(|l| Change {
                    package: l.package.clone(),
                    before: Some(l.version),
                    after: None,
                    reason: ChangeReason::Removed,
                })
                .collect(),
        });
    }

    let mut names = catalog_map.keys().cloned().collect::<Vec<_>>();
    names.sort();
    let mut search = Search {
        roots,
        catalog: &catalog_map,
        locks: &lock_map,
        target,
        budget: max_attempts,
        attempts: 0,
        first_error: None,
        best: None,
        catalog_names: names,
    };
    search.visit(&mut BTreeMap::new())?;

    let Some((_, selected, selected_map, active)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: roots[0].package.clone(),
            requirements: vec![roots[0].constraint],
            evidence: vec![Evidence {
                constraint: roots[0].constraint,
                path: vec![root_token(0, &roots[0])],
            }],
        }));
    };

    let mut all_names = BTreeSet::new();
    all_names.extend(lock_map.keys().cloned());
    all_names.extend(selected_map.keys().cloned());
    let mut changes = Vec::new();
    for name in all_names {
        let before = lock_map.get(&name).copied();
        let after = selected_map.get(&name).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog_map
                    .get(&name)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else {
                    let constraints = active
                        .origins
                        .get(&name)
                        .into_iter()
                        .flatten()
                        .map(|o| o.constraint)
                        .collect::<Vec<_>>();
                    if !satisfies(&constraints, old) {
                        ChangeReason::Rejected
                    } else {
                        ChangeReason::Rebalanced
                    }
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package: name,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}
