pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateFeature {
        package: String,
        version: Version,
        feature: String,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
        evidence: Vec<Evidence>,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
        evidence: Vec<Evidence>,
    },

        package: String,

fn valid_name(s: &str) -> bool {
    let b = s.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_key(c: Constraint) -> (u8, Version) {
    match c {
        Constraint::Exact(v) => (0, v),
        Constraint::AtLeast(v) => (1, v),
        Constraint::Before(v) => (2, v),
        Constraint::Caret(v) => (3, v),
    }
}

fn constraint_text(c: Constraint) -> String {
    let (prefix, v) = match c {
        Constraint::Exact(v) => ("=", v),
        Constraint::AtLeast(v) => (">=", v),
        Constraint::Before(v) => ("<", v),
        Constraint::Caret(v) => ("^", v),
    };
    format!("{prefix}{}", version_text(v))
}

fn canonical_constraints(cs: &[Constraint]) -> Vec<Constraint> {
    let mut result = cs.to_vec();
    result.sort_by_key(|c| constraint_key(*c));
    result.dedup();
    result
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical_constraints(cs)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn satisfies(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| matches(v, *c))
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{b:02X}"));
        }
    }
    out
}

fn sorted_evidence(evidence: &[Evidence]) -> Vec<Evidence> {
    let mut out = evidence.to_vec();
    out.sort_by(|a, b| {
        constraint_key(a.constraint)
            .cmp(&constraint_key(b.constraint))
            .then_with(|| a.path.cmp(&b.path))
    });
    out
}

fn append_evidence(mut first: String, evidence: &[Evidence]) -> String {
    for e in sorted_evidence(evidence) {
        first.push_str("\n  C:");
        first.push_str(&constraint_text(e.constraint));
        first.push_str(" via ");
        first.push_str(&e.path.join(" -> "));
    }
    first
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateFeature {
            package,
            version,
            feature,
        } => format!(
            "duplicate-feature|package={}|version={}|feature={}",
            encode(package),
            version_text(*version),
            encode(feature)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package, evidence } => append_evidence(
            format!("missing|package={}", encode(package)),
            evidence,
        ),
        ResolveError::NoMatch {
            package,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "no-match|package={}|constraints={}",
                encode(package),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
            evidence,
        } => append_evidence(
            format!(
                "conflict|package={}|selected={}|constraints={}",
                encode(package),
                version_text(*selected),
                constraints_text(requirements)
            ),
            evidence,
        ),
        ResolveError::MissingFeature {
            package,
            selected,
            feature,
            path,
        } => format!(
            "missing-feature|package={}|selected={}|feature={}\n  F:{} via {}",
            encode(package),
            version_text(*selected),
            encode(feature),
            encode(feature),
            path.join(" -> ")
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

#[derive(Clone)]
struct Origin {
    constraint: Constraint,
    path: Vec<String>,
}

#[derive(Clone, Default)]
struct Active {
    origins: Vec<Origin>,
    features: BTreeSet<String>,
}

use std::collections::{BTreeMap, BTreeSet};

struct Search<'a> {
    packages: BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: BTreeMap<String, Version>,
    target: Target,
    attempts: usize,
    max_attempts: usize,
    first_error: Option<ResolveError>,
    best: Option<(usize, Vec<Selection>, BTreeMap<String, Version>)>,
}

impl<'a> Search<'a> {
    fn target_matches(&self, p: Platform) -> bool {
        matches!(
            (p, self.target),
            (Platform::Any, _)
                | (Platform::Linux, Target::Linux)
                | (Platform::Mac, Target::Mac)
                | (Platform::Windows, Target::Windows)
        )
    }

    fn release(&self, name: &str, version: Version) -> Option<&Release> {
        self.packages
            .get(name)?
            .releases
            .iter()
            .find(|r| r.version == version)
    }

    fn token_root(&self, index: usize, req: &Requirement) -> String {
        let mut fs = req.features.clone();
        fs.sort();
        fs.dedup();
        format!(
            "root[{index}]:{}@{}+{{{}}}",
            req.package,
            constraint_text(req.constraint),
            fs.join(",")
        )
    }

    fn token_dep(
        source: &str,
        version: Version,
        index: usize,
        dep: &Dependency,
    ) -> String {
        let mut fs = dep.features.clone();
        fs.sort();
        fs.dedup();
        format!(
            "{}@{}/deps[{}]:{}@{}+{{{}}};gate={};platform={}",
            source,
            version_text(version),
            index,
            dep.package,
            constraint_text(dep.constraint),
            fs.join(","),
            dep.gate.as_deref().unwrap_or("*"),
            match dep.platform {
                Platform::Any => "any",
                Platform::Linux => "linux",
                Platform::Mac => "mac",
                Platform::Windows => "windows",
            }
        )
    }

    fn closure(
        &self,
        selected: &BTreeMap<String, Version>,
    ) -> (BTreeMap<String, Active>, BTreeMap<String, Vec<String>>) {
        let mut active: BTreeMap<String, Active> = BTreeMap::new();
        let mut paths: BTreeMap<String, Vec<String>> = BTreeMap::new();

        for (i, req) in self.roots.iter().enumerate() {
            let token = self.token_root(i, req);
            let path = vec![token];
            let entry = active.entry(req.package.clone()).or_default();
            entry.origins.push(Origin {
                constraint: req.constraint,
                path: path.clone(),
            });
            entry.features.extend(req.features.iter().cloned());
            if paths
                .get(&req.package)
                .map_or(true, |p| (path.len(), &path) < (p.len(), p))
            {
                paths.insert(req.package.clone(), path);
            }
        }

        loop {
            let snapshot = active.clone();
            let mut changed = false;

            for (name, version) in selected {
                let Some(release) = self.release(name, *version) else {
                    continue;
                };
                let requested: Vec<String> = snapshot
                    .get(name)
                    .map(|a| a.features.iter().cloned().collect())
                    .unwrap_or_default();
                let mut activated = BTreeSet::new();
                let mut queue = requested;
                while let Some(feature_name) = queue.pop() {
                    if !activated.insert(feature_name.clone()) {
                        continue;
                    }
                    if let Some(decl) = release.features.iter().find(|f| f.name == feature_name) {
                        for enabled in &decl.enables {
                            if !activated.contains(enabled) {
                                queue.push(enabled.clone());
                            }
                        }
                    }
                }

                let entry = active.entry(name.clone()).or_default();
                for f in &activated {
                    if entry.features.insert(f.clone()) {
                        changed = true;
                    }
                }

                for (i, dep) in release.deps.iter().enumerate() {
                    if !self.target_matches(dep.platform)
                        || dep
                            .gate
                            .as_ref()
                            .is_some_and(|g| !activated.contains(g))
                    {
                        continue;
                    }
                    let token = Self::token_dep(name, *version, i, dep);
                    let mut path = paths.get(name).cloned().unwrap_or_default();
                    path.push(token);
                    let dep_entry = active.entry(dep.package.clone()).or_default();
                    let before = dep_entry.origins.len();
                    dep_entry.origins.push(Origin {
                        constraint: dep.constraint,
                        path: path.clone(),
                    });
                    dep_entry.features.extend(dep.features.iter().cloned());
                    if dep_entry.origins.len() != before {
                        changed = true;
                    }
                    if paths
                        .get(&dep.package)
                        .map_or(true, |p| (path.len(), &path) < (p.len(), p))
                    {
                        paths.insert(dep.package.clone(), path);
                    }
                }
            }

            if !changed {
                break;
            }
        }

        (active, paths)
    }

    fn requirements(active: &Active) -> Vec<Constraint> {
        canonical_constraints(
            &active
                .origins
                .iter()
                .map(|o| o.constraint)
                .collect::<Vec<_>>(),
        )
    }

    fn evidence(active: &Active) -> Vec<Evidence> {
        let mut out = active
            .origins
            .iter()
            .map(|o| Evidence {
                constraint: o.constraint,
                path: o.path.clone(),
            })
            .collect::<Vec<_>>();
        out.sort_by(|a, b| {
            constraint_key(a.constraint)
                .cmp(&constraint_key(b.constraint))
                .then_with(|| a.path.cmp(&b.path))
        });
        out
    }

    fn inspect(
        &self,
        selected: &BTreeMap<String, Version>,
    ) -> Result<(BTreeMap<String, Active>, BTreeMap<String, Vec<String>>), ResolveError> {
        let (active, paths) = self.closure(selected);
        for name in active.keys() {
            if !self.packages.contains_key(name) {
                return Err(ResolveError::Missing {
                    package: name.clone(),
                    evidence: Self::evidence(&active[name]),
                });
            }
        }

        for (name, version) in selected {
            if let Some(a) = active.get(name) {
                if !satisfies(*version, &Self::requirements(a)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: Self::requirements(a),
                        evidence: Self::evidence(a),
                    });
                }
            }
        }

        for (name, version) in selected {
            let Some(release) = self.release(name, *version) else {
                continue;
            };
            if let Some(a) = active.get(name) {
                for feature in &a.features {
                    if !release.features.iter().any(|f| f.name == *feature) {
                        return Err(ResolveError::MissingFeature {
                            package: name.clone(),
                            selected: *version,
                            feature: feature.clone(),
                            path: paths.get(name).cloned().unwrap_or_default(),
                        });
                    }
                }
            }
        }

        for (name, a) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let candidates = self.packages[name]
                .releases
                .iter()
                .any(|r| satisfies(r.version, &Self::requirements(a)));
            if !candidates {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: Self::requirements(a),
                    evidence: Self::evidence(a),
                });
            }
        }
        Ok((active, paths))
    }

    fn vector_better(
        &self,
        candidate: &BTreeMap<String, Version>,
        incumbent: &BTreeMap<String, Version>,
    ) -> bool {
        let names: Vec<_> = self.packages.keys().collect();
        for name in names {
            match (candidate.get(name), incumbent.get(name)) {
                (Some(a), Some(b)) if a != b => return a > b,
                (Some(_), None) => return true,
                (None, Some(_)) => return false,
                _ => {}
            }
        }
        false
    }

    fn visit(&mut self, selected: &mut BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let (active, _) = match self.inspect(selected) {
            Ok(x) => x,
            Err(e) => {
                if self.first_error.is_none() {
                    self.first_error = Some(e);
                }
                return Ok(());
            }
        };

        let next = active.keys().find(|n| !selected.contains_key(*n)).cloned();
        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(p, v)| selected.get(*p) != Some(*v))
                .count();
            let replace = self.best.as_ref().map_or(true, |(best_cost, _, best)| {
                cost < *best_cost
                    || (cost == *best_cost && self.vector_better(selected, best))
            });
            if replace {
                let selections = selected
                    .iter()
                    .map(|(package, version)| Selection {
                        package: package.clone(),
                        version: *version,
                    })
                    .collect();
                self.best = Some((cost, selections, selected.clone()));
            }
            return Ok(());
        };

        let a = &active[&name];
        let mut candidates: Vec<Version> = self.packages[&name]
            .releases
            .iter()
            .filter(|r| satisfies(r.version, &Self::requirements(a)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));
        if let Some(lock) = self.locks.get(&name) {
            if let Some(i) = candidates.iter().position(|v| v == lock) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            self.visit(selected)?;
            selected.remove(&name);
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    target: Target,
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
        for (f, feature) in root.features.iter().enumerate() {
            if !valid_name(feature) {
                return Err(ResolveError::InvalidName {
                    site: format!("root[{i}].feature[{f}]"),
                    name: feature.clone(),
                });
            }
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
                if let Some(gate) = &dep.gate {
                    if !valid_name(gate) {
                        return Err(ResolveError::InvalidName {
                            site: format!("catalog[{i}].release[{j}].dep[{k}].gate"),
                            name: gate.clone(),
                        });
                    }
                }
                for (f, feature) in dep.features.iter().enumerate() {
                    if !valid_name(feature) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{i}].release[{j}].dep[{k}].feature[{f}]"
                            ),
                            name: feature.clone(),
                        });
                    }
                }
            }
            for (f, feature) in release.features.iter().enumerate() {
                if !valid_name(&feature.name) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].feature[{f}]"),
                        name: feature.name.clone(),
                    });
                }
                for (e, enabled) in feature.enables.iter().enumerate() {
                    if !valid_name(enabled) {
                        return Err(ResolveError::InvalidName {
                            site: format!(
                                "catalog[{i}].release[{j}].feature[{f}].enable[{e}]"
                            ),
                            name: enabled.clone(),
                        });
                    }
                }
            }
        }
    }

    let mut seen_packages = BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for package in catalog {
        for release in &package.releases {
            let mut names = BTreeSet::new();
            for feature in &release.features {
                if !names.insert(feature.name.clone()) {
                    return Err(ResolveError::DuplicateFeature {
                        package: package.name.clone(),
                        version: release.version,
                        feature: feature.name.clone(),
                    });
                }
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{i}]"),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = BTreeSet::new();
    let mut lock_map = BTreeMap::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.clone()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
        lock_map.insert(lock.package.clone(), lock.version);
    }

    let packages: BTreeMap<String, &Package> = catalog
        .iter()
        .map(|p| (p.name.clone(), p))
        .collect();
    let mut search = Search {
        packages,
        roots,
        locks: lock_map.clone(),
        target,
        attempts: 0,
        max_attempts,
        first_error: None,
        best: None,
    };
    let mut selected = BTreeMap::new();
    search.visit(&mut selected)?;
    let Some((_, selections, chosen)) = search.best else {
        return Err(search.first_error.unwrap_or_else(|| ResolveError::NoMatch {
            package: roots
                .first()
                .map(|r| r.package.clone())
                .unwrap_or_default(),
            requirements: Vec::new(),
            evidence: Vec::new(),
        }));
    };

    let (final_active, _) = search.closure(&chosen);
    let mut all_names: BTreeSet<String> = lock_map.keys().cloned().collect();
    all_names.extend(chosen.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = chosen.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = search
                    .packages
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if !final_active
                    .get(&package)
                    .map(|a| satisfies(old, &Search::requirements(a)))
                    .unwrap_or(true)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution {
        selected: selections,
        changes,
    })
}
