#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let mut kept = Vec::with_capacity(input.len());
    let mut map = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_mode = false;
    let mut quote_closed = false;
    let mut depth = 0usize;
    let mut outer_comment = 0usize;

    while i < input.len() {
        if quote_mode && !quote_closed {
            let b = input[i];
            kept.push(b);
            map.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    kept.push(input[i]);
                    map.push(i);
                    i += 1;
                }
                continue;
            }
            if b == b'"' {
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        let b = input[i];
        if depth == 0 && !quote_mode && kept.is_empty() && b == b'"' {
            quote_mode = true;
            quote_closed = false;
            kept.push(b);
            map.push(i);
            i += 1;
            continue;
        }

        let is_ws_construct = is_hws(b) || b == b'\r' || b == b'\n';
        let is_comment = depth > 0 || b == b'(';
        if is_ws_construct || is_comment {
            let run_start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if depth > 0 {
                    if c == b'\\' {
                        let slash = i;
                        if i + 1 >= input.len()
                            || !(0x20..=0x7e).contains(&input[i + 1])
                        {
                            return Err(EmailError::CommentEscape { at: slash });
                        }
                        i += 2;
                        continue;
                    }
                    if let Some(next) = is_fold(input, i) {
                        i = next;
                        continue;
                    }
                    if c == b'\r' || c == b'\n' {
                        return Err(EmailError::BadFold { at: i });
                    }
                    if c == b'(' {
                        if depth == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        depth += 1;
                        i += 1;
                        continue;
                    }
                    if c == b')' {
                        depth -= 1;
                        i += 1;
                        if depth == 0 {
                            continue;
                        }
                        continue;
                    }
                    if !(0x20..=0x7e).contains(&c) && c != b'\t' {
                        return Err(EmailError::CommentChar { at: i });
                    }
                    i += 1;
                    continue;
                }

                if is_hws(c) {
                    i += 1;
                    continue;
                }
                if let Some(next) = is_fold(input, i) {
                    i = next;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    outer_comment = i;
                    depth = 1;
                    i += 1;
                    continue;
                }
                break;
            }
            runs.push(Run {
                start: run_start,
                end: i,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        kept.push(b);
        map.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }

    for run in &runs {
        let before = map.iter().position(|&p| p >= run.start).map(|x| x);
        let left = before.and_then(|idx| idx.checked_sub(1)).and_then(|idx| kept.get(idx));
        let right = before.and_then(|idx| kept.get(idx));
        let legal = before.is_none()
            || before == Some(0)
            || left == Some(&b'@')
            || right == Some(&b'@');
        if !legal {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    Ok((kept, map, runs))
}

fn pos(map: &[usize], i: usize, input_len: usize) -> usize {
    map.get(i).copied().unwrap_or(input_len)
}

fn parse_internal(input: &str, allow_single: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, map, _) = cleanup(bytes)?;
    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = 0usize;
    if clean[0] == b'"' {
        let mut i = 1;
        let mut closing = None;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
                continue;
            }
            if clean[i] == b'"' {
                closing = Some(i);
                break;
            }
            i += 1;
        }
        match closing {
            Some(end) => protected_end = end + 1,
            None => return Err(EmailError::UnclosedQuote { at: pos(&map, 0, bytes.len()) }),
        }
    }

    let separators: Vec<usize> = (protected_end..clean.len())
        .filter(|&i| clean[i] == b'@')
        .collect();
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: pos(&map, separators[1], bytes.len()),
        });
    }
    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&map, sep, bytes.len()),
        });
    }

    let raw_local = &clean[..sep];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;

    if raw_local[0] == b'"' {
        quoted = true;
        let closing = protected_end - 1;
        if closing + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&map, closing + 1, bytes.len()),
            });
        }
        let mut i = 1;
        while i < closing {
            let b = raw_local[i];
            if b == b'\\' {
                let esc = raw_local.get(i + 1).copied();
                match esc {
                    Some(b'"') | Some(b'\\') | Some(b'@') => {
                        local.push(esc.unwrap());
                        origins.push(pos(&map, i, bytes.len()));
                        i += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: pos(&map, i, bytes.len()) }),
                }
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: pos(&map, i, bytes.len()) });
                }
                local.push(b);
                origins.push(pos(&map, i, bytes.len()));
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: pos(&map, closing, bytes.len()) });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        if raw_local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: pos(&map, 64, bytes.len()) });
        }
        let mut prev_dot = false;
        for (i, &b) in raw_local.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == raw_local.len() || prev_dot {
                    return Err(EmailError::LocalDot { at: pos(&map, i, bytes.len()) });
                }
                prev_dot = true;
            } else {
                prev_dot = false;
                if !b.is_ascii_alphanumeric()
                    && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
                {
                    return Err(EmailError::LocalChar { at: pos(&map, i, bytes.len()) });
                }
            }
            local.push(b);
            origins.push(pos(&map, i, bytes.len()));
        }
    }

    let domain_start = sep + 1;
    if domain_start == clean.len() {
        return Err(EmailError::EmptyDomain {
            at: pos(&map, domain_start, bytes.len()),
        });
    }

    let mut domain_end = clean.len();
    let root_pos = if clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(pos(&map, domain_end, bytes.len()))
    } else {
        None
    };
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: pos(&map, domain_start, bytes.len()),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(&map, domain_start + 253, bytes.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..domain_end {
        if clean[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    labels.push((start, domain_end));

    for (idx, &(a, b)) in labels.iter().enumerate() {
        if a == b {
            let at = if idx + 1 == labels.len() {
                root_pos.unwrap_or_else(|| pos(&map, a, bytes.len()))
            } else {
                pos(&map, a, bytes.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if b - a > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(&map, a + 63, bytes.len()),
            });
        }
        if clean[a] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(&map, a, bytes.len()) });
        }
        if clean[b - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(&map, b - 1, bytes.len()) });
        }
        for i in a..b {
            let c = clean[i];
            if !c.is_ascii_alphanumeric() && c != b'-' {
                return Err(EmailError::DomainChar { at: pos(&map, i, bytes.len()) });
            }
        }
    }

    if labels.len() < 2 && !allow_single {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&map, domain_start, bytes.len()),
        });
    }
    let (last_start, last_end) = labels[labels.len() - 1];
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(&map, last_start, bytes.len()) });
    }

    let local_string = String::from_utf8(local).unwrap();
    let domain_string = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();
    Ok((
        Address {
            local: local_string,
            domain: domain_string,
            quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, false).map(|x| x.0)
}

fn valid_dot_atom(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() || b.len() > 64 {
        return false;
    }
    let mut prev_dot = false;
    for (i, &c) in b.iter().enumerate() {
        if c == b'.' {
            if i == 0 || i + 1 == b.len() || prev_dot {
                return false;
            }
            prev_dot = true;
        } else {
            prev_dot = false;
            if !c.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&c) {
                return false;
            }
        }
    }
    true
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_internal(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(index) = address.local.find('+') {
            if index == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.get(index).copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(index);
        }
    }
    if !valid_dot_atom(&address.local) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, original)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    result.duplicate_of[index] = Some(*original);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    result.source_indices.push(index);
                    result.addresses.push(address);
                }
            }
        }
    }

    result
}
