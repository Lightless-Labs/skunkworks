#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn lexical_cleanup(input: &str) -> Result<(Vec<u8>, Vec<usize>, Vec<(usize, usize)>), EmailError> {
    let bytes = input.as_bytes();
    let mut kept = Vec::new();
    let mut map = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut quote_protected = false;

    while i < bytes.len() {
        let b = bytes[i];

        if quote_protected {
            kept.push(b);
            map.push(i);
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    kept.push(bytes[i]);
                    map.push(i);
                    i += 1;
                }
                continue;
            }
            if b == b'"' {
                quote_protected = false;
            }
            i += 1;
            continue;
        }

        if kept.is_empty() && b == b'"' {
            quote_protected = true;
            kept.push(b);
            map.push(i);
            i += 1;
            continue;
        }

        let start = i;
        let mut comment_depth = 0usize;
        let mut outer_open = 0usize;
        let mut in_comment = false;

        loop {
            if i >= bytes.len() {
                if in_comment {
                    return Err(EmailError::UnclosedComment { at: outer_open });
                }
                if i > start {
                    runs.push((start, i));
                }
                break;
            }

            let c = bytes[i];

            if in_comment {
                if c == b'\\' {
                    if i + 1 >= bytes.len()
                        || !(0x20..=0x7e).contains(&bytes[i + 1])
                    {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                    continue;
                }
                if let Some(end) = fold_end(bytes, i) {
                    i = end;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    if comment_depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    comment_depth += 1;
                    i += 1;
                    continue;
                }
                if c == b')' {
                    comment_depth -= 1;
                    i += 1;
                    if comment_depth == 0 {
                        in_comment = false;
                    }
                    continue;
                }
                if !(c == b'\t' || (0x20..=0x7e).contains(&c)) {
                    return Err(EmailError::CommentChar { at: i });
                }
                i += 1;
                continue;
            }

            if is_hws(c) {
                i += 1;
                continue;
            }
            if let Some(end) = fold_end(bytes, i) {
                i = end;
                continue;
            }
            if c == b'\r' || c == b'\n' {
                return Err(EmailError::BadFold { at: i });
            }
            if c == b'(' {
                outer_open = i;
                comment_depth = 1;
                in_comment = true;
                i += 1;
                continue;
            }
            if c == b')' {
                return Err(EmailError::UnexpectedClose { at: i });
            }
            break;
        }

        if i > start {
            runs.push((start, i));
        }
        if i >= bytes.len() {
            break;
        }

        // The first retained quote opens the sole protected quoted prefix.
        if kept.is_empty() && bytes[i] == b'"' {
            quote_protected = true;
        }
        kept.push(bytes[i]);
        map.push(i);
        i += 1;
    }

    Ok((kept, map, runs))
}

fn is_atom_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(s: &[u8]) -> bool {
    if s.is_empty() {
        return false;
    }
    let mut prev_dot = true;
    for (i, &b) in s.iter().enumerate() {
        if b == b'.' {
            if prev_dot || i + 1 == s.len() {
                return false;
            }
            prev_dot = true;
        } else if is_atom_byte(b) {
            prev_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_internal(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let (clean, map, runs) = lexical_cleanup(input)?;
    for &(a, b) in &runs {
        let left_at = a > 0 && clean.get(a.saturating_sub(1)).copied() == Some(b'@');
        let right_at = b < bytes.len() && clean.get(0..).and_then(|_| None::<u8>).is_some();
        let _ = right_at;
        // Determine adjacency using original retained positions.
        let left_retained = map.iter().position(|&p| p >= a).and_then(|k| k.checked_sub(1));
        let right_retained = map.iter().position(|&p| p >= b);
        let adjacent = left_retained.is_some_and(|k| clean.get(k) == Some(&b'@'))
            || right_retained.is_some_and(|k| clean.get(k) == Some(&b'@'));
        let boundary = a == 0 || b == bytes.len();
        let _ = left_at;
        if !boundary && !adjacent {
            return Err(EmailError::CfwsBoundary { at: a });
        }
    }
    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut protected_end = 0usize;
    if clean.first() == Some(&b'"') {
        let mut i = 1;
        let mut found = false;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                protected_end = i + 1;
                found = true;
                break;
            } else {
                i += 1;
            }
        }
        if !found {
            return Err(EmailError::UnclosedQuote { at: map[0] });
        }
    }

    let ats: Vec<usize> = clean.iter().enumerate()
        .filter_map(|(i, &b)| (b == b'@' && i >= protected_end).then_some(i))
        .collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt { at: map[ats[1]] });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: map[sep] });
    }

    let local_raw = &clean[..sep];
    let local_map = &map[..sep];
    let (mut local, mut origins, quoted) = if local_raw[0] == b'"' {
        let close = protected_end - 1;
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: map[close + 1] });
        }
        let mut out = Vec::new();
        let mut org = Vec::new();
        let mut i = 1;
        while i < close {
            if local_raw[i] == b'\\' {
                let next = local_raw.get(i + 1).copied();
                if !matches!(next, Some(b'"' | b'\\' | b'@')) {
                    return Err(EmailError::BadEscape { at: local_map[i] });
                }
                out.push(next.unwrap());
                org.push(local_map[i]);
                i += 2;
            } else {
                let b = local_raw[i];
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: local_map[i] });
                }
                out.push(b);
                org.push(local_map[i]);
                i += 1;
            }
        }
        if out.is_empty() {
            return Err(EmailError::EmptyLocal { at: local_map[close] });
        }
        if out.len() > 64 {
            return Err(EmailError::LocalTooLong { at: org[64] });
        }
        (out, org, true)
    } else {
        if local_raw.len() > 64 {
            return Err(EmailError::LocalTooLong { at: local_map[64] });
        }
        let mut prev_dot = false;
        for (i, &b) in local_raw.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_raw.len() || prev_dot {
                    return Err(EmailError::LocalDot { at: local_map[i] });
                }
                prev_dot = true;
            } else {
                if !is_atom_byte(b) {
                    return Err(EmailError::LocalChar { at: local_map[i] });
                }
                prev_dot = false;
            }
        }
        (local_raw.to_vec(), local_map.to_vec(), false)
    };

    let domain_start = sep + 1;
    let mut dom_end = clean.len();
    let mut root_pos = None;
    if dom_end > domain_start && clean[dom_end - 1] == b'.' {
        root_pos = Some(map[dom_end - 1]);
        dom_end -= 1;
    }
    if domain_start == dom_end {
        return Err(EmailError::EmptyDomain { at: map.get(domain_start).copied().unwrap_or(input.len()) });
    }
    if dom_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong { at: map[domain_start + 253] });
    }

    let mut labels = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=dom_end {
        if i == dom_end || clean[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    for (idx, &(a, b)) in labels.iter().enumerate() {
        if a == b {
            let at = if idx + 1 == labels.len() {
                root_pos.unwrap_or_else(|| map.get(a).copied().unwrap_or(input.len()))
            } else {
                map.get(a).copied().unwrap_or(input.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if b - a > 63 {
            return Err(EmailError::LabelTooLong { at: map[a + 63] });
        }
        if clean[a] == b'-' {
            return Err(EmailError::LabelHyphen { at: map[a] });
        }
        if clean[b - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: map[b - 1] });
        }
        for i in a..b {
            if !clean[i].is_ascii_alphanumeric() && clean[i] != b'-' {
                return Err(EmailError::DomainChar { at: map[i] });
            }
        }
    }
    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot { at: map[domain_start] });
    }
    let (last_a, last_b) = labels[labels.len() - 1];
    if clean[last_a..last_b].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: map[last_a] });
    }

    let domain = clean[domain_start..dom_end].iter()
        .map(|&b| b.to_ascii_lowercase() as char).collect::<String>();
    Ok((Address {
        local: String::from_utf8(local).unwrap(),
        domain,
        quoted,
    }, origins))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_internal(input, false).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_internal(input, policy.allow_single_label)?;
    let mut bytes = address.local.into_bytes();
    if policy.local_case == LocalCase::Lower {
        for b in &mut bytes {
            *b = b.to_ascii_lowercase();
        }
    }
    let mut new_origins = origins;
    if policy.strip_subaddress {
        if let Some(plus) = bytes.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: new_origins.get(plus).copied().unwrap_or(input.len()),
                });
            }
            bytes.truncate(plus);
            new_origins.truncate(plus);
        }
    }
    if !valid_dot_atom(&bytes) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    address.local = String::from_utf8(bytes).unwrap();
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: std::collections::HashMap<(String, String), usize> =
        std::collections::HashMap::new();
    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => result.errors.push((i, e)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first) = keys.get(&key) {
                    result.duplicate_of[i] = Some(first);
                } else {
                    keys.insert(key, i);
                    result.source_indices.push(i);
                    result.addresses.push(address);
                }
            }
        }
    }
    result
}
