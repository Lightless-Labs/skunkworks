use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn normalize_from(path: &str, base: &str) -> String {
        let mut parts: Vec<&str> = if path.starts_with('/') {
            Vec::new()
        } else {
            base.split('/').filter(|s| !s.is_empty()).collect()
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part),
            }
        }
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        Ok(Self::normalize_from(path, &self.cwd))
    }

    fn parent(path: &str) -> (&str, &str) {
        if path == "/" {
            return ("/", "");
        }
        match path.rfind('/') {
            Some(0) => ("/", &path[1..]),
            Some(i) => (&path[..i], &path[i + 1..]),
            None => ("/", path),
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            path.starts_with('/')
        } else {
            path == dir
                || path
                    .strip_prefix(dir)
                    .is_some_and(|rest| rest.starts_with('/'))
        }
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn get_entry(&self, path: &str) -> Option<&Entry> {
        self.entries.get(path)
    }

    fn resolve(&self, initial: &str, create_final: bool) -> Result<String, Error> {
        let mut pending: Vec<String> = initial
            .split('/')
            .filter(|s| !s.is_empty())
            .map(str::to_string)
            .collect();
        let mut current = "/".to_string();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        while !pending.is_empty() {
            let component = pending.remove(0);
            let candidate = Self::child_path(&current, &component);
            let is_final = pending.is_empty();

            match self.get_entry(&candidate) {
                None => {
                    if create_final && is_final {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
                Some(Entry::Dir) => current = candidate,
                Some(Entry::File(_)) => {
                    if is_final {
                        return Ok(candidate);
                    }
                    return Err(Error::NotDir(candidate));
                }
                Some(Entry::Link(target)) => {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let physical_parent = Self::parent(&candidate).0.to_string();
                    let mut expanded = Self::normalize_from(target, &physical_parent);
                    let suffix = pending;
                    if !suffix.is_empty() {
                        if expanded == "/" {
                            expanded.push_str(&suffix.join("/"));
                        } else {
                            expanded.push('/');
                            expanded.push_str(&suffix.join("/"));
                        }
                    }
                    let expanded = Self::normalize_from(&expanded, "/");
                    pending = expanded
                        .split('/')
                        .filter(|s| !s.is_empty())
                        .map(str::to_string)
                        .collect();
                    current = "/".to_string();
                }
            }
        }
        Ok(current)
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent(path);
        let physical = self.resolve(parent, false)?;
        match self.get_entry(&physical) {
            Some(Entry::Dir) => Ok((physical, name.to_string())),
            Some(_) => Err(Error::NotDir(physical)),
            None => Err(Error::Missing(physical)),
        }
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|p| p != path && Self::is_within(p, path))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&canonical)?;
        let dest = Self::child_path(&parent, &name);
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let dest = self.resolve(&canonical, true)?;
        match self.entries.get(&dest) {
            Some(Entry::Dir) => Err(Error::IsDir(dest)),
            Some(Entry::File(_)) | None => {
                let (parent, _) = self.resolve_parent(&dest)?;
                let actual = Self::child_path(&parent, Self::parent(&dest).1);
                self.entries.insert(actual, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let mut result = Vec::new();
        for (p, entry) in &self.entries {
            if p == "/" || p == &resolved {
                continue;
            }
            let (parent, name) = Self::parent(p);
            if parent == resolved {
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                    Entry::Link(_) => Kind::Link,
                };
                result.push((name.to_string(), kind));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            _ => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let src = self.normalize(from)?;
        let dst = self.normalize(to)?;
        if src == "/" || dst == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&src)?;
        let source = Self::child_path(&src_parent, &src_name);
        let source_entry = self.entries.get(&source).cloned().ok_or_else(|| Error::Missing(source.clone()))?;
        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&dst)?;
        let destination = Self::child_path(&dst_parent, &dst_name);
        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination)
        {
            return Err(Error::Busy(destination));
        }
        if source == destination {
            return Ok(());
        }
        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let dest_entry = self.entries.get(&destination).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }
        if let Some(ref dest_entry) = dest_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&destination) {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        if dest_entry.is_some() {
            self.entries.retain(|p, _| !Self::is_within(p, &destination));
        }
        let moved: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(p, _)| Self::is_within(p, &source))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();
        self.entries.retain(|p, _| !Self::is_within(p, &source));
        for (old, entry) in moved {
            let suffix = if old == source {
                String::new()
            } else {
                old[source.len()..].to_string()
            };
            self.entries.insert(format!("{}{}", destination, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let target = Self::child_path(&parent, &name);
        let entry = self.entries.get(&target).ok_or_else(|| Error::Missing(target.clone()))?;
        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if matches!(entry, Entry::Dir) && self.has_children(&target) && !recursive {
            return Err(Error::NotEmpty(target));
        }
        self.entries.retain(|p, _| !Self::is_within(p, &target));
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let dest = Self::child_path(&parent, &name);
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = Self::parent(&canonical);
        let physical_parent = self.resolve(parent, false)?;
        match self.get_entry(&physical_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(physical_parent)),
            None => return Err(Error::Missing(physical_parent)),
        }
        let target = Self::child_path(&physical_parent, name);
        match self.entries.get(&target) {
            Some(Entry::Link(text)) => Ok(text.clone()),
            Some(_) | None if target == "/" || self.entries.contains_key(&target) => {
                Err(Error::InvalidTarget(0))
            }
            _ => Err(Error::Missing(target)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let mut rows = Vec::new();
        let mut selected: Vec<String> = self
            .entries
            .keys()
            .filter(|p| *p == &resolved || Self::is_within(p, &resolved))
            .cloned()
            .collect();
        selected.sort();

        for item in &selected {
            let size = match self.entries.get(item).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => self
                    .entries
                    .iter()
                    .filter(|(p, _)| *p != item && Self::is_within(p, item))
                    .map(|(_, e)| match e {
                        Entry::File(data) => data.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
            };
            rows.push((item.clone(), size));
        }
        Ok(rows)
    }
}

use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
struct PathInfo {
    absolute: bool,
    components: Vec<String>,
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<PathInfo, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(PathInfo {
            absolute: path.starts_with('/'),
            components: path
                .split('/')
                .filter(|c| !c.is_empty() && *c != ".")
                .map(str::to_string)
                .collect(),
        })
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_string).collect()
        }
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn normalize(info: &PathInfo, cwd: &str) -> Vec<String> {
        let mut components = if info.absolute {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for component in &info.components {
            if component == ".." {
                components.pop();
            } else {
                components.push(component.clone());
            }
        }
        components
    }

    fn parent_components(components: &[String]) -> Vec<String> {
        components[..components.len() - 1].to_vec()
    }

    fn resolve_components(
        &self,
        components: &[String],
        create_final: bool,
    ) -> Result<String, Error> {
        let mut current: Vec<String> = Vec::new();
        let mut pending = components.to_vec();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(Self::canonical(&current));
            }

            let component = pending.remove(0);
            current.push(component);
            let path = Self::canonical(&current);
            let final_component = pending.is_empty();

            let entry = match self.entries.get(&path) {
                Some(entry) => entry.clone(),
                None if create_final && final_component => return Ok(path),
                None => return Err(Error::Missing(path)),
            };

            if let Entry::Link(target) = entry {
                let seen = visited.contains(&path);
                if seen {
                    return Err(Error::Loop(path));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(path));
                }
                visited.insert(path.clone());
                expansions += 1;

                let target_info = Self::validate(&target).map_err(|e| match e {
                    Error::InvalidPath(i) => Error::InvalidTarget(i),
                    _ => Error::InvalidTarget(0),
                })?;
                let mut expanded = if target_info.absolute {
                    Vec::new()
                } else {
                    current[..current.len() - 1].to_vec()
                };
                for part in target_info.components {
                    if part == ".." {
                        expanded.pop();
                    } else {
                        expanded.push(part);
                    }
                }
                expanded.extend(pending);
                current.clear();
                pending = expanded;
                continue;
            }

            if !final_component && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(path));
            }
        }
    }

    fn resolve_parent(&self, components: &[String]) -> Result<String, Error> {
        let parent = Self::parent_components(components);
        self.resolve_components(&parent, false)
    }

    fn ensure_dir(&self, path: &str) -> Result<(), Error> {
        match self.entries.get(path) {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(path.to_string())),
            None => Err(Error::Missing(path.to_string())),
        }
    }

    fn subtree_paths(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                *candidate == path
                    || (path == "/" && candidate.starts_with('/'))
                    || (path != "/" && candidate.starts_with(&format!("{}/", path)))
            })
            .cloned()
            .collect()
    }

    fn contains_path(directory: &str, path: &str) -> bool {
        directory == path
            || directory == "/"
            || path.starts_with(&format!("{}/", directory))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let info = Self::validate(path)?;
        let components = Self::normalize(&info, &self.cwd);
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }
        let parent = self.resolve_parent(&components)?;
        self.ensure_dir(&parent)?;
        let final_path = Self::canonical(&components);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let info = Self::validate(path)?;
        let components = Self::normalize(&info, &self.cwd);
        let resolved = self.resolve_components(&components, true)?;
        if resolved == "/" {
            return Err(Error::IsDir(resolved));
        }
        let parent = Self::canonical(&Self::parent_components(&Self::components(&resolved)));
        self.ensure_dir(&parent)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) | None => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let info = Self::validate(path)?;
        let components = Self::normalize(&info, &self.cwd);
        let resolved = self.resolve_components(&components, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let info = Self::validate(path)?;
        let components = Self::normalize(&info, &self.cwd);
        let resolved = self.resolve_components(&components, false)?;
        self.ensure_dir(&resolved)?;
        let prefix = if resolved == "/" {
            "/".to_string()
        } else {
            format!("{}/", resolved)
        };
        let mut result = Vec::new();
        for (candidate, entry) in &self.entries {
            if candidate == &resolved || !candidate.starts_with(&prefix) {
                continue;
            }
            let rest = &candidate[prefix.len()..];
            if !rest.is_empty() && !rest.contains('/') {
                result.push((rest.to_string(), entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let info = Self::validate(path)?;
        let components = Self::normalize(&info, &self.cwd);
        let resolved = self.resolve_components(&components, false)?;
        self.ensure_dir(&resolved)?;
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from_info = Self::validate(from)?;
        let to_info = Self::validate(to)?;
        let source_components = Self::normalize(&from_info, &self.cwd);
        let destination_components = Self::normalize(&to_info, &self.cwd);

        if source_components.is_empty() || destination_components.is_empty() {
            return Err(Error::Root);
        }

        let source_parent = self.resolve_parent(&source_components)?;
        self.ensure_dir(&source_parent)?;
        let source = Self::canonical(&source_components);
        let source_entry = self.entries.get(&source).cloned().ok_or_else(|| {
            Error::Missing(source.clone())
        })?;

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        let destination_parent = self.resolve_parent(&destination_components)?;
        self.ensure_dir(&destination_parent)?;
        let destination = Self::canonical(&destination_components);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && Self::contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if overwrite {
            if let Some(dest_entry) = &destination_entry {
                match (&source_entry, dest_entry) {
                    (Entry::Dir, Entry::Dir) => {
                        let children = self
                            .entries
                            .keys()
                            .any(|p| p != &destination && Self::contains_path(&destination, p));
                        if children {
                            return Err(Error::NotEmpty(destination));
                        }
                    }
                    (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                    (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                    _ => {}
                }
            }
        }

        let source_paths = self.subtree_paths(&source);
        let mut moved = Vec::new();
        for old in &source_paths {
            let suffix = &old[source.len()..];
            let new = format!("{}{}", destination, suffix);
            moved.push((old.clone(), new, self.entries.get(old).unwrap().clone()));
        }

        if destination_entry.is_some() {
            for old in self.subtree_paths(&destination) {
                self.entries.remove(&old);
            }
        }
        for (old, _, _) in &moved {
            self.entries.remove(old);
        }
        for (_, new, entry) in moved {
            self.entries.insert(new, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let info = Self::validate(path)?;
        let components = Self::normalize(&info, &self.cwd);
        if components.is_empty() {
            return Err(Error::Root);
        }
        let parent = self.resolve_parent(&components)?;
        self.ensure_dir(&parent)?;
        let target = Self::canonical(&components);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && Self::contains_path(&target, &self.cwd) {
            return Err(Error::Busy(target));
        }

        let paths = self.subtree_paths(&target);
        if matches!(entry, Entry::Dir) && paths.len() > 1 && !recursive {
            return Err(Error::NotEmpty(target));
        }
        for p in paths {
            self.entries.remove(&p);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link_info = Self::validate(link)?;
        let link_components = Self::normalize(&link_info, &self.cwd);
        let target_info = Self::validate(target).map_err(|e| match e {
            Error::InvalidPath(i) => Error::InvalidTarget(i),
            _ => Error::InvalidTarget(0),
        })?;
        if target_info.components.is_empty() && target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if link_components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }
        let parent = self.resolve_parent(&link_components)?;
        self.ensure_dir(&parent)?;
        let link_path = Self::canonical(&link_components);
        if self.entries.contains_key(&link_path) {
            return Err(Error::Exists(link_path));
        }
        self.entries.insert(link_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let info = Self::validate(path)?;
        let components = Self::normalize(&info, &self.cwd);
        if components.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        let parent = self.resolve_parent(&components)?;
        self.ensure_dir(&parent)?;
        let final_path = Self::canonical(&components);
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let info = Self::validate(path)?;
        let components = Self::normalize(&info, &self.cwd);
        let resolved = self.resolve_components(&components, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }

        let paths = self.subtree_paths(&resolved);
        let mut sizes = BTreeMap::new();
        for current in &paths {
            let size = match self.entries.get(current).unwrap() {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(current.clone(), size);
        }

        let mut ordered = paths;
        ordered.sort();
        let mut result = Vec::new();
        for current in ordered {
            let total = if matches!(self.entries.get(&current), Some(Entry::Dir)) {
                sizes
                    .iter()
                    .filter(|(candidate, _)| {
                        *candidate == &current
                            || Self::contains_path(&current, candidate)
                    })
                    .map(|(_, size)| *size)
                    .sum()
            } else {
                *sizes.get(&current).unwrap()
            };
            result.push((current, total));
        }
        Ok(result)
    }
}
