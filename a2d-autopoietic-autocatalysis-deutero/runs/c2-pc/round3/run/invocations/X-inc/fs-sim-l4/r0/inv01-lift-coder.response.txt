use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn components(path: &str) -> Vec<String> {
    if path == "/" {
        Vec::new()
    } else {
        path[1..].split('/').map(str::to_owned).collect()
    }
}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn normalize(path: &str, cwd: &str) -> Result<String, Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }

    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(cwd)
    };
    for part in path.split('/') {
        match part {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_owned()),
        }
    }
    Ok(canonical(&parts))
}

fn parent(path: &str) -> (String, String) {
    let mut parts = components(path);
    let name = parts.pop().unwrap_or_default();
    (canonical(&parts), name)
}

fn is_within(path: &str, dir: &str) -> bool {
    path == dir || (dir == "/" && path.starts_with('/'))
        || (dir != "/" && path.strip_prefix(dir).is_some_and(|s| s.starts_with('/')))
}

struct Resolver<'a> {
    entries: &'a BTreeMap<String, Entry>,
    visited: HashSet<String>,
    expansions: usize,
}

impl<'a> Resolver<'a> {
    fn new(entries: &'a BTreeMap<String, Entry>) -> Self {
        Self {
            entries,
            visited: HashSet::new(),
            expansions: 0,
        }
    }

    fn resolve(&mut self, path: &str, follow_final: bool, allow_missing_final: bool) -> Result<String, Error> {
        let mut todo = components(path);
        let mut done: Vec<String> = Vec::new();

        loop {
            let Some(part) = todo.first().cloned() else {
                return Ok(canonical(&done));
            };
            todo.remove(0);
            done.push(part);
            let current = canonical(&done);
            let final_component = todo.is_empty();

            match self.entries.get(&current) {
                None if final_component && allow_missing_final => return Ok(current),
                None => return Err(Error::Missing(current)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) if !final_component => return Err(Error::NotDir(current)),
                Some(Entry::File(_)) => {}
                Some(Entry::Link(target)) if follow_final || !final_component => {
                    if self.visited.contains(&current) {
                        return Err(Error::Loop(current));
                    }
                    if self.expansions >= 8 {
                        return Err(Error::Depth(current));
                    }
                    self.visited.insert(current.clone());
                    self.expansions += 1;

                    let (link_parent, _) = parent(&current);
                    let mut expanded = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        components(&link_parent)
                    };
                    for c in target.split('/') {
                        match c {
                            "" | "." => {}
                            ".." => {
                                expanded.pop();
                            }
                            other => expanded.push(other.to_owned()),
                        }
                    }
                    expanded.extend(todo);
                    let expanded_path = canonical(&expanded);
                    todo = components(&expanded_path);
                    done.clear();
                }
                Some(Entry::Link(_)) => {}
            }
        }
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let p = normalize(path, &self.cwd)?;
        let (par, _) = parent(&p);
        let mut resolver = Resolver::new(&self.entries);
        let resolved_parent = resolver.resolve(&par, true, false)?;
        match self.entries.get(&resolved_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved_parent)),
            None => return Err(Error::Missing(resolved_parent)),
        }
        let resolved = if par == "/" { p.clone() } else {
            let (_, name) = parent(&p);
            if resolved_parent == "/" { format!("/{name}") } else { format!("{resolved_parent}/{name}") }
        };
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let p = normalize(path, &self.cwd)?;
        let mut resolver = Resolver::new(&self.entries);
        let resolved = resolver.resolve(&p, true, true)?;
        if let Some(entry) = self.entries.get(&resolved) {
            match entry {
                Entry::Dir => return Err(Error::IsDir(resolved)),
                Entry::File(_) => {}
                Entry::Link(_) => return Err(Error::Missing(resolved)),
            }
        } else {
            let (par, _) = parent(&resolved);
            let mut parent_resolver = Resolver::new(&self.entries);
            let resolved_parent = parent_resolver.resolve(&par, true, false)?;
            match self.entries.get(&resolved_parent) {
                Some(Entry::Dir) => {}
                Some(_) => return Err(Error::NotDir(resolved_parent)),
                None => return Err(Error::Missing(resolved_parent)),
            }
        }
        self.entries.insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let p = normalize(path, &self.cwd)?;
        let mut resolver = Resolver::new(&self.entries);
        let resolved = resolver.resolve(&p, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => Err(Error::Missing(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let p = normalize(path, &self.cwd)?;
        let mut resolver = Resolver::new(&self.entries);
        let resolved = resolver.resolve(&p, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let prefix = if resolved == "/" { "/".to_owned() } else { format!("{resolved}/") };
        let mut result = Vec::new();
        for (name, entry) in &self.entries {
            if let Some(rest) = name.strip_prefix(&prefix) {
                if !rest.is_empty() && !rest.contains('/') {
                    result.push((rest.to_owned(), entry.kind()));
                }
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let p = normalize(path, &self.cwd)?;
        let mut resolver = Resolver::new(&self.entries);
        let resolved = resolver.resolve(&p, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let src = normalize(from, &self.cwd)?;
        let dst = normalize(to, &self.cwd)?;
        if src == "/" || dst == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = parent(&src);
        let mut r1 = Resolver::new(&self.entries);
        let src_parent = r1.resolve(&src_parent, true, false)?;
        match self.entries.get(&src_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(src_parent)),
            None => return Err(Error::Missing(src_parent)),
        }
        let src = if src_parent == "/" { format!("/{src_name}") } else { format!("{src_parent}/{src_name}") };
        let source = self.entries.get(&src).cloned().ok_or_else(|| Error::Missing(src.clone()))?;
        if source == Entry::Dir && is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent, dst_name) = parent(&dst);
        let mut r2 = Resolver::new(&self.entries);
        let dst_parent = r2.resolve(&dst_parent, true, false)?;
        match self.entries.get(&dst_parent) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(dst_parent)),
            None => return Err(Error::Missing(dst_parent)),
        }
        let dst = if dst_parent == "/" { format!("/{dst_name}") } else { format!("{dst_parent}/{dst_name}") };
        if matches!(self.entries.get(&dst), Some(Entry::Dir)) && is_within(&self.cwd, &dst) {
            return Err(Error::Busy(dst));
        }
        if src == dst {
            return Ok(());
        }
        if source == Entry::Dir && is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        let existing = self.entries.get(&dst).cloned();
        if existing.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }
        if let Some(dest) = existing {
            match (&source, &dest) {
                (Entry::Dir, Entry::Dir) => {
                    if self.entries.keys().any(|p| p != &dst && is_within(p, &dst)) {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self.entries.iter()
            .filter(|(p, _)| *p == &src || is_within(p, &src))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();
        for (p, _) in &moving {
            self.entries.remove(p);
        }
        if existing.is_some() {
            self.entries.retain(|p, _| p != &dst && !is_within(p, &dst));
        }
        for (old, entry) in moving {
            let suffix = old.strip_prefix(&src).unwrap_or("");
            self.entries.insert(format!("{dst}{suffix}"), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let p = normalize(path, &self.cwd)?;
        if p == "/" {
            return Err(Error::Root);
        }
        let (par, name) = parent(&p);
        let mut resolver = Resolver::new(&self.entries);
        let par = resolver.resolve(&par, true, false)?;
        match self.entries.get(&par) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(par)),
            None => return Err(Error::Missing(par)),
        }
        let target = if par == "/" { format!("/{name}") } else { format!("{par}/{name}") };
        let entry = self.entries.get(&target).ok_or_else(|| Error::Missing(target.clone()))?;
        if entry == &Entry::Dir && is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }
        if entry == &Entry::Dir && !recursive && self.entries.keys().any(|p| p != &target && is_within(p, &target)) {
            return Err(Error::NotEmpty(target));
        }
        self.entries.retain(|p, _| p != &target && !is_within(p, &target));
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let p = normalize(link, &self.cwd)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (par, name) = parent(&p);
        let mut resolver = Resolver::new(&self.entries);
        let par = resolver.resolve(&par, true, false)?;
        match self.entries.get(&par) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(par)),
            None => return Err(Error::Missing(par)),
        }
        let p = if par == "/" { format!("/{name}") } else { format!("{par}/{name}") };
        if self.entries.contains_key(&p) {
            return Err(Error::Exists(p));
        }
        self.entries.insert(p, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let p = normalize(path, &self.cwd)?;
        if p == "/" {
            return Err(Error::InvalidTarget(0));
        }
        let (par, name) = parent(&p);
        let mut resolver = Resolver::new(&self.entries);
        let par = resolver.resolve(&par, true, false)?;
        match self.entries.get(&par) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(par)),
            None => return Err(Error::Missing(par)),
        }
        let p = if par == "/" { format!("/{name}") } else { format!("{par}/{name}") };
        match self.entries.get(&p) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(p)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let p = normalize(path, &self.cwd)?;
        let mut resolver = Resolver::new(&self.entries);
        let resolved = resolver.resolve(&p, true, false)?;
        if !self.entries.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }
        let mut rows: Vec<(String, u64)> = self.entries.iter()
            .filter(|(name, _)| *name == &resolved || is_within(name, &resolved))
            .map(|(name, entry)| {
                let size = match entry {
                    Entry::File(data) => data.len() as u64,
                    Entry::Link(target) => target.len() as u64,
                    Entry::Dir => self.entries.iter()
                        .filter(|(child, _)| is_within(child, name))
                        .map(|(_, child_entry)| match child_entry {
                            Entry::File(data) => data.len() as u64,
                            Entry::Link(target) => target.len() as u64,
                            Entry::Dir => 0,
                        }).sum(),
                };
                (name.clone(), size)
            }).collect();
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}
