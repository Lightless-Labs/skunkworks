#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index));

    let separator = match separators.next() {
        Some(index) => index,
        None => return Err(EmailError::MissingAt),
    };
    if let Some(at) = separators.next() {
        return Err(EmailError::MultipleAt { at });
    }

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];
    if local_bytes.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (index, &byte) in local_bytes.iter().enumerate() {
        let position = index;
        if byte == b'.' {
            if index == 0 || index + 1 == local_bytes.len() || local_bytes[index - 1] == b'.' {
                return Err(EmailError::LocalDot { at: position });
            }
        } else if !byte.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
        {
            return Err(EmailError::LocalChar { at: position });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let removed_root_dot = raw_domain.last() == Some(&b'.');
    let domain_bytes = if removed_root_dot {
        &raw_domain[..raw_domain.len() - 1]
    } else {
        raw_domain
    };

    if domain_bytes.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut label_start = 0;
    for (index, &byte) in domain_bytes.iter().enumerate() {
        if byte == b'.' {
            if index == label_start {
                return Err(EmailError::EmptyLabel {
                    at: domain_start + index,
                });
            }
            labels.push((label_start, &domain_bytes[label_start..index]));
            label_start = index + 1;
        }
    }

    if label_start == domain_bytes.len() {
        let at = if removed_root_dot {
            input.len() - 1
        } else {
            domain_start + domain_bytes.len()
        };
        return Err(EmailError::EmptyLabel { at });
    }
    labels.push((label_start, &domain_bytes[label_start..]));

    for &(relative_start, label) in &labels {
        let start = domain_start + relative_start;

        if label.len() > 63 {
            return Err(EmailError::LabelTooLong { at: start + 63 });
        }

        if label.first() == Some(&b'-') {
            return Err(EmailError::LabelHyphen { at: start });
        }
        if label.last() == Some(&b'-') {
            return Err(EmailError::LabelHyphen {
                at: start + label.len() - 1,
            });
        }

        for (index, &byte) in label.iter().enumerate() {
            if !byte.is_ascii_alphanumeric() && byte != b'-' {
                return Err(EmailError::DomainChar { at: start + index });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let local = String::from_utf8(local_bytes.to_vec()).expect("ASCII local part");
    let domain = String::from_utf8(domain_bytes.to_vec())
        .expect("ASCII domain")
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted: false,
    })
}
