#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: closed quote
    let mut record_index = 0usize;
    let mut quote_start = 0usize;
    let mut has_content = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != 2 && b == b'\r' {
            if i + 1 < bytes.len() && bytes[i + 1] == b'\n' {
                record.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut record));
                state = 0;
                has_content = false;
                record_index += 1;
                i += 2;
                continue;
            }
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            0 => {
                if b == quote {
                    quote_start = i;
                    state = 2;
                    has_content = true;
                    i += 1;
                } else if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    has_content = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    has_content = false;
                    record_index += 1;
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    state = 1;
                    has_content = true;
                    i += ch.len_utf8();
                }
            }
            1 => {
                if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    has_content = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    state = 0;
                    has_content = false;
                    record_index += 1;
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    has_content = true;
                    i += ch.len_utf8();
                }
            }
            2 => {
                if b == quote {
                    if i + 1 < bytes.len() && bytes[i + 1] == quote {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = 3;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    has_content = true;
                    i += ch.len_utf8();
                }
            }
            _ => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    has_content = true;
                    i += 1;
                } else if b == b'\n' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    state = 0;
                    has_content = false;
                    record_index += 1;
                    i += 1;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if has_content || !record.is_empty() || !field.is_empty() || state == 3 {
        record.push(field);
        records.push(record);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    if !config.headers {
        return Ok(Table {
            header: None,
            rows: records,
            duplicates: config.duplicates,
        });
    }

    let mut iter = records.into_iter();
    let Some(header) = iter.next() else {
        return Ok(Table {
            header: None,
            rows: Vec::new(),
            duplicates: config.duplicates,
        });
    };

    if config.duplicates == DuplicatePolicy::Reject {
        for second in 0..header.len() {
            if let Some(first) = header[..second].iter().position(|name| name == &header[second]) {
                return Err(CsvError::DuplicateHeader {
                    name: header[second].clone(),
                    first,
                    second,
                });
            }
        }
    }

    let expected = header.len();
    let mut rows = Vec::new();
    for (record, row) in iter.enumerate() {
        if row.len() != expected {
            return Err(CsvError::HeaderWidth {
                record: record + 1,
                expected,
                actual: row.len(),
            });
        }
        rows.push(row);
    }

    Ok(Table {
        header: Some(header),
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

pub enum ColumnRef {
    Index(usize),
    Name(String),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ColumnType {
    Text,
    Integer,
    Decimal,
    Boolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ColumnSpec {
    pub column: ColumnRef,
    pub kind: ColumnType,
    pub nullable: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Null,
    Text(String),
    Integer(i64),
    Decimal(i64),
    Boolean(bool),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CellErrorKind {
    UnknownColumn,
    MissingCell,
    Empty,
    InvalidInteger,
    IntegerOverflow,
    InvalidDecimal,
    DecimalOverflow,
    InvalidBoolean,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct CellError {
    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ErrorPolicy {
    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return false;
    }
    let start = usize::from(bytes[0] == b'-');
    if start == bytes.len() {
        return false;
    }
    if bytes[start] == b'0' {
        return start == 0 && bytes.len() == 1;
    }
    if !(b'1'..=b'9').contains(&bytes[start]) {
        return false;
    }
    bytes[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let bytes = s.as_bytes();
    let start = usize::from(bytes.first() == Some(&b'-'));
    if start >= bytes.len() {
        return false;
    }
    let dot = match bytes[start..].iter().position(|&b| b == b'.') {
        Some(pos) => start + pos,
        None => return false,
    };
    let integral = &bytes[start..dot];
    let fraction = &bytes[dot + 1..];
    if fraction.len() != 2 || !fraction.iter().all(u8::is_ascii_digit) || integral.is_empty() {
        return false;
    }
    if integral[0] == b'0' {
        integral.len() == 1
    } else {
        (b'1'..=b'9').contains(&integral[0])
            && integral[1..].iter().all(u8::is_ascii_digit)
    }
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) || s == "-0.00" {
        return Err(CellErrorKind::InvalidDecimal);
    }

    let negative = s.starts_with('-');
    let body = if negative { &s[1..] } else { s };
    let (whole, fraction) = body.split_once('.').unwrap();

    // Parse the unsigned magnitude as u64 so the magnitude of i64::MIN fits.
    let magnitude_text = format!("{whole}{fraction}");
    let magnitude = magnitude_text
        .parse::<u64>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;

    if negative {
        let min_magnitude = (i64::MAX as u64) + 1;
        if magnitude > min_magnitude {
            return Err(CellErrorKind::DecimalOverflow);
        }
        if magnitude == min_magnitude {
            Ok(i64::MIN)
        } else {
            Ok(-(magnitude as i64))
        }
    } else if magnitude > i64::MAX as u64 {
        Err(CellErrorKind::DecimalOverflow)
    } else {
        Ok(magnitude as i64)
    }
}

fn evaluate_cell(
    table: &Table,
    row: &[String],
    row_index: usize,
    spec: &ColumnSpec,
    spec_index: usize,
) -> (Option<Value>, Option<CellError>) {
    let column = match &spec.column {
        ColumnRef::Index(index) => Some(*index),
        ColumnRef::Name(name) => match table.column_index(name) {
            Some(index) => Some(index),
            None => {
                return (
                    None,
                    Some(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: None,
                        kind: CellErrorKind::UnknownColumn,
                    }),
                )
            }
        },
    };

    let index = column.unwrap();
    let Some(cell) = row.get(index) else {
        return (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(index),
                kind: CellErrorKind::MissingCell,
            }),
        );
    };

    if cell.is_empty() {
        if spec.nullable {
            return (Some(Value::Null), None);
        }
        if spec.kind != ColumnType::Text {
            return (
                None,
                Some(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(index),
                    kind: CellErrorKind::Empty,
                }),
            );
        }
    }

    let result = match spec.kind {
        ColumnType::Text => Ok(Value::Text(cell.clone())),
        ColumnType::Integer => parse_integer(cell).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(cell).map(Value::Decimal),
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    };

    match result {
        Ok(value) => (Some(value), None),
        Err(kind) => (
            None,
            Some(CellError {
                row: row_index,
                spec: spec_index,
                column: Some(index),
                kind,
            }),
        ),
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        for (spec_index, spec) in columns.iter().enumerate() {
            let (value, error) = evaluate_cell(table, row, row_index, spec, spec_index);
            output[spec_index] = value;
            if let Some(error) = error {
                result.errors.push(error);
                match policy {
                    ErrorPolicy::Collect => {}
                    ErrorPolicy::FirstPerRow => break,
                    ErrorPolicy::Stop => {
                        result.rows.push(output);
                        return result;
                    }
                }
            }
        }
        result.rows.push(output);
    }

    result
}
