#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let mut clean = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut protected = false;

    while i < input.len() {
        if protected {
            let b = input[i];
            clean.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    clean.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    protected = false;
                }
                i += 1;
            }
            continue;
        }

        let b = input[i];

        if depth > 0 {
            if b == b'\\' {
                let at = i;
                if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                    return Err(EmailError::CommentEscape { at });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if b == b'\r' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !is_printable(b) && b != b'\t' {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'(' || b == b'\r' || b == b'\n' {
            let start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let x = input[i];
                if x == b' ' || x == b'\t' {
                    i += 1;
                } else if x == b'\r' || x == b'\n' {
                    if x == b'\r' {
                        if let Some(end) = fold_end(input, i) {
                            i = end;
                        } else {
                            return Err(EmailError::BadFold { at: i });
                        }
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if x == b'(' {
                    outer_open = i;
                    depth = 1;
                    i += 1;
                    break;
                } else {
                    break;
                }
            }

            while depth > 0 {
                if i >= input.len() {
                    return Err(EmailError::UnclosedComment { at: outer_open });
                }
                let x = input[i];
                if x == b'\\' {
                    let at = i;
                    if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                        return Err(EmailError::CommentEscape { at });
                    }
                    i += 2;
                } else if x == b'\r' || x == b'\n' {
                    if x == b'\r' {
                        if let Some(end) = fold_end(input, i) {
                            i = end;
                        } else {
                            return Err(EmailError::BadFold { at: i });
                        }
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if x == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                } else if x == b')' {
                    depth -= 1;
                    i += 1;
                } else {
                    if !is_printable(x) && x != b'\t' {
                        return Err(EmailError::CommentChar { at: i });
                    }
                    i += 1;
                }
            }

            let end = i;
            if end > start {
                if let Some(last) = runs.last_mut() {
                    if last.end == start {
                        last.end = end;
                    } else {
                        runs.push(Run { start, end });
                    }
                } else {
                    runs.push(Run { start, end });
                }
            }
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if b == b'"' && clean.is_empty() {
            protected = true;
            clean.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        clean.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }

    Ok((clean, positions, runs))
}

fn orig(pos: &[usize], input_len: usize, i: usize) -> usize {
    pos.get(i).copied().unwrap_or(input_len)
}

fn parse_clean(
    clean: &[u8],
    pos: &[usize],
    input_len: usize,
) -> Result<Address, EmailError> {
    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let quoted_prefix_end = if clean[0] == b'"' {
        let mut i = 1usize;
        let mut closing = None;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                closing = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        match closing {
            Some(end) => Some(end),
            None => return Err(EmailError::UnclosedQuote { at: orig(pos, input_len, 0) }),
        }
    } else {
        None
    };

    let mut ats = Vec::new();
    let mut i = 0usize;
    if let Some(end) = quoted_prefix_end {
        i = end + 1;
    }
    while i < clean.len() {
        if clean[i] == b'@' {
            ats.push(i);
            if ats.len() == 2 {
                return Err(EmailError::MultipleAt { at: orig(pos, input_len, i) });
            }
        }
        i += 1;
    }
    let at_idx = match ats.first().copied() {
        Some(v) => v,
        None => return Err(EmailError::MissingAt),
    };
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal { at: orig(pos, input_len, at_idx) });
    }

    let quoted = clean[0] == b'"';
    let local = if !quoted {
        if at_idx > 64 {
            return Err(EmailError::LocalTooLong { at: orig(pos, input_len, 64) });
        }
        let mut prev_dot = false;
        for j in 0..at_idx {
            let b = clean[j];
            if b == b'.' {
                if j == 0 || j + 1 == at_idx || prev_dot {
                    return Err(EmailError::LocalDot { at: orig(pos, input_len, j) });
                }
                prev_dot = true;
            } else {
                prev_dot = false;
                if !(b.is_ascii_alphanumeric()
                    || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
                {
                    return Err(EmailError::LocalChar { at: orig(pos, input_len, j) });
                }
            }
        }
        String::from_utf8(clean[..at_idx].to_vec()).unwrap()
    } else {
        let end = quoted_prefix_end.unwrap();
        if end + 1 != at_idx {
            return Err(EmailError::TrailingLocal {
                at: orig(pos, input_len, end + 1),
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < end {
            let b = clean[j];
            if b == b'\\' {
                if j + 1 >= end || !matches!(clean[j + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: orig(pos, input_len, j) });
                }
                decoded.push(clean[j + 1]);
                origins.push(orig(pos, input_len, j));
                j += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar { at: orig(pos, input_len, j) });
                }
                decoded.push(b);
                origins.push(orig(pos, input_len, j));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: orig(pos, input_len, end) });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        String::from_utf8(decoded).unwrap()
    };

    let domain_start = at_idx + 1;
    if domain_start == clean.len() {
        return Err(EmailError::EmptyDomain { at: input_len });
    }
    let mut domain_end = clean.len();
    let root_dot = if clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };
    if domain_end == domain_start {
        return Err(EmailError::EmptyDomain {
            at: orig(pos, input_len, domain_start),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: orig(pos, input_len, domain_start + 253),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    let mut j = domain_start;
    while j <= domain_end {
        if j == domain_end || clean[j] == b'.' {
            if j == label_start {
                let err_at = if j == domain_end {
                    root_dot.map(|d| orig(pos, input_len, d))
                        .unwrap_or_else(|| orig(pos, input_len, j))
                } else {
                    orig(pos, input_len, j)
                };
                return Err(EmailError::EmptyLabel { at: err_at });
            }
            let len = j - label_start;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: orig(pos, input_len, label_start + 63),
                });
            }
            if clean[label_start] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: orig(pos, input_len, label_start),
                });
            }
            if clean[j - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: orig(pos, input_len, j - 1),
                });
            }
            for k in label_start..j {
                let b = clean[k];
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar { at: orig(pos, input_len, k) });
                }
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
        j += 1;
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: orig(pos, input_len, domain_start),
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: orig(pos, input_len, last_start),
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();
    Ok(Address { local, domain, quoted })
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, positions, runs) = cleanup(bytes)?;

    for run in runs {
        let left_at = run.start > 0
            && bytes[..run.start].iter().rev().find(|b| **b != b' ' && **b != b'\t')
                .is_some_and(|b| *b == b'@');
        let right_at = run.end < bytes.len()
            && bytes[run.end..].iter().find(|b| **b != b' ' && **b != b'\t')
                .is_some_and(|b| *b == b'@');

        // Boundary adjacency is defined in the cleaned sequence.
        let before = positions.iter().position(|p| *p >= run.end);
        let after = positions.iter().rposition(|p| *p < run.start);
        let adjacent_left = after.is_some_and(|idx| clean[idx] == b'@');
        let adjacent_right = before.is_some_and(|idx| clean[idx] == b'@');
        let at_boundary = after.is_none() || before.is_none() || adjacent_left || adjacent_right;
        let _ = (left_at, right_at);
        if !at_boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    parse_clean(&clean, &positions, bytes.len())
}
