#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

fn validate_config(config: Config) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&config.delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&config.quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if config.delimiter == config.quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = 0;
    if input.starts_with('\u{feff}') {
        pos = '\u{feff}'.len_utf8();
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut record_started = false;

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == 0 {
            return Err(CsvError::Nul {
                record: records.len(),
                byte: pos,
            });
        }

        let outside_quotes = !matches!(state, State::Quoted { .. });
        if outside_quotes && byte == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: records.len(),
                byte: pos,
            });
        }

        match state {
            State::Quoted { opening } => {
                if byte == quote {
                    if bytes.get(pos + 1) == Some(&quote) {
                        field.push(quote as char);
                        pos += 2;
                    } else {
                        state = State::AfterQuote;
                        pos += 1;
                    }
                } else {
                    let ch = input[pos..].chars().next().expect("valid UTF-8 boundary");
                    field.push(ch);
                    pos += ch.len_utf8();
                }
                let _ = opening;
            }
            State::AfterQuote => {
                if byte == delimiter {
                    row.push(std::mem::take(&mut field));
                    record_started = true;
                    state = State::FieldStart;
                    pos += 1;
                } else if byte == b'\n' || byte == b'\r' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record_started = false;
                    state = State::FieldStart;
                    pos += if byte == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte {
                        record: records.len(),
                        byte: pos,
                    });
                }
            }
            State::FieldStart => {
                if byte == quote {
                    state = State::Quoted { opening: pos };
                    record_started = true;
                    pos += 1;
                } else if byte == delimiter {
                    row.push(String::new());
                    record_started = true;
                    pos += 1;
                } else if byte == b'\n' || byte == b'\r' {
                    row.push(String::new());
                    records.push(std::mem::take(&mut row));
                    record_started = false;
                    pos += if byte == b'\r' { 2 } else { 1 };
                } else {
                    let ch = input[pos..].chars().next().expect("valid UTF-8 boundary");
                    field.push(ch);
                    record_started = true;
                    state = State::Unquoted;
                    pos += ch.len_utf8();
                }
            }
            State::Unquoted => {
                if byte == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: records.len(),
                        byte: pos,
                    });
                } else if byte == delimiter {
                    row.push(std::mem::take(&mut field));
                    record_started = true;
                    state = State::FieldStart;
                    pos += 1;
                } else if byte == b'\n' || byte == b'\r' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record_started = false;
                    state = State::FieldStart;
                    pos += if byte == b'\r' { 2 } else { 1 };
                } else {
                    let ch = input[pos..].chars().next().expect("valid UTF-8 boundary");
                    field.push(ch);
                    pos += ch.len_utf8();
                }
            }
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: records.len(),
            byte: opening,
        });
    }

    if record_started || !row.is_empty() || !field.is_empty() {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn make_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut header = None;
    let mut rows = records;

    if config.headers && !rows.is_empty() {
        let parsed_header = rows.remove(0);

        if config.duplicates == DuplicatePolicy::Reject {
            for second in 0..parsed_header.len() {
                if let Some(first) = parsed_header[..second]
                    .iter()
                    .position(|value| value == &parsed_header[second])
                {
                    return Err(CsvError::DuplicateHeader {
                        name: parsed_header[second].clone(),
                        first,
                        second,
                    });
                }
            }
        }

        for (index, row) in rows.iter().enumerate() {
            if row.len() != parsed_header.len() {
                return Err(CsvError::HeaderWidth {
                    record: index + 1,
                    expected: parsed_header.len(),
                    actual: row.len(),
                });
            }
        }
        header = Some(parsed_header);
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config)?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }

        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| make_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}
