#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    busy_until: Option<u64>,
    idle_since: u64,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<&Customer> = Vec::new();
    let mut arrival_index = 0usize;
    let mut time = 0u64;

    loop {
        let mut finishing: Vec<(u32, usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                state.busy_until.filter(|&finish| finish == time).map(|_| {
                    let server_id = state.server.id;
                    // The customer ID is recovered from the scheduled finish event below.
                    (server_id, index, 0)
                })
            })
            .collect();
        finishing.sort_by_key(|&(server_id, _, _)| server_id);

        for (_, index, _) in finishing {
            let server_id = states[index].server.id;
            let event_pos = events.iter().rposition(|event| {
                matches!(event, Event::Start { server, finish, .. } if *server == server_id && *finish == time)
            });
            let customer_id = event_pos
                .and_then(|pos| match events[pos] {
                    Event::Start { customer, .. } => Some(customer),
                    _ => None,
                })
                .expect("occupied server has a matching start");
            events.push(Event::Finish {
                time,
                customer: customer_id,
                server: server_id,
            });
            states[index].busy_until = None;
            states[index].idle_since = time;
        }

        let mut opening_indices: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.server.opens == time).then_some(index)
            })
            .collect();
        opening_indices.sort_by_key(|&index| std::cmp::Reverse(states[index].server.id));
        for index in opening_indices {
            states[index].opened = true;
            events.push(Event::Open {
                time,
                server: states[index].server.id,
            });
        }

        let first_arrival = arrival_index;
        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == time {
            arrival_index += 1;
        }
        for customer in arrivals[first_arrival..arrival_index].iter().copied() {
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(customer);
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting.remove(0);
            let server_id = states[server_index].server.id;
            let finish = customer
                .service
                .checked_add(time)
                .and_then(|end| end.checked_add(states[server_index].server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time,
                })?;

            events.push(Event::Start {
                time,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = time;
                states[server_index].busy_until = None;
            } else {
                states[server_index].busy_until = Some(finish);
            }
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        if arrival_index < arrivals.len() {
            consider(arrivals[arrival_index].arrival);
        }
        for state in &states {
            if !state.opened {
                consider(state.server.opens);
            }
            if let Some(finish) = state.busy_until {
                consider(finish);
            }
        }

        match next_time {
            Some(next) => time = next,
            None => break,
        }
    }

    Ok(events)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<&Customer> = customers.iter().collect();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            idle_since: server.opens,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<&Customer> = Vec::new();
    let mut next_customer = 0usize;
    let mut now = 0u64;

    loop {
        // Completion phase, in ascending server ID order.
        let mut completed: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| (state.busy_until == Some(now)).then_some(i))
            .collect();
        completed.sort_by_key(|&i| states[i].id);
        for index in completed {
            // The customer ID for each scheduled finish is stored in the event
            // schedule below; find it from the matching start record.
            let server_id = states[index].id;
            let customer_id = events.iter().rev().find_map(|event| match event {
                Event::Start {
                    customer,
                    server,
                    finish,
                    ..
                } if *server == server_id && *finish == now => Some(*customer),
                _ => None,
            }).expect("occupied server has a matching start");
            events.push(Event::Finish {
                time: now,
                customer: customer_id,
                server: server_id,
            });
            states[index].busy_until = None;
            states[index].idle_since = now;
        }

        // Open phase, in descending server ID order.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| (!state.opened && state.opens == now).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].id));
        for index in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time: now,
                server: states[index].id,
            });
        }

        // Arrival phase, already ordered by arrival and descending ID.
        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == now
        {
            let customer = ordered_customers[next_customer];
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(customer);
            next_customer += 1;
        }

        // Dispatch from the shared FIFO.
        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting[0];
            let server = &states[server_index];
            let finish = customer
                .service
                .checked_add(now)
                .and_then(|time| time.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            let server_id = server.id;
            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server_id,
                finish,
            });
            waiting.remove(0);

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = now;
            } else {
                states[server_index].busy_until = Some(finish);
            }
        }

        let all_done = next_customer == ordered_customers.len()
            && waiting.is_empty()
            && states.iter().all(|state| state.busy_until.is_none());
        if all_done {
            // Still emit every remaining server opening, even after customers
            // have completed.
            if states.iter().all(|state| state.opened) {
                break;
            }
        }

        let next_arrival = ordered_customers
            .get(next_customer)
            .map(|customer| customer.arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = states.iter().filter_map(|state| state.busy_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&time| time > now)
            .min();

        match next_time {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}
