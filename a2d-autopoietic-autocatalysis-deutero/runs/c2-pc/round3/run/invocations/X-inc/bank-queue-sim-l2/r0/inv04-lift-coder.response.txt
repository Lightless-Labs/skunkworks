```rust
use std::collections::{HashSet, VecDeque};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    let mut server_ids = HashSet::with_capacity(servers.len());
    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if !server_ids.insert(server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    let mut customer_ids = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if !customer_ids.insert(customer.id) {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<&Customer> = customers.iter().collect();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            idle_since: server.opens,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_customer = 0usize;
    let mut now = 0u64;

    loop {
        // Complete scheduled finishes in ascending server ID order.
        let mut completed: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| (state.busy_until == Some(now)).then_some(i))
            .collect();
        completed.sort_by_key(|&i| states[i].id);
        for index in completed {
            let server_id = states[index].id;
            let customer_id = events
                .iter()
                .rev()
                .find_map(|event| match event {
                    Event::Start {
                        customer,
                        server,
                        finish,
                        ..
                    } if *server == server_id && *finish == now => Some(*customer),
                    _ => None,
                })
                .expect("occupied server has a matching start");

            events.push(Event::Finish {
                time: now,
                customer: customer_id,
                server: server_id,
            });
            states[index].busy_until = None;
            states[index].idle_since = now;
        }

        // Open servers in descending server ID order.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| (!state.opened && state.opens == now).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(states[i].id));
        for index in opening {
            states[index].opened = true;
            events.push(Event::Open {
                time: now,
                server: states[index].id,
            });
        }

        // Emit the whole arrival cohort in descending customer ID order.
        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == now
        {
            let customer = ordered_customers[next_customer];
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push_back(customer);
            next_customer += 1;
        }

        // Dispatch from the shared FIFO.
        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer = *waiting.front().expect("queue is not empty");
            let server = &states[server_index];
            let finish = customer
                .service
                .checked_add(now)
                .and_then(|time| time.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            let server_id = server.id;
            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server_id,
                finish,
            });
            waiting.pop_front();

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = now;
            } else {
                states[server_index].busy_until = Some(finish);
            }
        }

        let all_done = next_customer == ordered_customers.len()
            && waiting.is_empty()
            && states.iter().all(|state| state.busy_until.is_none());
        if all_done && states.iter().all(|state| state.opened) {
            break;
        }

        let next_arrival = ordered_customers
            .get(next_customer)
            .map(|customer| customer.arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = states.iter().filter_map(|state| state.busy_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&time| time > now)
            .min();

        match next_time {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}
```
