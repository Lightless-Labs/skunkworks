#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Customer> = Vec::new();
    let mut arrival_index = 0;
    let mut time = None;

    loop {
        let next_finish = states.iter().filter_map(|state| state.busy_until).min();
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.server.opens)
            .min();
        let next_arrival = arrivals.get(arrival_index).map(|customer| customer.arrival);

        let next_time = [next_finish, next_open, next_arrival]
            .into_iter()
            .flatten()
            .min();

        let Some(tick) = next_time else {
            break;
        };
        if time.is_some_and(|previous| tick < previous) {
            unreachable!("simulation time must be nondecreasing");
        }
        time = Some(tick);

        let mut finishing: Vec<(u32, usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (state.busy_until == Some(tick)).then(|| {
                    let customer = match state.busy_until {
                        Some(_) => 0,
                        None => unreachable!(),
                    };
                    (state.server.id, index, customer)
                })
            })
            .collect();

        // The customer ID for a scheduled finish is tracked separately below.
        finishing.sort_by_key(|entry| entry.0);
        for (_, index, _) in finishing {
            let state = &mut states[index];
            // A finish event's customer is stored in the event schedule.
            let customer = match events.iter().rev().find_map(|event| match event {
                Event::Start {
                    time: start_time,
                    customer,
                    server,
                    finish,
                } if *server == state.server.id && *finish == tick && *start_time <= tick => {
                    Some(*customer)
                }
                _ => None,
            }) {
                Some(customer) => customer,
                None => unreachable!("occupied server has a matching start"),
            };
            events.push(Event::Finish {
                time: tick,
                customer,
                server: state.server.id,
            });
            state.busy_until = None;
            state.idle_since = tick;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.server.opens == tick).then_some(index)
            })
            .collect();
        opening.sort_by_key(|&index| std::cmp::Reverse(states[index].server.id));
        for index in opening {
            let state = &mut states[index];
            state.opened = true;
            events.push(Event::Open {
                time: tick,
                server: state.server.id,
            });
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == tick {
            let customer = arrivals[arrival_index].clone();
            events.push(Event::Arrive {
                time: tick,
                customer: customer.id,
            });
            waiting.push(customer);
            arrival_index += 1;
        }

        while !waiting.is_empty() {
            let Some(server_index) = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(index, _)| index)
            else {
                break;
            };

            let customer = waiting.remove(0);
            let server = &states[server_index].server;
            let finish = customer
                .service
                .checked_add(0)
                .and_then(|duration| tick.checked_add(duration))
                .and_then(|start| start.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: tick,
                })?;

            events.push(Event::Start {
                time: tick,
                customer: customer.id,
                server: server.id,
                finish,
            });

            let state = &mut states[server_index];
            if finish == tick {
                events.push(Event::Finish {
                    time: tick,
                    customer: customer.id,
                    server: server.id,
                });
                state.idle_since = tick;
                state.busy_until = None;
            } else {
                state.busy_until = Some(finish);
            }
        }
    }

    Ok(events)
}
