#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind {
    Start,
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub task: usize,
    pub worker: usize,
    pub kind: EventKind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before doing any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest currently available ID.
    let mut available = std::collections::BTreeSet::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.insert(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree.clone();
    while let Some(&id) = available.iter().next() {
        available.remove(&id);
        order.push(id);
        for &child in &children[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }
    if order.len() != n {
        let removed = {
            let mut is_removed = vec![false; n];
            for &id in &order {
                is_removed[id] = true;
            }
            is_removed
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_successor_tail = children[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut running: Vec<(u64, usize, usize)> = Vec::new(); // finish, task, worker
    let mut worker_available = vec![0u64; workers];
    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all running tasks at this time in task-ID order.
        let mut finishing: Vec<(u64, usize, usize)> = Vec::new();
        let mut still_running = Vec::with_capacity(running.len());
        for entry in running.drain(..) {
            if entry.0 == now {
                finishing.push(entry);
            } else {
                still_running.push(entry);
            }
        }
        running = still_running;
        finishing.sort_by_key(|entry| entry.1);
        for &(finish, task_id, worker_id) in &finishing {
            completed[task_id] = true;
            completed_count += 1;
            events.push(Event {
                time: finish,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Finish,
            });
        }

        loop {
            let worker_id = match (0..workers).find(|&w| worker_available[w] <= now) {
                Some(w) => w,
                None => break,
            };

            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id] || tasks[task_id].release > now {
                    continue;
                }
                if !tasks[task_id].deps.iter().all(|&p| completed[p]) {
                    continue;
                }
                let better = match best {
                    None => true,
                    Some(current) => {
                        tail[task_id] > tail[current]
                            || (tail[task_id] == tail[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                    }
                };
                if better {
                    best = Some(task_id);
                }
            }

            let task_id = match best {
                Some(id) => id,
                None => break,
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker_id] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker: worker_id,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });
            makespan = makespan.max(finish);

            if finish == now {
                // Immediate completion of a zero-duration task.
                completed[task_id] = true;
                completed_count += 1;
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((finish, task_id, worker_id));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };
        for &(finish, _, _) in &running {
            consider(finish);
        }
        for task_id in 0..n {
            if !started[task_id] {
                consider(tasks[task_id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        if let Some(time) = next_time {
            now = time;
        } else {
            // With an acyclic graph, this can only arise when remaining tasks
            // depend on a task that has not yet completed; advancing to its
            // running finish is handled above.
            break;
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("all acyclic tasks must be scheduled"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
