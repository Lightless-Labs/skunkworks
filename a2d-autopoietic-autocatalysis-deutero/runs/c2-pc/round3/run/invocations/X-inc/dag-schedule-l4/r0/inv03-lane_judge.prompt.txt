#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {

    Overflow {

        stage: Stage,

    ResourceExceeded {

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate dependency entries before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &limit)) in task.needs.iter().zip(capacity).enumerate() {
            if need > limit {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: limit,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal always removes the smallest available ID.
    let mut available = std::collections::BTreeSet::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.insert(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while let Some(id) = available.pop_first() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }
    if order.len() != n {
        let removed = {
            let mut v = vec![false; n];
            for &id in &order {
                v[id] = true;
            }
            v
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let worker_count = cooldowns.len();
    let resource_count = capacity.len();
    let mut trace = format!("workers={worker_count};resources={resource_count}\n");
    let mut used = vec![0u64; resource_count];

    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut predecessor_remaining: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut worker_free = vec![0u64; worker_count];

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }
    let mut running = Vec::<Running>::new();

    let render_used = |used: &[u64]| -> String {
        if used.is_empty() {
            "-".to_string()
        } else {
            used.iter()
                .map(u64::to_string)
                .collect::<Vec<_>>()
                .join(",")
        }
    };

    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete every task due now, in task ID order.
        let mut due: Vec<Running> = running
            .iter()
            .copied()
            .filter(|item| item.finish == now)
            .collect();
        due.sort_by_key(|item| item.task);

        if !due.is_empty() {
            running.retain(|item| item.finish != now);
            for item in due {
                let id = item.task;
                completed[id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                for (r, &amount) in tasks[id].needs.iter().enumerate() {
                    used[r] -= u64::from(amount);
                }
                trace.push_str(&format!(
                    "{now}|end|{id}|{}|{}\n",
                    item.worker,
                    render_used(&used)
                ));
                for &child in &children[id] {
                    predecessor_remaining[child] -= 1;
                }
            }
        }

        // Repeatedly start one fitting eligible task, reevaluating after each start.
        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || predecessor_remaining[id] != 0
                {
                    continue;
                }
                let fits = tasks[id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r]);
                if !fits {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority
                                        && id < current)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }

            let Some(id) = best else {
                break;
            };

            let worker = (0..worker_count)
                .map(|offset| (id % worker_count + offset) % worker_count)
                .find(|&w| worker_free[w] <= now)
                .expect("a start is attempted only when an available worker exists");

            let finish = now.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let available_at =
                finish
                    .checked_add(cooldowns[worker])
                    .ok_or(Error::Overflow {
                        task: id,
                        stage: Stage::Turnaround,
                    })?;

            started[id] = true;
            worker_free[worker] = available_at;
            for (r, &amount) in tasks[id].needs.iter().enumerate() {
                used[r] += u64::from(amount);
            }
            runs[id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            trace.push_str(&format!(
                "{now}|start|{id}|{worker}|{}\n",
                render_used(&used)
            ));

            if finish == now {
                // Immediate zero-duration completion, before choosing another start.
                completed[id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                for (r, &amount) in tasks[id].needs.iter().enumerate() {
                    used[r] -= u64::from(amount);
                }
                trace.push_str(&format!(
                    "{now}|end|{id}|{worker}|{}\n",
                    render_used(&used)
                ));
                for &child in &children[id] {
                    predecessor_remaining[child] -= 1;
                }
            } else {
                running.push(Running {
                    task: id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the next finish, release, or worker availability event.
        let mut next: Option<u64> = None;
        let mut consider = |event: u64| {
            if event > now && next.map_or(true, |current| event < current) {
                next = Some(event);
            }
        };
        for item in &running {
            consider(item.finish);
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                consider(tasks[id].release);
            }
        }
        for &free_at in &worker_free {
            consider(free_at);
        }

        // If there is no future event, the remaining tasks cannot be made runnable.
        // Acyclicity and valid capacities ensure this is unreachable, but keep the
        // scheduler panic-free if reached.
        let Some(next_time) = next else {
            return Err(Error::Cycle {
                blocked: (0..n).filter(|&id| !completed[id]).collect(),
            });
        };
        now = next_time;
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("all tasks have started"))
        .collect::<Vec<_>>();
    trace.push_str(&format!("makespan={makespan}\n"));

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for resource in 0..capacity.len() {
            if task.needs[resource] > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need: task.needs[resource],
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Kahn traversal, always removing the smallest currently available ID.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }
    let mut topo = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        topo.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }
    if topo.len() != n {
        let mut removed = vec![false; n];
        for &id in &topo {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in topo.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let worker_count = cooldowns.len();
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut used = vec![0u64; capacity.len()];
    let mut worker_free = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut trace = format!("workers={};resources={}\n", worker_count, capacity.len());

    fn append_resources(trace: &mut String, used: &[u64]) {
        if used.is_empty() {
            trace.push('-');
        } else {
            for (i, value) in used.iter().enumerate() {
                if i != 0 {
                    trace.push(',');
                }
                trace.push_str(&value.to_string());
            }
        }
    }

    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete every task finishing now, in ascending task ID order.
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|item| item.finish == now)
            .collect();
        finishing.sort_by_key(|item| item.task);

        for item in finishing {
            for (r, &need) in tasks[item.task].needs.iter().enumerate() {
                used[r] -= u64::from(need);
            }
            completed[item.task] = true;
            completed_count += 1;
            trace.push_str(&format!(
                "{}|end|{}|{}|",
                now, item.task, item.worker
            ));
            append_resources(&mut trace, &used);
            trace.push('\n');
            running.retain(|other| other.task != item.task);
        }

        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if !tasks[id].deps.iter().all(|&pred| completed[pred]) {
                    continue;
                }
                let fits = tasks[id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r]);
                if !fits {
                    continue;
                }
                match best {
                    None => best = Some(id),
                    Some(current) => {
                        let a = (tails[id], tasks[id].priority, std::cmp::Reverse(id));
                        let b = (
                            tails[current],
                            tasks[current].priority,
                            std::cmp::Reverse(current),
                        );
                        if a > b {
                            best = Some(id);
                        }
                    }
                }
            }

            let Some(id) = best else {
                break;
            };

            let preferred = id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&w| worker_free[w] <= now);
            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[id].duration)
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Finish,
                })?;
            let free_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: id,
                    stage: Stage::Turnaround,
                })?;

            started[id] = true;
            worker_free[worker] = free_at;
            for (r, &need) in tasks[id].needs.iter().enumerate() {
                used[r] += u64::from(need);
            }
            runs[id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            running.push(Running {
                task: id,
                worker,
                finish,
            });

            trace.push_str(&format!("{}|start|{}|{}|", now, id, worker));
            append_resources(&mut trace, &used);
            trace.push('\n');

            if finish > makespan {
                makespan = finish;
            }

            // Zero-duration tasks complete immediately before another admission.
            if finish == now {
                for (r, &need) in tasks[id].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                completed[id] = true;
                completed_count += 1;
                trace.push_str(&format!("{}|end|{}|{}|", now, id, worker));
                append_resources(&mut trace, &used);
                trace.push('\n');
                running.retain(|item| item.task != id);
            }
        }

        if completed_count == n {
            break;
        }

        let mut next = None::<u64>;
        for item in &running {
            if item.finish > now {
                next = Some(next.map_or(item.finish, |x| x.min(item.finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |x| x.min(tasks[id].release)));
            }
        }
        for &free_at in &worker_free {
            if free_at > now {
                next = Some(next.map_or(free_at, |x| x.min(free_at)));
            }
        }

        match next {
            Some(time) => now = time,
            None => {
                // With an acyclic graph, valid workers, and finite releases,
                // this can only occur if all tasks have already completed.
                break;
            }
        }
    }

    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate all dependency entries before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resource shape and demand in task order.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists. Dependency order does not
    // affect the ordering of the Kahn traversal.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }
    for list in &mut children {
        list.sort_unstable();
    }

    // Kahn traversal always removes the smallest currently available ID.
    let mut available = std::collections::BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }
    let mut topo = Vec::with_capacity(n);
    while let Some(task_id) = available.pop_first() {
        topo.push(task_id);
        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }
    if topo.len() != n {
        let mut blocked = Vec::new();
        for task_id in 0..n {
            if indegree[task_id] > 0 {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in topo.iter().rev() {
        let max_child_tail = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    );
    let mut used = vec![0u64; capacity.len()];
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];

    // Worker availability is the earliest time it may be assigned again.
    let mut worker_available = vec![0u64; worker_count];

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }
    let mut running: Vec<Running> = Vec::new();

    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut completed_count = 0usize;

    while completed_count < n {
        // Complete all tasks finishing now, in ascending task ID.
        let mut ending: Vec<Running> = running
            .iter()
            .copied()
            .filter(|entry| entry.finish == now)
            .collect();
        ending.sort_unstable_by_key(|entry| entry.task);

        if !ending.is_empty() {
            running.retain(|entry| entry.finish != now);
            for entry in ending {
                let task_id = entry.task;
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    entry.worker,
                    format_resources(&used)
                ));
            }
        }

        // Start one task at a time and re-evaluate eligibility after each.
        let mut started_any = false;
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&task_id| {
                    if started[task_id] || tasks[task_id].release > now {
                        return false;
                    }
                    if !tasks[task_id]
                        .deps
                        .iter()
                        .all(|&predecessor| completed[predecessor])
                    {
                        return false;
                    }
                    tasks[task_id]
                        .needs
                        .iter()
                        .enumerate()
                        .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r])
                })
                .collect();

            candidates.sort_unstable_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            // Choose the first available worker in the task's cyclic scan.
            let mut chosen_worker = None;
            for offset in 0..worker_count {
                let worker = (task_id % worker_count + offset) % worker_count;
                if worker_available[worker] <= now {
                    chosen_worker = Some(worker);
                    break;
                }
            }
            let Some(worker) = chosen_worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] += u64::from(need);
            }
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            makespan = makespan.max(finish);
            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                format_resources(&used)
            ));
            started_any = true;

            if finish == now {
                // Immediate completion for zero-duration work.
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                completed[task_id] = true;
                completed_count += 1;
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    format_resources(&used)
                ));
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // If anything started, keep scheduling at this same time. The loop
        // above has already exhausted all current admissions.
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |old| old.min(time)));
            }
        };

        for entry in &running {
            consider(entry.finish);
        }
        for task_id in 0..n {
            if !started[task_id] {
                consider(tasks[task_id].release);
            }
        }
        for &available_at in &worker_available {
            consider(available_at);
        }

        if let Some(time) = next_time {
            now = time;
        } else if !started_any {
            // With an acyclic graph and finite releases, this should not occur.
            // It is kept as a defensive guard against an accidental infinite loop.
            break;
        }
    }

    trace.push_str(&format!("makespan={}\n", makespan));
    let runs = runs
        .into_iter()
        .map(|run| run.expect("all tasks complete"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn format_resources(used: &[u64]) -> String {
    if used.is_empty() {
        return "-".to_string();
    }
    let mut out = String::new();
    for (index, value) in used.iter().enumerate() {
        if index > 0 {
            out.push(',');
        }
        out.push_str(&value.to_string());
    }
    out
}
