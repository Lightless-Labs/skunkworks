pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    let count = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = Vec::with_capacity(task.deps.len());
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen.contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen.push(value);
        }
    }

    let mut successors = vec![Vec::new(); count];
    let mut indegree = vec![0usize; count];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(count);
    let mut remaining_indegree = indegree;
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &successors[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != count {
        let mut removed = vec![false; count];
        for &task_id in &order {
            removed[task_id] = true;
        }
        let blocked = (0..count).filter(|&task_id| !removed[task_id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut earliest = vec![Timing { start: 0, finish: 0 }; count];
    for &task_id in &order {
        let start = tasks[task_id]
            .deps
            .iter()
            .map(|&predecessor| earliest[predecessor].finish)
            .max()
            .unwrap_or(0);
        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;
        earliest[task_id] = Timing { start, finish };
    }

    let makespan = earliest.iter().map(|timing| timing.finish).max().unwrap_or(0);
    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut latest = vec![Timing { start: 0, finish: 0 }; count];
    for &task_id in order.iter().rev() {
        let finish = if successors[task_id].is_empty() {
            target
        } else {
            successors[task_id]
                .iter()
                .map(|&child| latest[child].start)
                .min()
                .unwrap()
        };
        let start = finish - tasks[task_id].duration;
        latest[task_id] = Timing { start, finish };
    }

    let slack: Vec<u64> = (0..count)
        .map(|task_id| latest[task_id].start - earliest[task_id].start)
        .collect();
    let baseline = target - makespan;
    let critical = (0..count)
        .filter(|&task_id| slack[task_id] == baseline)
        .collect();

    let path = if count == 0 {
        Vec::new()
    } else {
        let sink = (0..count)
            .filter(|&task_id| successors[task_id].is_empty())
            .filter(|&task_id| earliest[task_id].finish == makespan)
            .max()
            .unwrap();

        let mut reversed_path = vec![sink];
        let mut current = sink;
        while !tasks[current].deps.is_empty() {
            let predecessor = tasks[current]
                .deps
                .iter()
                .copied()
                .max_by_key(|&id| (earliest[id].finish, id))
                .unwrap();
            reversed_path.push(predecessor);
            current = predecessor;
        }
        reversed_path.reverse();
        reversed_path
    };

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}

fn validate_dependencies(tasks: &[Task]) -> Result<Vec<Vec<usize>>, Error> {
    let count = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; count];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let mut children = vec![Vec::new(); count];
    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }
    Ok(children)
}

fn kahn_order(tasks: &[Task], children: &[Vec<usize>]) -> Result<Vec<usize>, Error> {
    let count = tasks.len();
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut available = vec![false; count];

    for id in 0..count {
        available[id] = indegree[id] == 0;
    }

    let mut order = Vec::with_capacity(count);
    loop {
        let next = (0..count).find(|&id| available[id]);
        let Some(id) = next else {
            break;
        };

        available[id] = false;
        order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available[child] = true;
            }
        }
    }

    if order.len() != count {
        let mut removed = vec![false; count];
        for &id in &order {
            removed[id] = true;
        }
        let blocked = (0..count).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    Ok(order)
}

pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    let count = tasks.len();

    let children = validate_dependencies(tasks)?;
    let order = kahn_order(tasks, &children)?;

    let mut earliest: Vec<Option<Timing>> = vec![None; count];
    let mut makespan = 0u64;

    for &id in &order {
        let start = tasks[id]
            .deps
            .iter()
            .map(|&predecessor| earliest[predecessor].as_ref().unwrap().finish)
            .max()
            .unwrap_or(0);

        let finish = start
            .checked_add(tasks[id].duration)
            .ok_or(Error::Overflow { task: id })?;

        earliest[id] = Some(Timing { start, finish });
        makespan = makespan.max(finish);
    }

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let earliest: Vec<Timing> = earliest.into_iter().map(Option::unwrap).collect();

    let mut latest: Vec<Option<Timing>> = vec![None; count];
    for &id in order.iter().rev() {
        let finish = if children[id].is_empty() {
            target
        } else {
            children[id]
                .iter()
                .map(|&child| latest[child].as_ref().unwrap().start)
                .min()
                .unwrap()
        };

        let start = finish - tasks[id].duration;
        latest[id] = Some(Timing { start, finish });
    }
    let latest: Vec<Timing> = latest.into_iter().map(Option::unwrap).collect();

    let slack: Vec<u64> = (0..count)
        .map(|id| latest[id].start - earliest[id].start)
        .collect();
    let baseline = target - makespan;
    let critical = (0..count)
        .filter(|&id| slack[id] == baseline)
        .collect();

    let path = if count == 0 {
        Vec::new()
    } else {
        let mut current = (0..count)
            .filter(|&id| children[id].is_empty() && earliest[id].finish == makespan)
            .max()
            .unwrap();

        let mut reversed_path = vec![current];
        while !tasks[current].deps.is_empty() {
            current = *tasks[current]
                .deps
                .iter()
                .max_by_key(|&&predecessor| (earliest[predecessor].finish, predecessor))
                .unwrap();
            reversed_path.push(current);
        }
        reversed_path.reverse();
        reversed_path
    };

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}

pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    let n = tasks.len();

    // Validate every dependency entry before doing any graph computation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &dependency) in task.deps.iter().enumerate() {
            if dependency >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            if dependency == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dependency) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
        }
    }

    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dependency in &task.deps {
            successors[dependency].push(task_id);
        }
    }

    // Sorted available IDs ensure each iteration removes the smallest ID
    // currently ready, including newly unblocked tasks.
    let mut available = std::collections::BTreeSet::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.insert(task_id);
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree;
    while let Some(&task_id) = available.iter().next() {
        available.remove(&task_id);
        order.push(task_id);

        for &child in &successors[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for task_id in 0..n {
            if remaining_indegree[task_id] != 0 {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut earliest_start = vec![0u64; n];
    let mut earliest = vec![
        Timing {
            start: 0,
            finish: 0,
        };
        n
    ];

    for &task_id in &order {
        let start = tasks[task_id]
            .deps
            .iter()
            .map(|&dependency| earliest[dependency].finish)
            .max()
            .unwrap_or(0);
        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;
        earliest_start[task_id] = start;
        earliest[task_id] = Timing { start, finish };
    }

    let makespan = earliest.iter().map(|timing| timing.finish).max().unwrap_or(0);

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut latest = vec![
        Timing {
            start: 0,
            finish: 0,
        };
        n
    ];
    for &task_id in order.iter().rev() {
        let finish = successors[task_id]
            .iter()
            .map(|&child| latest[child].start)
            .min()
            .unwrap_or(target);
        let start = finish - tasks[task_id].duration;
        latest[task_id] = Timing { start, finish };
    }

    let slack: Vec<u64> = (0..n)
        .map(|task_id| latest[task_id].start - earliest_start[task_id])
        .collect();

    let baseline = target - makespan;
    let critical = (0..n)
        .filter(|&task_id| slack[task_id] == baseline)
        .collect();

    let path = if n == 0 {
        Vec::new()
    } else {
        let mut sink = None;
        for task_id in 0..n {
            if successors[task_id].is_empty()
                && earliest[task_id].finish == makespan
                && sink.is_none_or(|current| task_id > current)
            {
                sink = Some(task_id);
            }
        }

        let mut reversed_path = Vec::new();
        let mut current = sink.expect("a nonempty DAG has a sink");
        loop {
            reversed_path.push(current);
            if tasks[current].deps.is_empty() {
                break;
            }
            current = *tasks[current]
                .deps
                .iter()
                .max_by_key(|&&dependency| (earliest[dependency].finish, dependency))
                .expect("a non-root task has a predecessor");
        }
        reversed_path.reverse();
        reversed_path
    };

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}
