#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Analysis {
    pub order: Vec<usize>,
    pub earliest: Vec<Timing>,
    pub latest: Vec<Timing>,
    pub slack: Vec<u64>,
    pub critical: Vec<usize>,
    pub path: Vec<usize>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Overflow {
        task: usize,
    },
    TargetTooSmall {
        minimum: u64,
    },
}

fn validate_dependencies(tasks: &[Task]) -> Result<Vec<Vec<usize>>, Error> {
    let count = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; count];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let mut children = vec![Vec::new(); count];
    for (task_id, task) in tasks.iter().enumerate() {
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }
    Ok(children)
}

fn kahn_order(tasks: &[Task], children: &[Vec<usize>]) -> Result<Vec<usize>, Error> {
    let count = tasks.len();
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut available = vec![false; count];

    for id in 0..count {
        available[id] = indegree[id] == 0;
    }

    let mut order = Vec::with_capacity(count);
    loop {
        let next = (0..count).find(|&id| available[id]);
        let Some(id) = next else {
            break;
        };

        available[id] = false;
        order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available[child] = true;
            }
        }
    }

    if order.len() != count {
        let mut removed = vec![false; count];
        for &id in &order {
            removed[id] = true;
        }
        let blocked = (0..count).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    Ok(order)
}

pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    let count = tasks.len();

    let children = validate_dependencies(tasks)?;
    let order = kahn_order(tasks, &children)?;

    let mut earliest: Vec<Option<Timing>> = vec![None; count];
    let mut makespan = 0u64;

    for &id in &order {
        let start = tasks[id]
            .deps
            .iter()
            .map(|&predecessor| earliest[predecessor].as_ref().unwrap().finish)
            .max()
            .unwrap_or(0);

        let finish = start
            .checked_add(tasks[id].duration)
            .ok_or(Error::Overflow { task: id })?;

        earliest[id] = Some(Timing { start, finish });
        makespan = makespan.max(finish);
    }

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let earliest: Vec<Timing> = earliest.into_iter().map(Option::unwrap).collect();

    let mut latest: Vec<Option<Timing>> = vec![None; count];
    for &id in order.iter().rev() {
        let finish = if children[id].is_empty() {
            target
        } else {
            children[id]
                .iter()
                .map(|&child| latest[child].as_ref().unwrap().start)
                .min()
                .unwrap()
        };

        let start = finish - tasks[id].duration;
        latest[id] = Some(Timing { start, finish });
    }
    let latest: Vec<Timing> = latest.into_iter().map(Option::unwrap).collect();

    let slack: Vec<u64> = (0..count)
        .map(|id| latest[id].start - earliest[id].start)
        .collect();
    let baseline = target - makespan;
    let critical = (0..count)
        .filter(|&id| slack[id] == baseline)
        .collect();

    let path = if count == 0 {
        Vec::new()
    } else {
        let mut current = (0..count)
            .filter(|&id| children[id].is_empty() && earliest[id].finish == makespan)
            .max()
            .unwrap();

        let mut reversed_path = vec![current];
        while !tasks[current].deps.is_empty() {
            current = *tasks[current]
                .deps
                .iter()
                .max_by_key(|&&predecessor| (earliest[predecessor].finish, predecessor))
                .unwrap();
            reversed_path.push(current);
        }
        reversed_path.reverse();
        reversed_path
    };

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}
