fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 24
        && bytes[0].is_ascii_lowercase()
        && bytes[bytes.len() - 1] != b'-'
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    constraint_texts(constraints.to_vec())
}

fn constraint_texts(constraints: Vec<Constraint>) -> String {
    canonical(constraints)
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            v >= x
                && if x.major > 0 {
                    v.major == x.major
                } else if x.minor > 0 {
                    v.major == 0 && v.minor == x.minor
                } else {
                    v == x
                }
        }
    }
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

type Assignment = std::collections::BTreeMap<String, Version>;
type Active = std::collections::BTreeMap<String, Vec<Constraint>>;

struct Search<'a> {
    catalog: &'a std::collections::BTreeMap<String, &'a Package>,
    roots: &'a [Requirement],
    locks: &'a std::collections::BTreeMap<String, Version>,
    budget: usize,
    attempts: usize,
    best: Option<(usize, Vec<Version>, Assignment, Active)>,
    first_error: Option<ResolveError>,
    names: Vec<String>,
}

impl<'a> Search<'a> {
    fn active(&self, assigned: &Assignment) -> Active {
        let mut active = Active::new();
        for root in self.roots {
            active
                .entry(root.package.clone())
                .or_default()
                .push(root.constraint);
        }
        for (name, version) in assigned {
            if let Some(pkg) = self.catalog.get(name) {
                if let Some(release) = pkg.releases.iter().find(|r| r.version == *version) {
                    for dep in &release.deps {
                        active
                            .entry(dep.package.clone())
                            .or_default()
                            .push(dep.constraint);
                    }
                }
            }
        }
        for cs in active.values_mut() {
            *cs = canonical(std::mem::take(cs));
        }
        active
    }

    fn inspect(&mut self, assigned: &mut Assignment) -> Result<(), ResolveError> {
        let active = self.active(assigned);

        if let Some(name) = active
            .keys()
            .find(|name| !self.catalog.contains_key(*name))
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in assigned.iter() {
            if let Some(cs) = active.get(name) {
                if !cs.iter().all(|c| matches(*version, *c)) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs.clone(),
                    });
                }
            }
        }

        for (name, cs) in &active {
            if assigned.contains_key(name) {
                continue;
            }
            let pkg = self.catalog[name];
            if !pkg
                .releases
                .iter()
                .any(|r| cs.iter().all(|c| matches(r.version, *c)))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs.clone(),
                });
            }
        }

        let next = active.keys().find(|n| !assigned.contains_key(*n)).cloned();
        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|(n, v)| assigned.get(*n) != Some(*v))
                .count();
            let vector = self
                .names
                .iter()
                .map(|n| assigned.get(n).copied())
                .map(|v| v.unwrap_or(Version {
                    major: 0,
                    minor: 0,
                    patch: 0,
                }))
                .collect::<Vec<_>>();
            let replace = self.best.as_ref().is_none_or(|(bc, bv, _, _)| {
                cost < *bc || (cost == *bc && vector > *bv)
            });
            if replace {
                self.best = Some((cost, vector, assigned.clone(), active));
            }
            return Ok(());
        };

        let mut candidates: Vec<Version> = self.catalog[&name]
            .releases
            .iter()
            .filter(|r| active[&name].iter().all(|c| matches(r.version, *c)))
            .map(|r| r.version)
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(locked) = self.locks.get(&name) {
            if let Some(i) = candidates.iter().position(|v| v == locked) {
                let v = candidates.remove(i);
                candidates.insert(0, v);
            }
        }

        for version in candidates {
            if self.attempts == self.budget {
                return Err(ResolveError::Limit {
                    attempts: self.budget,
                });
            }
            self.attempts += 1;
            assigned.insert(name.clone(), version);
            match self.inspect(assigned) {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    assigned.remove(&name);
                    return Err(ResolveError::Limit { attempts });
                }
                Err(e) => {
                    if self.first_error.is_none() {
                        self.first_error = Some(e);
                    }
                }
            }
            assigned.remove(&name);
        }
        Ok(())
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, pkg) in catalog.iter().enumerate() {
        if !valid_name(&pkg.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: pkg.name.clone(),
            });
        }
        for (j, release) in pkg.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen = std::collections::BTreeSet::new();
    for pkg in catalog {
        if !seen.insert(pkg.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: pkg.name.clone(),
            });
        }
    }
    for pkg in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &pkg.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: pkg.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut lock_map = std::collections::BTreeMap::new();
    for lock in locks {
        if lock_map.insert(lock.package.clone(), lock.version).is_some() {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let cat: std::collections::BTreeMap<String, &Package> =
        catalog.iter().map(|p| (p.name.clone(), p)).collect();
    let mut names = catalog.iter().map(|p| p.name.clone()).collect::<Vec<_>>();
    names.sort();
    let mut search = Search {
        catalog: &cat,
        roots,
        locks: &lock_map,
        budget: max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
        names,
    };
    search.inspect(&mut Assignment::new())?;

    let Some((_, _, assignment, active)) = search.best else {
        return Err(search.first_error.unwrap_or(ResolveError::NoMatch {
            package: roots
                .first()
                .map(|r| r.package.clone())
                .unwrap_or_default(),
            requirements: Vec::new(),
        }));
    };

    let selected = assignment
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut all_names = std::collections::BTreeSet::new();
    all_names.extend(lock_map.keys().cloned());
    all_names.extend(assignment.keys().cloned());
    let mut changes = Vec::new();
    for package in all_names {
        let before = lock_map.get(&package).copied();
        let after = assignment.get(&package).copied();
        if before == after {
            continue;
        }
        let reason = match (before, after) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = cat
                    .get(&package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == old));
                if !exists {
                    ChangeReason::Unavailable
                } else if active
                    .get(&package)
                    .is_some_and(|cs| !cs.iter().all(|c| matches(old, *c)))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => continue,
        };
        changes.push(Change {
            package,
            before,
            after,
            reason,
        });
    }

    Ok(Resolution { selected, changes })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut cs: Vec<Constraint>) -> Vec<Constraint> {
    cs.sort();
    cs.dedup();
    cs
}

fn constraints_text(cs: &[Constraint]) -> String {
    canonical(cs.to_vec())
        .into_iter()
        .map(constraint_text)
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'_' | b'-') {
            out.push(*b as char);
        } else {
            out.push('%');
            out.push_str(&format!("{:02X}", b));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn matches_constraint(version: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => version == x,
        Constraint::AtLeast(x) => version >= x,
        Constraint::Before(x) => version < x,
        Constraint::Caret(x) => {
            version >= x
                && if x.major > 0 {
                    version.major == x.major
                } else if x.minor > 0 {
                    version.major == 0 && version.minor == x.minor
                } else {
                    version == x
                }
        }
    }
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|c| matches_constraint(version, *c))
}

fn validate(catalog: &[Package], roots: &[Requirement], locks: &[Lock]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }

    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }
    Ok(())
}

fn active_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    selected: &std::collections::BTreeMap<String, Version>,
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in selected {
        if let Some(package) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = package.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for constraints in active.values_mut() {
        *constraints = canonical(std::mem::take(constraints));
    }
    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, std::collections::BTreeMap<String, Version>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn inspect(&mut self, selected: &mut std::collections::BTreeMap<String, Version>) -> Result<(), ResolveError> {
        let active = active_constraints(self.roots, self.catalog, selected);

        for name in active.keys() {
            if !self.catalog.iter().any(|p| p.name == *name) {
                return Err(ResolveError::Missing {
                    package: name.clone(),
                });
            }
        }

        for (name, version) in selected.iter() {
            if let Some(constraints) = active.get(name) {
                if !satisfies(*version, constraints) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: constraints.clone(),
                    });
                }
            }
        }

        for (name, constraints) in &active {
            if selected.contains_key(name) {
                continue;
            }
            let package = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !package
                .releases
                .iter()
                .any(|r| satisfies(r.version, constraints))
            {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: constraints.clone(),
                });
            }
        }

        let next = active.keys().find(|name| !selected.contains_key(*name)).cloned();
        let Some(name) = next else {
            let cost = self
                .locks
                .iter()
                .filter(|lock| selected.get(&lock.package) != Some(&lock.version))
                .count();
            let replace = match &self.best {
                None => true,
                Some((best_cost, best_selection)) => {
                    if cost != *best_cost {
                        cost < *best_cost
                    } else {
                        self.vector_cmp(selected, best_selection).is_gt()
                    }
                }
            };
            if replace {
                self.best = Some((cost, selected.clone()));
            }
            return Ok(());
        };

        let constraints = active.get(&name).unwrap();
        let package = self.catalog.iter().find(|p| p.name == name).unwrap();
        let mut candidates = package
            .releases
            .iter()
            .filter(|r| satisfies(r.version, constraints))
            .map(|r| r.version)
            .collect::<Vec<_>>();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(index) = candidates.iter().position(|v| *v == lock.version) {
                let v = candidates.remove(index);
                candidates.insert(0, v);
            }
        }

        let mut branch_error = None;
        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.insert(name.clone(), version);
            let result = self.inspect(selected);
            selected.remove(&name);

            match result {
                Ok(()) => {}
                Err(ResolveError::Limit { .. }) => return result,
                Err(err) => {
                    if branch_error.is_none() {
                        branch_error = Some(err);
                    }
                }
            }
        }

        if let Some(err) = branch_error {
            if self.first_error.is_none() {
                self.first_error = Some(err.clone());
            }
            Err(err)
        } else {
            Ok(())
        }
    }

    fn vector_cmp(
        &self,
        a: &std::collections::BTreeMap<String, Version>,
        b: &std::collections::BTreeMap<String, Version>,
    ) -> std::cmp::Ordering {
        let names = self
            .catalog
            .iter()
            .map(|p| p.name.as_str())
            .collect::<std::collections::BTreeSet<_>>();
        for name in names {
            match (a.get(name), b.get(name)) {
                (None, None) => {}
                (None, Some(_)) => return std::cmp::Ordering::Less,
                (Some(_), None) => return std::cmp::Ordering::Greater,
                (Some(av), Some(bv)) => {
                    let cmp = av.cmp(bv);
                    if !cmp.is_eq() {
                        return cmp;
                    }
                }
            }
        }
        std::cmp::Ordering::Equal
    }
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    validate(catalog, roots, locks)?;

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut selected = std::collections::BTreeMap::new();
    let result = search.inspect(&mut selected);
    if let Err(ResolveError::Limit { attempts }) = result {
        return Err(ResolveError::Limit { attempts });
    }

    let final_selection = match search.best {
        Some((_, selection)) => selection,
        None => return Err(search.first_error.unwrap_or_else(|| {
            ResolveError::NoMatch {
                package: String::new(),
                requirements: Vec::new(),
            }
        })),
    };

    let active = active_constraints(roots, catalog, &final_selection);
    let selected_vec = final_selection
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect::<Vec<_>>();

    let mut changes = Vec::new();
    for lock in locks {
        let after = final_selection.get(&lock.package).copied();
        if after == Some(lock.version) {
            continue;
        }
        let reason = match after {
            None => ChangeReason::Removed,
            Some(_) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == lock.package)
                    .is_some_and(|p| p.releases.iter().any(|r| r.version == lock.version));
                if !exists {
                    ChangeReason::Unavailable
                } else if active
                    .get(&lock.package)
                    .is_some_and(|cs| !satisfies(lock.version, cs))
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
        };
        changes.push(Change {
            package: lock.package.clone(),
            before: Some(lock.version),
            after,
            reason,
        });
    }
    for (package, version) in &final_selection {
        if !locks.iter().any(|l| l.package == *package) {
            changes.push(Change {
                package: package.clone(),
                before: None,
                after: Some(*version),
                reason: ChangeReason::Added,
            });
        }
    }
    changes.sort_by(|a, b| a.package.cmp(&b.package));

    Ok(Resolution {
        selected: selected_vec,
        changes,
    })
}
