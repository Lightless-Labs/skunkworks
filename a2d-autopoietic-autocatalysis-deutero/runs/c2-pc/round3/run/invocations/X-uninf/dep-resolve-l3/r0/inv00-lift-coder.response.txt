#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Lock {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ChangeReason {
    Added,
    Removed,
    Unavailable,
    Rejected,
    Rebalanced,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Change {
    pub package: String,
    pub before: Option<Version>,
    pub after: Option<Version>,
    pub reason: ChangeReason,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Resolution {
    pub selected: Vec<Selection>,
    pub changes: Vec<Change>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    DuplicateLock {
        package: String,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn version_text(v: Version) -> String {
    format!("{}.{}.{}", v.major, v.minor, v.patch)
}

fn constraint_text(c: Constraint) -> String {
    match c {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical(mut constraints: Vec<Constraint>) -> Vec<Constraint> {
    constraints.sort();
    constraints.dedup();
    constraints
}

fn constraints_text(constraints: &[Constraint]) -> String {
    canonical(constraints.to_vec())
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode(s: &str) -> String {
    let mut out = String::new();
    for b in s.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'-') {
            out.push(b as char);
        } else {
            out.push_str(&format!("%{:02X}", b));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => {
            format!("invalid-name|site={}|name={}", encode(site), encode(name))
        }
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode(package),
            version_text(*version)
        ),
        ResolveError::DuplicateLock { package } => {
            format!("duplicate-lock|package={}", encode(package))
        }
        ResolveError::Missing { package } => format!("missing|package={}", encode(package)),
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={}", attempts),
    }
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b.len() <= 24
        && b[0].is_ascii_lowercase()
        && b[b.len() - 1] != b'-'
        && b.iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-')
}

fn matches_constraint(v: Version, c: Constraint) -> bool {
    match c {
        Constraint::Exact(x) => v == x,
        Constraint::AtLeast(x) => v >= x,
        Constraint::Before(x) => v < x,
        Constraint::Caret(x) => {
            if v < x {
                false
            } else if x.major > 0 {
                v.major == x.major
            } else if x.minor > 0 {
                v.major == 0 && v.minor == x.minor
            } else {
                v == x
            }
        }
    }
}

fn matches_all(v: Version, cs: &[Constraint]) -> bool {
    cs.iter().all(|c| matches_constraint(v, *c))
}

fn collect_constraints(
    roots: &[Requirement],
    catalog: &[Package],
    selected: &[(String, Version)],
) -> std::collections::BTreeMap<String, Vec<Constraint>> {
    let mut active = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }
    for (name, version) in selected {
        if let Some(pkg) = catalog.iter().find(|p| p.name == *name) {
            if let Some(release) = pkg.releases.iter().find(|r| r.version == *version) {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }
    for cs in active.values_mut() {
        *cs = canonical(std::mem::take(cs));
    }
    active
}

struct Search<'a> {
    catalog: &'a [Package],
    roots: &'a [Requirement],
    locks: &'a [Lock],
    max_attempts: usize,
    attempts: usize,
    best: Option<(usize, Vec<(String, Version)>)>,
    first_error: Option<ResolveError>,
}

impl<'a> Search<'a> {
    fn inspect(&mut self, selected: &mut Vec<(String, Version)>) -> Result<(), ResolveError> {
        let active = collect_constraints(self.roots, self.catalog, selected);

        if let Some(name) = active
            .keys()
            .filter(|name| !self.catalog.iter().any(|p| p.name == ***name))
            .min()
        {
            return Err(ResolveError::Missing {
                package: name.clone(),
            });
        }

        for (name, version) in selected.iter() {
            if let Some(cs) = active.get(name) {
                if !matches_all(*version, cs) {
                    return Err(ResolveError::Conflict {
                        package: name.clone(),
                        selected: *version,
                        requirements: cs.clone(),
                    });
                }
            }
        }

        for (name, cs) in &active {
            if selected.iter().any(|(n, _)| n == name) {
                continue;
            }
            let pkg = self.catalog.iter().find(|p| p.name == *name).unwrap();
            if !pkg.releases.iter().any(|r| matches_all(r.version, cs)) {
                return Err(ResolveError::NoMatch {
                    package: name.clone(),
                    requirements: cs.clone(),
                });
            }
        }

        let next = active
            .keys()
            .find(|name| !selected.iter().any(|(n, _)| n == *name))
            .cloned();

        let Some(name) = next else {
            self.consider(selected);
            return Ok(());
        };

        let cs = &active[&name];
        let pkg = self.catalog.iter().find(|p| p.name == name).unwrap();
        let mut candidates: Vec<Version> = pkg
            .releases
            .iter()
            .map(|r| r.version)
            .filter(|v| matches_all(*v, cs))
            .collect();
        candidates.sort_by(|a, b| b.cmp(a));

        if let Some(lock) = self.locks.iter().find(|l| l.package == name) {
            if let Some(i) = candidates.iter().position(|v| *v == lock.version) {
                let locked = candidates.remove(i);
                candidates.insert(0, locked);
            }
        }

        let mut first_child_error = None;
        for version in candidates {
            if self.attempts == self.max_attempts {
                return Err(ResolveError::Limit {
                    attempts: self.max_attempts,
                });
            }
            self.attempts += 1;
            selected.push((name.clone(), version));
            let result = self.inspect(selected);
            selected.pop();

            match result {
                Ok(()) => {}
                Err(ResolveError::Limit { attempts }) => {
                    return Err(ResolveError::Limit { attempts });
                }
                Err(e) => {
                    if first_child_error.is_none() {
                        first_child_error = Some(e);
                    }
                }
            }
        }

        if let Some(e) = first_child_error {
            Err(e)
        } else {
            // Candidates were known to be nonempty at this node.
            unreachable!()
        }
    }

    fn consider(&mut self, selected: &[(String, Version)]) {
        let cost = self
            .locks
            .iter()
            .filter(|lock| {
                !selected
                    .iter()
                    .any(|(name, version)| name == &lock.package && version == &lock.version)
            })
            .count();
        let mut candidate = selected.to_vec();
        candidate.sort_by(|a, b| a.0.cmp(&b.0));

        let replace = match &self.best {
            None => true,
            Some((best_cost, best)) if cost < *best_cost => true,
            Some((best_cost, best)) if cost == *best_cost => {
                compare_vectors(&candidate, best, self.catalog).is_gt()
            }
            _ => false,
        };
        if replace {
            self.best = Some((cost, candidate));
        }
    }
}

fn compare_vectors(
    a: &[(String, Version)],
    b: &[(String, Version)],
    catalog: &[Package],
) -> std::cmp::Ordering {
    let mut names: Vec<&str> = catalog.iter().map(|p| p.name.as_str()).collect();
    names.sort_unstable();
    names.dedup();
    for name in names {
        let av = a.iter().find(|(n, _)| n == name).map(|(_, v)| *v);
        let bv = b.iter().find(|(n, _)| n == name).map(|(_, v)| *v);
        let ord = match (av, bv) {
            (None, None) => std::cmp::Ordering::Equal,
            (None, Some(_)) => std::cmp::Ordering::Less,
            (Some(_), None) => std::cmp::Ordering::Greater,
            (Some(x), Some(y)) => x.cmp(&y),
        };
        if !ord.is_eq() {
            return ord;
        }
    }
    std::cmp::Ordering::Equal
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    locks: &[Lock],
    max_attempts: usize,
) -> Result<Resolution, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{}]", i),
                name: root.package.clone(),
            });
        }
    }
    for (i, pkg) in catalog.iter().enumerate() {
        if !valid_name(&pkg.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{}]", i),
                name: pkg.name.clone(),
            });
        }
        for (j, release) in pkg.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{}].release[{}].dep[{}]", i, j, k),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut seen_packages = std::collections::BTreeSet::new();
    for pkg in catalog {
        if !seen_packages.insert(pkg.name.as_str()) {
            return Err(ResolveError::DuplicatePackage {
                package: pkg.name.clone(),
            });
        }
    }

    for pkg in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &pkg.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: pkg.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    for (i, lock) in locks.iter().enumerate() {
        if !valid_name(&lock.package) {
            return Err(ResolveError::InvalidName {
                site: format!("lock[{}]", i),
                name: lock.package.clone(),
            });
        }
    }
    let mut seen_locks = std::collections::BTreeSet::new();
    for lock in locks {
        if !seen_locks.insert(lock.package.as_str()) {
            return Err(ResolveError::DuplicateLock {
                package: lock.package.clone(),
            });
        }
    }

    let mut search = Search {
        catalog,
        roots,
        locks,
        max_attempts,
        attempts: 0,
        best: None,
        first_error: None,
    };
    let mut selected = Vec::new();
    match search.inspect(&mut selected) {
        Err(ResolveError::Limit { attempts }) => {
            return Err(ResolveError::Limit { attempts });
        }
        Err(e) => search.first_error = Some(e),
        Ok(()) => {}
    }
    let chosen = match search.best {
        Some((_, selected)) => selected,
        None => return Err(search.first_error.unwrap_or(ResolveError::Missing {
            package: String::new(),
        })),
    };

    let final_active = collect_constraints(roots, catalog, &chosen);
    let mut selected_out: Vec<Selection> = chosen
        .iter()
        .map(|(package, version)| Selection {
            package: package.clone(),
            version: *version,
        })
        .collect();
    selected_out.sort_by(|a, b| a.package.cmp(&b.package));

    let mut before: std::collections::BTreeMap<String, Version> =
        locks.iter().map(|l| (l.package.clone(), l.version)).collect();
    let mut after: std::collections::BTreeMap<String, Version> = chosen.into_iter().collect();
    let mut names: std::collections::BTreeSet<String> = before.keys().cloned().collect();
    names.extend(after.keys().cloned());
    let mut changes = Vec::new();

    for name in names {
        let b = before.remove(&name);
        let a = after.remove(&name);
        if b == a {
            continue;
        }
        let reason = match (b, a) {
            (None, Some(_)) => ChangeReason::Added,
            (Some(_), None) => ChangeReason::Removed,
            (Some(old), Some(_)) => {
                let exists = catalog
                    .iter()
                    .find(|p| p.name == name)
                    .map(|p| p.releases.iter().any(|r| r.version == old))
                    .unwrap_or(false);
                if !exists {
                    ChangeReason::Unavailable
                } else if final_active
                    .get(&name)
                    .map(|cs| !matches_all(old, cs))
                    .unwrap_or(false)
                {
                    ChangeReason::Rejected
                } else {
                    ChangeReason::Rebalanced
                }
            }
            (None, None) => unreachable!(),
        };
        changes.push(Change {
            package: name,
            before: b,
            after: a,
            reason,
        });
    }

    Ok(Resolution {
        selected: selected_out,
        changes,
    })
}
