fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Cleaned {
        bytes: Vec::new(),
        positions: Vec::new(),
        runs: Vec::new(),
    };
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut in_quote = false;
    let mut run_start: Option<usize> = None;

    while i < input.len() {
        if depth == 0 && !in_quote && out.bytes.is_empty() && input[i] == b'"' {
            in_quote = true;
            out.bytes.push(input[i]);
            out.positions.push(i);
            i += 1;
            continue;
        }

        if in_quote {
            let b = input[i];
            out.bytes.push(b);
            out.positions.push(i);
            i += 1;
            if b == b'\\' && i < input.len() {
                out.bytes.push(input[i]);
                out.positions.push(i);
                i += 1;
            } else if b == b'"' {
                in_quote = false;
            }
            continue;
        }

        if depth == 0 && (input[i] == b' ' || input[i] == b'\t' || input[i] == b'\r' || input[i] == b'\n' || input[i] == b'(') {
            if run_start.is_none() {
                run_start = Some(i);
            }
        } else if depth == 0 {
            if let Some(start) = run_start.take() {
                out.runs.push((start, i));
            }
        }

        if depth > 0 {
            let b = input[i];
            if b == b'\\' {
                if i + 1 >= input.len() || !(0x20..=0x7e).contains(&input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        match input[i] {
            b'(' => {
                depth = 1;
                outer_open = i;
                i += 1;
            }
            b')' => return Err(EmailError::UnexpectedClose { at: i }),
            b'\r' | b'\n' => {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            }
            b' ' | b'\t' => i += 1,
            b => {
                out.bytes.push(b);
                out.positions.push(i);
                i += 1;
            }
        }
    }

    if depth != 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = run_start {
        out.runs.push((start, input.len()));
    }
    Ok(out)
}

fn pos(cleaned: &Cleaned, i: usize, input_len: usize) -> usize {
    cleaned.positions.get(i).copied().unwrap_or(input_len)
}

fn valid_atom(local: &[u8]) -> bool {
    if local.is_empty() || local.len() > 64 {
        return false;
    }
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return false;
            }
        } else if !b.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            return false;
        }
    }
    true
}

fn parse_with_policy(input: &str, policy: &Policy, strict_single: bool) -> Result<(Address, usize), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let cleaned = cleanup(raw)?;
    for &(start, end) in &cleaned.runs {
        let left_at = cleaned.bytes.last() == Some(&b'@') && end == raw.len();
        let right_at = cleaned.bytes.first() == Some(&b'@') && start == 0;
        let boundary = start == 0 || end == raw.len();
        let adjacent = cleaned.positions.iter().any(|&p| p + 1 == start && raw[p] == b'@')
            || cleaned.positions.iter().any(|&p| p == end && raw[p] == b'@');
        let _ = (left_at, right_at);
        if !boundary && !adjacent {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }
    if cleaned.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let b = &cleaned.bytes;
    let mut quoted_end = None;
    if b.first() == Some(&b'"') {
        let mut i = 1;
        while i < b.len() {
            if b[i] == b'\\' {
                i += 2;
            } else if b[i] == b'"' {
                quoted_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(&cleaned, 0, raw.len()),
            });
        }
    }

    let protected = quoted_end.map(|n| n + 1).unwrap_or(0);
    let ats: Vec<usize> = (protected..b.len()).filter(|&i| b[i] == b'@').collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: pos(&cleaned, ats[1], raw.len()),
        });
    }
    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&cleaned, at, raw.len()),
        });
    }

    let local_bytes = &b[..at];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;
    if local_bytes[0] == b'"' {
        quoted = true;
        let end = quoted_end.unwrap();
        if end + 1 != at {
            return Err(EmailError::TrailingLocal {
                at: pos(&cleaned, end + 1, raw.len()),
            });
        }
        let mut i = 1;
        while i < end {
            let source = pos(&cleaned, i, raw.len());
            let c = local_bytes[i];
            if c == b'\\' {
                if i + 1 >= end || !matches!(local_bytes[i + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: source });
                }
                local.push(local_bytes[i + 1]);
                origins.push(source);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&c) {
                    return Err(EmailError::QuotedChar { at: source });
                }
                local.push(c);
                origins.push(source);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(&cleaned, end, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(&cleaned, 64, raw.len()),
            });
        }
        for (i, &c) in local_bytes.iter().enumerate() {
            let p = pos(&cleaned, i, raw.len());
            if c == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: p });
                }
            } else if !c.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&c) {
                return Err(EmailError::LocalChar { at: p });
            }
            local.push(c);
            origins.push(p);
        }
    }

    let domain_start = at + 1;
    let mut domain_end = b.len();
    let mut root_pos = None;
    if domain_end > domain_start && b[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_pos = Some(pos(&cleaned, domain_end, raw.len()));
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos(&cleaned, domain_start, raw.len()),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(&cleaned, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..domain_end {
        if b[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    labels.push((start, domain_end));

    for (li, &(s, e)) in labels.iter().enumerate() {
        if s == e {
            let p = if li + 1 == labels.len() {
                root_pos.unwrap_or_else(|| pos(&cleaned, s, raw.len()))
            } else {
                pos(&cleaned, s.saturating_sub(1), raw.len())
            };
            return Err(EmailError::EmptyLabel { at: p });
        }
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(&cleaned, s + 63, raw.len()),
            });
        }
        if b[s] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(&cleaned, s, raw.len()) });
        }
        if b[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(&cleaned, e - 1, raw.len()) });
        }
        for i in s..e {
            if !b[i].is_ascii_alphanumeric() && b[i] != b'-' {
                return Err(EmailError::DomainChar { at: pos(&cleaned, i, raw.len()) });
            }
        }
    }
    if labels.len() < 2 && (strict_single || !policy.allow_single_label) {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&cleaned, domain_start, raw.len()),
        });
    }
    let (ts, te) = *labels.last().unwrap();
    if b[ts..te].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(&cleaned, ts, raw.len()) });
    }

    let mut domain = String::from_utf8(b[domain_start..domain_end].to_vec()).unwrap();
    domain.make_ascii_lowercase();
    let plus_origin = local.iter().position(|&c| c == b'+').map(|i| origins[i]);

    if policy.local_case == LocalCase::Lower {
        local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(i) = local.iter().position(|&c| c == b'+') {
            if i == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: plus_origin.unwrap_or(origins[0]),
                });
            }
            local.truncate(i);
        }
    }
    if local.is_empty() {
        return Err(EmailError::EmptyNormalizedLocal {
            at: origins.first().copied().unwrap_or(pos(&cleaned, 0, raw.len())),
        });
    }
    let atom = valid_atom(&local);
    let quoted = if !atom {
        true
    } else if policy.unquote_safe {
        false
    } else {
        quoted
    };

    Ok((
        Address {
            local: String::from_utf8(local).unwrap(),
            domain: std::mem::take(&mut domain),
            quoted,
        },
        at,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default(), true).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, false).map(|x| x.0)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => result.errors.push((i, e)),
            Ok(address) => {
                if let Some((_, _, first)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    result.duplicate_of[i] = Some(*first);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), i));
                    result.addresses.push(address);
                    result.source_indices.push(i);
                }
            }
        }
    }
    result
}
