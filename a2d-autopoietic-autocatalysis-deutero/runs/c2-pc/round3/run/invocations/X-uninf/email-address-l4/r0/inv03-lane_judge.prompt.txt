struct Retained {

#[derive(Clone, Copy)]

    start: usize,

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<Retained>, Vec<CfwsRun>), EmailError> {
    let mut retained = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut in_quote = false;
    let mut quote_closed = false;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut comment_stack: Vec<usize> = Vec::new();

    while i < input.len() {
        if in_quote {
            let b = input[i];
            retained.push(Retained { byte: b, pos: i });
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    retained.push(Retained {
                        byte: input[i],
                        pos: i,
                    });
                    if input[i] == b'"' && depth == 0 {
                        // Escaped quotes do not close the prefix.
                    }
                    i += 1;
                }
                continue;
            }
            if b == b'"' {
                in_quote = false;
                quote_closed = true;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            let b = input[i];
            if b == b'\\' {
                if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                comment_stack.push(i);
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                comment_stack.pop();
                i += 1;
                if depth == 0 {
                    runs.push(CfwsRun {
                        start: outer_open,
                        end: i,
                    });
                }
                continue;
            }
            if !is_printable(b) && b != b'\t' {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        let b = input[i];

        if !quote_closed && retained.is_empty() && b == b'"' {
            in_quote = true;
            retained.push(Retained { byte: b, pos: i });
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            let run_start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];
                if is_hws(c) {
                    i += 1;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    outer_open = i;
                    depth = 1;
                    comment_stack.push(i);
                    i += 1;
                    while i < input.len() && depth > 0 {
                        let d = input[i];
                        if d == b'\\' {
                            if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                                return Err(EmailError::CommentEscape { at: i });
                            }
                            i += 2;
                        } else if d == b'\r' || d == b'\n' {
                            if let Some(end) = fold_end(input, i) {
                                i = end;
                            } else {
                                return Err(EmailError::BadFold { at: i });
                            }
                        } else if d == b'(' {
                            if depth == 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            depth += 1;
                            comment_stack.push(i);
                            i += 1;
                        } else if d == b')' {
                            depth -= 1;
                            comment_stack.pop();
                            i += 1;
                        } else if !is_printable(d) && d != b'\t' {
                            return Err(EmailError::CommentChar { at: i });
                        } else {
                            i += 1;
                        }
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer_open });
                    }
                    continue;
                }
                break;
            }
            runs.push(CfwsRun {
                start: run_start,
                end: i,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        retained.push(Retained { byte: b, pos: i });
        i += 1;
    }

    if depth != 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }

    // If that quote never closed, the address stage reports UnclosedQuote.
    let _ = quote_closed;

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn validate_atom(bytes: &[u8], positions: &[usize], fallback: usize) -> Result<(), EmailError> {
    if bytes.len() > 64 {
        return Err(EmailError::LocalTooLong {
            at: positions.get(64).copied().unwrap_or(fallback),
        });
    }
    for i in 0..bytes.len() {
        let b = bytes[i];
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return Err(EmailError::LocalDot {
                    at: positions.get(i).copied().unwrap_or(fallback),
                });
            }
        } else if !is_atom_char(b) {
            return Err(EmailError::LocalChar {
                at: positions.get(i).copied().unwrap_or(fallback),
            });
        }
    }
    Ok(())
}

fn parse_core(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let (items, runs) = cleanup(raw)?;
    let bytes: Vec<u8> = items.iter().map(|x| x.byte).collect();

    for run in &runs {
        let left = items.iter().position(|x| x.pos >= run.end);
        let right = items.iter().rposition(|x| x.pos < run.start);
        let at_left = left.is_some_and(|idx| bytes[idx] == b'@');
        let at_right = right.is_some_and(|idx| bytes[idx] == b'@');
        let at_boundary = run.start == 0 || run.end == raw.len();
        if !at_left && !at_right && !at_boundary {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if items.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quoted_end = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quoted_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: items[0].pos });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            ats.push(i);
        }
        i += 1;
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: items[ats[1]].pos,
        });
    }
    let at_idx = ats[0];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal {
            at: items[at_idx].pos,
        });
    }

    let mut local = Vec::new();
    let mut local_positions = Vec::new();
    let quoted = bytes[0] == b'"';
    if quoted {
        let end = quoted_end.expect("quoted prefix has a closing quote");
        if end + 1 != at_idx {
            return Err(EmailError::TrailingLocal {
                at: error_pos(&items, end + 1, raw.len()),
            });
        }
        let mut j = 1;
        while j < end {
            if bytes[j] == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: items[j].pos,
                    });
                }
                let c = bytes[j + 1];
                if c != b'"' && c != b'\\' && c != b'@' {
                    return Err(EmailError::BadEscape {
                        at: items[j].pos,
                    });
                }
                local.push(c);
                local_positions.push(items[j].pos);
                j += 2;
            } else {
                if !is_printable(bytes[j]) {
                    return Err(EmailError::QuotedChar {
                        at: items[j].pos,
                    });
                }
                local.push(bytes[j]);
                local_positions.push(items[j].pos);
                j += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: items[end].pos,
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_positions[64],
            });
        }
    } else {
        local.extend_from_slice(&bytes[..at_idx]);
        local_positions.extend(items[..at_idx].iter().map(|x| x.pos));
        validate_atom(&local, &local_positions, raw.len())?;
    }

    let domain_start_idx = at_idx + 1;
    let domain_start_pos = error_pos(&items, domain_start_idx, raw.len());
    let mut domain_end_idx = bytes.len();
    let had_root_dot = domain_end_idx > domain_start_idx && bytes[domain_end_idx - 1] == b'.';
    if had_root_dot {
        domain_end_idx -= 1;
    }
    if domain_start_idx == domain_end_idx {
        return Err(EmailError::EmptyDomain {
            at: if had_root_dot {
                items[domain_end_idx].pos
            } else {
                domain_start_pos
            },
        });
    }
    let domain_len = domain_end_idx - domain_start_idx;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: items[domain_start_idx + 253].pos,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start_idx;
    for idx in domain_start_idx..domain_end_idx {
        if bytes[idx] == b'.' {
            labels.push((start, idx));
            start = idx + 1;
        }
    }
    labels.push((start, domain_end_idx));

    for &(s, e) in &labels {
        if s == e {
            let pos = if s == domain_start_idx {
                items[s].pos
            } else if e == domain_end_idx && had_root_dot {
                items[domain_end_idx].pos
            } else {
                items[s - 1].pos
            };
            return Err(EmailError::EmptyLabel { at: pos });
        }
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: items[s + 63].pos,
            });
        }
        if bytes[s] == b'-' {
            return Err(EmailError::LabelHyphen { at: items[s].pos });
        }
        if bytes[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: items[e - 1].pos,
            });
        }
        for idx in s..e {
            if !bytes[idx].is_ascii_alphanumeric() && bytes[idx] != b'-' {
                return Err(EmailError::DomainChar { at: items[idx].pos });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: items[domain_start_idx].pos,
        });
    }
    let (last_start, last_end) = *labels.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: items[last_start].pos,
        });
    }

    let mut domain = String::from_utf8(bytes[domain_start_idx..domain_end_idx].to_vec()).unwrap();
    domain.make_ascii_lowercase();
    let local_string = String::from_utf8(local).unwrap();
    Ok((
        Address {
            local: local_string,
            domain,
            quoted,
        },
        local_positions,
    ))
}

fn is_valid_dot_atom(local: &str) -> bool {
    let bytes = local.as_bytes();
    if bytes.is_empty() || bytes.len() > 64 {
        return false;
    }
    for i in 0..bytes.len() {
        if bytes[i] == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return false;
            }
        } else if !is_atom_char(bytes[i]) {
            return false;
        }
    }
    true
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, positions) = parse_core(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(i) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if i == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: positions.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(i);
        }
    }

    let safe = is_valid_dot_atom(&address.local);
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    use std::collections::HashMap;

    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut first_by_key: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_input) = first_by_key.get(&key) {
                    result.duplicate_of[index] = Some(first_input);
                } else {
                    first_by_key.insert(key, index);
                    result.source_indices.push(index);
                    result.addresses.push(address);
                }
            }
        }
    }

    result
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Cleaned {
        bytes: Vec::new(),
        positions: Vec::new(),
        runs: Vec::new(),
    };
    let mut i = 0;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut in_quote = false;
    let mut run_start: Option<usize> = None;

    while i < input.len() {
        if depth == 0 && !in_quote && out.bytes.is_empty() && input[i] == b'"' {
            in_quote = true;
            out.bytes.push(input[i]);
            out.positions.push(i);
            i += 1;
            continue;
        }

        if in_quote {
            let b = input[i];
            out.bytes.push(b);
            out.positions.push(i);
            i += 1;
            if b == b'\\' && i < input.len() {
                out.bytes.push(input[i]);
                out.positions.push(i);
                i += 1;
            } else if b == b'"' {
                in_quote = false;
            }
            continue;
        }

        if depth == 0 && (input[i] == b' ' || input[i] == b'\t' || input[i] == b'\r' || input[i] == b'\n' || input[i] == b'(') {
            if run_start.is_none() {
                run_start = Some(i);
            }
        } else if depth == 0 {
            if let Some(start) = run_start.take() {
                out.runs.push((start, i));
            }
        }

        if depth > 0 {
            let b = input[i];
            if b == b'\\' {
                if i + 1 >= input.len() || !(0x20..=0x7e).contains(&input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                return Err(EmailError::CommentChar { at: i });
            }
            i += 1;
            continue;
        }

        match input[i] {
            b'(' => {
                depth = 1;
                outer_open = i;
                i += 1;
            }
            b')' => return Err(EmailError::UnexpectedClose { at: i }),
            b'\r' | b'\n' => {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                } else {
                    return Err(EmailError::BadFold { at: i });
                }
            }
            b' ' | b'\t' => i += 1,
            b => {
                out.bytes.push(b);
                out.positions.push(i);
                i += 1;
            }
        }
    }

    if depth != 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = run_start {
        out.runs.push((start, input.len()));
    }
    Ok(out)
}

fn pos(cleaned: &Cleaned, i: usize, input_len: usize) -> usize {
    cleaned.positions.get(i).copied().unwrap_or(input_len)
}

fn valid_atom(local: &[u8]) -> bool {
    if local.is_empty() || local.len() > 64 {
        return false;
    }
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == local.len() || local[i - 1] == b'.' {
                return false;
            }
        } else if !b.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
        {
            return false;
        }
    }
    true
}

fn parse_with_policy(input: &str, policy: &Policy, strict_single: bool) -> Result<(Address, usize), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let cleaned = cleanup(raw)?;
    for &(start, end) in &cleaned.runs {
        let left_at = cleaned.bytes.last() == Some(&b'@') && end == raw.len();
        let right_at = cleaned.bytes.first() == Some(&b'@') && start == 0;
        let boundary = start == 0 || end == raw.len();
        let adjacent = cleaned.positions.iter().any(|&p| p + 1 == start && raw[p] == b'@')
            || cleaned.positions.iter().any(|&p| p == end && raw[p] == b'@');
        let _ = (left_at, right_at);
        if !boundary && !adjacent {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }
    if cleaned.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let b = &cleaned.bytes;
    let mut quoted_end = None;
    if b.first() == Some(&b'"') {
        let mut i = 1;
        while i < b.len() {
            if b[i] == b'\\' {
                i += 2;
            } else if b[i] == b'"' {
                quoted_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(&cleaned, 0, raw.len()),
            });
        }
    }

    let protected = quoted_end.map(|n| n + 1).unwrap_or(0);
    let ats: Vec<usize> = (protected..b.len()).filter(|&i| b[i] == b'@').collect();
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: pos(&cleaned, ats[1], raw.len()),
        });
    }
    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&cleaned, at, raw.len()),
        });
    }

    let local_bytes = &b[..at];
    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;
    if local_bytes[0] == b'"' {
        quoted = true;
        let end = quoted_end.unwrap();
        if end + 1 != at {
            return Err(EmailError::TrailingLocal {
                at: pos(&cleaned, end + 1, raw.len()),
            });
        }
        let mut i = 1;
        while i < end {
            let source = pos(&cleaned, i, raw.len());
            let c = local_bytes[i];
            if c == b'\\' {
                if i + 1 >= end || !matches!(local_bytes[i + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: source });
                }
                local.push(local_bytes[i + 1]);
                origins.push(source);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&c) {
                    return Err(EmailError::QuotedChar { at: source });
                }
                local.push(c);
                origins.push(source);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(&cleaned, end, raw.len()),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(&cleaned, 64, raw.len()),
            });
        }
        for (i, &c) in local_bytes.iter().enumerate() {
            let p = pos(&cleaned, i, raw.len());
            if c == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: p });
                }
            } else if !c.is_ascii_alphanumeric() && !b"!#$%&'*+-/=?^_`{|}~".contains(&c) {
                return Err(EmailError::LocalChar { at: p });
            }
            local.push(c);
            origins.push(p);
        }
    }

    let domain_start = at + 1;
    let mut domain_end = b.len();
    let mut root_pos = None;
    if domain_end > domain_start && b[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_pos = Some(pos(&cleaned, domain_end, raw.len()));
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos(&cleaned, domain_start, raw.len()),
        });
    }
    let domain_len = domain_end - domain_start;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(&cleaned, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..domain_end {
        if b[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }
    labels.push((start, domain_end));

    for (li, &(s, e)) in labels.iter().enumerate() {
        if s == e {
            let p = if li + 1 == labels.len() {
                root_pos.unwrap_or_else(|| pos(&cleaned, s, raw.len()))
            } else {
                pos(&cleaned, s.saturating_sub(1), raw.len())
            };
            return Err(EmailError::EmptyLabel { at: p });
        }
        if e - s > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(&cleaned, s + 63, raw.len()),
            });
        }
        if b[s] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(&cleaned, s, raw.len()) });
        }
        if b[e - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: pos(&cleaned, e - 1, raw.len()) });
        }
        for i in s..e {
            if !b[i].is_ascii_alphanumeric() && b[i] != b'-' {
                return Err(EmailError::DomainChar { at: pos(&cleaned, i, raw.len()) });
            }
        }
    }
    if labels.len() < 2 && (strict_single || !policy.allow_single_label) {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&cleaned, domain_start, raw.len()),
        });
    }
    let (ts, te) = *labels.last().unwrap();
    if b[ts..te].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: pos(&cleaned, ts, raw.len()) });
    }

    let mut domain = String::from_utf8(b[domain_start..domain_end].to_vec()).unwrap();
    domain.make_ascii_lowercase();
    let plus_origin = local.iter().position(|&c| c == b'+').map(|i| origins[i]);

    if policy.local_case == LocalCase::Lower {
        local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(i) = local.iter().position(|&c| c == b'+') {
            if i == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: plus_origin.unwrap_or(origins[0]),
                });
            }
            local.truncate(i);
        }
    }
    if local.is_empty() {
        return Err(EmailError::EmptyNormalizedLocal {
            at: origins.first().copied().unwrap_or(pos(&cleaned, 0, raw.len())),
        });
    }
    let atom = valid_atom(&local);
    let quoted = if !atom {
        true
    } else if policy.unquote_safe {
        false
    } else {
        quoted
    };

    Ok((
        Address {
            local: String::from_utf8(local).unwrap(),
            domain: std::mem::take(&mut domain),
            quoted,
        },
        at,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_with_policy(input, &Policy::default(), true).map(|x| x.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_with_policy(input, policy, false).map(|x| x.0)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (i, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(e) => result.errors.push((i, e)),
            Ok(address) => {
                if let Some((_, _, first)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    result.duplicate_of[i] = Some(*first);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), i));
                    result.addresses.push(address);
                    result.source_indices.push(i);
                }
            }
        }
    }
    result
}
