        if byte.is_ascii_alphanumeric() || matches!(*byte, b'.' | b'_' | b'-') {

fn matches_constraint(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(x) => version == x,
        Constraint::AtLeast(x) => version >= x,
        Constraint::Before(x) => version < x,
        Constraint::Caret(x) => {
            version >= x
                && if x.major > 0 {
                    version.major == x.major
                } else if x.minor > 0 {
                    version.major == 0 && version.minor == x.minor
                } else {
                    version == x
                }
        }
    }
}

fn matches_all(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|c| matches_constraint(version, *c))
}

    requirements: &std::collections::BTreeMap<String, Vec<Constraint>>,

    requirements
        .iter()
        .map(|(name, constraints)| (name.clone(), canonical_constraints(constraints)))
        .collect()

fn resolve_node(
    catalog: &[Package],
    by_name: &std::collections::BTreeMap<String, usize>,
    roots: &[Requirement],
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<(), ResolveError> {
    let mut active: std::collections::BTreeMap<String, Vec<Constraint>> =
        std::collections::BTreeMap::new();

    for root in roots {
        active
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    for (name, version) in selected.iter() {
        if let Some(&index) = by_name.get(name) {
            if let Some(release) = catalog[index]
                .releases
                .iter()
                .find(|release| release.version == *version)
            {
                for dep in &release.deps {
                    active
                        .entry(dep.package.clone())
                        .or_default()
                        .push(dep.constraint);
                }
            }
        }
    }

    let active = canonical_map(&active);

    if let Some(name) = active.keys().find(|name| !by_name.contains_key(*name)) {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, version) in selected.iter() {
        if let Some(constraints) = active.get(name) {
            if !matches_all(*version, constraints) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in &active {
        if selected.contains_key(name) {
            continue;
        }
        let package = &catalog[by_name[name]];
        if !package
            .releases
            .iter()
            .any(|release| matches_all(release.version, constraints))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    let Some((name, constraints)) = active
        .iter()
        .find(|(name, _)| !selected.contains_key(*name))
    else {
        return Ok(());
    };

    let package = &catalog[by_name[name]];
    let version = package
        .releases
        .iter()
        .filter(|release| matches_all(release.version, constraints))
        .map(|release| release.version)
        .max()
        .expect("candidate existence checked above");

    selected.insert(name.clone(), version);
    resolve_node(catalog, by_name, roots, selected)
}

pub fn resolve(catalog: &[Package], roots: &[Requirement]) -> Result<Vec<Selection>, ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }

    let mut by_name = std::collections::BTreeMap::new();
    for (i, package) in catalog.iter().enumerate() {
        if by_name.insert(package.name.clone(), i).is_some() {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut selected = std::collections::BTreeMap::new();
    resolve_node(catalog, &by_name, roots, &mut selected)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}

    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
    }
}

    }

    let mut selected: std::collections::BTreeMap<String, Version> =
        std::collections::BTreeMap::new();

    loop {
        if let Some(package) = active
            .keys()
            .find(|name| !catalog_map.contains_key(name.as_str()))
        {
            return Err(ResolveError::Missing {
                package: package.clone(),
            });
        }

        for (package, version) in &selected {
            let constraints = canonical(active.get(package).cloned().unwrap_or_default());
            if constraints.iter().any(|c| !matches(*version, *c)) {
                return Err(ResolveError::Conflict {
                    package: package.clone(),
                    selected: *version,
                    requirements: constraints,
                });
            }
        }

        let mut next_package: Option<String> = None;
        for (package, constraints) in &active {
            if selected.contains_key(package) {
                continue;
            }
            let constraints = canonical(constraints.clone());
            let releases = &catalog_map[package.as_str()].releases;
            let has_match = releases.iter().any(|release| {
                constraints
                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            });
            if !has_match {
                return Err(ResolveError::NoMatch {
                    package: package.clone(),
                    requirements: constraints,
                });
            }
            if next_package.is_none() {
                next_package = Some(package.clone());
            }
        }

        let Some(package_name) = next_package else {
            break;
        };

        let constraints = canonical(active[&package_name].clone());
        let package = catalog_map[package_name.as_str()];
        let release = package
            .releases
            .iter()
            .filter(|release| {
                constraints
                    .iter()
                    .all(|constraint| matches(release.version, *constraint))
            })
            .max_by_key(|release| release.version)
            .expect("candidate existence checked above");

        selected.insert(package_name, release.version);
        for dep in &release.deps {
            active
                .entry(dep.package.clone())
                .or_default()
                .push(dep.constraint);
        }
    }

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if !(1..=24).contains(&bytes.len())
        || !bytes[0].is_ascii_lowercase()
        || bytes[bytes.len() - 1] == b'-'
    {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_input(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, req) in roots.iter().enumerate() {
        if !valid_name(&req.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: req.package.clone(),
            });
        }
    }

    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn canonicalize(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            if version < v {
                false
            } else if v.major > 0 {
                version.major == v.major
            } else if v.minor > 0 {
                version.major == 0 && version.minor == v.minor
            } else {
                version == v
            }
        }
    }
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut canonical = constraints.to_vec();
    canonicalize(&mut canonical);
    canonical
        .iter()
        .map(|c| constraint_text(*c))
        .collect::<Vec<_>>()
        .join(",")
}

fn encode_text(text: &str) -> String {
    let mut encoded = String::new();
    for byte in text.as_bytes() {
        if byte.is_ascii_alphabetic()
            || byte.is_ascii_digit()
            || matches!(*byte, b'.' | b'_' | b'-')
        {
            encoded.push(*byte as char);
        } else {
            encoded.push('%');
            encoded.push(char::from(b"0123456789ABCDEF"[(byte >> 4) as usize]));
            encoded.push(char::from(b"0123456789ABCDEF"[(byte & 0x0f) as usize]));
        }
    }
    encoded
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            encode_text(site),
            encode_text(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", encode_text(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            encode_text(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", encode_text(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            encode_text(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            encode_text(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
    }
}

fn search(
    catalog: &[Package],
    constraints: &mut std::collections::BTreeMap<String, Vec<Constraint>>,
    selected: &mut std::collections::BTreeMap<String, Version>,
) -> Result<(), ResolveError> {
    for package in selected.keys() {
        constraints.entry(package.clone()).or_default();
    }

    if let Some(package) = constraints
        .keys()
        .find(|name| !catalog.iter().any(|p| p.name == ***name))
        .cloned()
    {
        return Err(ResolveError::Missing { package });
    }

    for (package, version) in selected.iter() {
        let mut requirements = constraints.get(package).cloned().unwrap_or_default();
        canonicalize(&mut requirements);
        if requirements.iter().any(|c| !matches(*version, *c)) {
            return Err(ResolveError::Conflict {
                package: package.clone(),
                selected: *version,
                requirements,
            });
        }
    }

    for (package, requirements) in constraints.iter() {
        if selected.contains_key(package) {
            continue;
        }
        let mut canonical = requirements.clone();
        canonicalize(&mut canonical);
        let catalog_package = catalog.iter().find(|p| p.name == *package).unwrap();
        if !catalog_package.releases.iter().any(|release| {
            canonical
                .iter()
                .all(|constraint| matches(release.version, *constraint))
        }) {
            return Err(ResolveError::NoMatch {
                package: package.clone(),
                requirements: canonical,
            });
        }
    }

    let next_package = constraints
        .keys()
        .find(|package| !selected.contains_key(*package))
        .cloned();

    let Some(package) = next_package else {
        return Ok(());
    };

    let mut requirements = constraints.get(&package).cloned().unwrap_or_default();
    canonicalize(&mut requirements);

    let catalog_package = catalog.iter().find(|p| p.name == package).unwrap();
    let mut candidates: Vec<&Release> = catalog_package
        .releases
        .iter()
        .filter(|release| {
            requirements
                .iter()
                .all(|constraint| matches(release.version, *constraint))
        })
        .collect();
    candidates.sort_by(|a, b| b.version.cmp(&a.version));
    let release = candidates[0];

    selected.insert(package.clone(), release.version);
    for dep in &release.deps {
        constraints
            .entry(dep.package.clone())
            .or_default()
            .push(dep.constraint);
    }

    search(catalog, constraints, selected)
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
) -> Result<Vec<Selection>, ResolveError> {
    validate_input(catalog, roots)?;

    let mut package_names = std::collections::BTreeSet::new();
    for package in catalog {
        if !package_names.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut constraints = std::collections::BTreeMap::<String, Vec<Constraint>>::new();
    for root in roots {
        constraints
            .entry(root.package.clone())
            .or_default()
            .push(root.constraint);
    }

    let mut selected = std::collections::BTreeMap::<String, Version>::new();
    search(catalog, &mut constraints, &mut selected)?;

    Ok(selected
        .into_iter()
        .map(|(package, version)| Selection { package, version })
        .collect())
}
