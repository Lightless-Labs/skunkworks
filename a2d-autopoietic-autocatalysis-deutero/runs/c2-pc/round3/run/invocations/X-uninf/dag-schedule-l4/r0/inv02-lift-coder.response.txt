#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

#[derive(Clone)]
struct Running {
    task: usize,
    worker: usize,
    finish: u64,
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Validate resource shape and demand in task order.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &cap)) in task.needs.iter().zip(capacity).enumerate() {
            if need > cap {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: cap,
                });
            }
        }
    }

    // Build graph edges from predecessors to children.
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest available ID first.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let mut smallest_pos = 0;
        for pos in 1..available.len() {
            if available[pos] < available[smallest_pos] {
                smallest_pos = pos;
            }
        }
        let id = available.swap_remove(smallest_pos);
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_successor_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = String::new();
    trace.push_str(&format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    ));

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut used = vec![0u64; capacity.len()];
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut completed_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete every task finishing now, in ascending task ID order.
        let mut ending: Vec<Running> = running
            .iter()
            .filter(|item| item.finish == now)
            .cloned()
            .collect();
        ending.sort_by_key(|item| item.task);

        if !ending.is_empty() {
            running.retain(|item| item.finish != now);
            for item in ending {
                let task_id = item.task;
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                for resource in 0..capacity.len() {
                    used[resource] -= tasks[task_id].needs[resource] as u64;
                }
                append_event(
                    &mut trace,
                    now,
                    "end",
                    task_id,
                    item.worker,
                    &used,
                );
            }
        }

        // Keep admitting tasks at this time until no worker/task pair fits.
        loop {
            let mut best: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id] || tasks[task_id].release > now {
                    continue;
                }
                if !tasks[task_id].deps.iter().all(|&dep| completed[dep]) {
                    continue;
                }
                if !fits(&tasks[task_id].needs, &used, capacity) {
                    continue;
                }

                let better = match best {
                    None => true,
                    Some(other) => {
                        tails[task_id] > tails[other]
                            || (tails[task_id] == tails[other]
                                && (tasks[task_id].priority > tasks[other].priority
                                    || (tasks[task_id].priority
                                        == tasks[other].priority
                                        && task_id < other)))
                    }
                };
                if better {
                    best = Some(task_id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= now);

            let Some(worker) = worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let next_worker_time = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = next_worker_time;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            for resource in 0..capacity.len() {
                used[resource] += tasks[task_id].needs[resource] as u64;
            }
            append_event(&mut trace, now, "start", task_id, worker, &used);

            if tasks[task_id].duration == 0 {
                // Zero-duration completion is immediate, before another start.
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                for resource in 0..capacity.len() {
                    used[resource] -= tasks[task_id].needs[resource] as u64;
                }
                append_event(&mut trace, now, "end", task_id, worker, &used);
            } else {
                running.push(Running {
                    task: task_id,
                    worker,
                    finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // Advance to the earliest event that can change eligibility or capacity.
        let mut next_time: Option<u64> = None;
        for item in &running {
            if item.finish > now {
                next_time = Some(next_time.map_or(item.finish, |value| value.min(item.finish)));
            }
        }
        for (task_id, task) in tasks.iter().enumerate() {
            if !started[task_id] && task.release > now {
                next_time = Some(next_time.map_or(task.release, |value| value.min(task.release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next_time = Some(next_time.map_or(available_at, |value| value.min(available_at)));
            }
        }

        // If no event can advance time, a valid DAG would be unable to progress.
        // This cannot occur once cycle detection has succeeded.
        if let Some(next) = next_time {
            now = next;
        } else {
            break;
        }
    }

    for run in &mut runs {
        if run.is_none() {
            // Unreachable for a validated acyclic graph; avoids panicking if
            // internal state is ever changed.
            return Err(Error::Cycle { blocked: Vec::new() });
        }
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();
    trace.push_str(&format!("makespan={}\n", makespan));

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}

fn fits(needs: &[u32], used: &[u64], capacity: &[u32]) -> bool {
    needs.iter().enumerate().all(|(resource, &need)| {
        (need as u64) <= (capacity[resource] as u64 - used[resource])
    })
}

fn append_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    trace.push_str(&format!("{}|{}|{}|{}|", time, kind, task, worker));
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, value) in used.iter().enumerate() {
            if index > 0 {
                trace.push(',');
            }
            trace.push_str(&value.to_string());
        }
    }
    trace.push('\n');
}
