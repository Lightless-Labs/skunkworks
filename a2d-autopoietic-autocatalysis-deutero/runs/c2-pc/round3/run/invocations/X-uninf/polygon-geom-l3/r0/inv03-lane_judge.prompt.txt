const COORD_LIMIT: i64 = 1_000_000_000;
const MAX_ITEMS: usize = 4096;

fn valid_point(p: Point) -> bool {
    (-COORD_LIMIT..=COORD_LIMIT).contains(&p.x)
        && (-COORD_LIMIT..=COORD_LIMIT).contains(&p.y)
}

fn prepare_polygon(points: &[Point]) -> Result<Vec<Point>, GeomError> {
    if points.len() > MAX_ITEMS {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !valid_point(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut effective = points.to_vec();
    if effective.len() >= 2 && effective.last() == effective.first() {
        effective.pop();
    }
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, q: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (q.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (q.x as i128 - a.x as i128)
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

fn rounded_sqrt_million(d2: u128) -> u64 {
    let target = d2 * 1_000_000;
    let mut lo = 0u128;
    let mut hi = 2_000_000_001u128;
    while lo < hi {
        let mid = lo + (hi - lo + 1) / 2;
        if mid * mid <= target {
            lo = mid;
        } else {
            hi = mid - 1;
        }
    }
    let lower = lo;
    let threshold = lower * lower + lower;
    (if target >= threshold { lower + 1 } else { lower }) as u64
}

fn edge_length_milli(a: Point, b: Point) -> u64 {
    let dx = (a.x as i128 - b.x as i128) as i64 as i128;
    let dy = (a.y as i128 - b.y as i128) as i64 as i128;
    let d2 = (dx * dx + dy * dy) as u128;
    rounded_sqrt_million(d2)
}

fn perimeter_milli(vertices: &[Point]) -> u64 {
    if vertices.len() < 2 {
        return 0;
    }
    (0..vertices.len())
        .map(|i| edge_length_milli(vertices[i], vertices[(i + 1) % vertices.len()]))
        .sum()
}

fn signed_area2(vertices: &[Point]) -> i128 {
    (0..vertices.len())
        .map(|i| {
            let a = vertices[i];
            let b = vertices[(i + 1) % vertices.len()];
            a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128
        })
        .sum()
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let vertices = prepare_polygon(points)?;
    let signed = signed_area2(&vertices);
    let edge_milli: Vec<u64> = (0..vertices.len())
        .map(|i| edge_length_milli(vertices[i], vertices[(i + 1) % vertices.len()]))
        .collect();
    let perimeter_milli = edge_milli.iter().copied().sum();

    let mut longest_edge = None;
    for i in 0..vertices.len() {
        let replace = match longest_edge {
            None => true,
            Some(j) if edge_milli[i] > edge_milli[j] => true,
            Some(j) if edge_milli[i] < edge_milli[j] => false,
            Some(j) => {
                let pair_i = {
                    let (a, b) = (vertices[i], vertices[(i + 1) % vertices.len()]);
                    if a <= b { (a, b) } else { (b, a) }
                };
                let pair_j = {
                    let (a, b) = (vertices[j], vertices[(j + 1) % vertices.len()]);
                    if a <= b { (a, b) } else { (b, a) }
                };
                pair_i < pair_j || (pair_i == pair_j && i < j)
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    Ok(Measures {
        signed_area2: signed,
        area2: signed.unsigned_abs(),
        orientation: if signed > 0 {
            Orientation::CounterClockwise
        } else if signed < 0 {
            Orientation::Clockwise
        } else {
            Orientation::Degenerate
        },
        edge_milli,
        perimeter_milli,
        longest_edge,
    })
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_ITEMS {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !valid_point(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..polygon.len() {
            if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_ITEMS {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !valid_point(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut sorted: Vec<(Point, usize)> = Vec::new();
    for (i, &p) in points.iter().enumerate() {
        if let Some(entry) = sorted.iter_mut().find(|entry| entry.0 == p) {
            entry.1 = entry.1.min(i);
        } else {
            sorted.push((p, i));
        }
    }
    sorted.sort_by_key(|entry| entry.0);

    if sorted.len() <= 1 {
        let vertices: Vec<Point> = sorted.iter().map(|entry| entry.0).collect();
        let source_indices: Vec<usize> = sorted.iter().map(|entry| entry.1).collect();
        return Ok(Hull {
            vertices,
            source_indices,
            area2: 0,
            perimeter_milli: 0,
        });
    }

    let all_collinear = sorted
        .iter()
        .skip(2)
        .all(|entry| cross(sorted[0].0, sorted[1].0, entry.0) == 0);

    let hull_entries: Vec<(Point, usize)> = if all_collinear {
        match mode {
            Collinear::Include => sorted.clone(),
            Collinear::Exclude => vec![sorted[0], *sorted.last().unwrap()],
        }
    } else {
        let mut lower: Vec<(Point, usize)> = Vec::new();
        for &entry in &sorted {
            while lower.len() >= 2 {
                let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, entry.0);
                if if mode == Collinear::Include { turn < 0 } else { turn <= 0 } {
                    lower.pop();
                } else {
                    break;
                }
            }
            lower.push(entry);
        }

        let mut upper: Vec<(Point, usize)> = Vec::new();
        for &entry in sorted.iter().rev() {
            while upper.len() >= 2 {
                let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, entry.0);
                if if mode == Collinear::Include { turn < 0 } else { turn <= 0 } {
                    upper.pop();
                } else {
                    break;
                }
            }
            upper.push(entry);
        }
        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = hull_entries.iter().map(|entry| entry.0).collect();
    let source_indices: Vec<usize> = hull_entries.iter().map(|entry| entry.1).collect();
    let signed = if vertices.len() >= 3 {
        signed_area2(&vertices)
    } else {
        0
    };
    Ok(Hull {
        area2: signed.unsigned_abs(),
        perimeter_milli: perimeter_milli(&vertices),
        vertices,
        source_indices,
    })
}

const MAX_COORD: i64 = 1_000_000_000;
const MAX_VERTICES: usize = 4096;

fn in_range(p: Point) -> bool {
    (-MAX_COORD..=MAX_COORD).contains(&p.x) && (-MAX_COORD..=MAX_COORD).contains(&p.y)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn shoelace(points: &[Point]) -> i128 {
    if points.is_empty() {
        return 0;
    }
    let mut sum = 0i128;
    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
    }
    sum
}

fn rounded_edge_milli(a: Point, b: Point) -> u64 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    let n = ((dx * dx + dy * dy) as u128) * 1_000_000u128;

    let mut lo = 0u128;
    let mut hi = 3_000_000_001u128;
    while lo < hi {
        let mid = lo + (hi - lo) / 2;
        if mid * mid < n {
            lo = mid + 1;
        } else {
            hi = mid;
        }
    }
    let ceil = lo;
    let floor = if ceil * ceil == n { ceil } else { ceil - 1 };
    if ceil * ceil == n {
        return ceil as u64;
    }

    // Compare sqrt(n) to the half integer floor + 1/2 exactly:
    // 4*n versus (2*floor+1)^2.
    let twice_mid = 2 * floor + 1;
    if 4 * n >= twice_mid * twice_mid {
        ceil as u64
    } else {
        floor as u64
    }
}

fn perimeter_milli(points: &[Point]) -> u64 {
    if points.is_empty() {
        return 0;
    }
    let mut total = 0u64;
    for i in 0..points.len() {
        total += rounded_edge_milli(points[i], points[(i + 1) % points.len()]);
    }
    total
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    let points = prepare_polygon(points)?;
    let signed_area2 = shoelace(points);
    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter = 0u64;
    let mut longest_edge = None;
    let mut longest_len = 0u64;
    let mut longest_pair: Option<(Point, Point)> = None;

    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        let len = rounded_edge_milli(a, b);
        edge_milli.push(len);
        perimeter += len;

        let (low, high) = if a <= b { (a, b) } else { (b, a) };
        let pair = (low, high);
        if longest_edge.is_none()
            || len > longest_len
            || (len == longest_len
                && (pair < longest_pair.unwrap()
                    || (pair == longest_pair.unwrap() && i < longest_edge.unwrap())))
        {
            longest_edge = Some(i);
            longest_len = len;
            longest_pair = Some(pair);
        }
    }

    Ok(Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli: perimeter,
        longest_edge,
    })
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare_polygon(points)?;
    if queries.len() > MAX_VERTICES {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    let mut results = Vec::with_capacity(queries.len());
    for &q in queries {
        let mut boundary = None;
        for i in 0..polygon.len() {
            if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], q) {
                boundary = Some(i);
                break;
            }
        }
        if let Some(edge) = boundary {
            results.push(Location::Boundary { edge });
            continue;
        }

        let mut inside = false;
        for i in 0..polygon.len() {
            let a = polygon[i];
            let b = polygon[(i + 1) % polygon.len()];
            if (a.y > q.y) != (b.y > q.y) {
                let c = cross(a, b, q);
                if (b.y > a.y && c > 0) || (b.y < a.y && c < 0) {
                    inside = !inside;
                }
            }
        }
        results.push(if inside {
            Location::Inside
        } else {
            Location::Outside
        });
    }
    Ok(results)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_VERTICES {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|&(p, _)| p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    if unique.len() <= 1 {
        let vertices: Vec<Point> = unique.iter().map(|&(p, _)| p).collect();
        let source_indices: Vec<usize> = unique.iter().map(|&(_, i)| i).collect();
        return Ok(Hull {
            perimeter_milli: perimeter_milli(&vertices),
            vertices,
            source_indices,
            area2: 0,
        });
    }

    let all_collinear = unique
        .iter()
        .skip(2)
        .all(|&(p, _)| cross(unique[0].0, unique[1].0, p) == 0);

    let selected: Vec<(Point, usize)> = if all_collinear {
        match mode {
            Collinear::Include => unique,
            Collinear::Exclude => vec![unique[0], *unique.last().unwrap()],
        }
    } else {
        let mut lower: Vec<(Point, usize)> = Vec::new();
        for &point in &unique {
            while lower.len() >= 2 {
                let c = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, point.0);
                let remove = match mode {
                    Collinear::Exclude => c <= 0,
                    Collinear::Include => c < 0,
                };
                if remove {
                    lower.pop();
                } else {
                    break;
                }
            }
            lower.push(point);
        }

        let mut upper: Vec<(Point, usize)> = Vec::new();
        for &point in unique.iter().rev() {
            while upper.len() >= 2 {
                let c = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, point.0);
                let remove = match mode {
                    Collinear::Exclude => c <= 0,
                    Collinear::Include => c < 0,
                };
                if remove {
                    upper.pop();
                } else {
                    break;
                }
            }
            upper.push(point);
        }

        lower.pop();
        upper.pop();
        lower.extend(upper);
        lower
    };

    let vertices: Vec<Point> = selected.iter().map(|&(p, _)| p).collect();
    let source_indices: Vec<usize> = selected.iter().map(|&(_, i)| i).collect();
    let area2 = shoelace(&vertices).unsigned_abs();
    let perimeter_milli = perimeter_milli(&vertices);

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}
