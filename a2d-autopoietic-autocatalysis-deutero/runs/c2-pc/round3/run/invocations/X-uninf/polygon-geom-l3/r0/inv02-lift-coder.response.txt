#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Orientation {
    Clockwise,
    Degenerate,
    CounterClockwise,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Measures {
    pub signed_area2: i128,
    pub area2: u128,
    pub orientation: Orientation,
    pub edge_milli: Vec<u64>,
    pub perimeter_milli: u64,
    pub longest_edge: Option<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum GeomError {
    TooManyVertices { len: usize },
    OutOfRange { index: usize },
    TooFewVertices { len: usize },
    TooManyQueries { len: usize },
    QueryOutOfRange { index: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Location {
    Outside,
    Inside,
    Boundary { edge: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Collinear {
    Exclude,
    Include,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Hull {
    pub vertices: Vec<Point>,
    pub source_indices: Vec<usize>,
    pub area2: u128,
    pub perimeter_milli: u64,
}

const LIMIT: i64 = 1_000_000_000;
const MAX_INPUT: usize = 4096;

fn in_range(p: Point) -> bool {
    (-LIMIT..=LIMIT).contains(&p.x) && (-LIMIT..=LIMIT).contains(&p.y)
}

fn prepare(points: &[Point]) -> Result<&[Point], GeomError> {
    if points.len() > MAX_INPUT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }
    let effective = if points.len() >= 2 && points.first() == points.last() {
        &points[..points.len() - 1]
    } else {
        points
    };
    if effective.len() < 3 {
        return Err(GeomError::TooFewVertices {
            len: effective.len(),
        });
    }
    Ok(effective)
}

fn cross(a: Point, b: Point, c: Point) -> i128 {
    (b.x as i128 - a.x as i128) * (c.y as i128 - a.y as i128)
        - (b.y as i128 - a.y as i128) * (c.x as i128 - a.x as i128)
}

fn squared_distance(a: Point, b: Point) -> u128 {
    let dx = b.x as i128 - a.x as i128;
    let dy = b.y as i128 - a.y as i128;
    (dx * dx + dy * dy) as u128
}

fn rounded_sqrt(value: u128) -> u64 {
    let mut lo = 0u128;
    let mut hi = 1u128 << 64;
    while lo + 1 < hi {
        let mid = (lo + hi) / 2;
        if mid <= value / mid {
            lo = mid;
        } else {
            hi = mid;
        }
    }
    let remainder = value - lo * lo;
    if remainder > lo {
        (lo + 1) as u64
    } else {
        lo as u64
    }
}

fn edge_mill(a: Point, b: Point) -> u64 {
    rounded_sqrt(squared_distance(a, b) * 1_000_000)
}

fn endpoint_pair(a: Point, b: Point) -> (Point, Point) {
    if a <= b { (a, b) } else { (b, a) }
}

fn measures_for(points: &[Point]) -> Measures {
    let mut signed_area2 = 0i128;
    let mut edge_milli = Vec::with_capacity(points.len());
    let mut perimeter_milli = 0u64;
    let mut longest_edge = None;

    for i in 0..points.len() {
        let a = points[i];
        let b = points[(i + 1) % points.len()];
        signed_area2 += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;

        let length = edge_mill(a, b);
        edge_milli.push(length);
        perimeter_milli += length;

        let replace = match longest_edge {
            None => true,
            Some(old) if length > edge_milli[old] => true,
            Some(old) if length < edge_milli[old] => false,
            Some(old) => {
                let pair = endpoint_pair(a, b);
                let old_pair = endpoint_pair(points[old], points[(old + 1) % points.len()]);
                pair < old_pair || (pair == old_pair && i < old)
            }
        };
        if replace {
            longest_edge = Some(i);
        }
    }

    let orientation = match signed_area2.cmp(&0) {
        std::cmp::Ordering::Less => Orientation::Clockwise,
        std::cmp::Ordering::Equal => Orientation::Degenerate,
        std::cmp::Ordering::Greater => Orientation::CounterClockwise,
    };

    Measures {
        signed_area2,
        area2: signed_area2.unsigned_abs(),
        orientation,
        edge_milli,
        perimeter_milli,
        longest_edge,
    }
}

pub fn measure(points: &[Point]) -> Result<Measures, GeomError> {
    Ok(measures_for(prepare(points)?))
}

fn on_segment(a: Point, b: Point, q: Point) -> bool {
    cross(a, b, q) == 0
        && q.x >= a.x.min(b.x)
        && q.x <= a.x.max(b.x)
        && q.y >= a.y.min(b.y)
        && q.y <= a.y.max(b.y)
}

pub fn classify(points: &[Point], queries: &[Point]) -> Result<Vec<Location>, GeomError> {
    let polygon = prepare(points)?;
    if queries.len() > MAX_INPUT {
        return Err(GeomError::TooManyQueries { len: queries.len() });
    }
    for (index, &query) in queries.iter().enumerate() {
        if !in_range(query) {
            return Err(GeomError::QueryOutOfRange { index });
        }
    }

    Ok(queries
        .iter()
        .map(|&q| {
            for i in 0..polygon.len() {
                if on_segment(polygon[i], polygon[(i + 1) % polygon.len()], q) {
                    return Location::Boundary { edge: i };
                }
            }

            let mut inside = false;
            for i in 0..polygon.len() {
                let a = polygon[i];
                let b = polygon[(i + 1) % polygon.len()];
                if (a.y > q.y) != (b.y > q.y) {
                    let determinant = cross(a, b, q);
                    if (b.y > a.y && determinant > 0) || (b.y < a.y && determinant < 0) {
                        inside = !inside;
                    }
                }
            }
            if inside {
                Location::Inside
            } else {
                Location::Outside
            }
        })
        .collect())
}

fn between(a: Point, b: Point, p: Point) -> bool {
    p.x >= a.x.min(b.x)
        && p.x <= a.x.max(b.x)
        && p.y >= a.y.min(b.y)
        && p.y <= a.y.max(b.y)
}

pub fn convex_hull(points: &[Point], mode: Collinear) -> Result<Hull, GeomError> {
    if points.len() > MAX_INPUT {
        return Err(GeomError::TooManyVertices { len: points.len() });
    }
    for (index, &point) in points.iter().enumerate() {
        if !in_range(point) {
            return Err(GeomError::OutOfRange { index });
        }
    }

    let mut unique: Vec<(Point, usize)> = Vec::new();
    for (index, &point) in points.iter().enumerate() {
        if !unique.iter().any(|(p, _)| *p == point) {
            unique.push((point, index));
        }
    }
    unique.sort_by_key(|&(point, _)| point);

    let n = unique.len();
    let mut output: Vec<(Point, usize)>;

    if n <= 1 {
        output = unique;
    } else {
        let all_collinear = unique
            .iter()
            .skip(2)
            .all(|&(p, _)| cross(unique[0].0, unique[n - 1].0, p) == 0);

        if all_collinear {
            output = match mode {
                Collinear::Include => unique,
                Collinear::Exclude => vec![unique[0], unique[n - 1]],
            };
        } else {
            let mut lower = Vec::new();
            for &item in &unique {
                while lower.len() >= 2 {
                    let turn = cross(lower[lower.len() - 2].0, lower[lower.len() - 1].0, item.0);
                    let remove = match mode {
                        Collinear::Exclude => turn <= 0,
                        Collinear::Include => turn < 0,
                    };
                    if !remove {
                        break;
                    }
                    lower.pop();
                }
                lower.push(item);
            }

            let mut upper = Vec::new();
            for &item in unique.iter().rev() {
                while upper.len() >= 2 {
                    let turn = cross(upper[upper.len() - 2].0, upper[upper.len() - 1].0, item.0);
                    let remove = match mode {
                        Collinear::Exclude => turn <= 0,
                        Collinear::Include => turn < 0,
                    };
                    if !remove {
                        break;
                    }
                    upper.pop();
                }
                upper.push(item);
            }
            lower.pop();
            upper.pop();
            lower.extend(upper);
            output = lower;
        }
    }

    let vertices: Vec<Point> = output.iter().map(|&(p, _)| p).collect();
    let source_indices: Vec<usize> = output.iter().map(|&(_, i)| i).collect();
    let area2 = if vertices.len() < 3 {
        0
    } else {
        let mut sum = 0i128;
        for i in 0..vertices.len() {
            let a = vertices[i];
            let b = vertices[(i + 1) % vertices.len()];
            sum += a.x as i128 * b.y as i128 - a.y as i128 * b.x as i128;
        }
        sum.unsigned_abs()
    };
    let perimeter_milli = if vertices.is_empty() {
        0
    } else {
        (0..vertices.len())
            .map(|i| edge_mill(vertices[i], vertices[(i + 1) % vertices.len()]))
            .sum()
    };

    Ok(Hull {
        vertices,
        source_indices,
        area2,
        perimeter_milli,
    })
}
