```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components: Vec<&str> = Vec::new();
        if !path.starts_with('/') {
            components.extend(self.cwd.split('/').filter(|part| !part.is_empty()));
        }
        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                part => components.push(part),
            }
        }

        if components.is_empty() {
            Ok("/".to_owned())
        } else {
            Ok(format!("/{}", components.join("/")))
        }
    }

    fn parent_and_name(path: &str) -> (&str, &str) {
        match path.rfind('/') {
            Some(0) => ("/", &path[1..]),
            Some(index) => (&path[..index], &path[index + 1..]),
            None => ("/", path),
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(), Error> {
        let (parent, _) = Self::parent_and_name(path);
        match self.entries.get(parent) {
            None => Err(Error::Missing(parent.to_owned())),
            Some(Entry::File(_)) => Err(Error::NotDir(parent.to_owned())),
            Some(Entry::Dir) => Ok(()),
        }
    }

    fn resolve_existing(&self, path: &str) -> Result<&Entry, Error> {
        if path == "/" {
            return self
                .entries
                .get("/")
                .ok_or_else(|| Error::Missing("/".to_owned()));
        }

        let mut current = String::new();
        for (index, component) in path[1..].split('/').enumerate() {
            current.push('/');
            current.push_str(component);
            let entry = self
                .entries
                .get(&current)
                .ok_or_else(|| Error::Missing(current.clone()))?;
            if index + 1 < path[1..].split('/').count() && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(current));
            }
        }
        self.entries
            .get(path)
            .ok_or_else(|| Error::Missing(path.to_owned()))
    }

    fn contains_path(parent: &str, candidate: &str) -> bool {
        if parent == "/" {
            candidate.starts_with('/')
        } else {
            candidate == parent
                || (candidate.starts_with(parent)
                    && candidate.as_bytes().get(parent.len()) == Some(&b'/'))
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                if candidate.as_str() == path || !Self::contains_path(path, candidate) {
                    return false;
                }
                let suffix = if path == "/" {
                    &candidate[1..]
                } else {
                    &candidate[path.len() + 1..]
                };
                !suffix.is_empty() && !suffix.contains('/')
            })
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.normalize(path)?;
        if resolved == "/" {
            return Err(Error::Exists(resolved));
        }
        self.resolve_parent(&resolved)?;
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let resolved = self.normalize(path)?;
        if resolved == "/" {
            return Err(Error::IsDir(resolved));
        }
        self.resolve_parent(&resolved)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            None => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let resolved = self.normalize(path)?;
        match self.resolve_existing(&resolved)? {
            Entry::Dir => Err(Error::IsDir(resolved)),
            Entry::File(data) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let resolved = self.normalize(path)?;
        match self.resolve_existing(&resolved)? {
            Entry::File(_) => return Err(Error::NotDir(resolved)),
            Entry::Dir => {}
        }

        let mut result: Vec<(String, Kind)> = self
            .children(&resolved)
            .into_iter()
            .filter_map(|child| {
                self.entries.get(&child).map(|entry| {
                    let (_, basename) = Self::parent_and_name(&child);
                    (basename.to_owned(), entry.kind())
                })
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.normalize(path)?;
        match self.resolve_existing(&resolved)? {
            Entry::Dir => {
                self.cwd = resolved;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&source)?;
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        self.resolve_parent(&destination)?;

        let destination_entry = self.entries.get(&destination).cloned();
        if matches!(destination_entry, Some(Entry::Dir))
            && Self::contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        match (&source_entry, &destination_entry) {
            (Entry::Dir, Some(Entry::File(_))) => {
                return Err(Error::NotDir(destination));
            }
            (Entry::File(_), Some(Entry::Dir)) => {
                return Err(Error::IsDir(destination));
            }
            (Entry::Dir, Some(Entry::Dir)) if !self.children(&destination).is_empty() => {
                return Err(Error::NotEmpty(destination));
            }
            _ => {}
        }

        let source_descendants: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::contains_path(&source, path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if destination_entry.is_some() {
            self.entries.remove(&destination);
        }

        for (old_path, _) in &source_descendants {
            self.entries.remove(old_path);
        }

        for (old_path, entry) in source_descendants {
            let suffix = &old_path[source.len()..];
            let new_path = format!("{destination}{suffix}");
            self.entries.insert(new_path, entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let resolved = self.normalize(path)?;
        if resolved == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&resolved)?;
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && Self::contains_path(&resolved, &self.cwd) {
            return Err(Error::Busy(resolved));
        }

        let children = if matches!(entry, Entry::Dir) {
            self.children(&resolved)
        } else {
            Vec::new()
        };
        if !children.is_empty() && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        if recursive {
            let paths: Vec<String> = self
                .entries
                .keys()
                .filter(|candidate| Self::contains_path(&resolved, candidate))
                .cloned()
                .collect();
            for candidate in paths {
                self.entries.remove(&candidate);
            }
        } else {
            self.entries.remove(&resolved);
        }

        Ok(())
    }
}
```
