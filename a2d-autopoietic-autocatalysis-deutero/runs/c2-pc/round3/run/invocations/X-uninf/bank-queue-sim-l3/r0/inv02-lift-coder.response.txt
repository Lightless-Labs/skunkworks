#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    started: bool,
    remaining: u64,
    pause_time: Option<u64>,
}

struct Running {
    job: Job,
    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    use std::cmp::Ordering;
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(pa), Some(pb)) => pa.cmp(&pb).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut now = 0u64;

    while next_arrival < arrivals.len() || running.is_some() || !waiting.is_empty() {
        let mut next_tick: Option<u64> = None;
        if next_arrival < arrivals.len() {
            next_tick = Some(arrivals[next_arrival].arrival);
        }
        if let Some(active) = running.as_ref() {
            for t in [Some(active.finish), Some(active.work_at)] {
                if let Some(t) = t.filter(|&t| t > now) {
                    next_tick = Some(next_tick.map_or(t, |n| n.min(t)));
                }
            }
            if active.finish == now || active.work_at == now {
                next_tick = Some(now);
            }
        }

        if let Some(t) = next_tick {
            now = t;
        } else {
            // Work can only remain here if an idle dispatch is possible.
            if running.is_none() && !waiting.is_empty() {
                // Keep the current timestamp; dispatch below.
            } else {
                break;
            }
        }

        // Scheduled completion happens before arrivals at this tick.
        if running.as_ref().is_some_and(|r| r.finish == now) {
            let completed = running.take().unwrap();
            trace.push(Event {
                time: now,
                customer: completed.job.customer.id,
                kind: Kind::Finish,
            });
        }

        // Process the full arrival cohort.
        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            trace.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        // Protection ends at work_at, including boundaries without arrivals.
        if let Some(active) = running.as_ref() {
            if now >= active.work_at && now < active.finish {
                if let Some(best_index) = waiting
                    .iter()
                    .enumerate()
                    .min_by(|(_, a), (_, b)| waiter_cmp(a, b))
                    .map(|(i, _)| i)
                {
                    if waiting[best_index].customer.priority < active.job.customer.priority {
                        let best_id = waiting[best_index].customer.id;
                        let active = running.take().unwrap();
                        let remaining = active.finish - now;
                        trace.push(Event {
                            time: now,
                            customer: active.job.customer.id,
                            kind: Kind::Preempt {
                                remaining,
                                by: best_id,
                            },
                        });
                        let mut paused = active.job;
                        paused.remaining = remaining;
                        paused.pause_time = Some(now);
                        waiting.push(paused);
                    }
                }
            }
        }

        // Dispatch while idle, including repeated inline zero-duration finishes.
        loop {
            if running.is_some() {
                break;
            }
            if waiting.is_empty() {
                break;
            }
            let best_index = waiting
                .iter()
                .enumerate()
                .min_by(|(_, a), (_, b)| waiter_cmp(a, b))
                .map(|(i, _)| i)
                .unwrap();
            let mut job = waiting.remove(best_index);
            let cost = if last_dispatched == Some(job.customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };

            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time: now,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: job.customer.id,
                time: now,
            })?;

            last_dispatched = Some(job.customer.id);
            let kind = if job.started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            job.started = true;
            trace.push(Event {
                time: now,
                customer: job.customer.id,
                kind,
            });

            if finish == now {
                trace.push(Event {
                    time: now,
                    customer: job.customer.id,
                    kind: Kind::Finish,
                });
                continue;
            }
            running = Some(Running {
                job,
                work_at,
                finish,
            });
        }
    }

    Ok(trace)
}
