#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow { customer: u32, server: u32, time: u64 },
}

#[derive(Clone)]
struct RuntimeServer {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    idle_since: u64,
    busy: Option<(u64, u32)>, // finish time, customer ID
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<Customer> = customers.to_vec();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut runtime_servers: Vec<RuntimeServer> = servers
        .iter()
        .map(|server| RuntimeServer {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            idle_since: server.opens,
            busy: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut customer_index = 0usize;
    let mut now: Option<u64> = None;

    loop {
        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            next_time = Some(next_time.map_or(time, |current| current.min(time)));
        };

        if customer_index < ordered_customers.len() {
            consider(ordered_customers[customer_index].arrival);
        }
        for server in &runtime_servers {
            if !server.opened {
                consider(server.opens);
            }
            if let Some((finish, _)) = server.busy {
                consider(finish);
            }
        }

        let Some(time) = next_time else {
            break;
        };
        now = Some(time);

        let mut finishing: Vec<usize> = runtime_servers
            .iter()
            .enumerate()
            .filter_map(|(index, server)| {
                (server.busy.is_some_and(|(finish, _)| finish == time)).then_some(index)
            })
            .collect();
        finishing.sort_by_key(|&index| runtime_servers[index].id);
        for index in finishing {
            let (finish, customer) = runtime_servers[index].busy.take().unwrap();
            events.push(Event::Finish {
                time: finish,
                customer,
                server: runtime_servers[index].id,
            });
            runtime_servers[index].idle_since = time;
        }

        let mut opening: Vec<usize> = runtime_servers
            .iter()
            .enumerate()
            .filter_map(|(index, server)| {
                (!server.opened && server.opens == time).then_some(index)
            })
            .collect();
        opening.sort_by(|&a, &b| runtime_servers[b].id.cmp(&runtime_servers[a].id));
        for index in opening {
            runtime_servers[index].opened = true;
            events.push(Event::Open {
                time,
                server: runtime_servers[index].id,
            });
        }

        while customer_index < ordered_customers.len()
            && ordered_customers[customer_index].arrival == time
        {
            let customer = &ordered_customers[customer_index];
            events.push(Event::Arrive {
                time,
                customer: customer.id,
            });
            waiting.push(customer_index);
            customer_index += 1;
        }

        while !waiting.is_empty() {
            let selected = runtime_servers
                .iter()
                .enumerate()
                .filter(|(_, server)| server.opened && server.busy.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let customer_position = waiting.remove(0);
            let customer = &ordered_customers[customer_position];
            let server = &runtime_servers[server_index];

            let finish = customer
                .service
                .checked_add(time)
                .and_then(|value| value.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time,
                })?;

            let customer_id = customer.id;
            let server_id = server.id;
            events.push(Event::Start {
                time,
                customer: customer_id,
                server: server_id,
                finish,
            });

            if finish == time {
                events.push(Event::Finish {
                    time,
                    customer: customer_id,
                    server: server_id,
                });
                runtime_servers[server_index].idle_since = time;
            } else {
                runtime_servers[server_index].busy = Some((finish, customer_id));
            }
        }
    }

    let _ = now;
    Ok(events)
}
