#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut ordered_customers: Vec<Customer> = customers.to_vec();
    ordered_customers.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0usize;
    let mut now = 0u64;

    loop {
        // Finish events precede opens and arrivals at the same tick.
        let mut finishing: Vec<(u32, usize, u32)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                state
                    .busy_until
                    .filter(|&finish| finish == now)
                    .map(|_| {
                        let (customer, server) = states[index]
                            .busy_until
                            .map(|_| (0, state.server.id))
                            .unwrap();
                        (server, index, customer)
                    })
            })
            .collect();

        // The occupying customer is retained separately through the event list below.
        // This loop is intentionally replaced by the active-job records.
        finishing.clear();

        // Emit any finishes stored in active_jobs in ascending server ID order.
        // active_jobs is kept local to the simulation state below.

        let _ = &mut waiting;
        let _ = &mut next_customer;
        let _ = &mut events;
        let _ = &mut now;
        let _ = &mut states;

        // The actual event processing is performed by the complete state machine below.
        break;
    }

    // Re-run through a compact state machine with explicit active-job ownership.
    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            busy_until: None,
        })
        .collect();
    let mut active_customer: Vec<Option<u32>> = vec![None; states.len()];
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0usize;
    let mut now = 0u64;
    let mut events = Vec::new();

    loop {
        let mut finish_indices: Vec<usize> = (0..states.len())
            .filter(|&i| states[i].busy_until == Some(now))
            .collect();
        finish_indices.sort_by_key(|&i| states[i].server.id);
        for i in finish_indices {
            let customer = active_customer[i].take().unwrap();
            events.push(Event::Finish {
                time: now,
                customer,
                server: states[i].server.id,
            });
            states[i].busy_until = None;
            states[i].idle_since = now;
        }

        let mut opening_indices: Vec<usize> = (0..states.len())
            .filter(|&i| !states[i].opened && states[i].server.opens == now)
            .collect();
        opening_indices.sort_by(|&a, &b| states[b].server.id.cmp(&states[a].server.id));
        for i in opening_indices {
            states[i].opened = true;
            events.push(Event::Open {
                time: now,
                server: states[i].server.id,
            });
        }

        while next_customer < ordered_customers.len()
            && ordered_customers[next_customer].arrival == now
        {
            let customer_index = next_customer;
            events.push(Event::Arrive {
                time: now,
                customer: ordered_customers[customer_index].id,
            });
            waiting.push(customer_index);
            next_customer += 1;
        }

        loop {
            if waiting.is_empty() {
                break;
            }
            let selected = (0..states.len())
                .filter(|&i| states[i].opened && states[i].busy_until.is_none())
                .min_by(|&a, &b| {
                    states[a]
                        .idle_since
                        .cmp(&states[b].idle_since)
                        .then_with(|| states[a].server.overhead.cmp(&states[b].server.overhead))
                        .then_with(|| states[b].server.id.cmp(&states[a].server.id))
                });
            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &ordered_customers[customer_index];
            let server = &states[server_index].server;
            let finish = now
                .checked_add(customer.service)
                .and_then(|t| t.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = now;
            } else {
                states[server_index].busy_until = Some(finish);
                active_customer[server_index] = Some(customer.id);
            }
        }

        if next_customer == ordered_customers.len()
            && waiting.is_empty()
            && states.iter().all(|state| state.opened && state.busy_until.is_none())
        {
            break;
        }

        let next_time = ordered_customers
            .get(next_customer)
            .map(|customer| customer.arrival)
            .into_iter()
            .chain(
                states
                    .iter()
                    .filter(|state| !state.opened)
                    .map(|state| state.server.opens),
            )
            .chain(states.iter().filter_map(|state| state.busy_until))
            .filter(|&time| time > now)
            .min();

        match next_time {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}
