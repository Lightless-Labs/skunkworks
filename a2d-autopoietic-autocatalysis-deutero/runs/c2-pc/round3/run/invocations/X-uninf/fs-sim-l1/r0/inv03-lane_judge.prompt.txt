use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn components(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut start = 1;
        for (offset, byte) in path.bytes().enumerate().skip(1) {
            if byte == b'/' {
                if offset == start {
                    return Err(Error::InvalidPath(offset));
                }
                let component = &path[start..offset];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component);
                start = offset + 1;
            }
        }

        if start == path.len() {
            return Err(Error::InvalidPath(path.len() - 1));
        }

        let component = &path[start..];
        if component == "." || component == ".." {
            return Err(Error::InvalidPath(start));
        }
        components.push(component);
        Ok(components)
    }

    fn canonical(components: &[&str], count: usize) -> String {
        if count == 0 {
            "/".to_owned()
        } else {
            let mut result = String::new();
            for component in &components[..count] {
                result.push('/');
                result.push_str(component);
            }
            result
        }
    }

    fn dir_at<'a>(&'a self, components: &[&str], count: usize) -> Result<&'a BTreeMap<String, Entry>, Error> {
        let mut current = &self.root;
        for i in 0..count {
            let prefix = Self::canonical(components, i + 1);
            match current {
                Entry::Dir(children) => match children.get(components[i]) {
                    Some(entry) => current = entry,
                    None => return Err(Error::Missing(prefix)),
                },
                Entry::File(_) => return Err(Error::NotDir(Self::canonical(components, i))),
            }
            if i + 1 < count && matches!(current, Entry::File(_)) {
                return Err(Error::NotDir(prefix));
            }
        }

        match current {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::canonical(components, count))),
        }
    }

    fn entry_at<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        if components.is_empty() {
            return Ok(&self.root);
        }
        let parent = self.dir_at(components, components.len() - 1)?;
        let name = components[components.len() - 1];
        parent
            .get(name)
            .ok_or_else(|| Error::Missing(Self::canonical(components, components.len())))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let parent = self.dir_at_mut(&components, components.len() - 1)?;
        let name = components[components.len() - 1];
        let canonical = Self::canonical(&components, components.len());
        if parent.contains_key(name) {
            return Err(Error::Exists(canonical));
        }
        parent.insert(name.to_owned(), Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let canonical = Self::canonical(&components, components.len());
        let parent = self.dir_at_mut(&components, components.len() - 1)?;
        let name = components[components.len() - 1];
        match parent.get_mut(name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(canonical)),
            Some(Entry::File(bytes)) => {
                bytes.clear();
                bytes.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name.to_owned(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::components(path)?;
        match self.entry_at(&components)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical(&components, components.len()))),
            Entry::File(bytes) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::components(path)?;
        match self.entry_at(&components)? {
            Entry::File(_) => Err(Error::NotDir(Self::canonical(&components, components.len()))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }

    fn dir_at_mut<'a>(
        &'a mut self,
        components: &[&str],
        count: usize,
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let mut current = &mut self.root;
        for i in 0..count {
            let prefix = Self::canonical(components, i + 1);
            match current {
                Entry::Dir(children) => match children.get_mut(components[i]) {
                    Some(entry) => current = entry,
                    None => return Err(Error::Missing(prefix)),
                },
                Entry::File(_) => return Err(Error::NotDir(Self::canonical(components, i))),
            }
            if i + 1 < count && matches!(current, Entry::File(_)) {
                return Err(Error::NotDir(prefix));
            }
        }

        match current {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::canonical(components, count))),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self { entries }
    }

    fn canonicalize(path: &str) -> Result<Vec<String>, Error> {
        if let Some(index) = path.bytes().position(|byte| byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut start = 1;
        let bytes = path.as_bytes();

        for end in 1..=bytes.len() {
            if end == bytes.len() || bytes[end] == b'/' {
                if end == start {
                    return Err(Error::InvalidPath(if end == bytes.len() {
                        bytes.len() - 1
                    } else {
                        end
                    }));
                }

                let component = &path[start..end];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component.to_string());
                start = end + 1;
            }
        }

        Ok(components)
    }

    fn prefix(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn lookup(&self, components: &[String]) -> Result<&Entry, Error> {
        let mut prefix = Vec::new();
        let mut current = self
            .entries
            .get("/")
            .expect("root directory must always exist");

        if components.is_empty() {
            return Ok(current);
        }

        for (index, component) in components.iter().enumerate() {
            if !matches!(current, Entry::Dir) {
                return Err(Error::NotDir(Self::prefix(&prefix)));
            }
            prefix.push(component.clone());
            current = self
                .entries
                .get(&Self::prefix(&prefix))
                .ok_or_else(|| Error::Missing(Self::prefix(&prefix)))?;

            if index + 1 < components.len() && !matches!(current, Entry::Dir) {
                return Err(Error::NotDir(Self::prefix(&prefix)));
            }
        }

        Ok(current)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::canonicalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }

        let parent_components = &components[..components.len() - 1];
        self.lookup(parent_components)?;
        let parent_path = Self::prefix(parent_components);
        if !matches!(self.entries.get(&parent_path), Some(Entry::Dir)) {
            return Err(Error::NotDir(parent_path));
        }

        let canonical = Self::prefix(&components);
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }

        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::canonicalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        let parent_components = &components[..components.len() - 1];
        self.lookup(parent_components)?;
        let parent_path = Self::prefix(parent_components);
        if !matches!(self.entries.get(&parent_path), Some(Entry::Dir)) {
            return Err(Error::NotDir(parent_path));
        }

        let canonical = Self::prefix(&components);
        match self.entries.get_mut(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::canonicalize(path)?;
        match self.lookup(&components)? {
            Entry::Dir => Err(Error::IsDir(Self::prefix(&components))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::canonicalize(path)?;
        let canonical = Self::prefix(&components);
        match self.lookup(&components)? {
            Entry::File(_) => return Err(Error::NotDir(canonical.clone())),
            Entry::Dir => {}
        }

        let child_prefix = if canonical == "/" {
            "/".to_string()
        } else {
            format!("{}/", canonical)
        };
        let mut children = Vec::new();

        for (entry_path, entry) in &self.entries {
            if entry_path == &canonical || !entry_path.starts_with(&child_prefix) {
                continue;
            }
            let remainder = &entry_path[child_prefix.len()..];
            if remainder.contains('/') || remainder.is_empty() {
                continue;
            }
            let kind = match entry {
                Entry::Dir => Kind::Dir,
                Entry::File(_) => Kind::File,
            };
            children.push((remainder.to_string(), kind));
        }

        children.sort_by(|left, right| left.0.cmp(&right.0));
        Ok(children)
    }

pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn components(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut result = Vec::new();
        let mut start = 1;
        for (i, part) in path[1..].split('/').enumerate() {
            let offset = 1 + path[1..]
                .split('/')
                .take(i)
                .map(|s| s.len() + 1)
                .sum::<usize>();
            if part.is_empty() || part == "." || part == ".." {
                let invalid_at = if part.is_empty() && offset == path.len() {
                    path.len() - 1
                } else {
                    offset
                };
                return Err(Error::InvalidPath(invalid_at));
            }
            result.push(part);
            start = offset + part.len() + 1;
        }
        let _ = start;
        Ok(result)
    }

    fn canonical(parts: &[&str]) -> String {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            let mut path = String::new();
            for part in parts {
                path.push('/');
                path.push_str(part);
            }
            path
        }

                Entry::Dir(children) => {
                    prefix.push(part);
                    current = children
                        .get(*part)
                        .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
                }
                Entry::File(_) => return Err(Error::NotDir(Self::canonical(&prefix))),
            }
        }

        Ok(current)
    }

        let mut prefix: Vec<&str> = Vec::new();

        for part in parent_parts {
            match current {
                Entry::Dir(children) => {
                    prefix.push(part);
                    current = children
                        .get_mut(*part)
                        .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
                }
                Entry::File(_) => return Err(Error::NotDir(Self::canonical(&prefix))),
            }
        }

        let parent_path = Self::canonical(parent_parts);
        match current {
            Entry::Dir(children) => Ok((children, parent_path)),
            Entry::File(_) => Err(Error::NotDir(parent_path)),
        }
    }
