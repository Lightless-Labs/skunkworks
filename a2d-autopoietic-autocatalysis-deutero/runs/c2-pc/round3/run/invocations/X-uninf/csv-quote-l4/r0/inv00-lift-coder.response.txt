#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|s| s == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|s| s == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_with_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut pos = if bytes.starts_with(&[0xef, 0xbb, 0xbf]) {
        3
    } else {
        0
    };

    if pos == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut record_index = 0usize;
    let mut field_started = false;
    let mut in_quotes = false;
    let mut after_quote = false;
    let mut quote_start = 0usize;

    while pos < bytes.len() {
        let b = bytes[pos];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: pos,
            });
        }

        if !in_quotes && b == b'\r' && bytes.get(pos + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: pos,
            });
        }

        if in_quotes {
            if b == quote {
                if bytes.get(pos + 1) == Some(&quote) {
                    field.push(quote as char);
                    pos += 2;
                } else {
                    in_quotes = false;
                    after_quote = true;
                    pos += 1;
                }
            } else {
                let ch = input[pos..].chars().next().expect("valid UTF-8");
                field.push(ch);
                pos += ch.len_utf8();
            }
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                field_started = false;
                after_quote = false;
                pos += 1;
            } else if b == b'\n' || (b == b'\r' && bytes.get(pos + 1) == Some(&b'\n')) {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record_index += 1;
                field_started = false;
                after_quote = false;
                pos += if b == b'\r' { 2 } else { 1 };
            } else {
                return Err(CsvError::TrailingByte {
                    record: record_index,
                    byte: pos,
                });
            }
            continue;
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            field_started = false;
            pos += 1;
        } else if b == b'\n' || (b == b'\r' && bytes.get(pos + 1) == Some(&b'\n')) {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record_index += 1;
            field_started = false;
            pos += if b == b'\r' { 2 } else { 1 };
        } else if b == quote {
            if field_started {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: pos,
                });
            }
            in_quotes = true;
            quote_start = pos;
            field_started = true;
            pos += 1;
        } else {
            let ch = input[pos..].chars().next().expect("valid UTF-8");
            field.push(ch);
            field_started = true;
            pos += ch.len_utf8();
        }
    }

    if in_quotes {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if after_quote || field_started || !row.is_empty() || !field.is_empty() {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_with_dialect(input, b',', b'"')
}

fn duplicate_header_error(header: &[String]) -> Option<CsvError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();
    for (second, name) in header.iter().enumerate() {
        if let Some((_, first)) = seen.iter().find(|(old, _)| *old == name) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first: *first,
                second,
            });
        }
        seen.push((name, second));
    }
    None
}

fn make_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers && !records.is_empty() {
        let mut it = records.into_iter();
        (it.next(), it.collect::<Vec<_>>())
    } else {
        (None, records)
    };

    if let Some(h) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(error) = duplicate_header_error(h) {
                return Err(error);
            }
        }
        for (i, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: i + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    result: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote).map_err(|component| {
            CsvError::InvalidConfig { component }
        })?;
        Ok(Self {
            config,
            input: String::new(),
            result: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.result.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.result {
            return result.clone();
        }
        let result = parse_with_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| make_table(records, self.config));
        self.result = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.result = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len() && start == 0;
    }
    (b'1'..=b'9').contains(&b[start])
        && b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if start >= b.len() {
        return false;
    }
    let dot = match b[start..].iter().position(|&c| c == b'.') {
        Some(n) => start + n,
        None => return false,
    };
    if b.len() != dot + 3 {
        return false;
    }
    let integral = &b[start..dot];
    let integer_ok = if integral == b"0" {
        true
    } else {
        !integral.is_empty()
            && (b'1'..=b'9').contains(&integral[0])
            && integral[1..].iter().all(u8::is_ascii_digit)
    };
    integer_ok
        && b[dot + 1..].iter().all(u8::is_ascii_digit)
        && !(start == 1 && integral == b"0" && b[dot + 1..] == *b"00")
}

fn evaluate_cell(
    cell: Option<&String>,
    column: Option<usize>,
    spec: &ColumnSpec,
    row: usize,
    spec_index: usize,
) -> (Option<Value>, Option<CellError>) {
    let fail = |kind| {
        (
            None,
            Some(CellError {
                row,
                spec: spec_index,
                column,
                kind,
            }),
        )
    };

    let Some(cell) = cell else {
        return fail(CellErrorKind::MissingCell);
    };

    if spec.nullable && cell.is_empty() {
        return (Some(Value::Null), None);
    }
    if cell.is_empty() && spec.kind != ColumnType::Text {
        return fail(CellErrorKind::Empty);
    }

    let value = match spec.kind {
        ColumnType::Text => Value::Text(cell.clone()),
        ColumnType::Integer => {
            if !integer_syntax(cell) {
                return fail(CellErrorKind::InvalidInteger);
            }
            match cell.parse::<i64>() {
                Ok(v) => Value::Integer(v),
                Err(_) => return fail(CellErrorKind::IntegerOverflow),
            }
        }
        ColumnType::Decimal => {
            if !decimal_syntax(cell) {
                return fail(CellErrorKind::InvalidDecimal);
            }
            let negative = cell.starts_with('-');
            let unsigned = if negative { &cell[1..] } else { cell.as_str() };
            let dot = unsigned.find('.').unwrap();
            let combined = format!("{}{}", &unsigned[..dot], &unsigned[dot + 1..]);
            let magnitude = match combined.parse::<u64>() {
                Ok(v) => v,
                Err(_) => return fail(CellErrorKind::DecimalOverflow),
            };
            if negative {
                if magnitude > (i64::MAX as u64) + 1 {
                    return fail(CellErrorKind::DecimalOverflow);
                }
                if magnitude == (i64::MAX as u64) + 1 {
                    Value::Decimal(i64::MIN)
                } else {
                    Value::Decimal(-(magnitude as i64))
                }
            } else if magnitude > i64::MAX as u64 {
                return fail(CellErrorKind::DecimalOverflow);
            } else {
                Value::Decimal(magnitude as i64)
            }
        }
        ColumnType::Boolean => match cell.as_str() {
            "yes" | "1" => Value::Boolean(true),
            "no" | "0" => Value::Boolean(false),
            _ => return fail(CellErrorKind::InvalidBoolean),
        },
    };
    (Some(value), None)
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = vec![None; columns.len()];
        let mut had_error = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if policy == ErrorPolicy::Stop && !result.errors.is_empty() {
                break;
            }
            if policy == ErrorPolicy::FirstPerRow && had_error {
                break;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let Some(index) = resolved else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                had_error = true;
                if policy == ErrorPolicy::Stop {
                    result.rows.push(out);
                    return result;
                }
                continue;
            };

            let (value, error) =
                evaluate_cell(row.get(index), Some(index), spec, row_index, spec_index);
            out[spec_index] = value;
            if let Some(error) = error {
                result.errors.push(error);
                had_error = true;
                if policy == ErrorPolicy::Stop {
                    result.rows.push(out);
                    return result;
                }
            }
        }
        result.rows.push(out);
    }
    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns { record: usize },

        first: usize,
        second: usize,

}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote).map_err(|component| {
        WriteError::InvalidConfig { component }
    })?;

    let mut records: Vec<&[String]> = Vec::new();
    if let Some(header) = &table.header {
        records.push(header);
    }
    records.extend(table.rows.iter().map(Vec::as_slice));

    for (record_index, record) in records.iter().enumerate() {
        for (column, field) in record.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (i, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: i + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, fields) in records.iter().enumerate() {
        if fields.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(CsvError::DuplicateHeader {
                    name,
                    first,
                    second,
                }) = duplicate_header_error(header)
                {
                    return Err(WriteError::DuplicateHeader {
                        name,
                        first,
                        second,
                    });
                }
            }
        }
    }

    let mut output_records: Vec<Vec<String>> = Vec::new();
    if config.mode == WriteMode::Preserve {
        if let Some(header) = &table.header {
            output_records.push(header.clone());
        }
        output_records.extend(table.rows.clone());
    } else if let Some(header) = &table.header {
        let mut chosen: Vec<usize> = Vec::new();
        for (i, name) in header.iter().enumerate() {
            let matches: Vec<usize> = header
                .iter()
                .enumerate()
                .filter_map(|(j, other)| (other == name).then_some(j))
                .collect();
            let keep = match table.duplicates {
                DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => matches[0],
                DuplicatePolicy::KeepLast => *matches.last().unwrap(),
            };
            if i == keep {
                chosen.push(i);
            }
        }
        chosen.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));

        let projected_header = chosen.iter().map(|&i| header[i].clone()).collect();
        output_records.push(projected_header);

        let mut rows: Vec<Vec<String>> = table
            .rows
            .iter()
            .map(|row| chosen.iter().map(|&i| row[i].clone()).collect())
            .collect();
        rows.sort_by(|a, b| compare_rows(a, b));
        output_records.extend(rows);
    } else {
        let mut rows = table.rows.clone();
        rows.sort_by(|a, b| compare_rows(a, b));
        output_records = rows;
    }

    if output_records.is_empty() {
        return Ok(String::new());
    }

    let mut encoded_records = Vec::with_capacity(output_records.len());
    for (record_index, record) in output_records.iter().enumerate() {
        let mut encoded_fields = Vec::with_capacity(record.len());
        for (column_index, field) in record.iter().enumerate() {
            let bytes = field.as_bytes();
            let needs_quote = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
                || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t')
                || (record_index == 0 && column_index == 0 && field.starts_with('\u{feff}'))
                || (record.len() == 1 && field.is_empty());

            if needs_quote {
                let mut encoded = String::new();
                encoded.push(config.quote as char);
                for ch in field.chars() {
                    if ch as u32 == config.quote as u32 {
                        encoded.push(config.quote as char);
                        encoded.push(config.quote as char);
                    } else {
                        encoded.push(ch);
                    }
                }
                encoded.push(config.quote as char);
                encoded_fields.push(encoded);
            } else {
                encoded_fields.push(field.clone());
            }
        }
        encoded_records.push(encoded_fields.join(&(config.delimiter as char).to_string()));
    }

    let ending = match config.mode {
        WriteMode::Canonical => "\n",
        WriteMode::Preserve => match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        },
    };

    let mut output = encoded_records.join(ending);
    if config.mode == WriteMode::Canonical || config.final_terminator {
        output.push_str(ending);
    }
    Ok(output)
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        match left.as_bytes().cmp(right.as_bytes()) {
            std::cmp::Ordering::Equal => {}
            other => return other,
        }
    }
    a.len().cmp(&b.len())
}
