#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct CfwsRun {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<CfwsRun>), EmailError> {
    let mut clean = Vec::with_capacity(input.len());
    let mut map = Vec::with_capacity(input.len());
    let mut runs = Vec::new();

    let mut i = 0;
    let mut protected = false;
    let mut escaped = false;

    while i < input.len() {
        if protected {
            let b = input[i];
            clean.push(b);
            map.push(i);
            i += 1;
            if escaped {
                escaped = false;
            } else if b == b'\\' {
                escaped = true;
            } else if b == b'"' {
                protected = false;
            }
            continue;
        }

        let b = input[i];

        if b == b' ' || b == b'\t' || b == b'(' {
            let run_start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                match input[i] {
                    b' ' | b'\t' => i += 1,
                    b'\r' | b'\n' => {
                        let at = i;
                        if input[i] != b'\r'
                            || i + 2 >= input.len()
                            || input[i + 1] != b'\n'
                            || !is_hws(input[i + 2])
                        {
                            return Err(EmailError::BadFold { at });
                        }
                        i += 2;
                        while i < input.len() && is_hws(input[i]) {
                            i += 1;
                        }
                    }
                    b'(' => {
                        let outer = i;
                        let mut depth = 0usize;
                        loop {
                            if i >= input.len() {
                                return Err(EmailError::UnclosedComment { at: outer });
                            }
                            let c = input[i];
                            match c {
                                b'\\' => {
                                    let slash = i;
                                    if i + 1 >= input.len() || !is_printable(input[i + 1]) {
                                        return Err(EmailError::CommentEscape { at: slash });
                                    }
                                    i += 2;
                                }
                                b'\r' | b'\n' => {
                                    let at = i;
                                    if input[i] != b'\r'
                                        || i + 2 >= input.len()
                                        || input[i + 1] != b'\n'
                                        || !is_hws(input[i + 2])
                                    {
                                        return Err(EmailError::BadFold { at });
                                    }
                                    i += 2;
                                    while i < input.len() && is_hws(input[i]) {
                                        i += 1;
                                    }
                                }
                                b'(' => {
                                    depth += 1;
                                    if depth > 3 {
                                        return Err(EmailError::CommentDepth { at: i });
                                    }
                                    i += 1;
                                }
                                b')' => {
                                    depth -= 1;
                                    i += 1;
                                    if depth == 0 {
                                        break;
                                    }
                                }
                                c if is_printable(c) || c == b'\t' => i += 1,
                                _ => return Err(EmailError::CommentChar { at: i }),
                            }
                        }
                    }
                    _ => break,
                }
            }
            runs.push(CfwsRun {
                start: run_start,
                end: i,
            });
            continue;
        }

        if b == b'\r' || b == b'\n' {
            let at = i;
            if b != b'\r'
                || i + 2 >= input.len()
                || input[i + 1] != b'\n'
                || !is_hws(input[i + 2])
            {
                return Err(EmailError::BadFold { at });
            }
            let run_start = i;
            i += 2;
            while i < input.len() && is_hws(input[i]) {
                i += 1;
            }
            runs.push(CfwsRun {
                start: run_start,
                end: i,
            });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if b == b'"' && clean.is_empty() {
            protected = true;
            clean.push(b);
            map.push(i);
            i += 1;
            continue;
        }

        clean.push(b);
        map.push(i);
        i += 1;
    }

    Ok((clean, map, runs))
}

fn pos(map: &[usize], index: usize, input_len: usize) -> usize {
    map.get(index).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, map, runs) = cleanup(bytes)?;
    for run in runs {
        let left_at = run.start > 0 && bytes[run.start - 1] == b'@';
        let right_at = run.end < bytes.len() && bytes[run.end] == b'@';
        let at_boundary = run.start == 0 || run.end == bytes.len();
        if !at_boundary && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean[0] == b'"' {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(&map, 0, bytes.len()),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0;
    if let Some(end) = quote_end {
        i = end + 1;
    }
    while i < clean.len() {
        if clean[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: pos(&map, separators[1], bytes.len()),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&map, sep, bytes.len()),
        });
    }

    let local_bytes = &clean[..sep];
    let local;
    let quoted;

    if local_bytes[0] == b'"' {
        let end = quote_end.expect("quoted prefix has closing quote");
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&map, end + 1, bytes.len()),
            });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            let b = local_bytes[j];
            if b == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: pos(&map, j, bytes.len()),
                    });
                }
                let next = local_bytes[j + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: pos(&map, j, bytes.len()),
                    });
                }
                decoded.push(next);
                origins.push(pos(&map, j, bytes.len()));
                j += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar {
                        at: pos(&map, j, bytes.len()),
                    });
                }
                decoded.push(b);
                origins.push(pos(&map, j, bytes.len()));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(&map, end, bytes.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        local = String::from_utf8(decoded).expect("ASCII local");
        quoted = true;
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(&map, 64, bytes.len()),
            });
        }
        for j in 0..local_bytes.len() {
            let b = local_bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot {
                        at: pos(&map, j, bytes.len()),
                    });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar {
                    at: pos(&map, j, bytes.len()),
                });
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).expect("ASCII local");
        quoted = false;
    }

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let mut root_dot = None;
    if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(domain_end);
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos(&map, domain_start, bytes.len()),
        });
    }

    let domain_bytes = &clean[domain_start..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(&map, domain_start + 253, bytes.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for j in domain_start..=domain_end {
        if j == domain_end || clean[j] == b'.' {
            if label_start == j {
                let at = if j == domain_end {
                    root_dot
                        .map(|r| pos(&map, r, bytes.len()))
                        .unwrap_or_else(|| pos(&map, label_start, bytes.len()))
                } else {
                    pos(&map, j, bytes.len())
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    for &(start, end) in &labels {
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(&map, start + 63, bytes.len()),
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&map, start, bytes.len()),
            });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&map, end - 1, bytes.len()),
            });
        }
        for j in start..end {
            let b = clean[j];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar {
                    at: pos(&map, j, bytes.len()),
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&map, domain_start, bytes.len()),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: pos(&map, last_start, bytes.len()),
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(u8::to_ascii_lowercase)
        .collect::<Vec<_>>();
    Ok(Address {
        local,
        domain: String::from_utf8(domain).expect("ASCII domain"),
        quoted,
    })
}
