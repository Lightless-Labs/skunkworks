#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };

    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct Person {
    customer: Customer,
    initial_deadline: Option<u64>,
    accepted: bool,
    terminal: bool,
    remaining: u64,
    waiting: bool,
    queue_entry: u64,
    deadline: Option<u64>,
    waited: u64,
    pause_time: Option<u64>,
    started: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    interval_start: u64,
}

fn push_event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn best_waiter(people: &[Person]) -> Option<usize> {
    let mut best: Option<usize> = None;
    for (i, p) in people.iter().enumerate() {
        if !p.waiting || p.terminal {
            continue;
        }
        let better = match best {
            None => true,
            Some(j) => {
                let a = &p.customer;
                let b = &people[j].customer;
                if a.priority != b.priority {
                    a.priority < b.priority
                } else {
                    match (p.pause_time, people[j].pause_time) {
                        (Some(x), Some(y)) => {
                            x < y || (x == y && a.id < b.id)
                        }
                        (Some(_), None) => true,
                        (None, Some(_)) => false,
                        (None, None) => {
                            a.arrival < b.arrival
                                || (a.arrival == b.arrival && a.id > b.id)
                        }
                    }
                }
            }
        };
        if better {
            best = Some(i);
        }
    }
    best
}

fn consume_wait(p: &mut Person, now: u64, wait_ticks: &mut u128) {
    let segment = now - p.queue_entry;
    p.waited = p.waited.saturating_add(segment);
    *wait_ticks += segment as u128;
    if let Some(deadline) = p.deadline {
        let remaining = deadline.saturating_sub(now);
        p.deadline = Some(now.saturating_add(remaining));
    }
    p.waiting = false;
}

fn expire_waiters(
    people: &mut [Person],
    now: u64,
    events: &mut Vec<Event>,
    stats: &mut Stats,
    wait_ticks: &mut u128,
) {
    let mut expired: Vec<(u64, u32, usize)> = people
        .iter()
        .enumerate()
        .filter_map(|(i, p)| {
            if p.waiting && p.deadline.is_some_and(|d| d <= now) {
                Some((p.deadline.unwrap(), p.customer.id, i))
            } else {
                None
            }
        })
        .collect();
    expired.sort_by_key(|&(deadline, id, _)| (deadline, id));

    for (_, _, i) in expired {
        let p = &mut people[i];
        if !p.waiting {
            continue;
        }
        consume_wait(p, now, wait_ticks);
        p.terminal = true;
        stats.reneged += 1;
        push_event(events, now, p.customer.id, Kind::Renege { waited: p.waited });
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::<u32>::new();
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority { index, value: c.priority });
        }
        let deadline = match c.patience {
            Some(p) => Some(c.arrival.checked_add(p).ok_or(Error::DeadlineOverflow {
                index,
                id: c.id,
            })?),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut people: Vec<Person> = customers
        .iter()
        .zip(initial_deadlines.iter())
        .map(|(c, d)| Person {
            customer: c.clone(),
            initial_deadline: *d,
            accepted: false,
            terminal: false,
            remaining: c.service,
            waiting: false,
            queue_entry: c.arrival,
            deadline: None,
            waited: 0,
            pause_time: None,
            started: false,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..people.len()).collect();
    arrivals.sort_by(|&i, &j| {
        let a = &people[i].customer;
        let b = &people[j].customer;
        a.arrival
            .cmp(&b.arrival)
            .then(a.priority.cmp(&b.priority))
            .then(b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut active: Option<Active> = None;
    let mut previous_dispatch: Option<u32> = None;
    let mut first_dispatch = true;
    let mut arrival_pos = 0usize;
    let mut now = 0u64;

    loop {
        let next_arrival = arrivals
            .get(arrival_pos)
            .map(|&i| people[i].customer.arrival);
        let next_finish = active.as_ref().map(|a| a.finish);
        let next_work = active
            .as_ref()
            .filter(|a| a.work_at > now)
            .map(|a| a.work_at);
        let next_deadline = people
            .iter()
            .filter(|p| p.waiting)
            .filter_map(|p| p.deadline)
            .filter(|&d| d > now)
            .min();

        let mut next_time = None;
        for t in [next_arrival, next_finish, next_work, next_deadline].into_iter().flatten() {
            if t >= now && next_time.is_none_or(|n| t < n) {
                next_time = Some(t);
            }
        }

        if next_time.is_none() {
            break;
        }
        now = next_time.unwrap();

        // Phase A: finish first.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let a = active.take().unwrap();
            let p = &mut people[a.index];
            stats.busy_ticks += (now - a.interval_start) as u128;
            p.terminal = true;
            stats.completed += 1;
            stats.turnaround_ticks += (now - p.customer.arrival) as u128;
            stats.completion_order.push(p.customer.id);
            push_event(&mut events, now, p.customer.id, Kind::Finish);
        }

        // Phase B: expire existing waiters.
        expire_waiters(&mut people, now, &mut events, &mut stats, &mut stats.wait_ticks);

        // Phase C: process the full arrival cohort.
        while arrival_pos < arrivals.len()
            && people[arrivals[arrival_pos]].customer.arrival == now
        {
            let i = arrivals[arrival_pos];
            arrival_pos += 1;
            let id = people[i].customer.id;
            push_event(&mut events, now, id, Kind::Arrive);

            let waiting_count = people.iter().filter(|p| p.waiting).count();
            let idle = active.is_none();
            let gate = config.waiting_capacity + usize::from(idle);
            if waiting_count >= gate {
                stats.balked += 1;
                people[i].terminal = true;
                push_event(&mut events, now, id, Kind::Balk);
            } else {
                let p = &mut people[i];
                p.accepted = true;
                p.waiting = true;
                p.queue_entry = now;
                p.deadline = p.initial_deadline;
                stats.accepted += 1;
            }
        }

        // Phase D: compare and possibly preempt.
        let protection_ended = active.as_ref().is_some_and(|a| a.work_at <= now);
        if protection_ended {
            if let (Some(a), Some(w)) = (active.as_ref(), best_waiter(&people)) {
                if people[w].customer.priority < people[a.index].customer.priority {
                    let a = active.take().unwrap();
                    let remaining = a.finish - now;
                    let waiter_id = people[w].customer.id;
                    stats.busy_ticks += (now - a.interval_start) as u128;

                    {
                        let p = &mut people[a.index];
                        p.remaining = remaining;
                        p.waiting = true;
                        p.queue_entry = now;
                        p.pause_time = Some(now);
                        p.deadline = match p.customer.patience {
                            Some(_) => Some(now.checked_add(match p.customer.patience {
                                Some(budget) => {
                                    let spent = p.waited;
                                    budget.saturating_sub(spent)
                                }
                                None => 0,
                            }).ok_or(Error::TimeOverflow {
                                customer: p.customer.id,
                                time: now,
                                stage: OverflowStage::Patience,
                            })?),
                            None => None,
                        };
                    }

                    stats.preemptions += 1;
                    push_event(
                        &mut events,
                        now,
                        people[a.index].customer.id,
                        Kind::Preempt { remaining, by: waiter_id },
                    );
                }
            }
        }

        // Phase E: dispatch while idle, including inline zero-time finishes.
        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&people) else {
                break;
            };

            let id = people[i].customer.id;
            let remaining = people[i].remaining;
            let was_started = people[i].started;
            consume_wait(&mut people[i], now, &mut stats.wait_ticks);

            let switch = if first_dispatch {
                0
            } else if previous_dispatch == Some(id) {
                0
            } else {
                config.switch_cost
            };
            first_dispatch = false;
            previous_dispatch = Some(id);

            let work_at = now.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::Finish,
            })?;

            people[i].started = true;
            if !was_started {
                stats.started += 1;
                push_event(
                    &mut events,
                    now,
                    id,
                    Kind::Start { remaining, work_at, finish },
                );
            } else {
                push_event(
                    &mut events,
                    now,
                    id,
                    Kind::Resume { remaining, work_at, finish },
                );
            }

            active = Some(Active {
                index: i,
                work_at,
                finish,
                interval_start: now,
            });

            if finish == now {
                let a = active.take().unwrap();
                let p = &mut people[a.index];
                p.terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - p.customer.arrival) as u128;
                stats.completion_order.push(p.customer.id);
                push_event(&mut events, now, p.customer.id, Kind::Finish);
                continue;
            }
            break;
        }

        // Phase F: expire newly queued waiters.
        expire_waiters(&mut people, now, &mut events, &mut stats, &mut stats.wait_ticks);

        if let Some(last) = events.last() {
            stats.elapsed = last.time;
        }
    }

    if let Some(last) = events.last() {
        stats.elapsed = last.time;
    }

    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap_or(0)
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap_or(0)
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap_or(0)
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
