/// Parses an 81-character Sudoku string into a grid. Non-digit characters are treated as empty cells.
pub fn parse(s: &str) -> [[u8; 9]; 9] {
    let mut grid = [[0u8; 9]; 9];
    for (i, ch) in s.chars().take(81).enumerate() {
        let value = ch.to_digit(10).unwrap_or(0) as u8;
        grid[i / 9][i % 9] = if value <= 9 { value } else { 0 };
    }
    grid
}

/// Checks whether a completed Sudoku grid is valid.
pub fn validate(grid: &[[u8; 9]; 9]) -> bool {
    for r in 0..9 {
        for c in 0..9 {
            let value = grid[r][c];
            if !(1..=9).contains(&value) {
                return false;
            }
            for i in 0..9 {
                if i != c && grid[r][i] == value {
                    return false;
                }
                if i != r && grid[i][c] == value {
                    return false;
                }
            }
            let box_row = (r / 3) * 3;
            let box_col = (c / 3) * 3;
            for br in box_row..box_row + 3 {
                for bc in box_col..box_col + 3 {
                    if (br != r || bc != c) && grid[br][bc] == value {
                        return false;
                    }
                }
            }
        }
    }
    true
}

fn candidates(grid: &[[u8; 9]; 9], row: usize, col: usize) -> [bool; 10] {
    let mut allowed = [true; 10];
    allowed[0] = false;

    for i in 0..9 {
        let row_value = grid[row][i] as usize;
        let col_value = grid[i][col] as usize;
        if row_value <= 9 {
            allowed[row_value] = false;
        }
        if col_value <= 9 {
            allowed[col_value] = false;
        }
    }

    let box_row = (row / 3) * 3;
    let box_col = (col / 3) * 3;
    for r in box_row..box_row + 3 {
        for c in box_col..box_col + 3 {
            let value = grid[r][c] as usize;
            if value <= 9 {
                allowed[value] = false;
            }
        }
    }
    allowed
}

fn search(grid: &mut [[u8; 9]; 9]) -> bool {
    let mut best: Option<(usize, usize, [bool; 10], usize)> = None;

    for r in 0..9 {
        for c in 0..9 {
            if grid[r][c] != 0 {
                continue;
            }
            let allowed = candidates(grid, r, c);
            let count = allowed.iter().skip(1).filter(|&&yes| yes).count();
            if count == 0 {
                return false;
            }
            if best.as_ref().is_none_or(|(_, _, _, best_count)| count < *best_count) {
                best = Some((r, c, allowed, count));
                if count == 1 {
                    break;
                }
            }
        }
        if best.as_ref().is_some_and(|(_, _, _, count)| *count == 1) {
            break;
        }
    }

    let Some((row, col, allowed, _)) = best else {
        return true;
    };

    for value in 1..=9 {
        if allowed[value] {
            grid[row][col] = value as u8;
            if search(grid) {
                return true;
            }
            grid[row][col] = 0;
        }
    }
    false
}

/// Solves a Sudoku grid, returning `None` when its clues are invalid or unsolvable.
pub fn solve(mut grid: [[u8; 9]; 9]) -> Option<[[u8; 9]; 9]> {
    if grid.iter().flatten().any(|&value| value > 9) {
        return None;
    }

    for r in 0..9 {
        for c in 0..9 {
            let value = grid[r][c];
            if value == 0 {
                continue;
            }
            grid[r][c] = 0;
            let allowed = candidates(&grid, r, c);
            grid[r][c] = value;
            if !allowed[value as usize] {
                return None;
            }
        }
    }

    if search(&mut grid) {
        Some(grid)
    } else {
        None
    }
}
