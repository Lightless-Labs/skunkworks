/// Generates every legal move for the requested side.
pub fn generate_moves(board: &Board, white: bool) -> Vec<Move> {
    pseudo_moves(board, white)
        .into_iter()

/// Applies a move and returns the resulting board position.
pub fn apply_move(board: &Board, mv: &Move) -> Board {
    let mut next = board.clone();
    if mv.from_rank >= 8 || mv.from_file >= 8 || mv.to_rank >= 8 || mv.to_file >= 8 {
        return next;
    }
    let Some(mut piece) = next.squares[mv.from_rank][mv.from_file] else { return next };
    let captured = next.squares[mv.to_rank][mv.to_file];

    if piece.kind == PieceKind::Pawn
        && mv.from_file != mv.to_file
        && captured.is_none()
        && board.en_passant == Some((mv.to_rank, mv.to_file))
    {
        let captured_rank = if piece.white { mv.to_rank - 1 } else { mv.to_rank + 1 };
        next.squares[captured_rank][mv.to_file] = None;
    }

    if piece.kind == PieceKind::King && mv.from_file.abs_diff(mv.to_file) == 2 {
        let rank = mv.from_rank;
        if mv.to_file == 6 {
            next.squares[rank][5] = next.squares[rank][7];
            next.squares[rank][7] = None;
        } else {
            next.squares[rank][3] = next.squares[rank][0];
            next.squares[rank][0] = None;
        }
    }

    if piece.kind == PieceKind::King {
        if piece.white {
            next.white_king_side = false;
            next.white_queen_side = false;
        } else {
            next.black_king_side = false;
            next.black_queen_side = false;
        }
    }
    if piece.kind == PieceKind::Rook {
        update_rook_rights(&mut next, piece.white, mv.from_rank, mv.from_file);
    }
    if let Some(cap) = captured {
        if cap.kind == PieceKind::Rook {
            update_rook_rights(&mut next, cap.white, mv.to_rank, mv.to_file);
        }
    }

    next.en_passant = None;
    if piece.kind == PieceKind::Pawn && mv.from_rank.abs_diff(mv.to_rank) == 2 {
        next.en_passant = Some(((mv.from_rank + mv.to_rank) / 2, mv.from_file));
    }
    if piece.kind == PieceKind::Pawn && (mv.to_rank == 0 || mv.to_rank == 7) {
        piece.kind = mv.promotion.unwrap_or(PieceKind::Queen);
    }
    next.squares[mv.from_rank][mv.from_file] = None;
    next.squares[mv.to_rank][mv.to_file] = Some(piece);
    next
}

fn update_rook_rights(board: &mut Board, white: bool, rank: usize, file: usize) {
    let home = if white { 0 } else { 7 };
    if rank != home { return; }
    match (white, file) {
        (true, 0) => board.white_queen_side = false,
        (true, 7) => board.white_king_side = false,
        (false, 0) => board.black_queen_side = false,
        (false, 7) => board.black_king_side = false,
        _ => {}
    }
}

/// Evaluates material balance from White's perspective.
pub fn evaluate_material(board: &Board) -> i32 {

            let value = match piece.kind {
                PieceKind::Pawn => 100,
                PieceKind::Knight => 320,
