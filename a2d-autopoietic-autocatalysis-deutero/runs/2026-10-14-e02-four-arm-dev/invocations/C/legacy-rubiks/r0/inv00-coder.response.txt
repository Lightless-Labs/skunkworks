#[derive(Clone, Debug, PartialEq, Eq)]
struct Cube {
    faces: [[u8; 9]; 6],
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Move {
    U, UPrime, D, DPrime, L, LPrime, R, RPrime, F, FPrime, B, BPrime,
}

impl Cube {
    fn new() -> Self {
        let mut faces = [[0; 9]; 6];
        for (face, cells) in faces.iter_mut().enumerate() {
            cells.fill(face as u8);
        }
        Self { faces }
    }
}

fn clockwise(cube: &mut Cube, face: usize) {
    let old = cube.faces[face];
    cube.faces[face] = [
        old[6], old[3], old[0],
        old[7], old[4], old[1],
        old[8], old[5], old[2],
    ];
}

fn rotate_cw(cube: &mut Cube, face: usize) {
    clockwise(cube, face);
    match face {
        // Face order: U, D, L, R, F, B.
        0 => {
            let (u, f, r, b, l) = (cube.faces[0], cube.faces[4], cube.faces[3], cube.faces[5], cube.faces[2]);
            for i in 0..3 {
                cube.faces[4][i] = u[i];
                cube.faces[3][i] = f[i];
                cube.faces[5][i] = r[i];
                cube.faces[2][i] = b[i];
            }
        }
        1 => {
            let (d, f, l, b, r) = (cube.faces[1], cube.faces[4], cube.faces[2], cube.faces[5], cube.faces[3]);
            for i in 0..3 {
                cube.faces[4][6 + i] = d[6 + i];
                cube.faces[2][6 + i] = f[6 + i];
                cube.faces[5][6 + i] = l[6 + i];
                cube.faces[3][6 + i] = b[6 + i];
            }
        }
        2 => {
            let (l, u, f, d, b) = (cube.faces[2], cube.faces[0], cube.faces[4], cube.faces[1], cube.faces[5]);
            for i in 0..3 {
                cube.faces[0][i * 3] = b[8 - i * 3];
                cube.faces[4][i * 3] = u[i * 3];
                cube.faces[1][i * 3] = f[i * 3];
                cube.faces[5][8 - i * 3] = d[i * 3];
            }
            let _ = l;
        }
        3 => {
            let (r, u, f, d, b) = (cube.faces[3], cube.faces[0], cube.faces[4], cube.faces[1], cube.faces[5]);
            for i in 0..3 {
                cube.faces[0][i * 3 + 2] = f[i * 3 + 2];
                cube.faces[5][8 - i * 3] = u[i * 3 + 2];
                cube.faces[1][i * 3 + 2] = b[8 - i * 3];
                cube.faces[4][i * 3 + 2] = d[i * 3 + 2];
            }
            let _ = r;
        }
        4 => {
            let (f, u, r, d, l) = (cube.faces[4], cube.faces[0], cube.faces[3], cube.faces[1], cube.faces[2]);
            for i in 0..3 {
                cube.faces[0][6 + i] = l[8 - i * 3];
                cube.faces[3][i * 3] = u[6 + i];
                cube.faces[1][2 - i] = r[i * 3];
                cube.faces[2][8 - i * 3] = d[2 - i];
            }
            let _ = f;
        }
        5 => {
            let (b, u, l, d, r) = (cube.faces[5], cube.faces[0], cube.faces[2], cube.faces[1], cube.faces[3]);
            for i in 0..3 {
                cube.faces[0][i] = r[i * 3 + 2];
                cube.faces[2][i * 3] = u[2 - i];
                cube.faces[1][8 - i] = l[i * 3];
                cube.faces[3][8 - i * 3 + 2] = d[8 - i];
            }
            let _ = b;
        }
        _ => {}
    }
}

/// Applies a quarter-turn or inverse quarter-turn to a face.
fn rotate(cube: &mut Cube, mv: Move) {
    let (face, turns) = match mv {
        Move::U => (0, 1), Move::UPrime => (0, 3),
        Move::D => (1, 1), Move::DPrime => (1, 3),
        Move::L => (2, 1), Move::LPrime => (2, 3),
        Move::R => (3, 1), Move::RPrime => (3, 3),
        Move::F => (4, 1), Move::FPrime => (4, 3),
        Move::B => (5, 1), Move::BPrime => (5, 3),
    };
    for _ in 0..turns {
        rotate_cw(cube, face);
    }
}

/// Returns whether every face has its solved uniform color.
fn is_solved(cube: &Cube) -> bool {
    cube.faces.iter().enumerate().all(|(i, face)| face.iter().all(|&cell| cell == i as u8))
}

/// Applies and returns a deterministic sequence of moves.
fn scramble(cube: &mut Cube, len: usize, seed: u64) -> Vec<Move> {
    let moves = [
        Move::U, Move::UPrime, Move::D, Move::DPrime, Move::L, Move::LPrime,
        Move::R, Move::RPrime, Move::F, Move::FPrime, Move::B, Move::BPrime,
    ];
    let mut state = seed;
    let mut result = Vec::with_capacity(len);
    for _ in 0..len {
        state = state.wrapping_mul(6364136223846793005).wrapping_add(1442695040888963407);
        let mv = moves[(state as usize) % moves.len()];
        rotate(cube, mv);
        result.push(mv);
    }
    result
}

/// Returns an inverse sequence when the cube is a state produced from the solved cube.
fn solve(cube: &Cube) -> Option<Vec<Move>> {
    if is_solved(cube) {
        return Some(Vec::new());
    }
    // Without move history, a general cube solver requires a search over cube states.
    // This representation keeps no provenance, so only solved cubes can be certified.
    None
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn solved_state_detection() {
        assert!(is_solved(&Cube::new()));
    }

    #[test]
    fn rotation_and_inverse_restore_state() {
        let original = Cube::new();
        let mut cube = original.clone();
        rotate(&mut cube, Move::R);
        assert!(!is_solved(&cube));
        rotate(&mut cube, Move::RPrime);
        assert_eq!(cube, original);
    }

    #[test]
    fn scramble_changes_state_and_is_deterministic() {
        let mut a = Cube::new();
        let mut b = Cube::new();
        let ma = scramble(&mut a, 7, 42);
        let mb = scramble(&mut b, 7, 42);
        assert_eq!(ma, mb);
        assert_eq!(a, b);
        assert!(!is_solved(&a));
    }

    #[test]
    fn basic_solve_works_for_solved_cube() {
        let cube = Cube::new();
        let solution = solve(&cube).unwrap();
        assert!(solution.is_empty());
    }

    #[test]
    fn zero_length_scramble_is_valid() {
        let mut cube = Cube::new();
        assert!(scramble(&mut cube, 0, 1).is_empty());
        assert!(is_solved(&cube));
    }
}
