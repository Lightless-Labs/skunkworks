#[derive(Clone, Debug, PartialEq, Eq)]
struct Cube {
    faces: [[u8; 9]; 6],
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Move {
    U, UPrime,
    D, DPrime,
    L, LPrime,
    R, RPrime,
    F, FPrime,
    B, BPrime,
}

impl Cube {
    fn new() -> Self {
        Self {
            faces: [
                [0; 9],
                [1; 9],
                [2; 9],
                [3; 9],
                [4; 9],
                [5; 9],
            ],
        }
    }
}

fn rotate_face_clockwise(face: &mut [u8; 9]) {
    let old = *face;
    face[0] = old[6];
    face[1] = old[3];
    face[2] = old[0];
    face[3] = old[7];
    face[4] = old[4];
    face[5] = old[1];
    face[6] = old[8];
    face[7] = old[5];
    face[8] = old[2];
}

fn turn_clockwise(cube: &mut Cube, face: usize, strips: [[usize; 3]; 4]) {
    rotate_face_clockwise(&mut cube.faces[face]);
    let old = cube.faces;
    for i in 0..4 {
        for j in 0..3 {
            cube.faces[strips[(i + 1) % 4][0]][strips[(i + 1) % 4][1 + j]] =
                old[strips[i][0]][strips[i][1 + j]];
        }
    }
}

fn clockwise(cube: &mut Cube, mv: Move) {
    match mv {
        Move::U => turn_clockwise(cube, 0, [[4, 0, 1], [2, 0, 1], [5, 0, 1], [3, 0, 1]]),
        Move::D => turn_clockwise(cube, 1, [[4, 6, 7], [3, 6, 7], [5, 6, 7], [2, 6, 7]]),
        Move::L => turn_clockwise(cube, 2, [[0, 0, 3], [5, 2, 5], [1, 0, 3], [4, 6, 7]]),
        Move::R => turn_clockwise(cube, 3, [[0, 2, 5], [4, 2, 5], [1, 2, 5], [5, 6, 7]]),
        Move::F => turn_clockwise(cube, 4, [[0, 6, 7], [3, 0, 3], [1, 2, 1], [2, 8, 5]]),
        Move::B => turn_clockwise(cube, 5, [[0, 2, 1], [2, 0, 3], [1, 6, 7], [3, 8, 5]]),
        _ => unreachable!(),
    }
}

fn rotate(cube: &mut Cube, mv: Move) {
    let (base, turns) = match mv {
        Move::U => (Move::U, 1),
        Move::UPrime => (Move::U, 3),
        Move::D => (Move::D, 1),
        Move::DPrime => (Move::D, 3),
        Move::L => (Move::L, 1),
        Move::LPrime => (Move::L, 3),
        Move::R => (Move::R, 1),
        Move::RPrime => (Move::R, 3),
        Move::F => (Move::F, 1),
        Move::FPrime => (Move::F, 3),
        Move::B => (Move::B, 1),
        Move::BPrime => (Move::B, 3),
    };
    for _ in 0..turns {
        clockwise(cube, base);
    }
}

fn is_solved(cube: &Cube) -> bool {
    cube.faces
        .iter()
        .all(|face| face.iter().all(|&cell| cell == face[4]))
}

fn scramble(cube: &mut Cube, len: usize, seed: u64) -> Vec<Move> {
    const MOVES: [Move; 12] = [
        Move::U, Move::UPrime, Move::D, Move::DPrime,
        Move::L, Move::LPrime, Move::R, Move::RPrime,
        Move::F, Move::FPrime, Move::B, Move::BPrime,
    ];

    let mut state = seed;
    let mut result = Vec::with_capacity(len);
    for _ in 0..len {
        // SplitMix64, a small deterministic generator.
        state = state.wrapping_add(0x9E3779B97F4A7C15);
        let mut z = state;
        z = (z ^ (z >> 30)).wrapping_mul(0xBF58476D1CE4E5B9);
        z = (z ^ (z >> 27)).wrapping_mul(0x94D049BB133111EB);
        z ^= z >> 31;

        let mv = MOVES[(z as usize) % MOVES.len()];
        rotate(cube, mv);
        result.push(mv);
    }
    result
}

fn solve(cube: &Cube) -> Option<Vec<Move>> {
    // Each face turn is a permutation of the 54 stickers. Reverse BFS from
    // the supplied state is impractical for the full cube group, so use an
    // IDA* search with a conservative misplaced-sticker heuristic.
    if is_solved(cube) {
        return Some(Vec::new());
    }

    const MOVES: [Move; 12] = [
        Move::U, Move::UPrime, Move::D, Move::DPrime,
        Move::L, Move::LPrime, Move::R, Move::RPrime,
        Move::F, Move::FPrime, Move::B, Move::BPrime,
    ];

    fn heuristic(cube: &Cube) -> usize {
        let misplaced: usize = cube.faces.iter().map(|f| {
            f.iter().filter(|&&x| x != f[4]).count()
        }).sum();
        (misplaced + 11) / 12
    }

    fn search(
        cube: &mut Cube,
        remaining: usize,
        previous_face: Option<usize>,
        path: &mut Vec<Move>,
        nodes: &mut usize,
        limit: usize,
    ) -> bool {
        *nodes += 1;
        if is_solved(cube) {
            return true;
        }
        if remaining == 0 || heuristic(cube) > remaining || *nodes > limit {
            return false;
        }

        for (index, &mv) in MOVES.iter().enumerate() {
            let face = index / 2;
            if previous_face == Some(face) {
                continue;
            }
            rotate(cube, mv);
            path.push(mv);
            if search(cube, remaining - 1, Some(face), path, nodes, limit) {
                return true;
            }
            path.pop();
            rotate(cube, inverse(mv));
            if *nodes > limit {
                return false;
            }
        }
        false
    }

    fn inverse(mv: Move) -> Move {
        match mv {
            Move::U => Move::UPrime, Move::UPrime => Move::U,
            Move::D => Move::DPrime, Move::DPrime => Move::D,
            Move::L => Move::LPrime, Move::LPrime => Move::L,
            Move::R => Move::RPrime, Move::RPrime => Move::R,
            Move::F => Move::FPrime, Move::FPrime => Move::F,
            Move::B => Move::BPrime, Move::BPrime => Move::B,
        }
    }

    let mut work = cube.clone();
    let mut path = Vec::new();
    let mut bound = heuristic(&work);
    // A finite cap prevents unbounded work on hard states.
    while bound <= 12 {
        let mut nodes = 0;
        if search(&mut work, bound, None, &mut path, &mut nodes, 2_000_000) {
            return Some(path);
        }
        if nodes > 2_000_000 {
            return None;
        }
        bound += 1;
    }
    None
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn solved_state_detection() {
        assert!(is_solved(&Cube::new()));
    }

    #[test]
    fn single_rotation_and_inverse_restore_state() {
        let original = Cube::new();
        let mut cube = original.clone();
        rotate(&mut cube, Move::R);
        assert!(!is_solved(&cube));
        rotate(&mut cube, Move::RPrime);
        assert_eq!(cube, original);
    }

    #[test]
    fn scramble_changes_state() {
        let mut cube = Cube::new();
        let moves = scramble(&mut cube, 10, 42);
        assert_eq!(moves.len(), 10);
        assert!(!is_solved(&cube));
    }

    #[test]
    fn basic_solve_works() {
        let mut cube = Cube::new();
        let moves = scramble(&mut cube, 3, 42);
        let solution = solve(&cube).expect("short scramble should be solvable");
        for mv in solution {
            rotate(&mut cube, mv);
        }
        assert!(is_solved(&cube));
        assert_eq!(moves.len(), 3);
    }
}
