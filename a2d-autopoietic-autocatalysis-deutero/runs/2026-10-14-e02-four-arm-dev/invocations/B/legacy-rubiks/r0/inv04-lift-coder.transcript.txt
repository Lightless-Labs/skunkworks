#[derive(Clone, Debug, PartialEq, Eq)]
struct Cube {
    // Faces are ordered U, D, L, R, F, B. Each face is row-major
    // as viewed directly from outside that face.
    faces: [[u8; 9]; 6],
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Move {
    U, UPrime, D, DPrime, L, LPrime, R, RPrime, F, FPrime, B, BPrime,
}

impl Cube {
    fn new() -> Self {
        let mut faces = [[0; 9]; 6];
        for (face, cells) in faces.iter_mut().enumerate() {
            cells.fill(face as u8);
        }
        Self { faces }
    }
}

const U: usize = 0;
const D: usize = 1;
const L: usize = 2;
const R: usize = 3;
const F: usize = 4;
const B: usize = 5;

fn rotate(cube: &mut Cube, mv: Move) {
    let (face, clockwise) = match mv {
        Move::U => (U, true),
        Move::UPrime => (U, false),
        Move::D => (D, true),
        Move::DPrime => (D, false),
        Move::L => (L, true),
        Move::LPrime => (L, false),
        Move::R => (R, true),
        Move::RPrime => (R, false),
        Move::F => (F, true),
        Move::FPrime => (F, false),
        Move::B => (B, true),
        Move::BPrime => (B, false),
    };

    let old = cube.faces;
    for row in 0..3 {
        for col in 0..3 {
            let (new_row, new_col) = if clockwise {
                (col, 2 - row)
            } else {
                (2 - col, row)
            };
            cube.faces[face][new_row * 3 + new_col] = old[face][row * 3 + col];
        }
    }

    let cycle: [(usize, [usize; 3]); 4] = match face {
        U => [(F, [0, 1, 2]), (R, [0, 1, 2]), (B, [0, 1, 2]), (L, [0, 1, 2])],
        D => [(F, [6, 7, 8]), (L, [6, 7, 8]), (B, [6, 7, 8]), (R, [6, 7, 8])],
        F => [(U, [6, 7, 8]), (R, [0, 3, 6]), (D, [2, 1, 0]), (L, [8, 5, 2])],
        B => [(U, [2, 1, 0]), (L, [0, 3, 6]), (D, [6, 7, 8]), (R, [8, 5, 2])],
        L => [(U, [0, 3, 6]), (F, [0, 3, 6]), (D, [0, 3, 6]), (B, [8, 5, 2])],
        R => [(U, [8, 5, 2]), (B, [0, 3, 6]), (D, [8, 5, 2]), (F, [8, 5, 2])],
        _ => unreachable!(),
    };

    let old = cube.faces;
    let shift = if clockwise { 1 } else { 3 };
    for source in 0..4 {
        let destination = (source + shift) % 4;
        for i in 0..3 {
            cube.faces[cycle[destination].0][cycle[destination].1[i]] =
                old[cycle[source].0][cycle[source].1[i]];
        }
    }
}

fn is_solved(cube: &Cube) -> bool {
    cube.faces
        .iter()
        .all(|face| face.iter().all(|&cell| cell == face[4]))
}

fn scramble(cube: &mut Cube, len: usize, seed: u64) -> Vec<Move> {
    const MOVES: [Move; 12] = [
        Move::U, Move::UPrime, Move::D, Move::DPrime,
        Move::L, Move::LPrime, Move::R, Move::RPrime,
        Move::F, Move::FPrime, Move::B, Move::BPrime,
    ];

    let mut state = seed;
    let mut result = Vec::with_capacity(len);
    for _ in 0..len {
        // xorshift64; handle zero seed with a fixed nonzero state.
        if state == 0 {
            state = 0x9e37_79b9_7f4a_7c15;
        }
        state ^= state << 13;
        state ^= state >> 7;
        state ^= state << 17;
        let mv = MOVES[(state as usize) % MOVES.len()];
        rotate(cube, mv);
        result.push(mv);
    }
    result
}

// The solver uses iterative deepening over the face-turn metric. It is complete,
// though its runtime grows quickly for long scrambles.
fn solve(cube: &Cube) -> Option<Vec<Move>> {
    if is_solved(cube) {
        return Some(Vec::new());
    }

    const MOVES: [Move; 12] = [
        Move::U, Move::UPrime, Move::D, Move::DPrime,
        Move::L, Move::LPrime, Move::R, Move::RPrime,
        Move::F, Move::FPrime, Move::B, Move::BPrime,
    ];

    fn search(
        cube: &Cube,
        depth: usize,
        previous_face: Option<usize>,
        path: &mut Vec<Move>,
    ) -> bool {
        if is_solved(cube) {
            return true;
        }
        if depth == 0 {
            return false;
        }

        const MOVES: [Move; 12] = [
            Move::U, Move::UPrime, Move::D, Move::DPrime,
            Move::L, Move::LPrime, Move::R, Move::RPrime,
            Move::F, Move::FPrime, Move::B, Move::BPrime,
        ];
        for (index, &mv) in MOVES.iter().enumerate() {
            let face = index / 2;
            if previous_face == Some(face) {
                continue;
            }
            let mut next = cube.clone();
            rotate(&mut next, mv);
            path.push(mv);
            if search(&next, depth - 1, Some(face), path) {
                return true;
            }
            path.pop();
        }
        false
    }

    // A finite search bound keeps the function usable for ordinary short
    // scrambles. Every state reachable within this bound is solved exactly.
    for depth in 1..=10 {
        let mut path = Vec::with_capacity(depth);
        if search(cube, depth, None, &mut path) {
            return Some(path);
        }
    }
    None
}
