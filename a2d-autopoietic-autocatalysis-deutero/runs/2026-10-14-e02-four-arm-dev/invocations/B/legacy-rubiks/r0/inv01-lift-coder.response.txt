#[derive(Clone, Debug, PartialEq, Eq)]
struct Cube {
    faces: [[u8; 9]; 6],
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Move {
    U,
    UPrime,
    D,
    DPrime,
    L,
    LPrime,
    R,
    RPrime,
    F,
    FPrime,
    B,
    BPrime,
}

impl Cube {
    fn new() -> Self {
        Self {
            faces: [
                [0; 9],
                [1; 9],
                [2; 9],
                [3; 9],
                [4; 9],
                [5; 9],
            ],
        }
    }

    fn turn_cw(&mut self, face: usize) {
        let old = self.faces[face];
        self.faces[face] = [
            old[6], old[3], old[0], old[7], old[4], old[1], old[8], old[5], old[2],
        ];
    }
}

fn rotate(cube: &mut Cube, mv: Move) {
    let (face, clockwise) = match mv {
        Move::U => (0, true),
        Move::UPrime => (0, false),
        Move::D => (1, true),
        Move::DPrime => (1, false),
        Move::L => (2, true),
        Move::LPrime => (2, false),
        Move::R => (3, true),
        Move::RPrime => (3, false),
        Move::F => (4, true),
        Move::FPrime => (4, false),
        Move::B => (5, true),
        Move::BPrime => (5, false),
    };

    if clockwise {
        cube.turn_cw(face);
    } else {
        for _ in 0..3 {
            cube.turn_cw(face);
        }
    }
}

fn is_solved(cube: &Cube) -> bool {
    cube.faces
        .iter()
        .all(|face| face.iter().all(|&cell| cell == face[0]))
}

fn scramble(cube: &mut Cube, len: usize, seed: u64) -> Vec<Move> {
    const MOVES: [Move; 12] = [
        Move::U,
        Move::UPrime,
        Move::D,
        Move::DPrime,
        Move::L,
        Move::LPrime,
        Move::R,
        Move::RPrime,
        Move::F,
        Move::FPrime,
        Move::B,
        Move::BPrime,
    ];

    let mut state = seed;
    let mut result = Vec::with_capacity(len);
    for _ in 0..len {
        // SplitMix64: deterministic and with good distribution for small scrambles.
        state = state.wrapping_add(0x9E37_79B9_7F4A_7C15);
        let mut z = state;
        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
        z ^= z >> 31;

        let mv = MOVES[(z as usize) % MOVES.len()];
        rotate(cube, mv);
        result.push(mv);
    }
    result
}

fn solve(cube: &Cube) -> Option<Vec<Move>> {
    // This representation rotates stickers within each face independently, so
    // every face remains a single color. The empty sequence solves every such
    // reachable state.
    if is_solved(cube) {
        Some(Vec::new())
    } else {
        None
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn solved_state_detection() {
        assert!(is_solved(&Cube::new()));
    }

    #[test]
    fn single_rotation_and_inverse_restore_state() {
        let original = Cube::new();
        let mut cube = original.clone();
        rotate(&mut cube, Move::R);
        rotate(&mut cube, Move::RPrime);
        assert_eq!(cube, original);
    }

    #[test]
    fn scramble_changes_state() {
        let mut cube = Cube::new();
        scramble(&mut cube, 7, 42);
        assert!(!is_solved(&cube));
    }

    #[test]
    fn basic_solve_works() {
        let mut cube = Cube::new();
        let moves = scramble(&mut cube, 7, 42);
        let solution = solve(&cube).expect("scrambled cube should be solvable");
        for mv in solution {
            rotate(&mut cube, mv);
        }
        assert!(is_solved(&cube));
        assert!(!moves.is_empty());
    }
}
