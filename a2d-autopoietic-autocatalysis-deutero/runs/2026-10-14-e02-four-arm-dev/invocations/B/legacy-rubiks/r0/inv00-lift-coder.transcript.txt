#[derive(Clone, Debug, PartialEq, Eq)]
struct Cube {
    // Face order: U, D, L, R, F, B. Each face is row-major.
    faces: [[u8; 9]; 6],
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Move {
    U,
    UPrime,
    D,
    DPrime,
    L,
    LPrime,
    R,
    RPrime,
    F,
    FPrime,
    B,
    BPrime,
}

impl Cube {
    fn new() -> Self {
        let mut faces = [[0; 9]; 6];
        for (face, cells) in faces.iter_mut().enumerate() {
            cells.fill(face as u8);
        }
        Self { faces }
    }
}

fn clockwise(face: &mut [u8; 9]) {
    let old = *face;
    for row in 0..3 {
        for col in 0..3 {
            face[col * 3 + (2 - row)] = old[row * 3 + col];
        }
    }
}

fn turn_face(cube: &mut Cube, face: usize, clockwise_turn: bool) {
    if clockwise_turn {
        clockwise(&mut cube.faces[face]);
    } else {
        for _ in 0..3 {
            clockwise(&mut cube.faces[face]);
        }
    }
}

// Convert a sticker position and outward normal to a 3D surface coordinate.
// Coordinates are in {-1, 0, 1}; axes are x right, y up, z front.
fn sticker_position(face: usize, row: i32, col: i32) -> ([i32; 3], [i32; 3]) {
    match face {
        0 => ([col - 1, 1, row - 1], [0, 1, 0]),       // U
        1 => ([col - 1, -1, 1 - row], [0, -1, 0]),    // D
        2 => ([-1, 1 - row, col - 1], [-1, 0, 0]),     // L
        3 => ([1, 1 - row, 1 - col], [1, 0, 0]),      // R
        4 => ([col - 1, 1 - row, 1], [0, 0, 1]),      // F
        _ => ([1 - col, 1 - row, -1], [0, 0, -1]),    // B
    }
}

fn face_row_col(normal: [i32; 3], pos: [i32; 3]) -> (usize, usize, usize) {
    match normal {
        [0, 1, 0] => (0, pos[2] + 1, pos[0] + 1),
        [0, -1, 0] => (1, 1 - pos[2], pos[0] + 1),
        [-1, 0, 0] => (2, 1 - pos[1], pos[2] + 1),
        [1, 0, 0] => (3, 1 - pos[1], 1 - pos[2]),
        [0, 0, 1] => (4, 1 - pos[1], pos[0] + 1),
        _ => (5, 1 - pos[1], 1 - pos[0]),
    }
}

fn rotate_vector(v: [i32; 3], axis: usize, clockwise_turn: bool) -> [i32; 3] {
    // Clockwise as viewed from outside the turned face.
    let sign = if clockwise_turn { -1 } else { 1 };
    match axis {
        0 => [v[0], -sign * v[2], sign * v[1]],
        1 => [sign * v[2], v[1], -sign * v[0]],
        _ => [-sign * v[1], sign * v[0], v[2]],
    }
}

fn rotate(cube: &mut Cube, mv: Move) {
    let (face, clockwise_turn, axis, layer, layer_sign) = match mv {
        Move::U => (0, true, 1, 1, 1),
        Move::UPrime => (0, false, 1, 1, 1),
        Move::D => (1, true, 1, -1, -1),
        Move::DPrime => (1, false, 1, -1, -1),
        Move::L => (2, true, 0, -1, -1),
        Move::LPrime => (2, false, 0, -1, -1),
        Move::R => (3, true, 0, 1, 1),
        Move::RPrime => (3, false, 0, 1, 1),
        Move::F => (4, true, 2, 1, 1),
        Move::FPrime => (4, false, 2, 1, 1),
        Move::B => (5, true, 2, -1, -1),
        Move::BPrime => (5, false, 2, -1, -1),
    };

    let mut old = cube.clone();
    turn_face(&mut old, face, clockwise_turn);

    for source_face in 0..6 {
        for row in 0..3 {
            for col in 0..3 {
                let (pos, normal) = sticker_position(source_face, row, col);
                if pos[axis] * layer_sign != layer {
                    continue;
                }

                let signed_turn = clockwise_turn;
                let new_pos = rotate_vector(pos, axis, signed_turn);
                let new_normal = rotate_vector(normal, axis, signed_turn);
                let (dest_face, dest_row, dest_col) = face_row_col(new_normal, new_pos);

                // The turned face was already rotated as a whole above.
                if source_face == face {
                    continue;
                }

                cube.faces[dest_face][dest_row * 3 + dest_col] =
                    old.faces[source_face][row * 3 + col];
            }
        }
    }
}

fn is_solved(cube: &Cube) -> bool {
    cube.faces
        .iter()
        .all(|face| face.iter().all(|&cell| cell == face[4]))
}

fn scramble(cube: &mut Cube, len: usize, seed: u64) -> Vec<Move> {
    const MOVES: [Move; 12] = [
        Move::U,
        Move::UPrime,
        Move::D,
        Move::DPrime,
        Move::L,
        Move::LPrime,
        Move::R,
        Move::RPrime,
        Move::F,
        Move::FPrime,
        Move::B,
        Move::BPrime,
    ];

    let mut state = seed;
    let mut result = Vec::with_capacity(len);
    for _ in 0..len {
        // SplitMix64: deterministic across platforms and Rust versions.
        state = state.wrapping_add(0x9E37_79B9_7F4A_7C15);
        let mut z = state;
        z = (z ^ (z >> 30)).wrapping_mul(0xBF58_476D_1CE4_E5B9);
        z = (z ^ (z >> 27)).wrapping_mul(0x94D0_49BB_1331_11EB);
        z ^= z >> 31;
        let mv = MOVES[(z as usize) % MOVES.len()];
        rotate(cube, mv);
        result.push(mv);
    }
    result
}

fn solve(cube: &Cube) -> Option<Vec<Move>> {
    // Bidirectional breadth-first search is practical for states a few moves
    // from solved. The bound keeps memory use finite for arbitrary input.
    use std::collections::{HashMap, VecDeque};

    if is_solved(cube) {
        return Some(Vec::new());
    }

    const MOVES: [Move; 12] = [
        Move::U,
        Move::UPrime,
        Move::D,
        Move::DPrime,
        Move::L,
        Move::LPrime,
        Move::R,
        Move::RPrime,
        Move::F,
        Move::FPrime,
        Move::B,
        Move::BPrime,
    ];
    const DEPTH: usize = 6;

    fn inverse(mv: Move) -> Move {
        match mv {
            Move::U => Move::UPrime,
            Move::UPrime => Move::U,
            Move::D => Move::DPrime,
            Move::DPrime => Move::D,
            Move::L => Move::LPrime,
            Move::LPrime => Move::L,
            Move::R => Move::RPrime,
            Move::RPrime => Move::R,
            Move::F => Move::FPrime,
            Move::FPrime => Move::F,
            Move::B => Move::BPrime,
            Move::BPrime => Move::B,
        }
    }

    fn expand(
        start: &Cube,
        depth: usize,
    ) -> HashMap<Cube, Vec<Move>> {
        let mut seen = HashMap::new();
        let mut queue = VecDeque::new();
        seen.insert(start.clone(), Vec::new());
        queue.push_back(start.clone());

        while let Some(state) = queue.pop_front() {
            let path = seen[&state].clone();
            if path.len() == depth {
                continue;
            }
            for mv in MOVES {
                let mut next = state.clone();
                rotate(&mut next, mv);
                if !seen.contains_key(&next) {
                    let mut next_path = path.clone();
                    next_path.push(mv);
                    seen.insert(next.clone(), next_path);
                    queue.push_back(next);
                }
            }
        }
        seen
    }

    let forward = expand(cube, DEPTH);
    let backward = expand(&Cube::new(), DEPTH);

    for (state, path_from_cube) in &forward {
        if let Some(path_from_solved) = backward.get(state) {
            let mut answer = path_from_cube.clone();
            answer.extend(path_from_solved.iter().rev().copied().map(inverse));
            return Some(answer);
        }
    }
    None
}
