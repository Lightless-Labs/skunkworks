#[derive(Clone, Debug, PartialEq, Eq)]
struct Cube {
    // Face order: U, D, L, R, F, B.
    faces: [[u8; 9]; 6],
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Move {
    U,
    UPrime,
    D,
    DPrime,
    L,
    LPrime,
    R,
    RPrime,
    F,
    FPrime,
    B,
    BPrime,
}

impl Cube {
    fn new() -> Self {
        Self {
            faces: [
                [b'U'; 9],
                [b'D'; 9],
                [b'L'; 9],
                [b'R'; 9],
                [b'F'; 9],
                [b'B'; 9],
            ],
        }
    }
}

fn rotate(cube: &mut Cube, mv: Move) {
    let (face, clockwise) = match mv {
        Move::U => (0, true),
        Move::UPrime => (0, false),
        Move::D => (1, true),
        Move::DPrime => (1, false),
        Move::L => (2, true),
        Move::LPrime => (2, false),
        Move::R => (3, true),
        Move::RPrime => (3, false),
        Move::F => (4, true),
        Move::FPrime => (4, false),
        Move::B => (5, true),
        Move::BPrime => (5, false),
    };

    let old = cube.faces;
    let mut next = old;

    // Rotate the selected face in the face's own view.
    for row in 0..3 {
        for col in 0..3 {
            let (new_row, new_col) = if clockwise {
                (col, 2 - row)
            } else {
                (2 - col, row)
            };
            next[face][new_row * 3 + new_col] = old[face][row * 3 + col];
        }
    }

    // Each entry is a directed strip around the turned face. The listed
    // indices run in the same physical direction around the face.
    let (strips, reverse) = match face {
        0 => ([[(4, 0), (4, 1), (4, 2)], [(3, 0), (3, 1), (3, 2)], [(5, 0), (5, 1), (5, 2)], [(2, 0), (2, 1), (2, 2)]], false),
        1 => ([[(4, 6), (4, 7), (4, 8)], [(2, 6), (2, 7), (2, 8)], [(5, 6), (5, 7), (5, 8)], [(3, 6), (3, 7), (3, 8)]], false),
        2 => ([[(0, 0), (0, 3), (0, 6)], [(4, 0), (4, 3), (4, 6)], [(1, 0), (1, 3), (1, 6)], [(5, 8), (5, 5), (5, 2)]], false),
        3 => ([[(0, 2), (0, 5), (0, 8)], [(5, 8), (5, 5), (5, 2)], [(1, 2), (1, 5), (1, 8)], [(4, 2), (4, 5), (4, 8)]], false),
        4 => ([[(0, 6), (0, 7), (0, 8)], [(3, 0), (3, 3), (3, 6)], [(1, 2), (1, 1), (1, 0)], [(2, 8), (2, 5), (2, 2)]], false),
        _ => ([[(0, 2), (0, 1), (0, 0)], [(2, 0), (2, 3), (2, 6)], [(1, 6), (1, 7), (1, 8)], [(3, 8), (3, 5), (3, 2)]], false),
    };
    let _ = reverse;

    // Clockwise maps each strip to the next one in this ordering.
    let steps = if clockwise { 1 } else { 3 };
    for strip in 0..4 {
        let source = (strip + 4 - steps) % 4;
        for i in 0..3 {
            let (dst_face, dst_idx) = strips[strip][i];
            let (src_face, src_idx) = strips[source][i];
            next[dst_face][dst_idx] = old[src_face][src_idx];
        }
    }

    cube.faces = next;
}

fn is_solved(cube: &Cube) -> bool {
    cube.faces.iter().all(|face| face.iter().all(|&cell| cell == face[0]))
}

fn scramble(cube: &mut Cube, len: usize, seed: u64) -> Vec<Move> {
    const MOVES: [Move; 12] = [
        Move::U, Move::UPrime, Move::D, Move::DPrime,
        Move::L, Move::LPrime, Move::R, Move::RPrime,
        Move::F, Move::FPrime, Move::B, Move::BPrime,
    ];

    let mut state = seed;
    let mut result = Vec::with_capacity(len);
    for _ in 0..len {
        // xorshift64*, with a nonzero fallback for seed zero.
        if state == 0 {
            state = 0x9E37_79B9_7F4A_7C15;
        }
        state ^= state >> 12;
        state ^= state << 25;
        state ^= state >> 27;
        let random = state.wrapping_mul(0x2545_F491_4F6C_DD1D);
        let mv = MOVES[(random as usize) % MOVES.len()];
        rotate(cube, mv);
        result.push(mv);
    }
    result
}

fn solve(cube: &Cube) -> Option<Vec<Move>> {
    // Reverse any path from the solved state to this state. A breadth-first
    // search is practical for shallow scrambles and returns None if its
    // bounded search does not find the state.
    const MOVES: [Move; 12] = [
        Move::U, Move::UPrime, Move::D, Move::DPrime,
        Move::L, Move::LPrime, Move::R, Move::RPrime,
        Move::F, Move::FPrime, Move::B, Move::BPrime,
    ];
    const MAX_DEPTH: usize = 7;

    if is_solved(cube) {
        return Some(Vec::new());
    }

    use std::collections::{HashSet, VecDeque};
    let start = Cube::new();
    let mut queue = VecDeque::new();
    let mut seen = HashSet::new();
    queue.push_back((start.clone(), Vec::<Move>::new()));
    seen.insert(start.faces);

    while let Some((current, path)) = queue.pop_front() {
        if path.len() >= MAX_DEPTH {
            continue;
        }
        for &mv in &MOVES {
            let mut next = current.clone();
            rotate(&mut next, mv);
            if !seen.insert(next.faces) {
                continue;
            }
            let mut next_path = path.clone();
            next_path.push(mv);
            if &next == cube {
                let solution = next_path
                    .into_iter()
                    .rev()
                    .map(inverse_move)
                    .collect();
                return Some(solution);
            }
            queue.push_back((next, next_path));
        }
    }
    None
}

fn inverse_move(mv: Move) -> Move {
    match mv {
        Move::U => Move::UPrime,
        Move::UPrime => Move::U,
        Move::D => Move::DPrime,
        Move::DPrime => Move::D,
        Move::L => Move::LPrime,
        Move::LPrime => Move::L,
        Move::R => Move::RPrime,
        Move::RPrime => Move::R,
        Move::F => Move::FPrime,
        Move::FPrime => Move::F,
        Move::B => Move::BPrime,
        Move::BPrime => Move::B,
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn solved_state_detection() {
        assert!(is_solved(&Cube::new()));
    }

    #[test]
    fn single_rotation_and_inverse_restore_state() {
        let original = Cube::new();
        let mut cube = original.clone();
        rotate(&mut cube, Move::R);
        assert_ne!(cube, original);
        rotate(&mut cube, Move::RPrime);
        assert_eq!(cube, original);
    }

    #[test]
    fn scramble_changes_state() {
        let mut cube = Cube::new();
        scramble(&mut cube, 8, 1234);
        assert!(!is_solved(&cube));
    }

    #[test]
    fn basic_solve_works() {
        let mut cube = Cube::new();
        let moves = scramble(&mut cube, 3, 42);
        let solution = solve(&cube).expect("shallow scramble should be solvable");
        for mv in solution {
            rotate(&mut cube, mv);
        }
        assert!(is_solved(&cube));
        assert_eq!(moves.len(), 3);
    }
}
