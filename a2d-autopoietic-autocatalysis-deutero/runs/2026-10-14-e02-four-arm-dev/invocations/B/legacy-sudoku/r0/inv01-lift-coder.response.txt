fn parse(s: &str) -> [[u8; 9]; 9] {
    let mut grid = [[0u8; 9]; 9];
    for (i, ch) in s.chars().take(81).enumerate() {
        grid[i / 9][i % 9] = ch.to_digit(10).filter(|&d| d <= 9).unwrap_or(0) as u8;
    }
    grid
}

fn validate(grid: &[[u8; 9]; 9]) -> bool {
    for row in 0..9 {
        let mut seen = [false; 10];
        for col in 0..9 {
            let value = grid[row][col] as usize;
            if !(1..=9).contains(&value) || seen[value] {
                return false;
            }
            seen[value] = true;
        }
    }

    for col in 0..9 {
        let mut seen = [false; 10];
        for row in 0..9 {
            let value = grid[row][col] as usize;
            if seen[value] {
                return false;
            }
            seen[value] = true;
        }
    }

    for box_row in 0..3 {
        for box_col in 0..3 {
            let mut seen = [false; 10];
            for row in 0..3 {
                for col in 0..3 {
                    let value = grid[box_row * 3 + row][box_col * 3 + col] as usize;
                    if seen[value] {
                        return false;
                    }
                    seen[value] = true;
                }
            }
        }
    }

    true
}

fn solve(mut grid: [[u8; 9]; 9]) -> Option<[[u8; 9]; 9]> {
    if !givens_valid(&grid) {
        return None;
    }
    if search(&mut grid) {
        Some(grid)
    } else {
        None
    }
}

fn givens_valid(grid: &[[u8; 9]; 9]) -> bool {
    for row in 0..9 {
        for col in 0..9 {
            let value = grid[row][col];
            if value > 9 {
                return false;
            }
            if value != 0 {
                for other_col in 0..9 {
                    if other_col != col && grid[row][other_col] == value {
                        return false;
                    }
                }
                for other_row in 0..9 {
                    if other_row != row && grid[other_row][col] == value {
                        return false;
                    }
                }
                let start_row = row / 3 * 3;
                let start_col = col / 3 * 3;
                for other_row in start_row..start_row + 3 {
                    for other_col in start_col..start_col + 3 {
                        if (other_row != row || other_col != col)
                            && grid[other_row][other_col] == value
                        {
                            return false;
                        }
                    }
                }
            }
        }
    }
    true
}

fn search(grid: &mut [[u8; 9]; 9]) -> bool {
    let mut best: Option<(usize, usize, [u8; 9], usize)> = None;

    for row in 0..9 {
        for col in 0..9 {
            if grid[row][col] == 0 {
                let mut candidates = [0u8; 9];
                let mut count = 0;

                for value in 1..=9 {
                    if allowed(grid, row, col, value) {
                        candidates[count] = value;
                        count += 1;
                    }
                }

                if count == 0 {
                    return false;
                }
                if best.as_ref().is_none_or(|(_, _, _, best_count)| count < *best_count) {
                    best = Some((row, col, candidates, count));
                    if count == 1 {
                        break;
                    }
                }
            }
        }
        if best.as_ref().is_some_and(|(_, _, _, count)| *count == 1) {
            break;
        }
    }

    let Some((row, col, candidates, count)) = best else {
        return true;
    };

    for &value in &candidates[..count] {
        grid[row][col] = value;
        if search(grid) {
            return true;
        }
        grid[row][col] = 0;
    }
    false
}

fn allowed(grid: &[[u8; 9]; 9], row: usize, col: usize, value: u8) -> bool {
    for i in 0..9 {
        if grid[row][i] == value || grid[i][col] == value {
            return false;
        }
    }

    let start_row = row / 3 * 3;
    let start_col = col / 3 * 3;
    for r in start_row..start_row + 3 {
        for c in start_col..start_col + 3 {
            if grid[r][c] == value {
                return false;
            }
        }
    }
    true
}

fn main() {}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn parse_reads_digits_row_major() {
        let grid = parse("530070000600195000098000060800060003400803001700020006060000280000419005000080079");
        assert_eq!(grid[0][0], 5);
        assert_eq!(grid[0][1], 3);
        assert_eq!(grid[0][2], 0);
        assert_eq!(grid[1][3], 1);
        assert_eq!(grid[8][8], 9);
    }

    #[test]
    fn validate_accepts_complete_solution_and_rejects_duplicate() {
        let mut grid = parse("534678912672195348198342567859761423426853791713924856961537284287419635345286179");
        assert!(validate(&grid));
        grid[0][0] = grid[0][3];
        assert!(!validate(&grid));
    }

    #[test]
    fn solves_a_relabelled_puzzle() {
        let solved = solve(parse("570030000400915000012000040200040007600207009300080004040000820000691005000020031")).expect("puzzle is solvable");
        let digits: String = solved
            .iter()
            .flat_map(|row| row.iter())
            .map(|&d| char::from_digit(d as u32, 10).unwrap())
            .collect();
        assert_eq!(digits, "576432198438915762912768543251349687684257319397186254149573826823691475765824931");
    }
}
