fn parse(s: &str) -> [[u8; 9]; 9] {
    let mut grid = [[0u8; 9]; 9];
    for (i, ch) in s.chars().take(81).enumerate() {
        grid[i / 9][i % 9] = ch.to_digit(10).unwrap_or(0) as u8;
    }
    grid
}

fn validate(grid: &[[u8; 9]; 9]) -> bool {
    for i in 0..9 {
        let mut row_seen = [false; 10];
        let mut col_seen = [false; 10];

        for j in 0..9 {
            let row_value = grid[i][j] as usize;
            let col_value = grid[j][i] as usize;
            if !(1..=9).contains(&row_value)
                || !(1..=9).contains(&col_value)
                || row_seen[row_value]
                || col_seen[col_value]
            {
                return false;
            }
            row_seen[row_value] = true;
            col_seen[col_value] = true;
        }
    }

    for box_row in (0..9).step_by(3) {
        for box_col in (0..9).step_by(3) {
            let mut seen = [false; 10];
            for row in box_row..box_row + 3 {
                for col in box_col..box_col + 3 {
                    let value = grid[row][col] as usize;
                    if !(1..=9).contains(&value) || seen[value] {
                        return false;
                    }
                    seen[value] = true;
                }
            }
        }
    }

    true
}

fn solve(mut grid: [[u8; 9]; 9]) -> Option<[[u8; 9]; 9]> {
    if !givens_are_valid(&grid) {
        return None;
    }

    if solve_backtracking(&mut grid) {
        Some(grid)
    } else {
        None
    }
}

fn givens_are_valid(grid: &[[u8; 9]; 9]) -> bool {
    for row in 0..9 {
        for col in 0..9 {
            let value = grid[row][col];
            if value > 9 {
                return false;
            }
            if value == 0 {
                continue;
            }

            for other_col in 0..9 {
                if other_col != col && grid[row][other_col] == value {
                    return false;
                }
            }
            for other_row in 0..9 {
                if other_row != row && grid[other_row][col] == value {
                    return false;
                }
            }

            let box_row = (row / 3) * 3;
            let box_col = (col / 3) * 3;
            for other_row in box_row..box_row + 3 {
                for other_col in box_col..box_col + 3 {
                    if (other_row != row || other_col != col)
                        && grid[other_row][other_col] == value
                    {
                        return false;
                    }
                }
            }
        }
    }
    true
}

fn solve_backtracking(grid: &mut [[u8; 9]; 9]) -> bool {
    let mut best: Option<(usize, usize, Vec<u8>)> = None;

    for row in 0..9 {
        for col in 0..9 {
            if grid[row][col] != 0 {
                continue;
            }

            let candidates: Vec<u8> = (1..=9)
                .filter(|&value| can_place(grid, row, col, value))
                .collect();

            if candidates.is_empty() {
                return false;
            }

            if best
                .as_ref()
                .is_none_or(|(_, _, current)| candidates.len() < current.len())
            {
                best = Some((row, col, candidates));
                if best.as_ref().unwrap().2.len() == 1 {
                    break;
                }
            }
        }
        if best.as_ref().is_some_and(|(_, _, candidates)| candidates.len() == 1) {
            break;
        }
    }

    let Some((row, col, candidates)) = best else {
        return true;
    };

    for value in candidates {
        grid[row][col] = value;
        if solve_backtracking(grid) {
            return true;
        }
        grid[row][col] = 0;
    }

    false
}

fn can_place(grid: &[[u8; 9]; 9], row: usize, col: usize, value: u8) -> bool {
    for index in 0..9 {
        if grid[row][index] == value || grid[index][col] == value {
            return false;
        }
    }

    let box_row = (row / 3) * 3;
    let box_col = (col / 3) * 3;
    for r in box_row..box_row + 3 {
        for c in box_col..box_col + 3 {
            if grid[r][c] == value {
                return false;
            }
        }
    }

    true
}
