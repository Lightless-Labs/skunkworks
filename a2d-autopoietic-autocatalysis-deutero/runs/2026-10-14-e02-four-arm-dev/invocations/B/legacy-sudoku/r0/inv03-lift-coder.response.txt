fn parse(s: &str) -> [[u8; 9]; 9] {
    let mut grid = [[0u8; 9]; 9];
    let mut cells = s.chars().filter(|c| !c.is_whitespace());

    for row in &mut grid {
        for cell in row {
            *cell = match cells.next() {
                Some(c) => c.to_digit(10).filter(|&d| d <= 9).unwrap_or(0) as u8,
                None => 0,
            };
        }
    }

    grid
}

fn validate(grid: &[[u8; 9]; 9]) -> bool {
    for i in 0..9 {
        let mut row_seen = [false; 10];
        let mut col_seen = [false; 10];

        for j in 0..9 {
            let row_value = grid[i][j] as usize;
            let col_value = grid[j][i] as usize;

            if !(1..=9).contains(&row_value)
                || !(1..=9).contains(&col_value)
                || row_seen[row_value]
                || col_seen[col_value]
            {
                return false;
            }

            row_seen[row_value] = true;
            col_seen[col_value] = true;
        }
    }

    for box_row in (0..9).step_by(3) {
        for box_col in (0..9).step_by(3) {
            let mut seen = [false; 10];
            for row in box_row..box_row + 3 {
                for col in box_col..box_col + 3 {
                    let value = grid[row][col] as usize;
                    if !(1..=9).contains(&value) || seen[value] {
                        return false;
                    }
                    seen[value] = true;
                }
            }
        }
    }

    true
}

fn solve(mut grid: [[u8; 9]; 9]) -> Option<[[u8; 9]; 9]> {
    // Reject out-of-range values and conflicting clues before searching.
    for row in 0..9 {
        for col in 0..9 {
            let value = grid[row][col];
            if value > 9 {
                return None;
            }
            if value != 0 {
                grid[row][col] = 0;
                let valid_placement = can_place(&grid, row, col, value);
                grid[row][col] = value;
                if !valid_placement {
                    return None;
                }
            }
        }
    }

    if search(&mut grid) {
        Some(grid)
    } else {
        None
    }
}

fn can_place(grid: &[[u8; 9]; 9], row: usize, col: usize, value: u8) -> bool {
    for i in 0..9 {
        if grid[row][i] == value || grid[i][col] == value {
            return false;
        }
    }

    let box_row = (row / 3) * 3;
    let box_col = (col / 3) * 3;
    for r in box_row..box_row + 3 {
        for c in box_col..box_col + 3 {
            if grid[r][c] == value {
                return false;
            }
        }
    }

    true
}

fn search(grid: &mut [[u8; 9]; 9]) -> bool {
    // Choose the empty cell with the fewest legal candidates.
    let mut best: Option<(usize, usize, Vec<u8>)> = None;

    for row in 0..9 {
        for col in 0..9 {
            if grid[row][col] != 0 {
                continue;
            }

            let candidates: Vec<u8> = (1..=9)
                .filter(|&value| can_place(grid, row, col, value))
                .collect();

            if candidates.is_empty() {
                return false;
            }

            if best
                .as_ref()
                .is_none_or(|(_, _, current)| candidates.len() < current.len())
            {
                best = Some((row, col, candidates));
                if best.as_ref().unwrap().2.len() == 1 {
                    break;
                }
            }
        }

        if best.as_ref().is_some_and(|(_, _, candidates)| candidates.len() == 1) {
            break;
        }
    }

    let Some((row, col, candidates)) = best else {
        return true;
    };

    for value in candidates {
        grid[row][col] = value;
        if search(grid) {
            return true;
        }
        grid[row][col] = 0;
    }

    false
}
