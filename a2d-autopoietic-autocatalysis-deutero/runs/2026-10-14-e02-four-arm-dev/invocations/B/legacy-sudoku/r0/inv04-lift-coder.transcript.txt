pub fn parse(s: &str) -> [[u8; 9]; 9] {
    let mut grid = [[0u8; 9]; 9];
    for (i, ch) in s.chars().take(81).enumerate() {
        let value = ch.to_digit(10).unwrap_or(0) as u8;
        grid[i / 9][i % 9] = value;
    }
    grid
}

pub fn solve(mut grid: [[u8; 9]; 9]) -> Option<[[u8; 9]; 9]> {
    if grid.iter().flatten().any(|&value| value > 9) || !givens_are_valid(&grid) {
        return None;
    }

    if solve_recursive(&mut grid) {
        Some(grid)
    } else {
        None
    }
}

fn givens_are_valid(grid: &[[u8; 9]; 9]) -> bool {
    for row in 0..9 {
        for col in 0..9 {
            let value = grid[row][col];
            if value == 0 {
                continue;
            }

            for other_col in 0..9 {
                if other_col != col && grid[row][other_col] == value {
                    return false;
                }
            }
            for other_row in 0..9 {
                if other_row != row && grid[other_row][col] == value {
                    return false;
                }
            }

            let box_row = (row / 3) * 3;
            let box_col = (col / 3) * 3;
            for other_row in box_row..box_row + 3 {
                for other_col in box_col..box_col + 3 {
                    if (other_row != row || other_col != col)
                        && grid[other_row][other_col] == value
                    {
                        return false;
                    }
                }
            }
        }
    }
    true
}

fn solve_recursive(grid: &mut [[u8; 9]; 9]) -> bool {
    let mut best_cell = None;
    let mut best_candidates = 0u16;

    for row in 0..9 {
        for col in 0..9 {
            if grid[row][col] != 0 {
                continue;
            }

            let candidates = candidate_mask(grid, row, col);
            let count = candidates.count_ones();
            if count == 0 {
                return false;
            }
            if best_cell.is_none() || count < best_candidates.count_ones() {
                best_cell = Some((row, col));
                best_candidates = candidates;
                if count == 1 {
                    break;
                }
            }
        }
        if best_candidates.count_ones() == 1 {
            break;
        }
    }

    let Some((row, col)) = best_cell else {
        return validate(grid);
    };

    for value in 1..=9 {
        if best_candidates & (1 << value) != 0 {
            grid[row][col] = value as u8;
            if solve_recursive(grid) {
                return true;
            }
            grid[row][col] = 0;
        }
    }

    false
}

fn candidate_mask(grid: &[[u8; 9]; 9], row: usize, col: usize) -> u16 {
    let mut used = 0u16;

    for i in 0..9 {
        let row_value = grid[row][i];
        let col_value = grid[i][col];
        if row_value != 0 {
            used |= 1 << row_value;
        }
        if col_value != 0 {
            used |= 1 << col_value;
        }
    }

    let box_row = (row / 3) * 3;
    let box_col = (col / 3) * 3;
    for r in box_row..box_row + 3 {
        for c in box_col..box_col + 3 {
            let value = grid[r][c];
            if value != 0 {
                used |= 1 << value;
            }
        }
    }

    let all_digits = ((1u16 << 10) - 1) & !1;
    all_digits & !used
}
