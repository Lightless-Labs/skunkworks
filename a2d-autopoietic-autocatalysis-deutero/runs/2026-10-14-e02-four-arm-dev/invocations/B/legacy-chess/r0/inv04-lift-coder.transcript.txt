pub fn apply_move(board: &Board, mv: &Move) -> Board {
    let mut next = board.clone();
    if mv.from_rank >= 8 || mv.from_file >= 8 || mv.to_rank >= 8 || mv.to_file >= 8 {
        return next;
    }

    let Some(mut piece) = next.squares[mv.from_rank][mv.from_file] else {
        return next;
    };

    let moving_side = piece.white;
    let captured = next.squares[mv.to_rank][mv.to_file];

    if piece.kind == PieceKind::King {
        let side = usize::from(!piece.white);
        next.castling_rights[side] = [false, false];
    }

    if piece.kind == PieceKind::Rook {
        update_rook_rights(&mut next, piece.white, mv.from_rank, mv.from_file);
    }
    if let Some(captured_piece) = captured {
        if captured_piece.kind == PieceKind::Rook {
            update_rook_rights(&mut next, captured_piece.white, mv.to_rank, mv.to_file);
        }
    }

    next.en_passant = None;
    next.squares[mv.from_rank][mv.from_file] = None;

    if mv.is_en_passant && piece.kind == PieceKind::Pawn {
        let captured_rank = if piece.white {
            mv.to_rank.saturating_add(1)
        } else {
            mv.to_rank.saturating_sub(1)
        };
        if captured_rank < 8 {
            next.squares[captured_rank][mv.to_file] = None;
        }
    }

    if mv.is_castling && piece.kind == PieceKind::King {
        let (rook_from, rook_to) = if mv.to_file > mv.from_file {
            (7, 5)
        } else {
            (0, 3)
        };
        next.squares[mv.to_rank][rook_to] = next.squares[mv.to_rank][rook_from];
        next.squares[mv.to_rank][rook_from] = None;
    }

    if piece.kind == PieceKind::Pawn {
        if (mv.from_rank as isize - mv.to_rank as isize).unsigned_abs() == 2 {
            next.en_passant = Some(((mv.from_rank + mv.to_rank) / 2, mv.from_file));
        }
        if let Some(kind) = mv.promotion {
            piece.kind = kind;
        } else if mv.to_rank == 0 || mv.to_rank == 7 {
            piece.kind = PieceKind::Queen;
        }
    }

    next.squares[mv.to_rank][mv.to_file] = Some(piece);
    next.white_to_move = !moving_side;
    next
}

fn update_rook_rights(board: &mut Board, white: bool, rank: usize, file: usize) {
    let home_rank = if white { 7 } else { 0 };
    if rank != home_rank {
        return;
    }
    let side = usize::from(!white);
    if file == 7 {
        board.castling_rights[side][0] = false;
    } else if file == 0 {
        board.castling_rights[side][1] = false;
    }
}

pub fn generate_moves(board: &Board, white: bool) -> Vec<Move> {
    pseudo_legal_moves(board, white)
        .into_iter()
        .filter(|mv| {
            let next = apply_move(board, mv);
            !is_check(&next, white)
        })
        .collect()
}

pub fn is_checkmate(board: &Board, white: bool) -> bool {
    is_check(board, white) && generate_moves(board, white).is_empty()
}

pub fn is_stalemate(board: &Board, white: bool) -> bool {
    !is_check(board, white) && generate_moves(board, white).is_empty()
}

pub fn evaluate_material(board: &Board) -> i32 {
    let mut score = 0;
    for rank in 0..8 {
        for file in 0..8 {
            if let Some(piece) = board.squares[rank][file] {
                let value = match piece.kind {
                    PieceKind::Pawn => 100,
                    PieceKind::Knight => 320,
                    PieceKind::Bishop => 330,
                    PieceKind::Rook => 500,
                    PieceKind::Queen => 900,
                    PieceKind::King => 0,
                };
                score += if piece.white { value } else { -value };
            }
        }
    }
    score
}
