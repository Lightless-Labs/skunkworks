#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut data = Vec::with_capacity(self.pending.len() + input.len());
        data.extend_from_slice(&self.pending);
        data.extend_from_slice(input);

        let complete_len = (data.len() / 5) * 5;
        let mut output = encode_bytes(&data[..complete_len], self.config.alphabet);
        self.pending = data[complete_len..].to_vec();
        Ok(std::mem::take(&mut output))
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = encode_partial(&self.pending, self.config.alphabet);
        if self.config.padding == Padding::Required {
            while output.len() % 8 != 0 {
                output.push('=');
            }
        }
        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

pub struct Decoder {
    config: Config,
    consumed: usize,
    data_count: usize,
    bit_buffer: u64,
    bit_count: u8,
    last_data_offset: Option<usize>,
    pad_start: Option<usize>,
    pad_count: usize,
    expected_pads: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            consumed: 0,
            data_count: 0,
            bit_buffer: 0,
            bit_count: 0,
            last_data_offset: None,
            pad_start: None,
            pad_count: 0,
            expected_pads: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.consumed;
            next.consumed = next.consumed.checked_add(1).unwrap_or(usize::MAX);

            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }
                if next.pad_start.is_none() {
                    let residue = next.data_count % 8;
                    let Some(expected) = expected_pad_count(residue) else {
                        return Err(StreamError::Padding { offset });
                    };
                    if residue == 0 {
                        return Err(StreamError::Padding { offset });
                    }
                    next.pad_start = Some(offset);
                    next.expected_pads = expected;
                }
                if next.pad_count >= next.expected_pads {
                    return Err(StreamError::Padding { offset });
                }
                next.pad_count += 1;
                continue;
            }

            let upper = byte.to_ascii_uppercase();
            let value = match symbol_value(next.config.alphabet, upper) {
                Some(v) => v,
                None => return Err(StreamError::InvalidByte { offset, byte }),
            };

            if byte != upper && next.config.canonical {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: upper,
                });
            }

            if next.pad_start.is_some() {
                return Err(StreamError::Padding { offset });
            }

            next.data_count = next.data_count.checked_add(1).unwrap_or(usize::MAX);
            next.last_data_offset = Some(offset);
            next.bit_buffer = (next.bit_buffer << 5) | u64::from(value);
            next.bit_count += 5;

            while next.bit_count >= 8 {
                next.bit_count -= 8;
                let byte_out = ((next.bit_buffer >> next.bit_count) & 0xff) as u8;
                output.push(byte_out);
                if next.bit_count == 0 {
                    next.bit_buffer = 0;
                } else {
                    next.bit_buffer &= (1u64 << next.bit_count) - 1;
                }
            }
        }

        self.assign_state(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let Some(expected) = expected_pad_count(residue) else {
            return Err(StreamError::InvalidLength {
                offset: self.consumed,
            });
        };

        if self.config.padding == Padding::Required && self.pad_count != expected {
            return Err(StreamError::Padding {
                offset: self.pad_start.unwrap_or(self.consumed),
            });
        }

        let unused = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused != 0 && self.bit_buffer & ((1u64 << unused) - 1) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.consumed),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }
}

impl Decoder {
    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            consumed: self.consumed,
            data_count: self.data_count,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            last_data_offset: self.last_data_offset,
            pad_start: self.pad_start,
            pad_count: self.pad_count,
            expected_pads: self.expected_pads,
            closed: self.closed,
        }
    }

    fn assign_state(&mut self, other: Self) {
        self.consumed = other.consumed;
        self.data_count = other.data_count;
        self.bit_buffer = other.bit_buffer;
        self.bit_count = other.bit_count;
        self.last_data_offset = other.last_data_offset;
        self.pad_start = other.pad_start;
        self.pad_count = other.pad_count;
        self.expected_pads = other.expected_pads;
        self.closed = other.closed;
    }
}

fn expected_pad_count(residue: usize) -> Option<usize> {
    match residue {
        0 => Some(0),
        2 => Some(6),
        4 => Some(4),
        5 => Some(3),
        7 => Some(1),
        _ => None,
    }
}

fn symbol_value(alphabet: Alphabet, byte: u8) -> Option<u8> {
    match alphabet {
        Alphabet::Standard => {
            const TABLE: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
            TABLE.iter().position(|&c| c == byte).map(|v| v as u8)
        }
        Alphabet::Hex => {
            const TABLE: &[u8; 32] = b"0123456789ABCDEFGHIJKLMNOPQRSTUV";
            TABLE.iter().position(|&c| c == byte).map(|v| v as u8)
        }
    }
}

fn alphabet_symbol(alphabet: Alphabet, value: u8) -> u8 {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"[value as usize],
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV"[value as usize],
    }
}

fn encode_bytes(bytes: &[u8], alphabet: Alphabet) -> String {
    let mut output = String::with_capacity(bytes.len() * 8 / 5);
    let mut buffer: u64 = 0;
    let mut bits: u8 = 0;

    for &byte in bytes {
        buffer = (buffer << 8) | u64::from(byte);
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            output.push(alphabet_symbol(
                alphabet,
                ((buffer >> bits) & 31) as u8,
            ) as char);
            if bits == 0 {
                buffer = 0;
            } else {
                buffer &= (1u64 << bits) - 1;
            }
        }
    }

    output
}

fn encode_partial(bytes: &[u8], alphabet: Alphabet) -> String {
    let mut output = encode_bytes(bytes, alphabet);
    if !bytes.is_empty() {
        let bits_from_bytes = bytes.len() * 8;
        let emitted_bits = (bits_from_bytes / 5) * 5;
        let remaining_bits = bits_from_bytes - emitted_bits;
        if remaining_bits > 0 {
            let mut buffer: u64 = 0;
            for &byte in bytes {
                buffer = (buffer << 8) | u64::from(byte);
            }
            let value = ((buffer & ((1u64 << remaining_bits) - 1))
                << (5 - remaining_bits)) as u8;
            output.push(alphabet_symbol(alphabet, value) as char);
        }
    }
    output
}
