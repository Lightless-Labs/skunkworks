#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 6] = [
    "thousand", "million", "billion", "trillion", "quadrillion", "quintillion",
];

fn append_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            out.push_str(" and ");
            append_group(remainder, out);
        }
    } else if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        let units = n % 10;
        if units != 0 {
            out.push('-');
            out.push_str(SMALL[units as usize]);
        }
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut remaining = magnitude;
    for group in groups.iter_mut().rev() {
        *group = (remaining % 1000) as u16;
        remaining /= 1000;
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    let mut emitted_higher = false;
    for (index, &group) in groups.iter().enumerate() {
        if group == 0 {
            continue;
        }

        let is_units = index == 6;
        if !result.is_empty() && !(is_units && group < 100 && emitted_higher) {
            result.push(' ');
        }
        if is_units && group < 100 && emitted_higher {
            result.push_str(" and ");
        }

        append_group(group, &mut result);
        if index < 6 {
            result.push(' ');
            result.push_str(SCALES[5 - index]);
            emitted_higher = true;
        }
    }

    Ok(result)
}

fn word_value(word: &str) -> Option<u128> {
    if word == "zero" {
        return Some(0);
    }
    for (i, &w) in SMALL.iter().enumerate().skip(1) {
        if word == w {
            return Some(i as u128);
        }
    }
    for (i, &w) in TENS.iter().enumerate() {
        if word == w {
            return Some(((i + 2) * 10) as u128);
        }
    }
    match word {
        "hundred" => Some(100),
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        "and" => Some(0),
        "minus" => Some(0),
        _ => None,
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }
    let bytes = input.as_bytes();

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (i, &(token, index)) in tokens.iter().enumerate() {
        if token == "minus" && i == 0 {
            continue;
        }
        if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index });
        }
    }

    let negative = tokens.first().is_some_and(|(token, _)| *token == "minus");
    let arithmetic = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for &(token, _) in arithmetic {
        if let Some(value) = SMALL.iter().position(|&w| w == token).filter(|&i| i > 0) {
            group += value as u128;
        } else if let Some(value) = TENS.iter().position(|&w| w == token) {
            group += ((value + 2) * 10) as u128;
        } else {
            match token {
                "hundred" => group = group.saturating_mul(100),
                "thousand" => {
                    total = total.saturating_add(group.saturating_mul(1_000));
                    group = 0;
                }
                "million" => {
                    total = total.saturating_add(group.saturating_mul(1_000_000));
                    group = 0;
                }
                "billion" => {
                    total = total.saturating_add(group.saturating_mul(1_000_000_000));
                    group = 0;
                }
                "trillion" => {
                    total = total.saturating_add(group.saturating_mul(1_000_000_000_000));
                    group = 0;
                }
                "quadrillion" => {
                    total = total.saturating_add(group.saturating_mul(1_000_000_000_000_000));
                    group = 0;
                }
                "quintillion" => {
                    total = total.saturating_add(group.saturating_mul(LIMIT));
                    group = 0;
                }
                "zero" | "and" => {}
                "minus" => unreachable!(),
                _ => unreachable!(),
            }
        }

        if total.saturating_add(group) > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(value)?;
    if canonical != input {
        let differing = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }
    Ok(value)
}
