#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn valid_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<u8>, Vec<usize>, Vec<Run>), EmailError> {
    let mut kept = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut first_retained = true;
    let mut in_quote = false;
    let mut run_start: Option<usize> = None;

    while i < input.len() {
        let b = input[i];

        if in_quote {
            kept.push(b);
            positions.push(i);
            i += 1;
            if b == b'\\' {
                if i < input.len() {
                    kept.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
            } else if b == b'"' {
                in_quote = false;
            }
            continue;
        }

        if b == b'"' && first_retained {
            if let Some(start) = run_start.take() {
                runs.push(Run { start, end: i });
            }
            kept.push(b);
            positions.push(i);
            i += 1;
            first_retained = false;
            in_quote = true;
            continue;
        }

        if is_hws(b) || b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
            }

            if is_hws(b) {
                i += 1;
                continue;
            }

            let outer = i;
            let mut depth = 1usize;
            i += 1;
            while depth > 0 {
                if i >= input.len() {
                    return Err(EmailError::UnclosedComment { at: outer });
                }
                let c = input[i];
                if c == b'\\' {
                    if i + 1 >= input.len()
                        || !(0x20..=0x7e).contains(&input[i + 1])
                    {
                        return Err(EmailError::CommentEscape { at: i });
                    }
                    i += 2;
                } else if c == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                } else if c == b')' {
                    depth -= 1;
                    i += 1;
                } else if c == b'\r' || c == b'\n' {
                    if let Some(end) = valid_fold(input, i) {
                        i = end;
                    } else {
                        return Err(EmailError::BadFold { at: i });
                    }
                } else if c == b'\t' || (0x20..=0x7e).contains(&c) {
                    i += 1;
                } else {
                    return Err(EmailError::CommentChar { at: i });
                }
            }
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if let Some(end) = valid_fold(input, i) {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                i = end;
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if let Some(start) = run_start.take() {
            runs.push(Run { start, end: i });
        }
        kept.push(b);
        positions.push(i);
        first_retained = false;
        i += 1;
    }

    if let Some(start) = run_start {
        runs.push(Run {
            start,
            end: input.len(),
        });
    }
    Ok((kept, positions, runs))
}

fn pos(positions: &[usize], index: usize, input_len: usize) -> usize {
    positions.get(index).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, positions, runs) = cleanup(bytes)?;

    for run in &runs {
        let left_index = positions.iter().rposition(|&p| p < run.start);
        let right_index = positions.iter().position(|&p| p >= run.end);

        let left_ok = left_index
            .is_none_or(|index| clean[index] == b'@');
        let right_ok = right_index
            .is_none_or(|index| clean[index] == b'@');

        if !left_ok && !right_ok {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if clean.first() == Some(&b'"') {
        let mut i = 1;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(&positions, 0, bytes.len()),
            });
        }
    }

    let protected_end = quote_end.map_or(0, |i| i + 1);
    let separators: Vec<usize> = (0..clean.len())
        .filter(|&i| clean[i] == b'@' && i >= protected_end)
        .collect();

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: pos(&positions, separators[1], bytes.len()),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&positions, sep, bytes.len()),
        });
    }

    let (local, quoted) = if let Some(end_quote) = quote_end {
        if end_quote + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&positions, end_quote + 1, bytes.len()),
            });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < end_quote {
            let b = clean[i];
            if b == b'\\' {
                if i + 1 >= end_quote {
                    return Err(EmailError::BadEscape {
                        at: pos(&positions, i, bytes.len()),
                    });
                }
                let next = clean[i + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: pos(&positions, i, bytes.len()),
                    });
                }
                decoded.push(next);
                origins.push(pos(&positions, i, bytes.len()));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: pos(&positions, i, bytes.len()),
                    });
                }
                decoded.push(b);
                origins.push(pos(&positions, i, bytes.len()));
                i += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(&positions, end_quote, bytes.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }

        (String::from_utf8(decoded).unwrap(), true)
    } else {
        let local_bytes = &clean[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(&positions, 64, bytes.len()),
            });
        }

        let specials = b"!#$%&'*+-/=?^_`{|}~";
        for i in 0..local_bytes.len() {
            let b = local_bytes[i];
            if b == b'.' {
                if i == 0
                    || i + 1 == local_bytes.len()
                    || local_bytes[i - 1] == b'.'
                {
                    return Err(EmailError::LocalDot {
                        at: pos(&positions, i, bytes.len()),
                    });
                }
            } else if !b.is_ascii_alphanumeric() && !specials.contains(&b) {
                return Err(EmailError::LocalChar {
                    at: pos(&positions, i, bytes.len()),
                });
            }
        }

        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let mut root_dot = None;
    if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(domain_end);
    }

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos(&positions, domain_start, bytes.len()),
        });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(&positions, domain_start + 253, bytes.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;

    for i in domain_start..=domain_end {
        if i == domain_end || clean[i] == b'.' {
            if start == i {
                let at = if i == domain_end {
                    root_dot
                        .map(|x| pos(&positions, x, bytes.len()))
                        .unwrap_or_else(|| pos(&positions, i, bytes.len()))
                } else {
                    pos(&positions, i, bytes.len())
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let len = i - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: pos(&positions, start + 63, bytes.len()),
                });
            }
            if clean[start] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: pos(&positions, start, bytes.len()),
                });
            }
            if clean[i - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: pos(&positions, i - 1, bytes.len()),
                });
            }
            for j in start..i {
                let b = clean[j];
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar {
                        at: pos(&positions, j, bytes.len()),
                    });
                }
            }

            labels.push((start, i));
            start = i + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&positions, domain_start, bytes.len()),
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: pos(&positions, last_start, bytes.len()),
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
