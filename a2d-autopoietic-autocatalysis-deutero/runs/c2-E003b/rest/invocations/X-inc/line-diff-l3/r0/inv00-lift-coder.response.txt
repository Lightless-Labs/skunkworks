#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
enum RowKind {
    Context,
    Remove,
    Add,
}

#[derive(Clone)]
struct Row {
    kind: RowKind,
    payload: String,
    line: usize,
}

struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.starts_with('0')) {
        return None;
    }
    if !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line_text: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line_text.strip_prefix("@@ -")?;
    let (old_start, rest) = rest.split_once(',')?;
    let (old_count, rest) = rest.split_once(" +")?;
    let (new_start, rest) = rest.split_once(',')?;
    let (new_count, suffix) = rest.split_once(" @@")?;
    if !suffix.is_empty() {
        return None;
    }

    let a = parse_number(old_start)?;
    let b = parse_number(old_count)?;
    let c = parse_number(new_start)?;
    let d = parse_number(new_count)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn physical_lines(text: &str) -> Vec<&str> {
    text.split_terminator('\n').collect()
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let lines = physical_lines(patch);
    for (idx, expected) in [(0, "--- before"), (1, "+++ after")] {
        if lines.get(idx).copied() != Some(expected) {
            return Err(ApplyError::BadHeader { line: idx + 1 });
        }
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    if i >= lines.len() || !lines[i].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut previous_old_end = 0usize;
    let mut previous_old_start = 0usize;
    let mut net: i128 = 0;
    let mut have_previous = false;

    while i < lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };
        i += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let text = lines[i];
            let Some((prefix, payload)) = text.split_at_checked(1) else {
                return Err(ApplyError::BadBody { line: row_line });
            };
            let kind = match prefix {
                " " => {
                    old_seen += 1;
                    new_seen += 1;
                    RowKind::Context
                }
                "-" => {
                    old_seen += 1;
                    RowKind::Remove
                }
                "+" => {
                    new_seen += 1;
                    RowKind::Add
                }
                _ => return Err(ApplyError::BadBody { line: row_line }),
            };
            if payload.contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            rows.push(Row {
                kind,
                payload: payload.to_owned(),
                line: row_line,
            });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| !matches!(r.kind, RowKind::Context)) {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if have_previous {
            if old_start < previous_old_end || old_start < previous_old_start {
                return Err(ApplyError::Misordered { line: header_line });
            }
        }
        let required_new = old_start as i128 + net;
        if required_new < 0 || new_start as i128 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        previous_old_start = old_start;
        previous_old_end = old_start.saturating_add(b);
        have_previous = true;
        net += d as i128 - b as i128;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });

        if hunks.len() == 17 {
            return Err(ApplyError::TooLarge);
        }
    }

    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source
        .strip_suffix('\n')
        .unwrap()
        .split('\n')
        .map(str::to_owned)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;

    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let leading = hunk
            .rows
            .iter()
            .take_while(|r| matches!(r.kind, RowKind::Context))
            .count();
        let trailing = hunk
            .rows
            .iter()
            .rev()
            .take_while(|r| matches!(r.kind, RowKind::Context))
            .count();

        let max_l = options.fuzz.min(leading);
        let max_r = options.fuzz.min(trailing);
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, u128)> = None;
        let mut best_starts = Vec::<usize>::new();
        let mut best_choice = None::<(usize, usize, usize)>;

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                let mut new_payloads = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len().saturating_sub(r)] {
                    match row.kind {
                        RowKind::Context => {
                            old_payloads.push(row.payload.as_str());
                            new_payloads.push(row.payload.as_str());
                        }
                        RowKind::Remove => old_payloads.push(row.payload.as_str()),
                        RowKind::Add => new_payloads.push(row.payload.as_str()),
                    }
                }

                let last_start = working.len().saturating_sub(hunk.old_count);
                if hunk.old_count > working.len() {
                    continue;
                }
                for s in 0..=last_start {
                    let distance = (s as i128 - expected).unsigned_abs();
                    if distance > options.max_offset as u128 {
                        continue;
                    }
                    let begin = s + l;
                    let end = s + hunk.old_count - r;
                    if end > working.len() || end < begin {
                        continue;
                    }
                    if old_payloads.len() != end - begin
                        || !old_payloads
                            .iter()
                            .zip(&working[begin..end])
                            .all(|(expected, actual)| *expected == actual)
                    {
                        continue;
                    }

                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(s);
                            best_choice = Some((s, l, r));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(s);
                            best_choice = Some((s, l, r));
                        }
                        Some(current) if rank == current => {
                            if !best_starts.contains(&s) {
                                best_starts.push(s);
                            }
                            if let Some((old_s, old_l, _)) = best_choice {
                                if s == old_s && l < old_l {
                                    best_choice = Some((s, l, r));
                                }
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_starts.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }
        if best_starts.len() > 1 {
            best_starts.sort_unstable();
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let (s, l, r) = best_choice.expect("a best candidate exists");
        let mut replacement = Vec::new();
        for row in &hunk.rows[l..hunk.rows.len().saturating_sub(r)] {
            match row.kind {
                RowKind::Context | RowKind::Add => replacement.push(row.payload.clone()),
                RowKind::Remove => {}
            }
        }

        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
