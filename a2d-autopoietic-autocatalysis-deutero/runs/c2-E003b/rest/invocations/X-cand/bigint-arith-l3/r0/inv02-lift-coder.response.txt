```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    // Little-endian base-10 digits; zero is represented as [0] with negative=false.
    digits: Vec<u8>,
    negative: bool,
}

impl Big {
    pub fn parse(text: &str, max_digits: usize) -> Result<Self, Error> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        // Invalid bytes take precedence over malformed separators.
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !byte.is_ascii_digit() {
                return Err(Error::InvalidDigit { index });
            }
        }

        // Validate separators against immediate magnitude bytes.
        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || !bytes[index - 1].is_ascii_digit()
                    || !bytes[index + 1].is_ascii_digit())
            {
                return Err(Error::Separator { index });
            }
        }

        let first_digit_index = start;
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();

        if digit_count > 1 && bytes[first_digit_index] == b'0' {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }

        if digit_count > max_digits {
            let excess_number = max_digits + 1;
            let mut seen = 0usize;
            for (index, &byte) in bytes.iter().enumerate().skip(start) {
                if byte.is_ascii_digit() {
                    seen += 1;
                    if seen == excess_number {
                        return Err(Error::TooManyDigits {
                            limit: max_digits,
                            index,
                        });
                    }
                }
            }
        }

        let mut digits = Vec::with_capacity(digit_count.max(1));
        for &byte in bytes[start..].iter().rev() {
            if byte != b'_' {
                digits.push(byte - b'0');
            }
        }
        if digits.is_empty() {
            digits.push(0);
        }

        let mut result = Self { digits, negative };
        result.normalize();
        Ok(result)
    }

    pub fn format(&self, min_digits: usize, group: usize) -> String {
        let magnitude_len = self.digits.len().max(1);
        let width = magnitude_len.max(min_digits);
        let padding = width - magnitude_len;

        let mut out = String::with_capacity(width + width / 2 + usize::from(self.negative));
        if self.negative && !self.is_zero() {
            out.push('-');
        }

        for pos in (0..width).rev() {
            if group > 0 && pos < width - 1 && (pos + 1) % group == 0 {
                out.push('_');
            }
            if pos >= self.digits.len() {
                out.push('0');
            } else {
                out.push((b'0' + self.digits[pos]) as char);
            }
        }

        let _ = padding;
        out
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.is_negative(), rhs.is_negative()) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => cmp_magnitude(&self.digits, &rhs.digits),
            (true, true) => cmp_magnitude(&rhs.digits, &self.digits),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = if self.is_negative() == rhs.is_negative() {
            Self {
                digits: add_magnitudes(&self.digits, &rhs.digits),
                negative: self.is_negative(),
            }
        } else {
            match cmp_magnitude(&self.digits, &rhs.digits) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => Self {
                    digits: sub_magnitudes(&self.digits, &rhs.digits),
                    negative: self.is_negative(),
                },
                Ordering::Less => Self {
                    digits: sub_magnitudes(&rhs.digits, &self.digits),
                    negative: rhs.is_negative(),
                },
            }
        };
        result.normalize();
        result
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut accum = vec![0u32; self.digits.len() + rhs.digits.len()];
        for (i, &a) in self.digits.iter().enumerate() {
            for (j, &b) in rhs.digits.iter().enumerate() {
                accum[i + j] += u32::from(a) * u32::from(b);
            }
        }

        for i in 0..accum.len() - 1 {
            let carry = accum[i] / 10;
            accum[i] %= 10;
            accum[i + 1] += carry;
        }

        let mut digits: Vec<u8> = accum.into_iter().map(|v| (v % 10) as u8).collect();
        while digits.len() > 1 && digits.last() == Some(&0) {
            digits.pop();
        }

        let mut result = Self {
            digits,
            negative: self.is_negative() != rhs.is_negative(),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }

        let (quotient_digits, remainder_digits) =
            div_magnitudes(&self.digits, &rhs.digits);

        let mut quotient = Self {
            digits: quotient_digits,
            negative: self.is_negative() != rhs.is_negative(),
        };
        let mut remainder = Self {
            digits: remainder_digits,
            negative: self.is_negative(),
        };
        quotient.normalize();
        remainder.normalize();
        Ok((quotient, remainder))
    }

    fn zero() -> Self {
        Self {
            digits: vec![0],
            negative: false,
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn is_negative(&self) -> bool {
        self.negative && !self.is_zero()
    }

    fn normalize(&mut self) {
        while self.digits.len() > 1 && self.digits.last() == Some(&0) {
            self.digits.pop();
        }
        if self.digits.is_empty() {
            self.digits.push(0);
        }
        if self.is_zero() {
            self.negative = false;
        }
    }
}

fn cmp_magnitude(a: &[u8], b: &[u8]) -> Ordering {
    match a.len().cmp(&b.len()) {
        Ordering::Equal => {
            for i in (0..a.len()).rev() {
                match a[i].cmp(&b[i]) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        }
        other => other,
    }
}

fn add_magnitudes(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u16;
    for i in 0..a.len().max(b.len()) {
        let sum = u16::from(*a.get(i).unwrap_or(&0))
            + u16::from(*b.get(i).unwrap_or(&0))
            + carry;
        out.push((sum % 10) as u8);
        carry = sum / 10;
    }
    if carry != 0 {
        out.push(carry as u8);
    }
    out
}

// Requires a >= b.
fn sub_magnitudes(a: &[u8], b: &[u8]) -> Vec<u8> {
    let mut out = Vec::with_capacity(a.len());
    let mut borrow = 0i16;
    for i in 0..a.len() {
        let mut value = i16::from(a[i]) - borrow - i16::from(*b.get(i).unwrap_or(&0));
        if value < 0 {
            value += 10;
            borrow = 1;
        } else {
            borrow = 0;
        }
        out.push(value as u8);
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out
}

fn mul_magnitude_digit(a: &[u8], digit: u8) -> Vec<u8> {
    if digit == 0 {
        return vec![0];
    }
    let mut out = Vec::with_capacity(a.len() + 1);
    let mut carry = 0u16;
    for &value in a {
        let product = u16::from(value) * u16::from(digit) + carry;
        out.push((product % 10) as u8);
        carry = product / 10;
    }
    while carry != 0 {
        out.push((carry % 10) as u8);
        carry /= 10;
    }
    out
}

fn div_magnitudes(dividend: &[u8], divisor: &[u8]) -> (Vec<u8>, Vec<u8>) {
    if cmp_magnitude(dividend, divisor) == Ordering::Less {
        return (vec![0], dividend.to_vec());
    }

    let mut quotient_msd = Vec::with_capacity(dividend.len());
    let mut remainder = vec![0];

    for &digit in dividend.iter().rev() {
        if remainder.len() == 1 && remainder[0] == 0 {
            remainder[0] = digit;
        } else {
            remainder.insert(0, digit);
        }
        trim_magnitude(&mut remainder);

        let mut qdigit = 0u8;
        for candidate in (1..=9).rev() {
            let product = mul_magnitude_digit(divisor, candidate);
            if cmp_magnitude(&product, &remainder) != Ordering::Greater {
                qdigit = candidate;
                remainder = sub_magnitudes(&remainder, &product);
                break;
            }
        }
        quotient_msd.push(qdigit);
    }

    while quotient_msd.len() > 1 && quotient_msd.first() == Some(&0) {
        quotient_msd.remove(0);
    }
    let mut quotient: Vec<u8> = quotient_msd.into_iter().rev().collect();
    trim_magnitude(&mut quotient);
    trim_magnitude(&mut remainder);
    (quotient, remainder)
}

fn trim_magnitude(digits: &mut Vec<u8>) {
    while digits.len() > 1 && digits.last() == Some(&0) {
        digits.pop();
    }
    if digits.is_empty() {
        digits.push(0);
    }
}
```
