```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const EXPONENT_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let sum = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64
                + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut x = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if x < 0 {
                x += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(x as u32);
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn mul_abs_small(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            self.limbs.clear();
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let n = *limb as u64 * factor as u64 + carry;
            *limb = (n % Self::BASE) as u32;
            carry = n / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_abs_small(&mut self, value: u32) {
        let mut carry = value as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let n = self.limbs[i] as u64 + carry;
            self.limbs[i] = (n % Self::BASE) as u32;
            carry = n / Self::BASE;
            i += 1;
        }
    }

    fn mul_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        if a.is_empty() || b.is_empty() {
            return Vec::new();
        }
        let mut out = vec![0u32; a.len() + b.len()];
        for (i, &x) in a.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in b.iter().enumerate() {
                let k = i + j;
                let n = out[k] as u64 + x as u64 * y as u64 + carry;
                out[k] = (n % Self::BASE) as u32;
                carry = n / Self::BASE;
            }
            let mut k = i + b.len();
            while carry != 0 {
                if k == out.len() {
                    out.push(0);
                }
                let n = out[k] as u64 + carry;
                out[k] = (n % Self::BASE) as u32;
                carry = n / Self::BASE;
                k += 1;
            }
        }
        while out.last() == Some(&0) {
            out.pop();
        }
        out
    }

    fn div_rem_abs(a: &Self, b: &Self) -> (Self, Self) {
        debug_assert!(!b.is_zero());
        if a.cmp_abs(b) == Ordering::Less {
            let mut rem = a.clone();
            rem.negative = false;
            return (Self::zero(), rem);
        }

        let mut quotient_be = Vec::with_capacity(a.limbs.len());
        let mut rem = Self::zero();
        for &limb in a.limbs.iter().rev() {
            rem.limbs.insert(0, limb);
            rem.normalize();

            let mut lo = 0u64;
            let mut hi = Self::BASE - 1;
            while lo < hi {
                let mid = (lo + hi + 1) / 2;
                let mut product = b.clone();
                product.negative = false;
                product.mul_abs_small(mid as u32);
                if product.cmp_abs(&rem) != Ordering::Greater {
                    lo = mid;
                } else {
                    hi = mid - 1;
                }
            }

            quotient_be.push(lo as u32);
            if lo != 0 {
                let mut product = b.clone();
                product.negative = false;
                product.mul_abs_small(lo as u32);
                rem.limbs = Self::sub_abs(&rem.limbs, &product.limbs);
                rem.normalize();
            }
        }
        quotient_be.reverse();
        while quotient_be.last() == Some(&0) {
            quotient_be.pop();
        }
        (Self { negative: false, limbs: quotient_be }, rem)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    fn valid_radix(radix: u32) -> Result<(), Error> {
        if (2..=36).contains(&radix) {
            Ok(())
        } else {
            Err(Error::InvalidRadix { radix })
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        Self::valid_radix(radix)?;
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'-' => (true, 1),
            b'+' => (false, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (i, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && Self::digit_value(byte).filter(|&d| d < radix).is_none() {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_'
                && (i == start
                    || i + 1 == bytes.len()
                    || bytes[i - 1] == b'_'
                    || bytes[i + 1] == b'_')
            {
                malformed_separator = Some(i);
                break;
            }
        }
        if let Some(index) = malformed_separator {
            return Err(Error::Separator { index });
        }

        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        if digit_count > 1 && Self::digit_value(bytes[first_digit_index]) == Some(0) {
            return Err(Error::LeadingZero { index: first_digit_index });
        }
        if digit_count > max_digits {
            let target = max_digits;
            let index = bytes[start..]
                .iter()
                .enumerate()
                .filter(|(_, b)| **b != b'_')
                .nth(target)
                .map(|(i, _)| start + i)
                .unwrap_or(first_digit_index);
            return Err(Error::TooManyDigits { limit: max_digits, index });
        }

        let mut result = Self::zero();
        for &byte in &bytes[start..] {
            if byte == b'_' {
                continue;
            }
            result.mul_abs_small(radix);
            result.add_abs_small(Self::digit_value(byte).unwrap());
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        Self::valid_radix(radix)?;
        let mut n = self.clone();
        n.negative = false;
        let mut digits = Vec::new();
        if n.is_zero() {
            digits.push(0u32);
        } else {
            while !n.is_zero() {
                let mut rem = 0u64;
                for limb in n.limbs.iter_mut().rev() {
                    let cur = rem * Self::BASE + *limb as u64;
                    *limb = (cur / radix as u64) as u32;
                    rem = cur % radix as u64;
                }
                n.normalize();
                digits.push(rem as u32);
            }
            digits.reverse();
        }

        if digits.len() < min_digits {
            let mut padded = vec![0; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(match *digit {
                0..=9 => (b'0' + *digit as u8) as char,
                _ => (b'a' + (*digit - 10) as u8) as char,
            });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        let mut result = if self.negative == rhs.negative {
            Self {
                negative: self.negative,
                limbs: Self::add_abs(&self.limbs, &rhs.limbs),
            }
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Greater => Self {
                    negative: self.negative,
                    limbs: Self::sub_abs(&self.limbs, &rhs.limbs),
                },
                Ordering::Less => Self {
                    negative: rhs.negative,
                    limbs: Self::sub_abs(&rhs.limbs, &self.limbs),
                },
                Ordering::Equal => Self::zero(),
            }
        };
        result.normalize();
        result
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut result = Self {
            negative: self.negative ^ rhs.negative,
            limbs: Self::mul_abs(&self.limbs, &rhs.limbs),
        };
        result.normalize();
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut q, mut r) = Self::div_rem_abs(self, rhs);
        q.negative = (self.negative ^ rhs.negative) && !q.is_zero();
        r.negative = self.negative && !r.is_zero();
        Ok((q, r))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::EXPONENT_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::EXPONENT_LIMIT,
            });
        }
        let mut power = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while power > 0 {
            if power & 1 == 1 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.clone();
        let mut b = rhs.clone();
        a.negative = false;
        b.negative = false;
        while !b.is_zero() {
            let (_, r) = a.div_rem(&b).expect("nonzero gcd divisor");
            a = b;
            b = r;
            b.negative = false;
        }
        a.negative = false;
        a
    }
}
```
