```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn components(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&byte| byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut components = Vec::new();
        let mut start = 1;

        for end in 1..=bytes.len() {
            if end == bytes.len() || bytes[end] == b'/' {
                let component = &path[start..end];
                if component.is_empty() || component == "." || component == ".." {
                    let offset = if component.is_empty() {
                        if end == bytes.len() {
                            bytes.len() - 1
                        } else {
                            end
                        }
                    } else {
                        start
                    };
                    return Err(Error::InvalidPath(offset));
                }
                components.push(component);
                start = end + 1;
            }
        }
        Ok(components)
    }

    fn canonical_prefix(components: &[&str]) -> String {
        if components.is_empty() {
            "/".to_string()
        } else {
            let mut path = String::new();
            for component in components {
                path.push('/');
                path.push_str(component);
            }
            path
        }
    }

    fn entry_at<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        for (index, component) in components.iter().enumerate() {
            let prefix = Self::canonical_prefix(&components[..=index]);
            match current {
                Entry::Dir(children) => {
                    current = children.get(*component).ok_or(Error::Missing(prefix))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(&components[..index])));
                }
            }
        }
        Ok(current)
    }

    fn parent_mut<'a>(
        &'a mut self,
        components: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let mut current = &mut self.root;
        for (index, component) in components.iter().enumerate() {
            let prefix = Self::canonical_prefix(&components[..=index]);
            match current {
                Entry::Dir(children) => {
                    current = match children.get_mut(*component) {
                        Some(Entry::Dir(children)) => children,
                        Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
                        None => return Err(Error::Missing(prefix)),
                    };
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(&components[..index])));
                }
            }
        }
        match current {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => unreachable!(),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }

        let (name, parents) = components.split_last().unwrap();
        let parent = self.parent_mut(parents)?;
        let canonical = Self::canonical_prefix(&components);
        if parent.contains_key(*name) {
            return Err(Error::Exists(canonical));
        }
        parent.insert((*name).to_string(), Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }

        let (name, parents) = components.split_last().unwrap();
        let parent = self.parent_mut(parents)?;
        let canonical = Self::canonical_prefix(&components);
        match parent.get_mut(*name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                *contents = data.to_vec();
                Ok(())
            }
            None => {
                parent.insert((*name).to_string(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::components(path)?;
        match self.entry_at(&components)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical_prefix(&components))),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::components(path)?;
        match self.entry_at(&components)? {
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
            Entry::File(_) => Err(Error::NotDir(Self::canonical_prefix(&components))),
        }
    }
}
```
