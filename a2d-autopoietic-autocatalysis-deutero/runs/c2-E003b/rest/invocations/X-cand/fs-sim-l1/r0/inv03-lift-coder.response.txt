```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn components(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut start = 1;
        for (i, byte) in path.bytes().enumerate().skip(1) {
            if byte == b'/' {
                let component = &path[start..i];
                if component.is_empty() || component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                components.push(component);
                start = i + 1;
            }
        }

        let component = &path[start..];
        if component.is_empty() || component == "." || component == ".." {
            return Err(Error::InvalidPath(if component.is_empty() {
                path.len() - 1
            } else {
                start
            }));
        }
        components.push(component);
        Ok(components)
    }

    fn prefix(components: &[&str]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn parent_dir<'a>(
        &self,
        components: &[&str],
    ) -> Result<&'a BTreeMap<String, Entry>, Error>
    where
        Self: 'a,
    {
        let mut entry = &self.root;
        for i in 0..components.len().saturating_sub(1) {
            let prefix = Self::prefix(&components[..=i]);
            match entry {
                Entry::Dir(children) => match children.get(components[i]) {
                    Some(next) => entry = next,
                    None => return Err(Error::Missing(prefix)),
                },
                Entry::File(_) => return Err(Error::NotDir(Self::prefix(&components[..i])),
                ),
            }
        }

        match entry {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => {
                let parent = Self::prefix(&components[..components.len().saturating_sub(1)]);
                Err(Error::NotDir(parent))
            }
        }
    }

    fn lookup<'a>(&'a self, components: &[&str]) -> Result<&'a Entry, Error> {
        let mut entry = &self.root;
        for (i, component) in components.iter().enumerate() {
            let prefix = Self::prefix(&components[..=i]);
            match entry {
                Entry::Dir(children) => match children.get(*component) {
                    Some(next) => entry = next,
                    None => return Err(Error::Missing(prefix)),
                },
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::prefix(&components[..i])));
                }
            }
        }
        Ok(entry)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let final_path = Self::prefix(&components);
        let parent = self.parent_dir(&components)?;
        if parent.contains_key(components[components.len() - 1]) {
            return Err(Error::Exists(final_path));
        }

        let parent = self.parent_dir_mut(&components)?;
        parent.insert(
            components[components.len() - 1].to_owned(),
            Entry::Dir(BTreeMap::new()),
        );
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::components(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let final_path = Self::prefix(&components);
        let parent = self.parent_dir(&components)?;
        let name = components[components.len() - 1];
        if let Some(entry) = parent.get(name) {
            match entry {
                Entry::Dir(_) => return Err(Error::IsDir(final_path)),
                Entry::File(_) => {}
            }
        }

        let parent = self.parent_dir_mut(&components)?;
        match parent.get_mut(name) {
            Some(Entry::File(contents)) => contents.clear(),
            Some(Entry::Dir(_)) => return Err(Error::IsDir(final_path)),
            None => {
                parent.insert(name.to_owned(), Entry::File(Vec::new()));
            }
        }
        if let Some(Entry::File(contents)) = parent.get_mut(name) {
            contents.extend_from_slice(data);
        }
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::components(path)?;
        let final_path = Self::prefix(&components);
        match self.lookup(&components)? {
            Entry::Dir(_) => Err(Error::IsDir(final_path)),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::components(path)?;
        let final_path = Self::prefix(&components);
        match self.lookup(&components)? {
            Entry::File(_) => Err(Error::NotDir(final_path)),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }

    fn parent_dir_mut<'a>(
        &'a mut self,
        components: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let mut entry = &mut self.root;
        for i in 0..components.len().saturating_sub(1) {
            let prefix = Self::prefix(&components[..=i]);
            match entry {
                Entry::Dir(children) => match children.get_mut(components[i]) {
                    Some(next) => entry = next,
                    None => return Err(Error::Missing(prefix)),
                },
                Entry::File(_) => return Err(Error::NotDir(Self::prefix(&components[..i]))),
            }
        }

        match entry {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::prefix(
                &components[..components.len().saturating_sub(1)],
            ))),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
