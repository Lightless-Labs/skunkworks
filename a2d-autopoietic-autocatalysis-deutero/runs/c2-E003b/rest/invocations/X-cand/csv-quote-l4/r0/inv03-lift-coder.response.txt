use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|value| value == name),
            DuplicatePolicy::Reject | DuplicatePolicy::KeepFirst => {
                header.iter().position(|value| value == name)
            }
        }
    }
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let start = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if start == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut row = Vec::new();
    let mut field = String::new();
    let mut i = start;
    let mut record_index = 0usize;
    let mut record_started = false;

    #[derive(Clone, Copy)]
    enum State {
        FieldStart,
        Unquoted,
        Quoted { opening: usize },
        AfterQuote,
    }

    let mut state = State::FieldStart;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        let outside = !matches!(state, State::Quoted { .. });
        if outside && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            State::FieldStart => {
                record_started = true;
                if b == quote {
                    state = State::Quoted { opening: i };
                    i += 1;
                } else if b == delimiter {
                    row.push(String::new());
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(String::new());
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    record_started = false;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    state = State::Unquoted;
                    i += ch.len_utf8();
                }
            }
            State::Unquoted => {
                if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                }
            }
            State::Quoted { opening } => {
                if b == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = State::AfterQuote;
                        i += 1;
                    }
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                    let _ = opening;
                }
            }
            State::AfterQuote => {
                if b == delimiter {
                    row.push(std::mem::take(&mut field));
                    state = State::FieldStart;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    row.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut row));
                    record_index += 1;
                    record_started = false;
                    state = State::FieldStart;
                    i += if b == b'\r' { 2 } else { 1 };
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    if let State::Quoted { opening } = state {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: opening,
        });
    }

    if record_started {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn make_table(
    records: Vec<Vec<String>>,
    config: Config,
) -> Result<Table, CsvError> {
    let (header, rows, header_record_index) = if config.headers {
        if records.is_empty() {
            (None, Vec::new(), None)
        } else {
            let mut it = records.into_iter();
            (it.next(), it.collect::<Vec<_>>(), Some(0usize))
        }
    } else {
        (None, records, None)
    };

    if let Some(header) = &header {
        if config.duplicates == DuplicatePolicy::Reject {
            let mut seen: HashMap<&str, usize> = HashMap::new();
            for (second, name) in header.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(CsvError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second,
                    });
                }
                seen.insert(name, second);
            }
        }

        for (idx, row) in rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1 + header_record_index.unwrap_or(0),
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| make_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

#[derive(Debug, Clone, PartialEq, Eq)]

    pub row: usize,
    pub spec: usize,
    pub column: Option<usize>,
    pub kind: CellErrorKind,
}

    Collect,
    FirstPerRow,
    Stop,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct TypedResult {
    pub rows: Vec<Vec<Option<Value>>>,
    pub errors: Vec<CellError>,
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return b[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let (negative, body) = if b[0] == b'-' {
        (true, &b[1..])
    } else {
        (false, b)
    };
    let Some(dot) = body.iter().position(|&x| x == b'.') else {
        return false;
    };
    let integral = &body[..dot];
    let fractional = &body[dot + 1..];
    if fractional.len() != 2 || !fractional.iter().all(u8::is_ascii_digit) {
        return false;
    }
    if integral.is_empty()
        || !integral.iter().all(u8::is_ascii_digit)
        || (integral.len() > 1 && integral[0] == b'0')
    {
        return false;
    }
    !(negative && integral == b"0" && fractional == b"00")
}

fn eval_cell(
    text: Option<&str>,
    column: Option<usize>,
    spec: &ColumnSpec,
    row: usize,
    spec_index: usize,
) -> Result<Value, CellError> {
    let error = |kind| CellError {
        row,
        spec: spec_index,
        column,
        kind,
    };

    let Some(text) = text else {
        return Err(error(CellErrorKind::MissingCell));
    };

    if text.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(error(CellErrorKind::Empty));
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(text.to_string())),
        ColumnType::Boolean => match text {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(error(CellErrorKind::InvalidBoolean)),
        },
        ColumnType::Integer => {
            if !integer_syntax(text) {
                return Err(error(CellErrorKind::InvalidInteger));
            }
            match text.parse::<i64>() {
                Ok(value) => Ok(Value::Integer(value)),
                Err(_) => Err(error(CellErrorKind::IntegerOverflow)),
            }
        }
        ColumnType::Decimal => {
            if !decimal_syntax(text) {
                return Err(error(CellErrorKind::InvalidDecimal));
            }
            let negative = text.starts_with('-');
            let body = if negative { &text[1..] } else { text };
            let dot = body.find('.').unwrap();
            let whole = body[..dot].parse::<i128>().unwrap();
            let frac = body[dot + 1..].parse::<i128>().unwrap();
            let magnitude = whole * 100 + frac;
            let signed = if negative { -magnitude } else { magnitude };
            if signed < i64::MIN as i128 || signed > i64::MAX as i128 {
                Err(error(CellErrorKind::DecimalOverflow))
            } else {
                Ok(Value::Decimal(signed as i64))
            }
        }
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = vec![None; columns.len()];
        let mut stop_row = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if stop_row {
                break;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(index) => Some(*index),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let Some(column) = resolved else {
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                if policy != ErrorPolicy::Collect {
                    stop_row = true;
                }
                if policy == ErrorPolicy::Stop {
                    result.rows.push(output);
                    return result;
                }
                continue;
            };

            let cell = row.get(column).map(String::as_str);
            match eval_cell(cell, Some(column), spec, row_index, spec_index) {
                Ok(value) => output[spec_index] = Some(value),
                Err(error) => {
                    result.errors.push(error);
                    if policy != ErrorPolicy::Collect {
                        stop_row = true;
                    }
                    if policy == ErrorPolicy::Stop {
                        result.rows.push(output);
                        return result;
                    }
                }
            }
        }

        result.rows.push(output);
    }

    result
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul { record: usize, column: usize, byte: usize },
    Width { record: usize, expected: usize, actual: usize },
    ZeroColumns { record: usize },
    DuplicateHeader { name: String, first: usize, second: usize },
}

fn write_records(table: &Table, mode: WriteMode) -> Result<Vec<Vec<String>>, WriteError> {
    let mut records = Vec::new();
    if let Some(header) = &table.header {
        records.push(header.clone());
    }
    records.extend(table.rows.iter().cloned());

    for (record_index, record) in records.iter().enumerate() {
        for (column_index, field) in record.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record: record_index,
                    column: column_index,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (row_index, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: row_index + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record_index, record) in records.iter().enumerate() {
        if record.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: record_index,
            });
        }
    }

    if mode == WriteMode::Preserve {
        return Ok(records);
    }

    if let Some(header) = &table.header {
        if table.duplicates == DuplicatePolicy::Reject {
            let mut seen: HashMap<&str, usize> = HashMap::new();
            for (second, name) in header.iter().enumerate() {
                if let Some(&first) = seen.get(name.as_str()) {
                    return Err(WriteError::DuplicateHeader {
                        name: name.clone(),
                        first,
                        second,
                    });
                }
                seen.insert(name, second);
            }
        }

        let mut chosen: HashMap<&str, usize> = HashMap::new();
        for (index, name) in header.iter().enumerate() {
            match table.duplicates {
                DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                    chosen.entry(name.as_str()).or_insert(index);
                }
                DuplicatePolicy::KeepLast => {
                    chosen.insert(name.as_str(), index);
                }
            }
        }
        let mut retained: Vec<(String, usize)> = chosen
            .into_iter()
            .map(|(name, index)| (name.to_string(), index))
            .collect();
        retained.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

        let output_header = retained.iter().map(|(name, _)| name.clone()).collect();
        let indexes: Vec<usize> = retained.iter().map(|(_, index)| *index).collect();

        let mut projected_rows: Vec<Vec<String>> = table
            .rows
            .iter()
            .map(|row| indexes.iter().map(|&index| row[index].clone()).collect())
            .collect();

        projected_rows.sort_by(|a, b| compare_rows(a, b));
        let mut output = vec![output_header];
        output.extend(projected_rows);
        Ok(output)
    } else {
        let mut rows = table.rows.clone();
        rows.sort_by(|a, b| compare_rows(a, b));
        Ok(rows)
    }
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (left, right) in a.iter().zip(b.iter()) {
        let cmp = left.as_bytes().cmp(right.as_bytes());
        if cmp != std::cmp::Ordering::Equal {
            return cmp;
        }
    }
    a.len().cmp(&b.len())
}

fn encode_field(
    field: &str,
    delimiter: u8,
    quote: u8,
    first_record: bool,
    first_field: bool,
    sole_field: bool,
) -> String {
    let bytes = field.as_bytes();
    let edge_space = bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
        || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t');
    let bom_prefix = first_record && first_field && field.starts_with('\u{feff}');
    let sole_empty = sole_field && field.is_empty();
    let needs_quote = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || edge_space
        || bom_prefix
        || sole_empty;

    if !needs_quote {
        return field.to_string();
    }

    let mut out = String::new();
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
            out.push(quote as char);
        } else {
            out.push(ch);
        }
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let records = write_records(table, config.mode)?;
    if records.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    let terminator: &str = match (config.mode, config.line_ending) {
        (WriteMode::Canonical, _) | (_, LineEnding::Lf) => "\n",
        (_, LineEnding::CrLf) => "\r\n",
    };

    for (record_index, record) in records.iter().enumerate() {
        if record_index > 0 {
            out.push_str(terminator);
        }
        for (column_index, field) in record.iter().enumerate() {
            if column_index > 0 {
                out.push(config.delimiter as char);
            }
            out.push_str(&encode_field(
                field,
                config.delimiter,
                config.quote,
                record_index == 0,
                column_index == 0,
                record.len() == 1,
            ));
        }
    }

    let append_final = match config.mode {
        WriteMode::Canonical => true,
        WriteMode::Preserve => config.final_terminator,
    };
    if append_final {
        out.push_str(terminator);
    }

    Ok(out)
}
