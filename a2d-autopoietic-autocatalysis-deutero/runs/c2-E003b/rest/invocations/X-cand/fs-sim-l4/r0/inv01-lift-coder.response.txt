```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
struct Resolved {
    path: String,
    entry: Entry,
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        if path == "/" {
            Vec::new()
        } else {
            path[1..].split('/').map(str::to_owned).collect()
        }
    }

    fn path_from(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_owned()),
            }
        }

        Self::path_from(&parts)
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_owned();
        }
        match path.rfind('/') {
            Some(0) => "/".to_owned(),
            Some(i) => path[..i].to_owned(),
            None => "/".to_owned(),
        }
    }

    fn basename(path: &str) -> String {
        if path == "/" {
            String::new()
        } else {
            path.rsplit('/').next().unwrap_or("").to_owned()
        }
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{name}")
        } else {
            format!("{parent}/{name}")
        }
    }

    fn is_within(path: &str, ancestor: &str) -> bool {
        if ancestor == "/" {
            return path.starts_with('/');
        }
        path == ancestor
            || path
                .strip_prefix(ancestor)
                .is_some_and(|rest| rest.starts_with('/'))
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| {
                *candidate != path
                    && Self::parent(candidate) == path
                    && self.entries.contains_key(*candidate)
            })
            .cloned()
            .collect()
    }

    fn resolve(&self, path: &str) -> Result<Resolved, Error> {
        let mut current = path.to_owned();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            let parts = Self::components(&current);
            let mut prefix_parts: Vec<String> = Vec::new();
            let mut restarted = false;

            for (index, part) in parts.iter().enumerate() {
                prefix_parts.push(part.clone());
                let prefix = Self::path_from(&prefix_parts);
                let entry = match self.entries.get(&prefix) {
                    Some(entry) => entry,
                    None => return Err(Error::Missing(prefix)),
                };

                if let Entry::Link(target) = entry {
                    if visited.contains(&prefix) {
                        return Err(Error::Loop(prefix));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(prefix));
                    }
                    visited.insert(prefix.clone());
                    expansions += 1;

                    let link_parent = Self::parent(&prefix);
                    let mut expanded = Self::normalize(target, &link_parent);
                    if index + 1 < parts.len() {
                        let suffix = parts[index + 1..].join("/");
                        expanded = Self::normalize(&format!("{expanded}/{suffix}"), "/");
                    }
                    current = expanded;
                    restarted = true;
                    break;
                }

                if index + 1 < parts.len() && !matches!(entry, Entry::Dir) {
                    return Err(Error::NotDir(prefix));
                }
            }

            if restarted {
                continue;
            }

            let entry = self
                .entries
                .get(&current)
                .cloned()
                .ok_or_else(|| Error::Missing(current.clone()))?;
            return Ok(Resolved {
                path: current,
                entry,
            });
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<String, Error> {
        let parent = Self::parent(path);
        let resolved = self.resolve(&parent)?;
        if !matches!(resolved.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved.path));
        }
        Ok(resolved.path)
    }

    fn resolved_child(parent: &str, path: &str) -> String {
        Self::join(parent, &Self::basename(path))
    }

    fn is_dir(path: &str, entry: &Entry) -> bool {
        matches!(entry, Entry::Dir)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let parent = self.resolve_parent(&canonical)?;
        let final_path = Self::resolved_child(&parent, &canonical);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::IsDir("/".to_owned()));
        }

        let mut current = canonical.clone();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            let parent = Self::parent(&current);
            let resolved_parent = self.resolve_with_tracking(&parent, &mut visited, &mut expansions)?;
            if !matches!(resolved_parent.entry, Entry::Dir) {
                return Err(Error::NotDir(resolved_parent.path));
            }
            let candidate = Self::resolved_child(&resolved_parent.path, &current);

            match self.entries.get(&candidate) {
                Some(Entry::Link(target)) => {
                    if visited.contains(&candidate) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    visited.insert(candidate.clone());
                    expansions += 1;
                    current = Self::normalize(target, &Self::parent(&candidate));
                }
                Some(Entry::Dir) => return Err(Error::IsDir(candidate)),
                Some(Entry::File(_)) => {
                    self.entries
                        .insert(candidate, Entry::File(data.to_vec()));
                    return Ok(());
                }
                None => {
                    self.entries
                        .insert(candidate, Entry::File(data.to_vec()));
                    return Ok(());
                }
            }
        }
    }

    fn resolve_with_tracking(
        &self,
        path: &str,
        visited: &mut BTreeSet<String>,
        expansions: &mut usize,
    ) -> Result<Resolved, Error> {
        let mut current = path.to_owned();

        loop {
            let parts = Self::components(&current);
            let mut prefix_parts: Vec<String> = Vec::new();
            let mut restarted = false;

            for (index, part) in parts.iter().enumerate() {
                prefix_parts.push(part.clone());
                let prefix = Self::path_from(&prefix_parts);
                let entry = match self.entries.get(&prefix) {
                    Some(entry) => entry,
                    None => return Err(Error::Missing(prefix)),
                };

                if let Entry::Link(target) = entry {
                    if visited.contains(&prefix) {
                        return Err(Error::Loop(prefix));
                    }
                    if *expansions >= 8 {
                        return Err(Error::Depth(prefix));
                    }
                    visited.insert(prefix.clone());
                    *expansions += 1;

                    let mut expanded = Self::normalize(target, &Self::parent(&prefix));
                    if index + 1 < parts.len() {
                        expanded = Self::normalize(
                            &format!("{expanded}/{}", parts[index + 1..].join("/")),
                            "/",
                        );
                    }
                    current = expanded;
                    restarted = true;
                    break;
                }

                if index + 1 < parts.len() && !matches!(entry, Entry::Dir) {
                    return Err(Error::NotDir(prefix));
                }
            }

            if restarted {
                continue;
            }

            let entry = self
                .entries
                .get(&current)
                .cloned()
                .ok_or_else(|| Error::Missing(current.clone()))?;
            return Ok(Resolved {
                path: current,
                entry,
            });
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical)?;
        match resolved.entry {
            Entry::File(data) => Ok(data),
            Entry::Dir => Err(Error::IsDir(resolved.path)),
            Entry::Link(_) => unreachable!(),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical)?;
        if !matches!(resolved.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved.path));
        }

        let mut result: Vec<_> = self
            .children(&resolved.path)
            .into_iter()
            .map(|child| {
                let name = Self::basename(&child);
                let kind = self.entries.get(&child).unwrap().kind();
                (name, kind)
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical)?;
        if !matches!(resolved.entry, Entry::Dir) {
            return Err(Error::NotDir(resolved.path));
        }
        self.cwd = resolved.path;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        Self::validate(to)?;
        let source = Self::normalize(from, &self.cwd);
        let destination = Self::normalize(to, &self.cwd);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let source_parent = self.resolve_parent(&source)?;
        let source_path = Self::resolved_child(&source_parent, &source);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let destination_parent = self.resolve_parent(&destination)?;
        let destination_path = Self::resolved_child(&destination_parent, &destination);

        if let Some(Entry::Dir) = self.entries.get(&destination_path) {
            if Self::is_within(&self.cwd, &destination_path) {
                return Err(Error::Busy(destination_path));
            }
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir)
            && Self::is_within(&destination_path, &source_path)
        {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(ref existing) = destination_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination_path).is_empty() {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(path, _)| Self::is_within(path, &source_path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect();

        if let Some(existing) = destination_entry {
            if matches!(existing, Entry::Dir) {
                self.entries.remove(&destination_path);
            } else {
                self.entries.remove(&destination_path);
            }
        }

        for (old, _) in &moving {
            self.entries.remove(old);
        }
        for (old, entry) in moving {
            let suffix = if old == source_path {
                String::new()
            } else {
                old[source_path.len()..].to_owned()
            };
            let new_path = format!("{destination_path}{suffix}");
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }

        let parent = self.resolve_parent(&canonical)?;
        let final_path = Self::resolved_child(&parent, &canonical);
        let entry = self
            .entries
            .get(&final_path)
            .cloned()
            .ok_or_else(|| Error::Missing(final_path.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &final_path) {
            return Err(Error::Busy(final_path));
        }

        let children = self.children(&final_path);
        if matches!(entry, Entry::Dir) && !children.is_empty() && !recursive {
            return Err(Error::NotEmpty(final_path));
        }

        let remove_paths: Vec<String> = if matches!(entry, Entry::Dir) && recursive {
            self.entries
                .keys()
                .filter(|candidate| Self::is_within(candidate, &final_path))
                .cloned()
                .collect()
        } else {
            vec![final_path]
        };

        for candidate in remove_paths {
            self.entries.remove(&candidate);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let canonical_link = Self::normalize(link, &self.cwd);

        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let parent = self.resolve_parent(&canonical_link)?;
        let final_path = Self::resolved_child(&parent, &canonical_link);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }

        self.entries
            .insert(final_path, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let parent = self.resolve_parent(&canonical)?;
        let final_path = Self::resolved_child(&parent, &canonical);
        match self.entries.get(&final_path) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(final_path)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical)?;

        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| Self::is_within(candidate, &resolved.path))
            .cloned()
            .collect();

        let mut rows = Vec::with_capacity(paths.len());
        for item in &paths {
            let entry = self.entries.get(item).unwrap();
            let size = match entry {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => paths
                    .iter()
                    .filter(|descendant| {
                        *descendant != item && Self::is_within(descendant, item)
                    })
                    .filter_map(|descendant| self.entries.get(descendant))
                    .map(|descendant| match descendant {
                        Entry::File(data) => data.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
            };
            rows.push((item.clone(), size));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}
```
