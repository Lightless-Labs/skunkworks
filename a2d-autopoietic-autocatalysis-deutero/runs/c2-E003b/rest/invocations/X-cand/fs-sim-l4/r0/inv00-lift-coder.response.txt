use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = normalize(path, &self.cwd)?;
        let (parent, name) = split_parent(&path);
        let parent = self.resolve_full(&parent)?;
        self.require_dir(&parent)?;

        let resolved = join_path(&parent, &name);
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = normalize(path, &self.cwd)?;
        let resolved = self.resolve_full_allow_missing_final(&path)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, _) = split_parent(&resolved);
                self.require_dir(&parent)?;
                self.entries
                    .insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = normalize(path, &self.cwd)?;
        let resolved = self.resolve_full(&path)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) | None => unreachable!(),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = normalize(path, &self.cwd)?;
        let resolved = self.resolve_full(&path)?;
        self.require_dir(&resolved)?;

        let mut children = Vec::new();
        for (candidate, entry) in &self.entries {
            if candidate == &resolved {
                continue;
            }
            let (parent, name) = split_parent(candidate);
            if parent == resolved {
                children.push((name, entry.kind()));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = normalize(path, &self.cwd)?;
        let resolved = self.resolve_full(&path)?;
        self.require_dir(&resolved)?;
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = normalize(from, &self.cwd)?;
        let to = normalize(to, &self.cwd)?;
        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (src_parent_raw, src_name) = split_parent(&from);
        let src_parent = self.resolve_full(&src_parent_raw)?;
        self.require_dir(&src_parent)?;
        let source = join_path(&src_parent, &src_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        let (dst_parent_raw, dst_name) = split_parent(&to);
        let dst_parent = self.resolve_full(&dst_parent_raw)?;
        self.require_dir(&dst_parent)?;
        let destination = join_path(&dst_parent, &dst_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let dest_entry = self.entries.get(&destination).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if overwrite {
            if let Some(dest) = &dest_entry {
                match (&source_entry, dest) {
                    (Entry::Dir, Entry::Dir) => {
                        if self.has_children(&destination) {
                            return Err(Error::NotEmpty(destination));
                        }
                    }
                    (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                    (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                    _ => {}
                }
            }
        }

        let moving = self.subtree(&source);
        if let Some(dest) = &dest_entry {
            if matches!(dest, Entry::Dir) {
                self.remove_subtree(&destination);
            } else {
                self.entries.remove(&destination);
            }
        }
        self.remove_subtree(&source);

        for (old_path, entry) in moving {
            let suffix = old_path.strip_prefix(&source).unwrap_or("");
            let new_path = if suffix.is_empty() {
                destination.clone()
            } else if destination == "/" {
                format!("/{suffix}")
            } else {
                format!("{destination}{suffix}")
            };
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = normalize(path, &self.cwd)?;
        if path == "/" {
            return Err(Error::Root);
        }

        let (parent_raw, name) = split_parent(&path);
        let parent = self.resolve_full(&parent_raw)?;
        self.require_dir(&parent)?;
        let resolved = join_path(&parent, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && contains_path(&resolved, &self.cwd) {
            return Err(Error::Busy(resolved));
        }
        if matches!(entry, Entry::Dir) && self.has_children(&resolved) && !recursive {
            return Err(Error::NotEmpty(resolved));
        }

        if matches!(entry, Entry::Dir) && recursive {
            self.remove_subtree(&resolved);
        } else {
            self.entries.remove(&resolved);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link = normalize(link, &self.cwd)?;
        validate_target(target)?;

        let (parent_raw, name) = split_parent(&link);
        let parent = self.resolve_full(&parent_raw)?;
        self.require_dir(&parent)?;
        let resolved = join_path(&parent, &name);
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries
            .insert(resolved, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let path = normalize(path, &self.cwd)?;
        let resolved = self.resolve_parent_and_final(&path)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(Entry::Dir) | Some(Entry::File(_)) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let path = normalize(path, &self.cwd)?;
        let resolved = self.resolve_full(&path)?;
        let mut rows = Vec::new();
        for (candidate, entry) in &self.entries {
            if contains_path(&resolved, candidate) {
                rows.push((candidate.clone(), entry.own_size()));
            }
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));

        let sizes: BTreeMap<String, u64> = rows.iter().cloned().collect();
        for (candidate, entry) in &self.entries {
            if matches!(entry, Entry::Dir) && contains_path(&resolved, candidate) {
                let size = self
                    .entries
                    .iter()
                    .filter(|(child, _)| child != &candidate && contains_path(candidate, child))
                    .map(|(_, child_entry)| child_entry.own_size())
                    .sum::<u64>();
                let _ = sizes.get(candidate);
                if let Some(row) = rows.iter_mut().find(|row| row.0 == *candidate) {
                    row.1 = size;
                }
            }
        }
        Ok(rows)
    }

    fn require_dir(&self, path: &str) -> Result<(), Error> {
        match self.entries.get(path) {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(path.to_string())),
            None => Err(Error::Missing(path.to_string())),
        }
    }

    fn resolve_full(&self, path: &str) -> Result<String, Error> {
        self.resolve(path, false)
    }

    fn resolve_full_allow_missing_final(&self, path: &str) -> Result<String, Error> {
        self.resolve(path, true)
    }

    fn resolve_parent_and_final(&self, path: &str) -> Result<String, Error> {
        let (parent, name) = split_parent(path);
        let resolved_parent = self.resolve_full(&parent)?;
        self.require_dir(&resolved_parent)?;
        Ok(join_path(&resolved_parent, &name))
    }

    fn resolve(&self, path: &str, allow_missing_final: bool) -> Result<String, Error> {
        let mut pending: Vec<String> = components(path);
        let mut current = Vec::<String>::new();
        let mut visited = HashSet::<String>::new();
        let mut expansions = 0usize;

        while !pending.is_empty() {
            let component = pending.remove(0);
            current.push(component);
            let candidate = as_path(&current);
            let final_component = pending.is_empty();

            let entry = match self.entries.get(&candidate) {
                Some(entry) => entry,
                None => {
                    if allow_missing_final && final_component {
                        return Ok(candidate);
                    }
                    return Err(Error::Missing(candidate));
                }
            };

            match entry {
                Entry::Dir => {}
                Entry::File(_) => {
                    if !final_component {
                        return Err(Error::NotDir(candidate));
                    }
                }
                Entry::Link(target) => {
                    if !visited.insert(candidate.clone()) {
                        return Err(Error::Loop(candidate));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(candidate));
                    }
                    expansions += 1;

                    let parent = as_path(&current[..current.len() - 1]);
                    let mut replacement = if target.starts_with('/') {
                        components(target)
                    } else {
                        let mut base = components(&parent);
                        base.extend(components(target));
                        base
                    };
                    replacement.extend(pending);
                    current.clear();
                    pending = replacement;
                }
            }
        }

        Ok(as_path(&current))
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries.keys().any(|candidate| {
            candidate != path && split_parent(candidate).0 == path
        })
    }

    fn subtree(&self, root: &str) -> Vec<(String, Entry)> {
        self.entries
            .iter()
            .filter(|(path, _)| contains_path(root, path))
            .map(|(path, entry)| (path.clone(), entry.clone()))
            .collect()
    }

    fn remove_subtree(&mut self, root: &str) {
        let paths: Vec<String> = self
            .entries
            .keys()
            .filter(|path| contains_path(root, path))
            .cloned()
            .collect();
        for path in paths {
            self.entries.remove(&path);
        }
    }
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }

    fn own_size(&self) -> u64 {
        match self {
            Entry::Dir => 0,
            Entry::File(data) => data.len() as u64,
            Entry::Link(target) => target.len() as u64,
        }
    }
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(index) = path.as_bytes().iter().position(|byte| *byte == 0) {
        return Err(Error::InvalidPath(index));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn validate_target(target: &str) -> Result<(), Error> {
    if target.is_empty() {
        return Err(Error::InvalidTarget(0));
    }
    if let Some(index) = target.as_bytes().iter().position(|byte| *byte == 0) {
        return Err(Error::InvalidTarget(index));
    }
    Ok(())
}

fn normalize(path: &str, cwd: &str) -> Result<String, Error> {
    validate_path(path)?;
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(cwd)
    };
    for part in path.split('/') {
        match part {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_string()),
        }
    }
    Ok(as_path(&parts))
}

fn components(path: &str) -> Vec<String> {
    let mut parts = Vec::new();
    for part in path.split('/') {
        match part {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other.to_string()),
        }
    }
    parts
}

fn as_path(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn split_parent(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_string(), String::new());
    }
    match path.rfind('/') {
        Some(0) => ("/".to_string(), path[1..].to_string()),
        Some(index) => (path[..index].to_string(), path[index + 1..].to_string()),
        None => ("/".to_string(), path.to_string()),
    }
}

fn join_path(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn contains_path(parent: &str, path: &str) -> bool {
    if parent == "/" {
        path.starts_with('/')
    } else {
        path == parent
            || path
                .strip_prefix(parent)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }
}
