```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Clone, Debug)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|s| !s.is_empty() && *s != ".")
        .fold(Vec::new(), |mut acc, part| {
            if part == ".." {
                acc.pop();
            } else {
                acc.push(part.to_string());
            }
            acc
        })
}

fn canonical(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn path_parts(path: &str) -> Vec<String> {
    components(path)
}

fn parent_path(path: &str) -> String {
    if path == "/" {
        return "/".to_string();
    }
    let i = path.rfind('/').unwrap_or(0);
    if i == 0 {
        "/".to_string()
    } else {
        path[..i].to_string()
    }
}

fn basename(path: &str) -> String {
    path.rsplit('/').next().unwrap_or("").to_string()
}

fn is_within(path: &str, dir: &str) -> bool {
    dir == "/" || path == dir || path.strip_prefix(dir).is_some_and(|s| s.starts_with('/'))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        validate_path(path)?;
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            path_parts(&self.cwd)
        };
        for part in path.split('/') {
            if part.is_empty() || part == "." {
                continue;
            }
            if part == ".." {
                parts.pop();
            } else {
                parts.push(part.to_string());
            }
        }
        Ok(canonical(&parts))
    }

    fn resolve(&self, path: &str, follow_final: bool, allow_missing_final: bool) -> Result<String, Error> {
        let initial = path_parts(path);
        let mut todo: Vec<String> = initial;
        let mut physical: Vec<String> = Vec::new();
        let mut visited = BTreeSet::new();
        let mut expansions = 0usize;

        loop {
            if todo.is_empty() {
                return Ok(canonical(&physical));
            }

            let component = todo.remove(0);
            physical.push(component);
            let current = canonical(&physical);
            let is_final = todo.is_empty();

            match self.entries.get(&current) {
                None => {
                    if is_final && allow_missing_final {
                        return Ok(current);
                    }
                    return Err(Error::Missing(current));
                }
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => {
                    if !is_final {
                        return Err(Error::NotDir(current));
                    }
                }
                Some(Entry::Link(target)) if follow_final || !is_final => {
                    if !visited.insert(current.clone()) {
                        return Err(Error::Loop(current));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(current));
                    }
                    expansions += 1;

                    let mut target_parts = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        physical[..physical.len() - 1].to_vec()
                    };
                    for part in target.split('/') {
                        if part.is_empty() || part == "." {
                            continue;
                        }
                        if part == ".." {
                            target_parts.pop();
                        } else {
                            target_parts.push(part.to_string());
                        }
                    }
                    target_parts.extend(todo);
                    let expanded = canonical(&target_parts);
                    todo = path_parts(&expanded);
                    physical.clear();
                }
                Some(Entry::Link(_)) => {}
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = parent_path(path);
        let name = basename(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        if !matches!(self.entries.get(&resolved_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved_parent));
        }
        Ok((resolved_parent, name))
    }

    fn child_path(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn direct_children(&self, dir: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|path| *path != dir && parent_path(path) == dir)
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = Self::child_path(&parent, &name);
        if self.entries.contains_key(&target) {
            return Err(Error::Exists(target));
        }
        self.entries.insert(target, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let resolved = match self.resolve(&normalized, true, true) {
            Ok(p) => p,
            Err(Error::Missing(p)) => {
                let (parent, name) = self.resolve_parent(&p)?;
                Self::child_path(&parent, &name)
            }
            Err(e) => return Err(e),
        };

        match self.entries.get_mut(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(bytes)) => {
                *bytes = data.to_vec();
                Ok(())
            }
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                self.entries.insert(resolved, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
            Some(Entry::Link(_)) | None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        let mut result: Vec<_> = self
            .direct_children(&resolved)
            .into_iter()
            .filter_map(|child| {
                let kind = match self.entries.get(&child)? {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                    Entry::Link(_) => Kind::Link,
                };
                Some((basename(&child), kind))
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = Self::child_path(&source_parent, &source_name);
        let source_entry = self.entries.get(&source_path).cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (dest_parent, dest_name) = self.resolve_parent(&destination)?;
        let dest_path = Self::child_path(&dest_parent, &dest_name);

        if matches!(self.entries.get(&dest_path), Some(Entry::Dir)) && is_within(&self.cwd, &dest_path) {
            return Err(Error::Busy(dest_path));
        }

        if source_path == dest_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && is_within(&dest_path, &source_path) {
            return Err(Error::Cycle(dest_path));
        }

        let dest_entry = self.entries.get(&dest_path).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(dest_path));
        }
        if let Some(existing) = &dest_entry {
            match (&source_entry, existing) {
                (Entry::Dir, Entry::Dir) => {
                    if self.direct_children(&dest_path).len() > 0 {
                        return Err(Error::NotEmpty(dest_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dest_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(dest_path)),
                _ => {}
            }
        }

        let moved: Vec<(String, Entry)> = self.entries
            .iter()
            .filter(|(p, _)| **p == source_path || is_within(p, &source_path))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();

        if let Some(existing) = dest_entry {
            if matches!(existing, Entry::Dir) {
                self.entries.remove(&dest_path);
            } else {
                self.entries.remove(&dest_path);
            }
        }

        for (old, _) in &moved {
            self.entries.remove(old);
        }
        for (old, entry) in moved {
            let suffix = &old[source_path.len()..];
            let new_path = format!("{}{}", dest_path, suffix);
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        if normalized == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = self.resolve_parent(&normalized)?;
        let target = Self::child_path(&parent, &name);
        let entry = self.entries.get(&target)
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &target) {
            return Err(Error::Busy(target));
        }

        let children = self.direct_children(&target);
        if matches!(entry, Entry::Dir) && !children.is_empty() && !recursive {
            return Err(Error::NotEmpty(target));
        }

        if recursive {
            let to_remove: Vec<String> = self.entries.keys()
                .filter(|p| **p == target || is_within(p, &target))
                .cloned()
                .collect();
            for p in to_remove {
                self.entries.remove(&p);
            }
        } else {
            self.entries.remove(&target);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let normalized_link = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }

        let (parent, name) = self.resolve_parent(&normalized_link)?;
        let link_path = Self::child_path(&parent, &name);
        if self.entries.contains_key(&link_path) {
            return Err(Error::Exists(link_path));
        }
        self.entries.insert(link_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = self.normalize(path)?;
        let parent = parent_path(&normalized);
        let name = basename(&normalized);
        let resolved_parent = self.resolve(&parent, true, false)?;
        if !matches!(self.entries.get(&resolved_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved_parent));
        }
        let target = Self::child_path(&resolved_parent, &name);
        match self.entries.get(&target) {
            None => Err(Error::Missing(target)),
            Some(Entry::Link(value)) => Ok(value.clone()),
            Some(Entry::Dir) | Some(Entry::File(_)) => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(&normalized, true, false)?;
        let entry = self.entries.get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let paths: Vec<String> = self.entries.keys()
            .filter(|p| **p == resolved || is_within(p, &resolved))
            .cloned()
            .collect();

        let mut rows = Vec::new();
        for p in paths {
            let size = match self.entries.get(&p).unwrap() {
                Entry::Dir => self.entries.iter()
                    .filter(|(child, e)| is_within(child, &p) && **child != p)
                    .map(|(_, e)| match e {
                        Entry::File(bytes) => bytes.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
                Entry::File(bytes) => bytes.len() as u64,
                Entry::Link(target) => target.len() as u64,
            };
            rows.push((p, size));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        let _ = entry;
        Ok(rows)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
