```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
}

pub struct IndexedHeap {
    items: Vec<Entry>,
    indices: std::collections::HashMap<String, usize>,
    next_age: u128,
}

struct Entry {
    item: Item,
    age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            items: Vec::new(),
            indices: std::collections::HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.items.len()
    }

    pub fn is_empty(&self) -> bool {
        self.items.is_empty()
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonical_key(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let entry = Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };
        self.next_age = self.next_age.wrapping_add(1);
        let index = self.items.len();
        self.items.push(entry);
        self.indices.insert(key, index);
        self.sift_up(index);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.items.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.items.is_empty() {
            return None;
        }
        Some(self.remove_at(0).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonical_key(key)?;
        Ok(self.indices.get(&key).map(|&i| self.items[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonical_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.items[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.items[index].item.priority = priority;
        self.reorder(index);
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonical_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.items[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.items[index].item.priority = priority;
        self.reorder(index);
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonical_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let previous = std::mem::replace(&mut self.items[index].item.value, value.to_owned());
        self.reorder(index);
        Ok(previous)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonical_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        Ok(self.remove_at(index).item)
    }

    fn remove_at(&mut self, index: usize) -> Entry {
        let removed = self.items.swap_remove(index);
        self.indices.remove(&removed.item.key);
        if index < self.items.len() {
            self.indices.insert(self.items[index].item.key.clone(), index);
            self.reorder(index);
        }
        removed
    }

    fn reorder(&mut self, index: usize) {
        if index > 0 && self.better(index, (index - 1) / 2) {
            self.sift_up(index);
        } else {
            self.sift_down(index);
        }
    }

    fn sift_up(&mut self, mut index: usize) {
        while index > 0 {
            let parent = (index - 1) / 2;
            if !self.better(index, parent) {
                break;
            }
            self.swap_slots(index, parent);
            index = parent;
        }
    }

    fn sift_down(&mut self, mut index: usize) {
        loop {
            let left = index * 2 + 1;
            if left >= self.items.len() {
                break;
            }
            let right = left + 1;
            let best = if right < self.items.len() && self.better(right, left) {
                right
            } else {
                left
            };
            if !self.better(best, index) {
                break;
            }
            self.swap_slots(index, best);
            index = best;
        }
    }

    fn better(&self, a: usize, b: usize) -> bool {
        let a = &self.items[a];
        let b = &self.items[b];
        a.item.priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
            .is_lt()
    }

    fn swap_slots(&mut self, a: usize, b: usize) {
        self.items.swap(a, b);
        self.indices.insert(self.items[a].item.key.clone(), a);
        self.indices.insert(self.items[b].item.key.clone(), b);
    }
}

fn canonical_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }

    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}
```
