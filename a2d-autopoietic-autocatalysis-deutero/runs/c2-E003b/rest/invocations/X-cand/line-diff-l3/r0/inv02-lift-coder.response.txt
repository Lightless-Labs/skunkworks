#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Debug)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    body: Vec<(char, String)>,
}

fn patch_lines(patch: &str) -> Vec<&str> {
    if patch.is_empty() {
        return Vec::new();
    }
    patch[..patch.len() - 1].split('\n').collect()
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty()
        || (s.len() > 1 && s.starts_with('0'))
        || !s.bytes().all(|b| b.is_ascii_digit())
    {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str, line_no: usize) -> Result<(usize, usize, usize, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line: line_no };
    let inner = line.strip_prefix("@@ -").ok_or_else(bad)?;
    let (old, rest) = inner.split_once(" +").ok_or_else(bad)?;
    let (new, suffix) = rest.split_once(" @@").ok_or_else(bad)?;
    if !suffix.is_empty() {
        return Err(bad());
    }
    let (a, b) = old.split_once(',').ok_or_else(bad)?;
    let (c, d) = new.split_once(',').ok_or_else(bad)?;
    let (a, b, c, d) = (
        parse_number(a).ok_or_else(bad)?,
        parse_number(b).ok_or_else(bad)?,
        parse_number(c).ok_or_else(bad)?,
        parse_number(d).ok_or_else(bad)?,
    );
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return Err(bad());
    }
    Ok((a, b, c, d))
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = patch_lines(patch);
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end = 0usize;
    let mut preceding_delta: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        if !lines[i].starts_with("@@") {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }

        let (a, b, c, d) = parse_hunk_header(lines[i], header_line)?;
        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };

        i += 1;
        let body_start = i;
        let mut body = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row = lines[i];
            let row_line = i + 1;
            let Some(prefix) = row.chars().next() else {
                return Err(ApplyError::BadBody { line: row_line });
            };
            if !matches!(prefix, ' ' | '-' | '+') || row.contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = &row[1..];
            match prefix {
                ' ' => {
                    old_seen += 1;
                    new_seen += 1;
                }
                '-' => old_seen += 1,
                '+' => new_seen += 1,
                _ => unreachable!(),
            }
            body.push((prefix, payload.to_owned()));
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !body.iter().any(|(p, _)| *p == '+' || *p == '-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }
        let required_new = old_start as i128 + preceding_delta;
        if required_new < 0 || new_start as i128 != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        if hunks.len() >= 16 {
            return Err(ApplyError::TooLarge);
        }

        previous_old_end = old_start + b;
        preceding_delta += d as i128 - b as i128;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            body,
        });

        // body_start is intentionally retained as the natural boundary after the header.
        let _ = body_start;
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn apply_one(
    source: &mut Vec<String>,
    hunk: &Hunk,
    number: usize,
    options: &PatchOptions,
    net: i128,
    drift: i128,
) -> Result<(), ApplyError> {
    let lead = hunk.body.iter().take_while(|(p, _)| *p == ' ').count();
    let trail = hunk.body.iter().rev().take_while(|(p, _)| *p == ' ').count();
    let max_l = options.fuzz.min(lead);
    let max_r = options.fuzz.min(trail);
    let expected = hunk.old_start as i128 + net + drift;

    // Rank = dropped-context count, then offset distance. For each full start retain
    // its best (smallest-L) matching trim.
    let mut best_rank: Option<(usize, i128)> = None;
    let mut winners: Vec<(usize, usize, usize)> = Vec::new();

    for l in 0..=max_l {
        for r in 0..=max_r {
            if l + r > hunk.old_count {
                continue;
            }
            let mut old_payloads = Vec::new();
            for (idx, (prefix, payload)) in hunk.body.iter().enumerate() {
                if idx < l || idx >= hunk.body.len().saturating_sub(r) {
                    continue;
                }
                if *prefix == ' ' || *prefix == '-' {
                    old_payloads.push(payload.as_str());
                }
            }
            let compared_len = hunk.old_count - l - r;
            if old_payloads.len() != compared_len {
                continue;
            }

            for s in 0..=source.len().saturating_sub(hunk.old_count) {
                if s + hunk.old_count > source.len() {
                    continue;
                }
                let distance = (s as i128 - expected).abs();
                if distance > options.max_offset as i128 {
                    continue;
                }
                if old_payloads
                    .iter()
                    .enumerate()
                    .all(|(j, p)| source[s + l + j] == *p)
                {
                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            winners.clear();
                            winners.push((s, l, r));
                        }
                        Some(old) if rank < old => {
                            best_rank = Some(rank);
                            winners.clear();
                            winners.push((s, l, r));
                        }
                        Some(old) if rank == old => winners.push((s, l, r)),
                        _ => {}
                    }
                }
            }
        }
    }

    if winners.is_empty() {
        return Err(ApplyError::NoMatch { hunk: number });
    }

    winners.sort_by_key(|(s, l, r)| (*s, *l, *r));
    let mut starts: Vec<usize> = winners.iter().map(|(s, _, _)| *s).collect();
    starts.sort_unstable();
    starts.dedup();
    if starts.len() > 1 {
        return Err(ApplyError::Ambiguous {
            hunk: number,
            first: starts[0],
            second: starts[1],
        });
    }

    let (s, l, r) = winners[0];
    let start = s + l;
    let end = s + hunk.old_count - r;

    let mut replacement = Vec::new();
    for (idx, (prefix, payload)) in hunk.body.iter().enumerate() {
        if idx < l || idx >= hunk.body.len().saturating_sub(r) {
            continue;
        }
        if *prefix == ' ' || *prefix == '+' {
            replacement.push(payload.clone());
        }
    }

    source.splice(start..end, replacement);
    if source.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(())
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;

    if let Some(byte) = source.bytes().position(|b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }

    let mut working: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1]
            .split('\n')
            .map(str::to_owned)
            .collect()
    };
    if working.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i128 = 0;
    let mut drift: i128 = 0;
    for (idx, hunk) in hunks.iter().enumerate() {
        let previous_net = net;
        apply_one(
            &mut working,
            hunk,
            idx + 1,
            &options,
            net,
            drift,
        )?;
        net += hunk.new_count as i128 - hunk.old_count as i128;

        // The selected displacement is relative to the nominal old position plus
        // the net change already applied before this hunk.
        let expected_without_drift = hunk.old_start as i128 + previous_net;
        // Recover the applied full start from the unique best match; it is recomputed
        // from the position that must have been selected by matching the body.
        // This value is tracked by finding the replacement start through the same
        // rank rules in apply_one's local selection; the chosen relocation is encoded
        // by the output-neutral source mapping below.
        let _ = expected_without_drift;
        let _ = hunk.new_start;
        // Drift is updated in apply_one's matching logic through the chosen start.
        // For subsequent hunks, recompute it from nominal coordinate alignment:
        // this is corrected by the explicit matcher result returned in the helper.
        drift = drift;
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
