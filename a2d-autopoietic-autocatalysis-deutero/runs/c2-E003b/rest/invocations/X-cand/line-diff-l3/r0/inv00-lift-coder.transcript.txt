struct Hunk {

    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,

}

fn parse_decimal(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.starts_with('0')) || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let mut value = 0usize;
    for b in s.bytes() {
        value = value.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if value > 128 {
            return None;
        }
    }
    Some(value)
}

fn parse_hunk_header(line: &str, line_no: usize) -> Result<(usize, usize, usize, usize), ApplyError> {
    let bad = || ApplyError::BadHunkHeader { line: line_no };
    let inner = line.strip_prefix("@@ -").ok_or_else(bad)?;
    let inner = inner.strip_suffix(" @@").ok_or_else(bad)?;
    let (old_part, new_part) = inner.split_once(" +").ok_or_else(bad)?;
    if new_part.contains(' ') || old_part.contains(' ') {
        return Err(bad());
    }
    let (a, b) = old_part.split_once(',').ok_or_else(bad)?;
    let (c, d) = new_part.split_once(',').ok_or_else(bad)?;
    if b.contains(',') || d.contains(',') {
        return Err(bad());

    let a = parse_decimal(a).ok_or_else(bad)?;
    let b = parse_decimal(b).ok_or_else(bad)?;

fn full_start(start: usize, count: usize) -> usize {
    if count > 0 { start - 1 } else { start }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: 1 + patch.bytes().filter(|&b| b == b'\n').count(),
        });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();
    if lines.first().copied() != Some("--- before") {
        return Err(ApplyError::BadHeader { line: 1 });
    }
    if lines.get(1).copied() != Some("+++ after") {
        return Err(ApplyError::BadHeader { line: 2 });
    }
    if lines.len() < 3 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2usize;
    let mut previous_old_end = 0usize;
    let mut net: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        let (a, b, c, d) = parse_hunk_header(lines[i], header_line)?;
        let old_start = full_start(a, b);
        let new_start = full_start(c, d);
        i += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row_line = i + 1;
            let text = lines[i];
            let bytes = text.as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') || text[1..].contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let prefix = bytes[0];
            if prefix != b'+' {
                old_seen += 1;
            }
            if prefix != b'-' {
                new_seen += 1;
            }
            rows.push(Row {
                prefix,
                payload: text[1..].to_string(),
                line: row_line,
            });
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.prefix == b'+' || r.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        if old_start < previous_old_end
            || (new_start as i128) != (old_start as i128 + net)
        {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = old_start + b;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            rows,
        });
        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    if hunks.is_empty() {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    if source.is_empty() {
        return Ok(Vec::new());
    }
    let lines: Vec<String> = source[..source.len() - 1]
        .split('\n')
        .map(str::to_string)
        .collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn abs_distance(a: i128, b: i128) -> i128 {
    (a - b).abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let mut lead = 0usize;
        while lead < hunk.rows.len() && hunk.rows[lead].prefix == b' ' {
            lead += 1;
        }
        let mut trail = 0usize;
        while trail < hunk.rows.len() - lead
            && hunk.rows[hunk.rows.len() - 1 - trail].prefix == b' '
        {
            trail += 1;
        }

        let expected = hunk.old_start as i128 + net + drift;
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_choices: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > lead + trail {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    if row.prefix == b' ' || row.prefix == b'-' {
                        old_payloads.push(row.payload.as_str());
                    }
                }
                let kept_old = hunk.old_count - l - r;
                if old_payloads.len() != kept_old {
                    continue;
                }

                if kept_old > working.len() {
                    continue;
                }
                for s in 0..=working.len() - kept_old {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_distance(s as i128, expected);
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    let matches = old_payloads
                        .iter()
                        .enumerate()
                        .all(|(idx, payload)| working[s + l + idx] == *payload);
                    if !matches {
                        continue;
                    }

                    let rank = (l + r, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_choices.clear();
                            best_choices.push((s, l, r));
                        }
                        Some(current) if rank < current => {
                            best_rank = Some(rank);
                            best_choices.clear();
                            best_choices.push((s, l, r));
                        }
                        Some(current) if rank == current => best_choices.push((s, l, r)),
                        _ => {}
                    }
                }
            }
        }

        if best_choices.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }

        let mut starts: Vec<usize> = best_choices.iter().map(|c| c.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: starts[0],
                second: starts[1],
            });
        }

        best_choices.sort_by_key(|choice| choice.1);
        let (s, l, r) = best_choices[0];
        let body = &hunk.rows[l..hunk.rows.len() - r];
        let replacement: Vec<String> = body
            .iter()
            .filter(|row| row.prefix == b' ' || row.prefix == b'+')
            .map(|row| row.payload.clone())
            .collect();

        let replace_start = s + l;
        let replace_end = s + hunk.old_count - r;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = s as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
