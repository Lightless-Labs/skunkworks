#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
enum Row {
    Context(String),
    Remove(String),
    Add(String),
}

struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.starts_with('0')) || !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(s: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = s.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, ending) = rest.split_once(" @@")?;
    if !ending.is_empty() {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if b.contains(',') || d.contains(',') {
        return None;
    }
    let (a, b, c, d) = (parse_number(a)?, parse_number(b)?, parse_number(c)?, parse_number(d)?);
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    Some((a, b, c, d))
}

fn nominal(start: usize, count: usize) -> i128 {
    if count > 0 { (start - 1) as i128 } else { start as i128 }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader { line: patch.bytes().filter(|&b| b == b'\n').count() + 1 });
    }

    let lines: Vec<&str> = patch[..patch.len() - 1].split('\n').collect();
    for (idx, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(idx) != Some(expected) {
            return Err(ApplyError::BadHeader { line: idx + 1 });
        }
    }
    if lines.len() <= 2 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end: Option<i128> = None;
    let mut net: i128 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        let (a, b, c, d) = parse_hunk_header(lines[i])
            .ok_or(ApplyError::BadHunkHeader { line: header_line })?;
        let old_start = nominal(a, b);
        let new_start = nominal(c, d);
        i += 1;

        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let line = lines[i];
            let payload = line.get(1..).ok_or(ApplyError::BadBody { line: i + 1 })?;
            if payload.contains('\r') {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            match line.as_bytes().first().copied() {
                Some(b' ') => {
                    old_seen += 1;
                    new_seen += 1;
                    rows.push(Row::Context(payload.to_owned()));
                }
                Some(b'-') => {
                    old_seen += 1;
                    rows.push(Row::Remove(payload.to_owned()));
                }
                Some(b'+') => {
                    new_seen += 1;
                    rows.push(Row::Add(payload.to_owned()));
                }
                _ => return Err(ApplyError::BadBody { line: i + 1 }),
            }
            i += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| matches!(r, Row::Add(_) | Row::Remove(_))) {
            return Err(ApplyError::BadBody { line: header_line });
        }
        if let Some(end) = previous_old_end {
            if old_start < end || new_start != old_start + net {
                return Err(ApplyError::Misordered { line: header_line });
            }
        } else if new_start != old_start {
            return Err(ApplyError::Misordered { line: header_line });
        }
        previous_old_end = Some(old_start + b as i128);
        net += d as i128 - b as i128;

        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }
        hunks.push(Hunk {
            line: header_line,
            old_start: old_start as usize,
            old_count: b,
            new_start: new_start as usize,
            new_count: d,
            rows,
        });
    }
    Ok(hunks)
}

pub fn apply_unified(source: &str, patch: &str, options: PatchOptions) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;

    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut working: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1].split('\n').map(str::to_owned).collect()
    };
    if working.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let lead = hunk.rows.iter().take_while(|r| matches!(r, Row::Context(_))).count();
        let trail = hunk.rows.iter().rev().take_while(|r| matches!(r, Row::Context(_))).count();
        let expected = hunk.old_start as i128 + net + drift;

        let mut best_rank: Option<(usize, i128)> = None;
        let mut best: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=options.fuzz.min(lead) {
            for r in 0..=options.fuzz.min(trail) {
                if l + r > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    match row {
                        Row::Context(s) | Row::Remove(s) => old_payloads.push(s.as_str()),
                        Row::Add(_) => {}
                    }
                }
                let mut new_payloads = Vec::new();
                for row in &hunk.rows[l..hunk.rows.len() - r] {
                    match row {
                        Row::Context(s) | Row::Add(s) => new_payloads.push(s.clone()),
                        Row::Remove(_) => {}
                    }
                }
                let remaining_old = hunk.old_count - l - r;
                for s in 0..=working.len().saturating_sub(hunk.old_count) {
                    if s + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = (s as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    if old_payloads.len() != remaining_old {
                        continue;
                    }
                    if old_payloads.iter().enumerate().all(|(idx, p)| working[s + l + idx] == *p) {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(br) if rank < br => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(br) if rank == br => best.push((s, l, r)),
                            _ => {}
                        }
                    }
                }
                let _ = new_payloads;
            }
        }

        if best.is_empty() {
            return Err(ApplyError::NoMatch { hunk: hunk_index + 1 });
        }
        let mut starts: Vec<usize> = best.iter().map(|x| x.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: starts[0],
                second: starts[1],
            });
        }
        let s = starts[0];
        let (_, l, r) = best.into_iter().filter(|x| x.0 == s).min_by_key(|x| x.1).unwrap();

        let mut replacement = Vec::new();
        for row in &hunk.rows[l..hunk.rows.len() - r] {
            match row {
                Row::Context(text) | Row::Add(text) => replacement.push(text.clone()),
                Row::Remove(_) => {}
            }
        }
        let from = s + l;
        let to = s + hunk.old_count - r;
        working.splice(from..to, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        drift = s as i128 - (hunk.old_start as i128 + net);
        net += hunk.new_count as i128 - hunk.old_count as i128;
    }

    let mut out = String::new();
    for line in working {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}
