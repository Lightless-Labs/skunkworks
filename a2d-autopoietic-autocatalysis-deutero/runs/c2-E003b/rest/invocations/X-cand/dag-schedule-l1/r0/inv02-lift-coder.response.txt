```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Plan {
    pub order: Vec<usize>,
    pub times: Vec<Timing>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Overflow {
        task: usize,
    },
}

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let task_count = tasks.len();

    // Validate every dependency before performing scheduling work.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &dependency) in task.deps.iter().enumerate() {
            if dependency >= task_count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
            if dependency == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&dependency) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value: dependency,
                });
            }
        }
    }

    // Build successor lists and indegrees.
    let mut children = vec![Vec::new(); task_count];
    let mut indegree = vec![0usize; task_count];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dependency in &task.deps {
            children[dependency].push(task_id);
        }
    }

    // Keep available IDs sorted so each step removes the smallest one.
    let mut available: Vec<usize> = (0..task_count)
        .filter(|&task_id| indegree[task_id] == 0)
        .collect();
    let mut order = Vec::with_capacity(task_count);

    while !available.is_empty() {
        let task_id = available.remove(0);
        order.push(task_id);

        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let insertion = available
                    .binary_search(&child)
                    .unwrap_or_else(|position| position);
                available.insert(insertion, child);
            }
        }
    }

    if order.len() != task_count {
        let removed = {
            let mut removed = vec![false; task_count];
            for &task_id in &order {
                removed[task_id] = true;
            }
            removed
        };
        let blocked = (0..task_count).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute timings in Kahn order. Each task's start is the maximum
    // finish among its predecessors.
    let mut times: Vec<Option<Timing>> = vec![None; task_count];
    let mut makespan = 0u64;

    for &task_id in &order {
        let mut start = 0u64;
        for &dependency in &tasks[task_id].deps {
            let finish = times[dependency]
                .as_ref()
                .expect("predecessor must be scheduled")
                .finish;
            start = start.max(finish);
        }

        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;

        makespan = makespan.max(finish);
        times[task_id] = Some(Timing { start, finish });
    }

    let times = times
        .into_iter()
        .map(|timing| timing.expect("all tasks must be scheduled"))
        .collect();

    Ok(Plan {
        order,
        times,
        makespan,
    })
}
```
