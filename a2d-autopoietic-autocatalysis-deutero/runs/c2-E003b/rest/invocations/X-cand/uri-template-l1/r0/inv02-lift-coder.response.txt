#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    names: Vec<(usize, String)>,
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let start = i;
                let close = match bytes[start + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => start + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[start + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: start + 1 + offset,
                    });
                }

                let mut content_start = start + 1;
                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                if matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                    return Err(Error::UnsupportedOperator { pos: content_start });
                }

                let content = &bytes[content_start..close];
                let mut names = Vec::new();
                let mut seen = Vec::<String>::new();
                let mut spec_start = content_start;

                for spec in content.split(|&b| b == b',') {
                    if spec.is_empty() {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let first_valid = |b: u8| b.is_ascii_alphabetic() || b == b'_';
                    let rest_valid = |b: u8| first_valid(b) || b.is_ascii_digit();

                    if !first_valid(spec[0]) {
                        return Err(Error::InvalidName { pos: spec_start });
                    }
                    if let Some(offset) = spec.iter().enumerate().skip(1).find_map(|(idx, &b)| {
                        if rest_valid(b) { None } else { Some(idx) }
                    }) {
                        return Err(Error::InvalidName {
                            pos: spec_start + offset,
                        });
                    }

                    let name = std::str::from_utf8(spec).expect("validated ASCII name").to_owned();
                    if seen.iter().any(|existing| existing == &name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name.clone());
                    names.push((spec_start, name));
                    spec_start += spec.len() + 1;
                }

                expressions.push(Expression {
                    start,
                    end: close,
                    names,
                });
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn encode(value: &str) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            let a = bytes[i + 1];
            let c = bytes[i + 2];
            if a.is_ascii_hexdigit() && c.is_ascii_hexdigit() {
                out.push('%');
                out.push((a as char).to_ascii_uppercase());
                out.push((c as char).to_ascii_uppercase());
                i += 3;
                continue;
            }
        }

        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }

    out
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut out = String::new();
    let mut cursor = 0;

    for expression in expressions {
        out.push_str(&template[cursor..expression.start]);
        let mut first = true;

        for (_, name) in expression.names {
            let value = vars.iter().rev().find(|(key, _)| *key == name).map(|(_, value)| *value);
            if let Some(value) = value {
                if !first {
                    out.push(',');
                }
                out.push_str(&encode(value));
                first = false;
            }
        }

        cursor = expression.end + 1;
    }

    out.push_str(&template[cursor..]);
    Ok(out)
}
