#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Expression {
    specs: Vec<(String, usize)>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut literal_start = 0;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }

                expressions.push((literal_start, open, parse_expression(bytes, open, close)?));
                i = close + 1;
                literal_start = i;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    for (literal_start, open, expression) in expressions {
        output.push_str(&template[cursor..open]);
        let mut first = true;
        for (name, _) in expression.specs {
            if let Some(value) = lookup(vars, &name) {
                if !first {
                    output.push(',');
                }
                first = false;
                output.push_str(&encode(value));
            }
        }
        cursor = open + 1;
        while cursor < bytes.len() && bytes[cursor] != b'}' {
            cursor += 1;
        }
        cursor += 1;
        let _ = literal_start;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}

fn parse_expression(bytes: &[u8], open: usize, close: usize) -> Result<Expression, Error> {
    let mut start = open + 1;
    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(bytes[start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator { pos: start });
    }

    let mut specs = Vec::new();
    let mut seen: Vec<String> = Vec::new();
    let mut spec_start = start;

    loop {
        let mut end = spec_start;
        while end < close && bytes[end] != b',' {
            end += 1;
        }

        if end == spec_start {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name_bytes = &bytes[spec_start..end];
        for (offset, &byte) in name_bytes.iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName {
                    pos: spec_start + offset,
                });
            }
        }

        let name = std::str::from_utf8(name_bytes)
            .expect("validated variable names are ASCII")
            .to_owned();
        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.clone());
        specs.push((name, spec_start));

        if end == close {
            break;
        }
        spec_start = end + 1;
        if spec_start == close {
            return Err(Error::EmptyVariable { pos: close });
        }
    }

    start = close;
    let _ = start;
    Ok(Expression { specs })
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|&(key, value)| if key == name { Some(value) } else { None })
}

fn encode(value: &str) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' && i + 2 < bytes.len() && is_hex(bytes[i + 1]) && is_hex(bytes[i + 2]) {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let byte = bytes[i];
        if is_unreserved(byte) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}
