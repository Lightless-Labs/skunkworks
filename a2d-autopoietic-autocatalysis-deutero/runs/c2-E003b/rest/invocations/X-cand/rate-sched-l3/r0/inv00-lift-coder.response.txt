#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobData {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<JobData>,
    id_to_index: std::collections::HashMap<String, usize>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    has_cycle: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        use std::collections::{HashMap, HashSet};

        let mut id_to_index = HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if id_to_index.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            id_to_index.insert(job.id.clone(), i);
        }

        let mut resolved_dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            let mut dependencies = Vec::with_capacity(job.dependencies.len());
            for (j, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match id_to_index.get(dependency) {
                    Some(&index) => dependencies.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            resolved_dependencies.push(dependencies);
        }

        if has_cycle(&resolved_dependencies) {
            return Err(Error::Cycle);
        }

        let data = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| JobData {
                job,
                dependencies: resolved_dependencies[i].clone(),
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs: data,
            id_to_index,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_cycle: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.has_cycle
            && self.jobs.iter().any(|job| {
                matches!(job.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for job in &mut self.jobs {
            job.state = State::Waiting { ready_at: now };
            job.attempts = 0;
        }
        self.has_cycle = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (i, job) in self.jobs.iter().enumerate() {
            let ready_at = match job.state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !job.dependencies.iter().all(|&dependency| {
                matches!(self.jobs[dependency].state, State::Succeeded)
            }) {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                let priority = u64::from(job.job.priority).saturating_add(age).min(9);
                priority as u8
            };

            match best {
                None => best = Some((i, effective)),
                Some((best_i, best_priority))
                    if effective > best_priority
                        || (effective == best_priority && i < best_i) =>
                {
                    best = Some((i, effective));
                }
                _ => {}
            }
        }

        let result = if let Some((i, _)) = best {
            let job = &mut self.jobs[i];
            job.attempts += 1;
            job.state = State::Running {
                attempt: job.attempts,
            };
            Some(Ticket {
                id: job.job.id.clone(),
                attempt: job.attempts,
            })
        } else {
            None
        };

        self.last_time = now;
        Ok(result)
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let index = *self.id_to_index.get(id).ok_or(Error::Unknown)?;
        if !matches!(self.jobs[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.jobs[index].state = State::Succeeded;
        } else if self.jobs[index].attempts == self.jobs[index].job.max_attempts {
            self.jobs[index].state = State::Failed;
            self.propagate_blocked();
        } else {
            let delay = retry_delay(
                self.retry_base,
                self.retry_cap,
                self.jobs[index].attempts,
            );
            self.jobs[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let index = *self.id_to_index.get(id).ok_or(Error::Unknown)?;
        Ok(self.jobs[index].state.clone())
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();
            for (i, job) in self.jobs.iter().enumerate() {
                if matches!(job.state, State::Waiting { .. })
                    && job.dependencies.iter().any(|&dependency| {
                        matches!(
                            self.jobs[dependency].state,
                            State::Failed | State::Blocked
                        )
                    })
                {
                    newly_blocked.push(i);
                }
            }
            if newly_blocked.is_empty() {
                break;
            }
            for i in newly_blocked {
                self.jobs[i].state = State::Blocked;
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    (1..=16).contains(&id.len())
        && id
            .bytes()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}

fn has_cycle(dependencies: &[Vec<usize>]) -> bool {
    fn visit(node: usize, dependencies: &[Vec<usize>], marks: &mut [u8]) -> bool {
        match marks[node] {
            1 => return true,
            2 => return false,
            _ => {}
        }
        marks[node] = 1;
        for &dependency in &dependencies[node] {
            if visit(dependency, dependencies, marks) {
                return true;
            }
        }
        marks[node] = 2;
        false
    }

    let mut marks = vec![0; dependencies.len()];
    for node in 0..dependencies.len() {
        if visit(node, dependencies, &mut marks) {
            return true;
        }
    }
    false
}

fn retry_delay(base: u64, cap: u64, attempt: u8) -> u64 {
    let mut delay = base;
    for _ in 1..attempt {
        if delay >= cap {
            return cap;
        }
        delay = delay.saturating_mul(2).min(cap);
    }
    delay.min(cap)
}
