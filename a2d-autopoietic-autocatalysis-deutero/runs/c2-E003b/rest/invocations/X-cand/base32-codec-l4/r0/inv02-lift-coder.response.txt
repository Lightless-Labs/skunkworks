#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    buffer: [u8; 5],
    buffered: usize,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            buffer: [0; 5],
            buffered: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        for &byte in input {
            self.buffer[self.buffered] = byte;
            self.buffered += 1;

            if self.buffered == 5 {
                encode_group(&self.buffer, 8, self.config.alphabet, &mut output);
                self.buffered = 0;
            }
        }
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if self.buffered != 0 {
            let symbols = (self.buffered * 8 + 4) / 5;
            encode_group(
                &self.buffer,
                symbols,
                self.config.alphabet,
                &mut output,
            );
            if self.config.padding == Padding::Required {
                while output.len() % 8 != 0 {
                    output.push('=');
                }
            }
        }

        self.buffered = 0;
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_group(bytes: &[u8; 5], symbols: usize, alphabet: Alphabet, out: &mut String) {
    let table = alphabet_bytes(alphabet);
    for i in 0..symbols {
        let bit_start = i * 5;
        let mut value = 0u8;
        for j in 0..5 {
            let bit_index = bit_start + j;
            value <<= 1;
            if bit_index < 40 {
                let byte = bytes[bit_index / 8];
                value |= (byte >> (7 - bit_index % 8)) & 1;
            }
        }
        out.push(table[value as usize] as char);
    }
}

#[derive(Clone)]
pub struct Decoder {
    config: Config,
    data_count: usize,
    bit_buffer: u32,
    bit_count: u8,
    total: usize,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            data_count: 0,
            bit_buffer: 0,
            bit_count: 0,
            total: 0,
            last_data_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: None,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.total;
            match classify(byte, next.config.alphabet) {
                Classification::Invalid => {
                    return Err(StreamError::InvalidByte { offset, byte });
                }
                Classification::Data(value, canonical) => {
                    if next.config.canonical && byte != canonical {
                        return Err(StreamError::NonCanonical {
                            offset,
                            expected: canonical,
                        });
                    }
                    if next.first_padding_offset.is_some() {
                        return Err(StreamError::Padding { offset });
                    }

                    next.data_count += 1;
                    next.last_data_offset = Some(offset);
                    next.bit_buffer = (next.bit_buffer << 5) | value as u32;
                    next.bit_count += 5;
                    while next.bit_count >= 8 {
                        next.bit_count -= 8;
                        output.push(((next.bit_buffer >> next.bit_count) & 0xff) as u8);
                    }
                    if next.bit_count == 0 {
                        next.bit_buffer = 0;
                    } else {
                        next.bit_buffer &= (1u32 << next.bit_count) - 1;
                    }
                }
                Classification::Padding => {
                    if next.config.padding == Padding::Omitted {
                        return Err(StreamError::Padding { offset });
                    }

                    if next.first_padding_offset.is_none() {
                        let residue = next.data_count % 8;
                        let expected = match residue {
                            2 => 6,
                            4 => 4,
                            5 => 3,
                            7 => 1,
                            _ => return Err(StreamError::Padding { offset }),
                        };
                        next.first_padding_offset = Some(offset);
                        next.expected_padding = Some(expected);
                    }

                    next.padding_count += 1;
                    if next.padding_count > next.expected_padding.unwrap_or(0) {
                        return Err(StreamError::Padding { offset });
                    }
                }
            }
            next.total += 1;
        }

        *self = next;
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
            return Err(StreamError::InvalidLength { offset: self.total });
        }

        if self.config.padding == Padding::Required {
            let expected = match residue {
                0 => 0,
                2 => 6,
                4 => 4,
                5 => 3,
                7 => 1,
                _ => unreachable!(),
            };
            if self.padding_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_padding_offset.unwrap_or(self.total),
                });
            }
        }

        let unused_bits = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused_bits != 0 && self.bit_buffer != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.total),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }
}

enum Classification {
    Data(u8, u8),
    Padding,
    Invalid,
}

fn classify(byte: u8, alphabet: Alphabet) -> Classification {
    if byte == b'=' {
        return Classification::Padding;
    }

    let upper = byte.to_ascii_uppercase();
    if let Some(value) = alphabet_bytes(alphabet).iter().position(|&b| b == upper) {
        Classification::Data(value as u8, upper)
    } else {
        Classification::Invalid
    }
}
