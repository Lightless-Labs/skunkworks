use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d <= 0 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        let n = (self.num as i128)
            .checked_mul(b)
            .and_then(|x| x.checked_add((rhs.num as i128).checked_mul(a)?))
            .ok_or(RationalError::Overflow)?;
        let d = a.checked_mul(rhs.den as i128).ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        let n = (self.num as i128)
            .checked_mul(b)
            .and_then(|x| x.checked_sub((rhs.num as i128).checked_mul(a)?))
            .ok_or(RationalError::Overflow)?;
        let d = a.checked_mul(rhs.den as i128).ok_or(RationalError::Overflow)?;
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.den as u128);
        let g2 = gcd_u128(rhs.num.unsigned_abs(), self.den as u128);
        let a = self.num as i128 / g1 as i128;
        let b = rhs.num as i128 / g2 as i128;
        let c = self.den as i128 / g2 as i128;
        let d = rhs.den as i128 / g1 as i128;
        rational_from_i128(a * b, c * d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.num.unsigned_abs());
        let g2 = gcd_u128(self.den as u128, rhs.den as u128);
        let mut a = self.num as i128 / g1 as i128;
        let mut b = rhs.den as i128 / g2 as i128;
        let c = self.den as i128 / g2 as i128;
        let mut d = rhs.num as i128 / g1 as i128;
        if d < 0 {
            a = -a;
            d = -d;
        }
        rational_from_i128(a * b, c * d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self {
            num: n,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(fail(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(fail(i, ParseKind::InvalidByte));
        }
        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
        }

        let unsigned_num_start = if bytes[0] == b'-' { 1 } else { 0 };
        let num_digits = num_end - unsigned_num_start;
        if num_digits >= 2 && bytes[unsigned_num_start] == b'0' {
            return Err(fail(unsigned_num_start + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits == 1 && bytes[1] == b'0' {
            return Err(fail(0, ParseKind::NonCanonical));
        }
        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(fail(ds + 1, ParseKind::NonCanonical));
            }
        }

        let neg = bytes[0] == b'-';
        let mut num_mag: u128 = 0;
        for &b in &bytes[unsigned_num_start..num_end] {
            num_mag = num_mag * 10 + (b - b'0') as u128;
            let limit = if neg { 1u128 << 63 } else { i64::MAX as u128 };
            if num_mag > limit {
                return Err(fail(0, ParseKind::OutOfRange));
            }
        }

        let mut den_value = 1u128;
        if let (Some(ds), Some(de)) = (den_start, den_end) {
            den_value = 0;
            for &b in &bytes[ds..de] {
                den_value = den_value * 10 + (b - b'0') as u128;
                if den_value > i64::MAX as u128 {
                    return Err(fail(ds, ParseKind::OutOfRange));
                }
            }
            if den_value == 0 {
                return Err(fail(ds, ParseKind::ZeroDenominator));
            }
            if den_value == 1 {
                return Err(fail(ds - 1, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(fail(ds - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_value) > 1 {
                return Err(fail(ds - 1, ParseKind::NonCanonical));
            }
        }

        let n = if neg {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };
        let d = den_value as i64;
        Ok(Self { num: n, den: d })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let len = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != len {
            return Err(MatrixError::DataLength {
                expected: len,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(out_len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let (matrix, pivots, swaps) = eliminate(self, self.cols)?;
        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut augmented = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                augmented.push(self.data[r * n + c]);
            }
            for c in 0..n {
                augmented.push(if r == c {
                    Rational { num: 1, den: 1 }
                } else {
                    Rational { num: 0, den: 1 }
                });
            }
        }
        let mat = Matrix {
            rows: n,
            cols: width,
            data: augmented,
        };
        let (reduced, pivots, _) = eliminate(&mat, n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            result.extend_from_slice(&reduced.data[r * width + n..r * width + width]);
        }
        Ok(Matrix {
            rows: n,
            cols: n,
            data: result,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = m.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..m {
            data.extend_from_slice(&self.data[r * n..r * n + n]);
            data.extend_from_slice(&rhs.data[r * k..r * k + k]);
        }
        let augmented = Matrix {
            rows: m,
            cols: width,
            data,
        };
        let (reduced, pivots, _) = eliminate(&augmented, n).map_err(SolveError::Matrix)?;

        for r in 0..m {
            let all_zero = (0..n).all(|c| reduced.data[r * width + c].num == 0);
            if all_zero {
                for c in 0..k {
                    let value = reduced.data[r * width + n + c];
                    if value.num != 0 {
                        return Ok(Solution::None {
                            row: r,
                            rhs_col: c,
                            witness: reduced.data[r * width..r * width + width].to_vec(),
                        });
                    }
                }
            }
        }

        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![Rational { num: 0, den: 1 }; particular_len];
        for (row, &pivot_col) in pivots.iter().enumerate() {
            for c in 0..k {
                particular[pivot_col * k + c] = reduced.data[row * width + n + c];
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular,
        };
        if pivots.len() == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots,
            });
        }

        let free: Vec<usize> = (0..n).filter(|c| !pivots.contains(c)).collect();
        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * f + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in pivots.iter().enumerate() {
                let coefficient = reduced.data[i * width + free_col];
                let neg = coefficient.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: j,
                        step: pivots.len(),
                        source,
                    })
                })?;
                basis[pivot_col * f + j] = neg;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: f,
                data: basis,
            },
            pivots,
        })
    }
}

fn eliminate(
    matrix: &Matrix,
    candidate_cols: usize,
) -> Result<(Matrix, Vec<usize>, Vec<(usize, usize)>), MatrixError> {
    let mut out = matrix.clone();
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for col in 0..candidate_cols {
        if p == out.rows {
            break;
        }
        let mut best: Option<usize> = None;
        for r in p..out.rows {
            let value = out.data[r * out.cols + col];
            if value.num == 0 {
                continue;
            }
            let better = match best {
                None => true,
                Some(b) => {
                    let prev = out.data[b * out.cols + col];
                    (value.den, value.num.unsigned_abs(), r)
                        < (prev.den, prev.num.unsigned_abs(), b)
                }
            };
            if better {
                best = Some(r);
            }
        }
        let Some(q) = best else {
            continue;
        };
        if q != p {
            for c in 0..out.cols {
                out.data.swap(p * out.cols + c, q * out.cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = out.data[p * out.cols + col];
        for c in 0..out.cols {
            let value = out.data[p * out.cols + c];
            out.data[p * out.cols + c] =
                value
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    })?;
        }

        for r in 0..out.rows {
            if r == p {
                continue;
            }
            let factor = out.data[r * out.cols + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..out.cols {
                let product = factor
                    .mul(&out.data[p * out.cols + c])
                    .map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: p,
                        source,
                    })?;
                out.data[r * out.cols + c] = out.data[r * out.cols + c]
                    .sub(&product)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: p,
                        source,
                    })?;
            }
        }
        pivots.push(col);
        p += 1;
    }
    Ok((out, pivots, swaps))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}
