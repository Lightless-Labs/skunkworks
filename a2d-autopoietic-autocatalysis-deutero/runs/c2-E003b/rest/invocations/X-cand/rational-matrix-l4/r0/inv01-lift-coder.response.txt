use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn abs_i64(n: i64) -> u128 {
    if n < 0 {
        (-(n as i128)) as u128
    } else {
        n as u128
    }
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        rational_from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse {
                        at: i,
                        kind: ParseKind::InvalidByte,
                    });
                }
                slash = Some(i);
                continue;
            }
            if b.is_ascii_digit() {
                continue;
            }
            if b == b'-' && i == 0 {
                continue;
            }
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let end_num = slash.unwrap_or(bytes.len());
        let digit_start = if bytes[0] == b'-' { 1 } else { 0 };
        if digit_start == end_num {
            return Err(RationalError::Parse {
                at: text.len(),
                kind: ParseKind::InvalidByte,
            });
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(RationalError::Parse {
                    at: text.len(),
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let num_digits = end_num - digit_start;
        if num_digits >= 2 && bytes[digit_start] == b'0' {
            return Err(RationalError::Parse {
                at: digit_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }
        if digit_start == 1 && num_digits == 1 && bytes[1] == b'0' {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }

        if let Some(s) = slash {
            let den_start = s + 1;
            if bytes.len() - den_start >= 2 && bytes[den_start] == b'0' {
                return Err(RationalError::Parse {
                    at: den_start + 1,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let negative = digit_start == 1;
        let limit: u128 = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut nmag = 0u128;
        for &b in &bytes[digit_start..end_num] {
            nmag = nmag.saturating_mul(10).saturating_add((b - b'0') as u128);
            if nmag > limit {
                return Err(RationalError::Parse {
                    at: 0,
                    kind: ParseKind::OutOfRange,
                });
            }
        }
        let numerator = if negative {
            if nmag == 1u128 << 63 {
                i64::MIN
            } else {
                -(nmag as i64)
            }
        } else {
            nmag as i64
        };

        let Some(s) = slash else {
            return Ok(Rational { num: numerator, den: 1 });
        };

        let den_start = s + 1;
        let mut dmag = 0u128;
        for &b in &bytes[den_start..] {
            dmag = dmag.saturating_mul(10).saturating_add((b - b'0') as u128);
            if dmag > i64::MAX as u128 {
                return Err(RationalError::Parse {
                    at: den_start,
                    kind: ParseKind::OutOfRange,
                });
            }
        }
        if dmag == 0 {
            return Err(RationalError::Parse {
                at: den_start,
                kind: ParseKind::ZeroDenominator,
            });
        }
        if dmag == 1 {
            return Err(RationalError::Parse {
                at: s,
                kind: ParseKind::NonCanonical,
            });
        }
        if numerator == 0 {
            return Err(RationalError::Parse {
                at: s,
                kind: ParseKind::NonCanonical,
            });
        }
        if gcd_u128(abs_i64(numerator), dmag) > 1 {
            return Err(RationalError::Parse {
                at: s,
                kind: ParseKind::NonCanonical,
            });
        }
        Ok(Rational {
            num: numerator,
            den: dmag as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let zero = Rational { num: 0, den: 1 };
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { one } else { zero });
            }
        }
        let augmented = Matrix {
            rows: n,
            cols: width,
            data,
        };
        let reduction = eliminate(augmented, n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            result.extend_from_slice(&reduction.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Matrix {
            rows: n,
            cols: n,
            data: result,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * n..r * n + n]);
            data.extend_from_slice(&rhs.data[r * k..r * k + k]);
        }
        let augmented = Matrix {
            rows: self.rows,
            cols: width,
            data,
        };
        let reduction = eliminate(augmented, n).map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            let coeff_zero = (0..n).all(|c| reduction.matrix.data[r * width + c].num == 0);
            if coeff_zero {
                if let Some(rhs_col) = (0..k).find(|&c| reduction.matrix.data[r * width + n + c].num != 0) {
                    return Ok(Solution::None {
                        row: r,
                        rhs_col,
                        witness: reduction.matrix.data[r * width..(r + 1) * width].to_vec(),
                    });
                }
            }
        }

        let rank = reduction.pivots.len();
        let mut free = Vec::with_capacity(n.saturating_sub(rank));
        let mut is_pivot = vec![false; n];
        for &p in &reduction.pivots {
            is_pivot[p] = true;
        }
        for c in 0..n {
            if !is_pivot[c] {
                free.push(c);
            }
        }

        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let zero = Rational { num: 0, den: 1 };
        let mut particular = vec![zero; particular_len];
        for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
            for c in 0..k {
                particular[pivot_col * k + c] =
                    reduction.matrix.data[pivot_row * width + n + c];
            }
        }
        let particular = Matrix {
            rows: n,
            cols: k,
            data: particular,
        };

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots: reduction.pivots,
            });
        }

        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![zero; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * f + j] = Rational { num: 1, den: 1 };
            for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
                let coefficient = reduction.matrix.data[pivot_row * width + free_col];
                basis[pivot_col * f + j] = coefficient.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: j,
                        step: rank,
                        source,
                    })
                })?;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix {
                rows: n,
                cols: f,
                data: basis,
            },
            pivots: reduction.pivots,
        })
    }
}

fn eliminate(mut matrix: Matrix, candidate_cols: usize) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let cols = matrix.cols;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for col in 0..candidate_cols {
        if p == rows {
            break;
        }
        let mut best: Option<usize> = None;
        for row in p..rows {
            let value = matrix.data[row * cols + col];
            if value.num == 0 {
                continue;
            }
            let key = (value.den, abs_i64(value.num), row);
            if best.map_or(true, |b| {
                let v = matrix.data[b * cols + col];
                key < (v.den, abs_i64(v.num), b)
            }) {
                best = Some(row);
            }
        }
        let Some(q) = best else {
            continue;
        };
        if q != p {
            for c in 0..cols {
                matrix.data.swap(p * cols + c, q * cols + c);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * cols + col];
        let pivot_row: Vec<Rational> = (0..cols).map(|c| matrix.data[p * cols + c]).collect();
        let mut normalized = Vec::with_capacity(cols);
        for c in 0..cols {
            normalized.push(pivot_row[c].div(&pivot).map_err(|source| {
                MatrixError::Arithmetic {
                    row: p,
                    col: c,
                    step: p,
                    source,
                }
            })?);
        }
        for c in 0..cols {
            matrix.data[p * cols + c] = normalized[c];
        }

        for row in 0..rows {
            if row == p {
                continue;
            }
            let factor = matrix.data[row * cols + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..cols {
                let product = factor.mul(&normalized[c]).map_err(|source| {
                    MatrixError::Arithmetic {
                        row,
                        col: c,
                        step: p,
                        source,
                    }
                })?;
                matrix.data[row * cols + c] =
                    matrix.data[row * cols + c].sub(&product).map_err(|source| {
                        MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        }
                    })?;
            }
        }
        pivots.push(col);
        p += 1;
    }

    Ok(Reduction {
        matrix,
        pivots,
        swaps,
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,
    },
}
