```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128_parts(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }

    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128
        || n > i64::MAX as i128
        || d < 1
        || d > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

fn gcd_i64_u64(num: i64, den: u64) -> u64 {
    gcd_u128(num.unsigned_abs() as u128, den as u128) as u64
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }

        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128
            || n > i64::MAX as i128
            || d < 1
            || d > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_i64_u64(self.den, rhs.den as u64) as i128;
        let left_scale = rhs.den as i128 / g;
        let right_scale = self.den as i128 / g;
        let n = self.num as i128 * left_scale + rhs.num as i128 * right_scale;
        let d = self.den as i128 * left_scale;
        from_i128_parts(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_i64_u64(self.den, rhs.den as u64) as i128;
        let left_scale = rhs.den as i128 / g;
        let right_scale = self.den as i128 / g;
        let n = self.num as i128 * left_scale - rhs.num as i128 * right_scale;
        let d = self.den as i128 * left_scale;
        from_i128_parts(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128) as i128;
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128) as i128;

        let a = self.num as i128 / g1;
        let b = rhs.num as i128 / g2;
        let c = self.den as i128 / g2;
        let d = rhs.den as i128 / g1;

        from_i128_parts(a * b, c * d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.num.unsigned_abs() as u128) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;

        let a = self.num as i128 / g1;
        let b = rhs.den as i128 / g2;
        let c = self.den as i128 / g2;
        let d = rhs.num as i128 / g1;

        from_i128_parts(a * b, c * d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        match self.num.checked_neg() {
            Some(num) => Ok(Self { num, den: self.den }),
            None => Err(RationalError::Overflow),
        }
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse {
                        at: i,
                        kind: ParseKind::InvalidByte,
                    });
                }
                slash = Some(i);
            } else if b.is_ascii_digit() {
                // A digit is allowed in either token.
            } else if b == b'-' && i == 0 {
                // Only the numerator may begin with a minus.
            } else {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_start = usize::from(bytes[0] == b'-');
        if num_start == num_end {
            return Err(RationalError::Parse {
                at: bytes.len(),
                kind: ParseKind::InvalidByte,
            });
        }

        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(RationalError::Parse {
                    at: bytes.len(),
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let num_digits = &bytes[num_start..num_end];
        let den_digits = slash.map(|s| &bytes[s + 1..]);

        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(RationalError::Parse {
                at: num_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }

        if bytes[0] == b'-' && num_digits == b"0" {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }

        if let (Some(s), Some(digits)) = (slash, den_digits) {
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(RationalError::Parse {
                    at: s + 2,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let negative = bytes[0] == b'-';
        let limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };

        let num_mag = match parse_magnitude(num_digits, limit) {
            Some(v) => v,
            None => {
                return Err(RationalError::Parse {
                    at: 0,
                    kind: ParseKind::OutOfRange,
                });
            }
        };

        let den_mag = if let (Some(s), Some(digits)) = (slash, den_digits) {
            match parse_magnitude(digits, i64::MAX as u128) {
                Some(v) => Some((v, s + 1)),
                None => {
                    return Err(RationalError::Parse {
                        at: s + 1,
                        kind: ParseKind::OutOfRange,
                    });
                }
            }
        } else {
            None
        };

        if let Some((0, at)) = den_mag {
            return Err(RationalError::Parse {
                at,
                kind: ParseKind::ZeroDenominator,
            });
        }

        if let (Some(s), Some((d, _))) = (slash, den_mag) {
            if d == 1 {
                return Err(RationalError::Parse {
                    at: s,
                    kind: ParseKind::NonCanonical,
                });
            }
            if num_mag == 0 {
                return Err(RationalError::Parse {
                    at: s,
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd_u128(num_mag, d) > 1 {
                return Err(RationalError::Parse {
                    at: s,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let n = if negative {
            if num_mag == 1u128 << 63 {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        Ok(Self {
            num: n,
            den: den_mag.map_or(1, |(d, _)| d as i64),
        })
    }
}

fn parse_magnitude(digits: &[u8], limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for &b in digits {
        let digit = (b - b'0') as u128;
        value = value.checked_mul(10)?.checked_add(digit)?;
        if value > limit {
            return None;
        }
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
