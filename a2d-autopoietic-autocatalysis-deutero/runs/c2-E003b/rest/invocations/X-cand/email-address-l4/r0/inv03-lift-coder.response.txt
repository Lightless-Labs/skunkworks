#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize, usize)>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !input.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while input.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut out = Vec::new();
    let mut positions = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected = false;
    let mut quote_open = None;
    let mut comment_depth = 0usize;
    let mut outer_comment = 0usize;
    let mut run_start: Option<usize> = None;
    let mut run_end = 0usize;
    let mut run_clean_start = 0usize;

    let flush_run = |runs: &mut Vec<(usize, usize, usize)>,
                     run_start: &mut Option<usize>,
                     run_end: usize,
                     run_clean_start: usize| {
        if let Some(start) = run_start.take() {
            runs.push((start, run_end, run_clean_start));
        }
    };

    while i < input.len() {
        let b = input[i];

        if protected {
            out.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < input.len() {
                    out.push(input[i]);
                    positions.push(i);
                    i += 1;
                }
                continue;
            }
            if b == b'"' {
                protected = false;
                quote_open = None;
            }
            i += 1;
            continue;
        }

        let begins_protected = comment_depth == 0
            && out.is_empty()
            && b == b'"';
        if begins_protected {
            flush_run(&mut runs, &mut run_start, run_end, run_clean_start);
            protected = true;
            quote_open = Some(i);
            out.push(b);
            positions.push(i);
            i += 1;
            continue;
        }

        if comment_depth > 0 {
            if b == b'\\' {
                flush_run(&mut runs, &mut run_start, run_end, run_clean_start);
                if i + 1 >= input.len()
                    || !(0x20..=0x7e).contains(&input[i + 1])
                {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    if run_start.is_none() {
                        run_start = Some(i);
                        run_clean_start = out.len();
                    }
                    run_end = end;
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if comment_depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                comment_depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                comment_depth -= 1;
                i += 1;
                continue;
            }
            if !(b == b'\t' || (0x20..=0x7e).contains(&b)) {
                return Err(EmailError::CommentChar { at: i });
            }
            if is_hws(b) {
                if run_start.is_none() {
                    run_start = Some(i);
                    run_clean_start = out.len();
                }
                run_end = i + 1;
            }
            i += 1;
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                if run_start.is_none() {
                    run_start = Some(i);
                    run_clean_start = out.len();
                }
                run_end = end;
                i = end;
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }

        if b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
                run_clean_start = out.len();
            }
            if comment_depth == 0 {
                outer_comment = i;
            }
            comment_depth = 1;
            i += 1;
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if is_hws(b) {
            if run_start.is_none() {
                run_start = Some(i);
                run_clean_start = out.len();
            }
            run_end = i + 1;
            i += 1;
            continue;
        }

        flush_run(&mut runs, &mut run_start, run_end, run_clean_start);
        out.push(b);
        positions.push(i);
        i += 1;
    }

    if comment_depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_comment });
    }
    if protected {
        let _ = quote_open;
    }
    flush_run(&mut runs, &mut run_start, run_end, run_clean_start);

    Ok(Cleaned {
        bytes: out,
        positions,
        runs,
    })
}

fn err_at(error: fn { at: usize } -> EmailError, at: usize) -> EmailError {
    error { at }
}

fn is_atom_special(b: u8) -> bool {
    b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || is_atom_special(b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut last_dot = true;
    for &b in bytes {
        if b == b'.' {
            if last_dot {
                return false;
            }
            last_dot = true;
        } else if is_atom_char(b) {
            last_dot = false;
        } else {
            return false;
        }
    }
    !last_dot
}

fn parse_cleaned(clean: Cleaned, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let bytes = &clean.bytes;
    for &(start, _end, clean_pos) in &clean.runs {
        let left = clean_pos > 0 && bytes.get(clean_pos - 1) == Some(&b'@');
        let right = bytes.get(clean_pos) == Some(&b'@');
        if clean_pos == 0 || clean_pos == bytes.len() || left || right {
            continue;
        }
        return Err(EmailError::CfwsBoundary { at: start });
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_close = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: clean.positions[0],
            });
        }
    }

    let mut ats = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !quote_close.map(|q| i <= q).unwrap_or(false) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: clean.positions[ats[1]],
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: clean.positions[sep],
        });
    }

    let mut local = Vec::new();
    let mut origins = Vec::new();
    let quoted;

    if bytes[0] == b'"' {
        let close = quote_close.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: clean.positions[close + 1],
            });
        }
        quoted = true;
        let mut i = 1;
        while i < close {
            let b = bytes[i];
            if b == b'\\' {
                let Some(&next) = bytes.get(i + 1) else {
                    return Err(EmailError::BadEscape {
                        at: clean.positions[i],
                    });
                };
                if !matches!(next, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: clean.positions[i],
                    });
                }
                local.push(next);
                origins.push(clean.positions[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: clean.positions[i],
                    });
                }
                local.push(b);
                origins.push(clean.positions[i]);
                i += 1;
            }
        }
        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: clean.positions[close],
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: clean.positions[64],
            });
        }
        let mut previous_dot = false;
        for i in 0..sep {
            let b = bytes[i];
            if b == b'.' {
                if i == 0 || previous_dot || i + 1 == sep {
                    return Err(EmailError::LocalDot {
                        at: clean.positions[i],
                    });
                }
                previous_dot = true;
            } else {
                if !is_atom_char(b) {
                    return Err(EmailError::LocalChar {
                        at: clean.positions[i],
                    });
                }
                previous_dot = false;
            }
            local.push(b);
            origins.push(clean.positions[i]);
        }
    }

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let raw_start_pos = if domain_start < bytes.len() {
        clean.positions[domain_start]
    } else {
        clean.positions.last().copied().unwrap_or(0) + 1
    };
    if raw_domain.is_empty() {
        return Err(EmailError::EmptyDomain { at: raw_start_pos });
    }
    let root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(root_dot);
    if domain_len == 0 {
        let at = if root_dot {
            clean.positions[bytes.len() - 1]
        } else {
            raw_start_pos
        };
        return Err(EmailError::EmptyDomain { at });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: clean.positions[domain_start + 253],
        });
    }

    let domain_end = domain_start + domain_len;
    let mut label_starts = Vec::new();
    let mut label_ends = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || bytes[i] == b'.' {
            if i == start {
                let at = if root_dot && i == domain_end {
                    clean.positions[bytes.len() - 1]
                } else if i < bytes.len() {
                    clean.positions[i]
                } else {
                    raw_start_pos
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let len = i - start;
            if len > 63 {
                return Err(EmailError::LabelTooLong {
                    at: clean.positions[start + 63],
                });
            }
            if bytes[start] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: clean.positions[start],
                });
            }
            if bytes[i - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: clean.positions[i - 1],
                });
            }
            for j in start..i {
                let b = bytes[j];
                if !(b.is_ascii_alphanumeric() || b == b'-') {
                    return Err(EmailError::DomainChar {
                        at: clean.positions[j],
                    });
                }
            }
            label_starts.push(start);
            label_ends.push(i);
            start = i + 1;
        }
    }

    if label_starts.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot { at: raw_start_pos });
    }

    let last_start = *label_starts.last().unwrap();
    let last_end = *label_ends.last().unwrap();
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: clean.positions[last_start],
        });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();
    Ok((
        Address {
            local: String::from_utf8(local).unwrap(),
            domain,
            quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let bytes = input.as_bytes();
    if let Some(at) = bytes.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let clean = cleanup(bytes)?;
    parse_cleaned(clean, false).map(|v| v.0)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    let bytes = input.as_bytes();
    if let Some(at) = bytes.iter().position(|b| *b >= 128) {
        return Err(EmailError::NonAscii { at });
    }
    let clean = cleanup(bytes)?;
    let (mut address, origins) = parse_cleaned(clean, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|b| *b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.first().copied().unwrap_or(input.len()),
                });
            }
            address.local.truncate(plus);
        }
    }

    let safe = valid_dot_atom(address.local.as_bytes());
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut result = String::new();
    if address.quoted {
        result.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                result.push('\\');
            }
            result.push(ch);
        }
        result.push('"');
    } else {
        result.push_str(&address.local);
    }
    result.push('@');
    result.push_str(&address.domain);
    result
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    duplicate_of[index] = Some(*first_index);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
