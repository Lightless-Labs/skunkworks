#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct CfwsRun {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<CfwsRun>,
}

fn is_horizontal(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn valid_fold(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut end = i + 2;
    if !input.get(end).copied().map(is_horizontal).unwrap_or(false) {
        return None;
    }
    while input.get(end).copied().map(is_horizontal).unwrap_or(false) {
        end += 1;
    }
    Some(end)
}

fn clean_cfws(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut in_protected_quote = false;
    let mut escaped = false;

    while i < input.len() {
        let b = input[i];

        if in_protected_quote {
            bytes.push(b);
            positions.push(i);
            i += 1;
            if escaped {
                escaped = false;
            } else if b == b'\\' {
                escaped = true;
            } else if b == b'"' {
                in_protected_quote = false;
            }
            continue;
        }

        let begins_run = is_horizontal(b)
            || b == b'('
            || b == b')'
            || b == b'\r'
            || b == b'\n';

        if begins_run {
            let run_start = i;
            let mut depth = 0usize;
            let mut outer_open = None;

            loop {
                if i >= input.len() {
                    break;
                }
                let c = input[i];

                if depth > 0 {
                    if c == b'\\' {
                        let slash = i;
                        if i + 1 >= input.len()
                            || !(0x20..=0x7e).contains(&input[i + 1])
                        {
                            return Err(EmailError::CommentEscape { at: slash });
                        }
                        i += 2;
                        continue;
                    }

                    if c == b'\r' || c == b'\n' {
                        if let Some(end) = valid_fold(input, i) {
                            i = end;
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }

                    if c == b'(' {
                        if depth == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        depth += 1;
                        i += 1;
                        continue;
                    }

                    if c == b')' {
                        depth -= 1;
                        i += 1;
                        continue;
                    }

                    if c == b'\t' || (0x20..=0x7e).contains(&c) {
                        i += 1;
                        continue;
                    }

                    return Err(EmailError::CommentChar { at: i });
                }

                if is_horizontal(c) {
                    i += 1;
                    continue;
                }

                if c == b'\r' || c == b'\n' {
                    if let Some(end) = valid_fold(input, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }

                if c == b'(' {
                    if outer_open.is_none() {
                        outer_open = Some(i);
                    }
                    depth = 1;
                    i += 1;
                    continue;
                }

                if c == b')' {
                    return Err(EmailError::UnexpectedClose { at: i });
                }

                break;
            }

            if depth > 0 {
                return Err(EmailError::UnclosedComment {
                    at: outer_open.unwrap_or(run_start),
                });
            }

            runs.push(CfwsRun {
                start: run_start,
                end: i,
            });
            continue;
        }

        bytes.push(b);
        positions.push(i);
        if bytes.len() == 1 && b == b'"' {
            in_protected_quote = true;
            escaped = false;
        }
        i += 1;
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn offset(cleaned: &Cleaned, index: usize, input_len: usize) -> usize {
    cleaned.positions.get(index).copied().unwrap_or(input_len)
}

fn is_atext(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for &b in bytes {
        if b == b'.' {
            if previous_dot {
                return false;
            }
            previous_dot = true;
        } else if is_atext(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    !previous_dot
}

fn parse_cleaned(
    input: &str,
    cleaned: &Cleaned,
    allow_single_label: bool,
) -> Result<(Address, usize), EmailError> {
    let b = &cleaned.bytes;
    let input_len = input.len();

    for run in &cleaned.runs {
        let left = cleaned
            .positions
            .iter()
            .position(|&p| p >= run.end);
        let right = cleaned
            .positions
            .iter()
            .rposition(|&p| p < run.start);

        let at_left = left.map(|i| b[i] == b'@').unwrap_or(false);
        let at_right = right.map(|i| b[i] == b'@').unwrap_or(false);
        let at_boundary = right.is_none() || left.is_none();

        if !(at_boundary || at_left || at_right) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if b[0] == b'"' {
        let mut i = 1usize;
        while i < b.len() {
            if b[i] == b'\\' {
                i += 2;
            } else if b[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: offset(cleaned, 0, input_len),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0usize;
    while i < b.len() {
        if let Some(end) = quote_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if b[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: offset(cleaned, separators[1], input_len),
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: offset(cleaned, sep, input_len),
        });
    }

    let mut local = Vec::new();
    let mut local_origins = Vec::new();
    let quoted;

    if b[0] == b'"' {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: offset(cleaned, end + 1, input_len),
            });
        }

        quoted = true;
        let mut p = 1usize;
        while p < end {
            if b[p] == b'\\' {
                let slash_pos = offset(cleaned, p, input_len);
                if p + 1 >= end {
                    return Err(EmailError::BadEscape { at: slash_pos });
                }
                let next = b[p + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: slash_pos });
                }
                local.push(next);
                local_origins.push(slash_pos);
                p += 2;
            } else {
                if !(0x20..=0x7e).contains(&b[p]) {
                    return Err(EmailError::QuotedChar {
                        at: offset(cleaned, p, input_len),
                    });
                }
                local.push(b[p]);
                local_origins.push(offset(cleaned, p, input_len));
                p += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: offset(cleaned, end, input_len),
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_origins[64],
            });
        }
    } else {
        quoted = false;
        let local_bytes = &b[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: offset(cleaned, 64, input_len),
            });
        }
        let mut previous_dot = false;
        for (idx, &c) in local_bytes.iter().enumerate() {
            if c == b'.' {
                if idx == 0 || idx + 1 == local_bytes.len() || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: offset(cleaned, idx, input_len),
                    });
                }
                previous_dot = true;
            } else {
                if !is_atext(c) {
                    return Err(EmailError::LocalChar {
                        at: offset(cleaned, idx, input_len),
                    });
                }
                previous_dot = false;
            }
        }
        local.extend_from_slice(local_bytes);
        for idx in 0..local_bytes.len() {
            local_origins.push(offset(cleaned, idx, input_len));
        }
    }

    let domain_start = sep + 1;
    let mut domain_end = b.len();
    let mut root_dot_pos = None;
    if domain_end > domain_start && b[domain_end - 1] == b'.' {
        root_dot_pos = Some(offset(cleaned, domain_end - 1, input_len));
        domain_end -= 1;
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: offset(cleaned, domain_start, input_len),
        });
    }

    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: offset(cleaned, domain_start + 253, input_len),
        });
    }

    let mut labels = Vec::new();
    let mut label_start = domain_start;
    let mut cursor = domain_start;
    while cursor <= domain_end {
        if cursor == domain_end || b[cursor] == b'.' {
            if cursor == label_start {
                let at = if cursor == domain_end {
                    root_dot_pos.unwrap_or_else(|| offset(cleaned, cursor, input_len))
                } else {
                    offset(cleaned, cursor, input_len)
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let label = &b[label_start..cursor];
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: offset(cleaned, label_start + 63, input_len),
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: offset(cleaned, label_start, input_len),
                });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: offset(cleaned, cursor - 1, input_len),
                });
            }
            for (j, &c) in label.iter().enumerate() {
                if !c.is_ascii_alphanumeric() && c != b'-' {
                    return Err(EmailError::DomainChar {
                        at: offset(cleaned, label_start + j, input_len),
                    });
                }
            }
            labels.push((label_start, cursor));
            label_start = cursor + 1;
        }
        cursor += 1;
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: offset(cleaned, domain_start, input_len),
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if b[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: offset(cleaned, last_start, input_len),
        });
    }

    let domain = b[domain_start..domain_end]
        .iter()
        .map(|c| c.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok((
        Address {
            local: String::from_utf8(local).unwrap(),
            domain,
            quoted,
        },
        local_origins.first().copied().unwrap_or(input_len),
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = clean_cfws(raw)?;
    parse_cleaned(input, &cleaned, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    if input.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let raw = input.as_bytes();
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = clean_cfws(raw)?;
    let (mut address, mut plus_origin) =
        parse_cleaned(input, &cleaned, policy.allow_single_label)?;

    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }

    if policy.strip_subaddress {
        if let Some(plus) = address.local.as_bytes().iter().position(|&b| b == b'+') {
            if plus == 0 {
                let original_plus = if address.quoted {
                    let mut decoded_index = 0usize;
                    let mut p = 1usize;
                    while p < cleaned.bytes.len() {
                        if cleaned.bytes[p] == b'@' {
                            break;
                        }
                        if cleaned.bytes[p] == b'\\' && p + 1 < cleaned.bytes.len() {
                            if decoded_index == plus {
                                plus_origin = cleaned.positions[p];
                                break;
                            }
                            decoded_index += 1;
                            p += 2;
                        } else {
                            if decoded_index == plus {
                                plus_origin = cleaned.positions[p];
                                break;
                            }
                            decoded_index += 1;
                            p += 1;
                        }
                    }
                    plus_origin
                } else {
                    cleaned.positions.get(plus).copied().unwrap_or(input.len())
                };
                return Err(EmailError::EmptyNormalizedLocal { at: original_plus });
            }
            address.local.truncate(plus);
        }
    }

    if !valid_dot_atom(address.local.as_bytes()) {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }

    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut output = String::new();
    if address.quoted {
        output.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                output.push('\\');
            }
            output.push(c);
        }
        output.push('"');
    } else {
        output.push_str(&address.local);
    }
    output.push('@');
    output.push_str(&address.domain);
    output
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    use std::collections::HashMap;

    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut first_seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = first_seen.get(&key) {
                    result.duplicate_of[index] = Some(first_index);
                } else {
                    first_seen.insert(key, index);
                    result.source_indices.push(index);
                    result.addresses.push(address);
                }
            }
        }
    }

    result
}
