#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct CfwsRun {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &str) -> Result<(Vec<u8>, Vec<usize>, Vec<CfwsRun>), EmailError> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut positions = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut in_quote = false;
    let mut quote_closed = false;
    let mut quote_escape = false;

    while i < bytes.len() {
        if in_quote {
            let b = bytes[i];
            out.push(b);
            positions.push(i);
            i += 1;
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                in_quote = false;
                quote_closed = true;
            }
            continue;
        }

        if !quote_closed && out.is_empty() && bytes[i] == b'"' {
            in_quote = true;
            out.push(bytes[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        let b = bytes[i];
        if is_hws(b) || b == b'\r' || b == b'\n' || b == b'(' {
            let start = i;
            let mut depth = 0usize;
            let mut comment_outer = None;
            loop {
                if i >= bytes.len() {
                    break;
                }
                let c = bytes[i];
                if depth > 0 {
                    if c == b'\\' {
                        let slash = i;
                        if i + 1 >= bytes.len()
                            || !(0x20..=0x7e).contains(&bytes[i + 1])
                        {
                            return Err(EmailError::CommentEscape { at: slash });
                        }
                        i += 2;
                        continue;
                    }
                    if c == b'\r' || c == b'\n' {
                        if let Some(end) = fold_end(bytes, i) {
                            i = end;
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }
                    if c == b'(' {
                        if depth == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        depth += 1;
                        i += 1;
                        continue;
                    }
                    if c == b')' {
                        depth -= 1;
                        i += 1;
                        continue;
                    }
                    if !(c == b'\t' || (0x20..=0x7e).contains(&c)) {
                        return Err(EmailError::CommentChar { at: i });
                    }
                    i += 1;
                    continue;
                }

                if is_hws(c) {
                    i += 1;
                    continue;
                }
                if c == b'\r' || c == b'\n' {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }
                if c == b'(' {
                    comment_outer = Some(i);
                    depth = 1;
                    i += 1;
                    continue;
                }
                break;
            }
            if depth > 0 {
                return Err(EmailError::UnclosedComment {
                    at: comment_outer.unwrap_or(start),
                });
            }
            runs.push(CfwsRun { start, end: i });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        out.push(b);
        positions.push(i);
        i += 1;
    }

    Ok((out, positions, runs))
}

fn mapped(pos: &[usize], index: usize, input_len: usize) -> usize {
    pos.get(index).copied().unwrap_or(input_len)
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == bytes.len() {
                return false;
            }
            previous_dot = true;
        } else if is_atom_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_cleaned(
    input: &str,
    policy: &Policy,
    strict_single_label: bool,
) -> Result<(Address, Vec<usize>), EmailError> {
    let (bytes, positions, runs) = cleanup(input)?;

    for run in &runs {
        let left_at = run.start > 0 && input.as_bytes()[run.start - 1] == b'@';
        let right_at = run.end < input.len() && input.as_bytes()[run.end] == b'@';
        let at_boundary = run.start == 0 || run.end == input.len();
        if !at_boundary && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_close = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            if bytes[i] == b'\\' {
                i += 2;
            } else if bytes[i] == b'"' {
                quote_close = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: mapped(&positions, 0, input.len()),
            });
        }
    }

    let mut ats = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !quote_close.map(|q| i <= q).unwrap_or(false) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: mapped(&positions, ats[1], input.len()),
        });
    }
    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: mapped(&positions, sep, input.len()),
        });
    }

    let local_bytes = &bytes[..sep];
    let (mut local, quoted, origins) = if local_bytes[0] == b'"' {
        let close = quote_close.unwrap_or(0);
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: mapped(&positions, close + 1, input.len()),
            });
        }
        let mut decoded = Vec::new();
        let mut decoded_origins = Vec::new();
        let mut i = 1;
        while i < close {
            let b = local_bytes[i];
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape {
                        at: mapped(&positions, i, input.len()),
                    });
                }
                let next = local_bytes[i + 1];
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape {
                        at: mapped(&positions, i, input.len()),
                    });
                }
                decoded.push(next);
                decoded_origins.push(mapped(&positions, i, input.len()));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: mapped(&positions, i, input.len()),
                    });
                }
                decoded.push(b);
                decoded_origins.push(mapped(&positions, i, input.len()));
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: mapped(&positions, close, input.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: decoded_origins[64],
            });
        }
        (decoded, true, decoded_origins)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: mapped(&positions, 64, input.len()),
            });
        }
        let mut previous_dot = false;
        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: mapped(&positions, i, input.len()),
                    });
                }
                previous_dot = true;
            } else if is_atom_char(b) {
                previous_dot = false;
            } else {
                return Err(EmailError::LocalChar {
                    at: mapped(&positions, i, input.len()),
                });
            }
        }
        (
            local_bytes.to_vec(),
            false,
            (0..local_bytes.len())
                .map(|i| mapped(&positions, i, input.len()))
                .collect(),
        )
    };

    let domain_start = sep + 1;
    let mut domain_end = bytes.len();
    let root_dot = if domain_end > domain_start && bytes[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(domain_end)
    } else {
        None
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: mapped(&positions, domain_start, input.len()),
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: mapped(&positions, domain_start + 253, input.len()),
        });
    }

    let domain = &bytes[domain_start..domain_end];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0;
    for i in 0..=domain.len() {
        if i == domain.len() || domain[i] == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }

    for (idx, &(start, end)) in labels.iter().enumerate() {
        if start == end {
            let at = if idx + 1 == labels.len() {
                root_dot
                    .map(|r| mapped(&positions, r, input.len()))
                    .unwrap_or_else(|| mapped(&positions, domain_start + start, input.len()))
            } else {
                mapped(&positions, domain_start + start, input.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: mapped(&positions, domain_start + start + 63, input.len()),
            });
        }
        if domain[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: mapped(&positions, domain_start + start, input.len()),
            });
        }
        if domain[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: mapped(&positions, domain_start + end - 1, input.len()),
            });
        }
        for i in start..end {
            let b = domain[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: mapped(&positions, domain_start + i, input.len()),
                });
            }
        }
    }

    if strict_single_label && labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: mapped(&positions, domain_start, input.len()),
        });
    }
    if !strict_single_label && !policy.allow_single_label && labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: mapped(&positions, domain_start, input.len()),
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if domain[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: mapped(&positions, domain_start + last_start, input.len()),
        });
    }

    for b in &mut local {
        if policy.local_case == LocalCase::Lower && b.is_ascii_uppercase() {
            *b = b.to_ascii_lowercase();
        }
    }

    if policy.strip_subaddress {
        if let Some(plus) = local.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: origins.get(plus).copied().unwrap_or(input.len()),
                });
            }
            local.truncate(plus);
        }
    }

    let safe = valid_dot_atom(&local);
    let final_quoted = if !safe {
        true
    } else if policy.unquote_safe {
        false
    } else {
        quoted
    };

    let normalized_domain = domain.iter().map(|b| b.to_ascii_lowercase()).collect();
    Ok((
        Address {
            local: String::from_utf8(local).unwrap_or_default(),
            domain: String::from_utf8(normalized_domain).unwrap_or_default(),
            quoted: final_quoted,
        },
        origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let policy = Policy::default();
    parse_cleaned(input, &policy, true).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    parse_cleaned(input, policy, false).map(|(address, _)| address)
}

pub fn format_email(address: &Address) -> String {
    if !address.quoted {
        format!("{}@{}", address.local, address.domain)
    } else {
        let mut out = String::with_capacity(address.local.len() + address.domain.len() + 3);
        out.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
        out.push('@');
        out.push_str(&address.domain);
        out
    }
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut result = DedupResult {
        addresses: Vec::new(),
        source_indices: Vec::new(),
        duplicate_of: vec![None; inputs.len()],
        errors: Vec::new(),
    };
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => result.errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first_index)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    result.duplicate_of[index] = Some(*first_index);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    result.source_indices.push(index);
                    result.addresses.push(address);
                }
            }
        }
    }
    result
}
