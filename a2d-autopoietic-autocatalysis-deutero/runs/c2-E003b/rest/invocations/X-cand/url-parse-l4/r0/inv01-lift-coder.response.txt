use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let lo = hex_value(bytes[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = origins.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let additionally_safe = match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        };
        if unreserved || additionally_safe {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(char::from(b"0123456789ABCDEF"[(b >> 4) as usize]));
            out.push(char::from(b"0123456789ABCDEF"[(b & 0x0f) as usize]));
        }
    }
    out
}

fn validate_global(input: &str) -> Result<(), ParseError> {
    for (i, &b) in input.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn parse_port(input: &str, start: usize) -> Result<u16, ParseError> {
    if input.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in input.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    if input.len() >= 2 && input.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }

    let mut value: u32 = 0;
    for b in input.bytes() {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(value as u16)
}

fn map_codec_to_parse(error: CodecError, offset: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    validate_global(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let mut at_positions = authority.match_indices('@').map(|(i, _)| i);
    let first_at = at_positions.next();
    if let Some(second) = at_positions.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second,
        });
    }
    let (userinfo, hostport, hostport_start) = match first_at {
        Some(i) if i == 0 => {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + i,
            });
        }
        Some(i) => (
            Some(authority[..i].to_owned()),
            &authority[i + 1..],
            authority_start + i + 1,
        ),
        None => (None, authority, authority_start),
    };

    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let open = hostport_start;
        let close = match hostport[1..].find(']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close + 1,
            });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), port)
    } else {
        let split = hostport.find(':');
        let (host_raw, port_raw, port_start) = match split {
            Some(i) => (&hostport[..i], Some(&hostport[i + 1..]), hostport_start + i + 1),
            None => (hostport, None, hostport_start),
        };
        if host_raw.is_empty() {
            return Err(ParseError::InvalidHost { at: hostport_start });
        }
        for (i, b) in host_raw.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_start + i,
                });
            }
        }
        let port = match port_raw {
            Some(p) => Some(parse_port(p, port_start)?),
            None => None,
        };
        (host_raw.to_ascii_lowercase(), port)
    };

    let fragment_pos = input[authority_end..]
        .find('#')
        .map(|i| authority_end + i);
    let before_fragment_end = fragment_pos.unwrap_or(bytes.len());
    let query_pos = input[authority_end..before_fragment_end]
        .find('?')
        .map(|i| authority_end + i);

    let path_end = query_pos.unwrap_or(before_fragment_end);
    let path_raw = &input[authority_end..path_end];
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };
    let query = query_pos.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_pos.map(|f| input[f + 1..].to_owned());

    if let Some(ref user) = userinfo {
        let user_offset = authority_start;
        decode_component(Component::UserInfo, user)
            .map_err(|e| map_codec_to_parse(e, user_offset))?;
    }
    if !path_raw.is_empty() {
        decode_component(Component::Path, path_raw)
            .map_err(|e| map_codec_to_parse(e, authority_end))?;
    }
    if let (Some(q), Some(pos)) = (&query, query_pos) {
        decode_component(Component::Query, q)
            .map_err(|e| map_codec_to_parse(e, pos + 1))?;
    }
    if let (Some(f), Some(pos)) = (&fragment, fragment_pos) {
        decode_component(Component::Fragment, f)
            .map_err(|e| map_codec_to_parse(e, pos + 1))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded = decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded[1..].split('/').collect();
    let terminal_navigation = decoded_segments
        .last()
        .map(|s| *s == "." || *s == "..")
        .unwrap_or(false);

    let mut stack: Vec<&str> = Vec::new();
    for (raw, dec) in raw_segments.iter().zip(decoded_segments.iter()) {
        if *dec == "." {
            continue;
        }
        if *dec == ".." {
            stack.pop();
        } else {
            stack.push(raw);
        }
    }
    if terminal_navigation {
        stack.push("");
    }
    let joined = stack.join("/");
    if joined.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{joined}"))
    }
}

fn scan_reference(input: &str) -> Result<(), ParseError> {
    validate_global(input)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    scan_reference(reference).map_err(ResolveError::Reference)?;

    let boundary = reference
        .char_indices()
        .find(|(_, c)| matches!(c, '/' | '?' | '#'))
        .map(|(i, _)| i)
        .unwrap_or(reference.len());
    if reference[..boundary].contains(':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let fragment_pos = reference.find('#');
    let before_fragment = fragment_pos.unwrap_or(reference.len());
    let query_pos = reference[..before_fragment].find('?');
    let path_end = query_pos.unwrap_or(before_fragment);
    let ref_path = &reference[..path_end];
    let ref_query = query_pos.map(|q| reference[q + 1..before_fragment].to_owned());
    let ref_fragment = fragment_pos.map(|f| reference[f + 1..].to_owned());

    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_path.clone(),
            if query_pos.is_none() {
                base_url.query.clone()
            } else {
                ref_query
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
            format!("{slash}{ref_path}")
        };
        (merged, ref_query)
    };

    let mut absolute = format!("{}://", base_url.scheme);
    if let Some(user) = &base_url.userinfo {
        absolute.push_str(user);
        absolute.push('@');
    }
    if base_url.ipv6 {
        absolute.push('[');
        absolute.push_str(&base_url.host);
        absolute.push(']');
    } else {
        absolute.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        absolute.push(':');
        absolute.push_str(&port.to_string());
    }
    absolute.push_str(&selected_path);
    if let Some(query) = &selected_query {
        absolute.push('?');
        absolute.push_str(query);
    }
    if let Some(fragment) = &ref_fragment {
        absolute.push('#');
        absolute.push_str(fragment);
    }

    let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn form_decode(s: &str) -> String {
    s.replace('+', " ")
}

fn map_codec_to_query(error: CodecError, offset: usize) -> QueryError {
    match error {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut offset = 0;

    for field in input.split('&') {
        let field_start = offset;
        offset += field.len() + 1;
        if field.is_empty() {
            continue;
        }

        let eq = field.find('=');
        let (key_raw, value_raw, value_offset) = match eq {
            Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
            None => (field, "", field_start + field.len()),
        };

        let key = decode_component(Component::Query, &form_decode(key_raw))
            .map_err(|e| map_codec_to_query(e, field_start))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }
        let value = decode_component(Component::Query, &form_decode(value_raw))
            .map_err(|e| map_codec_to_query(e, value_offset))?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }

    Ok(QueryMap { entries })
}

fn form_encode(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(char::from(b"0123456789ABCDEF"[(b >> 4) as usize]));
            out.push(char::from(b"0123456789ABCDEF"[(b & 0x0f) as usize]));
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(&str, Vec<&str>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| *k == key) {
            existing.extend(values.iter().map(String::as_str));
        } else {
            merged.push((key.as_str(), values.iter().map(String::as_str).collect()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        for value in values {
            fields.push(format!("{}={}", form_encode(key), form_encode(value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, userinfo)
            .map_err(|e| CanonicalError::Parse(map_codec_to_parse(e, 0)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }

    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host).map_err(|_| {
            CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 })
        })?;
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }

    if let Some(port) = url.port {
        if !((url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443)) {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut canonical_segments = Vec::new();
    for segment in path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(match e {
                CodecError::BadEscape { at } => PathError::BadEscape { at },
                CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
            }))?;
        let mut encoded = String::new();
        for &b in decoded.as_bytes() {
            if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
                encoded.push(b as char);
            } else {
                encoded.push('%');
                encoded.push(char::from(b"0123456789ABCDEF"[(b >> 4) as usize]));
                encoded.push(char::from(b"0123456789ABCDEF"[(b & 0x0f) as usize]));
            }
        }
        canonical_segments.push(encoded);
    }
    out.push_str(&canonical_segments.join("/"));

    if let Some(map) = &query_map {
        out.push('?');
        out.push_str(&serialise_query(map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| CanonicalError::Parse(map_codec_to_parse(e, 0)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }

    Ok(out)
}
