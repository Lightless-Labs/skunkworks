use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct QueryMap {
    pub entries: Vec<(String, Vec<String>)>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum QueryError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
    EmptyKey { at: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CanonicalError {
    Parse(ParseError),
    Path(PathError),
    Query(QueryError),
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let src = input.as_bytes();
    let mut bytes = Vec::with_capacity(src.len());
    let mut offsets = Vec::with_capacity(src.len());
    let mut i = 0;

    while i < src.len() {
        if src[i] == b'%' {
            if i + 2 >= src.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(src[i + 1]);
            let lo = hex_value(src[i + 2]);
            if hi.is_none() || lo.is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            bytes.push((hi.unwrap() << 4) | lo.unwrap());
            offsets.push(i);
            i += 3;
        } else {
            bytes.push(src[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&bytes) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = offsets.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = is_unreserved(b)
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn scan_global(input: &str) -> Result<(), ParseError> {
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn map_codec_parse(err: CodecError, offset: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_global(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];
    let auth_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }
    let (userinfo, host_start_rel) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_start = authority_start + host_start_rel;
    let hostport = &authority[host_start_rel..];
    let (host, port, ipv6) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], host_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport.as_bytes().iter().position(|&b| b == b':');
        let (raw_host, raw_port) = match split {
            Some(i) => (&hostport[..i], Some((&hostport[i + 1..], host_start + i + 1))),
            None => (hostport, None),
        };
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, b) in raw_host.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_start + i });
            }
        }
        let port = match raw_port {
            Some((p, pos)) => Some(parse_port(p, pos)?),
            None => None,
        };
        (raw_host.to_ascii_lowercase(), port, false)
    };

    let component_end = bytes[authority_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|i| authority_end + i)
        .unwrap_or(bytes.len());
    let fragment = if component_end < bytes.len() {
        Some(input[component_end + 1..].to_owned())
    } else {
        None
    };
    let before_fragment = &input[authority_end..component_end];
    let before_bytes = before_fragment.as_bytes();
    let query_rel = before_bytes.iter().position(|&b| b == b'?');
    let (path_raw, query) = match query_rel {
        Some(i) => (
            &before_fragment[..i],
            Some(before_fragment[i + 1..].to_owned()),
        ),
        None => (before_fragment, None),
    };
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    if let Some(ui) = &userinfo {
        let offset = authority_start;
        decode_component(Component::UserInfo, ui)
            .map_err(|e| map_codec_parse(e, offset))?;
    }
    if !path_raw.is_empty() {
        decode_component(Component::Path, path_raw)
            .map_err(|e| map_codec_parse(e, authority_end))?;
    }
    if let Some(q) = &query {
        let query_offset = authority_end + query_rel.unwrap() + 1;
        decode_component(Component::Query, q)
            .map_err(|e| map_codec_parse(e, query_offset))?;
    }
    if let Some(f) = &fragment {
        decode_component(Component::Fragment, f)
            .map_err(|e| map_codec_parse(e, component_end + 1))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(port: &str, at: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + i });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }
    let mut value: u32 = 0;
    for b in port.bytes() {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u32);
        if value > 65535 {
            return Err(ParseError::InvalidPort { at });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(value as u16)
}

fn map_codec_path(err: CodecError) -> PathError {
    match err {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(map_codec_path)?;

    let mut segments: Vec<&str> = Vec::new();
    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut last_navigation = false;
    for segment in raw_segments {
        let decoded = decode_component(Component::Path, segment).map_err(map_codec_path)?;
        if decoded == "." {
            last_navigation = true;
        } else if decoded == ".." {
            if !segments.is_empty() {
                segments.pop();
            }
            last_navigation = true;
        } else {
            segments.push(segment);
            last_navigation = false;
        }
    }
    if last_navigation {
        segments.push("");
    }
    if segments.is_empty() {
        Ok("/".to_owned())
    } else {
        Ok(format!("/{}", segments.join("/")))
    }
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let mut base_url = parse_url(base).map_err(ResolveError::Base)?;
    base_url.path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    scan_global(reference).map_err(ResolveError::Reference)?;

    if let Some(colon) = reference.find(':') {
        let boundary = reference
            .find(|c| matches!(c, '/' | '?' | '#'))
            .unwrap_or(reference.len());
        if colon < boundary {
            let mut result = parse_url(reference).map_err(ResolveError::Reference)?;
            result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
            return Ok(result);
        }
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut result = parse_url(&absolute).map_err(ResolveError::Reference)?;
        result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
        return Ok(result);
    }

    let hash = reference.find('#');
    let (before_fragment, fragment) = match hash {
        Some(i) => (&reference[..i], Some(reference[i + 1..].to_owned())),
        None => (reference, None),
    };
    let qmark = before_fragment.find('?');
    let (ref_path, ref_query) = match qmark {
        Some(i) => (
            &before_fragment[..i],
            Some(before_fragment[i + 1..].to_owned()),
        ),
        None => (before_fragment, None),
    };

    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_url.path.clone(),
            if qmark.is_some() {
                ref_query
            } else {
                base_url.query.clone()
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_url.path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_url.path[..=slash], ref_path)
        };
        (merged, ref_query)
    };

    let mut absolute = format!("{}://", base_url.scheme);
    if let Some(ui) = &base_url.userinfo {
        absolute.push_str(ui);
        absolute.push('@');
    }
    if base_url.ipv6 {
        absolute.push('[');
        absolute.push_str(&base_url.host);
        absolute.push(']');
    } else {
        absolute.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        absolute.push(':');
        absolute.push_str(&port.to_string());
    }
    absolute.push_str(&selected_path);
    if let Some(q) = &selected_query {
        absolute.push('?');
        absolute.push_str(q);
    }
    if let Some(f) = &fragment {
        absolute.push('#');
        absolute.push_str(f);
    }

    let mut result = parse_url(&absolute).map_err(ResolveError::Reference)?;
    result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
    Ok(result)
}

fn map_codec_query(err: CodecError, offset: usize) -> QueryError {
    match err {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

fn plus_to_space(s: &str) -> String {
    s.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut field_start = 0;

    for field in input.split('&') {
        let field_len = field.len();
        if !field.is_empty() {
            let eq = field.find('=');
            let (raw_key, raw_value, value_offset) = match eq {
                Some(i) => (&field[..i], &field[i + 1..], field_start + i + 1),
                None => (field, "", field_start + field_len),
            };
            let key_input = plus_to_space(raw_key);
            let key = decode_component(Component::Query, &key_input)
                .map_err(|e| map_codec_query(e, field_start))?;
            if key.is_empty() {
                return Err(QueryError::EmptyKey { at: field_start });
            }
            let value_input = plus_to_space(raw_value);
            let value = decode_component(Component::Query, &value_input)
                .map_err(|e| map_codec_query(e, value_offset))?;

            if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
                values.push(value);
            } else {
                entries.push((key, vec![value]));
            }
        }
        field_start += field_len + 1;
    }
    Ok(QueryMap { entries })
}

fn encode_form(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(&str, Vec<&str>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, collected)) = merged.iter_mut().find(|(existing, _)| *existing == key) {
            collected.extend(values.iter().map(String::as_str));
        } else {
            merged.push((key, values.iter().map(String::as_str).collect()));
        }
    }

    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = encode_form(key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, encode_form(value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let normalized_path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query_map = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, userinfo).map_err(|e| {
            CanonicalError::Parse(match e {
                CodecError::BadEscape { at } => ParseError::BadEscape { at },
                CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at },
            })
        })?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }

    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host).expect("validated IPv6 address");
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }

    if let Some(port) = url.port {
        let default = (url.scheme == "http" && port == 80)
            || (url.scheme == "https" && port == 443);
        if !default {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut canonical_segments = Vec::new();
    for segment in normalized_path.split('/') {
        let decoded = decode_component(Component::Path, segment).map_err(|e| {
            CanonicalError::Path(match e {
                CodecError::BadEscape { at } => PathError::BadEscape { at },
                CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
            })
        })?;
        canonical_segments.push(encode_unreserved_only(&decoded));
    }
    out.push_str(&canonical_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment).map_err(|e| {
            CanonicalError::Parse(match e {
                CodecError::BadEscape { at } => ParseError::BadEscape { at },
                CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at },
            })
        })?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn encode_unreserved_only(text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        if is_unreserved(b) {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}
