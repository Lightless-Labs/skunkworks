#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|component| !component.is_empty())
            .next_back()
            .is_some_and(|component| component == "." || component == "..")
}

fn reduce_components<'a>(
    components: impl IntoIterator<Item = (&'a str, usize)>,
    absolute: bool,
) -> Result<Vec<&'a str>, PathError> {
    let mut result = Vec::new();

    for (component, offset) in components {
        match component {
            "" | "." => {}
            ".." => {
                if result.last().is_some_and(|last| *last != "..") {
                    result.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: offset });
                } else {
                    result.push("..");
                }
            }
            _ => result.push(component),
        }
    }

    Ok(result)
}

fn join_result(components: &[&str], absolute: bool, directory: bool) -> String {
    let mut result = String::new();

    if absolute {
        result.push('/');
    }
    result.push_str(&components.join("/"));

    if result.is_empty() {
        result.push('.');
    }

    if directory && result != "/" && result != "." {
        result.push('/');
    }

    result
}

fn components_with_offsets(path: &str, start: usize) -> Vec<(&str, usize)> {
    let bytes = path.as_bytes();
    let mut components = Vec::new();
    let mut i = start;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let component_start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        components.push((&path[component_start..i], component_start));
    }

    components
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes()[0] == b'/';
    let components = reduce_components(components_with_offsets(path, 0), absolute)?;
    Ok(join_result(
        &components,
        absolute,
        directory_intent(path),
    ))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_normalized = if cwd.starts_with('/') {
        normalize(cwd).map_err(|_| PathError::InvalidCwd)?
    } else {
        return Err(PathError::InvalidCwd);
    };

    let home_normalized = match home {
        Some(home_path) => {
            if !home_path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(normalize(home_path).map_err(|_| PathError::InvalidHome)?)
        }
        None => None,
    };

    validate_raw(path)?;

    let path_bytes = path.as_bytes();
    let tilde_expanded = path_bytes[0] == b'~';

    let (base, request_start) = if tilde_expanded {
        if path_bytes.len() > 1 && path_bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let base = home_normalized.ok_or(PathError::MissingHome)?;
        let start = if path_bytes.len() > 1 { 2 } else { 1 };
        (base, start)
    } else if path_bytes[0] == b'/' {
        ("/".to_string(), 0)
    } else {
        (cwd_normalized, 0)
    };

    let mut components: Vec<&str> = base
        .split('/')
        .filter(|component| !component.is_empty())
        .collect();

    let request_components = components_with_offsets(path, request_start);
    for (component, offset) in request_components {
        match component {
            "" | "." => {}
            ".." => {
                if components.pop().is_none() {
                    return Err(PathError::Escape { at: offset });
                }
            }
            _ => components.push(component),
        }
    }

    Ok(join_result(
        &components,
        true,
        directory_intent(path),
    ))
}
