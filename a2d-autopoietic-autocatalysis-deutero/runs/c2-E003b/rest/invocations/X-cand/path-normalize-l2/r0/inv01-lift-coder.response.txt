#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn reduce(
    stack: &mut Vec<String>,
    path: &str,
    offset: usize,
    absolute: bool,
) -> Result<(), PathError> {
    for component in components(path) {
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| last != "..") {
                    stack.pop();
                } else if absolute {
                    let at = offset + path.find(component).unwrap_or(0);
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            ordinary => stack.push(ordinary.to_owned()),
        }
    }
    Ok(())
}

fn build_path(stack: &[String], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        let joined = stack.join("/");
        if joined.is_empty() {
            "/".to_owned()
        } else {
            format!("/{joined}")
        }
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if directory && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let directory = directory_intent(path);
    let mut stack = Vec::new();
    reduce(&mut stack, path, 0, absolute)?;

    Ok(build_path(&stack, absolute, directory))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }

    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized).into_iter().map(str::to_owned).collect())
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd)?;

    let home_components = match home {
        Some(home_path) => {
            if !home_path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            Some(
                normalize(home_path)
                    .map_err(|_| PathError::InvalidHome)?
                    .split('/')
                    .filter(|component| !component.is_empty())
                    .map(str::to_owned)
                    .collect::<Vec<_>>(),
            )
        }
        None => None,
    };

    validate_raw(path)?;

    let (base, remainder, remainder_offset, absolute) =
        if path.starts_with('~') {
            let bytes = path.as_bytes();
            if bytes.len() == 1 || bytes[1] == b'/' {
                let base = home_components.ok_or(PathError::MissingHome)?;
                let skip = if bytes.len() > 1 { 2 } else { 1 };
                (base, &path[skip..], skip, true)
            } else {
                return Err(PathError::UnsupportedTilde { at: 1 });
            }
        } else if path.starts_with('/') {
            (Vec::new(), path, 0, true)
        } else {
            (cwd_components, path, 0, true)
        };

    let directory = directory_intent(path);
    let mut stack = base;
    reduce(&mut stack, remainder, remainder_offset, absolute)?;

    Ok(build_path(&stack, true, directory))
}
