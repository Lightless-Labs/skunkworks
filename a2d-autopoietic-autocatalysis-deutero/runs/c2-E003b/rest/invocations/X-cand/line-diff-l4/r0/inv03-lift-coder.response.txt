#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone, Debug)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: MergeSide,
}

#[derive(Clone, Debug)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<Change>,
}

#[derive(Clone, Copy)]
enum Op {
    Keep,
    Delete,
    Insert,
}

fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text.strip_suffix('\n')
        .unwrap_or(text)
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn validate(base: &str, left: &str, right: &str) -> Result<(Vec<String>, Vec<String>, Vec<String>), MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.find('\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let b = split_lines(base);
    let l = split_lines(left);
    let r = split_lines(right);

    if b.len() > 128 || l.len() > 128 || r.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: b.len(),
            left_len: l.len(),
            right_len: r.len(),
        });
    }

    Ok((b, l, r))
}

fn canonical_changes(old: &[String], new: &[String], side: MergeSide) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut base_pos = 0usize;
    let mut changes = Vec::new();
    let mut active_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 active_start: &mut Option<usize>,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>,
                 side: &MergeSide| {
        if let Some(start) = active_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
                side: side.clone(),
            });
            *deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(&mut changes, &mut active_start, &mut deletes, &mut inserts, &side);
            i += 1;
            j += 1;
            base_pos += 1;
            continue;
        }

        let op = if i == n {
            Op::Insert
        } else if j == m {
            Op::Delete
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            if del > ins {
                Op::Delete
            } else if ins > del {
                Op::Insert
            } else if (i + j) % 2 == 0 {
                Op::Delete
            } else {
                Op::Insert
            }
        };

        active_start.get_or_insert(base_pos);
        match op {
            Op::Keep => unreachable!(),
            Op::Delete => {
                i += 1;
                base_pos += 1;
                deletes += 1;
            }
            Op::Insert => {
                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(&mut changes, &mut active_start, &mut deletes, &mut inserts, &side);
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn make_components(changes: &[Change]) -> Vec<Component> {
    let n = changes.len();
    let mut seen = vec![false; n];
    let mut components = Vec::new();

    for seed in 0..n {
        if seen[seed] {
            continue;
        }
        seen[seed] = true;
        let mut stack = vec![seed];
        let mut indices = Vec::new();

        while let Some(i) = stack.pop() {
            indices.push(i);
            for j in 0..n {
                if !seen[j] && interacts(&changes[i], &changes[j]) {
                    seen[j] = true;
                    stack.push(j);
                }
            }
        }

        let start = indices.iter().map(|&i| changes[i].start).min().unwrap();
        let end = indices.iter().map(|&i| changes[i].end).max().unwrap();
        let mut component_changes: Vec<Change> =
            indices.into_iter().map(|i| changes[i].clone()).collect();
        component_changes.sort_by_key(|c| (c.start, c.end));
        components.push(Component {
            start,
            end,
            changes: component_changes,
        });
    }

    components.sort_by_key(|c| (c.start, usize::from(c.start != c.end), c.end));
    components
}

fn projection(base: &[String], component: &Component, side: MergeSide) -> Vec<String> {
    let mut changes: Vec<&Change> = component
        .changes
        .iter()
        .filter(|c| c.side == side)
        .collect();
    changes.sort_by_key(|c| (c.start, c.end));

    let mut out = Vec::new();
    let mut cursor = component.start;

    for change in changes {
        if change.start < cursor {
            continue;
        }
        out.extend(base[cursor..change.start].iter().cloned());
        out.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    out.extend(base[cursor..component.end].iter().cloned());
    out
}

fn append_lines(out: &mut Vec<String>, lines: &[String]) {
    out.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let (base_lines, left_lines, right_lines) = validate(base, left, right)?;

    let left_changes = canonical_changes(&base_lines, &left_lines, MergeSide::Left);
    let right_changes = canonical_changes(&base_lines, &right_lines, MergeSide::Right);
    let mut all_changes = left_changes;
    all_changes.extend(right_changes);

    let components = make_components(&all_changes);
    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut cursor = 0usize;

    for component in components {
        output.extend(base_lines[cursor..component.start].iter().cloned());

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(&base_lines, &component, MergeSide::Left);
        let right_projection = projection(&base_lines, &component, MergeSide::Right);

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    let base_label_start = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {base_label_start},{}",
                        component.end - component.start
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    let output_len = output.len() - output_start;
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len,
                    });
                }
            }
        }

        cursor = component.end;
    }

    output.extend(base_lines[cursor..].iter().cloned());

    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
