#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: usize,
}

#[derive(Clone)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<Change>,
}

fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_changes(base: &[String], other: &[String], side: usize) -> Vec<Change> {
    let n = base.len();
    let m = other.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == other[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut i = 0;
    let mut j = 0;
    let mut changes = Vec::new();
    let mut active_start = 0;
    let mut deletes = 0;
    let mut inserts = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 active: &mut bool,
                 start: usize,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>| {
        if *active {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
                side,
            });
            *deletes = 0;
            *active = false;
        }
    };

    let mut active = false;
    while i < n || j < m {
        if i < n && j < m && base[i] == other[j] {
            flush(
                &mut changes,
                &mut active,
                active_start,
                &mut deletes,
                &mut inserts,
            );
            i += 1;
            j += 1;
        } else if i == n {
            if !active {
                active_start = i;
                active = true;
            }
            inserts.push(other[j].clone());
            j += 1;
        } else if j == m {
            if !active {
                active_start = i;
                active = true;
            }
            deletes += 1;
            i += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            let choose_delete = if delete_score > insert_score {
                true
            } else if insert_score > delete_score {
                false
            } else {
                (i + j) % 2 == 0
            };

            if !active {
                active_start = i;
                active = true;
            }
            if choose_delete {
                deletes += 1;
                i += 1;
            } else {
                inserts.push(other[j].clone());
                j += 1;
            }
        }
    }

    flush(
        &mut changes,
        &mut active,
        active_start,
        &mut deletes,
        &mut inserts,
    );
    changes
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;

    match (a_zero, b_zero) {
        (true, true) => a.start == b.start,
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn build_components(changes: &[Change]) -> Vec<Component> {
    let mut visited = vec![false; changes.len()];
    let mut components = Vec::new();

    for seed in 0..changes.len() {
        if visited[seed] {
            continue;
        }
        visited[seed] = true;
        let mut stack = vec![seed];
        let mut members = Vec::new();

        while let Some(current) = stack.pop() {
            members.push(changes[current].clone());
            for next in 0..changes.len() {
                if !visited[next] && interact(&changes[current], &changes[next]) {
                    visited[next] = true;
                    stack.push(next);
                }
            }
        }

        let start = members.iter().map(|c| c.start).min().unwrap();
        let end = members.iter().map(|c| c.end).max().unwrap();
        members.sort_by_key(|c| (c.start, c.end, c.side));
        components.push(Component {
            start,
            end,
            changes: members,
        });
    }

    components.sort_by_key(|c| (c.start, usize::from(c.end > c.start), c.end));
    components
}

fn project(base: &[String], component: &Component, side: usize) -> Vec<String> {
    let mut result = Vec::new();
    let mut cursor = component.start;

    for change in component.changes.iter().filter(|c| c.side == side) {
        while cursor < change.start {
            result.push(base[cursor].clone());
            cursor += 1;
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    while cursor < component.end {
        result.push(base[cursor].clone());
        cursor += 1;
    }
    result
}

fn append_line(lines: &mut Vec<String>, line: impl Into<String>) {
    lines.push(line.into());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.find('\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = edit_changes(&base_lines, &left_lines, 0);
    changes.extend(edit_changes(&base_lines, &right_lines, 1));
    let components = build_components(&changes);

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0;

    for component in components {
        while base_cursor < component.start {
            append_line(&mut output, base_lines[base_cursor].clone());
            base_cursor += 1;
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(&base_lines, &component, 0);
        let right_projection = project(&base_lines, &component, 1);

        if left_projection == right_projection {
            output.extend(left_projection);
        } else if left_projection == base_projection {
            output.extend(right_projection);
        } else if right_projection == base_projection {
            output.extend(left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(left_projection),
                Resolution::PreferRight => output.extend(right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let base_start_display = if component.end > component.start {
                        component.start + 1
                    } else {
                        component.start
                    };

                    append_line(&mut output, format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(left_projection.iter().cloned());
                    append_line(
                        &mut output,
                        format!(
                            "||||||| BASE {base_start_display},{}",
                            component.end - component.start
                        ),
                    );
                    output.extend(base_projection.iter().cloned());
                    append_line(&mut output, "=======");
                    output.extend(right_projection.iter().cloned());
                    append_line(&mut output, format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    while base_cursor < base_lines.len() {
        append_line(&mut output, base_lines[base_cursor].clone());
        base_cursor += 1;
    }

    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
