#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn { side: MergeSide, byte: usize },
    MissingFinalLf { side: MergeSide },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

#[derive(Clone)]
struct Component {
    start: usize,
    end: usize,
    left: Vec<Change>,
    right: Vec<Change>,
}

enum Op {
    Keep,
    Delete,
    Insert(String),
}

fn split_lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }

    input[..input.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn canonical_changes(base: &[String], new: &[String]) -> Vec<Change> {
    let n = base.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut base_consumed = 0usize;
    let mut run_start = 0usize;
    let mut run_deletes = 0usize;
    let mut run_insertions = Vec::new();
    let mut changes = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 run_start: &mut usize,
                 run_deletes: &mut usize,
                 run_insertions: &mut Vec<String>| {
        if *run_deletes != 0 || !run_insertions.is_empty() {
            changes.push(Change {
                start: *run_start,
                end: *run_start + *run_deletes,
                replacement: std::mem::take(run_insertions),
            });
            *run_deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && base[i] == new[j] {
            flush(
                &mut changes,
                &mut run_start,
                &mut run_deletes,
                &mut run_insertions,
            );
            i += 1;
            j += 1;
            base_consumed += 1;
        } else if i == n {
            if run_deletes == 0 && run_insertions.is_empty() {
                run_start = base_consumed;
            }
            run_insertions.push(new[j].clone());
            j += 1;
        } else if j == m {
            if run_deletes == 0 && run_insertions.is_empty() {
                run_start = base_consumed;
            }
            run_deletes += 1;
            i += 1;
            base_consumed += 1;
        } else {
            if run_deletes == 0 && run_insertions.is_empty() {
                run_start = base_consumed;
            }
            let delete_len = lcs[i + 1][j];
            let insert_len = lcs[i][j + 1];
            let delete = if delete_len > insert_len {
                true
            } else if insert_len > delete_len {
                false
            } else {
                (i + j).is_multiple_of(2)
            };

            if delete {
                run_deletes += 1;
                i += 1;
                base_consumed += 1;
            } else {
                run_insertions.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(
        &mut changes,
        &mut run_start,
        &mut run_deletes,
        &mut run_insertions,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_positive = a.start < a.end;
    let b_positive = b.start < b.end;

    match (a_positive, b_positive) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn components(left: &[Change], right: &[Change]) -> Vec<Component> {
    let mut all: Vec<(bool, usize)> = Vec::new();
    all.extend((0..left.len()).map(|i| (true, i)));
    all.extend((0..right.len()).map(|i| (false, i)));

    let mut parent: Vec<usize> = (0..all.len()).collect();

    fn find(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        x
    }

    for a in 0..all.len() {
        for b in (a + 1)..all.len() {
            let (a_left, a_index) = all[a];
            let (b_left, b_index) = all[b];
            let ca = if a_left { &left[a_index] } else { &right[a_index] };
            let cb = if b_left { &left[b_index] } else { &right[b_index] };
            if interacts(ca, cb) {
                let ra = find(&mut parent, a);
                let rb = find(&mut parent, b);
                if ra != rb {
                    parent[rb] = ra;
                }
            }
        }
    }

    let mut groups: Vec<(usize, Vec<(bool, usize)>)> = Vec::new();
    for (node, item) in all.into_iter().enumerate() {
        let root = find(&mut parent, node);
        if let Some((_, members)) = groups.iter_mut().find(|(r, _)| *r == root) {
            members.push(item);
        } else {
            groups.push((root, vec![item]));
        }
    }

    let mut result = Vec::new();
    for (_, members) in groups {
        let mut group_left = Vec::new();
        let mut group_right = Vec::new();

        for (is_left, index) in members {
            if is_left {
                group_left.push(left[index].clone());
            } else {
                group_right.push(right[index].clone());
            }
        }

        group_left.sort_by_key(|c| (c.start, c.end));
        group_right.sort_by_key(|c| (c.start, c.end));

        let mut start = usize::MAX;
        let mut end = 0usize;
        for change in group_left.iter().chain(group_right.iter()) {
            start = start.min(change.start);
            end = end.max(change.end);
        }

        result.push(Component {
            start,
            end,
            left: group_left,
            right: group_right,
        });
    }

    result.sort_by_key(|c| (c.start, usize::from(c.start < c.end), c.end));
    result
}

fn projection(base: &[String], start: usize, end: usize, changes: &[Change]) -> Vec<String> {
    let mut result = Vec::new();
    let mut cursor = start;

    for change in changes {
        if cursor < change.start {
            result.extend(base[cursor..change.start].iter().cloned());
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    if cursor < end {
        result.extend(base[cursor..end].iter().cloned());
    }
    result
}

fn append_lines(output: &mut Vec<String>, lines: &[String]) {
    output.extend(lines.iter().cloned());
}

fn append_marker(output: &mut Vec<String>, marker: String) {
    output.push(marker);
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = input.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (input, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !input.is_empty() && !input.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = canonical_changes(&base_lines, &left_lines);
    let right_changes = canonical_changes(&base_lines, &right_lines);
    let components = components(&left_changes, &right_changes);

    let mut output = Vec::new();
    let mut cursor = 0usize;
    let mut conflicts = Vec::new();

    for component in components {
        if cursor < component.start {
            append_lines(&mut output, &base_lines[cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &component.left,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &component.right,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();

                    append_marker(&mut output, format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);

                    let base_label_start = if component.start < component.end {
                        component.start + 1
                    } else {
                        component.start
                    };
                    append_marker(
                        &mut output,
                        format!(
                            "||||||| BASE {base_label_start},{}",
                            component.end - component.start
                        ),
                    );
                    append_lines(&mut output, &base_projection);
                    append_marker(&mut output, "=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    append_marker(&mut output, format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: output.len() - output_start,
                    });
                }
            }
        }

        cursor = component.end;
    }

    if cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[cursor..]);
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
