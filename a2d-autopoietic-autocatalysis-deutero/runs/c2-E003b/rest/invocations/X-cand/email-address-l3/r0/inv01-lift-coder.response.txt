#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Debug)]
struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<(usize, usize)>,
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut end = i + 2;
    if !matches!(input.get(end), Some(b' ' | b'\t')) {
        return None;
    }
    while matches!(input.get(end), Some(b' ' | b'\t')) {
        end += 1;
    }
    Some(end)
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn clean_cfws(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0;
    let mut first_retained = true;

    while i < input.len() {
        if first_retained && input[i] == b'"' {
            let start = i;
            loop {
                if i >= input.len() {
                    break;
                }
                let b = input[i];
                bytes.push(b);
                positions.push(i);
                i += 1;
                if b == b'\\' {
                    if i < input.len() {
                        bytes.push(input[i]);
                        positions.push(i);
                        i += 1;
                    }
                } else if b == b'"' && i > start + 1 {
                    break;
                }
            }
            first_retained = false;
            continue;
        }

        let run_start = i;
        let mut removed = false;

        loop {
            if i >= input.len() {
                break;
            }
            let b = input[i];

            if is_hws(b) {
                removed = true;
                i += 1;
                while i < input.len() && is_hws(input[i]) {
                    i += 1;
                }
                continue;
            }

            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    removed = true;
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }

            if b == b'(' {
                removed = true;
                let outer = i;
                let mut stack = Vec::with_capacity(3);
                stack.push(i);
                i += 1;

                while !stack.is_empty() {
                    if i >= input.len() {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                    let c = input[i];

                    if c == b'\\' {
                        let slash = i;
                        if i + 1 >= input.len() || !(0x20..=0x7e).contains(&input[i + 1]) {
                            return Err(EmailError::CommentEscape { at: slash });
                        }
                        i += 2;
                        continue;
                    }

                    if c == b'\r' || c == b'\n' {
                        if let Some(end) = fold_end(input, i) {
                            i = end;
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }

                    if c == b'(' {
                        if stack.len() == 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        stack.push(i);
                        i += 1;
                        continue;
                    }

                    if c == b')' {
                        stack.pop();
                        i += 1;
                        continue;
                    }

                    if c == b'\t' || (0x20..=0x7e).contains(&c) {
                        i += 1;
                        continue;
                    }

                    return Err(EmailError::CommentChar { at: i });
                }
                continue;
            }

            if b == b')' {
                return Err(EmailError::UnexpectedClose { at: i });
            }

            break;
        }

        if removed {
            runs.push((run_start, i));
        }

        if i < input.len() {
            bytes.push(input[i]);
            positions.push(i);
            i += 1;
            first_retained = false;
        }
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn pos(cleaned: &Cleaned, index: usize, input_len: usize) -> usize {
    cleaned.positions.get(index).copied().unwrap_or(input_len)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let cleaned = clean_cfws(raw)?;
    let b = &cleaned.bytes;

    for &(start, end) in &cleaned.runs {
        let retained_before = cleaned.positions.iter().any(|&p| p < start);
        let retained_after = cleaned.positions.iter().any(|&p| p >= end);
        let left_at = retained_before
            && cleaned
                .bytes
                .iter()
                .zip(cleaned.positions.iter())
                .any(|(&c, &p)| p < start && c == b'@'
                    && cleaned.positions.iter().any(|&q| q >= end && q > p));
        let right_at = retained_after
            && cleaned
                .bytes
                .iter()
                .zip(cleaned.positions.iter())
                .any(|(&c, &p)| p >= end && c == b'@'
                    && cleaned.positions.iter().any(|&q| q < start && q < p));
        if retained_before && retained_after && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary { at: start });
        }
    }

    if b.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quote_end = None;
    if b[0] == b'"' {
        let mut i = 1;
        while i < b.len() {
            if b[i] == b'\\' {
                i += 2;
            } else if b[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: pos(&cleaned, 0, raw.len()),
            });
        }
    }

    let protected_end = quote_end.map_or(0, |i| i + 1);
    let ats: Vec<usize> = b
        .iter()
        .enumerate()
        .filter_map(|(i, &c)| (c == b'@' && i >= protected_end).then_some(i))
        .collect();

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: pos(&cleaned, ats[1], raw.len()),
        });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: pos(&cleaned, sep, raw.len()),
        });
    }

    let local_bytes = &b[..sep];
    let local;
    let quoted;

    if local_bytes[0] == b'"' {
        let close = quote_end.unwrap();
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos(&cleaned, close + 1, raw.len()),
            });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            let c = b[i];
            if c == b'\\' {
                if i + 1 >= close || !matches!(b[i + 1], b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: pos(&cleaned, i, raw.len()),
                    });
                }
                decoded.push(b[i + 1]);
                origins.push(pos(&cleaned, i, raw.len()));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&c) {
                    return Err(EmailError::QuotedChar {
                        at: pos(&cleaned, i, raw.len()),
                    });
                }
                decoded.push(c);
                origins.push(pos(&cleaned, i, raw.len()));
                i += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: pos(&cleaned, close, raw.len()),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        local = String::from_utf8(decoded).unwrap();
        quoted = true;
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: pos(&cleaned, 64, raw.len()),
            });
        }

        for (i, &c) in local_bytes.iter().enumerate() {
            let at = pos(&cleaned, i, raw.len());
            if c == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at });
                }
            } else if !(c.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&c))
            {
                return Err(EmailError::LocalChar { at });
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).unwrap();
        quoted = false;
    }

    let domain_start = sep + 1;
    let mut domain_end = b.len();
    let mut root_dot = None;
    if domain_end > domain_start && b[domain_end - 1] == b'.' {
        domain_end -= 1;
        root_dot = Some(domain_end);
    }

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: pos(&cleaned, domain_start, raw.len()),
        });
    }

    let domain_bytes = &b[domain_start..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: pos(&cleaned, domain_start + 253, raw.len()),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || b[i] == b'.' {
            labels.push((start, i));
            start = i + 1;
        }
    }

    for (index, &(start, end)) in labels.iter().enumerate() {
        if start == end {
            let at = if index + 1 == labels.len() {
                root_dot.unwrap_or_else(|| pos(&cleaned, start, raw.len()))
            } else {
                pos(&cleaned, start, raw.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }

        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: pos(&cleaned, start + 63, raw.len()),
            });
        }

        if b[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&cleaned, start, raw.len()),
            });
        }
        if b[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: pos(&cleaned, end - 1, raw.len()),
            });
        }

        for i in start..end {
            let c = b[i];
            if !(c.is_ascii_alphanumeric() || c == b'-') {
                return Err(EmailError::DomainChar {
                    at: pos(&cleaned, i, raw.len()),
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: pos(&cleaned, domain_start, raw.len()),
        });
    }

    let &(last_start, last_end) = labels.last().unwrap();
    if b[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: pos(&cleaned, last_start, raw.len()),
        });
    }

    let domain = String::from_utf8(domain_bytes.to_vec())
        .unwrap()
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
