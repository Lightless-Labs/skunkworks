#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct RemovedRun {
    start: usize,
    end: usize,
    left: Option<u8>,
    right: Option<u8>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn fold_end(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let mut clean = Vec::<u8>::with_capacity(bytes.len());
    let mut positions = Vec::<usize>::with_capacity(bytes.len());
    let mut runs = Vec::<RemovedRun>::new();

    let mut i = 0;
    let mut protected_quote = false;
    let mut quote_escaped = false;

    while i < bytes.len() {
        if protected_quote {
            let b = bytes[i];
            clean.push(b);
            positions.push(i);
            if quote_escaped {
                quote_escaped = false;
            } else if b == b'\\' {
                quote_escaped = true;
            } else if b == b'"' {
                protected_quote = false;
            }
            i += 1;
            continue;
        }

        let starts_quote = clean.is_empty() && bytes[i] == b'"';
        if starts_quote {
            protected_quote = true;
            clean.push(bytes[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        if is_hws(bytes[i]) || bytes[i] == b'\r' || bytes[i] == b'\n' || bytes[i] == b'(' {
            let start = i;
            let left = clean.last().copied();
            let mut depth = 0usize;
            let mut outer_open = None;

            loop {
                if i >= bytes.len() {
                    if depth > 0 {
                        return Err(EmailError::UnclosedComment {
                            at: outer_open.unwrap_or(start),
                        });
                    }
                    break;
                }

                let b = bytes[i];
                if depth == 0 {
                    if is_hws(b) {
                        i += 1;
                        continue;
                    }
                    if b == b'\r' || b == b'\n' {
                        if let Some(end) = fold_end(bytes, i) {
                            i = end;
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }
                    if b == b'(' {
                        depth = 1;
                        outer_open = Some(i);
                        i += 1;
                        continue;
                    }
                    break;
                }

                if b == b'\\' {
                    let slash = i;
                    let Some(&next) = bytes.get(i + 1) else {
                        return Err(EmailError::CommentEscape { at: slash });
                    };
                    if !(0x20..=0x7e).contains(&next) {
                        return Err(EmailError::CommentEscape { at: slash });
                    }
                    i += 2;
                    continue;
                }

                if b == b'\r' || b == b'\n' {
                    if let Some(end) = fold_end(bytes, i) {
                        i = end;
                        continue;
                    }
                    return Err(EmailError::BadFold { at: i });
                }

                if b == b'(' {
                    if depth == 3 {
                        return Err(EmailError::CommentDepth { at: i });
                    }
                    depth += 1;
                    i += 1;
                    continue;
                }
                if b == b')' {
                    depth -= 1;
                    i += 1;
                    continue;
                }
                if is_hws(b) || (0x20..=0x7e).contains(&b) {
                    i += 1;
                    continue;
                }
                return Err(EmailError::CommentChar { at: i });
            }

            runs.push(RemovedRun {
                start,
                end: i,
                left,
                right: bytes.get(i).copied(),
            });
            continue;
        }

        if bytes[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(bytes[i]);
        positions.push(i);
        i += 1;
    }

    for run in &runs {
        let boundary_ok = run.left.is_none()
            || run.right.is_none()
            || run.left == Some(b'@')
            || run.right == Some(b'@');
        if !boundary_ok {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let quote_close = if clean.first() == Some(&b'"') {
        let mut j = 1;
        let mut close = None;
        while j < clean.len() {
            if clean[j] == b'\\' {
                j = j.saturating_add(2);
            } else if clean[j] == b'"' {
                close = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        match close {
            Some(j) => Some(j),
            None => return Err(EmailError::UnclosedQuote { at: positions[0] }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut j = 0;
    while j < clean.len() {
        if quote_close.is_some_and(|close| j <= close) {
            j = quote_close.unwrap() + 1;
            continue;
        }
        if clean[j] == b'@' {
            separators.push(j);
        }
        j += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: positions[separators[1]],
        });
    }

    let at_idx = separators[0];
    let at_pos = positions[at_idx];
    if at_idx == 0 {
        return Err(EmailError::EmptyLocal { at: at_pos });
    }

    let mut local = String::new();
    let quoted = quote_close.is_some();

    if let Some(close) = quote_close {
        if close != at_idx.saturating_sub(1) {
            let trailing = close + 1;
            return Err(EmailError::TrailingLocal {
                at: positions.get(trailing).copied().unwrap_or(input.len()),
            });
        }

        let mut origins = Vec::new();
        let mut k = 1;
        while k < close {
            let b = clean[k];
            if b == b'\\' {
                let Some(&escaped) = clean.get(k + 1).filter(|_| k + 1 < close) else {
                    return Err(EmailError::BadEscape { at: positions[k] });
                };
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape { at: positions[k] });
                }
                local.push(escaped as char);
                origins.push(positions[k]);
                k += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: positions[k] });
                }
                local.push(b as char);
                origins.push(positions[k]);
                k += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: positions[close],
            });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        let local_bytes = &clean[..at_idx];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: positions[64],
            });
        }
        for (k, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if k == 0
                    || k + 1 == local_bytes.len()
                    || local_bytes[k - 1] == b'.'
                {
                    return Err(EmailError::LocalDot { at: positions[k] });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar { at: positions[k] });
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).unwrap();
    }

    let domain_start_idx = at_idx + 1;
    let domain_start_pos = positions.get(domain_start_idx).copied().unwrap_or(input.len());
    let mut domain_end = clean.len();
    let root_dot_pos = if domain_end > domain_start_idx && clean[domain_end - 1] == b'.' {
        domain_end -= 1;
        Some(positions[domain_end])
    } else {
        None
    };

    if domain_start_idx == domain_end {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }

    if domain_end - domain_start_idx > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start_idx + 253],
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start_idx;
    for k in domain_start_idx..domain_end {
        if clean[k] == b'.' {
            if k == label_start {
                return Err(EmailError::EmptyLabel { at: positions[k] });
            }
            labels.push((label_start, k));
            label_start = k + 1;
        }
    }
    if label_start == domain_end {
        return Err(EmailError::EmptyLabel {
            at: root_dot_pos.unwrap_or(input.len()),
        });
    }
    labels.push((label_start, domain_end));

    for &(start, end) in &labels {
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[start + 63],
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: positions[start] });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for k in start..end {
            let b = clean[k];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar { at: positions[k] });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: domain_start_pos,
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = clean[domain_start_idx..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
