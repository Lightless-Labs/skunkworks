#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before checking any other input property.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers {
            count: worker_count,
        });
    }

    // Resource validation is task-major, then resource-major.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    // Build predecessor counts and successor lists.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available = std::collections::BinaryHeap::new();
    for task_id in 0..n {
        if indegree[task_id] == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &children[task_id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &task_id in &order {
            removed[task_id] = true;
        }
        for task_id in 0..n {
            if !removed[task_id] {
                blocked.push(task_id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Compute bottom levels in reverse Kahn order.
    let mut tails = vec![0u64; n];
    for &task_id in order.iter().rev() {
        let max_child_tail = children[task_id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[task_id] = tasks[task_id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_predecessors: Vec<usize> =
        tasks.iter().map(|task| task.deps.len()).collect();

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
        available_at: u64,
    }

    let mut running: Vec<Running> = Vec::new();
    let mut worker_available = vec![0u64; worker_count];
    let mut used: Vec<u128> = vec![0; capacity.len()];
    let mut trace = String::new();
    trace.push_str(&format!(
        "workers={};resources={}\n",
        worker_count,
        capacity.len()
    ));

    let format_used = |used: &[u128]| -> String {
        if used.is_empty() {
            "-".to_string()
        } else {
            used.iter()
                .map(|value| value.to_string())
                .collect::<Vec<_>>()
                .join(",")
        }
    };

    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks finishing now in ascending task ID.
        let mut finishing: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, entry)| (entry.finish == now).then_some(index))
            .collect();
        finishing.sort_by_key(|&index| running[index].task);

        for index in finishing.into_iter().rev() {
            let entry = running.remove(index);
            let task_id = entry.task;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] -= need as u128;
            }
            completed[task_id] = true;
            completed_count += 1;
            makespan = makespan.max(entry.finish);
            for &child in &children[task_id] {
                remaining_predecessors[child] -= 1;
            }
            trace.push_str(&format!(
                "{}|end|{}|{}|{}\n",
                now,
                task_id,
                entry.worker,
                format_used(&used)
            ));
        }

        // Repeatedly admit one task, then recalculate eligibility and ranking.
        let mut started_one = false;
        loop {
            let mut candidates: Vec<usize> = (0..n)
                .filter(|&task_id| {
                    !started[task_id]
                        && tasks[task_id].release <= now
                        && remaining_predecessors[task_id] == 0
                        && tasks[task_id]
                            .needs
                            .iter()
                            .enumerate()
                            .all(|(resource, &need)| {
                                (need as u128) <= (capacity[resource] as u128) - used[resource]
                            })
                })
                .collect();

            candidates.sort_by(|&a, &b| {
                tails[b]
                    .cmp(&tails[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task_id) = candidates.first().copied() else {
                break;
            };

            let start = now;
            let finish = start
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;

            let worker = (0..worker_count)
                .map(|offset| (task_id % worker_count + offset) % worker_count)
                .find(|&worker| worker_available[worker] <= now);

            let Some(worker) = worker else {
                // No worker is free at this instant; the simulation advances below.
                break;
            };

            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            worker_available[worker] = available_at;
            started[task_id] = true;
            for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                used[resource] += need as u128;
            }
            runs[task_id] = Some(Run {
                start,
                finish,
                worker,
            });
            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                format_used(&used)
            ));

            if tasks[task_id].duration == 0 {
                // Immediate completion, including resource release and unblocking.
                for (resource, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[resource] -= need as u128;
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                for &child in &children[task_id] {
                    remaining_predecessors[child] -= 1;
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    format_used(&used)
                ));
            }

            started_one = true;
        }

        if completed_count == n {
            break;
        }

        // If any admission or completion changed the state, retry at this time
        // before advancing; otherwise advance to the next relevant event.
        if started_one {
            continue;
        }

        let mut next_time: Option<u64> = None;
        for entry in &running {
            if entry.finish > now {
                next_time = Some(next_time.map_or(entry.finish, |t| t.min(entry.finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                next_time = Some(
                    next_time.map_or(tasks[task_id].release, |t| t.min(tasks[task_id].release)),
                );
            }
        }
        for worker in 0..worker_count {
            if worker_available[worker] > now {
                next_time = Some(next_time.map_or(worker_available[worker], |t| {
                    t.min(worker_available[worker])
                }));
            }
        }

        let Some(next_time) = next_time else {
            // In an acyclic graph, this can only occur after all tasks complete.
            break;
        };
        now = next_time;
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("all acyclic tasks must start"))
        .collect();

    trace.push_str(&format!("makespan={}\n", makespan));

    Ok(Schedule {
        runs,
        makespan,
        trace,
    })
}
