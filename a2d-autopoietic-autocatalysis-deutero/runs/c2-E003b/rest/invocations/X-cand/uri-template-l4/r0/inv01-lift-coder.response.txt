#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
struct Spec {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Debug, Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

impl Operator {
    fn from_byte(byte: u8) -> Option<Self> {
        match byte {
            b'+' => Some(Self::Reserved),
            b'#' => Some(Self::Fragment),
            b'.' => Some(Self::Dot),
            b'/' => Some(Self::Slash),
            b';' => Some(Self::Semicolon),
            b'?' => Some(Self::Question),
            b'&' => Some(Self::Ampersand),
            _ => None,
        }
    }

    fn separator(self) -> char {
        match self {
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Question | Self::Ampersand => '&',
            _ => ',',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Question => Some('?'),
            Self::Ampersand => Some('&'),
            _ => None,
        }
    }

    fn names_fields(self) -> bool {
        matches!(self, Self::Semicolon | Self::Question | Self::Ampersand)
    }

    fn semicolon_rules(self) -> bool {
        self == Self::Semicolon
    }

    fn reserved_encoding(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }
}

fn parse_expression(content: &str, base: usize) -> Result<(Operator, Vec<Spec>), Error> {
    let bytes = content.as_bytes();
    let (operator, start) = if let Some(&first) = bytes.first() {
        if let Some(op) = Operator::from_byte(first) {
            (op, 1)
        } else {
            (Operator::Simple, 0)
        }
    } else {
        return Err(Error::EmptyExpression { pos: base });
    };

    if start == bytes.len() {
        return Err(Error::EmptyExpression { pos: base + bytes.len() });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut offset = start;

    for part in content[start..].split(',') {
        let spec_start = base + offset;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let part_bytes = part.as_bytes();
        let modifier_at = part_bytes.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_at.unwrap_or(part_bytes.len());
        let name = &part[..name_end];

        if name.is_empty() {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let name_bytes = name.as_bytes();
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: spec_start });
        }
        for (i, &byte) in name_bytes.iter().enumerate().skip(1) {
            if !is_name_continue(byte) {
                return Err(Error::InvalidName { pos: spec_start + i });
            }
        }

        let modifier = if let Some(index) = modifier_at {
            let suffix = &part[index..];
            if suffix == "*" {
                Modifier::Explode
            } else if suffix.as_bytes().first() == Some(&b':') {
                let digits = &suffix[1..];
                let db = digits.as_bytes();
                let valid = (1..=4).contains(&db.len())
                    && db[0].is_ascii_digit()
                    && db[0] != b'0'
                    && db.iter().all(u8::is_ascii_digit);
                if !valid {
                    return Err(Error::InvalidModifier {
                        pos: spec_start + index,
                    });
                }
                let count = digits.parse::<usize>().unwrap_or(0);
                if !(1..=9999).contains(&count) {
                    return Err(Error::InvalidModifier {
                        pos: spec_start + index,
                    });
                }
                Modifier::Prefix(count)
            } else {
                return Err(Error::InvalidModifier {
                    pos: spec_start + index,
                });
            }
        } else {
            Modifier::None
        };

        let owned_name = name.to_owned();
        if seen.iter().any(|existing| existing == &owned_name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(owned_name.clone());

        specs.push(Spec {
            name: owned_name,
            start: spec_start,
            modifier,
        });

        offset += part.len() + 1;
    }

    Ok((operator, specs))
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

#[derive(Debug)]
struct Expression {
    operator: Operator,
    specs: Vec<Spec>,
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Option<Expression>)>, Error> {
    let bytes = template.as_bytes();
    let mut spans = Vec::new();
    let mut literal_start = 0;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(relative) => i + 1 + relative,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(relative) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: i + 1 + relative,
                    });
                }

                if literal_start < open {
                    spans.push((literal_start, open, None));
                }
                let (operator, specs) =
                    parse_expression(&template[open + 1..close], open + 1)?;
                spans.push((
                    open,
                    close + 1,
                    Some(Expression { operator, specs }),
                ));
                i = close + 1;
                literal_start = i;
            }
            _ => i += 1,
        }
    }

    if literal_start < template.len() {
        spans.push((literal_start, template.len(), None));
    }
    Ok(spans)
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn encode(input: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() && bytes[i + 1].is_ascii_hexdigit() && bytes[i + 2].is_ascii_hexdigit() {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else if reserved && is_reserved(byte) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }
    output
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn take_prefix(value: &str, count: usize) -> String {
    value.chars().take(count).collect()
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if operator.semicolon_rules() && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expression.operator;
    let mut fields = Vec::<String>::new();

    for spec in &expression.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) {
            match value {
                Value::Scalar(_) => {}
                Value::List(_) | Value::Map(_) => {
                    return Err(Error::TypeMismatch { pos: spec.start });
                }
            }
        }

        match value {
            Value::Scalar(scalar) => {
                let value = match spec.modifier {
                    Modifier::Prefix(count) => take_prefix(scalar, count),
                    Modifier::None | Modifier::Explode => scalar.clone(),
                };
                let encoded = encode(&value, op.reserved_encoding());
                fields.push(if op.names_fields() {
                    named_field(&spec.name, encoded, op)
                } else {
                    encoded
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for item in items {
                        let encoded = encode(item, op.reserved_encoding());
                        fields.push(if op.names_fields() {
                            named_field(&spec.name, encoded, op)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, op.reserved_encoding()))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if op.names_fields() {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|existing| *existing == key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                    keys.push(key);
                }

                if entries.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for (key, value) in entries {
                        let encoded_key = encode(key, op.reserved_encoding());
                        let encoded_value = encode(value, op.reserved_encoding());
                        let field = if op.semicolon_rules() && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, op.reserved_encoding()));
                        components.push(encode(value, op.reserved_encoding()));
                    }
                    let joined = components.join(",");
                    fields.push(if op.names_fields() {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    let mut output = String::new();
    if let Some(prefix) = op.prefix() {
        output.push(prefix);
    }
    output.push_str(&fields.join(&op.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let spans = parse_template(template)?;
    let mut output = String::new();

    for (start, end, expression) in spans {
        if let Some(expression) = expression {
            output.push_str(&expand_expression(&expression, vars)?);
        } else {
            output.push_str(&template[start..end]);
        }
    }

    Ok(output)
}
