#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Reserved | Self::Fragment => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon => ';',
            Self::Query | Self::Continuation => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Simple | Self::Reserved => None,
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continuation => Some('&'),
        }
    }

    fn named_fields(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continuation)
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Reserved | Self::Fragment)
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug)]
struct Specifier {
    name: String,
    modifier: Modifier,
    pos: usize,
}

#[derive(Debug)]
struct Expression {
    operator: Operator,
    specs: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn parse_specifier(text: &str, start: usize) -> Result<Specifier, Error> {
    if text.is_empty() {
        return Err(Error::EmptyVariable { pos: start });
    }

    let bytes = text.as_bytes();
    let modifier_at = bytes.iter().position(|&b| b == b'*' || b == b':');
    let name_end = modifier_at.unwrap_or(bytes.len());

    if name_end == 0 {
        return Err(Error::InvalidName { pos: start });
    }

    if !is_name_start(bytes[0]) {
        return Err(Error::InvalidName { pos: start });
    }
    for (i, &b) in bytes[..name_end].iter().enumerate().skip(1) {
        if !is_name_byte(b) {
            return Err(Error::InvalidName { pos: start + i });
        }
    }

    let modifier = if let Some(i) = modifier_at {
        let suffix = &text[i..];
        if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let valid = (1..=4).contains(&digits.len())
                && digits.as_bytes()[0].is_ascii_digit()
                && digits.as_bytes()[0] != b'0'
                && digits.bytes().all(|b| b.is_ascii_digit());
            if !valid {
                return Err(Error::InvalidModifier { pos: start + i });
            }
            let value = digits
                .bytes()
                .fold(0usize, |n, b| n * 10 + (b - b'0') as usize);
            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier { pos: start + i });
            }
            Modifier::Prefix(value)
        } else {
            return Err(Error::InvalidModifier { pos: start + i });
        }
    } else {
        Modifier::None
    };

    Ok(Specifier {
        name: text[..name_end].to_owned(),
        modifier,
        pos: start,
    })
}

fn parse_expression(content: &str, content_start: usize, close: usize) -> Result<Expression, Error> {
    let bytes = content.as_bytes();
    let (operator, offset) = if let Some(&b) = bytes.first() {
        let op = match b {
            b'+' => Some(Operator::Reserved),
            b'#' => Some(Operator::Fragment),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semicolon),
            b'?' => Some(Operator::Query),
            b'&' => Some(Operator::Continuation),
            b'=' | b'!' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: content_start });
            }
            _ => None,
        };
        if let Some(op) = op {
            (op, 1)
        } else {
            (Operator::Simple, 0)
        }
    } else {
        (Operator::Simple, 0)
    };

    let remaining = &content[offset..];
    if remaining.is_empty() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut local_start = 0;

    for part in remaining.split(',') {
        let pos = content_start + offset + local_start;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos });
        }
        let spec = parse_specifier(part, pos)?;
        if !seen.insert(spec.name.clone()) {
            return Err(Error::DuplicateVariable { pos: spec.pos });
        }
        specs.push(spec);
        local_start += part.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn validate_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let rest = &template[cursor..];
        let next_open = rest.find('{').map(|i| cursor + i);
        let next_close = rest.find('}').map(|i| cursor + i);

        match (next_open, next_close) {
            (None, Some(close)) => return Err(Error::UnexpectedClose { pos: close }),
            (None, None) => break,
            (Some(open), Some(close)) if close < open => {
                return Err(Error::UnexpectedClose { pos: close });
            }
            (Some(open), _) => {
                let after_open = open + 1;
                let closing = template[after_open..]
                    .find('}')
                    .map(|i| after_open + i);
                let Some(close) = closing else {
                    return Err(Error::Unclosed { pos: open });
                };

                if let Some(inner) = template[after_open..close].find('{') {
                    return Err(Error::NestedOpen {
                        pos: after_open + inner,
                    });
                }

                let expression = parse_expression(
                    &template[after_open..close],
                    after_open,
                    close,
                )?;
                expressions.push((open, close + 1, expression));
                cursor = close + 1;
            }
        }
    }

    Ok(expressions)
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut output = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];

        if b == b'%' && i + 2 < bytes.len() {
            if hex_value(bytes[i + 1]).is_some() && hex_value(bytes[i + 2]).is_some() {
                output.push('%');
                output.push((bytes[i + 1] as char).to_ascii_uppercase());
                output.push((bytes[i + 2] as char).to_ascii_uppercase());
                i += 3;
                continue;
            }
        }

        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let allowed_reserved = reserved
            && matches!(
                b,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                    | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || allowed_reserved {
            output.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(b >> 4) as usize] as char);
            output.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| value)
}

fn scalar_field(name: &str, value: &str, operator: Operator) -> String {
    let encoded = encode(value, operator.reserved());
    if operator.named_fields() {
        if operator == Operator::Semicolon && encoded.is_empty() {
            name.to_owned()
        } else {
            format!("{name}={encoded}")
        }
    } else {
        encoded
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator;
    let mut fields = Vec::new();

    for spec in &expression.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        match (&spec.modifier, value) {
            (Modifier::Prefix(_), Value::List(_) | Value::Map(_)) => {
                return Err(Error::TypeMismatch { pos: spec.pos });
            }
            (Modifier::Prefix(_), Value::Scalar(_)) => {}
            _ => {}
        }

        match value {
            Value::Scalar(text) => {
                let selected = if let Modifier::Prefix(count) = spec.modifier {
                    text.chars().take(count).collect::<String>()
                } else {
                    text.clone()
                };
                fields.push(scalar_field(&spec.name, &selected, operator));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for item in items {
                        fields.push(scalar_field(&spec.name, item, operator));
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, operator.reserved()))
                        .collect::<Vec<_>>()
                        .join(",");
                    if operator.named_fields() {
                        fields.push(if operator == Operator::Semicolon && joined.is_empty() {
                            spec.name.clone()
                        } else {
                            format!("{}={joined}", spec.name)
                        });
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                let mut keys = std::collections::HashSet::new();
                for (key, _) in entries {
                    if !keys.insert(key) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }
                if entries.is_empty() {
                    continue;
                }

                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for (key, value) in entries {
                        let encoded_key = encode(key, operator.reserved());
                        let encoded_value = encode(value, operator.reserved());
                        if operator == Operator::Semicolon && encoded_value.is_empty() {
                            fields.push(encoded_key);
                        } else {
                            fields.push(format!("{encoded_key}={encoded_value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, operator.reserved()));
                        components.push(encode(value, operator.reserved()));
                    }
                    let joined = components.join(",");
                    if operator.named_fields() {
                        fields.push(if operator == Operator::Semicolon && joined.is_empty() {
                            spec.name.clone()
                        } else {
                            format!("{}={joined}", spec.name)
                        });
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut result = String::new();
    if let Some(prefix) = operator.prefix() {
        result.push(prefix);
    }
    result.push_str(&fields.join(&operator.separator().to_string()));
    Ok(result)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}
