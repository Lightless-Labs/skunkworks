use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let negative = (num < 0) ^ (den < 0);
    let n_abs = num.unsigned_abs();
    let d_abs = den.unsigned_abs();
    let gcd = gcd_u128(n_abs, d_abs);
    let n = n_abs / gcd;
    let d = d_abs / gcd;

    if d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }
    if negative {
        let min_magnitude = (i64::MAX as u128) + 1;
        if n > min_magnitude {
            return Err(RationalError::Overflow);
        }
        let signed_n = if n == min_magnitude {
            i64::MIN
        } else {
            -(n as i64)
        };
        Ok(Rational {
            num: signed_n,
            den: d as i64,
        })
    } else {
        if n > i64::MAX as u128 {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: n as i64,
            den: d as i64,
        })
    }
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let ad = self.den as i128;
        let bd = rhs.den as i128;
        rational_from_i128(a * bd + b * ad, ad * bd)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = rhs.num as i128;
        let ad = self.den as i128;
        let bd = rhs.den as i128;
        rational_from_i128(a * bd - b * ad, ad * bd)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational {
            num: -self.num,
            den: self.den,
        })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        } else if !bytes[i].is_ascii_digit() {
            return Err(parse_err(0, ParseKind::InvalidByte));
        }

        let num_start = 0;
        let num_digits_start = if bytes[0] == b'-' { 1 } else { 0 };
        if num_digits_start == bytes.len() {
            return Err(parse_err(bytes.len(), ParseKind::InvalidByte));
        }
        if !bytes[num_digits_start].is_ascii_digit() {
            return Err(parse_err(num_digits_start, ParseKind::InvalidByte));
        }
        i = num_digits_start;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        let num_end = i;

        let mut slash = None;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            if i == bytes.len() {
                return Err(parse_err(bytes.len(), ParseKind::InvalidByte));
            }
            if !bytes[i].is_ascii_digit() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            let start = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            den_start = Some(start);
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let num_token = &bytes[num_start..num_end];
        let negative = num_token[0] == b'-';
        let num_digits = &bytes[num_digits_start..num_end];

        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(parse_err(num_digits_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == b"0" {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_err(start + 1, ParseKind::NonCanonical));
            }
        }

        let numerator_limit: u128 = if negative {
            (i64::MAX as u128) + 1
        } else {
            i64::MAX as u128
        };
        let mut numerator_magnitude = 0u128;
        for &b in num_digits {
            let digit = (b - b'0') as u128;
            numerator_magnitude = match numerator_magnitude
                .checked_mul(10)
                .and_then(|v| v.checked_add(digit))
            {
                Some(v) if v <= numerator_limit => v,
                _ => return Err(parse_err(0, ParseKind::OutOfRange)),
            };
        }

        let mut denominator = 1u128;
        if let (Some(start), Some(end)) = (den_start, den_end) {
            denominator = 0;
            for &b in &bytes[start..end] {
                let digit = (b - b'0') as u128;
                denominator = match denominator
                    .checked_mul(10)
                    .and_then(|v| v.checked_add(digit))
                {
                    Some(v) if v <= i64::MAX as u128 => v,
                    _ => return Err(parse_err(start, ParseKind::OutOfRange)),
                };
            }
            if denominator == 0 {
                return Err(parse_err(start, ParseKind::ZeroDenominator));
            }
        }

        if let (Some(slash_at), Some(start), Some(end)) = (slash, den_start, den_end) {
            if &bytes[start..end] == b"1" {
                return Err(parse_err(slash_at, ParseKind::NonCanonical));
            }
            if numerator_magnitude == 0 {
                return Err(parse_err(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(numerator_magnitude, denominator) > 1 {
                return Err(parse_err(slash_at, ParseKind::NonCanonical));
            }
        }

        let num = if negative {
            if numerator_magnitude == (i64::MAX as u128) + 1 {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };
        Ok(Rational {
            num,
            den: denominator as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row * self.cols + col;
        Ok(self.data[index])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for col in 0..self.cols {
            for row in 0..self.rows {
                data.push(self.data[row * self.cols + col]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(output_len);
        let zero = Rational { num: 0, den: 1 };

        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = zero;
                for step in 0..self.cols {
                    let left = self.data[row * self.cols + step];
                    let right = rhs.data[step * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
