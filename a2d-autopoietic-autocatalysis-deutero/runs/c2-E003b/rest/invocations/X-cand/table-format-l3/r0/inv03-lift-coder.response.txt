#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }

    let overhead = 3 * n + 1;
    let minimum = overhead + n;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    let mut trimmed = Vec::with_capacity(rows.len());
    for (row_idx, row) in rows.iter().enumerate() {
        let mut trimmed_row = Vec::with_capacity(n);
        for (col_idx, cell) in row.iter().enumerate() {
            for (byte, ch) in cell.char_indices() {
                if ch.is_control() || ch == '\u{2028}' || ch == '\u{2029}' {
                    return Err(TableError::InvalidChar {
                        row: row_idx,
                        column: col_idx,
                        byte,
                    });
                }
            }
            let end = cell.trim_end_matches(' ');
            trimmed_row.push(end.to_owned());
        }
        trimmed.push(trimmed_row);
    }

    let mut widths = vec![1usize; n];
    for row in &trimmed {
        for (col, cell) in row.iter().enumerate() {
            widths[col] = widths[col].max(cell.chars().count());
        }
    }

    if options.max_width != 0 {
        while widths.iter().sum::<usize>() + overhead > options.max_width {
            let widest = widths
                .iter()
                .enumerate()
                .filter(|(_, width)| **width > 1)
                .max_by_key(|(col, width)| (**width, *col))
                .map(|(col, _)| col)
                .expect("minimum width was checked");
            widths[widest] -= 1;
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };

    let mut output = String::new();

    fn border(output: &mut String, widths: &[usize]) {
        output.push('+');
        for &width in widths {
            for _ in 0..(width + 2) {
                output.push('-');
            }
            output.push('+');
        }
        output.push('\n');
    }

    fn data_line(
        output: &mut String,
        row: &[String],
        widths: &[usize],
        alignments: &[Alignment],
    ) {
        output.push('|');
        for ((cell, &width), alignment) in row.iter().zip(widths).zip(alignments) {
            let chars: Vec<char> = cell.chars().collect();
            let content: String = if chars.len() > width {
                let mut clipped: String = chars.into_iter().take(width - 1).collect();
                clipped.push('\u{2026}');
                clipped
            } else {
                cell.clone()
            };
            let content_width = content.chars().count();
            let extra = width - content_width;
            let (left, right) = match alignment {
                Alignment::Left => (0, extra),
                Alignment::Right => (extra, 0),
                Alignment::Center => (extra / 2, extra - extra / 2),
            };

            output.push(' ');
            for _ in 0..left {
                output.push(' ');
            }
            output.push_str(&content);
            for _ in 0..right {
                output.push(' ');
            }
            output.push(' ');
            output.push('|');
        }
        output.push('\n');
    }

    border(&mut output, &widths);
    for (idx, row) in trimmed.iter().enumerate() {
        data_line(&mut output, row, &widths, &alignments);
        if options.header && idx == 0 && trimmed.len() > 1 {
            border(&mut output, &widths);
        }
    }
    border(&mut output, &widths);

    Ok(output)
}
