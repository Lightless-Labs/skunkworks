```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
struct Entry {
    value: String,
    deadline: Option<u64>,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    entry: Option<Entry>,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Option<Entry>>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: BTreeMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, Entry>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: BTreeMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn tx_exists(&self, id: u64) -> Result<(), Error> {
        if self.transactions.contains_key(&id) {
            Ok(())
        } else {
            Err(Error::UnknownTransaction(id))
        }
    }

    fn visible_entry(&self, key: &str, tx: Option<u64>, now: u64) -> Option<Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(write) = transaction.writes.get(key) {
                return write.clone().filter(|e| e.deadline.map_or(true, |d| d > now));
            }
            let version = self.data.get(key)?.iter()
                .rev()
                .find(|v| v.revision <= transaction.begin_revision)?;
            return version.entry.clone().filter(|e| e.deadline.map_or(true, |d| d > now));
        }
        self.data.get(key)?.last()?.entry.clone()
            .filter(|e| e.deadline.map_or(true, |d| d > now))
    }

    fn live_map(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, Entry> {
        let mut keys: BTreeSet<String> = self.data.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(t) = self.transactions.get(&id) {
                keys.extend(t.writes.keys().cloned());
            }
        }
        let mut result = BTreeMap::new();
        for key in keys {
            if let Some(entry) = self.visible_entry(&key, tx, now) {
                result.insert(key, entry);
            }
        }
        result
    }

    fn append(&mut self, key: String, entry: Option<Entry>, revision: u64) {
        self.data.entry(key).or_default().push(Version { revision, entry });
    }

    fn valid_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    fn validate_range(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|e| e <= start) {
            Err(Error::InvalidRange)
        } else {
            Ok(())
        }
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(id, Transaction {
            begin_revision: self.revision,
            read_only,
            writes: BTreeMap::new(),
        });
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        self.tx_exists(id)?;
        let transaction = self.transactions.get(&id).unwrap();
        if !transaction.writes.is_empty() {
            for key in transaction.writes.keys() {
                if self.data.get(key).and_then(|v| v.last())
                    .is_some_and(|v| v.revision > transaction.begin_revision)
                {
                    return Err(Error::Conflict { tx: id, key: key.clone() });
                }
            }
            let writes = self.transactions.get(&id).unwrap().writes.clone();
            let next = self.revision + 1;
            for (key, entry) in writes {
                self.append(key, entry, next);
            }
            self.revision = next;
        }
        self.transactions.remove(&id);
        Ok(self.revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        self.tx_exists(id)?;
        self.transactions.remove(&id);
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        if let Some(id) = tx { self.tx_exists(id)?; }
        if key.is_empty() { return Err(Error::KeyEmpty); }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        if ttl == Some(0) {
            if let Some(id) = tx {
                self.transactions.get_mut(&id).unwrap().writes.insert(key.to_owned(), None);
            } else {
                self.revision += 1;
                self.append(key.to_owned(), None, self.revision);
            }
            return Ok(());
        }
        let deadline = match ttl {
            Some(d) => Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            None => None,
        };
        let entry = Some(Entry { value: value.to_owned(), deadline });
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_owned(), entry);
        } else {
            self.revision += 1;
            self.append(key.to_owned(), entry, self.revision);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        if let Some(id) = tx { self.tx_exists(id)?; }
        if key.is_empty() { return Err(Error::KeyEmpty); }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let existed = self.visible_entry(key, tx, now).is_some();
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_owned(), None);
        } else {
            self.revision += 1;
            self.append(key.to_owned(), None, self.revision);
        }
        Ok(existed)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit(now)?;
        if let Some(id) = tx { self.tx_exists(id)?; }
        if key.is_empty() { return Err(Error::KeyEmpty); }
        Ok(self.visible_entry(key, tx, now).map(|e| e.value))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit(now)?;
        if let Some(id) = tx { self.tx_exists(id)?; }
        Ok(self.live_map(tx, now).values().filter(|e| e.value == value).count())
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        if let Some(id) = tx { self.tx_exists(id)?; }
        Self::validate_range(start, end)?;
        if let Some(c) = cursor {
            if c.start != start || c.end.as_deref() != end || c.prefix.as_deref() != prefix
                || c.offset != offset || c.last.is_empty()
                || !Self::matches(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }
        if limit == 0 {
            return Ok(Page { items: Vec::new(), next: None });
        }
        let after = cursor.map(|c| c.last.as_str());
        let all: Vec<(String, String)> = self.live_map(tx, now).into_iter()
            .filter(|(k, _)| Self::matches(k, start, end, prefix)
                && after.map_or(true, |last| k.as_str() > last))
            .map(|(k, e)| (k, e.value))
            .collect();
        let skip = if cursor.is_some() { 0 } else { offset };
        let items: Vec<_> = all.into_iter().skip(skip).take(limit).collect();
        let next = items.last().and_then(|(last, _)| {
            let has_more = self.live_map(tx, now).keys().any(|k|
                Self::matches(k, start, end, prefix) && k > last
            );
            has_more.then(|| Cursor {
                start: start.to_owned(),
                end: end.map(str::to_owned),
                prefix: prefix.map(str::to_owned),
                offset,
                last: last.clone(),
            })
        });
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        if let Some(id) = tx { self.tx_exists(id)?; }
        Self::validate_range(start, end)?;
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, entry) in self.live_map(tx, now) {
            if !Self::matches(&key, start, end, prefix) { continue; }
            let value = &entry.value;
            let valid_syntax = if value == "0" {
                true
            } else {
                let digits = value.strip_prefix('-').unwrap_or(value);
                !digits.is_empty()
                    && digits.as_bytes()[0].is_ascii_digit()
                    && digits.as_bytes()[0] != b'0'
                    && digits.bytes().all(|b| b.is_ascii_digit())
                    && !(value.starts_with('-') && digits == "0")
            };
            if !valid_syntax {
                return Err(Error::InvalidNumber { key, value: value.clone() });
            }
            let number = value.parse::<i64>()
                .map_err(|_| Error::InvalidNumber { key: key.clone(), value: value.clone() })?;
            sum = sum.checked_add(number).ok_or_else(|| Error::SumOverflow { key: key.clone() })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::valid_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let saved = self.live_map(None, now);
        self.snapshots.insert(name.to_owned(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() { return Err(Error::NameEmpty); }
        let snapshot = self.snapshots.get(name)
            .ok_or_else(|| Error::SnapshotMissing(name.to_owned()))?;
        if !self.transactions.is_empty() { return Err(Error::ActiveTransactions); }
        let saved: BTreeMap<String, Entry> = snapshot.iter()
            .filter(|(_, e)| e.deadline.map_or(true, |d| d > now))
            .map(|(k, e)| (k.clone(), e.clone()))
            .collect();
        let current = self.live_map(None, now);
        let mut union: BTreeSet<String> = current.keys().cloned().collect();
        union.extend(saved.keys().cloned());
        if union.is_empty() { return Ok(self.revision); }
        self.revision += 1;
        let revision = self.revision;
        for key in union {
            self.append(key.clone(), saved.get(&key).cloned(), revision);
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() { return Err(Error::NameEmpty); }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_owned()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;
        for versions in self.data.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() {
                keep.insert(versions.len() - 1);
            }
            for transaction in self.transactions.values() {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= transaction.begin_revision) {
                    keep.insert(index);
                }
            }
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain { removed += 1; }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self.data.values().flat_map(|v| v.iter())
            .filter(|v| v.entry.is_none()).count();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.len(),
            live_keys: self.live_map(None, now).len(),
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self.transactions.values().filter(|t| t.read_only).count(),
            staged_writes: self.transactions.values().map(|t| t.writes.len()).sum(),
            snapshots: self.snapshots.len(),
        })
    }
}
```
