#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_prefix = if bytes[0] == b'"' {
        let mut i = 1;
        let mut closing = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    closing = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match closing {
            Some(end) => Some(end),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_prefix {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
            if separators.len() == 2 {
                return Err(EmailError::MultipleAt { at: i });
            }
        }
        i += 1;
    }

    let separator = match separators.first() {
        Some(&at) => at,
        None => return Err(EmailError::MissingAt),
    };

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let (local, is_quoted) = if let Some(end) = quoted_prefix {
        if end + 1 != separator {
            return Err(EmailError::TrailingLocal { at: end + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            if bytes[j] == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape { at: j });
                }
                match bytes[j + 1] {
                    b'"' | b'\\' | b'@' => {
                        decoded.push(bytes[j + 1]);
                        origins.push(j);
                        j += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: j }),
                }
            } else {
                if !(0x20..=0x7e).contains(&bytes[j]) {
                    return Err(EmailError::QuotedChar { at: j });
                }
                decoded.push(bytes[j]);
                origins.push(j);
                j += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: end });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        let local_bytes = &bytes[..separator];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (idx, &byte) in local_bytes.iter().enumerate() {
            if byte == b'.' {
                if idx == 0 || idx + 1 == local_bytes.len() || local_bytes[idx - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: idx });
                }
            } else if !byte.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
            {
                return Err(EmailError::LocalChar { at: idx });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut labels = Vec::new();
    let mut label_start = 0;
    for idx in 0..=domain_len {
        if idx == domain_len || raw_domain[idx] == b'.' {
            let label = &raw_domain[label_start..idx];
            let absolute_start = domain_start + label_start;
            if label.is_empty() {
                let at = if idx == domain_len && has_root_dot {
                    domain_start + raw_domain.len() - 1
                } else {
                    absolute_start
                };
                return Err(EmailError::EmptyLabel { at });
            }

            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: absolute_start + 63,
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: absolute_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: absolute_start + label.len() - 1,
                });
            }
            if let Some(offset) = label
                .iter()
                .position(|&b| !b.is_ascii_alphanumeric() && b != b'-')
            {
                return Err(EmailError::DomainChar {
                    at: absolute_start + offset,
                });
            }

            labels.push(label);
            label_start = idx + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last = labels[labels.len() - 1];
    if last.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + domain_len - last.len(),
        });
    }

    let domain = String::from_utf8(raw_domain[..domain_len].to_vec())
        .unwrap()
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted: is_quoted,
    })
}
