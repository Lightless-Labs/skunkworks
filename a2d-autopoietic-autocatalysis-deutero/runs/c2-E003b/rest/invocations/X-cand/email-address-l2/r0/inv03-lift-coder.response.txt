#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_prefix_end = if bytes.first() == Some(&b'"') {
        let mut i = 1;
        let mut closing = None;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    closing = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match closing {
            Some(i) => Some(i),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        if let Some(end) = quoted_prefix_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if bytes[i] == b'@' {
            separators.push(i);
        }
        i += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let domain_start = sep + 1;
    let local_bytes = &bytes[..sep];
    let (local, quoted) = if local_bytes[0] == b'"' {
        let closing = quoted_prefix_end.expect("quoted prefix has a closing quote");
        if closing + 1 != sep {
            return Err(EmailError::TrailingLocal { at: closing + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut p = 1;
        while p < closing {
            let b = bytes[p];
            if b == b'\\' {
                if p + 1 >= closing {
                    return Err(EmailError::BadEscape { at: p });
                }
                let escaped = bytes[p + 1];
                match escaped {
                    b'"' | b'\\' | b'@' => {
                        decoded.push(escaped);
                        origins.push(p);
                        p += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: p }),
                }
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: p });
                }
                decoded.push(b);
                origins.push(p);
                p += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: closing });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }

        (
            String::from_utf8(decoded).expect("decoded bytes are ASCII"),
            true,
        )
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (idx, &b) in local_bytes.iter().enumerate() {
            let allowed = b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b);
            if b == b'.' {
                if idx == 0 || idx + 1 == local_bytes.len() || local_bytes[idx - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: idx });
                }
            } else if !allowed {
                return Err(EmailError::LocalChar { at: idx });
            }
        }

        (
            String::from_utf8(local_bytes.to_vec()).expect("local bytes are ASCII"),
            false,
        )
    };

    let raw_domain = &bytes[domain_start..];
    let has_root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(has_root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain_end = domain_start + domain_len;
    let mut label_ranges = Vec::new();
    let mut label_start = domain_start;

    for pos in domain_start..domain_end {
        if bytes[pos] == b'.' {
            if pos == label_start {
                return Err(EmailError::EmptyLabel { at: pos });
            }
            label_ranges.push((label_start, pos));
            label_start = pos + 1;
        }
    }

    if label_start == domain_end {
        let at = if has_root_dot {
            input.len() - 1
        } else {
            domain_end
        };
        return Err(EmailError::EmptyLabel { at });
    }
    label_ranges.push((label_start, domain_end));

    for &(start, end) in &label_ranges {
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong { at: start + 63 });
        }
        if bytes[start] == b'-' {
            return Err(EmailError::LabelHyphen { at: start });
        }
        if bytes[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen { at: end - 1 });
        }
        for pos in start..end {
            let b = bytes[pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(EmailError::DomainChar { at: pos });
            }
        }
    }

    if label_ranges.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *label_ranges.last().expect("domain has a label");
    if bytes[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld { at: last_start });
    }

    let domain = bytes[domain_start..domain_end]
        .iter()
        .map(|&b| b.to_ascii_lowercase() as char)
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
