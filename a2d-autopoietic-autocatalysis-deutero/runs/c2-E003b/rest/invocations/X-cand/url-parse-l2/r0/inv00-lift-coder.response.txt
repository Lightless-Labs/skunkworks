#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn is_hex(byte: u8) -> bool {
    byte.is_ascii_hexdigit()
}

fn hex_value(byte: u8) -> u8 {
    match byte {
        b'0'..=b'9' => byte - b'0',
        b'a'..=b'f' => byte - b'a' + 10,
        b'A'..=b'F' => byte - b'A' + 10,
        _ => 0,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() || !is_hex(bytes[i + 1]) || !is_hex(bytes[i + 2]) {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hex_value(bytes[i + 1]) << 4) | hex_value(bytes[i + 2]));
            offsets.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_at = error.valid_up_to();
            let original_at = offsets.get(decoded_at).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_at })
        }
    }
}

fn component_byte_is_safe(component: Component, byte: u8) -> bool {
    byte.is_ascii_alphanumeric()
        || matches!(byte, b'-' | b'.' | b'_' | b'~')
        || match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = text.as_bytes();
    let mut encoded = String::with_capacity(bytes.len());

    for &byte in bytes {
        if component_byte_is_safe(component, byte) {
            encoded.push(byte as char);
        } else {
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }

    encoded
}

fn ascii_lowercase(value: &str) -> String {
    value
        .bytes()
        .map(|byte| byte.to_ascii_lowercase() as char)
        .collect()
}

fn parse_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {
    let bytes = port_text.as_bytes();

    if bytes.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    if let Some(offset) = bytes.iter().position(|byte| !byte.is_ascii_digit()) {
        return Err(ParseError::InvalidPort {
            at: port_start + offset,
        });
    }

    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for &byte in bytes {
        value = match value
            .checked_mul(10)
            .and_then(|current| current.checked_add((byte - b'0') as u32))
        {
            Some(next) => next,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }

    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    Ok(value as u16)
}

fn validate_component(
    component: Component,
    value: &str,
    original_start: usize,
) -> Result<(), ParseError> {
    match decode_component(component, value) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape {
            at: original_start + at,
        }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 {
            at: original_start + at,
        }),
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for at in 1..colon {
        let byte = bytes[at];
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let after_marker = colon + 3;
    let authority_end = input[after_marker..]
        .find(['/', '?', '#'])
        .map(|offset| after_marker + offset)
        .unwrap_or(input.len());
    let authority = &input[after_marker..authority_end];

    let authority_bytes = authority.as_bytes();
    let at_positions: Vec<usize> = authority_bytes
        .iter()
        .enumerate()
        .filter_map(|(index, &byte)| (byte == b'@').then_some(index))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: after_marker + at_positions[1],
        });
    }

    let (userinfo, host_port, host_port_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: after_marker + at,
            });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            after_marker + at + 1,
        )
    } else {
        (None, authority, after_marker)
    };

    let mut ipv6 = false;
    let (host, port) = if host_port.starts_with('[') {
        let open_at = host_port_start;
        let close_relative = match host_port.as_bytes().iter().position(|&byte| byte == b']') {
            Some(index) => index,
            None => return Err(ParseError::InvalidIpv6 { at: open_at }),
        };
        let interior = &host_port[1..close_relative];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        }

        let suffix = &host_port[close_relative + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            let port_start = host_port_start + close_relative + 2;
            Some(parse_port(&suffix[1..], port_start)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_port_start + close_relative + 1,
            });
        };

        ipv6 = true;
        (ascii_lowercase(interior), parsed_port)
    } else {
        let host_colon = host_port.as_bytes().iter().position(|&byte| byte == b':');
        let (host_text, parsed_port) = if let Some(index) = host_colon {
            let port_start = host_port_start + index + 1;
            (&host_port[..index], Some(parse_port(&host_port[index + 1..], port_start)?))
        } else {
            (host_port, None)
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_port_start });
        }
        if let Some(offset) = host_text
            .bytes()
            .position(|byte| !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')))
        {
            return Err(ParseError::InvalidHost {
                at: host_port_start + offset,
            });
        }

        (ascii_lowercase(host_text), parsed_port)
    };

    let tail = &input[authority_end..];
    let fragment_delimiter = tail.find('#');
    let before_fragment = fragment_delimiter.map(|at| &tail[..at]).unwrap_or(tail);
    let fragment = fragment_delimiter.map(|at| tail[at + 1..].to_owned());

    let query_delimiter = before_fragment.find('?');
    let before_query = query_delimiter
        .map(|at| &before_fragment[..at])
        .unwrap_or(before_fragment);
    let query = query_delimiter.map(|at| before_fragment[at + 1..].to_owned());

    let path = if before_query.is_empty() {
        "/".to_owned()
    } else {
        before_query.to_owned()
    };

    if let Some(userinfo_value) = userinfo.as_deref() {
        validate_component(Component::UserInfo, userinfo_value, after_marker)?;
    }

    let path_start = authority_end;
    if before_query.is_empty() {
        // The synthetic default path requires no validation.
    } else {
        validate_component(Component::Path, &path, path_start)?;
    }

    if let Some(query_value) = query.as_deref() {
        let query_start = authority_end + query_delimiter.unwrap() + 1;
        validate_component(Component::Query, query_value, query_start)?;
    }

    if let Some(fragment_value) = fragment.as_deref() {
        let fragment_start = authority_end + fragment_delimiter.unwrap() + 1;
        validate_component(Component::Fragment, fragment_value, fragment_start)?;
    }

    Ok(Url {
        scheme: ascii_lowercase(&input[..colon]),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}
