#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<char>,
    variables: Vec<(String, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || byte == b'_'
}

fn validate_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut content_start = open + 1;
    let operator = match bytes.get(content_start).copied() {
        Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
            let op = bytes[content_start] as char;
            content_start += 1;
            Some(op)
        }
        Some(b'=' | b'!' | b'@' | b'|' | b'$' | b'%' | b'^') => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => None,
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut variables = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut item_start = content_start;

    loop {
        let item_end = (item_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if item_start == item_end {
            return Err(Error::EmptyVariable { pos: item_start });
        }

        if !is_name_start(bytes[item_start]) {
            return Err(Error::InvalidName { pos: item_start });
        }
        for i in item_start + 1..item_end {
            if !is_name_continue(bytes[i]) {
                return Err(Error::InvalidName { pos: i });
            }
        }

        let name = template[item_start..item_end].to_owned();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        variables.push((name, item_start));

        if item_end == close {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        variables,
    })
}

fn parse(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let close = match (i + 1..bytes.len()).find(|&j| bytes[j] == b'}') {
                    Some(close) => close,
                    None => return Err(Error::Unclosed { pos: i }),
                };
                expressions.push(validate_expression(template, i, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut out = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                byte,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!'
                    | b'$' | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_byte {
            out.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(byte >> 4) as usize] as char);
            out.push(HEX[(byte & 0x0f) as usize] as char);
        }

        i += 1;
    }

    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding, _)| *binding == name)
        .map(|(_, value)| *value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let reserved = matches!(operator, Some('+' | '#'));
    let mut fields = Vec::new();

    for (name, _) in &expression.variables {
        if let Some(value) = lookup(name, vars) {
            let encoded = encode(value, reserved);
            let field = match operator {
                Some(';' | '?' | '&') if operator == Some(';') && encoded.is_empty() => {
                    name.clone()
                }
                Some(';' | '?' | '&') => format!("{name}={encoded}"),
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match operator {
        Some('.') => (".", "."),
        Some('/') => ("/", "/"),
        Some(';') => (";", ";"),
        Some('?') => ("?", "&"),
        Some('&') => ("&", "&"),
        Some('#') => ("#", ","),
        _ => ("", ","),
    };

    let mut out = String::from(prefix);
    out.push_str(&fields.join(separator));
    out
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let mut out = String::new();
    let mut cursor = 0;

    for expression in &expressions {
        out.push_str(&template[cursor..expression.start]);
        out.push_str(&expand_expression(expression, vars));
        cursor = expression.end;
    }

    out.push_str(&template[cursor..]);
    Ok(out)
}
