#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone, Copy)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<u8>,
    names: Vec<(&'a str, usize)>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();

    if let Some(relative) = bytes[open + 1..close].iter().position(|&byte| byte == b'{') {
        return Err(Error::NestedOpen {
            pos: open + 1 + relative,
        });
    }

    let mut content_start = open + 1;
    let operator = match bytes.get(content_start).copied() {
        Some(byte @ (b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')) => {
            content_start += 1;
            Some(byte)
        }
        Some(byte @ (b'!' | b'=' | b'<' | b'>' | b'%' | b' ' | b'\t' | b'\n' | b'\r')) => {
            return Err(Error::InvalidName { pos: content_start });
        }
        Some(byte @ (b'@' | b'|' | b'\\' | b'^' | b'`' | b'~')) => {
            return Err(Error::InvalidName { pos: content_start });
        }
        _ => None,
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut item_start = content_start;

    loop {
        let item_end = bytes[item_start..close]
            .iter()
            .position(|&byte| byte == b',')
            .map_or(close, |relative| item_start + relative);

        if item_start == item_end {
            return Err(Error::EmptyVariable { pos: item_start });
        }

        let item = &template[item_start..item_end];
        let item_bytes = item.as_bytes();

        if !is_name_start(item_bytes[0]) {
            return Err(Error::InvalidName { pos: item_start });
        }
        if let Some(relative) = item_bytes[1..]
            .iter()
            .position(|&byte| !is_name_continue(byte))
        {
            return Err(Error::InvalidName {
                pos: item_start + 1 + relative,
            });
        }

        if seen.iter().any(|&name| name == item) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        seen.push(item);
        names.push((item, item_start));

        if item_end == close {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        names,
    })
}

fn validate_template(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative) = bytes[open + 1..].iter().position(|&byte| byte == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative;
                expressions.push(parse_expression(template, open, close)?);
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find_map(|&(binding, value)| (binding == name).then_some(value))
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, reserved_allowed: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                const HEX: &[u8; 16] = b"0123456789ABCDEF";
                output.push('%');
                output.push(HEX[high as usize] as char);
                output.push(HEX[low as usize] as char);
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved_allowed && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0F) as usize] as char);
        }
        index += 1;
    }

    output
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let reserved_allowed = matches!(operator, Some(b'+') | Some(b'#'));
    let mut fields = Vec::new();

    for &(name, _) in &expression.names {
        if let Some(value) = lookup(vars, name) {
            let encoded = encode(value, reserved_allowed);
            let field = match operator {
                Some(b';') if encoded.is_empty() => name.to_owned(),
                Some(b';' | b'?' | b'&') => format!("{name}={encoded}"),
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (prefix, separator) = match operator {
        Some(b'#') => ("#", ","),
        Some(b'.') => (".", "."),
        Some(b'/') => ("/", "/"),
        Some(b';') => (";", ";"),
        Some(b'?') => ("?", "&"),
        Some(b'&') => ("&", "&"),
        _ => ("", ","),
    };

    let mut output = String::from(prefix);
    for (index, field) in fields.into_iter().enumerate() {
        if index > 0 {
            output.push_str(separator);
        }
        output.push_str(&field);
    }
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in &expressions {
        output.push_str(&template[cursor..expression.start]);
        output.push_str(&expand_expression(expression, vars));
        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
