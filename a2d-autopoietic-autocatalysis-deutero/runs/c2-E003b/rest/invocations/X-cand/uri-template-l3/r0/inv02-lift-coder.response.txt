#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression<'a> {
    operator: Option<char>,
    variables: Vec<&'a str>,
}

fn parse_template(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + offset });
                }

                let mut content_start = open + 1;
                let mut operator = None;

                if content_start < close {
                    let ch = bytes[content_start] as char;
                    if matches!(ch, '+' | '#' | '.' | '/' | ';' | '?' | '&') {
                        operator = Some(ch);
                        content_start += 1;
                    } else if matches!(ch, '!' | '=' | '$' | '%' | ':' | '@' | '[' | ']' | '^' | '`' | '|') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let content = &template[content_start..close];
                let mut variables = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut relative_start = 0;

                for specifier in content.split(',') {
                    let start = content_start + relative_start;
                    if specifier.is_empty() {
                        return Err(Error::EmptyVariable { pos: start });
                    }

                    let spec_bytes = specifier.as_bytes();
                    let valid_start = |b: u8| b.is_ascii_alphabetic() || b == b'_';
                    let valid_continue = |b: u8| valid_start(b) || b.is_ascii_digit();

                    if !valid_start(spec_bytes[0]) {
                        return Err(Error::InvalidName { pos: start });
                    }
                    if let Some(offset) = spec_bytes.iter().enumerate().skip(1).find_map(|(idx, &b)| {
                        if valid_continue(b) { None } else { Some(idx) }
                    }) {
                        return Err(Error::InvalidName { pos: start + offset });
                    }

                    if seen.iter().any(|&name| name == specifier) {
                        return Err(Error::DuplicateVariable { pos: start });
                    }

                    seen.push(specifier);
                    variables.push(specifier);
                    relative_start += specifier.len() + 1;
                }

                expressions.push(Expression { operator, variables });
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                result.push('%');
                result.push(HEX[high as usize] as char);
                result.push(HEX[low as usize] as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    result
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    let mut output = String::new();
    let bytes = template.as_bytes();
    let mut cursor = 0;

    for expression in expressions {
        let open = bytes[cursor..]
            .iter()
            .position(|&b| b == b'{')
            .map(|offset| cursor + offset)
            .expect("validated expression has an opening brace");

        output.push_str(&template[cursor..open]);

        let allow_reserved = matches!(expression.operator, Some('+' | '#'));
        let mut fields: Vec<String> = Vec::new();

        for name in &expression.variables {
            if let Some((_, value)) = vars.iter().rev().find(|(bound_name, _)| bound_name == name) {
                let encoded = encode(value, allow_reserved);
                let field = match expression.operator {
                    Some(';' | '?' | '&') => {
                        if expression.operator == Some(';') && encoded.is_empty() {
                            (*name).to_string()
                        } else {
                            format!("{name}={encoded}")
                        }
                    }
                    _ => encoded,
                };
                fields.push(field);
            }
        }

        if !fields.is_empty() {
            match expression.operator {
                Some('.') | Some('/') | Some(';') | Some('?') | Some('&') => {
                    let separator = expression.operator.unwrap();
                    output.push(separator);
                    output.push_str(&fields.join(&separator.to_string()));
                }
                Some('#') => {
                    output.push('#');
                    output.push_str(&fields.join(","));
                }
                _ => output.push_str(&fields.join(",")),
            }
        }

        let close = bytes[open + 1..]
            .iter()
            .position(|&b| b == b'}')
            .map(|offset| open + 1 + offset)
            .expect("validated expression has a closing brace");
        cursor = close + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
