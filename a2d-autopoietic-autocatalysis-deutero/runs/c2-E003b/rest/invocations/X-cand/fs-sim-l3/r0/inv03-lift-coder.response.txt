```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn components(path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(path
            .split('/')
            .filter(|part| !part.is_empty() && *part != ".")
            .map(str::to_string)
            .collect())
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        let input = Self::components(path)?;
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else if self.cwd == "/" {
            Vec::new()
        } else {
            self.cwd[1..].split('/').map(str::to_string).collect()
        };

        for part in input {
            if part == ".." {
                parts.pop();
            } else {
                parts.push(part);
            }
        }
        Ok(Self::canonical(&parts))
    }

    fn parent(path: &str) -> Option<String> {
        if path == "/" {
            return None;
        }
        match path.rfind('/') {
            Some(0) => Some("/".to_string()),
            Some(i) => Some(path[..i].to_string()),
            _ => None,
        }
    }

    fn basename(path: &str) -> &str {
        path.rsplit('/').next().unwrap_or("")
    }

    fn resolve_existing(&self, path: &str) -> Result<&Entry, Error> {
        if path == "/" {
            return self
                .entries
                .get("/")
                .ok_or_else(|| Error::Missing("/".to_string()));
        }

        let mut prefix = String::new();
        let parts: Vec<&str> = path[1..].split('/').collect();
        for (i, part) in parts.iter().enumerate() {
            prefix.push('/');
            prefix.push_str(part);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(entry) if i + 1 < parts.len() && !matches!(entry, Entry::Dir) => {
                    return Err(Error::NotDir(prefix));
                }
                Some(entry) if i + 1 == parts.len() => return Ok(entry),
                Some(_) => {}
            }
        }
        Err(Error::Missing(path.to_string()))
    }

    fn resolve_parent(&self, path: &str) -> Result<String, Error> {
        let parent = Self::parent(path).expect("non-root path has a parent");
        match self.resolve_existing(&parent)? {
            Entry::Dir => Ok(parent),
            _ => Err(Error::NotDir(parent)),
        }
    }

    fn contains_path(directory: &str, path: &str) -> bool {
        directory == "/"
            || path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|rest| rest.starts_with('/'))
    }

    fn is_cwd_or_ancestor(&self, path: &str) -> bool {
        Self::contains_path(path, &self.cwd)
    }

    fn children(&self, path: &str) -> Vec<(String, Kind)> {
        let prefix = if path == "/" {
            "/".to_string()
        } else {
            format!("{}/", path)
        };
        let mut result = self
            .entries
            .iter()
            .filter_map(|(candidate, entry)| {
                let rest = candidate.strip_prefix(&prefix)?;
                if !rest.is_empty() && !rest.contains('/') {
                    Some((rest.to_string(), entry.kind()))
                } else {
                    None
                }
            })
            .collect::<Vec<_>>();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        result
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Exists(path));
        }
        self.resolve_parent(&path)?;
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }
        self.resolve_parent(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(_)) => {
                self.entries.insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
            None => {
                self.entries.insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        match self.resolve_existing(&path)? {
            Entry::File(data) => Ok(data.clone()),
            Entry::Dir => Err(Error::IsDir(path)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        match self.resolve_existing(&path)? {
            Entry::Dir => Ok(self.children(&path)),
            Entry::File(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        match self.resolve_existing(&path)? {
            Entry::Dir => {
                self.cwd = path;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = self.normalize(from)?;
        let to = self.normalize(to)?;
        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&from)?;
        let source = self
            .entries
            .get(&from)
            .cloned()
            .ok_or_else(|| Error::Missing(from.clone()))?;

        if matches!(source, Entry::Dir) && self.is_cwd_or_ancestor(&from) {
            return Err(Error::Busy(from));
        }

        self.resolve_parent(&to)?;
        if matches!(self.entries.get(&to), Some(Entry::Dir)) && self.is_cwd_or_ancestor(&to) {
            return Err(Error::Busy(to));
        }

        if from == to {
            return Ok(());
        }

        if matches!(source, Entry::Dir) && Self::contains_path(&from, &to) {
            return Err(Error::Cycle(to));
        }

        let destination = self.entries.get(&to).cloned();
        if destination.is_some() && !overwrite {
            return Err(Error::Exists(to));
        }

        if let Some(dest) = &destination {
            match (&source, dest) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(to)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(to)),
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&to).is_empty() {
                        return Err(Error::NotEmpty(to));
                    }
                }
                _ => {}
            }
        }

        let mut moved = Vec::new();
        for (old_path, entry) in &self.entries {
            if old_path == &from || Self::contains_path(&from, old_path) {
                let suffix = &old_path[from.len()..];
                moved.push((format!("{to}{suffix}"), entry.clone()));
            }
        }

        if destination.is_some() {
            self.entries.retain(|key, _| {
                key != &to && !Self::contains_path(&to, key)
            });
        }
        self.entries.retain(|key, _| key != &from && !Self::contains_path(&from, key));
        for (new_path, entry) in moved {
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&path)?;
        let entry = self
            .entries
            .get(&path)
            .cloned()
            .ok_or_else(|| Error::Missing(path.clone()))?;

        if matches!(entry, Entry::Dir) && self.is_cwd_or_ancestor(&path) {
            return Err(Error::Busy(path));
        }

        if matches!(entry, Entry::Dir) && !recursive && !self.children(&path).is_empty() {
            return Err(Error::NotEmpty(path));
        }

        self.entries
            .retain(|key, _| key != &path && !Self::contains_path(&path, key));
        Ok(())
    }
}
```
