    match mode {
        Mode::Markdown => 3,
        Mode::Grid | Mode::Compact => 1,
    }

fn fixed_overhead(mode: &Mode, columns: usize) -> usize {
    match mode {
        Mode::Grid | Mode::Markdown => 3 * columns + 1,
        Mode::Compact => 2 * columns.saturating_sub(1),
    }
}

fn invalid_scalar(ch: char) -> bool {
    ch.is_control() && ch != '\n' && ch != '\r' || ch == '\u{2028}' || ch == '\u{2029}'
}

fn validate_cell(
    cell: &str,
    row: usize,
    column: usize,
) -> Result<(), TableError> {
    let mut chars = cell.char_indices().peekable();
    while let Some((byte, ch)) = chars.next() {
        if ch == '\r' {
            match chars.peek() {
                Some((_, '\n')) => {
                    chars.next();
                }
                _ => {
                    return Err(TableError::InvalidChar { row, column, byte });
                }
            }
        } else if invalid_scalar(ch) {
            return Err(TableError::InvalidChar { row, column, byte });
        }
    }
    Ok(())
}

fn tokenize_line(line: &str, markdown: bool) -> Vec<Token> {
    let mut tokens = Vec::new();
    for ch in line.chars() {
        if markdown && ch == '|' {
            tokens.push(Token { text: "\\|".to_owned(), width: 2 });
        } else if markdown && ch == '\\' {
            tokens.push(Token { text: "\\\\".to_owned(), width: 2 });
        } else {
            tokens.push(Token { text: ch.to_string(), width: 1 });
        }
    }
    tokens
}

fn prepare_cell(cell: &str, markdown: bool) -> LogicalLines {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            tokenize_line(trimmed, markdown)
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|token| token.width).sum()
}

fn allocate_widths(
    natural: &[usize],
    minimum_width: usize,
    overhead: usize,
    budget: usize,
) -> Vec<usize> {
    let mut widths = natural.to_vec();
    if budget == 0 {
        return widths;
    }

    while widths.iter().sum::<usize>() + overhead > budget {
        let candidate = (0..widths.len())
            .filter(|&i| widths[i] > minimum_width)
            .max_by_key(|&i| (widths[i], i));
        match candidate {
            Some(i) => widths[i] -= 1,
            None => break,
        }
    }
    widths
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<Vec<Token>> {
    if tokens.is_empty() {
        return vec![Vec::new()];
    }

    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.to_vec()];
            }

            let content_width = width.saturating_sub(1);
            let mut used = 0;
            let mut prefix = Vec::new();
            for token in tokens {
                if used + token.width > content_width {
                    break;
                }
                used += token.width;
                prefix.push(token.clone());
            }
            prefix.push(Token { text: "…".to_owned(), width: 1 });
            vec![prefix]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut used = 0;
                let mut end = start;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                if end == start {
                    // All tokens are one scalar except Markdown escapes, and the
                    // Markdown minimum width guarantees they fit.
                    end += 1;
                }
                result.push(tokens[start..end].to_vec());
                start = end;
            }
            result
        }
    }
}

fn fragment_text(tokens: &[Token]) -> String {
    tokens.iter().map(|token| token.text.as_str()).collect()
}

fn align_text(text: &str, width: usize, alignment: &Alignment, omit_right: bool) -> String {
    let content_width = text.chars().count();
    let extra = width.saturating_sub(content_width);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };

    let mut result = String::new();
    result.extend(std::iter::repeat_n(' ', left));
    result.push_str(text);
    if !omit_right {
        result.extend(std::iter::repeat_n(' ', right));
    }
    result
}

fn emit_data_line(
    output: &mut String,
    cells: &[String],
    widths: &[usize],
    alignments: &[Alignment],
    mode: &Mode,
) {
    match mode {
        Mode::Grid | Mode::Markdown => {
            output.push('|');
            for (i, text) in cells.iter().enumerate() {
                output.push(' ');
                output.push_str(&align_text(text, widths[i], &alignments[i], false));
                output.push(' ');
                output.push('|');
            }
            output.push('\n');
        }
        Mode::Compact => {
            for (i, text) in cells.iter().enumerate() {
                if i > 0 {
                    output.push_str("  ");
                }
                output.push_str(&align_text(
                    text,
                    widths[i],
                    &alignments[i],
                    i + 1 == cells.len(),
                ));
            }
            output.push('\n');
        }
    }
}

fn emit_grid_border(output: &mut String, widths: &[usize]) {
    output.push('+');
    for width in widths {
        output.extend(std::iter::repeat_n('-', width + 2));
        output.push('+');
    }
    output.push('\n');
}

    output: &mut String,
    widths: &[usize],
    alignments: &[Alignment],

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let columns = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != columns {
            return Err(TableError::Ragged {
                row,
                expected: columns,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != columns {
        return Err(TableError::AlignmentCount {
            expected: columns,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != columns {
        return Err(TableError::OverflowCount {
            expected: columns,
            actual: options.overflow.len(),
        });
    }
    if matches!(options.mode, Mode::Markdown) && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let min_width = min_column_width(&options.mode);
    let overhead = fixed_overhead(&options.mode, columns);
    let minimum = overhead + columns * min_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            validate_cell(cell, row, column)?;
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; columns]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; columns]
    } else {
        options.overflow.clone()
    };

    let markdown = matches!(options.mode, Mode::Markdown);
    let prepared: Vec<Vec<LogicalLines>> = rows
        .iter()
        .map(|row| {
            row.iter()
                .map(|cell| prepare_cell(cell, markdown))
                .collect()
        })
        .collect();

    let mut natural = vec![min_width; columns];
    for row in &prepared {
        for (column, lines) in row.iter().enumerate() {
            for line in lines {
                natural[column] = natural[column].max(token_width(line));
            }
        }
    }

    let widths = allocate_widths(&natural, min_width, overhead, options.max_width);
    let mut output = String::new();

    if matches!(options.mode, Mode::Grid) {
        emit_grid_border(&mut output, &widths);
    }

    for (row_index, row) in prepared.iter().enumerate() {
        let cell_fragments: Vec<Vec<Vec<Token>>> = row
            .iter()
            .enumerate()
            .map(|(column, lines)| {
                let mut all = Vec::new();
                for line in lines {
                    all.extend(fragments(line, widths[column], &overflows[column]));
                }
                all
            })
            .collect();

        let height = cell_fragments.iter().map(Vec::len).max().unwrap_or(1);
        for fragment_index in 0..height {
            let cells: Vec<String> = cell_fragments
                .iter()
                .map(|parts| {
                    parts.get(fragment_index)
                        .map(|part| fragment_text(part))
                        .unwrap_or_default()
                })
                .collect();
            emit_data_line(&mut output, &cells, &widths, &alignments, &options.mode);
        }

        if matches!(options.mode, Mode::Grid)
            && options.header
            && row_index == 0
            && rows.len() > 1
        {
            emit_grid_border(&mut output, &widths);
        }

        if matches!(options.mode, Mode::Markdown) && row_index == 0 {
            emit_markdown_separator(&mut output, &widths, &alignments);
        }
    }

    if matches!(options.mode, Mode::Grid) {
        emit_grid_border(&mut output, &widths);
    }

    Ok(output)
}
