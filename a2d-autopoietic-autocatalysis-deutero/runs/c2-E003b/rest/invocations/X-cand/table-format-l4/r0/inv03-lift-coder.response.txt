#[derive(Debug, Clone, PartialEq, Eq)]
pub enum TableError {
    EmptyRow { row: usize },
    Ragged { row: usize, expected: usize, actual: usize },
    InvalidChar { row: usize, column: usize, byte: usize },
    AlignmentCount { expected: usize, actual: usize },
    TooNarrow { minimum: usize, provided: usize },
    OverflowCount { expected: usize, actual: usize },
    MarkdownNeedsHeader,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Alignment {
    Left,
    Right,
    Center,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Overflow {
    Clip,
    Wrap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Mode {
    Grid,
    Compact,
    Markdown,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Options {
    pub alignments: Vec<Alignment>,
    pub header: bool,
    pub max_width: usize,
    pub overflow: Vec<Overflow>,
    pub mode: Mode,
}

#[derive(Clone)]
struct Token {
    text: String,
    width: usize,
}

fn invalid_char(c: char) -> bool {
    c == '\r' || c.is_control() && c != '\n' || c == '\u{2028}' || c == '\u{2029}'
}

fn validate_cell(cell: &str, row: usize, column: usize) -> Result<(), TableError> {
    let mut iter = cell.char_indices().peekable();
    while let Some((byte, c)) = iter.next() {
        if c == '\r' {
            if matches!(iter.peek(), Some((_, '\n'))) {
                iter.next();
            } else {
                return Err(TableError::InvalidChar { row, column, byte });
            }
        } else if invalid_char(c) {
            return Err(TableError::InvalidChar { row, column, byte });
        }
    }
    Ok(())
}

fn logical_lines(cell: &str, markdown: bool) -> Vec<Vec<Token>> {
    let normalized = cell.replace("\r\n", "\n");
    normalized
        .split('\n')
        .map(|line| {
            let trimmed = line.trim_end_matches(' ');
            trimmed
                .chars()
                .flat_map(|c| {
                    if markdown && c == '|' {
                        vec![
                            Token { text: "\\".to_owned(), width: 1 },
                            Token { text: "|".to_owned(), width: 1 },
                        ]
                    } else if markdown && c == '\\' {
                        vec![
                            Token { text: "\\".to_owned(), width: 1 },
                            Token { text: "\\".to_owned(), width: 1 },
                        ]
                    } else {
                        vec![Token { text: c.to_string(), width: 1 }]
                    }
                })
                .collect()
        })
        .collect()
}

fn token_width(tokens: &[Token]) -> usize {
    tokens.iter().map(|t| t.width).sum()
}

fn fragments(tokens: &[Token], width: usize, overflow: &Overflow) -> Vec<String> {
    if tokens.is_empty() {
        return vec![String::new()];
    }
    match overflow {
        Overflow::Clip => {
            if token_width(tokens) <= width {
                return vec![tokens.iter().map(|t| t.text.as_str()).collect()];
            }
            let mut out = String::new();
            let mut used = 0;
            let limit = width.saturating_sub(1);
            for token in tokens {
                if used + token.width > limit {
                    break;
                }
                used += token.width;
                out.push_str(&token.text);
            }
            out.push('…');
            vec![out]
        }
        Overflow::Wrap => {
            let mut result = Vec::new();
            let mut start = 0;
            while start < tokens.len() {
                let mut end = start;
                let mut used = 0;
                while end < tokens.len() && used + tokens[end].width <= width {
                    used += tokens[end].width;
                    end += 1;
                }
                // Allocated widths always fit every indivisible token.
                if end == start {
                    end += 1;
                }
                result.push(tokens[start..end].iter().map(|t| t.text.as_str()).collect());
                start = end;
            }
            result
        }
    }
}

fn aligned(text: &str, width: usize, alignment: &Alignment) -> String {
    let used = text.chars().count();
    let extra = width.saturating_sub(used);
    let (left, right) = match alignment {
        Alignment::Left => (0, extra),
        Alignment::Right => (extra, 0),
        Alignment::Center => (extra / 2, extra - extra / 2),
    };
    format!("{}{}{}", " ".repeat(left), text, " ".repeat(right))
}

fn border(widths: &[usize]) -> String {
    let mut line = String::from("+");
    for width in widths {
        line.push_str(&"-".repeat(width + 2));
        line.push('+');
    }
    line.push('\n');
    line
}

fn data_line(parts: &[String], mode: &Mode) -> String {
    let mut line = String::new();
    match mode {
        Mode::Compact => line.push_str(&parts.join("  ")),
        Mode::Grid | Mode::Markdown => {
            line.push('|');
            for part in parts {
                line.push(' ');
                line.push_str(part);
                line.push_str(" |");
            }
        }
    }
    line.push('\n');
    line
}

pub fn render(rows: &[Vec<String>], options: &Options) -> Result<String, TableError> {
    if rows.is_empty() {
        return Ok(String::new());
    }

    for (row, cells) in rows.iter().enumerate() {
        if cells.is_empty() {
            return Err(TableError::EmptyRow { row });
        }
    }

    let n = rows[0].len();
    for (row, cells) in rows.iter().enumerate() {
        if cells.len() != n {
            return Err(TableError::Ragged {
                row,
                expected: n,
                actual: cells.len(),
            });
        }
    }

    if !options.alignments.is_empty() && options.alignments.len() != n {
        return Err(TableError::AlignmentCount {
            expected: n,
            actual: options.alignments.len(),
        });
    }
    if !options.overflow.is_empty() && options.overflow.len() != n {
        return Err(TableError::OverflowCount {
            expected: n,
            actual: options.overflow.len(),
        });
    }
    if options.mode == Mode::Markdown && !options.header {
        return Err(TableError::MarkdownNeedsHeader);
    }

    let overhead = match options.mode {
        Mode::Compact => 2 * (n - 1),
        Mode::Grid | Mode::Markdown => 3 * n + 1,
    };
    let min_width = if options.mode == Mode::Markdown { 3 } else { 1 };
    let minimum = overhead + n * min_width;
    if options.max_width != 0 && options.max_width < minimum {
        return Err(TableError::TooNarrow {
            minimum,
            provided: options.max_width,
        });
    }

    for (row, cells) in rows.iter().enumerate() {
        for (column, cell) in cells.iter().enumerate() {
            validate_cell(cell, row, column)?;
        }
    }

    let markdown = options.mode == Mode::Markdown;
    let content: Vec<Vec<Vec<Vec<Token>>>> = rows
        .iter()
        .map(|row| row.iter().map(|cell| logical_lines(cell, markdown)).collect())
        .collect();

    let mut widths = vec![min_width; n];
    for row in &content {
        for (column, cell_lines) in row.iter().enumerate() {
            for line in cell_lines {
                widths[column] = widths[column].max(token_width(line));
            }
        }
    }

    if options.max_width != 0 {
        let mut total = overhead + widths.iter().sum::<usize>();
        while total > options.max_width {
            let eligible = (0..n)
                .filter(|&c| widths[c] > min_width)
                .max_by_key(|&c| (widths[c], c));
            if let Some(column) = eligible {
                widths[column] -= 1;
                total -= 1;
            } else {
                break;
            }
        }
    }

    let alignments: Vec<Alignment> = if options.alignments.is_empty() {
        vec![Alignment::Left; n]
    } else {
        options.alignments.clone()
    };
    let overflows: Vec<Overflow> = if options.overflow.is_empty() {
        vec![Overflow::Wrap; n]
    } else {
        options.overflow.clone()
    };

    let mut output = String::new();
    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }

    for (row_index, row) in content.iter().enumerate() {
        let mut cell_fragments = Vec::with_capacity(n);
        let mut height = 1;
        for column in 0..n {
            let mut all = Vec::new();
            for line in &row[column] {
                all.extend(fragments(line, widths[column], &overflows[column]));
            }
            height = height.max(all.len());
            cell_fragments.push(all);
        }

        for fragment_index in 0..height {
            let mut parts = Vec::with_capacity(n);
            for column in 0..n {
                let text = cell_fragments[column]
                    .get(fragment_index)
                    .map(String::as_str)
                    .unwrap_or("");
                let mut part = aligned(text, widths[column], &alignments[column]);
                if options.mode == Mode::Compact && column + 1 == n {
                    let used = text.chars().count();
                    let right_padding = match alignments[column] {
                        Alignment::Left => widths[column].saturating_sub(used),
                        Alignment::Right => 0,
                        Alignment::Center => (widths[column].saturating_sub(used) + 1) / 2,
                    };
                    for _ in 0..right_padding {
                        part.pop();
                    }
                }
                parts.push(part);
            }
            output.push_str(&data_line(&parts, &options.mode));
        }

        if options.mode == Mode::Markdown && row_index == 0 {
            let mut marker_parts = Vec::with_capacity(n);
            for column in 0..n {
                let width = widths[column];
                let marker = match alignments[column] {
                    Alignment::Left => format!(":{}", "-".repeat(width - 1)),
                    Alignment::Right => format!("{}:", "-".repeat(width - 1)),
                    Alignment::Center => format!(":{}:", "-".repeat(width - 2)),
                };
                marker_parts.push(marker);
            }
            output.push_str(&data_line(&marker_parts, &Mode::Markdown));
        }

        if options.mode == Mode::Grid && options.header && row_index == 0 && rows.len() > 1 {
            output.push_str(&border(&widths));
        }
    }

    if options.mode == Mode::Grid {
        output.push_str(&border(&widths));
    }
    Ok(output)
}
