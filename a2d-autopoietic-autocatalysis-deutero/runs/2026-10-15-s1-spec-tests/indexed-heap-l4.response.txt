    let mut receiver = IndexedHeap::with_capacity(2);
    receiver.push("keep", 5, "").unwrap();
    let mut source = IndexedHeap::new();
    source.push("same", 0, "source").unwrap();
    source.push("new", 1, "").unwrap();
    receiver.push("same", 9, "receiver").unwrap();
    let report = receiver.merge(&mut source, CollisionPolicy::TakeOther).unwrap();
    assert_eq!((report.moved, report.replaced, report.discarded), (1, 1, 0));
    assert_eq!(report.evicted.len(), 1);
    assert_eq!(report.evicted[0].key, "keep");
    assert_eq!(receiver.len(), 2);
    assert!(source.is_empty());
}

#[test]
fn spec_r48_merge_preserves_capacities_and_rejects_before_trimming() {
    let mut receiver = IndexedHeap::with_capacity(0);
    let mut source = IndexedHeap::with_capacity(3);
    receiver.push("same", 0, "receiver").unwrap();
    source.push("SAME", -1, "source").unwrap();
    let before_receiver = receiver.iter().collect::<Vec<_>>();
    let before_source = source.iter().collect::<Vec<_>>();
    assert_eq!(
        receiver.merge(&mut source, CollisionPolicy::Reject),
        Err(Error::Collision { key: "same".into() })
    );
    assert_eq!(receiver.capacity(), Some(0));
    assert_eq!(source.capacity(), Some(3));
    assert_eq!(receiver.iter().collect::<Vec<_>>(), before_receiver);
    assert_eq!(source.iter().collect::<Vec<_>>(), before_source);

    let mut dst = IndexedHeap::with_capacity(1);
    let mut src = IndexedHeap::with_capacity(2);
    dst.push("d", 5, "").unwrap();
    src.push("s", 0, "").unwrap();
    dst.merge(&mut src, CollisionPolicy::KeepSelf).unwrap();
    assert_eq!(dst.capacity(), Some(1));
    assert_eq!(src.capacity(), Some(2));
    assert!(src.is_empty());
}
