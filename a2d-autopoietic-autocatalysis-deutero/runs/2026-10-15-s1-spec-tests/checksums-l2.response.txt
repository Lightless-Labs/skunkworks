#[test]
fn spec_r2_public_types_and_traits() {
    fn assert_traits<T: std::fmt::Debug + Clone + Copy + PartialEq + Eq>() {}
    assert_traits::<Span>();
    assert_traits::<Hashes>();
    assert_traits::<SpanError>();
    assert_traits::<SnapshotError>();

    let span = Span { start: 1, end: 2 };
    let hashes = Hashes { crc32: 3, adler32: 4 };
    assert_eq!(span, span.clone());
    assert_eq!(hashes, hashes.clone());
}

#[test]
fn spec_r6_adler32_empty_and_long_input() {
    assert_eq!(adler32(b""), 1);
    let data = vec![255u8; 1_000_000];
    let mut a = 1u64;
    let mut b = 0u64;
    for _ in &data {
        a = (a + 255) % 65521;
        b = (b + a) % 65521;
    }
    assert_eq!(adler32(&data), ((b as u32) << 16) | a as u32);
}

#[test]

#[test]
fn spec_r21_seed_resumes_from_finalized_checksums() {
    let prefix = b"seeded prefix";
    let seed = checksums(prefix);
    let mut stream = Stream::from_seed(seed.crc32, seed.adler32).unwrap();
    stream.update(b" suffix");
    assert_eq!(stream.checksum(), checksums(b"seeded prefix suffix"));
}

#[test]
fn spec_r22_seed_adler_components_are_validated_in_order() {
    assert_eq!(
        Stream::from_seed(0, 0x0001_0000),
        Err(SnapshotError::Adler { component: 0 })
    );
    assert_eq!(
        Stream::from_seed(0, 0x0001_0000 | 65_521),
        Err(SnapshotError::Adler { component: 1 })
    );
    assert_eq!(
        Stream::from_seed(0, 0),
        Ok(Stream::from_seed(0, 0).unwrap())
    );
}

#[test]
fn spec_r23_stream_chunking_and_empty_update_preserve_seed() {
    let mut whole = Stream::new();
    whole.update(&[0, 1, 0x80, 0xff]);

    let mut chunked = Stream::new();
    chunked.update(&[0]);
    chunked.update(&[]);
    chunked.update(&[1, 0x80]);
    chunked.update(&[0xff]);
    assert_eq!(chunked.checksum(), whole.checksum());

    let mut seeded = Stream::from_seed(0x12345678, 0x00020003).unwrap();
    let before = seeded.checksum();
    seeded.update(&[]);
    assert_eq!(seeded.checksum(), before);
}

#[test]
fn spec_r25_reset_and_clone_are_independent() {
    let mut original = Stream::new();
    original.update(b"first");
    let mut copy = original.clone();
    original.update(b"!");
    copy.update(b"?");
    assert_eq!(original.checksum(), checksums(b"first!"));
    assert_eq!(copy.checksum(), checksums(b"first?"));

    let mut seeded = Stream::from_seed(checksums(b"seed").crc32, checksums(b"seed").adler32).unwrap();
    seeded.reset();
    assert_eq!(seeded.checksum(), checksums(b""));
}

#[test]
fn spec_r29_restore_validates_all_hex_before_adler_seed() {
    assert_eq!(
        Stream::restore("c00000000:a0000000G"),
        Err(SnapshotError::Hex { offset: 18 })
    );
    assert_eq!(
        Stream::restore("c00000000:a00010000"),
        Err(SnapshotError::Adler { component: 0 })
    );

    let mut original = Stream::new();
    original.update(b"before");
    let snapshot = original.snapshot();
    let mut restored = Stream::restore(&snapshot).unwrap();
    original.update(b"after");
    restored.update(b"after");
    assert_eq!(restored.checksum(), original.checksum());
    assert_eq!(restored.snapshot(), original.snapshot());
}

    let mut stream = Stream::new();
    stream.update(b"stable");
    let _ = stream.snapshot();
    let _ = stream.checksum();
    let _ = stream.snapshot();
    stream.update(b" continuation");
    assert_eq!(stream.checksum(), checksums(b"stable continuation"));
}
