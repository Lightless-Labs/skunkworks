fn spec_r1_single_file_api_usage() {
    let input = [0u8];
    assert_eq!(decode(&encode(&input, Alphabet::Standard, Padding::Omitted), Alphabet::Standard, Padding::Omitted), Ok(input.to_vec()));
}

fn spec_r2_public_types_and_operations() {
    let _: Alphabet = Alphabet::Standard;
    let _: Padding = Padding::Required;
    let result: Result<Vec<u8>, DecodeError> = decode("", Alphabet::Hex, Padding::Omitted);
    assert_eq!(result, Ok(Vec::new()));
}

#[test]
fn spec_r5_uppercase_only_and_no_whitespace_or_hyphens() {
    for (input, byte) in [("a", b'a'), (" ", b' '), ("-", b'-')] {
        assert_eq!(decode(input, Alphabet::Standard, Padding::Omitted), Err(DecodeError::InvalidByte { offset: 0, byte }));
        assert_eq!(decode(input, Alphabet::Hex, Padding::Omitted), Err(DecodeError::InvalidByte { offset: 0, byte }));
    }

fn spec_r7_required_padding_to_eight_symbols() {

    assert_eq!(encode(&[0, 0, 0, 0, 0], Alphabet::Standard, Padding::Required), "AAAAAAAA");

#[test]
fn spec_r9_empty_input_and_output() {
    for alphabet in [Alphabet::Standard, Alphabet::Hex] {
        for padding in [Padding::Required, Padding::Omitted] {
            assert_eq!(encode(&[], alphabet, padding), "");
            assert_eq!(decode("", alphabet, padding), Ok(Vec::new()));
        }
    }

#[test]
fn spec_r10_earliest_validation_pass_wins() {
    assert_eq!(decode("A!=", Alphabet::Omitted, Padding::Required), Err(DecodeError::InvalidByte { offset: 1, byte: b'!' }));
    assert_eq!(decode("A!=", Alphabet::Standard, Padding::Omitted), Err(DecodeError::InvalidByte { offset: 1, byte: b'!' }));
    assert_eq!(decode("A=======", Alphabet::Standard, Padding::Required), Err(DecodeError::InvalidLength { offset: 8 }));
    assert_eq!(decode("AB======", Alphabet::Standard, Padding::Required), Err(DecodeError::TrailingBits { offset: 1 }));
}

#[test]
fn spec_r11_lexical_scan_checks_all_original_bytes_before_padding_rules() {
    assert_eq!(decode("A=!", Alphabet::Standard, Padding::Omitted), Err(DecodeError::InvalidByte { offset: 2, byte: b'!' }));
    assert_eq!(decode("A=!", Alphabet::Hex, Padding::Required), Err(DecodeError::InvalidByte { offset: 2, byte: b'!' }));
}

#[test]
fn spec_r12_offsets_are_utf8_byte_offsets_and_report_first_non_ascii_byte() {
    assert_eq!(decode("AAé", Alphabet::Standard, Padding::Omitted), Err(DecodeError::InvalidByte { offset: 2, byte: 0xC3 }));
    assert_eq!(decode("é", Alphabet::Hex, Padding::Required), Err(DecodeError::InvalidByte { offset: 0, byte: 0xC3 }));
}

#[test]
fn spec_r13_omitted_padding_is_rejected_after_lexical_pass() {
    assert_eq!(decode("AA=AA", Alphabet::Standard, Padding::Omitted), Err(DecodeError::Padding { offset: 2 }));
    assert_eq!(decode("A=!", Alphabet::Standard, Padding::Omitted), Err(DecodeError::InvalidByte { offset: 2, byte: b'!' }));

#[test]
fn spec_r14_required_padding_rejects_data_after_first_equals() {

}

#[test]
fn spec_r15_legal_data_length_residues_and_invalid_length_offset() {
    for (data, padding) in [("A", Padding::Omitted), ("AAA", Padding::Omitted), ("AAAAAA", Padding::Omitted), ("AAAAAAAAA", Padding::Omitted), ("A", Padding::Required)] {
        let expected_len = data.len() + usize::from(padding == Padding::Required);
        let input = format!("{data}{}", "=".repeat(expected_len - data.len()));
        assert_eq!(decode(&input, Alphabet::Standard, padding), Err(DecodeError::InvalidLength { offset: input.len() }));
    }
}

#[test]
fn spec_r16_required_padding_counts_follow_data_residue() {
    assert_eq!(decode("AA======", Alphabet::Standard, Padding::Required), Ok(vec![0]));
    assert_eq!(decode("AAAA====", Alphabet::Standard, Padding::Required), Ok(vec![0, 0]));
    assert_eq!(decode("AAAAA===", Alphabet::Standard, Padding::Required), Ok(vec![0, 0, 0]));
    assert_eq!(decode("AAAAAAA=", Alphabet::Standard, Padding::Required), Ok(vec![0, 0, 0, 0]));
    assert_eq!(decode("AAAAAAAA", Alphabet::Standard, Padding::Required), Ok(vec![0, 0, 0, 0, 0]));
}

#[test]
fn spec_r17_required_padding_shape_and_edge_cases() {

    assert_eq!(decode("AAAAAAAA======", Alphabet::Standard, Padding::Required), Err(DecodeError::Padding { offset: 8 }));

    assert_eq!(decode("AB=====", Alphabet::Standard, Padding::Required), Err(DecodeError::Padding { offset: 2 }));

#[test]
fn spec_r23_arbitrary_bytes_repeated_blocks_and_round_trip_canonicality() {
    let raw: Vec<u8> = (0..=255).chain(0..=255).collect();
    for alphabet in [Alphabet::Standard, Alphabet::Hex] {
        for padding in [Padding::Required, Padding::Omitted] {
            let encoded = encode(&raw, alphabet, padding);
            assert_eq!(decode(&encoded, alphabet, padding), Ok(raw.clone()));
            assert_eq!(encode(&decode(&encoded, alphabet, padding).unwrap(), alphabet, padding), encoded);
        }
    }
}

    assert_eq!(decode("MZXW6", Alphabet::Standard, Padding::Omitted), Ok(b"foo".to_vec()));
    assert_eq!(encode(b"foo", Alphabet::Standard, Padding::Omitted), "MZXW6");
    assert_eq!(encode(b"foo", Alphabet::Hex, Padding::Required), "CPNMU===");
