    assert_debug::<Sparse>();

#[test]
fn spec_r6_nonfinite_values_and_bounds_priority() {
    for value in [f64::NAN, f64::INFINITY, f64::NEG_INFINITY] {
        assert_eq!(
            Sparse::from_coo(1, 1, &[Entry { row: 0, col: 0, value }]),
            Err(Error::NonFinite { entry: 0 })
        );
    }
    assert!(Sparse::from_coo(1, 1, &[Entry { row: 0, col: 0, value: 0.0 }]).is_ok());
    assert!(Sparse::from_coo(1, 1, &[Entry { row: 0, col: 0, value: -0.0 }]).is_ok());
    assert!(Sparse::from_coo(1, 1, &[Entry { row: 0, col: 0, value: f64::MIN_POSITIVE / 2.0 }]).is_ok());
    assert_eq!(
        Sparse::from_coo(1, 1, &[Entry { row: 0, col: 1, value: f64::NAN }]),
        Err(Error::Bounds { entry: 0, axis: Axis::Column, index: 1, limit: 1 })
    );
}

#[test]
fn spec_r14_input_unchanged_clone_and_accessors_preserve_contents() {
    let input = [
        Entry { row: 1, col: 0, value: 3.0 },
        Entry { row: 0, col: 1, value: 2.0 },
    ];
    let original = input;
    let sparse = Sparse::from_coo(2, 2, &input).unwrap();
    assert_eq!(input, original);
    let cloned = sparse.clone();
    assert_eq!(cloned.shape(), (2, 2));
    assert_eq!(cloned.entries(), sparse.entries());
    let before = sparse.entries().to_vec();
    let _ = sparse.shape();
    let _ = sparse.entries();
    assert_eq!(sparse.entries(), before.as_slice());
}
