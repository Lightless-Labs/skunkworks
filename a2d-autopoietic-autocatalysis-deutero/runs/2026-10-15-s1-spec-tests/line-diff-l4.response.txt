            side: MergeSide::Base

        Err(MergeError::MissingFinalLf {
            side: MergeSide::Left
        })

    assert_eq!(
        merge_three_way("", "", "c", Resolution::PreferLeft),
        Err(MergeError::MissingFinalLf {
            side: MergeSide::Right
        })
    );

    let too_many = format!("{}\n", vec!["x"; 129].join("\n"));
    assert_eq!(
        merge_three_way(&too_many, &too_many, "r\n", Resolution::PreferLeft),
