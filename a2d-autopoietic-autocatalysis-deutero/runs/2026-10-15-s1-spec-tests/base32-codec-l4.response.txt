        canonical: true,
    };
    let mut one = Encoder::new(c);
    let whole = format!("{}{}", one.push(b"abcdefghij").unwrap(), one.finish().unwrap());

    let mut chunked = Encoder::new(c);
    let mut encoded = chunked.push(b"abc").unwrap();
    encoded.push_str(&chunked.push(b"defg").unwrap());
    encoded.push_str(&chunked.push(b"hij").unwrap());
    encoded.push_str(&chunked.finish().unwrap());
    assert_eq!(encoded, whole);

    let mut d = Decoder::new(c);
    assert_eq!(d.push(&whole).unwrap(), b"abcdefghij");
    assert_eq!(d.finish(), Ok(vec![]));

    let mut d = Decoder::new(c);
    assert_eq!(d.push(&whole[..3]).unwrap(), vec![b'a']);
    assert_eq!(d.push(&whole[3..]).unwrap(), b"bcdefghij");
    assert_eq!(d.finish(), Ok(vec![]));

    let mut d = Decoder::new(c);
    assert_eq!(d.push(&whole).unwrap(), b"abcdefghij");
    assert_eq!(
        d.push("?"),
        Err(StreamError::InvalidByte {
            offset: whole.len(),
            byte: b'?'
        })
    );
}

#[test]
fn spec_r43_long_streams_preserve_pending_state_and_absolute_indices() {
    let c = Config {
        alphabet: Alphabet::Standard,
        padding: Padding::Omitted,
        canonical: true,
    };
    let input = vec![0xA5; 5000];
    let mut e = Encoder::new(c);
    let mut encoded = String::new();
    for chunk in input.chunks(3) {
        encoded.push_str(&e.push(chunk).unwrap());
    }
    encoded.push_str(&e.finish().unwrap());

    let mut d = Decoder::new(c);
    let mut decoded = Vec::new();
    for chunk in encoded.as_bytes().chunks(7) {
        decoded.extend(d.push(std::str::from_utf8(chunk).unwrap()).unwrap());
    }
    assert_eq!(decoded, input);
    assert_eq!(d.finish(), Ok(vec![]));

    let mut d = Decoder::new(c);
    let mut bad = "A".repeat(10_000);
    bad.push('!');
    assert_eq!(
        d.push(&bad),
        Err(StreamError::InvalidByte {
            offset: 10_000,
            byte: b'!'
        })
    );
}
