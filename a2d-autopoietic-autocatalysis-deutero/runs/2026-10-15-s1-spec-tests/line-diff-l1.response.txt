    );
}

#[test]
fn spec_r9_script_consumes_each_side_once_in_order() {
    let old = vec!["a".to_string(), "b".to_string(), "c".to_string()];
    let new = vec!["b".to_string(), "d".to_string()];
    let edits = diff_lines(&old, &new).unwrap();

    let mut old_indices = Vec::new();
    let mut new_indices = Vec::new();
    for edit in edits {
        match edit {
            Edit::Keep {
                old_index,
                new_index,
                ..
            } => {
                old_indices.push(old_index);
                new_indices.push(new_index);
            }
            Edit::Delete { old_index, .. } => old_indices.push(old_index),
            Edit::Insert { new_index, .. } => new_indices.push(new_index),
        }
    }
    assert_eq!(old_indices, vec![0, 1, 2]);
    assert_eq!(new_indices, vec![0, 1]);
}

#[test]
fn spec_r10_minimizes_insertions_and_deletions() {
    let old = vec!["a".to_string(), "b".to_string()];
    let new = vec!["b".to_string(), "a".to_string()];
    let edits = diff_lines(&old, &new).unwrap();
    let edit_cost = edits
        .iter()
        .filter(|edit| matches!(edit, Edit::Delete { .. } | Edit::Insert { .. }))
        .count();
    assert_eq!(edit_cost, 2);
}

#[test]
fn spec_r11_lcs_choice_preserves_a_longest_common_subsequence() {
    let old = vec!["a".to_string(), "b".to_string(), "c".to_string()];
    let new = vec!["b".to_string(), "c".to_string(), "a".to_string()];
    let edits = diff_lines(&old, &new).unwrap();
    let keeps: Vec<(&str, usize, usize)> = edits
        .iter()
        .filter_map(|edit| match edit {
            Edit::Keep {
                old_index,
                new_index,
                line,
            } => Some((line.as_str(), *old_index, *new_index)),
            _ => None,
        })
        .collect();
    assert_eq!(keeps, vec![("b", 1, 0), ("c", 2, 1)]);
}
