#[test]
fn spec_r52_canonicalization_idempotent_without_side_effects() {
    let input = String::from("HTTP://EXAMPLE/a/./b?x=one+two#top");
    let before = input.clone();
    let once = canonical_url(&input).unwrap();
    assert_eq!(canonical_url(&once).unwrap(), once);
    assert_eq!(parse_url(&input).unwrap().path, "/a/./b");
    assert_eq!(input, before);
    assert_eq!(
        decode_component(Component::Query, "+").unwrap(),
        "+"
    );
}
