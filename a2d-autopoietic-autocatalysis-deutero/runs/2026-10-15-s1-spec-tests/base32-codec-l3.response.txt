#[test]
fn spec_r1_single_source_compatible_api() {
    let encoded: String = encode(&[], false);
    let decoded: Result<Vec<u8>, DecodeError> = decode(&encoded, false);
    assert_eq!(decoded, Ok(Vec::new()));
}

#[test]
fn spec_r2_error_variants_and_shapes() {
    assert_eq!(
        DecodeError::MissingCheck { offset: 3 },
        DecodeError::MissingCheck { offset: 3 }
    );
    assert_eq!(
        DecodeError::InvalidByte { offset: 1, byte: b'?' },
        DecodeError::InvalidByte { offset: 1, byte: b'?' }
    );
    assert_eq!(
        DecodeError::InvalidLength { offset: 4 },
        DecodeError::InvalidLength { offset: 4 }
    );
    assert_eq!(
        DecodeError::TrailingBits { offset: 2 },
        DecodeError::TrailingBits { offset: 2 }
    );
    assert_eq!(
        DecodeError::CheckMismatch {
            offset: 5,
            expected: 3,
            found: 4
        },
        DecodeError::CheckMismatch {
            offset: 5,
            expected: 3,
            found: 4
        }
    );
    let cloned = DecodeError::InvalidLength { offset: 7 }.clone();
    assert_eq!(format!("{cloned:?}"), "InvalidLength { offset: 7 }");
}

#[test]
fn spec_r3_data_alphabet_values() {
    let alphabet = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ";
    for (value, &symbol) in alphabet.iter().enumerate() {
        assert_eq!(encode(&[value as u8], false), format!("{}", symbol as char));
    }
}

#[test]
fn spec_r4_most_significant_bit_first_and_zero_fill() {
    assert_eq!(encode(&[0b1010_0000], false), "M0");
    assert_eq!(encode(&[0b0000_0011], false), "06");
    assert_eq!(encode(&[0b1111_1111], false), "ZW");
}

#[test]
fn spec_r5_unpadded_without_empty_final_group() {
    assert_eq!(encode(&[0], false), "00");
    assert_eq!(encode(&[0, 0], false), "0000");
    assert_eq!(decode("000", false), Err(DecodeError::InvalidLength { offset: 3 }));
}

#[test]
fn spec_r6_empty_encoding_by_checked_flag() {
    assert_eq!(encode(&[], false), "");
    assert_eq!(encode(&[], true), "0");
}

#[test]
fn spec_r7_check_alphabet_and_exactly_one_check() {
    let alphabet = b"0123456789ABCDEFGHJKMNPQRSTVWXYZ*~$=U";
    for (value, &symbol) in alphabet.iter().enumerate() {
        assert_eq!(encode(&[value as u8], true).len(), 3);
        assert_eq!(
            decode(&format!("{}{}{}", (value as u8 / 32), (value as u8 % 32), symbol as char), true)
                .is_ok(),
            true
        );
    }
    assert_eq!(encode(&[], true), "0");
}

#[test]
fn spec_r8_check_remainder_includes_zero_filled_symbol() {
    assert_eq!(encode(&[0], true), "000");
    assert_eq!(encode(&[1], true), "080");
    assert_eq!(encode(&[0xff], true), "ZWU");
}

#[test]
fn spec_r9_hyphens_ignored_everywhere() {
    assert_eq!(decode("---c--s-q-p-y---", false), Ok(b"foo".to_vec()));
    assert_eq!(decode("-8-0-$-", true), Ok(vec![64]));
    assert_eq!(decode("----", false), Ok(Vec::new()));
}

#[test]
fn spec_r10_data_case_folding_and_aliases() {
    assert_eq!(decode("csqpy", false), Ok(b"foo".to_vec()));
    assert_eq!(decode("O1", false), Ok(vec![0]));
    assert_eq!(decode("I1", false), Ok(vec![8]));
    assert_eq!(decode("L1", false), Ok(vec![8]));
    assert_eq!(decode("i1", false), Ok(vec![8]));
    assert_eq!(decode("l1", false), Ok(vec![8]));
    assert_eq!(decode("o1", false), Ok(vec![0]));
}

#[test]
fn spec_r11_check_aliases_case_and_extended_symbols() {
    assert_eq!(decode("00u", true), Ok(Vec::new()));
    assert_eq!(decode("00U", true), Ok(Vec::new()));
    assert_eq!(decode("00*", true), Err(DecodeError::CheckMismatch {
        offset: 2, expected: 0, found: 32
    }));
    assert_eq!(decode("00~", true), Err(DecodeError::CheckMismatch {
        offset: 2, expected: 0, found: 33
    }));
    assert_eq!(decode("00$", true), Err(DecodeError::CheckMismatch {
        offset: 2, expected: 0, found: 34
    }));
    assert_eq!(decode("00=", true), Err(DecodeError::CheckMismatch {
        offset: 2, expected: 0, found: 35
    }));
    assert_eq!(decode("I0", false), Err(DecodeError::InvalidLength { offset: 2 }));
    assert_eq!(decode("O0", false), Err(DecodeError::InvalidLength { offset: 2 }));
}

#[test]
fn spec_r12_error_offsets_are_original_utf8_byte_indices() {
    assert_eq!(
        decode("-?", false),
        Err(DecodeError::InvalidByte { offset: 1, byte: b'?' })
    );
    assert_eq!(
        decode("0-?", true),
        Err(DecodeError::InvalidByte { offset: 2, byte: b'?' })
    );
    assert_eq!(
        decode("0-0-", true),
        Err(DecodeError::CheckMismatch { offset: 2, expected: 0, found: 0 })
    );
}

#[test]
fn spec_r13_whitespace_newlines_and_non_ascii_are_invalid() {
    for input in [" ", "\n", "\r\n", "é", "🙂"] {

        assert_eq!(
            decode(input, false),
            Err(DecodeError::InvalidByte { offset: 0, byte: expected_byte })
        );

    assert_eq!(
        decode("00é", true),
        Err(DecodeError::InvalidByte { offset: 2, byte: 0xc3 })
    );
}

#[test]
fn spec_r28_arbitrary_binary_bytes_and_repeated_groups() {
    let bytes: Vec<u8> = (0..=255).collect();
    for checked in [false, true] {
        let encoded = encode(&bytes, checked);
        assert_eq!(decode(&encoded, checked), Ok(bytes.clone()));
    }
    let repeated = vec![0x5a; 257];
    assert_eq!(decode(&encode(&repeated, false), false), Ok(repeated));
}

#[test]
fn spec_r29_success_round_trip_canonicalizes_symbols() {
    let cases = [
        ("c-s-q-p-y", false),
        ("o1", false),
        ("c-s-q-p-y-u", true),
        ("-0-", true),
    ];
    for (input, checked) in cases {
        let decoded = decode(input, checked).unwrap();
        let canonical = encode(&decoded, checked);
        assert!(!canonical.contains('-'));
        assert!(canonical.bytes().all(|b| !b.is_ascii_lowercase()));
        assert_eq!(decode(&canonical, checked), Ok(decoded));
    }
}

#[test]
fn spec_r30_single_significant_byte_role_depends_on_mode() {
    assert_eq!(decode("0", true), Ok(Vec::new()));
    assert_eq!(decode("0", false), Err(DecodeError::InvalidLength { offset: 1 }));
    assert_eq!(decode("-", true), Err(DecodeError::MissingCheck { offset: 1 }));
    assert_eq!(decode("-", false), Ok(Vec::new()));
}
