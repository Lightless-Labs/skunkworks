```rust
#[test]
fn spec_r1_wrap_returns_owned_lines() {
    assert_eq!(wrap("a b", 3), Ok(vec!["a b".to_string()]));
}

#[test]
fn spec_r2_error_variants_and_api_types() {
    assert_eq!(wrap("x", 0), Err(WrapError::WidthZero));
    assert_eq!(wrap("x", 257), Err(WrapError::WidthTooLarge { width: 257 }));
    assert_eq!(wrap("\u{7f}", 1), Err(WrapError::InvalidByte { index: 0, byte: 127 }));
}

#[test]
fn spec_r3_width_validation_boundaries() {
    assert_eq!(wrap("", 0), Err(WrapError::WidthZero));
    assert_eq!(wrap("x", 256), Ok(vec!["x".to_string()]));
    assert_eq!(wrap("", 257), Err(WrapError::WidthTooLarge { width: 257 }));
}

#[test]
fn spec_r4_first_invalid_byte_reports_original_byte_index() {
    assert_eq!(
        wrap("ok\n\u{1}later\u{7f}", 10),
        Err(WrapError::InvalidByte { index: 3, byte: 1 })
    );
}

#[test]
fn spec_r5_separators_collapse_and_empty_input_has_no_lines() {
    assert_eq!(wrap("", 4), Ok(vec![]));
    assert_eq!(wrap(" \t\r\n ", 4), Ok(vec![]));
    assert_eq!(
        wrap(" \talpha\r\n\tbeta  ", 20),
        Ok(vec!["alpha beta".to_string()])
    );
}

#[test]
fn spec_r6_word_characters_are_preserved_literally() {
    assert_eq!(
        wrap("a-b\\c!", 20),
        Ok(vec!["a-b\\c!".to_string()])
    );
}

#[test]
fn spec_r7_words_are_packed_in_order_without_indentation() {
    assert_eq!(
        wrap("aa bb cc", 5),
        Ok(vec!["aa bb".to_string(), "cc".to_string()])
    );
}

#[test]
fn spec_r8_word_that_fits_is_appended_with_one_space() {
    assert_eq!(
        wrap("a b", 3),
        Ok(vec!["a b".to_string()])
    );
}

#[test]
fn spec_r9_short_suffix_that_does_not_fit_is_not_split() {
    assert_eq!(
        wrap("abc de", 4),
        Ok(vec!["abc".to_string(), "de".to_string()])
    );
}

#[test]
fn spec_r10_long_word_splits_at_width_on_empty_line() {
    assert_eq!(
        wrap("abcdef", 3),
        Ok(vec!["abc".to_string(), "def".to_string()])
    );
}

#[test]
fn spec_r11_available_columns_reserve_one_for_separator() {
    assert_eq!(
        wrap("a abcdef", 6),
        Ok(vec!["a abcd".to_string(), "ef".to_string()])
    );
}

#[test]
fn spec_r12_long_suffix_uses_available_columns_when_at_least_two() {
    assert_eq!(
        wrap("ab abcdef", 7),
        Ok(vec!["ab abcd".to_string(), "ef".to_string()])
    );
}

#[test]
fn spec_r13_one_available_column_is_left_unused() {
    assert_eq!(
        wrap("abc abcd", 6),
        Ok(vec!["abc".to_string(), "abcd".to_string()])
    );
}

#[test]
fn spec_r14_exact_width_line_emits_without_extra_empty_line() {
    assert_eq!(
        wrap("abcdefgh", 4),
        Ok(vec!["abcd".to_string(), "efgh".to_string()])
    );
}

#[test]
fn spec_r15_short_split_tail_can_share_line_with_next_word() {
    assert_eq!(
        wrap("abcdef g", 4),
        Ok(vec!["abcd".to_string(), "ef g".to_string()])
    );
}

#[test]
fn spec_r16_output_lines_have_no_edge_spaces_and_preserve_nonseparators() {
    let lines = wrap(" \talpha-beta gamma\\delta\r\n", 7).unwrap();
    assert_eq!(lines, vec!["alpha-b".to_string(), "eta".to_string(), "gamma\\".to_string(), "delta".to_string()]);
    assert!(lines.iter().all(|line| !line.starts_with(' ') && !line.ends_with(' ') && !line.contains('\n') && line.len() <= 7));
    assert_eq!(
        lines.iter().flat_map(|line| line.bytes().filter(|byte| *byte != b' ')).collect::<Vec<_>>(),
        b"alpha-betagamma\\delta"
    );
}
```
