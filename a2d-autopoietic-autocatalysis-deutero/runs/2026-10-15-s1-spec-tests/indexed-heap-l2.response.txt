#[test]
fn spec_r2_public_types_and_error_variants() {
    let item = Item {
        key: "a".into(),
        priority: 1,
        value: "v".into(),
    };
    let cloned = item.clone();
    assert_eq!(item, cloned);
    let _: Error = Error::EmptyKey;
    let _: Error = Error::InvalidKey { at: 0 };
    let _: Error = Error::Duplicate { key: "a".into() };
    let _: Error = Error::Missing { key: "a".into() };
    let _: Error = Error::WrongDirection {
        current: 1,
        requested: 2,
    };
}

#[test]
fn spec_r3_trim_only_ascii_spaces_on_all_key_methods() {
    let mut q = IndexedHeap::new();
    q.push("  Alpha  ", 0, "v").unwrap();
    assert_eq!(q.get(" Alpha ").unwrap().unwrap().key, "alpha");
    assert_eq!(q.decrease_key(" Alpha ", -1), Ok(0));
    assert_eq!(q.increase_key("Alpha", 1), Ok(-1));
    assert_eq!(q.replace_value(" Alpha ", "w"), Ok("v".into()));
    assert_eq!(q.remove(" Alpha ").unwrap().key, "alpha");

    assert_eq!(q.get("\t").unwrap_err(), Error::InvalidKey { at: 0 });
    assert_eq!(q.get("\nA").unwrap_err(), Error::InvalidKey { at: 0 });
    assert_eq!(q.push("\u{00a0}", 0, ""), Err(Error::InvalidKey { at: 0 }));
    assert_eq!(q.push("   ", 0, ""), Err(Error::EmptyKey));
}

#[test]
fn spec_r28_replace_and_remove_validation_precedes_lookup_and_errors_are_atomic() {
    let mut q = IndexedHeap::new();
    assert_eq!(
        q.replace_value("9bad", "x"),
        Err(Error::InvalidKey { at: 0 })
    );
    assert_eq!(
        q.remove("9bad"),
        Err(Error::InvalidKey { at: 0 })
    );
    assert_eq!(
        q.replace_value("Absent", "x"),
        Err(Error::Missing {
            key: "absent".into()
        })
    );
    assert_eq!(
        q.remove("Absent"),
        Err(Error::Missing {
            key: "absent".into()
        })
    );
    assert_eq!(q.len(), 0);

    q.push("A", 0, "v").unwrap();
    assert_eq!(
        q.replace_value("Absent", "new"),
        Err(Error::Missing { key: "absent".into() })
    );
    assert_eq!(q.get("a").unwrap().unwrap().value, "v");
    assert_eq!(
        q.remove("Absent"),
        Err(Error::Missing { key: "absent".into() })
    );
    assert_eq!(q.len(), 1);
    assert_eq!(q.peek().unwrap().key, "a");
}
