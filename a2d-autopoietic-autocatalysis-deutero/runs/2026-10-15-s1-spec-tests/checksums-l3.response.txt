#[test]
fn spec_r41_fletcher_and_multi_handle_long_inputs_exactly() {
    let data = vec![255u8; 200_000];
    let n = data.len() as u64;
    let a16 = ((255 * n) % 255) as u16;
    let b16 = ((255 * n * (n + 1) / 2) % 255) as u16;
    assert_eq!(fletcher16(&data), (b16 << 8) | a16);

    let mut a32 = 0u64;
    let mut b32 = 0u64;
    for pair in data.chunks(2) {
        let word = if pair.len() == 2 {
            ((pair[0] as u64) << 8) | pair[1] as u64
        } else {
            (pair[0] as u64) << 8
        };
        a32 = (a32 + word) % 65535;
        b32 = (b32 + a32) % 65535;
    }
    let expected32 = ((b32 as u32) << 16) | a32 as u32;
    assert_eq!(fletcher32(&data), expected32);

    let multi = multi_checksums(&data, &[], Crc16::X25).unwrap();
    assert_eq!(multi.fletcher16, fletcher16(&[]));
    assert_eq!(multi.fletcher32, fletcher32(&[]));
}
