    let event_len = m.events().len();

    );
    assert_eq!(m.state(), state);
    assert_eq!(m.audit(), audit);
    assert_eq!(m.events().len(), event_len);
    assert_eq!(before.adjustment, 0);
    assert_eq!(events.len(), 0);
}
