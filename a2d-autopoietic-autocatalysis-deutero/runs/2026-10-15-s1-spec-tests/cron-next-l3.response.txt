            minute: 0
        }),
        Ok(false)
    );
    assert_eq!(
        cron.matches(CivilDateTime {
            year: 2000,
            month: 1,
            day: 31,
            hour: 9,
            minute: 1
        }),
        Ok(false)
    );
    assert_eq!(
        cron.matches(CivilDateTime {
            year: 2000,
            month: 2,
            day: 29,
            hour: 9,
            minute: 0
        }),
        Ok(false)
    );
    assert_eq!(
        cron.next_fire(CivilDateTime {
            year: 2000,
            month: 1,
            day: 30,
            hour: 8,
            minute: 59
        }),
        Ok(Some(CivilDateTime {
            year: 2000,
            month: 1,
            day: 30,
            hour: 9,
            minute: 0
        }))
    );
}
