#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = self.pending.clone();
        combined.extend_from_slice(input);
        let complete_len = (combined.len() / 5) * 5;
        let output = encode_bytes(&combined[..complete_len], self.config.alphabet);
        self.pending = combined[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = encode_bytes(&self.pending, self.config.alphabet);
        if !self.pending.is_empty() && self.config.padding == Padding::Required {
            while output.len() % 8 != 0 {
                output.push('=');
            }
        }
        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

fn encode_bytes(bytes: &[u8], alphabet: Alphabet) -> String {
    let table: &[u8; 32] = match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    };

    let mut output = String::new();
    let mut acc: u32 = 0;
    let mut bits = 0u32;

    for &byte in bytes {
        acc = (acc << 8) | byte as u32;
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            output.push(table[((acc >> bits) & 31) as usize] as char);
        }
    }

    if bits > 0 {
        output.push(table[((acc << (5 - bits)) & 31) as usize] as char);
    }
    output
}

pub struct Decoder {
    config: Config,
    bit_buffer: u32,
    bit_count: u8,
    data_count: usize,
    last_data_offset: Option<usize>,
    tail_value: u8,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: usize,
    position: usize,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            bit_buffer: 0,
            bit_count: 0,
            data_count: 0,
            last_data_offset: None,
            tail_value: 0,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: 0,
            position: 0,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            match classify(byte, next.config.alphabet) {
                Classified::Invalid => {
                    return Err(StreamError::InvalidByte { offset, byte });
                }
                Classified::Data(value, canonical) => {
                    if next.config.canonical && byte != canonical {
                        return Err(StreamError::NonCanonical {
                            offset,
                            expected: canonical,
                        });
                    }
                    if next.first_padding_offset.is_some() {
                        return Err(StreamError::Padding { offset });
                    }

                    next.data_count += 1;
                    next.last_data_offset = Some(offset);
                    next.tail_value = value;
                    next.bit_buffer = (next.bit_buffer << 5) | value as u32;
                    next.bit_count += 5;
                    while next.bit_count >= 8 {
                        next.bit_count -= 8;
                        output.push(((next.bit_buffer >> next.bit_count) & 0xff) as u8);
                    }
                    if next.bit_count == 0 {
                        next.bit_buffer = 0;
                    } else {
                        next.bit_buffer &= (1u32 << next.bit_count) - 1;
                    }
                }
                Classified::Padding => {
                    if next.config.padding == Padding::Omitted {
                        return Err(StreamError::Padding { offset });
                    }
                    if next.first_padding_offset.is_none() {
                        let residue = next.data_count % 8;
                        let expected = match residue {
                            2 => 6,
                            4 => 4,
                            5 => 3,
                            7 => 1,
                            _ => return Err(StreamError::Padding { offset }),
                        };
                        next.first_padding_offset = Some(offset);
                        next.expected_padding = expected;
                    }
                    if next.padding_count >= next.expected_padding {
                        return Err(StreamError::Padding { offset });
                    }
                    next.padding_count += 1;
                }
            }
            next.position += 1;
        }

        self.restore_state(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let expected_padding = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => {
                return Err(StreamError::InvalidLength {
                    offset: self.position,
                });
            }
        };

        if self.config.padding == Padding::Required && self.padding_count != expected_padding {
            return Err(StreamError::Padding {
                offset: self.first_padding_offset.unwrap_or(self.position),
            });
        }

        let unused_bits = match residue {
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused_bits > 0 && (self.tail_value & ((1u8 << unused_bits) - 1)) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> DecoderState {
        DecoderState {
            config: self.config,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            data_count: self.data_count,
            last_data_offset: self.last_data_offset,
            tail_value: self.tail_value,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            position: self.position,
            closed: self.closed,
        }
    }

    fn restore_state(&mut self, state: DecoderState) {
        self.config = state.config;
        self.bit_buffer = state.bit_buffer;
        self.bit_count = state.bit_count;
        self.data_count = state.data_count;
        self.last_data_offset = state.last_data_offset;
        self.tail_value = state.tail_value;
        self.first_padding_offset = state.first_padding_offset;
        self.padding_count = state.padding_count;
        self.expected_padding = state.expected_padding;
        self.position = state.position;
        self.closed = state.closed;
    }
}

struct DecoderState {
    config: Config,
    bit_buffer: u32,
    bit_count: u8,
    data_count: usize,
    last_data_offset: Option<usize>,
    tail_value: u8,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: usize,
    position: usize,
    closed: bool,
}

enum Classified {
    Data(u8, u8),
    Padding,
    Invalid,
}

fn classify(byte: u8, alphabet: Alphabet) -> Classified {
    if byte == b'=' {
        return Classified::Padding;
    }

    let upper = byte.to_ascii_uppercase();
    let value = match alphabet {
        Alphabet::Standard => match upper {
            b'A'..=b'Z' => Some(upper - b'A'),
            b'2'..=b'7' => Some(26 + upper - b'2'),
            _ => None,
        },
        Alphabet::Hex => match upper {
            b'0'..=b'9' => Some(upper - b'0'),
            b'A'..=b'V' => Some(10 + upper - b'A'),
            _ => None,
        },
    };

    match value {
        Some(value) => {
            let canonical = match alphabet {
                Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567"[value as usize],
                Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV"[value as usize],
            };
            Classified::Data(value, canonical)
        }
        None => Classified::Invalid,
    }
}
