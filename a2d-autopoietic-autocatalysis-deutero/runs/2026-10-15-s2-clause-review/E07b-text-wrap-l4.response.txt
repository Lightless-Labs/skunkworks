#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    piece_count: usize,
    consumed: usize,
    marker_ended: bool,
    raw: bool,
    displayed_width: usize,
}

#[derive(Clone)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
    consumed_counts: Vec<usize>,
    prejustified: Vec<String>,
}

impl Candidate {
    fn better_than(&self, other: &Self) -> bool {
        self.cost
            .cmp(&other.cost)
            .then_with(|| self.marker_lines.cmp(&other.marker_lines))
            .then_with(|| self.lines.len().cmp(&other.lines.len()))
            .then_with(|| other.consumed_counts.cmp(&self.consumed_counts))
            .then_with(|| self.prejustified.cmp(&other.prejustified))
            .is_lt()
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn parse_segment(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &c in segment {
        if is_separator(c) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }

    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }

    words
}

fn line_indent(options: &WrapOptions, first: bool) -> usize {
    if first {
        options.first_indent
    } else {
        options.rest_indent
    }
}

fn make_prejustified(line: &Line) -> String {
    let mut s = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            s.push(' ');
        }
        s.push_str(piece);
    }
    s
}

fn line_cost(line: &Line, final_line: bool, options: &WrapOptions) -> u64 {
    let gap = options.width - line.displayed_width;
    let square = (gap * gap) as u64;
    let mut cost = if final_line { square / 4 } else { square };
    if line.marker_ended {
        cost += options.hyphen_cost as u64;
    }
    if line.raw {
        cost += (2 * options.width + 1) as u64;
    }
    if !final_line && line.piece_count == 1 {
        cost += 3;
    }
    cost
}

fn optimize(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    let mut best: Vec<Option<Candidate>> = vec![None; words.len() + 1];
    best[words.len()] = Some(Candidate {
        cost: 0,
        marker_lines: 0,
        lines: Vec::new(),
        consumed_counts: Vec::new(),
        prejustified: Vec::new(),
    });

    for start in (0..words.len()).rev() {
        let mut chosen: Option<Candidate> = None;
        let first = start == 0;
        let indent = line_indent(options, first);
        let capacity = options.width - indent;

        for end in start..words.len() {
            let mut text = String::new();
            let mut pieces = Vec::new();
            let mut displayed = indent;
            let mut valid = true;

            for index in start..=end {
                if index > start {
                    displayed += 1;
                    text.push(' ');
                }
                let word: String = words[index].chars.iter().collect();
                let len = words[index].chars.len();
                if displayed + len > options.width {
                    valid = false;
                    break;
                }
                displayed += len;
                text.push_str(&word);
                pieces.push(word);
            }

            if !valid {
                break;
            }

            let tail = best[end + 1].as_ref().unwrap();
            let line = Line {
                indent,
                piece_count: pieces.len(),
                pieces,
                consumed: words[start..=end].iter().map(|w| w.chars.len()).sum(),
                marker_ended: false,
                raw: false,
                displayed_width: displayed,
            };
            let mut candidate = tail.clone();
            candidate.cost += line_cost(&line, candidate.lines.is_empty(), options);
            candidate.lines.insert(0, line.clone());
            candidate.consumed_counts.insert(0, line.consumed);
            candidate.prejustified.insert(0, make_prejustified(&line));
            if chosen.as_ref().map_or(true, |c| candidate.better_than(c)) {
                chosen = Some(candidate);
            }
        }

        let word = &words[start];
        let capacity = options.width - indent;
        if word.chars.len() > capacity {
            for &marker in &word.markers {
                if marker == 0 || marker >= word.chars.len() || marker > capacity.saturating_sub(1) {
                    continue;
                }

                let piece: String = word.chars[..marker].iter().collect::<String>() + "-";
                let line = Line {
                    indent,
                    pieces: vec![piece],
                    piece_count: 1,
                    consumed: marker,
                    marker_ended: true,
                    raw: false,
                    displayed_width: indent + marker + 1,
                };

                if let Some(mut tail) =
                    optimize_from_state(words, start, marker, options, false)
                {
                    tail.cost += line_cost(&line, false, options);
                    tail.marker_lines += 1;
                    tail.lines.insert(0, line.clone());
                    tail.consumed_counts.insert(0, marker);
                    tail.prejustified.insert(0, make_prejustified(&line));
                    if chosen.as_ref().map_or(true, |c| tail.better_than(c)) {
                        chosen = Some(tail);
                    }
                }
            }
        }

        best[start] = chosen;
    }

    best[0]
        .take()
        .map(|candidate| candidate.lines)
        .unwrap_or_default()
}

fn optimize_from_state(
    words: &[Word],
    word_index: usize,
    offset: usize,
    options: &WrapOptions,
    first: bool,
) -> Option<Candidate> {
    if word_index >= words.len() {
        return Some(Candidate {
            cost: 0,
            marker_lines: 0,
            lines: Vec::new(),
            consumed_counts: Vec::new(),
            prejustified: Vec::new(),
        });
    }

    let word = &words[word_index];
    if offset >= word.chars.len() {
        return optimize_from_state(words, word_index + 1, 0, options, first);
    }

    let indent = line_indent(options, first);
    let capacity = options.width - indent;
    let remaining = word.chars.len() - offset;
    let mut best: Option<Candidate> = None;

    let mut pieces = Vec::new();
    let mut displayed = indent;
    for end in word_index..words.len() {
        if end > word_index {
            displayed += 1;
        }
        let current_word = &words[end];
        let start_offset = if end == word_index { offset } else { 0 };
        let piece: String = current_word.chars[start_offset..].iter().collect();
        let piece_len = current_word.chars.len() - start_offset;
        if displayed + piece_len > options.width {
            break;
        }
        displayed += piece_len;
        pieces.push(piece);

        if let Some(mut tail) =
            optimize_from_state(words, end + 1, 0, options, false)
        {
            let line = Line {
                indent,
                piece_count: pieces.len(),
                pieces: pieces.clone(),
                consumed: pieces.iter().map(|piece| scalar_len(piece)).sum(),
                marker_ended: false,
                raw: false,
                displayed_width: displayed,
            };
            tail.cost += line_cost(&line, tail.lines.is_empty(), options);
            tail.lines.insert(0, line.clone());
            tail.consumed_counts.insert(0, line.consumed);
            tail.prejustified.insert(0, make_prejustified(&line));
            if best.as_ref().map_or(true, |candidate| tail.better_than(candidate)) {
                best = Some(tail);
            }
        }
    }

    for &marker in &word.markers {
        if marker <= offset || marker >= word.chars.len() {
            continue;
        }
        let consumed = marker - offset;
        if consumed + 1 > capacity {
            continue;
        }
        let piece = word.chars[offset..marker].iter().collect::<String>() + "-";
        let line = Line {
            indent,
            pieces: vec![piece],
            piece_count: 1,
            consumed,
            marker_ended: true,
            raw: false,
            displayed_width: indent + consumed + 1,
        };

        if let Some(mut tail) =
            optimize_from_state(words, word_index, marker, options, false)
        {
            tail.cost += line_cost(&line, tail.lines.is_empty(), options);
            tail.marker_lines += 1;
            tail.lines.insert(0, line.clone());
            tail.consumed_counts.insert(0, consumed);
            tail.prejustified.insert(0, make_prejustified(&line));
            if best.as_ref().map_or(true, |candidate| tail.better_than(candidate)) {
                best = Some(tail);
            }
        }
    }

    let marker_fits = word.markers.iter().any(|&marker| {
        marker > offset
            && marker < word.chars.len()
            && marker - offset + 1 <= capacity
    });
    if remaining > capacity && capacity > 0 && !marker_fits {
        let piece = word.chars[offset..offset + capacity]
            .iter()
            .collect::<String>();
        let line = Line {
            indent,
            pieces: vec![piece],
            piece_count: 1,
            consumed: capacity,
            marker_ended: false,
            raw: true,
            displayed_width: indent + capacity,
        };

        if let Some(mut tail) = optimize_from_state(
            words,
            word_index,
            offset + capacity,
            options,
            false,
        ) {
            tail.cost += line_cost(&line, tail.lines.is_empty(), options);
            tail.lines.insert(0, line.clone());
            tail.consumed_counts.insert(0, capacity);
            tail.prejustified.insert(0, make_prejustified(&line));
            if best.as_ref().map_or(true, |candidate| tail.better_than(candidate)) {
                best = Some(tail);
            }
        }
    }

    best
}

fn render(lines: &[Line], options: &WrapOptions) -> Vec<String> {
    let mut result = Vec::with_capacity(lines.len());

    for (line_index, line) in lines.iter().enumerate() {
        let mut rendered = " ".repeat(line.indent);
        let gap_count = line.piece_count.saturating_sub(1);

        if options.justify && line_index + 1 < lines.len() && gap_count > 0 {
            let extra = options.width - line.displayed_width;
            let base = extra / gap_count;
            let remainder = extra % gap_count;

            for (i, piece) in line.pieces.iter().enumerate() {
                if i > 0 {
                    let gap_index = i - 1;
                    let start = line_index % gap_count;
                    let bonus = usize::from((gap_index + gap_count - start) % gap_count < remainder);
                    rendered.push_str(&" ".repeat(1 + base + bonus));
                }
                rendered.push_str(piece);
            }
        } else {
            for (i, piece) in line.pieces.iter().enumerate() {
                if i > 0 {
                    rendered.push(' ');
                }
                rendered.push_str(piece);
            }
        }

        result.push(rendered);
    }

    result
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars = input.chars().count();
    if chars > 512 {
        return Err(WrapError::InputTooLong { chars });
    }

    for (index, character) in input.char_indices() {
        if character.is_control() && !matches!(character, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..original.len() {
        if original[i].1 != '~' {
            continue;
        }

        let valid_neighbor = |j: isize| -> bool {
            if j < 0 || j as usize >= original.len() {
                return false;
            }
            let character = original[j as usize].1;
            character != '~' && !is_separator(character)
        };

        if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
            return Err(WrapError::InvalidMarker {
                index: original[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut chars = input.chars().peekable();

    while let Some(character) = chars.next() {
        if character == '\r' {
            if chars.peek() == Some(&'\n') {
                chars.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(character);
        }
    }

    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut index = 0;

    while index < normalized.len() {
        if normalized[index] == '\n' {
            let mut end = index + 1;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }

            if end - index >= 2 {
                let words = parse_segment(&normalized[segment_start..index]);
                if !words.is_empty() {
                    let lines = optimize(&words, options);
                    output.extend(render(&lines, options));
                }

                output.extend(std::iter::repeat(String::new()).take(end - index - 1));
                segment_start = end;
                index = end;
                continue;
            }
        }

        index += 1;
    }

    let words = parse_segment(&normalized[segment_start..]);
    if !words.is_empty() {
        let lines = optimize(&words, options);
        output.extend(render(&lines, options));
    }

    Ok(output)
}
