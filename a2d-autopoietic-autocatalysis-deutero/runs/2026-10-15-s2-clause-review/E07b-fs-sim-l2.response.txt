use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
    cwd: Vec<String>,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
            cwd: Vec::new(),
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&byte| byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut components = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    components.pop();
                }
                name => components.push(name.to_owned()),
            }
        }

        Ok(components)
    }

    fn canonical(components: &[String]) -> String {
        if components.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", components.join("/"))
        }
    }

    fn lookup<'a>(&'a self, components: &[String]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        let mut prefix = Vec::new();

        for component in components {
            let Entry::Dir(children) = current else {
                return Err(Error::NotDir(Self::canonical(&prefix)));
            };

            prefix.push(component.clone());
            current = children
                .get(component)
                .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
        }

        Ok(current)
    }

    fn parent_mut<'a>(
        &'a mut self,
        components: &[String],
    ) -> Result<(&'a mut BTreeMap<String, Entry>, String), Error> {
        let (name, parents) = components.split_last().expect("non-root path");
        let mut current = &mut self.root;
        let mut prefix = Vec::new();

        for component in parents {
            let Entry::Dir(children) = current else {
                return Err(Error::NotDir(Self::canonical(&prefix)));
            };

            prefix.push(component.clone());
            current = children
                .get_mut(component)
                .ok_or_else(|| Error::Missing(Self::canonical(&prefix)))?;
        }

        match current {
            Entry::Dir(children) => Ok((children, name.clone())),
            Entry::File(_) => Err(Error::NotDir(Self::canonical(parents))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let (children, name) = self.parent_mut(&components)?;
        let canonical = Self::canonical(&components);
        if children.contains_key(&name) {
            return Err(Error::Exists(canonical));
        }

        children.insert(name, Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);
        if components.is_empty() {
            return Err(Error::IsDir(canonical));
        }

        let (children, name) = self.parent_mut(&components)?;
        match children.get_mut(&name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                children.insert(name, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::Dir(_) => Err(Error::IsDir(canonical)),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::File(_) => Err(Error::NotDir(canonical)),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }

    pub fn cwd(&self) -> String {
        Self::canonical(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let components = self.normalize(path)?;
        let canonical = Self::canonical(&components);
        match self.lookup(&components)? {
            Entry::Dir(_) => {
                self.cwd = components;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(canonical)),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
