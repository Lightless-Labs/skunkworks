#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Side {
    Left,
    Right,
}

#[derive(Clone)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<(Side, Change)>,
}

fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_script(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0, 0, 0);
    let mut changes = Vec::new();
    let mut run_start: Option<usize> = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            if let Some(start) = run_start.take() {
                changes.push(Change {
                    start,
                    end: start + deletes,
                    replacement: std::mem::take(&mut inserts),
                });
                deletes = 0;
            }
            i += 1;
            j += 1;
            base_pos += 1;
            continue;
        }

        if run_start.is_none() {
            run_start = Some(base_pos);
        }

        if i == n {
            inserts.push(new[j].clone());
            j += 1;
        } else if j == m {
            deletes += 1;
            i += 1;
            base_pos += 1;
        } else {
            let delete_len = lcs[i + 1][j];
            let insert_len = lcs[i][j + 1];
            let delete = if delete_len > insert_len {
                true
            } else if insert_len > delete_len {
                false
            } else {
                (i + j) % 2 == 0
            };
            if delete {
                deletes += 1;
                i += 1;
                base_pos += 1;
            } else {
                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    if let Some(start) = run_start {
        changes.push(Change {
            start,
            end: start + deletes,
            replacement: inserts,
        });
    }

    changes
}

fn interact(a: &Change, b: &Change) -> bool {
    let a_positive = a.start < a.end;
    let b_positive = b.start < b.end;

    match (a_positive, b_positive) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn projection(
    base: &[String],
    start: usize,
    end: usize,
    changes: &[Change],
) -> Vec<String> {
    let mut result = Vec::new();
    let mut cursor = start;

    for change in changes {
        if cursor < change.start {
            result.extend(base[cursor..change.start].iter().cloned());
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    if cursor < end {
        result.extend(base[cursor..end].iter().cloned());
    }
    result
}

fn append_lines(output: &mut Vec<String>, lines: &[String]) {
    output.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let left_changes = edit_script(&base_lines, &left_lines);
    let right_changes = edit_script(&base_lines, &right_lines);

    let all_changes: Vec<(Side, Change)> = left_changes
        .iter()
        .cloned()
        .map(|c| (Side::Left, c))
        .chain(
            right_changes
                .iter()
                .cloned()
                .map(|c| (Side::Right, c)),
        )
        .collect();

    let count = all_changes.len();
    let mut parent: Vec<usize> = (0..count).collect();

    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }

    for i in 0..count {
        for j in (i + 1)..count {
            if interact(&all_changes[i].1, &all_changes[j].1) {
                let ri = find(&mut parent, i);
                let rj = find(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut groups_by_root: Vec<(usize, Vec<(Side, Change)>)> = Vec::new();
    for i in 0..count {
        let root = find(&mut parent, i);
        if let Some((_, group)) = groups_by_root.iter_mut().find(|(group_root, _)| *group_root == root) {
            group.push(all_changes[i].clone());
        } else {
            groups_by_root.push((root, vec![all_changes[i].clone()]));
        }
    }

    let mut components = Vec::new();
    for (_, mut group) in groups_by_root {
        group.sort_by_key(|(side, change)| {
            (
                change.start,
                change.end,
                if *side == Side::Left { 0 } else { 1 },
            )
        });
        let start = group.iter().map(|(_, c)| c.start).min().unwrap();
        let end = group.iter().map(|(_, c)| c.end).max().unwrap();
        components.push(Component {
            start,
            end,
            changes: group,
        });
    }

    components.sort_by_key(|component| {
        (
            component.start,
            if component.start == component.end { 0 } else { 1 },
            component.end,
        )
    });

    let mut output = Vec::<String>::new();
    let mut output_regions = Vec::<Conflict>::new();
    let mut base_cursor = 0usize;
    let mut ordinal = 0usize;

    for component in components {
        if base_cursor < component.start {
            append_lines(&mut output, &base_lines[base_cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let mut left_component_changes: Vec<Change> = component
            .changes
            .iter()
            .filter(|(side, _)| *side == Side::Left)
            .map(|(_, change)| change.clone())
            .collect();
        let mut right_component_changes: Vec<Change> = component
            .changes
            .iter()
            .filter(|(side, _)| *side == Side::Right)
            .map(|(_, change)| change.clone())
            .collect();
        left_component_changes.sort_by_key(|c| (c.start, c.end));
        right_component_changes.sort_by_key(|c| (c.start, c.end));

        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &left_component_changes,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &right_component_changes,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    ordinal += 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    let base_label = if component.end > component.start {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {base_label},{}",
                        component.end - component.start
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    output_regions.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: component.end - component.start,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[base_cursor..]);
    }

    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: output_regions.len(),
        regions: output_regions,
    })
}
