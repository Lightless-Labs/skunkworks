    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

#[derive(Debug, Clone, Copy)]
struct Accumulator {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
    fletcher16_a: u32,
    fletcher16_b: u32,
    fletcher32_a: u32,
    fletcher32_b: u32,
    fletcher32_high: Option<u8>,
    crc16: u16,
}

impl Accumulator {
    fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
            fletcher16_a: 0,
            fletcher16_b: 0,
            fletcher32_a: 0,
            fletcher32_b: 0,
            fletcher32_high: None,
            crc16: 0,
        }
    }

    fn from_seed(crc32: u32, adler32: u32) -> Self {
        Self {
            crc: crc32 ^ 0xffff_ffff,
            adler_a: adler32 & 0xffff,
            adler_b: adler32 >> 16,
            ..Self::new()
        }
    }

    fn update_base_byte(&mut self, byte: u8) {
        self.crc ^= u32::from(byte);
        for _ in 0..8 {
            self.crc = (self.crc >> 1)
                ^ if self.crc & 1 != 0 { 0xedb8_8320 } else { 0 };
        }
        self.adler_a = (self.adler_a + u32::from(byte)) % 65_521;
        self.adler_b = (self.adler_b + self.adler_a) % 65_521;
    }

    fn update_fletcher16_byte(&mut self, byte: u8) {
        self.fletcher16_a = (self.fletcher16_a + u32::from(byte)) % 255;
        self.fletcher16_b = (self.fletcher16_b + self.fletcher16_a) % 255;
    }

    fn update_fletcher32_byte(&mut self, byte: u8) {
        if let Some(high) = self.fletcher32_high.take() {
            let word = (u32::from(high) << 8) | u32::from(byte);
            self.update_fletcher32_word(word);
        } else {
            self.fletcher32_high = Some(byte);
        }
    }

    fn update_fletcher32_word(&mut self, word: u32) {
        self.fletcher32_a = (self.fletcher32_a + word) % 65_535;
        self.fletcher32_b = (self.fletcher32_b + self.fletcher32_a) % 65_535;
    }

    fn update_crc16(&mut self, byte: u8, variant: Crc16) {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                self.crc16 ^= u16::from(byte) << 8;
                for _ in 0..8 {
                    let top = self.crc16 & 0x8000 != 0;
                    self.crc16 = self.crc16.wrapping_shl(1);
                    if top {
                        self.crc16 ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                self.crc16 ^= u16::from(byte);
                for _ in 0..8 {
                    let low = self.crc16 & 1 != 0;
                    self.crc16 >>= 1;
                    if low {
                        self.crc16 ^= 0x8408;
                    }
                }
            }
        }
    }

    fn update_all(&mut self, byte: u8, variant: Crc16) {
        self.update_base_byte(byte);
        self.update_fletcher16_byte(byte);
        self.update_fletcher32_byte(byte);
        self.update_crc16(byte, variant);
    }

    fn hashes(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    fn finish_fletcher32(&self) -> u32 {
        let mut a = self.fletcher32_a;
        let mut b = self.fletcher32_b;
        if let Some(high) = self.fletcher32_high {
            a = (a + (u32::from(high) << 8)) % 65_535;
            b = (b + a) % 65_535;
        }
        (b << 16) | a
    }
}

#[derive(Debug, Clone)]
pub struct Stream {
    acc: Accumulator,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            acc: Accumulator::new(),
        }
    }

    pub fn checksum(&self) -> Hashes {
        self.acc.hashes()
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            acc: Accumulator::from_seed(crc32, adler32),
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.acc.update_base_byte(byte);
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            for &byte in &data[span.start..span.end] {
                self.acc.update_base_byte(byte);
            }
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        self.acc = Accumulator::new();
    }

    pub fn snapshot(&self) -> String {
        let hashes = self.checksum();
        format!("c{:08x}:a{:08x}", hashes.crc32, hashes.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let crc = parse_snapshot_hex(bytes, 1, 9)?;
        let adler = parse_snapshot_hex(bytes, 11, 19)?;
        Self::from_seed(crc, adler)
    }
}

fn snapshot_hex_value(byte: u8, offset: usize) -> Result<u32, SnapshotError> {
    match byte {
        b'0'..=b'9' => Ok(u32::from(byte - b'0')),
        b'a'..=b'f' => Ok(u32::from(byte - b'a' + 10)),
        _ => Err(SnapshotError::Hex { offset }),
    }
}

fn parse_snapshot_hex(bytes: &[u8], start: usize, end: usize) -> Result<u32, SnapshotError> {
    let mut value = 0u32;
    for offset in start..end {
        value = (value << 4) | snapshot_hex_value(bytes[offset], offset)?;
    }
    Ok(value)
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + u32::from(byte)) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut acc = Accumulator::new();
    for &byte in data {
        acc.update_fletcher32_byte(byte);
    }
    acc.finish_fletcher32()
}

pub enum Crc16 {
    Xmodem,
    False,
    X25,

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let initial = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    let mut acc = Accumulator {
        crc16: initial,
        ..Accumulator::new()
    };
    for &byte in data {
        acc.update_crc16(byte, variant);
    }
    match variant {
        Crc16::X25 => acc.crc16 ^ 0xffff,
        Crc16::Xmodem | Crc16::False => acc.crc16,
    }
}

    pub fletcher32: u32,

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let initial_crc16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    let mut acc = Accumulator {
        crc16: initial_crc16,
        ..Accumulator::new()
    };
    for span in spans {
        for &byte in &data[span.start..span.end] {
            acc.update_all(byte, variant);
        }
    }
    Ok(Multi {
        base: acc.hashes(),
        fletcher16: ((acc.fletcher16_b << 8) | acc.fletcher16_a) as u16,
        fletcher32: acc.finish_fletcher32(),
        crc16: match variant {
            Crc16::X25 => acc.crc16 ^ 0xffff,
            Crc16::Xmodem | Crc16::False => acc.crc16,
        },
    })
}

fn crc32_zero_byte(mut register: u32) -> u32 {
    for _ in 0..8 {
        register = (register >> 1)
            ^ if register & 1 != 0 { 0xedb8_8320 } else { 0 };
    }
    register
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for index in 0..32 {
        square[index] = gf2_matrix_times(matrix, matrix[index]);
    }
}

fn crc32_advance_zeros(mut crc: u32, mut length: u64) -> u32 {
    if length == 0 {
        return crc;
    }

    let mut operator = [0u32; 32];
    for bit in 0..32 {
        operator[bit] = crc32_zero_byte(1u32 << bit);
    }

    while length != 0 {
        if length & 1 != 0 {
            crc = gf2_matrix_times(&operator, crc);
        }
        length >>= 1;
        if length != 0 {
            let mut squared = [0u32; 32];
            gf2_matrix_square(&mut squared, &operator);
            operator = squared;
        }
    }
    crc
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    crc32_advance_zeros(left, right_len) ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }

    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Accumulator::new();
    let mut len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        len += selected.len() as u64;
        for &byte in selected {
            acc.update_base_byte(byte);
        }
    }
    let hashes = acc.hashes();
    Ok(format!(
        "len={len:016x};crc={:08x};adler={:08x}",
        hashes.crc32, hashes.adler32
    ))
}

pub enum VerifyError {
    Span(SpanError),
    Shape,
    Hex { offset: usize },
    Length,
    Mismatch {
        field: usize,
        expected: u32,
        actual: u32,

fn parse_verify_hex_u64(bytes: &[u8], start: usize, end: usize) -> Result<u64, usize> {
    let mut value = 0u64;
    for offset in start..end {
        let digit = match bytes[offset] {
            b'0'..=b'9' => u64::from(bytes[offset] - b'0'),
            b'a'..=b'f' => u64::from(bytes[offset] - b'a' + 10),
            _ => return Err(offset),
        };
        value = (value << 4) | digit;
    }
    Ok(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let length = parse_verify_hex_u64(bytes, 4, 20).map_err(|offset| VerifyError::Hex { offset })?;
    let crc = parse_verify_hex_u64(bytes, 25, 33).map_err(|offset| VerifyError::Hex { offset })? as u32;
    let adler =
        parse_verify_hex_u64(bytes, 40, 48).map_err(|offset| VerifyError::Hex { offset })? as u32;

    let mut acc = Accumulator::new();
    let mut selected_len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        selected_len += selected.len() as u64;
        for &byte in selected {
            acc.update_base_byte(byte);
        }
    }
    if length != selected_len {
        return Err(VerifyError::Length);
    }

    let actual = acc.hashes();
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: crc,
            actual: actual.crc32,
        });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}
