use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_wide(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / g;
        let right_scale = self.den as i128 / g;
        let n = (self.num as i128)
            .checked_mul(left_scale)
            .and_then(|a| {
                (rhs.num as i128)
                    .checked_mul(right_scale)
                    .and_then(|b| a.checked_add(b))
            })
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(left_scale)
            .ok_or(RationalError::Overflow)?;
        Self::from_wide(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let left_scale = rhs.den as i128 / g;
        let right_scale = self.den as i128 / g;
        let n = (self.num as i128)
            .checked_mul(left_scale)
            .and_then(|a| {
                (rhs.num as i128)
                    .checked_mul(right_scale)
                    .and_then(|b| a.checked_sub(b))
            })
            .ok_or(RationalError::Overflow)?;
        let d = (self.den as i128)
            .checked_mul(left_scale)
            .ok_or(RationalError::Overflow)?;
        Self::from_wide(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.den as u128);
        let g2 = gcd_u128(rhs.num.unsigned_abs(), self.den as u128);
        let a = (self.num as i128) / g1 as i128;
        let b = (rhs.num as i128) / g2 as i128;
        let c = (self.den as i128) / g2 as i128;
        let d = (rhs.den as i128) / g1 as i128;
        let n = a.checked_mul(b).ok_or(RationalError::Overflow)?;
        let den = c.checked_mul(d).ok_or(RationalError::Overflow)?;
        Self::from_wide(n, den)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(self.num.unsigned_abs(), rhs.num.unsigned_abs());
        let g2 = gcd_u128(self.den as u128, rhs.den as u128);
        let a = (self.num as i128) / g1 as i128;
        let b = (rhs.num as i128) / g1 as i128;
        let c = (self.den as i128) / g2 as i128;
        let d = (rhs.den as i128) / g2 as i128;
        let mut n = a.checked_mul(d).ok_or(RationalError::Overflow)?;
        let mut den = c.checked_mul(b).ok_or(RationalError::Overflow)?;
        if den < 0 {
            n = n.checked_neg().ok_or(RationalError::Overflow)?;
            den = den.checked_neg().ok_or(RationalError::Overflow)?;
        }
        Self::from_wide(n, den)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = (self.num as i128)
            .checked_neg()
            .ok_or(RationalError::Overflow)?;
        Self::from_wide(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let negative = bytes[0] == b'-';
        let digit_start = usize::from(negative);
        if negative && bytes.len() == 1 {
            return Err(parse_error(bytes.len(), ParseKind::InvalidByte));
        }

        let mut i = digit_start;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == digit_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            den_start = Some(start);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let num_digits = num_end - digit_start;

        if num_digits >= 2 && bytes[digit_start] == b'0' {
            return Err(parse_error(digit_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[1] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }
        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let num_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let num_mag = parse_magnitude_bounded(&bytes[digit_start..num_end], num_limit);
        if num_mag > num_limit {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }

        let den_value = if let (Some(start), Some(end)) = (den_start, den_end) {
            let value = parse_magnitude_bounded(&bytes[start..end], i64::MAX as u128);
            if value > i64::MAX as u128 {
                return Err(parse_error(start, ParseKind::OutOfRange));
            }
            Some((value as i64, start))
        } else {
            None
        };

        if let Some((0, at)) = den_value {
            return Err(parse_error(at, ParseKind::ZeroDenominator));
        }

        if let Some((den, _)) = den_value {
            let slash = den_start.unwrap() - 1;
            if den == 1 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den as u128) > 1 {
                return Err(parse_error(slash, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            -(num_mag as i128)
        } else {
            num_mag as i128
        };
        let den = den_value.map_or(1i128, |(d, _)| d as i128);
        Self::from_wide(signed_num, den)
    }

    fn from_wide(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if den < 0 {
            num = num.checked_neg().ok_or(RationalError::Overflow)?;
            den = den.checked_neg().ok_or(RationalError::Overflow)?;
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(num.unsigned_abs(), den as u128);
        num /= g as i128;
        den /= g as i128;

        if num < i64::MIN as i128 || num > i64::MAX as i128 || den > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

fn parse_error(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude_bounded(bytes: &[u8], limit: u128) -> u128 {
    let mut value = 0u128;
    for &byte in bytes {
        let digit = (byte - b'0') as u128;
        value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
            Some(next) if next <= limit => next,
            _ => limit + 1,
        };
    }
    value
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength {
        expected: usize,
        actual: usize,
    },
    Index {
        row: usize,
        col: usize,
    },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare {
        rows: usize,
        cols: usize,
    },
    Singular {
        rank: usize,
    },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let index = row
            .checked_mul(self.cols)
            .and_then(|base| base.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data
            .get(index)
            .copied()
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                let index = col * self.cols + row;
                data.push(self.data[index]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        if output_len == 0 {
            return Ok(Self {
                rows: self.rows,
                cols: rhs.cols,
                data: Vec::new(),
            });
        }

        let mut data = Vec::with_capacity(output_len);
        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for step in 0..self.cols {
                    let left_index = row
                        .checked_mul(self.cols)
                        .and_then(|base| base.checked_add(step))
                        .expect("valid matrix index");
                    let right_index = step
                        .checked_mul(rhs.cols)
                        .and_then(|base| base.checked_add(col))
                        .expect("valid matrix index");
                    let left = self.data[left_index];
                    let right = rhs.data[right_index];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }
}
