#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    occupied: bool,
    idle_since: u64,
    finish_at: Option<u64>,
    finishing_customer: Option<u32>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|earlier| earlier.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|earlier| earlier.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut state: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            occupied: false,
            finish_at: None,
            finishing_customer: None,
        })
        .collect();

    let mut waiting: Vec<&Customer> = Vec::new();
    let mut arrival_index = 0usize;
    let mut events = Vec::new();
    let mut now = 0u64;

    loop {
        let mut finishing: Vec<usize> = state
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (s.finish_at == Some(now)).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| state[i].server.id);
        for i in finishing {
            let customer = state[i].finishing_customer.take().unwrap();
            events.push(Event::Finish {
                time: now,
                customer,
                server: state[i].server.id,
            });
            state[i].finish_at = None;
            state[i].occupied = false;
            state[i].idle_since = now;
        }

        let mut opening: Vec<usize> = state
            .iter()
            .enumerate()
            .filter_map(|(i, s)| (!s.opened && s.server.opens == now).then_some(i))
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(state[i].server.id));
        for i in opening {
            state[i].opened = true;
            events.push(Event::Open {
                time: now,
                server: state[i].server.id,
            });
        }

        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == now {
            let customer = arrivals[arrival_index];
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(customer);
            arrival_index += 1;
        }

        while !waiting.is_empty() {
            let selected = state
                .iter()
                .enumerate()
                .filter(|(_, s)| s.opened && !s.occupied)
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer = waiting[0];
            let server_id = state[server_index].server.id;
            let service_end =
                now.checked_add(customer.service)
                    .ok_or(Error::TimeOverflow {
                        customer: customer.id,
                        server: server_id,
                        time: now,
                    })?;
            let finish = service_end
                .checked_add(state[server_index].server.overhead)
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server_id,
                    time: now,
                })?;

            waiting.remove(0);
            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server_id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server_id,
                });
                state[server_index].idle_since = now;
            } else {
                state[server_index].occupied = true;
                state[server_index].finish_at = Some(finish);
                state[server_index].finishing_customer = Some(customer.id);
            }
        }

        let all_done = arrival_index == arrivals.len()
            && waiting.is_empty()
            && state.iter().all(|s| !s.occupied);
        if all_done && state.iter().all(|s| s.opened) {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        if arrival_index < arrivals.len() {
            consider(arrivals[arrival_index].arrival);
        }
        for server in &state {
            if !server.opened {
                consider(server.server.opens);
            }
            if let Some(finish) = server.finish_at {
                consider(finish);
            }
        }

        if let Some(time) = next_time {
            now = time;
        } else {
            break;
        }
    }

    Ok(events)
}
