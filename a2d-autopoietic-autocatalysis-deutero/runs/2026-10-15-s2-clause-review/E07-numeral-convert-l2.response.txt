#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if !(2..=36).contains(&value) {
        Err(BaseError::BadBase { which, value })
    } else {
        Ok(())
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, base: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(tag_end) = bytes.iter().position(|&b| b == b'#') {
        let expected = base.to_string();
        if &bytes[..tag_end] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        tag_end + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let mut digits = Vec::new();

    // Invalid digits take precedence over separator and leading-zero errors.
    for (index, &byte) in bytes.iter().enumerate().skip(payload_start) {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < base => digits.push(value),
            _ => return Err(BaseError::InvalidDigit { operand, index }),
        }
    }

    for index in payload_start..bytes.len() {
        if bytes[index] == b'_' {
            let has_digit_left = index > payload_start
                && bytes[index - 1] != b'_'
                && digit_value(bytes[index - 1]).is_some_and(|d| d < base);
            let has_digit_right = index + 1 < bytes.len()
                && bytes[index + 1] != b'_'
                && digit_value(bytes[index + 1]).is_some_and(|d| d < base);
            if !has_digit_left || !has_digit_right {
                return Err(BaseError::Separator { operand, index });
            }
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Little-endian base-256 unsigned integer, used to support values of arbitrary length.
fn multiply_add(value: &mut Vec<u8>, multiplier: u8, addend: u8) {
    let mut carry = addend as u16;
    for byte in value.iter_mut() {
        let n = (*byte as u16) * (multiplier as u16) + carry;
        *byte = n as u8;
        carry = n >> 8;
    }
    while carry != 0 {
        value.push(carry as u8);
        carry >>= 8;
    }
}

fn add_magnitudes(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut carry = 0u16;
    for i in 0..left.len().max(right.len()) {
        let a = left.get(i).copied().unwrap_or(0) as u16;
        let b = right.get(i).copied().unwrap_or(0) as u16;
        let sum = a + b + carry;
        result.push(sum as u8);
        carry = sum >> 8;
    }
    if carry != 0 {
        result.push(carry as u8);
    }
    result
}

fn to_digits(value: &[u8], base: u8) -> Vec<u8> {
    let mut number = value.to_vec();
    while number.last() == Some(&0) {
        number.pop();
    }
    if number.is_empty() {
        return vec![0];
    }

    let mut output = Vec::new();
    while !number.is_empty() {
        let mut remainder = 0u16;
        for byte in number.iter_mut().rev() {
            let current = (remainder << 8) | *byte as u16;
            *byte = (current / base as u16) as u8;
            remainder = current % base as u16;
        }
        output.push(remainder as u8);
        while number.last() == Some(&0) {
            number.pop();
        }
    }
    output.reverse();
    output
}

fn format_digits(digits: &[u8], style: BaseStyle) -> String {
    let padding = style.min_width.saturating_sub(digits.len());
    let total = padding + digits.len();
    let separator_count = if style.group == 0 {
        0
    } else {
        (total - 1) / style.group
    };
    let mut output = String::with_capacity(total + separator_count);

    for i in 0..total {
        if style.group != 0 && i != 0 && (total - i) % style.group == 0 {
            output.push('_');
        }
        let digit = if i < padding { 0 } else { digits[i - padding] };
        let ch = match digit {
            0..=9 => (b'0' + digit) as char,
            _ if style.upper => (b'A' + digit - 10) as char,
            _ => (b'a' + digit - 10) as char,
        };
        output.push(ch);
    }

    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let digits = parse_operand(input, from, 0)?;
    let mut value = Vec::new();
    for digit in digits {
        multiply_add(&mut value, from, digit);
    }
    let result_digits = to_digits(&value, to);
    Ok(format_digits(&result_digits, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let left_digits = parse_operand(left, base, 0)?;
    let right_digits = parse_operand(right, base, 1)?;

    let mut left_value = Vec::new();
    for digit in left_digits {
        multiply_add(&mut left_value, base, digit);
    }
    let mut right_value = Vec::new();
    for digit in right_digits {
        multiply_add(&mut right_value, base, digit);
    }

    let sum = add_magnitudes(&left_value, &right_value);
    let result_digits = to_digits(&sum, base);
    Ok(format_digits(&result_digits, style))
}
