    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },

fn validate_config(delimiter: u8, quote: u8) -> Result<(), CsvError> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Delimiter,
        });
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    if delimiter == quote {
        return Err(CsvError::InvalidConfig {
            component: ConfigPart::Quote,
        });
    }
    Ok(())
}

fn parse_dialect(input: &str, delimiter: u8, quote: u8) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut record = Vec::new();
    let mut field = String::new();
    let mut state = 0u8; // 0: field start, 1: unquoted, 2: quoted, 3: after quote
    let mut quote_start = 0usize;
    let mut record_index = 0usize;
    let mut ended_with_terminator = false;

    while i < bytes.len() {
        let b = bytes[i];
        ended_with_terminator = false;

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if state != 2 && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        match state {
            0 => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    i += 1;
                } else if b == quote {
                    quote_start = i;
                    state = 2;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    i += if b == b'\r' { 2 } else { 1 };
                    ended_with_terminator = true;
                } else {
                    let start = i;
                    while i < bytes.len()
                        && bytes[i] != delimiter
                        && bytes[i] != quote
                        && bytes[i] != b'\n'
                        && bytes[i] != b'\r'
                        && bytes[i] != 0
                    {
                        i += 1;
                    }
                    field.push_str(&input[start..i]);
                    state = 1;
                }
            }
            1 => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    i += 1;
                } else if b == quote {
                    return Err(CsvError::UnexpectedQuote {
                        record: record_index,
                        byte: i,
                    });
                } else if b == b'\n' || b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = 0;
                    i += if b == b'\r' { 2 } else { 1 };
                    ended_with_terminator = true;
                } else {
                    let start = i;
                    while i < bytes.len()
                        && bytes[i] != delimiter
                        && bytes[i] != quote
                        && bytes[i] != b'\n'
                        && bytes[i] != b'\r'
                        && bytes[i] != 0
                    {
                        i += 1;
                    }
                    field.push_str(&input[start..i]);
                }
            }
            2 => {
                if b == quote {
                    if bytes.get(i + 1) == Some(&quote) {
                        field.push(quote as char);
                        i += 2;
                    } else {
                        state = 3;
                        i += 1;
                    }
                } else {
                    let start = i;
                    while i < bytes.len() && bytes[i] != quote && bytes[i] != 0 {
                        i += 1;
                    }
                    field.push_str(&input[start..i]);
                }
            }
            _ => {
                if b == delimiter {
                    record.push(std::mem::take(&mut field));
                    state = 0;
                    i += 1;
                } else if b == b'\n' || b == b'\r' {
                    record.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut record));
                    record_index += 1;
                    state = 0;
                    i += if b == b'\r' { 2 } else { 1 };
                    ended_with_terminator = true;
                } else {
                    return Err(CsvError::TrailingByte {
                        record: record_index,
                        byte: i,
                    });
                }
            }
        }
    }

    if state == 2 {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_start,
        });
    }

    if !ended_with_terminator {
        record.push(field);
        records.push(record);
    }
    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    let mut seen: HashMap<&str, usize> = HashMap::new();
    for (second, name) in header.iter().enumerate() {
        if let Some(&first) = seen.get(name.as_str()) {
            return Some(CsvError::DuplicateHeader {
                name: name.clone(),
                first,
                second,
            });
        }
        seen.insert(name.as_str(), second);
    }
    None
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let mut iter = records.into_iter();
    let header = if config.headers { iter.next() } else { None };
    let rows: Vec<Vec<String>> = iter.collect();

    if let Some(ref header_row) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(err) = duplicate_header(header_row) {
                return Err(err);
            }
        }
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != header_row.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: header_row.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    finished: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_config(config.delimiter, config.quote)?;
        Ok(Self {
            config,
            input: String::new(),
            finished: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.finished.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.finished {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.finished = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.finished = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let digits = if b[0] == b'-' { &b[1..] } else { b };
    if digits.is_empty() {
        return false;
    }
    if digits == b"0" {
        return b[0] != b'-';
    }
    digits[0].is_ascii_digit()
        && digits[0] != b'0'
        && digits.iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let body = if b.first() == Some(&b'-') { &b[1..] } else { b };
    let mut parts = body.split(|&x| x == b'.');
    let whole = parts.next().unwrap_or_default();
    let frac = parts.next().unwrap_or_default();
    if parts.next().is_some() || frac.len() != 2 || !frac.iter().all(u8::is_ascii_digit) {
        return false;
    }
    let whole_ok = whole == b"0"
        || (!whole.is_empty()
            && whole[0].is_ascii_digit()
            && whole[0] != b'0'
            && whole.iter().all(u8::is_ascii_digit));
    if !whole_ok {
        return false;
    }
    !(b.first() == Some(&b'-') && whole == b"0" && frac == b"00")
}

fn evaluate_cell(
    text: Option<&String>,
    column: Option<usize>,
    spec: &ColumnSpec,
    row: usize,
    spec_index: usize,
) -> Result<Value, CellError> {
    let make_error = |kind| CellError {
        row,
        spec: spec_index,
        column,
        kind,
    };

    let Some(text) = text else {
        return Err(make_error(CellErrorKind::MissingCell));
    };

    if text.is_empty() {
        if spec.nullable {
            return Ok(Value::Null);
        }
        if spec.kind != ColumnType::Text {
            return Err(make_error(CellErrorKind::Empty));
        }
    }

    match spec.kind {
        ColumnType::Text => Ok(Value::Text(text.clone())),
        ColumnType::Integer => {
            if !integer_syntax(text) {
                return Err(make_error(CellErrorKind::InvalidInteger));
            }
            match text.parse::<i64>() {
                Ok(v) => Ok(Value::Integer(v)),
                Err(_) => Err(make_error(CellErrorKind::IntegerOverflow)),
            }
        }
        ColumnType::Decimal => {
            if !decimal_syntax(text) {
                return Err(make_error(CellErrorKind::InvalidDecimal));
            }
            let negative = text.starts_with('-');
            let body = if negative { &text[1..] } else { text.as_str() };
            let mut parts = body.split('.');
            let whole: i128 = parts.next().unwrap().parse().unwrap_or(i128::MAX);
            let frac: i128 = parts.next().unwrap().parse().unwrap_or(0);
            let magnitude = whole.saturating_mul(100).saturating_add(frac);
            let signed = if negative { -magnitude } else { magnitude };
            if signed < i64::MIN as i128 || signed > i64::MAX as i128 {
                Err(make_error(CellErrorKind::DecimalOverflow))
            } else {
                Ok(Value::Decimal(signed as i64))
            }
        }
        ColumnType::Boolean => match text.as_str() {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(make_error(CellErrorKind::InvalidBoolean)),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::with_capacity(table.rows.len()),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut out = Vec::with_capacity(columns.len());
        let mut row_failed = false;

        for (spec_index, spec) in columns.iter().enumerate() {
            if row_failed {
                out.push(None);
                continue;
            }

            let resolved = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };

            let evaluated = match resolved {
                None => Err(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                }),
                Some(index) => evaluate_cell(
                    row.get(index),
                    Some(index),
                    spec,
                    row_index,
                    spec_index,
                ),
            };

            match evaluated {
                Ok(value) => out.push(Some(value)),
                Err(error) => {
                    out.push(None);
                    result.errors.push(error);
                    match policy {
                        ErrorPolicy::Collect => {}
                        ErrorPolicy::FirstPerRow | ErrorPolicy::Stop => row_failed = true,
                    }
                }
            }
            if policy == ErrorPolicy::Stop && row_failed {
                break;
            }
        }

        if row_failed && policy == ErrorPolicy::Stop {
            result.rows.push(out);
            break;
        }
        result.rows.push(out);
    }

    result
}

fn write_nul_preflight(table: &Table) -> Result<(), WriteError> {
    let mut record = 0usize;
    if let Some(header) = &table.header {
        for (column, field) in header.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
        record = 1;
    }
    for row in &table.rows {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
        record += 1;
    }
    Ok(())
}

fn compare_rows(a: &[String], b: &[String]) -> std::cmp::Ordering {
    for (x, y) in a.iter().zip(b.iter()) {
        match x.as_bytes().cmp(y.as_bytes()) {
            std::cmp::Ordering::Equal => {}
            ord => return ord,
        }
    }
    a.len().cmp(&b.len())
}

fn encode_field(
    field: &str,
    delimiter: u8,
    quote: u8,
    first_field_first_record: bool,
    sole: bool,
) -> String {
    let bytes = field.as_bytes();
    let needs_quote = bytes.contains(&delimiter)
        || bytes.contains(&quote)
        || bytes.contains(&b'\r')
        || bytes.contains(&b'\n')
        || bytes.first().is_some_and(|&b| b == b' ' || b == b'\t')
        || bytes.last().is_some_and(|&b| b == b' ' || b == b'\t')
        || (first_field_first_record && field.starts_with('\u{feff}'))
        || (sole && field.is_empty());

    if !needs_quote {
        return field.to_string();
    }
    let mut out = String::with_capacity(field.len() + 2);
    out.push(quote as char);
    for ch in field.chars() {
        if ch as u32 == quote as u32 {
            out.push(quote as char);
        }
        out.push(ch);
    }
    out.push(quote as char);
    out
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_config(config.delimiter, config.quote).map_err(convert_config_error)?;
    write_nul_preflight(table)?;

    if let Some(header) = &table.header {
        for (idx, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: idx + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    if let Some(header) = &table.header {
        if header.is_empty() {
            return Err(WriteError::ZeroColumns { record: 0 });
        }
    }
    let base_record = usize::from(table.header.is_some());
    for (idx, row) in table.rows.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns {
                record: base_record + idx,
            });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let (Some(header), DuplicatePolicy::Reject) = (&table.header, table.duplicates) {
            if let Some(error) = duplicate_header_write(header) {
                return Err(error);
            }
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(header) = &table.header {
                records.push(header.clone());
            }
            records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut retained: HashMap<&str, usize> = HashMap::new();
                for (index, name) in header.iter().enumerate() {
                    match table.duplicates {
                        DuplicatePolicy::KeepLast => {
                            retained.insert(name.as_str(), index);
                        }
                        DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                            retained.entry(name.as_str()).or_insert(index);
                        }
                    }
                }
                let mut columns: Vec<(String, usize)> = retained
                    .into_iter()
                    .map(|(name, index)| (name.to_string(), index))
                    .collect();
                columns.sort_by(|a, b| a.0.as_bytes().cmp(b.0.as_bytes()));

                let output_header: Vec<String> =
                    columns.iter().map(|(name, _)| name.clone()).collect();
                records.push(output_header);
                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| columns.iter().map(|(_, i)| row[*i].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| compare_rows(a, b));
                records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| compare_rows(a, b));
                records = rows;
            }
        }
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let ending = if canonical {
        "\n"
    } else {
        match config.line_ending {
            LineEnding::Lf => "\n",
            LineEnding::CrLf => "\r\n",
        }
    };

    let mut output = String::new();
    for (record_index, row) in records.iter().enumerate() {
        if record_index > 0 {
            output.push_str(ending);
        }
        for (column, field) in row.iter().enumerate() {
            if column > 0 {
                output.push(config.delimiter as char);
            }
            output.push_str(&encode_field(
                field,
                config.delimiter,
                config.quote,
                record_index == 0 && column == 0,
                row.len() == 1,
            ));
        }
    }
    if canonical || config.final_terminator {
        output.push_str(ending);
    }
    Ok(output)
}
