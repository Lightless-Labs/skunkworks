#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_parts(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        from_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        from_parts(a * d + c * b, b * d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        from_parts(a * d - c * b, b * d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_parts(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        from_parts(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        from_parts(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        use ParseKind::*;

        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: Empty });
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            let at = if i == bytes.len() { bytes.len() } else { i };
            return Err(RationalError::Parse {
                at,
                kind: InvalidByte,
            });
        }

        let slash = if i < bytes.len() && bytes[i] == b'/' {
            let s = i;
            i += 1;
            Some(s)
        } else {
            None
        };

        let den_start = if slash.is_some() { i } else { bytes.len() };
        if slash.is_some() {
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start {
                return Err(RationalError::Parse {
                    at: i,
                    kind: InvalidByte,
                });
            }
        }

        if i < bytes.len() {
            return Err(RationalError::Parse {
                at: i,
                kind: InvalidByte,
            });
        }

        let num_digits_start = usize::from(bytes[0] == b'-');
        let num_len = slash.unwrap_or(bytes.len()) - num_digits_start;
        if num_len >= 2 && bytes[num_digits_start] == b'0' {
            return Err(RationalError::Parse {
                at: num_digits_start + 1,
                kind: NonCanonical,
            });
        }
        if bytes[0] == b'-' && num_len == 1 && bytes[1] == b'0' {
            return Err(RationalError::Parse {
                at: 0,
                kind: NonCanonical,
            });
        }
        if slash.is_some() && bytes[den_start..].len() >= 2 && bytes[den_start] == b'0' {
            return Err(RationalError::Parse {
                at: den_start + 1,
                kind: NonCanonical,
            });
        }

        let num_mag_limit: u128 = if bytes[0] == b'-' {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut num_mag = 0u128;
        for &b in &bytes[num_digits_start..slash.unwrap_or(bytes.len())] {
            num_mag = num_mag
                .saturating_mul(10)
                .saturating_add((b - b'0') as u128);
            if num_mag > num_mag_limit {
                return Err(RationalError::Parse {
                    at: 0,
                    kind: OutOfRange,
                });
            }
        }

        let mut den_mag = 1u128;
        if slash.is_some() {
            den_mag = 0;
            for &b in &bytes[den_start..] {
                den_mag = den_mag
                    .saturating_mul(10)
                    .saturating_add((b - b'0') as u128);
                if den_mag > i64::MAX as u128 {
                    return Err(RationalError::Parse {
                        at: den_start,
                        kind: OutOfRange,
                    });
                }
            }
            if den_mag == 0 {
                return Err(RationalError::Parse {
                    at: den_start,
                    kind: ZeroDenominator,
                });
            }
            if den_mag == 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: NonCanonical,
                });
            }
            if num_mag == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: NonCanonical,
                });
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(RationalError::Parse {
                    at: slash.unwrap(),
                    kind: NonCanonical,
                });
            }
        }

        let signed_num = if bytes[0] == b'-' {
            -(num_mag as i128)
        } else {
            num_mag as i128
        };
        from_parts(signed_num, den_mag as i128)
    }
}

impl std::fmt::Display for Rational {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
