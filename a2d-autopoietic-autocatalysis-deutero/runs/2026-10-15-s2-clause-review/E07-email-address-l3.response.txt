#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone, Copy)]
struct Byte {
    value: u8,
    pos: usize,
}

#[derive(Clone, Copy)]
struct Run {
    start: usize,
    end: usize,
}

fn is_horizontal(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn fold_end(input: &[u8], i: usize) -> Option<usize> {
    if input.get(i) != Some(&b'\r') || input.get(i + 1) != Some(&b'\n') {
        return None;
    }

    let mut j = i + 2;
    if !input.get(j).copied().is_some_and(is_horizontal) {
        return None;
    }
    while input.get(j).copied().is_some_and(is_horizontal) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<(Vec<Byte>, Vec<Run>), EmailError> {
    let mut kept = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut run_start: Option<usize> = None;
    let mut i = 0usize;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut quoted_prefix = false;

    while i < input.len() {
        let b = input[i];

        if quoted_prefix {
            kept.push(Byte { value: b, pos: i });
            if b == b'\\' {
                if i + 1 < input.len() {
                    kept.push(Byte {
                        value: input[i + 1],
                        pos: i + 1,
                    });
                    i += 2;
                } else {
                    i += 1;
                }
                continue;
            }
            if b == b'"' {
                quoted_prefix = false;
            }
            i += 1;
            continue;
        }

        if depth > 0 {
            if b == b'\\' {
                if i + 1 >= input.len() || !printable(input[i + 1]) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
                continue;
            }
            if b == b'\r' || b == b'\n' {
                if let Some(end) = fold_end(input, i) {
                    i = end;
                    continue;
                }
                return Err(EmailError::BadFold { at: i });
            }
            if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
                continue;
            }
            if b == b')' {
                depth -= 1;
                i += 1;
                continue;
            }
            if is_horizontal(b) || printable(b) {
                i += 1;
                continue;
            }
            return Err(EmailError::CommentChar { at: i });
        }

        if b == b'"' && kept.is_empty() {
            if let Some(start) = run_start.take() {
                runs.push(Run { start, end: i });
            }
            quoted_prefix = true;
            kept.push(Byte { value: b, pos: i });
            i += 1;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'(' {
            if run_start.is_none() {
                run_start = Some(i);
            }
            if b == b'(' {
                depth = 1;
                outer_open = i;
            }
            i += 1;
            continue;
        }

        if b == b'\r' || b == b'\n' {
            if let Some(end) = fold_end(input, i) {
                if run_start.is_none() {
                    run_start = Some(i);
                }
                i = end;
                continue;
            }
            return Err(EmailError::BadFold { at: i });
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if let Some(start) = run_start.take() {
            runs.push(Run { start, end: i });
        }
        kept.push(Byte { value: b, pos: i });
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = run_start {
        runs.push(Run {
            start,
            end: input.len(),
        });
    }

    Ok((kept, runs))
}

fn pos_at(bytes: &[Byte], idx: usize, fallback: usize) -> usize {
    bytes.get(idx).map_or(fallback, |b| b.pos)
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, runs) = cleanup(raw)?;

    if clean.is_empty() {
        if let Some(run) = runs.first() {
            let _ = run;
        }
        return Err(EmailError::EmptyInput);
    }

    let mut close_quote = None;
    if clean.first().is_some_and(|b| b.value == b'"') {
        let mut i = 1usize;
        while i < clean.len() {
            if clean[i].value == b'\\' {
                i = (i + 2).min(clean.len());
            } else if clean[i].value == b'"' {
                close_quote = Some(i);
                break;
            } else {
                i += 1;
            }
        }
    }

    for run in &runs {
        let left = clean.iter().rev().find(|b| b.pos < run.start);
        let right = clean.iter().find(|b| b.pos >= run.end);
        if left.is_none() || right.is_none() {
            continue;
        }
        if left.is_some_and(|b| b.value == b'@') || right.is_some_and(|b| b.value == b'@') {
            continue;
        }
        return Err(EmailError::CfwsBoundary { at: run.start });
    }

    let mut ats = Vec::new();
    for (i, b) in clean.iter().enumerate() {
        if b.value == b'@' && !close_quote.is_some_and(|q| i <= q) {
            ats.push(i);
        }
    }

    if clean.first().is_some_and(|b| b.value == b'"') && close_quote.is_none() {
        return Err(EmailError::UnclosedQuote { at: clean[0].pos });
    }

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: clean[ats[1]].pos,
        });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: clean[sep].pos,
        });
    }

    let mut local = String::new();
    let quoted;

    if clean[0].value == b'"' {
        let q = close_quote.expect("closing quote was checked");
        if q + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: pos_at(&clean, q + 1, input.len()),
            });
        }

        quoted = true;
        let mut origins = Vec::new();
        let mut i = 1usize;
        while i < q {
            let b = clean[i].value;
            if b == b'\\' {
                if i + 1 >= q {
                    return Err(EmailError::BadEscape { at: clean[i].pos });
                }
                let next = clean[i + 1].value;
                if next != b'"' && next != b'\\' && next != b'@' {
                    return Err(EmailError::BadEscape { at: clean[i].pos });
                }
                local.push(next as char);
                origins.push(clean[i].pos);
                i += 2;
            } else {
                if !printable(b) {
                    return Err(EmailError::QuotedChar { at: clean[i].pos });
                }
                local.push(b as char);
                origins.push(clean[i].pos);
                i += 1;
            }
        }

        if local.is_empty() {
            return Err(EmailError::EmptyLocal { at: clean[q].pos });
        }
        if local.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
    } else {
        quoted = false;
        if sep > 64 {
            return Err(EmailError::LocalTooLong {
                at: clean[64].pos,
            });
        }

        let allowed_specials = b"!#$%&'*+-/=?^_`{|}~";
        for i in 0..sep {
            let b = clean[i].value;
            if b == b'.' {
                if i == 0 || i + 1 == sep || clean[i - 1].value == b'.' {
                    return Err(EmailError::LocalDot { at: clean[i].pos });
                }
            } else if !b.is_ascii_alphanumeric() && !allowed_specials.contains(&b) {
                return Err(EmailError::LocalChar { at: clean[i].pos });
            }
            local.push(b as char);
        }
    }

    let domain_start = sep + 1;
    let root_dot = domain_start < clean.len() && clean.last().is_some_and(|b| b.value == b'.');
    let domain_end = if root_dot { clean.len() - 1 } else { clean.len() };
    let domain_start_pos = pos_at(&clean, domain_start, input.len());

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: domain_start_pos,
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: clean[domain_start + 253].pos,
        });
    }

    let mut labels = Vec::new();
    let mut label_start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || clean[i].value == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }

    for &(start, end) in &labels {
        if start == end {
            let at = if start == domain_start {
                pos_at(&clean, start, input.len())
            } else if start == domain_end && root_dot {
                clean[clean.len() - 1].pos
            } else {
                pos_at(&clean, start, input.len())
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: clean[start + 63].pos,
            });
        }
        if clean[start].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: clean[start].pos,
            });
        }
        if clean[end - 1].value == b'-' {
            return Err(EmailError::LabelHyphen {
                at: clean[end - 1].pos,
            });
        }
        for b in &clean[start..end] {
            if !b.value.is_ascii_alphanumeric() && b.value != b'-' {
                return Err(EmailError::DomainChar { at: b.pos });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: domain_start_pos,
        });
    }

    let (last_start, last_end) = labels[labels.len() - 1];
    if clean[last_start..last_end]
        .iter()
        .all(|b| b.value.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: clean[last_start].pos,
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| (b.value as char).to_ascii_lowercase())
        .collect();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
