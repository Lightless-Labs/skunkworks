```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn validate(path: &str) -> Result<(), Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }
        Ok(())
    }

    fn components(path: &str) -> Vec<String> {
        let mut parts = Vec::new();
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part.to_string()),
            }
        }
        parts
    }

    fn from_components(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(path: &str, cwd: &str) -> String {
        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            Self::components(cwd)
        };
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part.to_string()),
            }
        }
        Self::from_components(&parts)
    }

    fn parent(path: &str) -> String {
        if path == "/" {
            return "/".to_string();
        }
        match path.rfind('/') {
            Some(0) => "/".to_string(),
            Some(i) => path[..i].to_string(),
            _ => "/".to_string(),
        }
    }

    fn basename(path: &str) -> String {
        if path == "/" {
            String::new()
        } else {
            path.rsplit('/').next().unwrap_or("").to_string()
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            return path.starts_with('/');
        }
        path == dir
            || path
                .strip_prefix(dir)
                .map(|rest| rest.starts_with('/'))
                .unwrap_or(false)
    }

    fn is_dir(&self, path: &str) -> bool {
        matches!(self.entries.get(path), Some(Entry::Dir))
    }

    fn expand_link(link_path: &str, target: &str, suffix: &[String]) -> String {
        let mut parts = if target.starts_with('/') {
            Vec::new()
        } else {
            Self::components(&Self::parent(link_path))
        };
        for part in target.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                _ => parts.push(part.to_string()),
            }
        }
        parts.extend(suffix.iter().cloned());
        let mut normalized = Vec::new();
        for part in parts {
            if part == ".." {
                normalized.pop();
            } else {
                normalized.push(part);
            }
        }
        Self::from_components(&normalized)
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        create_missing_final: bool,
    ) -> Result<String, Error> {
        let mut pending = Self::components(path);
        let mut resolved: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;

        loop {
            if pending.is_empty() {
                return Ok(Self::from_components(&resolved));
            }

            let component = pending.remove(0);
            resolved.push(component);
            let current = Self::from_components(&resolved);
            let is_final = pending.is_empty();

            match self.entries.get(&current) {
                None => {
                    if is_final && create_missing_final {
                        return Ok(current);
                    }
                    return Err(Error::Missing(current));
                }
                Some(Entry::Dir) => {
                    if is_final {
                        return Ok(current);
                    }
                }
                Some(Entry::File(_)) => {
                    if is_final {
                        return Ok(current);
                    }
                    return Err(Error::NotDir(current));
                }
                Some(Entry::Link(target)) if !is_final || follow_final => {
                    if !visited.insert(current.clone()) {
                        return Err(Error::Loop(current));
                    }
                    if expansions >= 8 {
                        return Err(Error::Depth(current));
                    }
                    expansions += 1;
                    let expanded = Self::expand_link(&current, target, &pending);
                    pending = Self::components(&expanded);
                    resolved.clear();
                }
                Some(Entry::Link(_)) => {
                    if is_final {
                        return Ok(current);
                    }
                }
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let parent = Self::parent(path);
        let name = Self::basename(path);
        let resolved_parent = self.resolve(&parent, true, false)?;
        if !self.is_dir(&resolved_parent) {
            return Err(Error::NotDir(resolved_parent));
        }
        Ok((resolved_parent, name))
    }

    fn join(parent: &str, name: &str) -> String {
        if parent == "/" {
            format!("/{}", name)
        } else {
            format!("{}/{}", parent, name)
        }
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| *candidate != path && Self::parent(candidate) == path)
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let final_path = Self::join(&parent, &name);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries.insert(final_path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, true)?;
        if matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::IsDir(resolved));
        }
        if !self.entries.contains_key(&resolved) {
            let (parent, name) = self.resolve_parent(&resolved)?;
            let final_path = Self::join(&parent, &name);
            self.entries.insert(final_path, Entry::File(data.to_vec()));
        } else {
            self.entries.insert(resolved, Entry::File(data.to_vec()));
        }
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            _ => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        if !self.is_dir(&resolved) {
            return Err(Error::NotDir(resolved));
        }
        let mut result: Vec<(String, Kind)> = self
            .children(&resolved)
            .into_iter()
            .filter_map(|child| {
                let kind = match self.entries.get(&child)? {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                    Entry::Link(_) => Kind::Link,
                };
                Some((Self::basename(&child), kind))
            })
            .collect();
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        if !self.is_dir(&resolved) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        Self::validate(to)?;
        let source = Self::normalize(from, &self.cwd);
        let destination = Self::normalize(to, &self.cwd);

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, source_name) = self.resolve_parent(&source)?;
        let source_path = Self::join(&source_parent, &source_name);
        let source_entry = self
            .entries
            .get(&source_path)
            .cloned()
            .ok_or_else(|| Error::Missing(source_path.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &source_path) {
            return Err(Error::Busy(source_path));
        }

        let (destination_parent, destination_name) = self.resolve_parent(&destination)?;
        let destination_path = Self::join(&destination_parent, &destination_name);

        if matches!(self.entries.get(&destination_path), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &destination_path)
        {
            return Err(Error::Busy(destination_path));
        }

        if source_path == destination_path {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&destination_path, &source_path) {
            return Err(Error::Cycle(destination_path));
        }

        let destination_entry = self.entries.get(&destination_path).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination_path));
        }

        if let Some(ref dest) = destination_entry {
            match (&source_entry, dest) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.children(&destination_path).is_empty() {
                        return Err(Error::NotEmpty(destination_path));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination_path)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination_path)),
                _ => {}
            }
        }

        let source_keys: Vec<String> = self
            .entries
            .keys()
            .filter(|key| Self::is_within(key, &source_path))
            .cloned()
            .collect();
        let mut moved = Vec::new();
        for old in &source_keys {
            let suffix = if old == &source_path {
                String::new()
            } else {
                old[source_path.len()..].to_string()
            };
            moved.push((old.clone(), format!("{}{}", destination_path, suffix)));
        }

        if destination_entry.is_some() {
            let dest_keys: Vec<String> = self
                .entries
                .keys()
                .filter(|key| Self::is_within(key, &destination_path))
                .cloned()
                .collect();
            for key in dest_keys {
                self.entries.remove(&key);
            }
        }

        let mut tree = Vec::new();
        for (old, new) in moved {
            if let Some(entry) = self.entries.remove(&old) {
                tree.push((new, entry));
            }
        }
        for (path, entry) in tree {
            self.entries.insert(path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let final_path = Self::join(&parent, &name);
        let entry = self
            .entries
            .get(&final_path)
            .cloned()
            .ok_or_else(|| Error::Missing(final_path.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &final_path) {
            return Err(Error::Busy(final_path));
        }
        let descendants = self.children(&final_path);
        if matches!(entry, Entry::Dir) && !recursive && !descendants.is_empty() {
            return Err(Error::NotEmpty(final_path));
        }

        let keys: Vec<String> = self
            .entries
            .keys()
            .filter(|key| {
                *key == &final_path
                    || (recursive
                        && matches!(entry, Entry::Dir)
                        && Self::is_within(key, &final_path))
            })
            .cloned()
            .collect();
        for key in keys {
            self.entries.remove(&key);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let canonical = Self::normalize(link, &self.cwd);
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if canonical == "/" {
            return Err(Error::Exists("/".to_string()));
        }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let final_path = Self::join(&parent, &name);
        if self.entries.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.entries
            .insert(final_path, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            _ => Err(Error::InvalidTarget(0)),
        }
    }

    fn subtree_size(&self, path: &str) -> u64 {
        match self.entries.get(path) {
            Some(Entry::File(data)) => data.len() as u64,
            Some(Entry::Link(target)) => target.len() as u64,
            Some(Entry::Dir) => self
                .children(path)
                .iter()
                .map(|child| self.subtree_size(child))
                .sum(),
            None => 0,
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        Self::validate(path)?;
        let canonical = Self::normalize(path, &self.cwd);
        let resolved = self.resolve(&canonical, true, false)?;
        let mut paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| Self::is_within(candidate, &resolved))
            .cloned()
            .collect();
        paths.sort();
        Ok(paths
            .into_iter()
            .map(|item| {
                let size = self.subtree_size(&item);
                (item, size)
            })
            .collect())
    }
}
```
