```rust
use std::collections::HashSet;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Resume {
        remaining: u64,
        work_at: u64,
        finish: u64,
    },
    Preempt {
        remaining: u64,
        by: u32,
    },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: Option<u64>,
}

struct Running {
    customer: Customer,
    finish: u64,
    work_at: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(at), Some(bt)) => at
                .cmp(&bt)
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn take_best(waiting: &mut Vec<Waiting>) -> Option<Waiting> {
    let index = (0..waiting.len()).min_by(|&a, &b| waiting_cmp(&waiting[a], &waiting[b]))?;
    Some(waiting.remove(index))
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut current_time = 0u64;

    while next_arrival < arrivals.len() || running.is_some() || !waiting.is_empty() {
        let next_arrival_time = arrivals.get(next_arrival).map(|c| c.arrival);
        let next_finish_time = running.as_ref().map(|r| r.finish);
        let next_work_time = running
            .as_ref()
            .filter(|r| r.work_at > current_time)
            .map(|r| r.work_at);

        let time = [next_arrival_time, next_finish_time, next_work_time]
            .into_iter()
            .flatten()
            .min()
            .expect("simulation has pending work");
        current_time = time;

        if running.as_ref().is_some_and(|r| r.finish == time) {
            let finished = running.take().unwrap();
            events.push(Event {
                time,
                customer: finished.customer.id,
                kind: Kind::Finish,
            });
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == time {
            let customer = arrivals[next_arrival].clone();
            events.push(Event {
                time,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                let best_priority = waiting.iter().map(|w| w.customer.priority).min();
                if best_priority.is_some_and(|p| p < active.customer.priority) {
                    let by = waiting
                        .iter()
                        .min_by(|a, b| waiting_cmp(a, b))
                        .unwrap()
                        .customer
                        .id;
                    let active = running.take().unwrap();
                    let remaining = active.finish - time;
                    events.push(Event {
                        time,
                        customer: active.customer.id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    waiting.push(Waiting {
                        customer: active.customer,
                        remaining,
                        started: true,
                        pause_time: Some(time),
                    });
                }
            }
        }

        while running.is_none() {
            let Some(mut selected) = take_best(&mut waiting) else {
                break;
            };

            let switch = if last_dispatched.is_none()
                || last_dispatched == Some(selected.customer.id)
            {
                0
            } else {
                switch_cost
            };

            let work_at = time
                .checked_add(switch)
                .ok_or(Error::TimeOverflow {
                    customer: selected.customer.id,
                    time,
                })?;
            let finish = work_at
                .checked_add(selected.remaining)
                .ok_or(Error::TimeOverflow {
                    customer: selected.customer.id,
                    time,
                })?;

            let kind = if selected.started {
                Kind::Resume {
                    remaining: selected.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: selected.remaining,
                    work_at,
                    finish,
                }
            };

            events.push(Event {
                time,
                customer: selected.customer.id,
                kind,
            });
            selected.started = true;
            last_dispatched = Some(selected.customer.id);

            if finish == time {
                events.push(Event {
                    time,
                    customer: selected.customer.id,
                    kind: Kind::Finish,
                });
            } else {
                running = Some(Running {
                    customer: selected.customer,
                    finish,
                    work_at,
                });
            }
        }
    }

    Ok(events)
}
```
