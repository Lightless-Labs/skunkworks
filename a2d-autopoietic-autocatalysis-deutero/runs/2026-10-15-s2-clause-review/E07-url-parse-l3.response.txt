    InvalidByte { at: usize },

fn scan_bytes(input: &str) -> Result<(), usize> {
    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(at);
        }
    }
    Ok(())
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]).ok_or(CodecError::BadEscape { at: i })?;
            let lo = hex_value(bytes[i + 2]).ok_or(CodecError::BadEscape { at: i })?;
            decoded.push((hi << 4) | lo);
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => Err(CodecError::InvalidUtf8 {
            at: origins[error.valid_up_to()],
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for byte in text.bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn codec_to_parse(error: CodecError, offset: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn validate_component(
    component: Component,
    value: &str,
    offset: usize,
) -> Result<(), ParseError> {
    decode_component(component, value)
        .map(|_| ())
        .map_err(|error| codec_to_parse(error, offset))
}

fn parse_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {
    if port_text.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    for (index, byte) in port_text.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + index,
            });
        }
    }

    if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in port_text.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|current| current.checked_add((byte - b'0') as u32))
        {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }

    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

fn parse_authority(
    authority: &str,
    authority_start: usize,
) -> Result<(Option<String>, String, Option<u16>, bool), ParseError> {
    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, byte)| (byte == b'@').then_some(i))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_port, host_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            authority_start + at + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    if host_port.starts_with('[') {
        let close = match host_port.find(']') {
            Some(index) => index,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &host_port[1..close];
        if interior.is_empty() || interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }

        let suffix = &host_port[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if let Some(port_text) = suffix.strip_prefix(':') {
            Some(parse_port(port_text, host_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        };

        return Ok((userinfo, interior.to_ascii_lowercase(), port, true));
    }

    let colon = host_port.find(':');
    let (host_text, port) = if let Some(colon) = colon {
        let port_text = &host_port[colon + 1..];
        (
            &host_port[..colon],
            Some(parse_port(port_text, host_start + colon + 1)?),
        )
    } else {
        (host_port, None)
    };

    if host_text.is_empty() {
        return Err(ParseError::InvalidHost { at: host_start });
    }

    for (index, byte) in host_text.bytes().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_start + index,
            });
        }
    }

    Ok((
        userinfo,
        host_text.to_ascii_lowercase(),
        port,
        false,
    ))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    if let Err(at) = scan_bytes(input) {
        return Err(ParseError::InvalidByte { at });
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(colon) => colon,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (index, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: index });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let scheme = input[..colon].to_ascii_lowercase();
    let authority_start = colon + 3;
    let authority_end = input[authority_start..]
        .find(|character| matches!(character, '/' | '?' | '#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(input.len());
    let authority = &input[authority_start..authority_end];

    let (userinfo, host, port, ipv6) = parse_authority(authority, authority_start)?;

    let after_authority = &input[authority_end..];
    let fragment_relative = after_authority.find('#');
    let before_fragment_end = fragment_relative
        .map(|index| authority_end + index)
        .unwrap_or(input.len());
    let fragment = fragment_relative.map(|index| {
        input[authority_end + index + 1..].to_owned()
    });

    let before_fragment = &input[authority_end..before_fragment_end];
    let query_relative = before_fragment.find('?');
    let path_end = query_relative
        .map(|index| authority_end + index)
        .unwrap_or(before_fragment_end);
    let query = query_relative.map(|index| {
        input[authority_end + index + 1..before_fragment_end].to_owned()
    });

    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    if let Some(userinfo_value) = userinfo.as_deref() {
        validate_component(
            Component::UserInfo,
            userinfo_value,
            authority_start,
        )?;
    }
    if path_end != authority_end {
        validate_component(Component::Path, &path, authority_end)?;
    }
    if let Some(query_value) = query.as_deref() {
        let query_start = path_end + 1;
        validate_component(Component::Query, query_value, query_start)?;
    }
    if let Some(fragment_value) = fragment.as_deref() {
        let fragment_start = before_fragment_end + 1;
        validate_component(Component::Fragment, fragment_value, fragment_start)?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }

    decode_component(Component::Path, path).map_err(|error| match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let mut stack: Vec<&str> = Vec::new();
    let mut terminal_navigation = false;

    for (index, segment) in raw_segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, segment).expect("validated path");
        let is_dot = decoded == ".";
        let is_dotdot = decoded == "..";
        terminal_navigation = index + 1 == raw_segments.len() && (is_dot || is_dotdot);

        if is_dot {
            continue;
        } else if is_dotdot {
            stack.pop();
        } else {
            stack.push(segment);
        }
    }

    if terminal_navigation {
        stack.push("");
    }

    let mut output = String::from("/");
    output.push_str(&stack.join("/"));
    Ok(output)
}

fn reference_split(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_index = reference.find('#');
    let before_fragment = fragment_index
        .map(|index| &reference[..index])
        .unwrap_or(reference);
    let fragment = fragment_index.map(|index| &reference[index + 1..]);

    let query_index = before_fragment.find('?');
    let path = query_index
        .map(|index| &before_fragment[..index])
        .unwrap_or(before_fragment);
    let query = query_index.map(|index| &before_fragment[index + 1..]);

    (path, query, fragment)
}

fn build_absolute_url(
    base: &Url,
    path: &str,
    query: Option<&str>,
    fragment: Option<&str>,
) -> String {
    let mut result = String::new();
    result.push_str(&base.scheme);
    result.push_str("://");
    if let Some(userinfo) = &base.userinfo {
        result.push_str(userinfo);
        result.push('@');
    }
    if base.ipv6 {
        result.push('[');
        result.push_str(&base.host);
        result.push(']');
    } else {
        result.push_str(&base.host);
    }
    if let Some(port) = base.port {
        result.push(':');
        result.push_str(&port.to_string());
    }
    result.push_str(path);
    if let Some(query) = query {
        result.push('?');
        result.push_str(query);
    }
    if let Some(fragment) = fragment {
        result.push('#');
        result.push_str(fragment);
    }
    result
}

fn normalize_resolved(mut url: Url) -> Result<Url, ResolveError> {
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let normalized_base_path =
        normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    if let Err(at) = scan_bytes(reference) {
        return Err(ResolveError::Reference(ParseError::InvalidByte { at }));
    }

    if reference.is_empty() {
        let mut result = base_url;
        result.path = normalized_base_path;
        return Ok(result);
    }

    let first_delimiter = reference
        .find(|character| matches!(character, '/' | '?' | '#'))
        .unwrap_or(reference.len());

    if reference[..first_delimiter].contains(':') {
        let url = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalize_resolved(url);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        return normalize_resolved(url);
    }

    let (reference_path, reference_query, reference_fragment) = reference_split(reference);

    let (selected_path, selected_query) = if reference_path.is_empty() {
        (
            normalized_base_path.as_str(),
            if reference_query.is_some() {
                reference_query
            } else {
                base_url.query.as_deref()
            },
        )
    } else {
        let merged_path = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let base_directory_end = normalized_base_path
                .rfind('/')
                .map(|index| index + 1)
                .unwrap_or(0);
            let mut merged = normalized_base_path[..base_directory_end].to_owned();
            merged.push_str(reference_path);
            merged
        };
        let absolute = build_absolute_url(
            &base_url,
            &merged_path,
            reference_query,
            reference_fragment,
        );
        let mut parsed = parse_url(&absolute).map_err(ResolveError::Reference)?;
        parsed.path = normalise_path(&parsed.path).map_err(ResolveError::Path)?;
        return Ok(parsed);
    };

    let absolute = build_absolute_url(
        &base_url,
        selected_path,
        selected_query,
        reference_fragment,
    );
    let mut parsed = parse_url(&absolute).map_err(ResolveError::Reference)?;
    parsed.path = normalise_path(&parsed.path).map_err(ResolveError::Path)?;
    Ok(parsed)
}
