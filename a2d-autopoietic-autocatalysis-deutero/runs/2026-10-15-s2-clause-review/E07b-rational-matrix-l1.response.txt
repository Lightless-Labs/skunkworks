use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128_fraction(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    // Convert to unsigned magnitudes before sign normalization so even
    // i128::MIN is handled without signed negation overflow.
    let num_negative = (num < 0) ^ (den < 0);
    let num_mag = num.unsigned_abs();
    let den_mag = den.unsigned_abs();

    if num_mag == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let g = gcd_u128(num_mag, den_mag);
    let reduced_num_mag = num_mag / g;
    let reduced_den_mag = den_mag / g;

    let max_num_mag = if num_negative {
        1u128 << 63
    } else {
        i64::MAX as u128
    };
    if reduced_num_mag > max_num_mag || reduced_den_mag > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }

    let reduced_num = if num_negative {
        if reduced_num_mag == (1u128 << 63) {
            i64::MIN
        } else {
            -(reduced_num_mag as i64)
        }
    } else {
        reduced_num_mag as i64
    };

    Ok(Rational {
        num: reduced_num,
        den: reduced_den_mag as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_i128_fraction(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        // Reduce denominator growth before forming the sum. The scaled
        // numerator products each fit i128; their sum is checked below.
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(b as u128, d as u128) as i128;
        let left_scale = d / g;
        let right_scale = b / g;

        let left = a * left_scale;
        let right = c * right_scale;
        let n = left.checked_add(right).ok_or(RationalError::Overflow)?;
        let den = b * left_scale;
        from_i128_fraction(n, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(b as u128, d as u128) as i128;
        let left_scale = d / g;
        let right_scale = b / g;

        let left = a * left_scale;
        let right = c * right_scale;
        let n = left.checked_sub(right).ok_or(RationalError::Overflow)?;
        let den = b * left_scale;
        from_i128_fraction(n, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let mut a = self.num as i128;
        let mut b = self.den as i128;
        let mut c = rhs.num as i128;
        let mut d = rhs.den as i128;

        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(c.unsigned_abs(), b as u128) as i128;
        a /= g1;
        d /= g1;
        c /= g2;
        b /= g2;

        from_i128_fraction(a * c, b * d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let mut a = self.num as i128;
        let mut b = self.den as i128;
        let mut c = rhs.den as i128;
        let mut d = rhs.num as i128;

        let g1 = gcd_u128(a.unsigned_abs(), d.unsigned_abs()) as i128;
        let g2 = gcd_u128(c as u128, b as u128) as i128;
        a /= g1;
        d /= g1;
        c /= g2;
        b /= g2;

        from_i128_fraction(a * c, b * d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        from_i128_fraction(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let len = bytes.len();

        if len == 0 {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::Empty,
            });
        }

        let mut i = 0;
        if bytes[0] == b'-' {
            i = 1;
        }

        let numerator_start = i;
        while i < len && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(RationalError::Parse {
                at: i,
                kind: ParseKind::InvalidByte,
            });
        }

        let numerator_end = i;
        let mut denominator_start = None;
        let mut denominator_end = None;

        if i < len {
            if bytes[i] != b'/' {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }

            i += 1;
            let start = i;
            if i == len {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            if !bytes[i].is_ascii_digit() {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
            while i < len && bytes[i].is_ascii_digit() {
                i += 1;
            }
            denominator_start = Some(start);
            denominator_end = Some(i);

            if i < len {
                return Err(RationalError::Parse {
                    at: i,
                    kind: ParseKind::InvalidByte,
                });
            }
        }

        let numerator_digits = &bytes[numerator_start..numerator_end];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(RationalError::Parse {
                at: numerator_start + 1,
                kind: ParseKind::NonCanonical,
            });
        }

        if bytes[0] == b'-' && numerator_digits.len() == 1 && numerator_digits[0] == b'0' {
            return Err(RationalError::Parse {
                at: 0,
                kind: ParseKind::NonCanonical,
            });
        }

        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            let digits = &bytes[start..end];
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(RationalError::Parse {
                    at: start + 1,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let negative = bytes[0] == b'-';
        let numerator_magnitude = parse_magnitude(
            &bytes[numerator_start..numerator_end],
            if negative { 1u128 << 63 } else { i64::MAX as u128 },
        )
        .ok_or(RationalError::Parse {
            at: 0,
            kind: ParseKind::OutOfRange,
        })?;

        let denominator_value = if let (Some(start), Some(end)) =
            (denominator_start, denominator_end)
        {
            let value = parse_magnitude(&bytes[start..end], i64::MAX as u128).ok_or(
                RationalError::Parse {
                    at: start,
                    kind: ParseKind::OutOfRange,
                },
            )?;
            Some(value as i64)
        } else {
            None
        };

        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            if end - start == 1 && bytes[start] == b'0' {
                return Err(RationalError::Parse {
                    at: start,
                    kind: ParseKind::ZeroDenominator,
                });
            }
        }

        if let (Some(start), Some(_end), Some(den)) =
            (denominator_start, denominator_end, denominator_value)
        {
            let slash = start - 1;
            if den == 1 {
                return Err(RationalError::Parse {
                    at: slash,
                    kind: ParseKind::NonCanonical,
                });
            }
            if numerator_magnitude == 0 {
                return Err(RationalError::Parse {
                    at: slash,
                    kind: ParseKind::NonCanonical,
                });
            }
            if gcd_u128(numerator_magnitude, den as u128) > 1 {
                return Err(RationalError::Parse {
                    at: slash,
                    kind: ParseKind::NonCanonical,
                });
            }
        }

        let numerator = if negative {
            if numerator_magnitude == (1u128 << 63) {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };

        match denominator_value {
            Some(den) => Ok(Rational {
                num: numerator,
                den,
            }),
            None => Ok(Rational {
                num: numerator,
                den: 1,
            }),
        }
    }
}

fn parse_magnitude(digits: &[u8], limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for &digit in digits {
        let digit = (digit - b'0') as u128;
        if value > limit / 10 || (value == limit / 10 && digit > limit % 10) {
            return None;
        }
        value = value * 10 + digit;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
