use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous {
        hunk: usize,
        first: usize,
        second: usize,
    },
}

#[derive(Debug)]
struct Row {
    prefix: u8,
    payload: String,
}

#[derive(Debug)]
struct Hunk {
    line: usize,
    old_start: usize,
    old_count: usize,
    new_count: usize,
    rows: Vec<Row>,
    lead: usize,
    trail: usize,
}

fn parse_number(text: &str) -> Option<usize> {
    if text.is_empty()
        || !text.bytes().all(|byte| byte.is_ascii_digit())
        || (text.len() > 1 && text.starts_with('0'))
    {
        return None;
    }

    let value = text.parse::<usize>().ok()?;
    (value <= 128).then_some(value)
}

fn parse_hunk_header(text: &str) -> Option<(usize, usize, usize, usize)> {
    let inner = text.strip_prefix("@@ -")?.strip_suffix(" @@")?;
    let (old, new) = inner.split_once(" +")?;
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;

    if a.contains(',')
        || b.contains(',')
        || c.contains(',')
        || d.contains(',')
    {
        return None;
    }

    let a = parse_number(a)?;
    let b = parse_number(b)?;
    let c = parse_number(c)?;
    let d = parse_number(d)?;

    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }

    Some((a, b, c, d))
}

fn physical_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text.split_terminator('\n').collect()
    }
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }

    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&byte| byte == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);

    for (index, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(index).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: index + 1 });
        }
    }

    if lines.len() <= 2 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut index = 2;
    let mut previous_old_end = 0usize;
    let mut preceding_delta: isize = 0;

    while index < lines.len() {
        let header_line = index + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[index]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };

        let old_start = if b > 0 { a - 1 } else { a };
        let new_start = if d > 0 { c - 1 } else { c };

        index += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;

        while index < lines.len() && !lines[index].starts_with("@@") {
            let line = lines[index];
            let Some(prefix) = line.as_bytes().first().copied() else {
                return Err(ApplyError::BadBody { line: index + 1 });
            };

            if !matches!(prefix, b' ' | b'-' | b'+') || line.as_bytes().contains(&b'\r') {
                return Err(ApplyError::BadBody { line: index + 1 });
            }

            let payload = &line[1..];
            rows.push(Row {
                prefix,
                payload: payload.to_string(),
            });

            match prefix {
                b' ' => {
                    old_seen += 1;
                    new_seen += 1;
                }
                b'-' => old_seen += 1,
                b'+' => new_seen += 1,
                _ => unreachable!(),
            }

            index += 1;
        }

        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }

        if !rows.iter().any(|row| row.prefix == b'+' || row.prefix == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let lead = rows.iter().take_while(|row| row.prefix == b' ').count();
        let trail = rows.iter().rev().take_while(|row| row.prefix == b' ').count();

        if old_start < previous_old_end {
            return Err(ApplyError::Misordered { line: header_line });
        }

        let required_new = old_start as isize + preceding_delta;
        if required_new < 0 || new_start as isize != required_new {
            return Err(ApplyError::Misordered { line: header_line });
        }

        previous_old_end = old_start + b;
        preceding_delta += d as isize - b as isize;

        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_count: d,
            rows,
            lead,
            trail,
        });

        if hunks.len() > 16 {
            return Err(ApplyError::TooLarge);
        }
    }

    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&byte| byte == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }

    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }

    let lines: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1]
            .split('\n')
            .map(str::to_string)
            .collect()
    };

    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    Ok(lines)
}

fn abs_distance(position: usize, expected: isize) -> usize {
    (position as isize - expected).unsigned_abs()
}

fn compare_rank(left: (usize, usize), right: (usize, usize)) -> Ordering {
    left.cmp(&right)
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;

    let mut net: isize = 0;
    let mut drift: isize = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let expected = hunk.old_start as isize + net + drift;
        let max_left = options.fuzz.min(hunk.lead);
        let max_right = options.fuzz.min(hunk.trail);

        let mut best_rank: Option<(usize, usize)> = None;
        let mut best_choices: Vec<(usize, usize, usize)> = Vec::new();

        for left in 0..=max_left {
            for right in 0..=max_right {
                let trimmed_rows = hunk.rows.len() - left - right;
                let mut old_payloads = Vec::new();

                for row in hunk.rows.iter().skip(left).take(trimmed_rows) {
                    if row.prefix == b' ' || row.prefix == b'-' {
                        old_payloads.push(row.payload.as_str());
                    }
                }

                let required_old = hunk.old_count - left - right;
                if old_payloads.len() != required_old || required_old > working.len() {
                    continue;
                }

                for start in 0..=working.len() - required_old {
                    if start + hunk.old_count > working.len() {
                        continue;
                    }

                    let distance = abs_distance(start, expected);
                    if distance > options.max_offset {
                        continue;
                    }

                    if old_payloads
                        .iter()
                        .enumerate()
                        .any(|(offset, payload)| working[start + left + offset] != *payload)
                    {
                        continue;
                    }

                    let rank = (left + right, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_choices.clear();
                            best_choices.push((start, left, right));
                        }
                        Some(current) if compare_rank(rank, current) == Ordering::Less => {
                            best_rank = Some(rank);
                            best_choices.clear();
                            best_choices.push((start, left, right));
                        }
                        Some(current) if rank == current => {
                            best_choices.push((start, left, right));
                        }
                        _ => {}
                    }
                }
            }
        }

        if best_choices.is_empty() {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        }

        let mut starts: Vec<usize> = best_choices.iter().map(|choice| choice.0).collect();
        starts.sort_unstable();
        starts.dedup();

        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: starts[0],
                second: starts[1],
            });
        }

        let start = starts[0];
        let (_, left, right) = best_choices
            .iter()
            .filter(|choice| choice.0 == start)
            .min_by_key(|choice| choice.1)
            .copied()
            .unwrap();

        let trim_end = hunk.rows.len() - right;
        let mut replacement = Vec::new();

        for row in hunk.rows.iter().skip(left).take(trim_end - left) {
            if row.prefix == b' ' || row.prefix == b'+' {
                replacement.push(row.payload.clone());
            }
        }

        let replace_start = start + left;
        let replace_end = start + hunk.old_count - right;
        working.splice(replace_start..replace_end, replacement);

        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as isize - hunk.old_count as isize;
        drift = start as isize - (hunk.old_start as isize + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }

    Ok(output)
}
