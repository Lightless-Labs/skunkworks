#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = n.checked_neg().ok_or(RationalError::Overflow)?;
        d = d.checked_neg().ok_or(RationalError::Overflow)?;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n as i64, den: d as i64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            + (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = (self.num as i128) * (rhs.den as i128)
            - (rhs.num as i128) * (self.den as i128);
        let d = (self.den as i128) * (rhs.den as i128);
        rational_from_i128(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128(
            (self.num as i128) * (rhs.num as i128),
            (self.den as i128) * (rhs.den as i128),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128(
            (self.num as i128) * (rhs.den as i128),
            (self.den as i128) * (rhs.num as i128),
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        rational_from_i128(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }
        let numerator_end = i;

        let mut denominator_start = None;
        let mut denominator_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            i += 1;
            denominator_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start.unwrap() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            denominator_end = Some(i);
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let num_digits = numerator_end - numerator_start;
        if num_digits >= 2 && bytes[numerator_start] == b'0' {
            return Err(parse_err(numerator_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[numerator_start] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_err(start + 1, ParseKind::NonCanonical));
            }
        }

        let mut num_mag: u128 = 0;
        for &b in &bytes[numerator_start..numerator_end] {
            num_mag = num_mag * 10 + (b - b'0') as u128;
            let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
            if num_mag > limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }

        let mut den_mag: u128 = 1;
        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            den_mag = 0;
            for &b in &bytes[start..end] {
                den_mag = den_mag * 10 + (b - b'0') as u128;
                if den_mag > i64::MAX as u128 {
                    return Err(parse_err(start, ParseKind::OutOfRange));
                }
            }
            if den_mag == 0 {
                return Err(parse_err(start, ParseKind::ZeroDenominator));
            }
            if den_mag == 1 {
                return Err(parse_err(start - 1, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_err(start - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(parse_err(start - 1, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_mag == 1u128 << 63 {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };

        Ok(Rational {
            num: signed_num,
            den: den_mag as i64,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let count = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let zero = Rational { num: 0, den: 1 };
        let mut data = Vec::with_capacity(count);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&mut self, candidate_cols: usize) -> Result<(Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == self.rows {
                break;
            }
            let mut chosen: Option<usize> = None;
            for row in p..self.rows {
                let x = self.data[row * self.cols + col];
                if x.num == 0 {
                    continue;
                }
                let key = (x.den, x.num.unsigned_abs(), row);
                if chosen.map_or(true, |old| {
                    let y = self.data[old * self.cols + col];
                    key < (y.den, y.num.unsigned_abs(), old)
                }) {
                    chosen = Some(row);
                }
            }
            let Some(q) = chosen else { continue };
            if q != p {
                for c in 0..self.cols {
                    self.data.swap(p * self.cols + c, q * self.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = self.data[p * self.cols + col];
            for c in 0..self.cols {
                let original = self.data[p * self.cols + c];
                self.data[p * self.cols + c] = original.div(&pivot).map_err(|source| {
                    MatrixError::Arithmetic { row: p, col: c, step: p, source }
                })?;
            }

            for row in 0..self.rows {
                if row == p {
                    continue;
                }
                let factor = self.data[row * self.cols + col];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..self.cols {
                    let pivot_entry = self.data[p * self.cols + c];
                    let product = factor.mul(&pivot_entry).map_err(|source| MatrixError::Arithmetic {
                        row, col: c, step: p, source,
                    })?;
                    let current = self.data[row * self.cols + c];
                    self.data[row * self.cols + c] = current.sub(&product).map_err(|source| {
                        MatrixError::Arithmetic { row, col: c, step: p, source }
                    })?;
                }
            }

            pivots.push(col);
            p += 1;
        }
        Ok((pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let (pivots, swaps) = matrix.eliminate(self.cols)?;
        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c { one } else { zero });
            }
        }
        let mut augmented = Self { rows: n, cols: width, data };
        let (pivots, _) = augmented.eliminate(n)?;
        if pivots.len() < n {
            return Err(MatrixError::Singular { rank: pivots.len() });
        }
        let mut result = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in 0..n {
                result.push(augmented.data[r * width + n + c]);
            }
        }
        Ok(Self { rows: n, cols: n, data: result })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let m = self.rows;
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let total = m.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;

        let mut augmented_data = Vec::with_capacity(total);
        for r in 0..m {
            for c in 0..n {
                augmented_data.push(self.data[r * n + c]);
            }
            for c in 0..k {
                augmented_data.push(rhs.data[r * k + c]);
            }
        }
        let mut augmented = Self { rows: m, cols: width, data: augmented_data };
        let (pivots, _) = augmented.eliminate(n).map_err(SolveError::Matrix)?;

        for row in 0..m {
            let mut zero_coefficients = true;
            for c in 0..n {
                if augmented.data[row * width + c].num != 0 {
                    zero_coefficients = false;
                    break;
                }
            }
            if zero_coefficients {
                for rhs_col in 0..k {
                    if augmented.data[row * width + n + rhs_col].num != 0 {
                        let witness = augmented.data[row * width..(row + 1) * width].to_vec();
                        return Ok(Solution::None { row, rhs_col, witness });
                    }
                }
            }
        }

        let particular_count = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let zero = Rational { num: 0, den: 1 };
        let mut particular_data = vec![zero; particular_count];
        for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
            for rhs_col in 0..k {
                particular_data[pivot_col * k + rhs_col] =
                    augmented.data[pivot_row * width + n + rhs_col];
            }
        }
        let particular = Self { rows: n, cols: k, data: particular_data };

        if pivots.len() == n {
            return Ok(Solution::Unique { values: particular, pivots });
        }

        let mut free = Vec::with_capacity(n - pivots.len());
        let mut is_pivot = vec![false; n];
        for &p in &pivots {
            is_pivot[p] = true;
        }
        for col in 0..n {
            if !is_pivot[col] {
                free.push(col);
            }
        }

        let basis_count = n.checked_mul(free.len()).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![zero; basis_count];
        let one = Rational { num: 1, den: 1 };
        for (basis_col, &free_col) in free.iter().enumerate() {
            basis_data[free_col * free.len() + basis_col] = one;
            for (pivot_row, &pivot_col) in pivots.iter().enumerate() {
                let coefficient = augmented.data[pivot_row * width + free_col];
                basis_data[pivot_col * free.len() + basis_col] =
                    coefficient.neg().map_err(|source| SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: basis_col,
                        step: pivots.len(),
                        source,
                    }))?;
            }
        }
        let basis = Self { rows: n, cols: free.len(), data: basis_data };
        Ok(Solution::Infinite { particular, free, basis, pivots })
    }
}
