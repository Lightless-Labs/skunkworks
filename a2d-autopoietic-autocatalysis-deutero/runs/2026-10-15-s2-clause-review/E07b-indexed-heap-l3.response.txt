use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
}

#[derive(Debug, Clone)]
struct Entry {
    item: Item,
    age: u128,
}

#[derive(Debug, Clone)]
pub struct IndexedHeap {
    entries: HashMap<String, Entry>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn canonical_key(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }

        if start == end {
            return Err(Error::EmptyKey);
        }

        for i in start..end {
            let b = bytes[i];
            let valid = if i == start {
                b.is_ascii_alphabetic()
            } else {
                b.is_ascii_alphabetic() || b.is_ascii_digit() || b == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: i });
            }
        }

        Ok(input[start..end].to_ascii_lowercase())
    }

    fn rank(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn best_key(&self) -> Option<&str> {
        self.entries
            .iter()
            .min_by(|(_, a), (_, b)| Self::rank(a, b))
            .map(|(key, _)| key.as_str())
    }

    fn rebase_ages(&mut self) {
        let mut entries: Vec<Entry> = self.entries.values().cloned().collect();
        entries.sort_by(|a, b| a.age.cmp(&b.age));
        for (age, entry) in entries.into_iter().enumerate() {
            if let Some(stored) = self.entries.get_mut(&entry.item.key) {
                stored.age = age as u128;
            }
        }
        self.next_age = self.entries.len() as u128;
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            self.rebase_ages();
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = Self::canonical_key(key)?;
        if self.entries.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let age = self.fresh_age();
        self.entries.insert(
            key.clone(),
            Entry {
                item: Item {
                    key,
                    priority,
                    value: value.to_owned(),
                },
                age,
            },
        );
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        let key = self.best_key()?;
        self.entries.get(key).map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.best_key()?.to_owned();
        self.entries.remove(&key).map(|entry| entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = Self::canonical_key(key)?;
        Ok(self.entries.get(&key).map(|entry| entry.item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let entry = self
            .entries
            .get_mut(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = entry.item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        entry.item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let entry = self
            .entries
            .get_mut(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = entry.item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        entry.item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = Self::canonical_key(key)?;
        let entry = self
            .entries
            .get_mut(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        Ok(std::mem::replace(&mut entry.item.value, value.to_owned()))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = Self::canonical_key(key)?;
        self.entries
            .remove(&key)
            .map(|entry| entry.item)
            .ok_or(Error::Missing { key })
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let mut source: Vec<Entry> = other.entries.values().cloned().collect();
        source.sort_by(Self::rank);

        if policy == CollisionPolicy::Reject {
            for entry in &source {
                if self.entries.contains_key(&entry.item.key) {
                    return Err(Error::Collision {
                        key: entry.item.key.clone(),
                    });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
        };

        for mut source_entry in source {
            let key = source_entry.item.key.clone();
            if let Some(receiver_entry) = self.entries.get_mut(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        receiver_entry.item.priority = source_entry.item.priority;
                        receiver_entry.item.value = source_entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                source_entry.age = self.fresh_age();
                self.entries.insert(key, source_entry);
                report.moved += 1;
            }
        }

        other.entries.clear();
        Ok(report)
    }
}
