#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn is_word_char(c: char) -> bool {
    !is_separator(c) && c != '~'
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    content_width: usize,
}

fn render_lines(lines: &[Line], width: usize, justify: bool) -> Vec<String> {
    let mut output = Vec::with_capacity(lines.len());

    for (line_index, line) in lines.iter().enumerate() {
        let mut rendered = " ".repeat(line.indent);

        if justify && line_index + 1 < lines.len() && line.pieces.len() >= 2 {
            let gaps = line.pieces.len() - 1;
            let extra = width - line.indent - line.content_width;
            let each = extra / gaps;
            let remainder = extra % gaps;
            let start_gap = line_index % gaps;

            for (i, piece) in line.pieces.iter().enumerate() {
                rendered.push_str(piece);
                if i < gaps {
                    let relative = (i + gaps - start_gap) % gaps;
                    rendered.push_str(&" ".repeat(1 + each + usize::from(relative < remainder)));
                }
            }
        } else {
            rendered.push_str(&line.pieces.join(" "));
        }

        output.push(rendered);
    }

    output
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();

    for &(index, c) in &original {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..original.len() {
        if original[i].1 == '~'
            && (i == 0
                || i + 1 == original.len()
                || !is_word_char(original[i - 1].1)
                || !is_word_char(original[i + 1].1))
        {
            return Err(WrapError::InvalidMarker {
                index: original[i].0,
            });
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < original.len() {
        let c = original[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < original.len() && original[i + 1].1 == '\n' {
                i += 2;
            } else {
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    let mut segments: Vec<Vec<char>> = Vec::new();
    let mut boundary_counts: Vec<usize> = Vec::new();
    let mut current = Vec::new();
    i = 0;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let count = i - start;
            if count >= 2 {
                segments.push(std::mem::take(&mut current));
                boundary_counts.push(count - 1);
            } else {
                current.push('\n');
            }
        } else {
            current.push(normalized[i]);
            i += 1;
        }
    }
    segments.push(current);

    let mut output = Vec::new();

    for (segment_index, segment) in segments.iter().enumerate() {
        let mut words = Vec::<Word>::new();
        let mut word_chars = Vec::new();
        let mut markers = Vec::new();

        for &c in segment {
            if is_separator(c) {
                if !word_chars.is_empty() {
                    words.push(Word {
                        chars: std::mem::take(&mut word_chars),
                        markers: std::mem::take(&mut markers),
                    });
                }
            } else if c == '~' {
                markers.push(word_chars.len());
            } else {
                word_chars.push(c);
            }
        }
        if !word_chars.is_empty() {
            words.push(Word {
                chars: word_chars,
                markers,
            });
        }

        let mut lines = Vec::<Line>::new();
        let mut pieces = Vec::<String>::new();
        let mut line_width = 0usize;
        let mut line_index = 0usize;

        let indent_for = |index: usize| {
            if index == 0 {
                options.first_indent
            } else {
                options.rest_indent
            }
        };

        let mut indent = indent_for(line_index);
        let mut capacity = options.width - indent;

        for word in &words {
            let mut offset = 0usize;

            while offset < word.chars.len() {
                let remaining = word.chars.len() - offset;
                let gap = usize::from(!pieces.is_empty());
                let room = capacity.saturating_sub(line_width + gap);

                if remaining <= room {
                    if gap != 0 {
                        line_width += 1;
                    }
                    pieces.push(word.chars[offset..].iter().collect());
                    line_width += remaining;
                    offset = word.chars.len();
                    continue;
                }

                let available = capacity.saturating_sub(line_width + gap);
                let usable = word
                    .markers
                    .iter()
                    .copied()
                    .filter(|&marker| {
                        marker > offset
                            && marker < word.chars.len()
                            && marker - offset + 1 <= available
                    })
                    .last();

                let accepted = usable.filter(|&marker| {
                    if pieces.is_empty() {
                        return true;
                    }

                    let used = marker - offset + 1;
                    let unused = capacity - (line_width + gap + used);
                    let baseline = capacity - line_width;
                    unused * unused + options.hyphen_cost < baseline * baseline
                });

                if let Some(marker) = accepted {
                    if gap != 0 {
                        line_width += 1;
                    }
                    let mut piece: String = word.chars[offset..marker].iter().collect();
                    piece.push('-');
                    line_width += marker - offset + 1;
                    pieces.push(piece);

                    lines.push(Line {
                        indent,
                        pieces: std::mem::take(&mut pieces),
                        content_width: line_width,
                    });
                    line_width = 0;
                    line_index += 1;
                    indent = indent_for(line_index);
                    capacity = options.width - indent;

                    offset = marker;
                    continue;
                }

                if !pieces.is_empty() {
                    lines.push(Line {
                        indent,
                        pieces: std::mem::take(&mut pieces),
                        content_width: line_width,
                    });
                    line_width = 0;
                    line_index += 1;
                    indent = indent_for(line_index);
                    capacity = options.width - indent;
                    continue;
                }

                if remaining > capacity {
                    let end = offset + capacity;
                    pieces.push(word.chars[offset..end].iter().collect());
                    line_width = capacity;
                    lines.push(Line {
                        indent,
                        pieces: std::mem::take(&mut pieces),
                        content_width: line_width,
                    });
                    line_width = 0;
                    line_index += 1;
                    indent = indent_for(line_index);
                    capacity = options.width - indent;
                    offset = end;
                } else {
                    pieces.push(word.chars[offset..].iter().collect());
                    line_width = remaining;
                    offset = word.chars.len();
                }
            }
        }

        if !pieces.is_empty() {
            lines.push(Line {
                indent,
                pieces,
                content_width: line_width,
            });
        }

        output.extend(render_lines(&lines, options.width, options.justify));

        if segment_index < boundary_counts.len() {
            output.extend(std::iter::repeat_n(
                String::new(),
                boundary_counts[segment_index],
            ));
        }
    }

    Ok(output)
}
