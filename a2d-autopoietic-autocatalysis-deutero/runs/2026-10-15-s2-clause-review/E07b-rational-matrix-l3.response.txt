use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_parts(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }

        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128
            || n > i64::MAX as i128
            || d < 1
            || d > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }

        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_parts(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(b as u128, d as u128) as i128;
        let left_scale = d / g;
        let right_scale = b / g;
        let n = a * left_scale + c * right_scale;
        let den = b * left_scale;
        Self::from_parts(n, den)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        let g = gcd_u128(b as u128, d as u128) as i128;
        let left_scale = d / g;
        let right_scale = b / g;
        let n = a * left_scale - c * right_scale;
        let den = b * left_scale;
        Self::from_parts(n, den)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let mut a = self.num as i128;
        let mut b = self.den as i128;
        let mut c = rhs.num as i128;
        let mut d = rhs.den as i128;

        let g1 = gcd_u128(a.unsigned_abs(), d as u128) as i128;
        let g2 = gcd_u128(c.unsigned_abs(), b as u128) as i128;
        a /= g1;
        d /= g1;
        c /= g2;
        b /= g2;

        Self::from_parts(a * c, b * d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }

        let mut a = self.num as i128;
        let mut b = self.den as i128;
        let mut c = rhs.num as i128;
        let mut d = rhs.den as i128;

        let g1 = gcd_u128(a.unsigned_abs(), c.unsigned_abs()) as i128;
        let g2 = gcd_u128(d as u128, b as u128) as i128;
        a /= g1;
        c /= g1;
        d /= g2;
        b /= g2;

        Self::from_parts(a * d, b * c)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_parts(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
            if i == bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let numerator_start = 0;
        let numerator_digits_start = if bytes[0] == b'-' { 1 } else { 0 };
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_digits_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let numerator_end = i;
        let slash = if i < bytes.len() && bytes[i] == b'/' {
            let position = i;
            i += 1;
            Some(position)
        } else {
            None
        };

        let denominator_start = if slash.is_some() { i } else { bytes.len() };
        if slash.is_some() {
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        if i < bytes.len() {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let numerator_digits = &bytes[numerator_digits_start..numerator_end];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(parse_error(
                numerator_digits_start + 1,
                ParseKind::NonCanonical,
            ));
        }
        if bytes[0] == b'-' && numerator_digits == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let Some(_) = slash {
            let denominator_digits = &bytes[denominator_start..];
            if denominator_digits.len() >= 2 && denominator_digits[0] == b'0' {
                return Err(parse_error(denominator_start + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let numerator_limit: u128 = if negative {
            1_u128 << 63
        } else {
            i64::MAX as u128
        };
        let numerator_magnitude = parse_magnitude(numerator_digits);
        if numerator_magnitude > numerator_limit {
            return Err(parse_error(numerator_start, ParseKind::OutOfRange));
        }

        let denominator_magnitude = if slash.is_some() {
            let value = parse_magnitude(&bytes[denominator_start..]);
            if value > i64::MAX as u128 {
                return Err(parse_error(denominator_start, ParseKind::OutOfRange));
            }
            value
        } else {
            1
        };

        if slash.is_some() && denominator_magnitude == 0 {
            return Err(parse_error(denominator_start, ParseKind::ZeroDenominator));
        }

        if let Some(slash_at) = slash {
            if denominator_magnitude == 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if numerator_magnitude == 0 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(numerator_magnitude, denominator_magnitude) > 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if numerator_magnitude == (1_u128 << 63) {
                i64::MIN
            } else {
                -(numerator_magnitude as i64)
            }
        } else {
            numerator_magnitude as i64
        };

        Ok(Self {
            num: signed_num,
            den: denominator_magnitude as i64,
        })
    }
}

fn parse_magnitude(bytes: &[u8]) -> u128 {
    let mut value = 0u128;
    for &byte in bytes {
        value = value * 10 + (byte - b'0') as u128;
    }
    value
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),
    },
    Arithmetic {
        row: usize,
        col: usize,
        step: usize,
        source: RationalError,
    },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    fn offset(&self, row: usize, col: usize) -> Option<usize> {
        if row >= self.rows || col >= self.cols {
            return None;
        }
        row.checked_mul(self.cols)?.checked_add(col)
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        self.offset(row, col)
            .and_then(|idx| self.data.get(idx).copied())
            .ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for row in 0..self.cols {
            for col in 0..self.rows {
                data.push(self.data[col * self.cols + row]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }

        let output_len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;

        let mut output = Vec::with_capacity(output_len);
        for row in 0..self.rows {
            for col in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[row * self.cols + k];
                    let right = rhs.data[k * rhs.cols + col];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row,
                        col,
                        step: k,
                        source,
                    })?;
                }
                output.push(acc);
            }
        }

        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data: output,
        })
    }

    fn reduce_augmented(
        &self,
        pivot_cols: usize,
    ) -> Result<Reduction, MatrixError> {
        let mut work = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for column in 0..pivot_cols {
            if p == work.rows {
                break;
            }

            let mut best: Option<usize> = None;
            for row in p..work.rows {
                let value = work.data[row * work.cols + column];
                if value.num == 0 {
                    continue;
                }
                let replace = match best {
                    None => true,
                    Some(best_row) => {
                        let best_value = work.data[best_row * work.cols + column];
                        match value.den.cmp(&best_value.den) {
                            Ordering::Less => true,
                            Ordering::Greater => false,
                            Ordering::Equal => {
                                match value.num.unsigned_abs().cmp(&best_value.num.unsigned_abs()) {
                                    Ordering::Less => true,
                                    Ordering::Greater => false,
                                    Ordering::Equal => row < best_row,
                                }
                            }
                        }
                    }
                };
                if replace {
                    best = Some(row);
                }
            }

            let Some(q) = best else {
                continue;
            };

            if q != p {
                for c in 0..work.cols {
                    work.data.swap(p * work.cols + c, q * work.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = work.data[p * work.cols + column];
            for c in 0..work.cols {
                let idx = p * work.cols + c;
                work.data[idx] = work.data[idx]
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    })?;
            }

            for row in 0..work.rows {
                if row == p {
                    continue;
                }
                let factor = work.data[row * work.cols + column];
                if factor.num == 0 {
                    continue;
                }
                for c in 0..work.cols {
                    let product = factor
                        .mul(&work.data[p * work.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                    let idx = row * work.cols + c;
                    work.data[idx] = work.data[idx]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                }
            }

            pivots.push(column);
            p += 1;
        }

        Ok(Reduction {
            matrix: work,
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.reduce_augmented(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }

        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;

        let mut data = vec![Rational { num: 0, den: 1 }; total];
        for row in 0..n {
            for col in 0..n {
                data[row * width + col] = self.data[row * n + col];
            }
            data[row * width + n + row] = Rational { num: 1, den: 1 };
        }

        let augmented = Self {
            rows: n,
            cols: width,
            data,
        };
        let reduction = augmented.reduce_augmented(n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }

        let mut inverse_data = Vec::with_capacity(n * n);
        for row in 0..n {
            for col in 0..n {
                inverse_data.push(reduction.matrix.data[row * width + n + col]);
            }
        }

        Ok(Self {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}
