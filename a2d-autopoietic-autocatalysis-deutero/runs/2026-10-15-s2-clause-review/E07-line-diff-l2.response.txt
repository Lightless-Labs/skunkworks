#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn line_count(text: &str) -> usize {
    if text.is_empty() {
        0
    } else {
        text.bytes().filter(|&b| b == b'\n').count()
    }
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1].split('\n').collect()
    }
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }

    if let Some(byte) = old.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::Old,
            byte,
        });
    }
    if let Some(byte) = new.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::New,
            byte,
        });
    }

    if !old.is_empty() && !old.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::Old });
    }
    if !new.is_empty() && !new.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::New });
    }

    let old_len = line_count(old);
    let new_len = line_count(new);
    if old_len > 128 || new_len > 128 {
        return Err(DiffError::TooLarge { old_len, new_len });
    }

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);

    let mut lcs = vec![vec![0usize; new_len + 1]; old_len + 1];
    for i in (0..old_len).rev() {
        for j in (0..new_len).rev() {
            lcs[i][j] = if old_lines[i] == new_lines[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::with_capacity(old_len + new_len);
    let (mut i, mut j) = (0, 0);
    while i < old_len && j < new_len {
        if old_lines[i] == new_lines[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old_lines[i].to_string(),
            });
            i += 1;
            j += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            if delete_score > insert_score
                || (delete_score == insert_score && (i + j) % 2 == 0)
            {
                edits.push(Edit::Delete {
                    old_index: i,
                    line: old_lines[i].to_string(),
                });
                i += 1;
            } else {
                edits.push(Edit::Insert {
                    new_index: j,
                    line: new_lines[j].to_string(),
                });
                j += 1;
            }
        }
    }
    while i < old_len {
        edits.push(Edit::Delete {
            old_index: i,
            line: old_lines[i].to_string(),
        });
        i += 1;
    }
    while j < new_len {
        edits.push(Edit::Insert {
            new_index: j,
            line: new_lines[j].to_string(),
        });
        j += 1;
    }

    let is_change = |edit: &Edit| matches!(edit, Edit::Delete { .. } | Edit::Insert { .. });
    let mut groups: Vec<(usize, usize)> = Vec::new();
    let mut pos = 0;
    while pos < edits.len() {
        if is_change(&edits[pos]) {
            let start = pos;
            while pos < edits.len() && is_change(&edits[pos]) {
                pos += 1;
            }
            groups.push((start, pos));
        } else {
            pos += 1;
        }
    }

    if groups.is_empty() {
        return Ok(String::new());
    }

    let merge_gap = 2 * context;
    let mut merged: Vec<(usize, usize)> = Vec::new();
    for (start, end) in groups {
        if let Some(last) = merged.last_mut() {
            let keep_gap = edits[last.1..start]
                .iter()
                .filter(|edit| matches!(edit, Edit::Keep { .. }))
                .count();
            if keep_gap <= merge_gap {
                last.1 = end;
                continue;
            }
        }
        merged.push((start, end));
    }

    let mut out = String::from("--- before\n+++ after\n");
    for (group_start, group_end) in merged {
        let mut start = group_start;
        let mut preceding = 0;
        while start > 0 && preceding < context {
            start -= 1;
            if matches!(edits[start], Edit::Keep { .. }) {
                preceding += 1;
            }
        }

        let mut end = group_end;
        let mut following = 0;
        while end < edits.len() && following < context {
            if matches!(edits[end], Edit::Keep { .. }) {
                following += 1;
            }
            end += 1;
        }

        let mut old_count = 0usize;
        let mut new_count = 0usize;
        let mut old_before = 0usize;
        let mut new_before = 0usize;

        for edit in &edits[..start] {
            match edit {
                Edit::Keep { .. } | Edit::Delete { .. } => old_before += 1,
                Edit::Insert { .. } => {}
            }
            match edit {
                Edit::Keep { .. } | Edit::Insert { .. } => new_before += 1,
                Edit::Delete { .. } => {}
            }
        }

        for edit in &edits[start..end] {
            match edit {
                Edit::Keep { .. } => {
                    old_count += 1;
                    new_count += 1;
                }
                Edit::Delete { .. } => old_count += 1,
                Edit::Insert { .. } => new_count += 1,
            }
        }

        let old_start = if old_count > 0 {
            old_before + 1
        } else {
            old_before
        };
        let new_start = if new_count > 0 {
            new_before + 1
        } else {
            new_before
        };

        out.push_str(&format!(
            "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
        ));
        for edit in &edits[start..end] {
            match edit {
                Edit::Keep { line, .. } => {
                    out.push(' ');
                    out.push_str(line);
                    out.push('\n');
                }
                Edit::Delete { line, .. } => {
                    out.push('-');
                    out.push_str(line);
                    out.push('\n');
                }
                Edit::Insert { line, .. } => {
                    out.push('+');
                    out.push_str(line);
                    out.push('\n');
                }
            }
        }
    }

    Ok(out)
}
