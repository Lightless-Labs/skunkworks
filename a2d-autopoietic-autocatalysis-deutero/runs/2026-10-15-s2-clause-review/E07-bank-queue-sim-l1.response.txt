#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Start { finish: u64, wake: bool },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64 },
}

pub fn simulate(customers: &[Customer], wake_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::{HashSet, VecDeque};

    let mut seen = HashSet::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<&Customer> = customers.iter().collect();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut events = Vec::with_capacity(customers.len() * 3);
    let mut waiting: VecDeque<&Customer> = VecDeque::new();
    let mut next_arrival = 0usize;
    let mut completion: Option<(u64, &Customer)> = None;
    let mut previous_finish: Option<u64> = None;
    let mut started_once = false;

    while next_arrival < arrivals.len() || completion.is_some() || !waiting.is_empty() {
        if completion.is_none() && waiting.is_empty() {
            if let Some(customer) = arrivals.get(next_arrival) {
                let tick = customer.arrival;
                while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == tick {
                    let customer = arrivals[next_arrival];
                    events.push(Event {
                        time: tick,
                        customer: customer.id,
                        kind: Kind::Arrive,
                    });
                    waiting.push_back(customer);
                    next_arrival += 1;
                }
            }
        }

        let tick = if let Some((finish, _)) = completion {
            finish
        } else {
            waiting.front().expect("waiting customer must exist").arrival
        };

        if let Some((finish, customer)) = completion {
            if finish == tick {
                events.push(Event {
                    time: tick,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                completion = None;
                previous_finish = Some(tick);
            }
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == tick {
            let customer = arrivals[next_arrival];
            events.push(Event {
                time: tick,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push_back(customer);
            next_arrival += 1;
        }

        while completion.is_none() {
            let Some(customer) = waiting.pop_front() else {
                break;
            };

            let wake = !started_once
                || previous_finish.is_some_and(|finish_time| tick > finish_time);

            let finish = tick
                .checked_add(customer.service)
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time: tick,
                })?
                .checked_add(if wake { wake_cost } else { 0 })
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    time: tick,
                })?;

            events.push(Event {
                time: tick,
                customer: customer.id,
                kind: Kind::Start { finish, wake },
            });
            started_once = true;

            if finish == tick {
                events.push(Event {
                    time: tick,
                    customer: customer.id,
                    kind: Kind::Finish,
                });
                previous_finish = Some(tick);
            } else {
                completion = Some((finish, customer));
            }
        }
    }

    Ok(events)
}
