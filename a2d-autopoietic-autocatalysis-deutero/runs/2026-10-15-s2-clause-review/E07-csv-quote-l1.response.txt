#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    const START: u8 = 0;
    const UNQUOTED: u8 = 1;
    const QUOTED: u8 = 2;
    const AFTER_QUOTE: u8 = 3;

    let bytes = input.as_bytes();
    let mut i = if bytes.starts_with(&[0xEF, 0xBB, 0xBF]) {
        3
    } else {
        0
    };

    if i == bytes.len() {
        return Ok(Vec::new());
    }

    let mut records = Vec::new();
    let mut fields = Vec::new();
    let mut field = String::new();
    let mut state = START;
    let mut record = 0usize;
    let mut quote_open = 0usize;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul { record, byte: i });
        }

        if state != QUOTED && b == b'\r' {
            if bytes.get(i + 1) != Some(&b'\n') {
                return Err(CsvError::BareCr { record, byte: i });
            }
        }

        match state {
            START => match b {
                b',' => {
                    fields.push(String::new());
                    i += 1;
                }
                b'\n' => {
                    fields.push(String::new());
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    i += 1;
                }
                b'\r' => {
                    fields.push(String::new());
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    i += 2;
                }
                b'"' => {
                    state = QUOTED;
                    quote_open = i;
                    i += 1;
                }
                _ => {
                    field.push(b as char);
                    if b >= 0x80 {
                        let ch = input[i..].chars().next().unwrap();
                        field.pop();
                        field.push(ch);
                        i += ch.len_utf8();
                    } else {
                        i += 1;
                    }
                    state = UNQUOTED;
                }
            },
            UNQUOTED => match b {
                b',' => {
                    fields.push(std::mem::take(&mut field));
                    state = START;
                    i += 1;
                }
                b'\n' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    state = START;
                    i += 1;
                }
                b'\r' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    state = START;
                    i += 2;
                }
                b'"' => {
                    return Err(CsvError::UnexpectedQuote { record, byte: i });
                }
                _ => {
                    if b >= 0x80 {
                        let ch = input[i..].chars().next().unwrap();
                        field.push(ch);
                        i += ch.len_utf8();
                    } else {
                        field.push(b as char);
                        i += 1;
                    }
                }
            },
            QUOTED => {
                if b == b'"' {
                    if bytes.get(i + 1) == Some(&b'"') {
                        field.push('"');
                        i += 2;
                    } else {
                        state = AFTER_QUOTE;
                        i += 1;
                    }
                } else if b >= 0x80 {
                    let ch = input[i..].chars().next().unwrap();
                    field.push(ch);
                    i += ch.len_utf8();
                } else {
                    field.push(b as char);
                    i += 1;
                }
            }
            AFTER_QUOTE => match b {
                b',' => {
                    fields.push(std::mem::take(&mut field));
                    state = START;
                    i += 1;
                }
                b'\n' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    state = START;
                    i += 1;
                }
                b'\r' => {
                    fields.push(std::mem::take(&mut field));
                    records.push(std::mem::take(&mut fields));
                    record += 1;
                    state = START;
                    i += 2;
                }
                _ => return Err(CsvError::TrailingByte { record, byte: i }),
            },
            _ => unreachable!(),
        }
    }

    if state == QUOTED {
        return Err(CsvError::UnclosedQuote {
            record,
            byte: quote_open,
        });
    }

    if state != START || !fields.is_empty() {
        fields.push(field);
        records.push(fields);
    }

    Ok(records)
}
