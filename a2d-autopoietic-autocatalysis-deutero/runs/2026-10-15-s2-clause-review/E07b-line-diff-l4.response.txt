#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Debug, Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: MergeSide,
}

#[derive(Debug, Clone)]
struct Component {
    changes: Vec<usize>,
    start: usize,
    end: usize,
}

fn split_lines(input: &str) -> Vec<String> {
    if input.is_empty() {
        return Vec::new();
    }

    input
        .strip_suffix('\n')
        .unwrap_or(input)
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn canonical_changes(base: &[String], new: &[String], side: MergeSide) -> Vec<Change> {
    let n = base.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut consumed = 0usize;
    let mut changes = Vec::new();
    let mut run_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 run_start: &mut Option<usize>,
                 deletes: &mut usize,
                 inserts: &mut Vec<String>| {
        if let Some(start) = run_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
                side: side.clone(),
            });
            *deletes = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && base[i] == new[j] {
            flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
            i += 1;
            j += 1;
            consumed += 1;
            continue;
        }

        if run_start.is_none() {
            run_start = Some(consumed);
        }

        if i == n {
            inserts.push(new[j].clone());
            j += 1;
        } else if j == m {
            deletes += 1;
            i += 1;
            consumed += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            let delete = if delete_score > insert_score {
                true
            } else if insert_score > delete_score {
                false
            } else {
                (i + j) % 2 == 0
            };

            if delete {
                deletes += 1;
                i += 1;
                consumed += 1;
            } else {
                inserts.push(new[j].clone());
                j += 1;
            }
        }
    }

    flush(&mut changes, &mut run_start, &mut deletes, &mut inserts);
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;

    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn components(changes: &[Change]) -> Vec<Component> {
    let mut seen = vec![false; changes.len()];
    let mut result = Vec::new();

    for first in 0..changes.len() {
        if seen[first] {
            continue;
        }

        let mut stack = vec![first];
        seen[first] = true;
        let mut members = Vec::new();

        while let Some(current) = stack.pop() {
            members.push(current);
            for other in 0..changes.len() {
                if !seen[other] && interacts(&changes[current], &changes[other]) {
                    seen[other] = true;
                    stack.push(other);
                }
            }
        }

        let start = members
            .iter()
            .map(|&idx| changes[idx].start)
            .min()
            .unwrap();
        let end = members
            .iter()
            .map(|&idx| changes[idx].end)
            .max()
            .unwrap();

        members.sort_by_key(|&idx| (changes[idx].start, changes[idx].end));
        result.push(Component {
            changes: members,
            start,
            end,
        });
    }

    result.sort_by_key(|component| {
        (
            component.start,
            usize::from(component.start != component.end),
            component.end,
        )
    });
    result
}

fn project(
    base: &[String],
    component: &Component,
    changes: &[Change],
    side: MergeSide,
) -> Vec<String> {
    let mut selected: Vec<&Change> = component
        .changes
        .iter()
        .map(|&idx| &changes[idx])
        .filter(|change| change.side == side)
        .collect();
    selected.sort_by_key(|change| (change.start, change.end));

    let mut output = Vec::new();
    let mut cursor = component.start;

    for change in selected {
        output.extend(base[cursor..change.start].iter().cloned());
        output.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }

    output.extend(base[cursor..component.end].iter().cloned());
    output
}

fn append_lines(output: &mut Vec<String>, lines: &[String]) {
    output.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&byte| byte == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = canonical_changes(&base_lines, &left_lines, MergeSide::Left);
    changes.extend(canonical_changes(
        &base_lines,
        &right_lines,
        MergeSide::Right,
    ));

    let comps = components(&changes);
    let mut output = Vec::<String>::new();
    let mut cursor = 0usize;
    let mut conflicts = Vec::new();

    for component in &comps {
        output.extend(base_lines[cursor..component.start].iter().cloned());

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_projection = project(&base_lines, component, &changes, MergeSide::Left);
        let right_projection = project(&base_lines, component, &changes, MergeSide::Right);

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);

                    let region_len = component.end - component.start;
                    let base_start_label = if region_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!(
                        "||||||| BASE {base_start_label},{region_len}"
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: region_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }

        cursor = component.end;
    }

    output.extend(base_lines[cursor..].iter().cloned());

    let text = if output.is_empty() {
        String::new()
    } else {
        let mut text = output.join("\n");
        text.push('\n');
        text
    };

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
