#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
enum EndKind {
    Complete,
    Marker,
    Raw,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    rendered: String,
    consumed: usize,
    kind: EndKind,
}

#[derive(Clone)]
struct Layout {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn scalar_len(s: &str) -> usize {
    s.chars().count()
}

fn line_text(line: &Line) -> String {
    let mut out = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i != 0 {
            out.push(' ');
        }
        out.push_str(piece);
    }
    out
}

fn better(a: &Layout, b: &Layout) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_lines != b.marker_lines {
        return a.marker_lines < b.marker_lines;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    for (la, lb) in a.lines.iter().zip(&b.lines) {
        if la.consumed != lb.consumed {
            return la.consumed > lb.consumed;
        }
    }
    for (la, lb) in a.lines.iter().zip(&b.lines) {
        if la.rendered != lb.rendered {
            return la.rendered < lb.rendered;
        }
    }
    false
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    fn rec(
        wi: usize,
        offset: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        memo: &mut Vec<Vec<Vec<Option<Layout>>>>,
    ) -> Layout {
        if wi == words.len() {
            return Layout {
                cost: 0,
                marker_lines: 0,
                lines: Vec::new(),
            };
        }
        if let Some(layout) = &memo[wi][offset][first as usize] {
            return layout.clone();
        }

        let indent = if first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let mut candidates = Vec::new();

        let mut pos_word = wi;
        let mut pos_offset = offset;
        let mut pieces = Vec::new();
        let mut consumed_total = 0usize;
        let mut displayed = 0usize;

        loop {
            if pos_word >= words.len() {
                break;
            }
            let word = &words[pos_word];
            let remaining = word.chars.len() - pos_offset;
            let base_space = if pieces.is_empty() { 0 } else { 1 };
            let available = cap.saturating_sub(displayed + base_space);

            if remaining <= available {
                let text: String = word.chars[pos_offset..].iter().collect();
                let mut next_pieces = pieces.clone();
                next_pieces.push(text);
                let new_displayed = displayed + base_space + remaining;
                let new_consumed = consumed_total + remaining;
                let next = rec(pos_word + 1, 0, false, words, options, memo);
                let mut line = Line {
                    indent,
                    pieces: next_pieces,
                    rendered: String::new(),
                    consumed: new_consumed,
                    kind: EndKind::Complete,
                };
                line.rendered = line_text(&line);
                let gap = options.width - (indent + new_displayed);
                let final_line = next.lines.is_empty();
                let mut cost = (gap * gap) as u64;
                if final_line {
                    cost /= 4;
                }
                if !final_line && line.pieces.len() == 1 {
                    cost += 3;
                }
                let mut layout = Layout {
                    cost: cost + next.cost,
                    marker_lines: next.marker_lines,
                    lines: vec![line],
                };
                layout.lines.extend(next.lines);
                candidates.push(layout);
            }

            for &marker in &word.markers {
                if marker <= pos_offset {
                    continue;
                }
                let prefix_len = marker - pos_offset;
                if prefix_len + 1 > available {
                    continue;
                }
                let mut next_pieces = pieces.clone();
                let mut text: String = word.chars[pos_offset..marker].iter().collect();
                text.push('-');
                next_pieces.push(text);
                let new_displayed = displayed + base_space + prefix_len + 1;
                let new_consumed = consumed_total + prefix_len;
                let next = rec(pos_word, marker, false, words, options, memo);
                let mut line = Line {
                    indent,
                    pieces: next_pieces,
                    rendered: String::new(),
                    consumed: new_consumed,
                    kind: EndKind::Marker,
                };
                line.rendered = line_text(&line);
                let gap = options.width - (indent + new_displayed);
                let final_line = next.lines.is_empty();
                let mut cost = (gap * gap) as u64;
                if final_line {
                    cost /= 4;
                }
                if !final_line && line.pieces.len() == 1 {
                    cost += 3;
                }
                let mut layout = Layout {
                    cost: cost + options.hyphen_cost as u64 + next.cost,
                    marker_lines: 1 + next.marker_lines,
                    lines: vec![line],
                };
                layout.lines.extend(next.lines);
                candidates.push(layout);
            }

            if pieces.is_empty() && remaining > cap {
                let marker_fits = word.markers.iter().any(|&m| {
                    m > pos_offset && m - pos_offset + 1 <= cap
                });
                if !marker_fits {
                    let text: String = word.chars[pos_offset..pos_offset + cap].iter().collect();
                    let next = rec(pos_word, pos_offset + cap, false, words, options, memo);
                    let mut line = Line {
                        indent,
                        pieces: vec![text],
                        rendered: String::new(),
                        consumed: cap,
                        kind: EndKind::Raw,
                    };
                    line.rendered = line_text(&line);
                    let gap = options.width - options.width;
                    let final_line = next.lines.is_empty();
                    let mut cost = (gap * gap) as u64;
                    if final_line {
                        cost /= 4;
                    }
                    if !final_line {
                        cost += 3;
                    }
                    let mut layout = Layout {
                        cost: cost + (2 * options.width + 1) as u64 + next.cost,
                        marker_lines: next.marker_lines,
                        lines: vec![line],
                    };
                    layout.lines.extend(next.lines);
                    candidates.push(layout);
                }
            }

            if remaining > available {
                break;
            }
            let text: String = word.chars[pos_offset..].iter().collect();
            pieces.push(text);
            displayed += base_space + remaining;
            consumed_total += remaining;
            pos_word += 1;
            pos_offset = 0;
            if pos_word >= words.len() {
                break;
            }
        }

        let best = candidates
            .into_iter()
            .reduce(|a, b| if better(&a, &b) { a } else { b })
            .expect("paragraph has a legal layout");
        memo[wi][offset][first as usize] = Some(best.clone());
        best
    }

    let max_word = words.iter().map(|w| w.chars.len()).max().unwrap_or(0);
    let mut memo = vec![
        vec![vec![None; 2]; max_word.saturating_add(1)];
        words.len().saturating_add(1)
    ];
    let mut layout = rec(0, 0, true, words, options, &mut memo);
    for line in &mut layout.lines {
        line.rendered = line_text(line);
    }

    if options.justify {
        for i in 0..layout.lines.len().saturating_sub(1) {
            let line = &layout.lines[i];
            let gaps = line.pieces.len().saturating_sub(1);
            if gaps == 0 {
                continue;
            }
            let current_width = scalar_len(&line.rendered);
            let extra = options.width - current_width;
            let each = extra / gaps;
            let rem = extra % gaps;
            let mut out = " ".repeat(line.indent);
            for (j, piece) in line.pieces.iter().enumerate() {
                if j != 0 {
                    out.push(' ');
                    out.push_str(&" ".repeat(each + usize::from(
                        (j - 1 + gaps - (i % gaps)) % gaps < rem,
                    )));
                }
                out.push_str(piece);
            }
            layout.lines[i].rendered = out;
        }
    }
    layout.lines
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars = input.chars().count();
    if chars > 512 {
        return Err(WrapError::InputTooLong { chars });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let indexed: Vec<(usize, char)> = input.char_indices().collect();
    for (i, &(byte, c)) in indexed.iter().enumerate() {
        if c != '~' {
            continue;
        }
        let valid_side = |j: isize| -> bool {
            if j < 0 || j as usize >= indexed.len() {
                return false;
            }
            let ch = indexed[j as usize].1;
            ch != '~' && !is_separator(ch)
        };
        if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
            return Err(WrapError::InvalidMarker { index: byte });
        }
    }

    let mut normalized = String::new();
    let mut chars_iter = input.chars().peekable();
    while let Some(c) = chars_iter.next() {
        if c == '\r' {
            if chars_iter.peek() == Some(&'\n') {
                chars_iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let chars: Vec<char> = normalized.chars().collect();
    let mut output = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;

    while i <= chars.len() {
        if i < chars.len() && chars[i] == '\n' {
            let mut j = i;
            while j < chars.len() && chars[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                let segment = &chars[start..i];
                let words = parse_words(segment);
                output.extend(
                    solve_paragraph(&words, options)
                        .into_iter()
                        .map(|line| line.rendered),
                );
                for _ in 1..(j - i) {
                    output.push(String::new());
                }
                start = j;
                i = j;
                continue;
            }
        }
        if i == chars.len() {
            let segment = &chars[start..i];
            let words = parse_words(segment);
            output.extend(
                solve_paragraph(&words, options)
                    .into_iter()
                    .map(|line| line.rendered),
            );
            break;
        }
        i += 1;
    }

    Ok(output)
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Vec::new();
    let mut markers = Vec::new();

    for &c in segment {
        if is_separator(c) {
            if !current.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut current),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if c == '~' {
            markers.push(current.len());
        } else {
            current.push(c);
        }
    }
    if !current.is_empty() {
        words.push(Word {
            chars: current,
            markers,
        });
    }
    words
}
