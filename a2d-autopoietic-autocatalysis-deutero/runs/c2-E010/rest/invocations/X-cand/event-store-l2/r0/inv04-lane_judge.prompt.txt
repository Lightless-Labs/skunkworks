                return Err(ApplyError::LimitExceeded);

        }

                Status::Closed => return Err(ApplyError::Closed),

                return Err(ApplyError::InvalidFloor);

                return Err(ApplyError::InsufficientFunds);
            }

                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}

            }
            account.status = Status::Closed;
        }

            if account.balance != 0 {

        actual: Option<u64>,

#[derive(Debug, Clone, PartialEq, Eq)]
struct Stream {
    account: Account,
    events: Vec<Event>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    entries: Vec<(String, Stream)>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
        }
    }

    fn validate_id(id: &str) -> Result<(), StoreError> {
        let bytes = id.as_bytes();
        if bytes.is_empty() {
            return Err(StoreError::InvalidId { byte: 0 });
        }
        for (index, byte) in bytes.iter().enumerate() {
            let good = match index {
                0 => byte.is_ascii_lowercase(),
                _ => byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-',
            };
            if !good {
                return Err(StoreError::InvalidId { byte: index });
            }
        }
        if bytes.len() > 24 {
            return Err(StoreError::InvalidId { byte: 24 });
        }
        Ok(())
    }

    fn find(&self, id: &str) -> Option<usize> {
        self.entries.iter().position(|(key, _)| key == id)
    }

    pub fn append(
        &mut self,
        id: &str,
        expected: Expected,
        events: &[Event],
    ) -> Result<Account, StoreError> {
        Self::validate_id(id)?;
        let position = self.find(id);
        let actual = position.map(|i| self.entries[i].1.account.version);

        let matches = match &expected {
            Expected::Any => true,
            Expected::Missing => position.is_none(),
            Expected::Exact(v) => actual == Some(*v),
        };
        if !matches {
            return Err(StoreError::Conflict { expected, actual });
        }

        if events.is_empty() {
            return Ok(position
                .map(|i| self.entries[i].1.account.clone())
                .unwrap_or_else(initial_account));
        }

        let mut account = position
            .map(|i| self.entries[i].1.account.clone())
            .unwrap_or_else(initial_account);
        for (index, event) in events.iter().enumerate() {
            apply_event(&mut account, event)
                .map_err(|kind| StoreError::Rejected(ReplayError { index, kind }))?;
        }

        match position {
            Some(i) => {
                self.entries[i].1.account = account.clone();
                self.entries[i].1.events.extend_from_slice(events);
            }
            None => {
                self.entries.push((
                    id.to_owned(),
                    Stream {
                        account: account.clone(),
                        events: events.to_vec(),
                    },
                ));
            }
        }
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        Self::validate_id(id)?;
        Ok(self
            .find(id)
            .map(|i| self.entries[i].1.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        Self::validate_id(id)?;
        Ok(self.find(id).map(|i| {
            self.entries[i]
                .1
                .events
                .iter()
                .enumerate()
                .filter(|(index, _)| *index as u128 + 1 > after as u128)
                .map(|(_, event)| event.clone())
                .collect()
        }))
    }
}

impl Default for Store {
    fn default() -> Self {
        Self::new()
    }
}

fn validate_id(id: &str) -> Result<(), StoreError> {
    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Err(StoreError::InvalidId { byte: 0 });
    }
    let scan_len = bytes.len().min(24);
    for index in 0..scan_len {
        let byte = bytes[index];
        let valid = if index == 0 {
            (b'a'..=b'z').contains(&byte)
        } else {
            (b'a'..=b'z').contains(&byte)
                || (b'0'..=b'9').contains(&byte)
                || byte == b'-'
        };
        if !valid {
            return Err(StoreError::InvalidId { byte: index });
        }
    }
    if bytes.len() > 24 {
        return Err(StoreError::InvalidId { byte: 24 });
    }
    Ok(())
}

            .unwrap_or_else(initial_account);

        for (index, event) in events.iter().enumerate() {
            if let Err(kind) = apply(&mut account, event) {
                return Err(StoreError::Rejected(ReplayError { index, kind }));
            }
        }

        if events.is_empty() {
            return Ok(account);
        }

        if let Some(i) = position {
            self.streams[i].1.account = account.clone();
            self.streams[i].1.events.extend_from_slice(events);
        } else {
            self.streams.push((
                id.to_owned(),
                Stream {
                    account: account.clone(),
                    events: events.to_vec(),
                },
            ));
        }
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id)?;
        Ok(self.position(id).map(|i| self.streams[i].1.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.position(id).map(|i| {
            self.streams[i]
                .1
                .events
                .iter()
                .enumerate()
                .filter(|(index, _)| (*index as u128) + 1 > after as u128)
                .map(|(_, event)| event.clone())
                .collect()
        }))
    }
}

impl Default for Store {
    fn default() -> Self {
        Self::new()
    }
}

fn initial_account() -> Account {
    Account {
        status: Status::New,
        balance: 0,
        limit: 0,
        version: 0,
    }

fn apply(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    match event {
        Event::Open { limit } => {
            if *limit < 1 || *limit > 1_000_000_000 {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.version = account.version.checked_add(1).ok_or(ApplyError::VersionOverflow)?;
            account.status = Status::Open;
            account.limit = *limit;
        }
        Event::Deposit { amount } => {
            if account.status == Status::New {
                return Err(ApplyError::NotOpened);
            }
            if account.status == Status::Closed {
                return Err(ApplyError::Closed);
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
            account.version = account.version.checked_add(1).ok_or(ApplyError::VersionOverflow)?;
        }
        Event::Withdraw { amount, floor } => {
            if account.status == Status::New {
                return Err(ApplyError::NotOpened);
            }
            if account.status == Status::Closed {
                return Err(ApplyError::Closed);
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
            account.version = account.version.checked_add(1).ok_or(ApplyError::VersionOverflow)?;
        }
        Event::Close { expected_version } => {
            if account.status == Status::New {
                return Err(ApplyError::NotOpened);
            }
            if account.status == Status::Closed {
                return Err(ApplyError::Closed);
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.version = account.version.checked_add(1).ok_or(ApplyError::VersionOverflow)?;
            account.status = Status::Closed;
        }
    }
    Ok(())
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    let mut account = initial_account();
    for (index, event) in events.iter().enumerate() {
        if let Err(kind) = apply(&mut account, event) {
            return Err(ReplayError { index, kind });
        }
    }
    Ok(account)
}

fn validate_id(id: &str) -> Result<(), StoreError> {
    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Err(StoreError::InvalidId { byte: 0 });
    }
    let mut index = 0;
    while index < bytes.len() && index < 24 {
        let byte = bytes[index];
        let ok = if index == 0 {
            (b'a'..=b'z').contains(&byte)
        } else {
            (b'a'..=b'z').contains(&byte)
                || (b'0'..=b'9').contains(&byte)
                || byte == b'-'
        };
        if !ok {
            return Err(StoreError::InvalidId { byte: index });
        }
        index += 1;
    }
    if bytes.len() > 24 {
        return Err(StoreError::InvalidId { byte: 24 });
    }
    Ok(())
}

impl Store {
    pub fn new() -> Self {
        Self { streams: Vec::new() }
    }

    pub fn append(
        &mut self,
        id: &str,
        expected: Expected,
        events: &[Event],
    ) -> Result<Account, StoreError> {
        validate_id(id)?;
        let position = self.streams.iter().position(|(key, _)| key == id);
        let actual = position.map(|pos| self.streams[pos].1.account.version);
        let matches = match &expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(version) => actual == Some(*version),
        };
        if !matches {
            return Err(StoreError::Conflict { expected, actual });
        }

        let mut candidate = position
            .map(|pos| self.streams[pos].1.account.clone())
            .unwrap_or_else(initial_account);

        for (index, event) in events.iter().enumerate() {
            if let Err(kind) = apply(&mut candidate, event) {
                return Err(StoreError::Rejected(ReplayError { index, kind }));
            }
        }

        if events.is_empty() {
            return Ok(candidate);
        }

        if let Some(pos) = position {
            self.streams[pos].1.account = candidate.clone();
            self.streams[pos].1.events.extend_from_slice(events);
        } else {
            self.streams.push((
                id.to_owned(),
                Stream {
                    account: candidate.clone(),
                    events: events.to_vec(),
                },
            ));
        }
        Ok(candidate)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id)?;
        Ok(self
            .streams
            .iter()
            .find(|(key, _)| key == id)
            .map(|(_, stream)| stream.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.iter().find(|(key, _)| key == id).map(|(_, stream)| {
            let skip = usize::try_from(after).unwrap_or(usize::MAX);
            if skip >= stream.events.len() {
                Vec::new()
            } else {
                stream.events.iter().skip(skip).cloned().collect()
            }
        }))
    }
}

fn initial_account() -> Account {
    Account {
        status: Status::New,
        balance: 0,
        limit: 0,
        version: 0,
    }
}

fn apply_event(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.status = Status::Open;
            account.limit = *limit;
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            // Open accounts maintain 0 <= balance <= limit.
            // Subtraction is safe because both values are nonnegative and balance <= limit.
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.status = Status::Closed;
        }
    }

    account.version = account
        .version
        .checked_add(1)
        .ok_or(ApplyError::VersionOverflow)?;
    Ok(())
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    let mut account = initial_account();
    for (index, event) in events.iter().enumerate() {
        apply_event(&mut account, event).map_err(|kind| ReplayError { index, kind })?;
    }
    Ok(account)
}

fn validate_id(id: &str) -> Result<(), StoreError> {
    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Err(StoreError::InvalidId { byte: 0 });
    }
    for (index, byte) in bytes.iter().enumerate().take(24) {
        let valid = if index == 0 {
            byte.is_ascii_lowercase()
        } else {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-'
        };
        if !valid {
            return Err(StoreError::InvalidId { byte: index });
        }
    }
    if bytes.len() > 24 {
        return Err(StoreError::InvalidId { byte: 24 });
    }
    Ok(())
}

impl Store {
    pub fn new() -> Self {
        Self {
            streams: BTreeMap::new(),
        }
    }

    pub fn append(
        &mut self,
        id: &str,
        expected: Expected,
        events: &[Event],
    ) -> Result<Account, StoreError> {
        validate_id(id)?;
        let existing = self.streams.get(id);
        let actual = existing.map(|stream| stream.account.version);
        let matches = match &expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(version) => actual == Some(*version),
        };
        if !matches {
            return Err(StoreError::Conflict { expected, actual });
        }

        let mut account = existing
            .map(|stream| stream.account.clone())
            .unwrap_or_else(initial_account);

        if events.is_empty() {
            return Ok(account);
        }

        let mut updated_events = existing
            .map(|stream| stream.events.clone())
            .unwrap_or_default();

        for (index, event) in events.iter().enumerate() {
            apply_event(&mut account, event).map_err(|kind| {
                StoreError::Rejected(ReplayError { index, kind })
            })?;
            updated_events.push(event.clone());
        }

        self.streams.insert(
            id.to_owned(),
            Stream {
                account: account.clone(),
                events: updated_events,
            },
        );
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|stream| stream.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|stream| {
            stream
                .events
                .iter()
                .enumerate()
                .filter(|(index, _)| (*index as u64) + 1 > after)
                .map(|(_, event)| event.clone())
                .collect()
        }))
    }
}
