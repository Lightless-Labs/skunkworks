#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while a != 0 {
        let r = b % a;
        b = a;
        a = r;
    }
    b
}

fn make(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = if den < 0 { (-num, -den) } else { (num, den) };
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
        Err(RationalError::Overflow)
    } else {
        Ok(Rational { num: n as i64, den: d as i64 })
    }
}

fn parse_error(at: usize, kind: ParseKind) -> RationalError {
    RationalError::Parse { at, kind }
}

fn parse_magnitude(bytes: &[u8]) -> u128 {
    bytes.iter().fold(0u128, |v, b| v * 10 + (b - b'0') as u128)
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 { return Err(RationalError::ZeroDenominator); }
        make(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        make(self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
             self.den as i128 * rhs.den as i128)
    }
    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        make(self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
             self.den as i128 * rhs.den as i128)
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        make(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }
    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        make(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }
    pub fn neg(&self) -> Result<Self, RationalError> {
        make(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        if b.is_empty() { return Err(parse_error(0, ParseKind::Empty)); }

        let mut p = 0;
        if b[0] == b'-' { p = 1; }
        let ns = p;
        while p < b.len() && b[p].is_ascii_digit() { p += 1; }
        if p == ns { return Err(parse_error(p, ParseKind::InvalidByte)); }
        let ne = p;
        let mut fraction = None;
        if p < b.len() && b[p] == b'/' {
            let slash = p;
            p += 1;
            let ds = p;
            while p < b.len() && b[p].is_ascii_digit() { p += 1; }
            if p == ds { return Err(parse_error(p, ParseKind::InvalidByte)); }
            fraction = Some((slash, ds, p));
        }
        if p != b.len() { return Err(parse_error(p, ParseKind::InvalidByte)); }

        if ne - ns > 1 && b[ns] == b'0' {
            return Err(parse_error(ns + 1, ParseKind::NonCanonical));
        }
        if b[0] == b'-' && ne - ns == 1 && b[ns] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }
        if let Some((_, ds, de)) = fraction {
            if de - ds > 1 && b[ds] == b'0' {
                return Err(parse_error(ds + 1, ParseKind::NonCanonical));
            }
        }

        let nm = parse_magnitude(&b[ns..ne]);
        let negative = b[0] == b'-';
        if nm > if negative { 1u128 << 63 } else { i64::MAX as u128 } {
            return Err(parse_error(0, ParseKind::OutOfRange));
        }
        let (dm, slash, ds) = if let Some((s, ds, de)) = fraction {
            let v = parse_magnitude(&b[ds..de]);
            if v > i64::MAX as u128 { return Err(parse_error(ds, ParseKind::OutOfRange)); }
            (v, Some(s), Some(ds))
        } else {
            (1, None, None)
        };
        if let (Some(s), Some(ds)) = (slash, ds) {
            if dm == 0 { return Err(parse_error(ds, ParseKind::ZeroDenominator)); }
            if dm == 1 { return Err(parse_error(s, ParseKind::NonCanonical)); }
            if nm == 0 { return Err(parse_error(s, ParseKind::NonCanonical)); }
            if gcd(nm, dm) > 1 { return Err(parse_error(s, ParseKind::NonCanonical)); }
        }

        let n = if negative { -(nm as i128) } else { nm as i128 };
        make(n, dm as i128).map_err(|_| parse_error(0, ParseKind::OutOfRange))
    }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let n = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != n { return Err(MatrixError::DataLength { expected: n, actual: data.len() }); }
        Ok(Self { rows, cols, data })
    }
    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols { return Err(MatrixError::Index { row, col }); }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut out = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                out.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data: out }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch { left: (self.rows, self.cols), right: (rhs.rows, rhs.cols) });
        }
        let n = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(n);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut sum = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let x = self.data[r * self.cols + k];
                    let y = rhs.data[k * rhs.cols + c];
                    let product = x.mul(&y).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    sum = sum.add(&product).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                out.push(sum);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: out })
    }
