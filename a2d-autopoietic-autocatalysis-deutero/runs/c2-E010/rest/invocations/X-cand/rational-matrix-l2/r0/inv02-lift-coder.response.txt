```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        Self::from_i128(
            self.num as i128 * b + rhs.num as i128 * a,
            a * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        Self::from_i128(
            self.num as i128 * b - rhs.num as i128 * a,
            a * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128) as i128;
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128) as i128;
        Self::from_i128(
            (self.num as i128 / g1) * (rhs.num as i128 / g2),
            (self.den as i128 / g2) * (rhs.den as i128 / g1),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.num.unsigned_abs() as u128) as i128;
        let g2 = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        Self::from_i128(
            (self.num as i128 / g1) * (rhs.den as i128 / g2),
            (self.den as i128 / g2) * (rhs.num as i128 / g1),
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = -(self.num as i128);
        Self::from_i128(n, self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(fail(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            if i == bytes.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            return Err(fail(i, ParseKind::InvalidByte));
        }
        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        let numerator_digits = &bytes[num_start..num_end];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(fail(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && numerator_digits == b"0" {
            return Err(fail(0, ParseKind::NonCanonical));
        }

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            if de - ds >= 2 && bytes[ds] == b'0' {
                return Err(fail(ds + 1, ParseKind::NonCanonical));
            }
        }

        let num_limit: u128 = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut num_mag = 0u128;
        for &b in numerator_digits {
            num_mag = num_mag.saturating_mul(10).saturating_add((b - b'0') as u128);
            if num_mag > num_limit {
                return Err(fail(0, ParseKind::OutOfRange));
            }
        }
        let num = if negative {
            if num_mag == 1u128 << 63 { i64::MIN } else { -(num_mag as i64) }
        } else {
            num_mag as i64
        };

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            let mut den_mag = 0u128;
            for &b in &bytes[ds..de] {
                den_mag = den_mag.saturating_mul(10).saturating_add((b - b'0') as u128);
                if den_mag > i64::MAX as u128 {
                    return Err(fail(ds, ParseKind::OutOfRange));
                }
            }
            if den_mag == 0 {
                return Err(fail(ds, ParseKind::ZeroDenominator));
            }
            if den_mag == 1 {
                return Err(fail(ds - 1, ParseKind::NonCanonical));
            }
            if num == 0 {
                return Err(fail(ds - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(fail(ds - 1, ParseKind::NonCanonical));
            }
            return Ok(Self { num, den: den_mag as i64 });
        }
        Ok(Self { num, den: 1 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        let idx = row.checked_mul(self.cols)
            .and_then(|v| v.checked_add(col))
            .ok_or(MatrixError::Index { row, col })?;
        self.data.get(idx).copied().ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let out_len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(out_len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r, col: c, step: k, source,
                    })?;
                }
                out.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: out })
    }
}
```

```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while a != 0 {
        let r = b % a;
        b = a;
        a = r;
    }
    b
}

impl Rational {
    fn make(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (n, d) = if den < 0 { (-num, -den) } else { (num, den) };
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd(n.unsigned_abs(), d as u128) as i128;
        let n = n / g;
        let d = d / g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            Err(RationalError::Overflow)
        } else {
            Ok(Self { num: n as i64, den: d as i64 })
        }
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::make(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.den as i128;
        let b = rhs.den as i128;
        let g = gcd(a as u128, b as u128) as i128;
        Self::make(
            self.num as i128 * (b / g) + rhs.num as i128 * (a / g),
            (a / g) * b,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.den as i128;
        let b = rhs.den as i128;
        let g = gcd(a as u128, b as u128) as i128;
        Self::make(
            self.num as i128 * (b / g) - rhs.num as i128 * (a / g),
            (a / g) * b,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd(self.num.unsigned_abs(), rhs.den as u128) as i128;
        let g2 = gcd(rhs.num.unsigned_abs(), self.den as u128) as i128;
        Self::make(
            (self.num as i128 / g1) * (rhs.num as i128 / g2),
            (self.den as i128 / g2) * (rhs.den as i128 / g1),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        let g1 = gcd(self.num.unsigned_abs(), rhs.num.unsigned_abs()) as i128;
        let g2 = gcd(self.den as u128, rhs.den as u128) as i128;
        Self::make(
            (self.num as i128 / g1) * (rhs.den as i128 / g2),
            (self.den as i128 / g2) * (rhs.num as i128 / g1),
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::make(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() { return Err(err(0, ParseKind::Empty)); }

        let neg = b[0] == b'-';
        let ns = usize::from(neg);
        let mut i = ns;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == ns { return Err(err(i, ParseKind::InvalidByte)); }
        let ne = i;
        let mut ds = None;
        let mut de = None;
        if i < b.len() {
            if b[i] != b'/' { return Err(err(i, ParseKind::InvalidByte)); }
            i += 1;
            ds = Some(i);
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == ds.unwrap() { return Err(err(i, ParseKind::InvalidByte)); }
            de = Some(i);
            if i < b.len() { return Err(err(i, ParseKind::InvalidByte)); }
        }

        if ne - ns >= 2 && b[ns] == b'0' { return Err(err(ns + 1, ParseKind::NonCanonical)); }
        if neg && &b[ns..ne] == b"0" { return Err(err(0, ParseKind::NonCanonical)); }
        if let (Some(s), Some(e)) = (ds, de) {
            if e - s >= 2 && b[s] == b'0' { return Err(err(s + 1, ParseKind::NonCanonical)); }
        }

        let limit = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        let mut mag = 0u128;
        for &x in &b[ns..ne] {
            mag = mag.saturating_mul(10).saturating_add((x - b'0') as u128);
            if mag > limit { return Err(err(0, ParseKind::OutOfRange)); }
        }
        let n = if neg {
            if mag == 1u128 << 63 { i64::MIN } else { -(mag as i64) }
        } else { mag as i64 };

        if let (Some(s), Some(e)) = (ds, de) {
            let mut d = 0u128;
            for &x in &b[s..e] {
                d = d.saturating_mul(10).saturating_add((x - b'0') as u128);
                if d > i64::MAX as u128 { return Err(err(s, ParseKind::OutOfRange)); }
            }
            if d == 0 { return Err(err(s, ParseKind::ZeroDenominator)); }
            if d == 1 || n == 0 { return Err(err(s - 1, ParseKind::NonCanonical)); }
            if gcd(mag, d) != 1 { return Err(err(s - 1, ParseKind::NonCanonical)); }
            Ok(Self { num: n, den: d as i64 })
        } else {
            Ok(Self { num: n, den: 1 })
        }
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 { write!(f, "{}", self.num) } else { write!(f, "{}/{}", self.num, self.den) }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let n = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != n { return Err(MatrixError::DataLength { expected: n, actual: data.len() }); }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        self.data.get(row * self.cols + col).copied().ok_or(MatrixError::Index { row, col })
    }

    pub fn transpose(&self) -> Self {
        let mut d = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                d.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data: d }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut d = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut sum = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let p = self.data[r * self.cols + k].mul(&rhs.data[k * rhs.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    sum = sum.add(&p)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                d.push(sum);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: d })
    }
}
```
