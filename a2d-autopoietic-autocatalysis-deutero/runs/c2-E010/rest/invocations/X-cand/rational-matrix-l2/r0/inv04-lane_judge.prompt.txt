    }

fn make(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = if den < 0 { (-num, -den) } else { (num, den) };
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
        Err(RationalError::Overflow)
    } else {
        Ok(Rational { num: n as i64, den: d as i64 })
    }
}

    RationalError::Parse { at, kind }

    bytes.iter().fold(0u128, |v, b| v * 10 + (b - b'0') as u128)

        Ok(Self { rows, cols, data })

        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut out = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                out.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data: out }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {

            return Err(MatrixError::DimensionMismatch { left: (self.rows, self.cols), right: (rhs.rows, rhs.cols) });

                out.push(sum);

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(mut num: i128, mut den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if den < 0 {
        num = -num;
        den = -den;
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
    num /= g;
    den /= g;
    if num < i64::MIN as i128 || num > i64::MAX as i128 || den > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: num as i64, den: den as i64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

        rational_from_i128(

            self.den as i128 * rhs.den as i128,

        Ok(Self { numerator: n as i64, denominator: d as i64 })

        Self::build(self.numerator as i128 * rhs.numerator as i128, self.denominator as i128 * rhs.denominator as i128)

        let b = text.as_bytes();
        if b.is_empty() { return Err(err(0, ParseKind::Empty)); }
        let mut i = usize::from(b[0] == b'-');
        let ns = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == ns { return Err(err(i, ParseKind::InvalidByte)); }
        let ne = i;
        let mut ds = None;
        let mut de = None;
        if i < b.len() {
            if b[i] != b'/' { return Err(err(i, ParseKind::InvalidByte)); }
            i += 1;
            ds = Some(i);
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == ds.unwrap() { return Err(err(i, ParseKind::InvalidByte)); }
            de = Some(i);
            if i < b.len() { return Err(err(i, ParseKind::InvalidByte)); }
        }

        if nds == 1 && ne == 2 && b[1] == b'0' { return Err(err(0, ParseKind::NonCanonical)); }
        if let (Some(s), Some(e)) = (ds, de) {
            if e - s > 1 && b[s] == b'0' { return Err(err(s + 1, ParseKind::NonCanonical)); }
        }

        let nm = mag(&b[nds..ne]);
        if nm > if nds == 1 { 1u128 << 63 } else { i64::MAX as u128 } {
            return Err(err(0, ParseKind::OutOfRange));
        }
        let n = if nds == 1 {
            if nm == 1u128 << 63 { i64::MIN } else { -(nm as i64) }
        } else { nm as i64 };

        if let (Some(s), Some(e)) = (ds, de) {
            let dm = mag(&b[s..e]);
            if dm > i64::MAX as u128 { return Err(err(s, ParseKind::OutOfRange)); }
            if dm == 0 { return Err(err(s, ParseKind::ZeroDenominator)); }
            if dm == 1 { return Err(err(s - 1, ParseKind::NonCanonical)); }

            if gcd_b(nm, dm) != 1 { return Err(err(s - 1, ParseKind::NonCanonical)); }
            return Ok(Self { numerator: n, denominator: dm as i64 });
        }
        Ok(Self { numerator: n, denominator: 1 })
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' { i += 1; }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
        if i == num_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }
        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
            if i == den_start.unwrap() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
        }

        let neg = bytes[0] == b'-';
        let num_digits_start = if neg { 1 } else { 0 };
        if num_end - num_digits_start >= 2 && bytes[num_digits_start] == b'0' {
            return Err(parse_err(num_digits_start + 1, ParseKind::NonCanonical));
        }
        if neg && num_end == 2 && bytes[1] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }
        if let (Some(s), Some(e)) = (den_start, den_end) {
            if e - s >= 2 && bytes[s] == b'0' {
                return Err(parse_err(s + 1, ParseKind::NonCanonical));
            }
        }

        let num_mag = parse_magnitude(&bytes[num_digits_start..num_end]);
        let num_limit = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        if num_mag > num_limit {
            return Err(parse_err(0, ParseKind::OutOfRange));
        }
        let num_value = if neg {
            if num_mag == (1u128 << 63) { i64::MIN } else { -(num_mag as i64) }
        } else {
            num_mag as i64
        };

        if let (Some(s), Some(e)) = (den_start, den_end) {
            let den_mag = parse_magnitude(&bytes[s..e]);
            if den_mag > i64::MAX as u128 {
                return Err(parse_err(s, ParseKind::OutOfRange));
            }
            if den_mag == 0 {
                return Err(parse_err(s, ParseKind::ZeroDenominator));
            }
            if den_mag == 1 {
                return Err(parse_err(s - 1, ParseKind::NonCanonical));
            }
            if num_value == 0 {
                return Err(parse_err(s - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, den_mag) > 1 {
                return Err(parse_err(s - 1, ParseKind::NonCanonical));
            }
            return Ok(Self { num: num_value, den: den_mag as i64 });
        }
        Ok(Self { num: num_value, den: 1 })
    }
}

fn parse_magnitude(bytes: &[u8]) -> u128 {
    bytes.iter().fold(0u128, |v, b| v * 10 + (b - b'0') as u128)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 { write!(f, "{}", self.num) } else { write!(f, "{}/{}", self.num, self.den) }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        if len == 0 {
            return Ok(Self { rows: self.rows, cols: rhs.cols, data });
        }
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let left = self.data[r * self.cols + k];
                    let right = rhs.data[k * rhs.cols + c];
                    let term = left.mul(&right).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }
}
