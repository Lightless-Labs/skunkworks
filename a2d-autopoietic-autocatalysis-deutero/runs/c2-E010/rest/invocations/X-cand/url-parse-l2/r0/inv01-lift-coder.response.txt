- Scheme ends at the first colon anywhere; an invalid character before it is reported as a scheme error.
- The authority ends at the first `/`, `?`, or `#`; extra slashes can therefore produce an empty host.
- Multiple `@` signs are rejected before host checks; a single `@` preserves all preceding userinfo verbatim.
- Port validation checks every byte for nondigits before leading-zero, zero, or range errors.
- Component validation runs after port parsing, one component at a time in userinfo, path, query, fragment order.
- IPv6 interior spelling is preserved except for ASCII lowercasing; the standard library parser validates it without canonicalizing it.

```rust
use std::net::Ipv6Addr;
use std::str::FromStr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let (Some(hi), Some(lo)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) else {
                return Err(CodecError::BadEscape { at: i });
            };
            decoded.push((hi << 4) | lo);
            offsets.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: offsets.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = matches!(b, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn validate_component(text: &str, base: usize) -> Result<(), ParseError> {
    decode_component(Component::Path, text).map(|_| ()).map_err(|e| match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    })
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if colon + 2 >= bytes.len() || bytes[colon + 1] != b'/' || bytes[colon + 2] != b'/' {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|n| auth_start + n)
        .unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];

    let mut at_positions = authority.match_indices('@').map(|(i, _)| i);
    let first_at = at_positions.next();
    let second_at = at_positions.next();
    if let Some(second) = second_at {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + second,
        });
    }
    let (userinfo, host_start) = match first_at {
        Some(i) if i == 0 => {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        Some(i) => (Some(authority[..i].to_owned()), auth_start + i + 1),
        None => (None, auth_start),
    };
    let hostport_start_rel = first_at.map(|i| i + 1).unwrap_or(0);
    let hostport = &authority[hostport_start_rel..];

    let mut ipv6 = false;
    let (host, port, host_end_offset) = if hostport.starts_with('[') {
        let open_abs = host_start;
        let close_rel = match hostport.find(']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: open_abs }),
        };
        let interior = &hostport[1..close_rel];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_abs });
        }
        ipv6 = true;
        let suffix = &hostport[close_rel + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], open_abs + close_rel + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: open_abs + close_rel + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, host_start + close_rel + 1)
    } else {
        let hostport_bytes = hostport.as_bytes();
        let colon_rel = hostport_bytes.iter().position(|&b| b == b':');
        let (host_raw, port) = match colon_rel {
            Some(i) => (
                &hostport[..i],
                Some(parse_port(
                    &hostport[i + 1..],
                    host_start + i + 1,
                )?),
            ),
            None => (hostport, None),
        };

        if host_raw.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, &b) in host_raw.as_bytes().iter().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + i,
                });
            }
        }
        (host_raw.to_ascii_lowercase(), port, host_start + host_raw.len())
    };

    let fragment_rel = input[auth_end..].find('#').map(|n| n + auth_end);
    let before_fragment_end = fragment_rel.unwrap_or(input.len());
    let query_rel = input[auth_end..before_fragment_end]
        .find('?')
        .map(|n| n + auth_end);

    let path_end = query_rel.unwrap_or(before_fragment_end);
    let path = if auth_end == path_end {
        "/".to_owned()
    } else {
        input[auth_end..path_end].to_owned()
    };
    let query = query_rel.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_rel.map(|f| input[f + 1..].to_owned());

    if let Some(ref value) = userinfo {
        validate_component(value, auth_start)?;
    }
    if path != "/" || auth_end != path_end {
        validate_component(&path, auth_end)?;
    }
    if let Some(q) = query_rel {
        validate_component(query.as_deref().unwrap_or(""), q + 1)?;
    }
    if let Some(f) = fragment_rel {
        validate_component(fragment.as_deref().unwrap_or(""), f + 1)?;
    }

    let _ = host_end_offset;
    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + i,
            });
        }
    }
    let bytes = port.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    let mut value: u32 = 0;
    for b in bytes {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u32);
        if value > u16::MAX as u32 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}
```

```rust
use std::net::Ipv6Addr;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
    pub ipv6: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
    InvalidIpv6 { at: usize },
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Component {
    UserInfo,
    Path,
    Query,
    Fragment,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CodecError {
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let b = input.as_bytes();
    let mut decoded = Vec::with_capacity(b.len());
    let mut origin = Vec::with_capacity(b.len());
    let mut i = 0usize;

    while i < b.len() {
        if b[i] != b'%' {
            decoded.push(b[i]);
            origin.push(i);
            i += 1;
            continue;
        }
        if i.checked_add(2).map_or(true, |j| j >= b.len()) {
            return Err(CodecError::BadEscape { at: i });
        }
        let hi = match b[i + 1] {
            b'0'..=b'9' => b[i + 1] - b'0',
            b'a'..=b'f' => b[i + 1] - b'a' + 10,
            b'A'..=b'F' => b[i + 1] - b'A' + 10,
            _ => return Err(CodecError::BadEscape { at: i }),
        };
        let lo = match b[i + 2] {
            b'0'..=b'9' => b[i + 2] - b'0',
            b'a'..=b'f' => b[i + 2] - b'a' + 10,
            b'A'..=b'F' => b[i + 2] - b'A' + 10,
            _ => return Err(CodecError::BadEscape { at: i }),
        };
        decoded.push((hi << 4) | lo);
        origin.push(i);
        i += 3;
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origin.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let allowed = match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        };
        if unreserved || allowed {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn codec_error_to_parse(err: CodecError, base: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

fn validate_component(value: &str, base: usize) -> Result<(), ParseError> {
    decode_component(Component::Path, value)
        .map(|_| ())
        .map_err(|e| codec_error_to_parse(e, base))
}

fn parse_port(s: &str, start: usize) -> Result<u16, ParseError> {
    if s.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut digits = true;
    let mut first_bad = start;
    for (i, b) in s.bytes().enumerate() {
        if !b.is_ascii_digit() && digits {
            digits = false;
            first_bad = start + i;
        }
    }
    if !digits {
        return Err(ParseError::InvalidPort { at: first_bad });
    }

    if s.len() >= 2 && s.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value: u32 = 0;
    for b in s.bytes() {
        value = value * 10 + (b - b'0') as u32;
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if value == 0 {
        Err(ParseError::InvalidPort { at: start })
    } else {
        Ok(value as u16)
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, b) in bytes.iter().enumerate() {
        if !matches!(*b, 0x21..=0x7e) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let colon = match bytes.iter().position(|b| *b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        if !(bytes[i].is_ascii_alphanumeric() || matches!(bytes[i], b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if colon + 2 >= bytes.len() || bytes[colon + 1] != b'/' || bytes[colon + 2] != b'/' {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|b| matches!(*b, b'/' | b'?' | b'#'))
        .map(|n| auth_start + n)
        .unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];

    let ats: Vec<usize> = authority.match_indices('@').map(|(i, _)| i).collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + ats[1],
        });
    }
    let (userinfo, hp_start) = match ats.first().copied() {
        Some(0) => return Err(ParseError::InvalidUserInfo { at: auth_start }),
        Some(i) => (Some(authority[..i].to_owned()), auth_start + i + 1),
        None => (None, auth_start),
    };
    let hostport = &input[hp_start..auth_end];

    let (host, port, ipv6) = if hostport.starts_with('[') {
        let close = match hostport.find(']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: hp_start }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hp_start });
        }
        let tail_start = hp_start + close + 1;
        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], tail_start + 1)?)
        } else {
            return Err(ParseError::InvalidHost { at: tail_start });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport.bytes().position(|b| b == b':');
        let (host_text, port) = match split {
            Some(i) => (
                &hostport[..i],
                Some(parse_port(&hostport[i + 1..], hp_start + i + 1)?),
            ),
            None => (hostport, None),
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hp_start });
        }
        for (i, b) in host_text.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: hp_start + i });
            }
        }
        (host_text.to_ascii_lowercase(), port, false)
    };

    let fragment_at = input[auth_end..].find('#').map(|x| auth_end + x);
    let pre_fragment_end = fragment_at.unwrap_or(input.len());
    let query_at = input[auth_end..pre_fragment_end]
        .find('?')
        .map(|x| auth_end + x);

    let path_end = query_at.unwrap_or(pre_fragment_end);
    let path = if path_end == auth_end {
        "/".to_owned()
    } else {
        input[auth_end..path_end].to_owned()
    };
    let query = query_at.map(|q| input[q + 1..pre_fragment_end].to_owned());
    let fragment = fragment_at.map(|f| input[f + 1..].to_owned());

    if let Some(ref value) = userinfo {
        let base = auth_start + ats[0] + 1;
        validate_component(value, base)?;
    }
    if path_end > auth_end {
        validate_component(&path, auth_end)?;
    }
    if let Some(q) = query_at {
        validate_component(query.as_deref().unwrap_or(""), q + 1)?;
    }
    if let Some(f) = fragment_at {
        validate_component(fragment.as_deref().unwrap_or(""), f + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}
```
