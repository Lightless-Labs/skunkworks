fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    decoded.push((h << 4) | l);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = e.valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: origins.get(at).copied().unwrap_or(input.len()),
            })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = matches!(b, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: port_start + i });
        }
    }
    let bytes = port.as_bytes();
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for &b in bytes {
        value = value * 10 + (b - b'0') as u32;
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let src = input.as_bytes();
    let mut output = Vec::with_capacity(src.len());
    let mut origins = Vec::with_capacity(src.len());
    let mut cursor = 0;

    while cursor < src.len() {
        let byte = src[cursor];
        if byte != b'%' {
            output.push(byte);
            origins.push(cursor);
            cursor += 1;
            continue;
        }
        if cursor + 2 >= src.len() {
            return Err(CodecError::BadEscape { at: cursor });
        }
        let high = match hex_value(src[cursor + 1]) {
            Some(value) => value,
            None => return Err(CodecError::BadEscape { at: cursor }),
        };
        let low = match hex_value(src[cursor + 2]) {
            Some(value) => value,
            None => return Err(CodecError::BadEscape { at: cursor }),
        };
        output.push((high << 4) | low);
        origins.push(cursor);
        cursor += 3;
    }

    if let Err(error) = std::str::from_utf8(&output) {
        let index = error.valid_up_to();
        return Err(CodecError::InvalidUtf8 {
            at: origins.get(index).copied().unwrap_or(src.len()),
        });
    }
    Ok(String::from_utf8(output).expect("UTF-8 validated"))
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut result = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &byte in text.as_bytes() {
        let common_safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~');
        let component_safe = match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        };
        if common_safe || component_safe {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    result
}

        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

fn parse_port(text: &str, start: usize) -> Result<u16, ParseError> {

        return Err(ParseError::InvalidPort { at: start });
    }
    for (idx, byte) in text.bytes().enumerate() {

    }
    let mut number = 0u32;
    for byte in text.bytes() {
        number = number.saturating_mul(10).saturating_add((byte - b'0') as u32);
        if number > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if number == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(number as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    let raw = input.as_bytes();
    for (offset, byte) in raw.iter().copied().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at: offset });
        }
    }

    let colon = match raw.iter().position(|&byte| byte == b':') {
        Some(value) => value,
        None => return Err(ParseError::InvalidScheme { at: raw.len() }),
    };
    if colon == 0 || !raw[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for index in 1..colon {
        let byte = raw[index];
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: index });
        }
    }

    let authority_marker = colon + 1;
    if raw.get(authority_marker) != Some(&b'/')
        || raw.get(authority_marker + 1) != Some(&b'/')
    {
        return Err(ParseError::MissingAuthority { at: authority_marker });
    }

    let authority_start = authority_marker + 2;
    let authority_end = raw[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|relative| authority_start + relative)
        .unwrap_or(raw.len());
    let authority = &input[authority_start..authority_end];

    let mut first_at = None;
    for (index, byte) in authority.bytes().enumerate() {
        if byte == b'@' {
            if first_at.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: authority_start + index,
                });
            }
            first_at = Some(index);
        }
    }
    let (userinfo, host_offset) = match first_at {
        Some(index) if index == 0 => {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + index,
            })
        }
        Some(index) => (Some(authority[..index].to_owned()), index + 1),
        None => (None, 0),
    };

    let host_port = &authority[host_offset..];
    let host_port_offset = authority_start + host_offset;
    let (host, port, ipv6) = if host_port.starts_with('[') {
        let close = match host_port.find(']') {
            Some(index) => index,
            None => return Err(ParseError::InvalidIpv6 { at: host_port_offset }),
        };
        let inside = &host_port[1..close];
        if inside.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_port_offset });
        }
        let suffix = &host_port[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                host_port_offset + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_port_offset + close + 1,
            });
        };
        (inside.to_ascii_lowercase(), port, true)
    } else {
        let separator = host_port.find(':');
        let (host_text, port_text, port_start) = if let Some(index) = separator {
            (
                &host_port[..index],
                Some(&host_port[index + 1..]),
                host_port_offset + index + 1,
            )
        } else {
            (host_port, None, host_port_offset + host_port.len())
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_port_offset });
        }
        for (index, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_port_offset + index,
                });
            }
        }
        let port = match port_text {
            Some(text) => Some(parse_port(text, port_start)?),
            None => None,
        };
        (host_text.to_ascii_lowercase(), port, false)
    };

    let tail = &input[authority_end..];
    let fragment_index = tail.find('#');
    let fragment = fragment_index.map(|index| tail[index + 1..].to_owned());
    let before_fragment = &tail[..fragment_index.unwrap_or(tail.len())];
    let query_index = before_fragment.find('?');
    let query = query_index.map(|index| before_fragment[index + 1..].to_owned());
    let path_raw = &before_fragment[..query_index.unwrap_or(before_fragment.len())];
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    if let Some(ref value) = userinfo {
        validate_component(value, authority_start)?;
    }
    if !path_raw.is_empty() {
        validate_component(path_raw, authority_end)?;
    }
    if let Some(index) = query_index {
        validate_component(
            &before_fragment[index + 1..],
            authority_end + index + 1,
        )?;
    }
    if let Some(index) = fragment_index {
        validate_component(&tail[index + 1..], authority_end + index + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut source = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = match hex_value(bytes[i + 1]) {
                Some(v) => v,
                None => return Err(CodecError::BadEscape { at: i }),
            };
            let lo = match hex_value(bytes[i + 2]) {
                Some(v) => v,
                None => return Err(CodecError::BadEscape { at: i }),
            };
            decoded.push((hi << 4) | lo);
            source.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            source.push(i);
            i += 1;
        }
    }

    if let Err(e) = std::str::from_utf8(&decoded) {
        let at = source.get(e.valid_up_to()).copied().unwrap_or(input.len());
        return Err(CodecError::InvalidUtf8 { at });
    }
    Ok(String::from_utf8(decoded).expect("UTF-8 was checked"))
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut result = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };

        if safe {
            result.push(char::from(b));
        } else {
            result.push('%');
            result.push(char::from(HEX[(b >> 4) as usize]));
            result.push(char::from(HEX[(b & 15) as usize]));
        }
    }
    result
}

fn port_value(text: &str, absolute_start: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }
    if let Some(i) = text.bytes().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort {
            at: absolute_start + i,
        });
    }
    let bytes = text.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }

    let mut n: u32 = 0;
    for &b in bytes {
        n = n * 10 + u32::from(b - b'0');
        if n > 65535 {
            return Err(ParseError::InvalidPort { at: absolute_start });
        }
    }
    if n == 0 {
        return Err(ParseError::InvalidPort { at: absolute_start });
    }
    Ok(n as u16)
}

fn check_component(raw: &str, absolute_start: usize) -> Result<(), ParseError> {
    match decode_component(Component::Path, raw) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape {
            at: absolute_start + at,
        }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 {
            at: absolute_start + at,
        }),
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    let b = input.as_bytes();
    for (i, &c) in b.iter().enumerate() {
        if !(0x21..=0x7e).contains(&c) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let scheme_end = match b.iter().position(|&c| c == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if scheme_end == 0 || !b[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..scheme_end {
        if !(b[i].is_ascii_alphanumeric() || matches!(b[i], b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    let after_colon = scheme_end + 1;
    if b.get(after_colon) != Some(&b'/') || b.get(after_colon + 1) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: after_colon });
    }

    let auth_start = after_colon + 2;
    let auth_end = b[auth_start..]
        .iter()
        .position(|&c| matches!(c, b'/' | b'?' | b'#'))
        .map(|n| auth_start + n)
        .unwrap_or(b.len());

    let authority = &input[auth_start..auth_end];
    let mut first_at = None;
    let mut second_at = None;
    for (i, c) in authority.bytes().enumerate() {
        if c == b'@' {
            if first_at.is_some() {
                second_at = Some(i);
                break;
            }
            first_at = Some(i);
        }
    }
    if let Some(i) = second_at {
        return Err(ParseError::InvalidUserInfo { at: auth_start + i });
    }

    let (userinfo, hp_start) = match first_at {
        Some(0) => return Err(ParseError::InvalidUserInfo { at: auth_start }),
        Some(i) => (Some(authority[..i].to_owned()), i + 1),
        None => (None, 0),
    };

    let hp = &authority[hp_start..];
    let hp_abs = auth_start + hp_start;
    let (host, port, ipv6) = if hp.starts_with('[') {
        let close = match hp.bytes().position(|c| c == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: hp_abs }),
        };
        let inside = &hp[1..close];
        if Ipv6Addr::from_str(inside).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hp_abs });
        }
        let suffix = &hp[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if let Some(rest) = suffix.strip_prefix(':') {
            Some(port_value(rest, hp_abs + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hp_abs + close + 1,
            });
        };
        (inside.to_ascii_lowercase(), port, true)
    } else {
        let (host_text, port_text, port_offset) = match hp.find(':') {
            Some(i) => (&hp[..i], Some(&hp[i + 1..]), i + 1),
            None => (hp, None, hp.len()),
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: hp_abs });
        }
        if let Some(i) = host_text.bytes().position(|c| {
            !(c.is_ascii_alphanumeric() || matches!(c, b'.' | b'-' | b'_'))
        }) {
            return Err(ParseError::InvalidHost { at: hp_abs + i });
        }
        let port = match port_text {
            Some(text) => Some(port_value(text, hp_abs + port_offset)?),
            None => None,
        };
        (host_text.to_ascii_lowercase(), port, false)
    };

    let fragment_mark = input[auth_end..].find('#').map(|i| auth_end + i);
    let content_end = fragment_mark.unwrap_or(input.len());
    let query_mark = input[auth_end..content_end]
        .find('?')
        .map(|i| auth_end + i);

    let path_end = query_mark.unwrap_or(content_end);
    let path = if path_end == auth_end {
        "/".to_owned()
    } else {
        input[auth_end..path_end].to_owned()
    };
    let query = query_mark.map(|q| input[q + 1..content_end].to_owned());
    let fragment = fragment_mark.map(|f| input[f + 1..].to_owned());

    if let Some(ref user) = userinfo {
        check_component(user, auth_start)?;
    }
    if path_end > auth_end {
        check_component(&input[auth_end..path_end], auth_end)?;
    }
    if let Some(q) = query_mark {
        check_component(&input[q + 1..content_end], q + 1)?;
    }
    if let Some(f) = fragment_mark {
        check_component(&input[f + 1..], f + 1)?;
    }

    Ok(Url {
        scheme: input[..scheme_end].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {

pub fn encode_component(component: Component, text: &str) -> String {

    for &b in text.as_bytes() {
        let safe = matches!(b, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),

    }

}

    if let Some(f) = fragment_rel {
        validate_component(fragment.as_deref().unwrap_or(""), f + 1)?;
    }

    let _ = host_end_offset;
    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort {

        }
    }

        value = value.saturating_mul(10).saturating_add((b - b'0') as u32);
        if value > u16::MAX as u32 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    decoded.push((h << 4) | l);
                    origins.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let common = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let extra = match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        };
        if common || extra {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn map_codec_error(error: CodecError, offset: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    for (i, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..]
        .iter()
        .position(|b| matches!(*b, b'/' | b'?' | b'#'))
        .map(|i| auth_start + i)
        .unwrap_or(input.len());
    let authority = &input[auth_start..auth_end];
    let auth_bytes = authority.as_bytes();

    let at_positions: Vec<usize> = auth_bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'@').then_some(i))
        .collect();
    if at_positions.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + at_positions[1],
        });
    }
    let (userinfo, host_start_rel) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };
    let host_start = auth_start + host_start_rel;
    let host_port = &authority[host_start_rel..];

    let (host, port, ipv6) = if host_port.starts_with('[') {
        let open = host_start;
        let close_rel = match host_port.as_bytes()[1..].iter().position(|&b| b == b']') {
            Some(i) => i + 1,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &host_port[1..close_rel];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        let suffix = &host_port[close_rel + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            parse_port(&suffix[1..], host_start + close_rel + 2)?
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close_rel + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = host_port.as_bytes().iter().position(|&b| b == b':');
        let (host_text, port) = if let Some(i) = split {
            (&host_port[..i], parse_port(&host_port[i + 1..], host_start + i + 1)?)
        } else {
            (host_port, None)
        };
        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        if let Some(i) = host_text
            .as_bytes()
            .iter()
            .position(|b| !(b.is_ascii_alphanumeric() || matches!(*b, b'.' | b'-' | b'_')))
        {
            return Err(ParseError::InvalidHost { at: host_start + i });
        }
        (host_text.to_ascii_lowercase(), port, false)
    };

    let component_start = auth_end;
    let fragment_marker = input[component_start..].find('#').map(|i| component_start + i);
    let before_fragment_end = fragment_marker.unwrap_or(input.len());
    let query_marker = input[component_start..before_fragment_end]
        .find('?')
        .map(|i| component_start + i);

    let path_end = query_marker.unwrap_or(before_fragment_end);
    let path = if path_end == component_start {
        "/".to_owned()
    } else {
        input[component_start..path_end].to_owned()
    };
    let query = query_marker.map(|q| input[q + 1..before_fragment_end].to_owned());
    let fragment = fragment_marker.map(|f| input[f + 1..].to_owned());

    if let Some(ref ui) = userinfo {
        let ui_offset = auth_start;
        validate_component(ui, ui_offset)?;
    }
    if path_end != component_start {
        validate_component(&path, component_start)?;
    }
    if let Some(q) = query_marker {
        validate_component(query.as_deref().unwrap_or(""), q + 1)?;
    }
    if let Some(f) = fragment_marker {
        validate_component(fragment.as_deref().unwrap_or(""), f + 1)?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(input: &str, port_start: usize) -> Result<u16, ParseError> {
    let b = input.as_bytes();
    if port_start == b.len() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut end = port_start;
    while end < b.len() && !matches!(b[end], b'/' | b'?' | b'#') {
        end += 1;
    }
    let digits = &b[port_start..end];

    if let Some(offset) = digits.iter().position(|c| !c.is_ascii_digit()) {
        return Err(ParseError::InvalidPort {
            at: port_start + offset,
        });
    }
    if digits.len() >= 2 && digits[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for &d in digits {
        value = value.saturating_mul(10).saturating_add((d - b'0') as u32);
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

fn parse_port_slice(s: &str, port_start: usize) -> Result<u16, ParseError> {

        return Err(ParseError::InvalidPort {
            at: port_start + offset,
        });

pub fn parse_url(input: &str) -> Result<Url, ParseError> {

        return Err(ParseError::Empty);
    }

        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

        return Err(ParseError::InvalidScheme { at: 0 });
    }

        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }

        || bytes[colon + 2] != b'/'

            at: authority_start + ats[1],

        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });

        (Some(authority[..at].to_owned()), at + 1)
    } else {
        (None, 0)
    };

    let host_absolute = authority_start + hostport_start;

                Some(&hostport[i + 1..]),
                host_absolute + i + 1,

            None => (hostport, None, 0),
        };

    let fragment_relative = after_authority.find('#');
    let before_fragment_end = fragment_relative
        .map(|i| authority_end + i)
        .unwrap_or(input.len());
    let fragment = fragment_relative.map(|i| input[authority_end + i + 1..].to_owned());

    let query_relative = input[authority_end..before_fragment_end].find('?');
    let before_query_end = query_relative
        .map(|i| authority_end + i)
        .unwrap_or(before_fragment_end);
    let query = query_relative.map(|i| {
        input[authority_end + i + 1..before_fragment_end].to_owned()
    });

    let path = if before_query_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..before_query_end].to_owned()
    };

    let mut url = Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    };

    if let Some(ref s) = url.userinfo {
        let start = authority_start;
        validate_component(input, start, s)?;
    }
    if before_query_end > authority_end {
        validate_component(input, authority_end, &url.path)?;
    }
    if let Some(q) = query_relative {
        validate_component(
            input,
            authority_end + q + 1,
            url.query.as_deref().unwrap_or(""),
        )?;
    }
    if let Some(f) = fragment_relative {
        validate_component(
            input,
            authority_end + f + 1,
            url.fragment.as_deref().unwrap_or(""),
        )?;
    }

    Ok(url)
}

    let source = input.as_bytes();
    let mut output = Vec::with_capacity(source.len());
    let mut source_offsets = Vec::with_capacity(source.len());
    let mut cursor = 0usize;

    while cursor < source.len() {
        let byte = source[cursor];
        if byte != b'%' {
            output.push(byte);
            source_offsets.push(cursor);
            cursor += 1;
            continue;
        }

        if source.len() - cursor < 3 {
            return Err(CodecError::BadEscape { at: cursor });
        }

        let high = hex_value(source[cursor + 1]);
        let low = hex_value(source[cursor + 2]);
        if high.is_none() || low.is_none() {
            return Err(CodecError::BadEscape { at: cursor });
        }
        output.push((high.unwrap() << 4) | low.unwrap());
        source_offsets.push(cursor);
        cursor += 3;
    }

    match String::from_utf8(output) {
        Ok(s) => Ok(s),
        Err(err) => {
            let valid = err.utf8_error().valid_up_to();
            Err(CodecError::InvalidUtf8 {
                at: source_offsets.get(valid).copied().unwrap_or(input.len()),
            })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    fn allowed(component: Component, byte: u8) -> bool {
        byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => byte == b'/' || byte == b'?',
                Component::Fragment => byte == b'/' || byte == b'?' || byte == b'#',
            }
    }

    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut result = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        if allowed(component, byte) {
            result.push(char::from(byte));
        } else {
            result.push('%');
            result.push(char::from(HEX[(byte >> 4) as usize]));
            result.push(char::from(HEX[(byte & 0x0f) as usize]));
        }
    }
    result
}

fn port(text: &str, offset: usize) -> Result<u16, ParseError> {
    let bytes = text.as_bytes();
    if bytes.is_empty() {
        return Err(ParseError::InvalidPort { at: offset });
    }
    for (i, b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: offset + i });
        }
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: offset });
    }

    let mut value = 0u32;
    for &b in bytes {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(n) if n <= 65535 => n,
            _ => return Err(ParseError::InvalidPort { at: offset }),
        };
    }
    if value == 0 {
        Err(ParseError::InvalidPort { at: offset })
    } else {
        Ok(value as u16)
    }
}

fn component_error(error: CodecError, offset: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn check_component(text: &str, offset: usize) -> Result<(), ParseError> {
    match decode_component(Component::UserInfo, text) {
        Ok(_) => Ok(()),
        Err(error) => Err(component_error(error, offset)),
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    let data = input.as_bytes();
    if data.is_empty() {
        return Err(ParseError::Empty);
    }
    for (offset, &byte) in data.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at: offset });
        }
    }

    let scheme_end = data.iter().position(|&byte| byte == b':');
    let scheme_end = match scheme_end {
        Some(pos) => pos,
        None => return Err(ParseError::InvalidScheme { at: data.len() }),
    };
    if scheme_end == 0 || !data[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for pos in 1..scheme_end {
        let byte = data[pos];
        if !byte.is_ascii_alphanumeric() && byte != b'+' && byte != b'-' && byte != b'.' {
            return Err(ParseError::InvalidScheme { at: pos });
        }
    }
    let scheme = input[..scheme_end].to_ascii_lowercase();

    let authority_start = scheme_end + 1;
    if data.get(authority_start) != Some(&b'/')
        || data.get(authority_start + 1) != Some(&b'/')
    {
        return Err(ParseError::MissingAuthority {
            at: authority_start,
        });
    }
    let authority_start = authority_start + 2;
    let mut authority_end = data.len();
    for pos in authority_start..data.len() {
        if data[pos] == b'/' || data[pos] == b'?' || data[pos] == b'#' {
            authority_end = pos;
            break;
        }
    }

    let mut user_at = None;
    for pos in authority_start..authority_end {
        if data[pos] == b'@' {
            if user_at.is_some() {
                return Err(ParseError::InvalidUserInfo { at: pos });
            }
            user_at = Some(pos);
        }
    }
    let (userinfo, host_start) = match user_at {
        Some(pos) if pos == authority_start => {
            return Err(ParseError::InvalidUserInfo { at: pos });
        }
        Some(pos) => (
            Some(input[authority_start..pos].to_owned()),
            pos + 1,
        ),
        None => (None, authority_start),
    };

    let authority_tail = &input[host_start..authority_end];
    let tail_bytes = authority_tail.as_bytes();
    let (host, port_value, ipv6) = if tail_bytes.first() == Some(&b'[') {
        let close = tail_bytes[1..]
            .iter()
            .position(|&byte| byte == b']')
            .map(|pos| pos + 1);
        let close = match close {
            Some(pos) => pos,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &authority_tail[1..close];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }

        let suffix = &authority_tail[close + 1..];
        let port_value = if suffix.is_empty() {
            None
        } else if suffix.as_bytes()[0] == b':' {
            Some(port(&suffix[1..], host_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), port_value, true)
    } else {
        let colon = tail_bytes.iter().position(|&byte| byte == b':');
        let (raw_host, port_value) = match colon {
            Some(pos) => (
                &authority_tail[..pos],
                Some(port(&authority_tail[pos + 1..], host_start + pos + 1)?),
            ),
            None => (authority_tail, None),
        };
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (pos, &byte) in raw_host.as_bytes().iter().enumerate() {
            if !byte.is_ascii_alphanumeric() && !matches!(byte, b'.' | b'-' | b'_') {
                return Err(ParseError::InvalidHost {
                    at: host_start + pos,
                });
            }
        }
        (raw_host.to_ascii_lowercase(), port_value, false)
    };

    let after_authority = authority_end;
    let fragment_pos = (after_authority..data.len()).find(|&pos| data[pos] == b'#');
    let before_fragment = fragment_pos.unwrap_or(data.len());
    let query_pos = (after_authority..before_fragment).find(|&pos| data[pos] == b'?');

    let path_end = query_pos.unwrap_or(before_fragment);
    let path = if path_end == after_authority {
        "/".to_owned()
    } else {
        input[after_authority..path_end].to_owned()
    };
    let query = query_pos.map(|pos| input[pos + 1..before_fragment].to_owned());
    let fragment = fragment_pos.map(|pos| input[pos + 1..].to_owned());

    if let Some(ref value) = userinfo {
        check_component(value, authority_start)?;
    }
    if path_end > after_authority {
        check_component(&path, after_authority)?;
    }
    if let Some(pos) = query_pos {
        check_component(query.as_deref().unwrap_or(""), pos + 1)?;
    }
    if let Some(pos) = fragment_pos {
        check_component(fragment.as_deref().unwrap_or(""), pos + 1)?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port: port_value,
        path,
        query,
        fragment,
        ipv6,
    })
}
