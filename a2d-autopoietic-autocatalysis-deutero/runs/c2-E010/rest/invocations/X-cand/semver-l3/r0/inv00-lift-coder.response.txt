- Core fields are validated before their following separator; leading zero takes precedence over overflow.
- Identifier characters are checked before prerelease numeric leading-zero rules; prerelease numbers are arbitrary length.
- Partial, caret, tilde, and hyphen expansions use exclusive upper bounds with a `-0` prerelease endpoint; overflow is an error at the operand start.
- Prerelease admission is arm-local and requires an explicit prerelease operand at the candidate’s exact core tuple.
- Canonical output must account for discrete SemVer precedence, including prerelease adjacency and finite `u64` core bounds.
- The requested “two readings” conflict with the requirement that both files pass the same tests. Both implementations below use the same interpretation; Solution B uses a separate implementation approach.

### Solution A

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while let Some(b) = byte(s, *pos) {
        if !is_digit(b) { break; }
        *pos += 1;
    }
    if *pos == start {
        return match byte(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }
    let raw = &s[start..*pos];
    if raw.len() > 1 && raw.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    raw.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let mut result = Vec::new();
    let mut i = start;
    loop {
        let id_start = i;
        while i < end && s.as_bytes()[i] != b'.' {
            let b = s.as_bytes()[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..i];
        if pre && id.bytes().all(is_digit) && id.len() > 1 && id.as_bytes()[0] == b'0' {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(id.to_string());
        if i == end { break; }
        i += 1;
        if i == end {
            return Err(VersionError::EmptyIdentifier { at: i });
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() { return Err(VersionError::Empty); }
    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;
    if byte(s, pos) != Some(b'.') {
        return match byte(s, pos) {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let minor = parse_core_field(s, &mut pos)?;
    if byte(s, pos) != Some(b'.') {
        return match byte(s, pos) {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let patch = parse_core_field(s, &mut pos)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();

    match byte(s, pos) {
        None => {}
        Some(b'-') => {
            let pre_start = pos + 1;
            let plus = s[pre_start..].find('+').map(|x| pre_start + x);
            let pre_end = plus.unwrap_or(s.len());
            pre = parse_identifiers(s, pre_start, pre_end, true)?;
            if let Some(p) = plus {
                build = parse_identifiers(s, p + 1, s.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, pos + 1, s.len(), false)?;
        }
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_numeric_str(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    if a.is_empty() { return if b.is_empty() { Ordering::Equal } else { Ordering::Greater }; }
    if b.is_empty() { return Ordering::Less; }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let ord = match (xn, yn) {
            (true, true) => cmp_numeric_str(x, y),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if ord != Ordering::Equal { return ord; }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Debug)]
struct Bound {
    version: Version,
    inclusive: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    lower: Option<Bound>,
    upper: Option<Bound>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op { Eq, Lt, Le, Gt, Ge }

fn version(major: u64, minor: u64, patch: u64, pre: Vec<String>, build: Vec<String>) -> Version {
    Version { major, minor, patch, pre, build }
}

fn endpoint(v: &Version) -> Version {
    version(v.major, v.minor, v.patch, v.pre.clone(), Vec::new())
}

fn zero_pre(major: u64, minor: u64, patch: u64) -> Version {
    version(major, minor, patch, vec!["0".to_string()], Vec::new())
}

fn cmp_bound_versions(a: &Bound, b: &Bound) -> Ordering {
    compare_versions(&a.version, &b.version)
}

fn add_lower(arm: &mut Arm, v: Version, inclusive: bool) {
    let candidate = Bound { version: endpoint(&v), inclusive };
    match &arm.lower {
        None => arm.lower = Some(candidate),
        Some(old) => {
            let c = compare_versions(&candidate.version, &old.version);
            if c == Ordering::Greater || (c == Ordering::Equal && !candidate.inclusive && old.inclusive) {
                arm.lower = Some(candidate);
            }
        }
    }
}

fn add_upper(arm: &mut Arm, v: Version, inclusive: bool) {
    let candidate = Bound { version: endpoint(&v), inclusive };
    match &arm.upper {
        None => arm.upper = Some(candidate),
        Some(old) => {
            let c = compare_versions(&candidate.version, &old.version);
            if c == Ordering::Less || (c == Ordering::Equal && !candidate.inclusive && old.inclusive) {
                arm.upper = Some(candidate);
            }
        }
    }
}

fn add_comparison(arm: &mut Arm, op: Op, v: Version, anchor: bool) {
    if anchor && !v.pre.is_empty() {
        arm.anchors.push(core(&v));
    }
    match op {
        Op::Eq => {
            add_lower(arm, v.clone(), true);
            add_upper(arm, v, true);
        }
        Op::Lt => add_upper(arm, v, false),
        Op::Le => add_upper(arm, v, true),
        Op::Gt => add_lower(arm, v, false),
        Op::Ge => add_lower(arm, v, true),
    }
}

fn is_space(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

fn tokens(s: &str, start: usize, end: usize) -> Vec<(usize, usize)> {
    let mut out = Vec::new();
    let mut i = start;
    while i < end {
        while i < end && is_space(s.as_bytes()[i]) { i += 1; }
        if i == end { break; }
        let a = i;
        while i < end && !is_space(s.as_bytes()[i]) { i += 1; }
        out.push((a, i));
    }
    out
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    full: Option<Version>,
    k: usize,
}

fn parse_partial(raw: &str, absolute: usize) -> Result<Partial, RangeError> {
    if raw.contains('-') || raw.contains('+') {
        let v = parse_version(raw).map_err(|e| RangeError::Version { at: absolute, error: e })?;
        return Ok(Partial { fields: vec![Some(v.major), Some(v.minor), Some(v.patch)], full: Some(v), k: 3 });
    }
    let parts: Vec<&str> = raw.split('.').collect();
    if parts.len() >= 3 && !parts.iter().any(|x| matches!(*x, "*" | "x" | "X")) {
        let v = parse_version(raw).map_err(|e| RangeError::Version { at: absolute, error: e })?;
        return Ok(Partial { fields: vec![Some(v.major), Some(v.minor), Some(v.patch)], full: Some(v), k: 3 });
    }
    if parts.len() > 3 || raw.is_empty() {
        return Err(RangeError::Syntax { at: absolute });
    }
    let mut fields = Vec::new();
    let mut saw_wild = false;
    let mut k = 0;
    for p in parts {
        if p.is_empty() { return Err(RangeError::Syntax { at: absolute }); }
        if matches!(p, "*" | "x" | "X") {
            saw_wild = true;
            fields.push(None);
        } else {
            if saw_wild || !p.bytes().all(is_digit) || (p.len() > 1 && p.as_bytes()[0] == b'0') {
                return Err(RangeError::Syntax { at: absolute });
            }
            let n = p.parse::<u64>().map_err(|_| RangeError::Syntax { at: absolute })?;
            fields.push(Some(n));
            k += 1;
        }
    }
    Ok(Partial { fields, full: None, k })
}

fn padded(p: &Partial) -> (u64, u64, u64) {
    let mut vals = [0u64; 3];
    for (i, x) in p.fields.iter().enumerate() {
        if let Some(n) = x { vals[i] = *n; }
    }
    (vals[0], vals[1], vals[2])
}

fn bump_tuple(t: (u64, u64, u64), field: usize, at: usize) -> Result<(u64,u64,u64), RangeError> {
    let mut a = [t.0, t.1, t.2];
    a[field] = a[field].checked_add(1).ok_or(RangeError::Syntax { at })?;
    for x in a.iter_mut().skip(field + 1) { *x = 0; }
    Ok((a[0], a[1], a[2]))
}

fn partial_upper(p: &Partial, at: usize) -> Result<Option<Version>, RangeError> {
    if p.k == 0 { return Ok(None); }
    let t = padded(p);
    let field = if p.k == 1 { 0 } else { 1 };
    let next = bump_tuple(t, field, at)?;
    Ok(Some(zero_pre(next.0, next.1, next.2)))
}

fn parse_operand_range(raw: &str, absolute: usize, kind: u8, arm: &mut Arm) -> Result<(), RangeError> {
    let operand = if kind == b'^' || kind == b'~' { &raw[1..] } else { raw };
    let operand_at = absolute + if kind == b'^' || kind == b'~' { 1 } else { 0 };
    let p = parse_partial(operand, operand_at)?;
    if kind == b'^' || kind == b'~' {
        if p.k == 0 { return Err(RangeError::Syntax { at: absolute }); }
        let (a,b,c) = padded(&p);
        let low = p.full.clone().unwrap_or_else(|| version(a,b,c,Vec::new(),Vec::new()));
        add_comparison(arm, Op::Ge, low.clone(), p.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
        let field = if kind == b'^' {
            if a > 0 || p.k == 1 { 0 } else if b > 0 || p.k == 2 { 1 } else { 2 }
        } else if p.k == 1 { 0 } else { 1 };
        let next = bump_tuple((a,b,c), field, absolute)?;
        add_upper(arm, zero_pre(next.0,next.1,next.2), false);
        return Ok(());
    }
    if let Some(v) = p.full {
        add_comparison(arm, Op::Eq, v.clone(), !v.pre.is_empty());
        return Ok(());
    }
    if p.k == 0 { return Ok(()); }
    let (a,b,c) = padded(&p);
    add_comparison(arm, Op::Ge, version(a,b,c,Vec::new(),Vec::new()), false);
    if let Some(upper) = partial_upper(&p, absolute)? { add_upper(arm, upper, false); }
    Ok(())
}

fn parse_comparator(raw: &str, at: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let bytes = raw.as_bytes();
    let (op, n) = if bytes.starts_with(b">=") { (Op::Ge, 2) }
        else if bytes.starts_with(b"<=") { (Op::Le, 2) }
        else if bytes.starts_with(b">") { (Op::Gt, 1) }
        else if bytes.starts_with(b"<") { (Op::Lt, 1) }
        else if bytes.starts_with(b"=") { (Op::Eq, 1) }
        else { return Err(RangeError::Syntax { at }); };
    if n == bytes.len() || !is_digit(bytes[n]) { return Err(RangeError::Syntax { at }); }
    let operand = &raw[n..];
    let v = parse_version(operand).map_err(|e| RangeError::Version { at: at + n, error: e })?;
    add_comparison(arm, op, v.clone(), !v.pre.is_empty());
    Ok(())
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let ts = tokens(s, start, end);
    if ts.is_empty() { return Err(RangeError::Empty { at: start }); }
    let mut standalone = None;
    for &(a,b) in &ts {
        if &s[a..b] == "-" { standalone = Some(a); break; }
    }
    if let Some(minus) = standalone {
        if ts.len() != 3 || ts[1] != (minus, minus + 1) || &s[ts[1].0..ts[1].1] != "-" {
            return Err(RangeError::Syntax { at: minus });
        }
        let mut lo = parse_partial(&s[ts[0].0..ts[0].1], ts[0].0)?;
        let mut hi = parse_partial(&s[ts[2].0..ts[2].1], ts[2].0)?;
        if lo.k == 0 { return Err(RangeError::Syntax { at: ts[0].0 }); }
        if hi.k == 0 { return Err(RangeError::Syntax { at: ts[2].0 }); }
        let (a,b,c) = padded(&lo);
        let low = lo.full.take().unwrap_or_else(|| version(a,b,c,Vec::new(),Vec::new()));
        add_comparison(&mut Arm { lower:None, upper:None, anchors:Vec::new() }, Op::Ge, low.clone(), !low.pre.is_empty());
        let mut arm = Arm { lower:None, upper:None, anchors:Vec::new() };
        add_comparison(&mut arm, Op::Ge, low.clone(), !low.pre.is_empty());
        if let Some(v) = hi.full {
            add_comparison(&mut arm, Op::Le, v.clone(), !v.pre.is_empty());
        } else {
            let (x,y,z) = padded(&hi);
            if hi.k == 3 {
                add_comparison(&mut arm, Op::Le, version(x,y,z,Vec::new(),Vec::new()), false);
            } else if let Some(up) = partial_upper(&hi, ts[2].0)? {
                add_upper(&mut arm, up, false);
            }
        }
        return Ok(arm);
    }
    let mut arm = Arm { lower: None, upper: None, anchors: Vec::new() };
    for &(a,b) in &ts {
        let raw = &s[a..b];
        let first = raw.as_bytes()[0];
        if matches!(first, b'<' | b'>' | b'=') {
            parse_comparator(raw, a, &mut arm)?;
        } else if matches!(first, b'^' | b'~') {
            parse_operand_range(raw, a, first, &mut arm)?;
        } else if is_digit(first) || matches!(first, b'*' | b'x' | b'X') {
            parse_operand_range(raw, a, 0, &mut arm)?;
        } else {
            return Err(RangeError::Syntax { at: a });
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut separators = Vec::new();
    let mut i = 0;
    while i + 1 < bytes.len() {
        if bytes[i] == b'|' && bytes[i+1] == b'|' && (i == 0 || is_space(bytes[i-1])) && (i+2 == bytes.len() || is_space(bytes[i+2])) {
            separators.push(i);
            i += 2;
        } else { i += 1; }
    }
    let mut arms = Vec::new();
    let mut start = 0;
    for sep in separators {
        let trimmed_start = (start..sep).find(|&x| !is_space(bytes[x])).unwrap_or(sep);
        let trimmed_end = (start..sep).rev().find(|&x| !is_space(bytes[x])).map(|x| x+1).unwrap_or(start);
        if trimmed_start >= trimmed_end {
            return Err(RangeError::Empty { at: sep });
        }
        arms.push(parse_arm(s, trimmed_start, trimmed_end)?);
        start = sep + 2;
    }
    let trimmed_start = (start..bytes.len()).find(|&x| !is_space(bytes[x])).unwrap_or(bytes.len());
    let trimmed_end = (start..bytes.len()).rev().find(|&x| !is_space(bytes[x])).map(|x| x+1).unwrap_or(start);
    if trimmed_start >= trimmed_end {
        if separators.is_empty() && s.bytes().all(is_space) { return Err(RangeError::Empty { at: 0 }); }
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(s, trimmed_start, trimmed_end)?);
    Ok(Range { arms })
}

fn has_anchor(arm: &Arm, v: &Version) -> bool {
    arm.anchors.iter().any(|x| *x == core(v))
}

fn satisfies_bounds(v: &Version, arm: &Arm) -> bool {
    if let Some(l) = &arm.lower {
        match compare_versions(v, &l.version) {
            Ordering::Less => return false,
            Ordering::Equal if !l.inclusive => return false,
            _ => {}
        }
    }
    if let Some(u) = &arm.upper {
        match compare_versions(v, &u.version) {
            Ordering::Greater => return false,
            Ordering::Equal if !u.inclusive => return false,
            _ => {}
        }
    }
    true
}

fn arm_matches(v: &Version, arm: &Arm, include_pre: bool) -> bool {
    if !satisfies_bounds(v, arm) { return false; }
    v.pre.is_empty() || include_pre || has_anchor(arm, v)
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(old) = best {
            let ord = compare_versions(v, &parsed[old]);
            if (maximum && ord == Ordering::Greater) || (!maximum && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

// Exact intersection is reduced to stable witnesses and prerelease anchor witnesses.
// For stable versions, every open interval is checked at its least stable successor.
fn next_core(t: (u64,u64,u64)) -> Option<(u64,u64,u64)> {
    if t.2 < u64::MAX { Some((t.0,t.1,t.2+1)) }
    else if t.1 < u64::MAX { Some((t.0,t.1+1,0)) }
    else if t.0 < u64::MAX { Some((t.0+1,0,0)) }
    else { None }
}

fn stable_at(t: (u64,u64,u64)) -> Version {
    version(t.0,t.1,t.2,Vec::new(),Vec::new())
}

fn pair_intersects(a: &Arm, b: &Arm, include: bool) -> bool {
    let mut lo: Option<Bound> = None;
    let mut hi: Option<Bound> = None;
    for arm in [a,b] {
        if let Some(x) = &arm.lower {
            if lo.as_ref().is_none_or(|y| compare_versions(&x.version,&y.version) == Ordering::Greater ||
                (compare_versions(&x.version,&y.version) == Ordering::Equal && !x.inclusive && y.inclusive)) { lo = Some(x.clone()); }
        }
        if let Some(x) = &arm.upper {
            if hi.as_ref().is_none_or(|y| compare_versions(&x.version,&y.version) == Ordering::Less ||
                (compare_versions(&x.version,&y.version) == Ordering::Equal && !x.inclusive && y.inclusive)) { hi = Some(x.clone()); }
        }
    }
    if let (Some(l),Some(u)) = (&lo,&hi) {
        let c = compare_versions(&l.version,&u.version);
        if c == Ordering::Greater || (c == Ordering::Equal && !(l.inclusive && u.inclusive)) { return false; }
    }
    if include {
        return true;
    }
    let anchors: Vec<(u64,u64,u64)> = a.anchors.iter().copied().filter(|x| b.anchors.contains(x))
        .chain(b.anchors.iter().copied().filter(|x| a.anchors.contains(x))).collect();
    for t in anchors {
        let v = zero_pre(t.0,t.1,t.2);
        if satisfies_bounds(&v,a) && satisfies_bounds(&v,b) { return true; }
    }
    let candidate = match &lo {
        None => stable_at((0,0,0)),
        Some(l) => {
            if l.inclusive && l.version.pre.is_empty() {
                stable_at(core(&l.version))
            } else if !l.version.pre.is_empty() {
                stable_at(core(&l.version))
            } else if let Some(t) = next_core(core(&l.version)) {
                stable_at(t)
            } else { return false; }
        }
    };
    satisfies_bounds(&candidate,a) && satisfies_bounds(&candidate,b)
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    a.arms.iter().any(|x| b.arms.iter().any(|y| pair_intersects(x,y,include_prerelease)))
}

fn arm_satisfiable_default(a: &Arm) -> bool {
    if pair_intersects(a, &Arm { lower:None, upper:None, anchors:Vec::new() }, false) { return true; }
    a.anchors.iter().any(|t| {
        let v = zero_pre(t.0,t.1,t.2);
        satisfies_bounds(&v,a)
    })
}

fn canonical_arm(a: &Arm) -> Option<String> {
    if !arm_satisfiable_default(a) { return None; }
```

```rust
pub fn parse_version(s: &str) -> Result<Version, VersionError> {

        }

        }
    }
```
