fn is_valid_atom_shape(field: Field, s: &str) -> bool {
    if s.is_empty() {
        return false;
    }
    if s.bytes().all(|b| b.is_ascii_digit()) {
        return true;

fn parse_number(
    s: &str,
    field: Field,
    offset: usize,
    step: bool,
) -> Result<u32, ParseError> {
    let b = s.as_bytes();
    if b.len() > 1 && b[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value: u32 = 0;
    for &ch in b {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((ch - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    if step {
        let width = SPECS[field_index(field)].1 - SPECS[field_index(field)].0 + 1;
        if value == 0 || value > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else {
        let (min, max) = SPECS[field_index(field)];
        if value < min || value > max {
            return Err(ParseError::OutOfRange { field, offset });
        }
    }
    Ok(value)
}

fn parse_atom(
    field: Field,
    s: &str,
    offset: usize,
) -> Result<u32, ParseError> {
    if s.bytes().all(|b| b.is_ascii_digit()) {
        parse_number(s, field, offset, false)
    } else {
        name_value(field, s).ok_or(ParseError::InvalidSyntax { field, offset })
    }
}

fn invalid_syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }

fn parse_field(text: &str, field: Field) -> Result<(Vec<bool>, bool), ParseError> {
    for (offset, &b) in text.as_bytes().iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let idx = field_index(field);
    let (min, max) = SPECS[idx];
    let mut selected = vec![false; (max + 1) as usize];
    let unrestricted = matches!(field, Field::DayOfMonth | Field::DayOfWeek) && text == "*";

    let mut start = 0usize;
    for item in text.split(',') {
        let item_offset = start;
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_offset,
            });
        }

        let bytes = item.as_bytes();
        let slash_positions: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();
        if slash_positions.len() > 1 {
            return Err(invalid_syntax(field, item_offset));
        }

        let (base, step_part, step_offset) = if let Some(&p) = slash_positions.first() {
            (&item[..p], Some(&item[p + 1..]), item_offset + p + 1)
        } else {
            (item, None, 0)
        };

        let (base_kind, left, right, left_offset, right_offset) = if base == "*" {
            (0u8, "", "", 0usize, 0usize)
        } else {
            let hyphens: Vec<usize> = base
                .as_bytes()
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
                .collect();
            if hyphens.len() > 1 {
                return Err(invalid_syntax(field, item_offset));
            }
            if let Some(&h) = hyphens.first() {
                let l = &base[..h];
                let r = &base[h + 1..];
                if l == "*" || r == "*" || !is_valid_atom_shape(field, l) || !is_valid_atom_shape(field, r) {
                    return Err(invalid_syntax(field, item_offset));
                }
                (2u8, l, r, item_offset, item_offset + h + 1)
            } else {
                if base.is_empty() || base == "*" || !is_valid_atom_shape(field, base) {
                    return Err(invalid_syntax(field, item_offset));
                }
                (1u8, base, "", item_offset, 0usize)
            }
        };

        if let Some(step_text) = step_part {
            if step_text.is_empty() || !step_text.bytes().all(|b| b.is_ascii_digit()) {
                return Err(invalid_syntax(field, item_offset));
            }
        }

        let mut a = min;
        let mut b = max;
        match base_kind {
            0 => {}
            1 => {
                a = parse_atom(field, left, left_offset)?;
                b = if step_part.is_some() { max } else { a };
            }
            2 => {
                a = parse_atom(field, left, left_offset)?;
                b = parse_atom(field, right, right_offset)?;
                if a > b {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: right_offset,
                    });
                }
            }
            _ => unreachable!(),
        }

        let step = if let Some(step_text) = step_part {
            if !step_text.bytes().all(|b| b.is_ascii_digit()) {
                return Err(invalid_syntax(field, item_offset));
            }
            parse_number(step_text, field, step_offset, true)?
        } else {
            if base_kind == 0 || base_kind == 2 {
                1
            } else {
                0
            }
        };

        if step == 0 {
            selected[a as usize] = true;
        } else {
            let mut value = a;
            while value <= b {
                selected[value as usize] = true;
                match value.checked_add(step) {
                    Some(next) => value = next,
                    None => break,
                }
            }
        }

        start += item.len() + 1;
    }

    Ok((selected, unrestricted))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount {
            found: pieces.len(),
        });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut sets = Vec::with_capacity(5);
    let mut unrestricted_days = [false; 2];
    for (i, (&field, text)) in fields.iter().zip(pieces.iter()).enumerate() {
        let (set, unrestricted) = parse_field(text, field)?;
        sets.push(set);
        if i == 2 {
            unrestricted_days[0] = unrestricted;
        } else if i == 4 {
            unrestricted_days[1] = unrestricted;
        }
    }

    let selected: [Vec<bool>; 5] = sets.try_into().ok().unwrap();
    Ok(Cron {
        selected,
        unrestricted_days,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &v)| if v { Some(i as u8) } else { None })
            .collect()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        self.selected[field_index(field)]
            .get(value as usize)
            .copied()
            .unwrap_or(false)
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.contains(Field::Minute, at.minute)
            || !self.contains(Field::Hour, at.hour)
            || !self.contains(Field::Month, at.month)
        {
            return false;
        }

        let dom = self.contains(Field::DayOfMonth, at.day);
        let dow = self.contains(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match self.unrestricted_days {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if let Some(next) = increment_minute(current) {
                current = next;
            } else {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days: u32 = 0;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn increment_minute(mut at: CivilDateTime) -> Option<CivilDateTime> {
    if at.minute < 59 {
        at.minute += 1;
        return Some(at);
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return Some(at);
    }
    at.hour = 0;
    if at.day < days_in_month(at.year, at.month) {
        at.day += 1;
        return Some(at);
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return Some(at);
    }
    at.month = 1;
    if at.year < 2399 {
        at.year += 1;
        return Some(at);
    }
    None
}

    let mut n = 0u32;
    for &c in b {
        n = n.checked_mul(10).and_then(|x| x.checked_add((c - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field: f, offset: off })?;
    }
    if is_step {
        let width = (MAXS[ix(f)] - MINS[ix(f)] + 1) as u32;
        if n == 0 || n > width { return Err(ParseError::BadStep { field: f, offset: off }); }
    } else if n < MINS[ix(f)] as u32 || n > MAXS[ix(f)] as u32 {
        return Err(ParseError::OutOfRange { field: f, offset: off });
    }
    Ok(n as u8)
}

fn endpoint(f: Field, s: &str, off: usize) -> Result<u8, ParseError> {
    if s.bytes().all(|c| c.is_ascii_digit()) { number(f, s, off, false) }
    else { named(f, s).ok_or_else(|| syntax(f, off)) }
}

fn parse_one(text: &str, f: Field, base_off: usize, bits: &mut u64) -> Result<(), ParseError> {
    if text.is_empty() { return Err(ParseError::EmptyItem { field: f, offset: base_off }); }
    let b = text.as_bytes();
    if b.iter().filter(|&&c| c == b'/').count() > 1 { return Err(syntax(f, base_off)); }
    let slash = b.iter().position(|&c| c == b'/');
    let (base, step_part) = match slash {
        Some(p) if p + 1 < b.len() => (&text[..p], Some((&text[p + 1..], base_off + p + 1))),
        Some(_) => return Err(syntax(f, base_off)),
        None => (text, None),
    };
    if base.is_empty() { return Err(syntax(f, base_off)); }
    let hyphens = base.bytes().filter(|&c| c == b'-').count();
    let star = base == "*";
    let (a_s, a_off, z_s, z_off) = if hyphens == 0 {
        (base, base_off, None, 0)
    } else {
        if star || hyphens != 1 { return Err(syntax(f, base_off)); }
        let p = base.find('-').unwrap();
        let (a,z) = (&base[..p], &base[p+1..]);
        if !atom_ok(f,a) || !atom_ok(f,z) { return Err(syntax(f,base_off)); }
        (a,base_off,Some(z),base_off+p+1)
    };
    if !star && !atom_ok(f,a_s) { return Err(syntax(f,base_off)); }
    let step = if let Some((s,o)) = step_part {
        if s.is_empty() || !s.bytes().all(|c| c.is_ascii_digit()) { return Err(syntax(f,base_off)); }
        Some(number(f,s,o,true)?)
    } else { None };
    let (lo,hi) = if star {
        (MINS[ix(f)], MAXS[ix(f)])
    } else {
        let a = endpoint(f,a_s,a_off)?;
        if let Some(z) = z_s {
            let end = endpoint(f,z,z_off)?;
            if a > end { return Err(ParseError::ReversedRange { field:f, offset:z_off }); }
            (a,end)
        } else if step.is_some() {
            (a,MAXS[ix(f)])
        } else { (a,a) }
    };
    let step = step.unwrap_or(1);
    let mut v = lo;
    loop {
        *bits |= 1u64 << v;
        if v > hi.saturating_sub(step) { break; }
        v += step;
    }
    Ok(())
}

fn parse_field(text: &str, f: Field) -> Result<u64, ParseError> {
    for (i,c) in text.bytes().enumerate() {
        if !(c.is_ascii_digit() || c.is_ascii_alphabetic() || matches!(c,b'*'|b'-'|b'/'|b',')) {
            return Err(ParseError::InvalidCharacter { field:f, offset:i });
        }
    }
    let mut bits = 0u64;
    let mut off = 0usize;
    for item in text.split(',') {
        parse_one(item,f,off,&mut bits)?;
        off += item.len()+1;
    }
    Ok(bits)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let p: Vec<&str> = input.split(' ').collect();
    if p.len()!=5 { return Err(ParseError::FieldCount { found:p.len() }); }
    let mut b=[0u64;5];
    for i in 0..5 { b[i]=parse_field(p[i],FIELDS[i])?; }
    Ok(Cron { bits:b, dom_star:p[2]=="*", dow_star:p[4]=="*" })
}

impl Cron {
    pub fn allowed(&self, f: Field) -> Vec<u8> {
        let i=ix(f);
        (MINS[i]..=MAXS[i]).filter(|&v| self.bits[i]&(1u64<<v)!=0).collect()
    }
    fn has(&self,f:Field,v:u8)->bool { self.bits[ix(f)]&(1u64<<v)!=0 }
    fn valid(a:CivilDateTime)->Result<(),DateError>{
        if !(2000..=2399).contains(&a.year){return Err(DateError::InvalidDateTime{component:DatePart::Year});}
        if !(1..=12).contains(&a.month){return Err(DateError::InvalidDateTime{component:DatePart::Month});}
        if !(1..=month_days(a.year,a.month)).contains(&a.day){return Err(DateError::InvalidDateTime{component:DatePart::Day});}
        if a.hour>23{return Err(DateError::InvalidDateTime{component:DatePart::Hour});}
        if a.minute>59{return Err(DateError::InvalidDateTime{component:DatePart::Minute});}
        Ok(())
    }
    fn day_ok(&self,a:CivilDateTime)->bool{
        let dm=self.has(Field::DayOfMonth,a.day);
        let dw=self.has(Field::DayOfWeek,weekday(a.year,a.month,a.day));
        match (self.dom_star,self.dow_star){
            (true,true)=>true,(true,false)=>dw,(false,true)=>dm,
            (false,false)=>if a.day==month_days(a.year,a.month){dm&&dw}else{dm||dw}
        }
    }
    pub fn matches(&self,a:CivilDateTime)->Result<bool,DateError>{
        Self::valid(a)?;
        Ok(self.has(Field::Minute,a.minute)&&self.has(Field::Hour,a.hour)&&self.has(Field::Month,a.month)&&self.day_ok(a))
    }
    pub fn next_fire(&self,after:CivilDateTime)->Result<Option<CivilDateTime>,DateError>{
        Self::valid(after)?;
        let mut a=after;
        loop{
            if a.minute<59{a.minute+=1;}else{
                a.minute=0;
                if a.hour<23{a.hour+=1;}else{
                    a.hour=0;
                    if a.day<month_days(a.year,a.month){a.day+=1;}else{
                        a.day=1;
                        if a.month<12{a.month+=1;}else{
                            if a.year==2399{return Ok(None);}
                            a.month=1;a.year+=1;
                        }
                    }
                }
            }
            if self.matches(a)?{return Ok(Some(a));}
        }
    }
}

fn leap(y:u16)->bool{y%4==0&&(y%100!=0||y%400==0)}
fn month_days(y:u16,m:u8)->u8{match m{2=>if leap(y){29}else{28},4|6|9|11=>30,_=>31}}
fn weekday(y:u16,m:u8,d:u8)->u8{
    let mut n=0u32;
    for yy in 2000..y{n+=if leap(yy){366}else{365};}
    for mm in 1..m{n+=month_days(y,mm) as u32;}
    n+=(d-1) as u32;
    ((n+6)%7) as u8
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {

const DOMAINS: [(u32, u32); 5] = [(0, 59), (0, 23), (1, 31), (1, 12), (0, 6)];

fn idx(f: Field) -> usize {
    match f {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn syntax(f: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field: f, offset }
}

fn mapped_name(f: Field, s: &str) -> Option<u32> {
    let table: &[&str] = match f {
        Field::Month => &["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    table.iter().position(|n| n.eq_ignore_ascii_case(s)).map(|n| {
        n as u32 + if f == Field::Month { 1 } else { 0 }
    })
}

fn atom_shape(f: Field, s: &str) -> bool {
    !s.is_empty()
        && (s.bytes().all(|b| b.is_ascii_digit()) || mapped_name(f, s).is_some())
}

fn number(s: &str, f: Field, offset: usize, step: bool) -> Result<u32, ParseError> {
    if s.len() > 1 && s.as_bytes()[0] == b'0' {
        return Err(ParseError::LeadingZero { field: f, offset });
    }
    let mut n = 0u32;
    for b in s.bytes() {
        n = n.checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field: f, offset })?;
    }
    if step {
        let (lo, hi) = DOMAINS[idx(f)];
        if n < 1 || n > hi - lo + 1 {
            return Err(ParseError::BadStep { field: f, offset });
        }
    } else {
        let (lo, hi) = DOMAINS[idx(f)];
        if n < lo || n > hi {
            return Err(ParseError::OutOfRange { field: f, offset });
        }
    }
    Ok(n)
}

fn atom(s: &str, f: Field, offset: usize) -> Result<u32, ParseError> {
    if s.bytes().all(|b| b.is_ascii_digit()) {
        number(s, f, offset, false)
    } else {
        mapped_name(f, s).ok_or_else(|| syntax(f, offset))
    }
}

fn field_set(text: &str, f: Field) -> Result<(u64, bool), ParseError> {
    for (i, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit() || b.is_ascii_alphabetic() || b"*-/, ".contains(&b) && b != b' ') {
            return Err(ParseError::InvalidCharacter { field: f, offset: i });
        }
    }

    let (lo, hi) = DOMAINS[idx(f)];
    let mut bits = 0u64;
    let unrestricted = (f == Field::DayOfMonth || f == Field::DayOfWeek) && text == "*";
    let mut base_offset = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field: f, offset: base_offset });
        }
        let slash: Vec<usize> = item.bytes().enumerate().filter_map(|(i, b)| (b == b'/').then_some(i)).collect();
        if slash.len() > 1 {
            return Err(syntax(f, base_offset));
        }
        let (base, step_text, step_offset) = match slash.first() {
            Some(&p) => (&item[..p], Some(&item[p + 1..]), base_offset + p + 1),
            None => (item, None, 0),
        };

        if let Some(st) = step_text {
            if st.is_empty() || !st.bytes().all(|b| b.is_ascii_digit()) {
                return Err(syntax(f, base_offset));
            }
        }

        let hyphen: Vec<usize> = base.bytes().enumerate().filter_map(|(i, b)| (b == b'-').then_some(i)).collect();
        let (kind, atext, btext, ao, bo) = if base == "*" {
            (0, "", "", 0, 0)
        } else if hyphen.len() == 1 {
            let p = hyphen[0];
            let a = &base[..p];
            let b = &base[p + 1..];
            if !atom_shape(f, a) || !atom_shape(f, b) {
                return Err(syntax(f, base_offset));
            }
            (2, a, b, base_offset, base_offset + p + 1)
        } else if hyphen.is_empty() && atom_shape(f, base) {
            (1, base, "", base_offset, 0)
        } else {
            return Err(syntax(f, base_offset));
        };

        let (mut start, mut end) = match kind {
            0 => (lo, hi),
            1 => {
                let v = atom(atext, f, ao)?;
                (v, if step_text.is_some() { hi } else { v })
            }
            _ => {
                let a = atom(atext, f, ao)?;
                let b = atom(btext, f, bo)?;
                if a > b {
                    return Err(ParseError::ReversedRange { field: f, offset: bo });
                }
                (a, b)
            }
        };
        let step = match step_text {
            Some(st) => number(st, f, step_offset, true)?,
            None if kind == 1 => 0,
            None => 1,
        };
        if step == 0 {
            bits |= 1u64 << start;
        } else {
            while start <= end {
                bits |= 1u64 << start;
                match start.checked_add(step) {
                    Some(n) => start = n,
                    None => break,
                }
            }
        }
        base_offset += item.len() + 1;
        let _ = &mut end;
    }
    Ok((bits, unrestricted))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let fields = [Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek];
    let mut sets = [0u64; 5];
    let mut unrestricted = [false; 2];
    for i in 0..5 {
        let (set, u) = field_set(parts[i], fields[i])?;
        sets[i] = set;
        if i == 2 { unrestricted[0] = u; }
        if i == 4 { unrestricted[1] = u; }
    }
    Ok(Cron { sets, unrestricted })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (lo, hi) = DOMAINS[idx(field)];
        (lo..=hi)
            .filter(|&n| self.sets[idx(field)] & (1u64 << n) != 0)
            .map(|n| n as u8)
            .collect()
    }

    fn has(&self, f: Field, n: u8) -> bool {
        self.sets[idx(f)] & (1u64 << n) != 0
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.has(Field::Minute, at.minute)
            || !self.has(Field::Hour, at.hour)
            || !self.has(Field::Month, at.month)
        {
            return false;
        }
        let dom = self.has(Field::DayOfMonth, at.day);
        let dow = self.has(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match self.unrestricted {
            [true, true] => true,
            [true, false] => dow,
            [false, true] => dom,
            [false, false] if at.day == month_days(at.year, at.month) => dom && dow,
            [false, false] => dom || dow,
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut x = after;
        loop {
            if x.minute < 59 {
                x.minute += 1;
            } else {
                x.minute = 0;
                if x.hour < 23 {
                    x.hour += 1;
                } else {
                    x.hour = 0;
                    if x.day < month_days(x.year, x.month) {
                        x.day += 1;
                    } else {
                        x.day = 1;
                        if x.month < 12 {
                            x.month += 1;
                        } else {
                            x.month = 1;
                            if x.year == 2399 {
                                return Ok(None);
                            }
                            x.year += 1;
                        }
                    }
                }
            }
            if self.matches_valid(x) {
                return Ok(Some(x));
            }
        }
    }
}

fn leap(y: u16) -> bool {
    y % 4 == 0 && (y % 100 != 0 || y % 400 == 0)
}

fn month_days(y: u16, m: u8) -> u8 {
    match m {
        2 => if leap(y) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(x: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&x.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&x.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if x.day == 0 || x.day > month_days(x.year, x.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if x.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if x.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(y: u16, m: u8, d: u8) -> u8 {
    let mut n = 0u32;
    for year in 2000..y {
        n += if leap(year) { 366 } else { 365 };
    }
    for month in 1..m {
        n += month_days(y, month) as u32;
    }
    ((6 + n + d as u32 - 1) % 7) as u8
}

    match k {
        0=>ParseError::InvalidSyntax{field:f,offset:o},
        1=>ParseError::LeadingZero{field:f,offset:o},
        2=>ParseError::NumberOverflow{field:f,offset:o},
        3=>ParseError::OutOfRange{field:f,offset:o},
        4=>ParseError::BadStep{field:f,offset:o},
        _=>ParseError::ReversedRange{field:f,offset:o},

fn numeric(s:&str,f:Field,o:usize,step:bool)->Result<u32,ParseError>{
    if s.len()>1 && s.as_bytes()[0]==b'0' { return Err(fail(f,o,1)); }
    let mut n=0u32;
    for b in s.bytes() {

    else if n<lo || n>hi { return Err(fail(f,o,3)); }
    Ok(n)
}
fn named(s:&str,f:Field)->Option<u32>{

    match f {

    else if let Some(v)=named(s,f) { Ok(v) }

fn parse_one(item:&str,baseoff:usize,f:Field,out:&mut BTreeSet<u8>)->Result<(),ParseError>{
    let mut sl=item.split('/');
    let base=sl.next().unwrap_or("");
    let stepstr=sl.next();
    if sl.next().is_some() || base.is_empty() || stepstr==Some("") { return Err(fail(f,baseoff,0)); }
    let (a,b,star,range)=if base=="*" {(String::new(),String::new(),true,false)}
    else {
        let mut hy=base.split('-');
        let x=hy.next().unwrap_or("");
        let y=hy.next();
        if hy.next().is_some() || x.is_empty() || y==Some("") { return Err(fail(f,baseoff,0)); }
        (x.to_string(),y.unwrap_or(x).to_string(),false,y.is_some())
    };
    let (start,end)=if star { let (lo,hi)=domain(f); (lo,hi) }
    else {
        let av=atom(&a,f,baseoff)?;
        let bv=if range { atom(&b,f,baseoff+a.len()+1)? } else { av };
        if range && av>bv { return Err(fail(f,baseoff+a.len()+1,5)); }
        (av,bv)
    };
    let step=if let Some(s)=stepstr {
        if !s.bytes().all(|b|b.is_ascii_digit()) { return Err(fail(f,baseoff,0)); }
        Some(numeric(s,f,baseoff+base.len()+1,true)?)
    } else { None };
    let end=if !star && !range && step.is_some(){domain(f).1}else{end};
    let inc=step.unwrap_or(1);
    let mut v=start;
    while v<=end { out.insert(v as u8); v+=inc; }
    Ok(())
}
fn parse_field(s:&str,f:Field)->Result<BTreeSet<u8>,ParseError>{
    for (i,b) in s.bytes().enumerate() {
        if !(b.is_ascii_digit()||b.is_ascii_alphabetic()||matches!(b,b'*'|b'-'|b'/'|b',')) {
            return Err(ParseError::InvalidCharacter{field:f,offset:i});
        }
    }
    let mut out=BTreeSet::new();
    let mut off=0;

    DayOfWeek,

pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {

    unrestricted_dom: bool,
    unrestricted_dow: bool,

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Clone, Copy)]
struct Spec {
    field: Field,
    min: u8,
    max: u8,
}

const SPECS: [Spec; 5] = [
    Spec { field: Field::Minute, min: 0, max: 59 },
    Spec { field: Field::Hour, min: 0, max: 23 },
    Spec { field: Field::DayOfMonth, min: 1, max: 31 },
    Spec { field: Field::Month, min: 1, max: 12 },
    Spec { field: Field::DayOfWeek, min: 0, max: 6 },
];

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn err(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn name_value(spec: Spec, s: &str) -> Option<u8> {
    if spec.field == Field::Month {
        const N: [&str; 12] = [
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC",
        ];
        N.iter().position(|n| n.eq_ignore_ascii_case(s)).map(|i| (i + 1) as u8)
    } else if spec.field == Field::DayOfWeek {
        const N: [&str; 7] = ["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"];
        N.iter().position(|n| n.eq_ignore_ascii_case(s)).map(|i| i as u8)
    } else {
        None
    }
}

fn base_atom_kind(spec: Spec, s: &str) -> bool {
    !s.is_empty()
        && (s.bytes().all(|b| b.is_ascii_digit()) || name_value(spec, s).is_some())
}

fn parse_num(spec: Spec, s: &str, offset: usize, step: bool) -> Result<u8, ParseError> {
    let b = s.as_bytes();
    if b.len() > 1 && b[0] == b'0' {
        return Err(err(spec.field, offset, 1));
    }
    let mut n: u32 = 0;
    for &c in b {
        n = n.checked_mul(10)
            .and_then(|v| v.checked_add((c - b'0') as u32))
            .ok_or_else(|| err(spec.field, offset, 2))?;
    }
    if step {
        let width = (spec.max - spec.min + 1) as u32;
        if n < 1 || n > width {
            return Err(err(spec.field, offset, 5));
        }
    } else if n < spec.min as u32 || n > spec.max as u32 {
        return Err(err(spec.field, offset, 3));
    }
    Ok(n as u8)
}

fn parse_endpoint(spec: Spec, s: &str, offset: usize) -> Result<u8, ParseError> {
    if s.bytes().all(|b| b.is_ascii_digit()) {
        parse_num(spec, s, offset, false)
    } else {
        name_value(spec, s).ok_or_else(|| err(spec.field, offset, 0))
    }
}

fn parse_field(text: &str, spec: Spec) -> Result<Vec<u8>, ParseError> {
    for (i, b) in text.bytes().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field: spec.field, offset: i });
        }
    }

    let mut selected = vec![false; (spec.max - spec.min + 1) as usize];
    let mut start = 0usize;
    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem { field: spec.field, offset: start });
        }
        let bytes = item.as_bytes();
        let slash_count = bytes.iter().filter(|&&b| b == b'/').count();
        let hyphen_count = bytes.iter().filter(|&&b| b == b'-').count();

        if slash_count > 1 {
            return Err(err(spec.field, start, 0));
        }
        let (base, step_text, step_offset) = if let Some(pos) = bytes.iter().position(|&b| b == b'/') {
            (&item[..pos], Some(&item[pos + 1..]), start + pos + 1)
        } else {
            (item, None, 0)
        };
        if base.is_empty() || step_text == Some("") {
            return Err(err(spec.field, start, 0));
        }

        let is_star = base == "*";
        let (left, right) = if hyphen_count == 0 {
            (base, None)
        } else {
            if is_star || hyphen_count != 1 {
                return Err(err(spec.field, start, 0));
            }
            let p = base.find('-').unwrap();
            let l = &base[..p];
            let r = &base[p + 1..];
            if !base_atom_kind(spec, l) || !base_atom_kind(spec, r) {
                return Err(err(spec.field, start, 0));
            }
            (l, Some((r, p + 1)))
        };

        if !is_star && !base_atom_kind(spec, left) {
            return Err(err(spec.field, start, 0));
        }
        if let Some((r, _)) = right {
            if !base_atom_kind(spec, r) {
                return Err(err(spec.field, start, 0));
            }
        }

        let step = if let Some(st) = step_text {
            if !st.bytes().all(|b| b.is_ascii_digit()) || st.is_empty() {
                return Err(err(spec.field, start, 0));
            }
            Some(parse_num(spec, st, step_offset, true)?)
        } else {
            None
        };

        let (lo, hi) = if is_star {
            (spec.min, spec.max)
        } else {
            let left_offset = if right.is_some() { start } else { start };
            let a = parse_endpoint(spec, left, left_offset)?;
            if let Some((r, rel)) = right {
                let b = parse_endpoint(spec, r, start + rel)?;
                if a > b {
                    return Err(err(spec.field, start + rel, 4));
                }
                (a, b)
            } else if step.is_some() {
                (a, spec.max)
            } else {
                (a, a)
            }
        };

        let stride = step.unwrap_or(1);
        let mut v = lo;
        loop {
            selected[(v - spec.min) as usize] = true;
            if v > hi.saturating_sub(stride) {
                break;
            }
            v += stride;
        }
        start += item.len() + 1;
    }
    Ok(selected.iter().enumerate().filter_map(|(i, &yes)| {
        if yes { Some(spec.min + i as u8) } else { None }
    }).collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let mut all = Vec::with_capacity(5);
    for (i, text) in parts.iter().enumerate() {
        all.push(parse_field(text, SPECS[i])?);
    }
    Ok(Cron {
        values: [all.remove(0), all.remove(0), all.remove(0), all.remove(0), all.remove(0)],
        unrestricted_dom: parts[2] == "*",
        unrestricted_dow: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.values[field_index(field)].clone()
    }

    fn contains(&self, field: Field, value: u8) -> bool {
        self.values[field_index(field)].binary_search(&value).is_ok()
    }

    fn valid(at: CivilDateTime) -> Result<(), DateError> {
        if !(2000..=2399).contains(&at.year) {
            return Err(DateError::InvalidDateTime { component: DatePart::Year });
        }
        if !(1..=12).contains(&at.month) {
            return Err(DateError::InvalidDateTime { component: DatePart::Month });
        }
        if !(1..=days_in_month(at.year, at.month)).contains(&at.day) {
            return Err(DateError::InvalidDateTime { component: DatePart::Day });
        }
        if at.hour > 23 {
            return Err(DateError::InvalidDateTime { component: DatePart::Hour });
        }
        if at.minute > 59 {
            return Err(DateError::InvalidDateTime { component: DatePart::Minute });
        }
        Ok(())
    }

    fn day_matches(&self, at: CivilDateTime) -> bool {
        let dom = self.contains(Field::DayOfMonth, at.day);
        let dow = self.contains(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        Self::valid(at)?;
        Ok(self.contains(Field::Minute, at.minute)
            && self.contains(Field::Hour, at.hour)
            && self.contains(Field::Month, at.month)
            && self.day_matches(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        Self::valid(after)?;
        let mut cur = after;
        loop {
            if cur.minute < 59 {
                cur.minute += 1;
            } else {
                cur.minute = 0;
                if cur.hour < 23 {
                    cur.hour += 1;
                } else {
                    cur.hour = 0;
                    if cur.day < days_in_month(cur.year, cur.month) {
                        cur.day += 1;
                    } else {
                        cur.day = 1;
                        if cur.month < 12 {
                            cur.month += 1;
                        } else {
                            cur.month = 1;
                            if cur.year == 2399 {
                                return Ok(None);
                            }
                            cur.year += 1;
                        }
                    }
                }
            }
            if self.matches(cur)? {
                return Ok(Some(cur));
            }
        }
    }
}

fn leap(y: u16) -> bool {
    y % 4 == 0 && (y % 100 != 0 || y % 400 == 0)
}

fn days_in_month(y: u16, m: u8) -> u8 {
    match m {
        2 => if leap(y) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(y: u16, m: u8, d: u8) -> u8 {
    let mut days = 0u32;
    for year in 2000..y {
        days += if leap(year) { 366 } else { 365 };
    }
    for month in 1..m {
        days += days_in_month(y, month) as u32;
    }
    days += (d - 1) as u32;
    ((days + 6) % 7) as u8
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn limits(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn syntax(field: Field, offset: usize) -> ParseError {
    ParseError::InvalidSyntax { field, offset }
}

fn parse_number(bytes: &[u8], field: Field, offset: usize, step: bool) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(syntax(field, offset));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut n = 0u32;
    for &b in bytes {
        n = n
            .checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }

    let (min, max) = limits(field);
    if step {
        let width = max - min + 1;
        if n == 0 || n > width {
            return Err(ParseError::BadStep { field, offset });
        }
    } else if n < min || n > max {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(n)
}

fn name_value(field: Field, bytes: &[u8]) -> Option<u32> {
    let names: &[&str] = match field {
        Field::Month => &[
            "JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV",
            "DEC",
        ],
        Field::DayOfWeek => &["SUN", "MON", "TUE", "WED", "THU", "FRI", "SAT"],
        _ => return None,
    };
    let upper: Vec<u8> = bytes.iter().map(u8::to_ascii_uppercase).collect();
    names
        .iter()
        .position(|n| n.as_bytes() == upper.as_slice())
        .map(|i| if field == Field::Month { (i + 1) as u32 } else { i as u32 })
}

fn parse_endpoint(
    field: Field,
    bytes: &[u8],
    offset: usize,
) -> Result<u32, ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) && !bytes.is_empty() {
        parse_number(bytes, field, offset, false)
    } else {
        name_value(field, bytes).ok_or_else(|| syntax(field, offset))
    }
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max) = limits(field);
    let mut selected = vec![false; (max + 1) as usize];

    let mut item_start = 0usize;
    for item in text.split(',') {
        let item_bytes = item.as_bytes();
        if item_bytes.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let slash_positions: Vec<usize> = item_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'/' { Some(i) } else { None })
            .collect();
        if slash_positions.len() > 1 {
            return Err(syntax(field, item_start));
        }
        let (base, step_bytes, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item_bytes[..slash], Some(&item_bytes[slash + 1..]), item_start + slash + 1)
        } else {
            (item_bytes, None, item_start)
        };

        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b'-' { Some(i) } else { None })
            .collect();

        let (kind, a_bytes, a_offset, b_bytes, b_offset) = if base == b"*" {
            (0u8, &[][..], item_start, &[][..], item_start)
        } else if hyphens.is_empty() {
            (1u8, base, item_start, &[][..], item_start)
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            (2u8, &base[..h], item_start, &base[h + 1..], item_start + h + 1)
        } else {
            return Err(syntax(field, item_start));
        };

        let endpoint_shape_ok = match kind {
            0 => true,
            1 => !a_bytes.is_empty() && a_bytes.iter().all(|b| b.is_ascii_alphanumeric()),
            2 => {
                !a_bytes.is_empty()
                    && !b_bytes.is_empty()
                    && a_bytes.iter().all(|b| b.is_ascii_alphanumeric())
                    && b_bytes.iter().all(|b| b.is_ascii_alphanumeric())
            }
            _ => false,
        };
        if !endpoint_shape_ok || (step_bytes.is_some_and(|s| s.is_empty())) {
            return Err(syntax(field, item_start));
        }

        if let Some(s) = step_bytes {
            if !s.iter().all(u8::is_ascii_digit) {
                return Err(syntax(field, item_start));
            }
        }

        let (start, end) = match kind {
            0 => (min, max),
            1 => {
                let v = parse_endpoint(field, a_bytes, a_offset)?;
                (v, if step_bytes.is_some() { max } else { v })
            }
            2 => {
                let a = parse_endpoint(field, a_bytes, a_offset)?;
                let b = parse_endpoint(field, b_bytes, b_offset)?;
                if a > b {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: b_offset,
                    });
                }
                (a, b)
            }
            _ => unreachable!(),
        };

        let step = if let Some(s) = step_bytes {
            parse_number(s, field, step_offset, true)?
        } else {
            1
        };

        let mut v = start;
        while v <= end {
            selected[v as usize] = true;
            match v.checked_add(step) {
                Some(next) => v = next,
                None => break,
            }
        }
        item_start += item_bytes.len() + 1;
    }
    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {

    }

    let mut sets = Vec::with_capacity(5);
    for (i, text) in parts.iter().enumerate() {
        sets.push(parse_field(FIELDS[i], text)?);
    }

    Ok(Cron {
        sets: sets.try_into().ok().unwrap(),
        day_of_month_unrestricted: parts[2] == "*",
        day_of_week_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (_, max) = limits(field);
        self.sets[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(i, &yes)| {
                if yes && (i as u32) <= max {
                    Some(i as u8)
                } else {
                    None
                }
            })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut ordinal = to_ordinal_minutes(after);
        let end = to_ordinal_minutes(CivilDateTime {
            year: 2399,
            month: 12,
            day: 31,
            hour: 23,
            minute: 59,
        });
        while ordinal < end {
            ordinal += 1;
            let at = from_ordinal_minutes(ordinal);
            if self.matches_valid(at) {
                return Ok(Some(at));
            }
        }
        Ok(None)
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        let minute = self.sets[0][at.minute as usize];
        let hour = self.sets[1][at.hour as usize];
        let dom = self.sets[2][at.day as usize];
        let month = self.sets[3][at.month as usize];
        let dow = weekday(at.year, at.month, at.day) as usize;
        let dow_match = self.sets[4][dow];

        if !(minute && hour && month) {
            return false;
        }

        let date_match = match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow_match,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow_match
                } else {
                    dom || dow_match
                }
            }
        };
        date_match
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day < 1 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn to_ordinal_minutes(at: CivilDateTime) -> u64 {
    let mut days = 0u64;
    for y in 2000..at.year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..at.month {
        days += days_in_month(at.year, m) as u64;
    }
    days += (at.day - 1) as u64;
    days * 1440 + at.hour as u64 * 60 + at.minute as u64
}

fn from_ordinal_minutes(mut n: u64) -> CivilDateTime {
    let minute = (n % 60) as u8;
    n /= 60;
    let hour = (n % 24) as u8;
    let mut days = n / 24;
    let mut year = 2000u16;
    while days >= if leap(year) { 366 } else { 365 } {
        days -= if leap(year) { 366 } else { 365 };
        year += 1;
    }
    let mut month = 1u8;
    while days >= days_in_month(year, month) as u64 {
        days -= days_in_month(year, month) as u64;
        month += 1;
    }
    CivilDateTime {
        year,
        month,
        day: days as u8 + 1,
        hour,
        minute,
    }
}

use std::convert::TryInto;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,

pub enum ParseError {

    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    allowed: [Vec<u8>; 5],
    dom_star: bool,
    dow_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn bounds(f: Field) -> (u32, u32) {
    match f {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}
fn idx(f: Field) -> usize {
    match f {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}
fn syn(f: Field, p: usize) -> ParseError {
    ParseError::InvalidSyntax { field: f, offset: p }
}
fn number(s: &[u8], f: Field, p: usize, step: bool) -> Result<u32, ParseError> {
    if s.is_empty() || !s.iter().all(u8::is_ascii_digit) {
        return Err(syn(f, p));
    }
    if s.len() > 1 && s[0] == b'0' {
        return Err(ParseError::LeadingZero { field: f, offset: p });
    }
    let mut x = 0u32;
    for b in s {
        x = x
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field: f, offset: p })?;
    }
    let (lo, hi) = bounds(f);
    if step {
        if x == 0 || x > hi - lo + 1 {
            return Err(ParseError::BadStep { field: f, offset: p });
        }
    } else if x < lo || x > hi {
        return Err(ParseError::OutOfRange { field: f, offset: p });
    }
    Ok(x)
}
fn named(f: Field, s: &[u8]) -> Option<u32> {
    let list: &[&str] = match f {
        Field::Month => &["JAN","FEB","MAR","APR","MAY","JUN","JUL","AUG","SEP","OCT","NOV","DEC"],
        Field::DayOfWeek => &["SUN","MON","TUE","WED","THU","FRI","SAT"],
        _ => return None,
    };
    list.iter().position(|v| {
        v.as_bytes().iter().zip(s).all(|(a,b)| a.eq_ignore_ascii_case(b)) && v.len() == s.len()
    }).map(|i| if f == Field::Month { (i+1) as u32 } else { i as u32 })
}
fn endpoint(f: Field, s: &[u8], p: usize) -> Result<u32, ParseError> {
    if !s.is_empty() && s.iter().all(u8::is_ascii_digit) {
        number(s, f, p, false)
    } else {
        named(f, s).ok_or_else(|| syn(f,p))
    }
}
fn field_set(f: Field, s: &str) -> Result<Vec<u8>, ParseError> {
    let b = s.as_bytes();
    for (i, c) in b.iter().enumerate() {
        if !(c.is_ascii_alphanumeric() || matches!(c, b'*'|b'-'|b'/'|b',')) {
            return Err(ParseError::InvalidCharacter { field: f, offset: i });
        }
    }
    let (lo, hi) = bounds(f);
    let mut set = vec![false; (hi+1) as usize];
    let mut baseoff = 0;
    for item in s.split(',') {
        let ib = item.as_bytes();
        if ib.is_empty() {
            return Err(ParseError::EmptyItem { field: f, offset: baseoff });
        }
        let slash = ib.iter().position(|c| *c == b'/');
        if ib.iter().filter(|c| **c == b'/').count() > 1 {
            return Err(syn(f, baseoff));
        }
        let (base, st, stpos) = match slash {
            Some(x) => (&ib[..x], Some(&ib[x+1..]), baseoff+x+1),
            None => (ib, None, baseoff),
        };
        let hy: Vec<usize> = base.iter().enumerate().filter_map(|(i,c)| if *c==b'-' {Some(i)} else {None}).collect();
        let (kind, a, ap, z, zp) = if base == b"*" {
            (0, &[][..], baseoff, &[][..], baseoff)
        } else if hy.is_empty() {
            (1, base, baseoff, &[][..], baseoff)
        } else if hy.len() == 1 {
            let h=hy[0]; (2,&base[..h],baseoff,&base[h+1..],baseoff+h+1)
        } else {
            return Err(syn(f,baseoff));
        };
        let valid = match kind {
            0 => true,
            1 => !a.is_empty() && a.iter().all(u8::is_ascii_alphanumeric),
            _ => !a.is_empty() && !z.is_empty()
                && a.iter().all(u8::is_ascii_alphanumeric)
                && z.iter().all(u8::is_ascii_alphanumeric),
        };
        if !valid || st.is_some_and(|x| x.is_empty()) || st.is_some_and(|x| !x.iter().all(u8::is_ascii_digit)) {
            return Err(syn(f,baseoff));
        }
        let (start,end)=match kind {
            0=>(lo,hi),
            1=>{let v=endpoint(f,a,ap)?;(v,if st.is_some(){hi}else{v})},
            _=>{
                let x=endpoint(f,a,ap)?;
                let y=endpoint(f,z,zp)?;
                if x>y {return Err(ParseError::ReversedRange{field:f,offset:zp});}
                (x,y)
            }
        };
        let step=if let Some(v)=st {number(v,f,stpos,true)?} else {1};
        let mut x=start;
        while x<=end {
            set[x as usize]=true;
            x=match x.checked_add(step) {Some(v)=>v,None=>break};
        }
        baseoff += ib.len()+1;
    }
    Ok(set.iter().enumerate().filter_map(|(i,v)| if *v {Some(i as u8)} else {None}).collect())
}
pub fn parse_cron(s: &str) -> Result<Cron, ParseError> {
    let p: Vec<&str> = s.split(' ').collect();
    if p.len()!=5 {return Err(ParseError::FieldCount{found:p.len()});}
    let f=[Field::Minute,Field::Hour,Field::DayOfMonth,Field::Month,Field::DayOfWeek];
    let mut v=Vec::new();
    for i in 0..5 {v.push(field_set(f[i],p[i])?);}
    Ok(Cron {
        allowed: v.try_into().ok().unwrap(),
        dom_star: p[2]=="*",
        dow_star: p[4]=="*",
    })
}
impl Cron {
    pub fn allowed(&self, f: Field) -> Vec<u8> { self.allowed[idx(f)].clone() }
    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_ok(at))
    }
    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut x=after;
        loop {
            if x.year==2399 && x.month==12 && x.day==31 && x.hour==23 && x.minute==59 { return Ok(None); }
            x=add_minute(x);
            if self.matches_ok(x) { return Ok(Some(x)); }
        }
    }
    fn matches_ok(&self,x:CivilDateTime)->bool {
        if !self.allowed[0].contains(&x.minute) || !self.allowed[1].contains(&x.hour) || !self.allowed[3].contains(&x.month) { return false; }
        let dom=self.allowed[2].contains(&x.day);
        let dow=self.allowed[4].contains(&weekday(x.year,x.month,x.day));
        match (self.dom_star,self.dow_star) {
            (true,true)=>true,
            (true,false)=>dow,
            (false,true)=>dom,
            (false,false)=>if x.day==month_days(x.year,x.month) {dom&&dow} else {dom||dow},
        }
    }
}
fn leap(y:u16)->bool { y%4==0 && (y%100!=0 || y%400==0) }
fn month_days(y:u16,m:u8)->u8 {
    match m {2=>if leap(y){29}else{28},4|6|9|11=>30,_=>31}
}
fn validate(x:CivilDateTime)->Result<(),DateError> {
    if !(2000..=2399).contains(&x.year) {return Err(DateError::InvalidDateTime{component:DatePart::Year});}
    if !(1..=12).contains(&x.month) {return Err(DateError::InvalidDateTime{component:DatePart::Month});}
    if x.day<1 || x.day>month_days(x.year,x.month) {return Err(DateError::InvalidDateTime{component:DatePart::Day});}
    if x.hour>23 {return Err(DateError::InvalidDateTime{component:DatePart::Hour});}
    if x.minute>59 {return Err(DateError::InvalidDateTime{component:DatePart::Minute});}
    Ok(())
}
fn weekday(y:u16,m:u8,d:u8)->u8 {
    let mut n=0u32;
    for yy in 2000..y {n+=if leap(yy){366}else{365};}
    for mm in 1..m {n+=month_days(y,mm) as u32;}
    n+=(d-1) as u32;
    ((n+6)%7) as u8
}
fn add_minute(mut x:CivilDateTime)->CivilDateTime {
    if x.minute<59 {x.minute+=1;return x;}
    x.minute=0;
    if x.hour<23 {x.hour+=1;return x;}
    x.hour=0;
    if x.day<month_days(x.year,x.month) {x.day+=1;return x;}
    x.day=1;
    if x.month<12 {x.month+=1;return x;}
    x.month=1;
    x.year+=1;
    x
}
