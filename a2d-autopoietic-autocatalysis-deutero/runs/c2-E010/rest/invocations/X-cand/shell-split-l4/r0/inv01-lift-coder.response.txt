- NUL is checked across the entire input before parsing, so it outranks earlier syntax errors.
- Braced expansion searches for its first unescaped `}` before validating its contents; defaults are decoded only when selected.
- Backslash-LF outside quotes does not start a word; double-quoted backslash only consumes the specified four characters.
- Wildcard offsets are byte offsets in the assembled word; expansions inherit the quote context where they occur.
- Joining chooses the shortest of the three specified encodings, with the given tie priority and fallback for the exceptional case.
- Solution B uses a separate scanner structure while following the same interpretations.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn append(text: &mut String, glob: &mut Vec<usize>, s: &str, active: bool) {
    for c in s.chars() {
        let p = text.len();
        text.push(c);
        if active && matches!(c, '*' | '?' | '[') {
            glob.push(p);
        }
    }
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v)
}

fn decode_default(s: &str) -> String {
    let mut out = String::new();
    let mut it = s.char_indices().peekable();
    while let Some((_, c)) = it.next() {
        if c == '\\' {
            if let Some((_, next)) = it.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut text = String::new();
    let mut glob = Vec::new();
    let mut active_word = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append(&mut text, &mut glob, &input[i..i + n], false);
                }
                i += n;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    i += n;
                } else if c == '\\' {
                    let p = i;
                    i += 1;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: p });
                    }
                    let next = input[i..].chars().next().unwrap();
                    let nn = next.len_utf8();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append(&mut text, &mut glob, &input[i..i + nn], false);
                        }
                        i += nn;
                    } else {
                        append(&mut text, &mut glob, "\\", false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let dollar = i;
                        let content_start = i + 2;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let d = input[j..].chars().next().unwrap();
                            if d == '\\' {
                                j += 1;
                                if j < input.len() {
                                    let x = input[j..].chars().next().unwrap();
                                    j += x.len_utf8();
                                }
                            } else if d == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += d.len_utf8();
                            }
                        }
                        let end = match close {
                            Some(k) => k,
                            None => return Err(SplitError::UnterminatedExpansion { pos: dollar }),
                        };
                        let body = &input[content_start..end];
                        let (name, default) = if let Some(k) = body.find(":-") {
                            (&body[..k], Some(&body[k + 2..]))
                        } else {
                            (body, None)
                        };
                        if name.is_empty() || !name.chars().next().map(is_name_start).unwrap_or(false)
                            || !name.chars().all(is_name_char)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        if default.is_none() && body != name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = match default {
                            Some(d) => match lookup(name, env) {
                                Some(v) if !v.is_empty() => v.to_string(),
                                _ => decode_default(d),
                            },
                            None => lookup(name, env).unwrap_or("").to_string(),
                        };
                        append(&mut text, &mut glob, &value, false);
                        active_word = true;
                        i = end + 1;
                    } else if let Some(first) = input[i + n..].chars().next() {
                        if is_name_start(first) {
                            let mut end = i + n + first.len_utf8();
                            while end < input.len() {
                                let d = input[end..].chars().next().unwrap();
                                if !is_name_char(d) {
                                    break;
                                }
                                end += d.len_utf8();
                            }
                            let name = &input[i + n..end];
                            let value = lookup(name, env).unwrap_or("");
                            append(&mut text, &mut glob, value, false);
                            active_word = true;
                            i = end;
                        } else {
                            append(&mut text, &mut glob, "$", false);
                            i += n;
                        }
                    } else {
                        append(&mut text, &mut glob, "$", false);
                        i += n;
                    }
                } else {
                    append(&mut text, &mut glob, &input[i..i + n], false);
                    i += n;
                }
            }
            Quote::None => {
                if is_sep(c) {
                    if active_word {
                        words.push(Word { text: std::mem::take(&mut text), glob: std::mem::take(&mut glob) });
                        active_word = false;
                    }
                    i += n;
                } else if c == '\'' {
                    quote = Quote::Single(i);
                    active_word = true;
                    i += n;
                } else if c == '"' {
                    quote = Quote::Double(i);
                    active_word = true;
                    i += n;
                } else if c == '\\' {
                    let p = i;
                    i += 1;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: p });
                    }
                    let next = input[i..].chars().next().unwrap();
                    let nn = next.len_utf8();
                    if next != '\n' {
                        append(&mut text, &mut glob, &input[i..i + nn], false);
                        active_word = true;
                    }
                    i += nn;
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[i + n..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[i + n..].starts_with('{') {
                        let dollar = i;
                        let content_start = i + 2;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let d = input[j..].chars().next().unwrap();
                            if d == '\\' {
                                j += 1;
                                if j < input.len() {
                                    let x = input[j..].chars().next().unwrap();
                                    j += x.len_utf8();
                                }
                            } else if d == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j += d.len_utf8();
                            }
                        }
                        let end = match close {
                            Some(k) => k,
                            None => return Err(SplitError::UnterminatedExpansion { pos: dollar }),
                        };
                        let body = &input[content_start..end];
                        let (name, default) = if let Some(k) = body.find(":-") {
                            (&body[..k], Some(&body[k + 2..]))
                        } else {
                            (body, None)
                        };
                        if name.is_empty() || !name.chars().next().map(is_name_start).unwrap_or(false)
                            || !name.chars().all(is_name_char)
                        {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        if default.is_none() && body != name {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        let value = match default {
                            Some(d) => match lookup(name, env) {
                                Some(v) if !v.is_empty() => v.to_string(),
                                _ => decode_default(d),
                            },
                            None => lookup(name, env).unwrap_or("").to_string(),
                        };
                        append(&mut text, &mut glob, &value, true);
                        active_word = true;
                        i = end + 1;
                    } else if let Some(first) = input[i + n..].chars().next() {
                        if is_name_start(first) {
                            let mut end = i + n + first.len_utf8();
                            while end < input.len() {
                                let d = input[end..].chars().next().unwrap();
                                if !is_name_char(d) {
                                    break;
                                }
                                end += d.len_utf8();
                            }
                            let name = &input[i + n..end];
                            let value = lookup(name, env).unwrap_or("");
                            append(&mut text, &mut glob, value, true);
                            active_word = true;
                            i = end;
                        } else {
                            append(&mut text, &mut glob, "$", true);
                            active_word = true;
                            i += n;
                        }
                    } else {
                        append(&mut text, &mut glob, "$", true);
                        active_word = true;
                        i += n;
                    }
                } else {
                    append(&mut text, &mut glob, &input[i..i + n], true);
                    active_word = true;
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single(p) => return Err(SplitError::UnterminatedSingle { pos: p }),
        Quote::Double(p) => return Err(SplitError::UnterminatedDouble { pos: p }),
        Quote::None => {}
    }
    if active_word {
        words.push(Word { text, glob });
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric() || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%') {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{}'", word))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        None
    } else {
        let mut out = String::from("\"");
        for c in word.chars() {
            if matches!(c, '\\' | '"' | '$') {
                out.push('\\');
            }
            out.push(c);
        }
        out.push('"');
        Some(out)
    }
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.as_bytes().iter().position(|&b| b == 0) {
            return Err(JoinError::Nul { word: word_index, pos });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates = Vec::new();
        if let Some(s) = bare_candidate(word) { candidates.push((0, s)); }
        if let Some(s) = single_candidate(word) { candidates.push((1, s)); }
        if let Some(s) = double_candidate(word) { candidates.push((2, s)); }

        let chosen = if candidates.is_empty() {
            let mut out = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    out.push('\'');
                    out.push('\\');
                    out.push('\'');
                    out.push('\'');
                } else {
                    out.push(c);
                }
            }
            out.push('\'');
            out
        } else {
            candidates.into_iter().min_by_key(|(priority, s)| (s.len(), *priority)).unwrap().1
        };
        encoded.push(chosen);
    }
    Ok(encoded.join(" "))
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

#[derive(Clone, Copy)]
enum Q { Bare, Single(usize), Double(usize) }

fn sep(c: char) -> bool { c == ' ' || c == '\t' || c == '\r' || c == '\n' }
fn start(c: char) -> bool { c == '_' || c.is_ascii_alphabetic() }
fn part(c: char) -> bool { c == '_' || c.is_ascii_alphanumeric() }

fn put(w: &mut Word, s: &str, wild: bool) {
    for c in s.chars() {
        let p = w.text.len();
        w.text.push(c);
        if wild && (c == '*' || c == '?' || c == '[') { w.glob.push(p); }
    }
}
fn envval<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    for (k, v) in env { if *k == name { return Some(*v); } }
    None
}
fn decode(s: &str) -> String {
    let mut o = String::new();
    let mut it = s.chars();
    while let Some(c) = it.next() {
        if c == '\\' { if let Some(n) = it.next() { o.push(n); } else { o.push('\\'); } }
        else { o.push(c); }
    }
    o
}
fn braced(input: &str, dollar: usize, env: &[(&str, &str)]) -> Result<(String, usize), SplitError> {
    let begin = dollar + 2;
    let mut p = begin;
    let mut close = None;
    while p < input.len() {
        let c = input[p..].chars().next().unwrap();
        if c == '\\' {
            p += 1;
            if p < input.len() { p += input[p..].chars().next().unwrap().len_utf8(); }
        } else if c == '}' { close = Some(p); break; }
        else { p += c.len_utf8(); }
    }
    let end = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let body = &input[begin..end];
    let split = body.find(":-");
    let (name, default) = match split { Some(i) => (&body[..i], Some(&body[i + 2..])), None => (body, None) };
    if name.is_empty() || !name.chars().next().map(start).unwrap_or(false) || !name.chars().all(part) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    if default.is_none() && name != body { return Err(SplitError::InvalidExpansion { pos: dollar }); }
    let value = match default {
        Some(d) => match envval(name, env) { Some(v) if !v.is_empty() => v.to_string(), _ => decode(d) },
        None => envval(name, env).unwrap_or("").to_string(),
    };
    Ok((value, end + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(p) = input.as_bytes().iter().position(|b| *b == 0) { return Err(SplitError::Nul { pos: p }); }
    let mut out = Vec::new();
    let mut w = Word { text: String::new(), glob: Vec::new() };
    let mut started = false;
    let mut q = Q::Bare;
    let mut p = 0;
    while p < input.len() {
        let c = input[p..].chars().next().unwrap();
        let sz = c.len_utf8();
        match q {
            Q::Single(_) => {
                if c == '\'' { q = Q::Bare; } else { put(&mut w, &input[p..p+sz], false); }
                p += sz;
            }
            Q::Double(_) => {
                if c == '"' { q = Q::Bare; p += sz; }
                else if c == '\\' {
                    let at = p; p += 1;
                    if p == input.len() { return Err(SplitError::TrailingEscape { pos: at }); }
                    let n = input[p..].chars().next().unwrap(); let ns = n.len_utf8();
                    if n == '"' || n == '\\' || n == '$' || n == '\n' {
                        if n != '\n' { put(&mut w, &input[p..p+ns], false); }
                        p += ns;
                    } else { put(&mut w, "\\", false); }
                } else if c == '`' { return Err(SplitError::CommandSubstitution { pos: p }); }
                else if c == '$' {
                    if input[p+1..].starts_with('(') { return Err(SplitError::CommandSubstitution { pos: p }); }
                    if input[p+1..].starts_with('{') {
                        let (v, next) = braced(input, p, env)?; put(&mut w, &v, false); started = true; p = next;
                    } else if p+1 < input.len() && start(input[p+1..].chars().next().unwrap()) {
                        let mut e = p+1;
                        while e < input.len() { let x = input[e..].chars().next().unwrap(); if !part(x) { break; } e += x.len_utf8(); }
                        let v = envval(&input[p+1..e], env).unwrap_or(""); put(&mut w, v, false); started = true; p = e;
                    } else { put(&mut w, "$", false); started = true; p += 1; }
                } else { put(&mut w, &input[p..p+sz], false); p += sz; }
            }
            Q::Bare => {
                if sep(c) {
                    if started { out.push(std::mem::replace(&mut w, Word { text: String::new(), glob: Vec::new() })); started = false; }
                    p += sz;
                } else if c == '\'' { q = Q::Single(p); started = true; p += sz; }
                else if c == '"' { q = Q::Double(p); started = true; p += sz; }
                else if c == '\\' {
                    let at = p; p += 1;
                    if p == input.len() { return Err(SplitError::TrailingEscape { pos: at }); }
                    let n = input[p..].chars().next().unwrap(); let ns = n.len_utf8();
                    if n != '\n' { put(&mut w, &input[p..p+ns], false); started = true; }
                    p += ns;
                } else if c == '`' { return Err(SplitError::CommandSubstitution { pos: p }); }
                else if c == '$' {
                    if input[p+1..].starts_with('(') { return Err(SplitError::CommandSubstitution { pos: p }); }
                    if input[p+1..].starts_with('{') {
                        let (v, next) = braced(input, p, env)?; put(&mut w, &v, true); started = true; p = next;
                    } else if p+1 < input.len() && start(input[p+1..].chars().next().unwrap()) {
                        let mut e = p+1;
                        while e < input.len() { let x = input[e..].chars().next().unwrap(); if !part(x) { break; } e += x.len_utf8(); }
                        let v = envval(&input[p+1..e], env).unwrap_or(""); put(&mut w, v, true); started = true; p = e;
                    } else { put(&mut w, "$", true); started = true; p += 1; }
                } else { put(&mut w, &input[p..p+sz], true); started = true; p += sz; }
            }
        }
    }
    match q { Q::Single(p) => return Err(SplitError::UnterminatedSingle { pos: p }), Q::Double(p) => return Err(SplitError::UnterminatedDouble { pos: p }), Q::Bare => {} }
    if started { out.push(w); }
    Ok(out)
}

fn bare(s: &str) -> Option<String> {
    if s.is_empty() || s.contains('\n') { return None; }
    let mut o = String::new();
    for c in s.chars() {
        if c.is_ascii_alphanumeric() || "_-./:=,+@%".contains(c) { o.push(c); }
        else { o.push('\\'); o.push(c); }
    }
    Some(o)
}
fn single(s: &str) -> Option<String> { if s.contains('\'') { None } else { Some(format!("'{}'", s)) } }
fn double(s: &str) -> Option<String> {
    if s.contains('`') { None } else {
        let mut o = String::from("\"");
        for c in s.chars() { if c == '\\' || c == '"' || c == '$' { o.push('\\'); } o.push(c); }
        o.push('"'); Some(o)
    }
}
pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (i, s) in words.iter().enumerate() {
        if let Some(p) = s.as_bytes().iter().position(|b| *b == 0) { return Err(JoinError::Nul { word: i, pos: p }); }
    }
    let mut all = Vec::new();
    for s in words {
        let mut cs = Vec::new();
        if let Some(x) = bare(s) { cs.push((0, x)); }
        if let Some(x) = single(s) { cs.push((1, x)); }
        if let Some(x) = double(s) { cs.push((2, x)); }
        let x = if cs.is_empty() {
            let mut x = String::from("'");
            for c in s.chars() { if c == '\'' { x.push('\''); x.push('\\'); x.push('\''); x.push('\''); } else { x.push(c); } }
            x.push('\''); x
        } else { cs.into_iter().min_by_key(|(r, x)| (x.len(), *r)).unwrap().1 };
        all.push(x);
    }
    Ok(all.join(" "))
}
```
