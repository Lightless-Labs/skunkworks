```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const ONES: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];
const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];
const SCALES: [&str; 7] = [
    "thousand",
    "million",
    "billion",
    "trillion",
    "quadrillion",
    "quintillion",
    "quintillion",
];

fn append_small(n: u16, out: &mut String) {
    if n < 20 {
        out.push_str(ONES[n as usize]);
    } else {
        out.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(ONES[(n % 10) as usize]);
        }
    }
}

fn append_group(n: u16, out: &mut String) {
    if n >= 100 {
        out.push_str(ONES[(n / 100) as usize]);
        out.push_str(" hundred");
        if n % 100 != 0 {
            out.push_str(" and ");
            append_small(n % 100, out);
        }
    } else {
        append_small(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = Vec::new();
    let mut n = magnitude;
    while n > 0 {
        groups.push((n % 1000) as u16);
        n /= 1000;
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let highest = groups.len();
    let units = groups.first().copied().unwrap_or(0);
    let cross_and = highest > 1 && (1..=99).contains(&units);
    let mut emitted = false;

    for idx in (0..highest).rev() {
        let group = groups[idx];
        if group == 0 {
            continue;
        }
        if emitted {
            if cross_and && idx == 0 {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }
        append_group(group, &mut out);
        if idx > 0 {
            out.push(' ');
            out.push_str(SCALES[idx - 1]);
        }
        emitted = true;
    }

    Ok(out)
}

fn word_value(word: &str) -> Option<(u128, TokenKind)> {
    for (i, w) in ONES.iter().enumerate() {
        if word == *w {
            return Some((i as u128, TokenKind::Number));
        }
    }
    for (i, w) in TENS.iter().enumerate() {
        if word == *w {
            return Some((((i + 2) * 10) as u128, TokenKind::Number));
        }
    }
    if word == "hundred" {
        return Some((100, TokenKind::Hundred));
    }
    for (i, w) in ["thousand", "million", "billion", "trillion", "quadrillion", "quintillion"]
        .iter()
        .enumerate()
    {
        if word == *w {
            return Some((1000u128.pow((i + 1) as u32), TokenKind::Scale));
        }
    }
    if word == "and" {
        return Some((0, TokenKind::And));
    }
    None
}

#[derive(Clone, Copy)]
enum TokenKind {
    Number,
    Hundred,
    Scale,
    And,
}

struct Token<'a> {
    text: &'a str,
    index: usize,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b'a'..=b'z').contains(&b) && b != b' ' && b != b'-' {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for i in 0..bytes.len() {
        if bytes[i] == b' ' || bytes[i] == b'-' {
            if i == 0
                || i + 1 == bytes.len()
                || !(b'a'..=b'z').contains(&bytes[i - 1])
                || !(b'a'..=b'z').contains(&bytes[i + 1])
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens = Vec::new();
    let mut start = 0;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b' ' || bytes[i] == b'-' {
            tokens.push(Token {
                text: &input[start..i],
                index: start,
            });
            start = i + 1;
        }
    }

    let mut negative = false;
    let mut arithmetic_start = 0;
    for (i, token) in tokens.iter().enumerate() {
        if token.text == "minus" {
            if i == 0 {
                negative = true;
                arithmetic_start = 1;
            } else {
                return Err(WordsError::InvalidWord { index: token.index });
            }
        } else if word_value(token.text).is_none() {
            return Err(WordsError::InvalidWord { index: token.index });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;
    for token in tokens.iter().skip(arithmetic_start) {
        let (value, kind) = word_value(token.text).expect("dictionary was checked");
        match kind {
            TokenKind::Number => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale => {
                total += group * value;
                group = 0;
            }
            TokenKind::And => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != input.as_bytes() {
        let common = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .take_while(|(a, b)| a == b)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(signed)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let mag = value.unsigned_abs();
    if mag > LIMIT {
        return Err(WordsError::Range);
    }
    if mag == 0 {
        return Ok(String::from("zero"));
    }

    let mut parts: Vec<(usize, u16)> = Vec::new();
    let mut n = mag;
    let mut scale = 0usize;
    while n != 0 {
        let g = (n % 1000) as u16;
        if g != 0 {
            parts.push((scale, g));
        }
        n /= 1000;
        scale += 1;
    }
    parts.reverse();

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }
    let has_higher = parts.iter().any(|(s, _)| *s > 0);
    let units = parts.iter().find(|(s, _)| *s == 0).map(|(_, g)| *g).unwrap_or(0);
    let use_cross_and = has_higher && (1..100).contains(&units);

    for (i, (scale, group)) in parts.iter().enumerate() {
        if i != 0 {
            if use_cross_and && *scale == 0 {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        }
        write_group(*group, &mut result);
        if *scale != 0 {
            result.push(' ');
            result.push_str(match scale {
                1 => "thousand",
                2 => "million",
                3 => "billion",
                4 => "trillion",
                5 => "quadrillion",
                6 => "quintillion",
                _ => unreachable!(),
            });
        }
    }
    Ok(result)
}

fn write_group(n: u16, out: &mut String) {
    const SMALL: [&str; 20] = [
        "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
        "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
        "seventeen", "eighteen", "nineteen",
    ];
    const DECADES: [&str; 8] = [
        "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
    ];
    let mut rem = n;
    if rem >= 100 {
        out.push_str(SMALL[(rem / 100) as usize]);
        out.push_str(" hundred");
        rem %= 100;
        if rem != 0 {
            out.push_str(" and ");
        }
    }
    if rem >= 20 {
        out.push_str(DECADES[(rem / 10 - 2) as usize]);
        rem %= 10;
        if rem != 0 {
            out.push('-');
            out.push_str(SMALL[rem as usize]);
        }
    } else if rem != 0 {
        out.push_str(SMALL[rem as usize]);
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }
    let b = input.as_bytes();
    for (i, &c) in b.iter().enumerate() {
        if !(b'a'..=b'z').contains(&c) && c != b' ' && c != b'-' {
            return Err(WordsError::InvalidByte { index: i });
        }
    }
    for i in 0..b.len() {
        if b[i] == b' ' || b[i] == b'-' {
            if i == 0 || i + 1 == b.len()
                || !(b'a'..=b'z').contains(&b[i - 1])
                || !(b'a'..=b'z').contains(&b[i + 1])
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut toks: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for i in 0..=b.len() {
        if i == b.len() || b[i] == b' ' || b[i] == b'-' {
            toks.push((&input[start..i], start));
            start = i + 1;
        }
    }

    let mut negative = false;
    for (i, (word, pos)) in toks.iter().enumerate() {
        if *word == "minus" {
            if i == 0 {
                negative = true;
            } else {
                return Err(WordsError::InvalidWord { index: *pos });
            }
        } else if !dictionary(word) {
            return Err(WordsError::InvalidWord { index: *pos });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;
    for (i, (word, _)) in toks.iter().enumerate() {
        if negative && i == 0 {
            continue;
        }
        if *word == "and" {
            // no-op
        } else if *word == "hundred" {
            group *= 100;
        } else if let Some(scale) = scale_value(word) {
            total += group * scale;
            group = 0;
        } else {
            group += number_value(word).unwrap();
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let mag = total + group;
    let value = if negative && mag != 0 { -(mag as i128) } else { mag as i128 };
    let encoded = words_encode(value)?;
    if encoded != input {
        let index = encoded.as_bytes().iter().zip(b.iter())
            .take_while(|(x, y)| x == y).count();
        return Err(WordsError::NonCanonical { index });
    }
    Ok(value)
}

fn dictionary(w: &str) -> bool {
    number_value(w).is_some() || w == "hundred" || w == "and" || scale_value(w).is_some()
}

fn number_value(w: &str) -> Option<u128> {
    const SMALL: [&str; 20] = [
        "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
        "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
        "seventeen", "eighteen", "nineteen",
    ];
    const TENS: [&str; 8] = [
        "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
    ];
    SMALL.iter().position(|x| *x == w).map(|x| x as u128)
        .or_else(|| TENS.iter().position(|x| *x == w).map(|x| ((x + 2) * 10) as u128))
}

fn scale_value(w: &str) -> Option<u128> {
    match w {
        "thousand" => Some(1_000),
        "million" => Some(1_000_000),
        "billion" => Some(1_000_000_000),
        "trillion" => Some(1_000_000_000_000),
        "quadrillion" => Some(1_000_000_000_000_000),
        "quintillion" => Some(1_000_000_000_000_000_000),
        _ => None,
    }
}
```
