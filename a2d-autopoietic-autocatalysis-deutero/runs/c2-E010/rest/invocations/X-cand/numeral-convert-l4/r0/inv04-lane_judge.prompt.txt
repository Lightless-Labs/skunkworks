fn encode_under_thousand(n: u16) -> String {

    if rem == 0 {

    } else {
        format!("{} hundred and {}", SMALL[hundreds as usize], encode_under_thousand(rem))
    }

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, byte) in input.bytes().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let tokens: Vec<(usize, &str)> = input
        .split(|c| c == ' ' || c == '-')
        .scan(0usize, |offset, token| {
            let start = *offset;
            *offset += token.len() + 1;
            Some((start, token))
        })
        .collect();

    for (i, (index, token)) in tokens.iter().enumerate() {
        if *token == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index: *index });
            }
        } else if word_value(token).is_none() {
            return Err(WordsError::InvalidWord { index: *index });
        }
    }

    let negative = tokens.first().is_some_and(|(_, word)| *word == "minus");
    let arithmetic_tokens = if negative { &tokens[1..] } else { &tokens[..] };

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for (_, token) in arithmetic_tokens {
        match *token {
            "zero" => {}
            "hundred" => group *= 100,
            "and" => {}
            scale_word => {
                if let Some(scale) = is_scale(scale_word) {
                    total += group * scale;
                    group = 0;
                } else {
                    group += word_value(scale_word).unwrap();
                }
            }
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(value)?;
    if canonical.as_bytes() != input.as_bytes() {
        let a = canonical.as_bytes();
        let b = input.as_bytes();
        let common = a.len().min(b.len());
        let index = (0..common).find(|&i| a[i] != b[i]).unwrap_or(common);
        return Err(WordsError::NonCanonical { index });
    }
    Ok(value)
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }
    let b = input.as_bytes();
    if let Some((i, _)) = b.iter().enumerate().find(|(_, c)| !(c.is_ascii_lowercase() || **c == b' ' || **c == b'-')) {
        return Err(WordsError::InvalidByte { index: i });
    }
    for i in 0..b.len() {
        if b[i] == b' ' || b[i] == b'-' {
            if i == 0 || i + 1 == b.len() || !b[i - 1].is_ascii_lowercase() || !b[i + 1].is_ascii_lowercase() {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut ops = Vec::new();
    let mut starts = Vec::new();
    let mut begin = 0;
    for i in 0..=b.len() {
        if i == b.len() || b[i] == b' ' || b[i] == b'-' {
            let op = word_op(&input[begin..i]).ok_or(WordsError::InvalidWord { index: begin })?;
            ops.push(op);
            starts.push(begin);
            begin = i + 1;
        }
    }

    let negative = matches!(ops.first(), Some(Op::Minus));
    for i in usize::from(negative)..ops.len() {
        if matches!(ops[i], Op::Minus) {
            return Err(WordsError::InvalidWord { index: starts[i] });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;
    for op in ops.iter().skip(usize::from(negative)) {
        match *op {
            Op::Add(v) => group = group.checked_add(v).ok_or(WordsError::Range)?,
            Op::TimesHundred => group = group.checked_mul(100).ok_or(WordsError::Range)?,
            Op::TimesScale(scale) => {
                total = total
                    .checked_add(group.checked_mul(scale).ok_or(WordsError::Range)?)
                    .ok_or(WordsError::Range)?;
                group = 0;
            }
            Op::Ignore | Op::Minus => {}
        }
        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let mag = total.checked_add(group).ok_or(WordsError::Range)?;
    if mag > LIMIT {
        return Err(WordsError::Range);
    }
    let result = if negative { -(mag as i128) } else { mag as i128 };
    let canonical = words_encode(result)?;
    let a = input.as_bytes();
    let c = canonical.as_bytes();
    let n = a.len().min(c.len());
    for i in 0..n {
        if a[i] != c[i] {
            return Err(WordsError::NonCanonical { index: i });
        }
    }
    if a.len() != c.len() {
        return Err(WordsError::NonCanonical { index: n });
    }
    Ok(result)
}

            out.push_str(SCALES[idx - 1]);
        }
        emitted = true;
    }

    Ok(out)
}

fn word_value(word: &str) -> Option<(u128, TokenKind)> {
    for (i, w) in ONES.iter().enumerate() {
        if word == *w {
            return Some((i as u128, TokenKind::Number));
        }
    }
    for (i, w) in TENS.iter().enumerate() {
        if word == *w {
            return Some((((i + 2) * 10) as u128, TokenKind::Number));
        }
    }
    if word == "hundred" {
        return Some((100, TokenKind::Hundred));
    }
    for (i, w) in ["thousand", "million", "billion", "trillion", "quadrillion", "quintillion"]
        .iter()
        .enumerate()
    {
        if word == *w {
            return Some((1000u128.pow((i + 1) as u32), TokenKind::Scale));
        }
    }
    if word == "and" {
        return Some((0, TokenKind::And));
    }
    None
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b'a'..=b'z').contains(&b) && b != b' ' && b != b'-' {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for i in 0..bytes.len() {
        if bytes[i] == b' ' || bytes[i] == b'-' {
            if i == 0
                || i + 1 == bytes.len()
                || !(b'a'..=b'z').contains(&bytes[i - 1])
                || !(b'a'..=b'z').contains(&bytes[i + 1])
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens = Vec::new();
    let mut start = 0;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b' ' || bytes[i] == b'-' {
            tokens.push(Token {
                text: &input[start..i],
                index: start,
            });
            start = i + 1;
        }
    }

    let mut negative = false;
    let mut arithmetic_start = 0;
    for (i, token) in tokens.iter().enumerate() {
        if token.text == "minus" {
            if i == 0 {
                negative = true;
                arithmetic_start = 1;
            } else {
                return Err(WordsError::InvalidWord { index: token.index });
            }
        } else if word_value(token.text).is_none() {
            return Err(WordsError::InvalidWord { index: token.index });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;
    for token in tokens.iter().skip(arithmetic_start) {
        let (value, kind) = word_value(token.text).expect("dictionary was checked");
        match kind {
            TokenKind::Number => group += value,
            TokenKind::Hundred => group *= 100,
            TokenKind::Scale => {
                total += group * value;
                group = 0;
            }
            TokenKind::And => {}
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let signed = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(signed)?;
    if canonical.as_bytes() != input.as_bytes() {
        let common = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .take_while(|(a, b)| a == b)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(signed)
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let mag = value.unsigned_abs();
    if mag > LIMIT {
        return Err(WordsError::Range);
    }
    if mag == 0 {
        return Ok(String::from("zero"));
    }

    let mut parts: Vec<(usize, u16)> = Vec::new();
    let mut n = mag;
    let mut scale = 0usize;
    while n != 0 {
        let g = (n % 1000) as u16;
        if g != 0 {
            parts.push((scale, g));
        }
        n /= 1000;
        scale += 1;
    }
    parts.reverse();

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }
    let has_higher = parts.iter().any(|(s, _)| *s > 0);
    let units = parts.iter().find(|(s, _)| *s == 0).map(|(_, g)| *g).unwrap_or(0);
    let use_cross_and = has_higher && (1..100).contains(&units);

    for (i, (scale, group)) in parts.iter().enumerate() {
        if i != 0 {
            if use_cross_and && *scale == 0 {
                result.push_str(" and ");
            } else {
                result.push(' ');
            }
        }
        write_group(*group, &mut result);
        if *scale != 0 {
            result.push(' ');
            result.push_str(match scale {
                1 => "thousand",
                2 => "million",
                3 => "billion",
                4 => "trillion",
                5 => "quadrillion",
                6 => "quintillion",
                _ => unreachable!(),
            });
        }
    }
    Ok(result)
}

    const SMALL: [&str; 20] = [

        "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",

    if rem >= 100 {

            out.push_str(" and ");

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }
    let b = input.as_bytes();
    for (i, &c) in b.iter().enumerate() {
        if !(b'a'..=b'z').contains(&c) && c != b' ' && c != b'-' {
            return Err(WordsError::InvalidByte { index: i });
        }
    }
    for i in 0..b.len() {
        if b[i] == b' ' || b[i] == b'-' {
            if i == 0 || i + 1 == b.len()
                || !(b'a'..=b'z').contains(&b[i - 1])
                || !(b'a'..=b'z').contains(&b[i + 1])

                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut toks: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for i in 0..=b.len() {
        if i == b.len() || b[i] == b' ' || b[i] == b'-' {
            toks.push((&input[start..i], start));
            start = i + 1;

    for (i, (word, pos)) in toks.iter().enumerate() {

                return Err(WordsError::InvalidWord { index: *pos });
            }

            return Err(WordsError::InvalidWord { index: *pos });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;
    for (i, (word, _)) in toks.iter().enumerate() {
        if negative && i == 0 {
            continue;
        }
        if *word == "and" {
            // no-op
        } else if *word == "hundred" {
            group *= 100;
        } else if let Some(scale) = scale_value(word) {
            total += group * scale;
            group = 0;
        } else {
            group += number_value(word).unwrap();
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }

    let mag = total + group;
    let value = if negative && mag != 0 { -(mag as i128) } else { mag as i128 };

        let index = encoded.as_bytes().iter().zip(b.iter())
            .take_while(|(x, y)| x == y).count();

}

    if spelling.as_bytes() != b {
        let common = spelling
            .as_bytes()
            .iter()
            .zip(b.iter())
            .take_while(|(x, y)| x == y)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(result)
}

    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
    "eighteen", "nineteen",
];

    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",

    "quintillion", "quadrillion", "trillion", "billion", "million", "thousand", "",

fn append_under_hundred(n: u128, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        let tens = (n / 10) as usize - 2;
        out.push_str(TENS[tens]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

fn append_group(n: u128, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let rem = n % 100;
        if rem != 0 {
            out.push_str(" and ");
            append_under_hundred(rem, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut rest = magnitude;
    for i in (0..7).rev() {
        groups[i] = (rest % 1000) as u16;
        rest /= 1000;
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let nonzero: Vec<usize> = (0..7).filter(|&i| groups[i] != 0).collect();
    for (pos, &i) in nonzero.iter().enumerate() {
        if pos != 0 {
            let is_final_small_units =
                i == 6 && groups[i] < 100 && groups[i] > 0;
            if is_final_small_units {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }
        append_group(groups[i] as u128, &mut out);
        if !SCALES[i].is_empty() {
            out.push(' ');
            out.push_str(SCALES[i]);
        }
    }
    Ok(out)
}

#[derive(Clone, Copy)]
enum TokenKind {
    Number,
    Hundred,
    Scale,
    And,
    Minus,
}

#[derive(Clone, Copy)]
struct Token {
    value: u128,
    kind: TokenKind,
    index: usize,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_lowercase() && b != b' ' && b != b'-' {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for i in 0..bytes.len() {
        if bytes[i] == b' ' || bytes[i] == b'-' {
            if i == 0
                || i + 1 == bytes.len()
                || !bytes[i - 1].is_ascii_lowercase()
                || !bytes[i + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens: Vec<Token> = Vec::new();
    let mut start = 0;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b' ' || bytes[i] == b'-' {
            let word = &input[start..i];
            let (value, kind) = lookup(word).ok_or(WordsError::InvalidWord { index: start })?;
            tokens.push(Token { value, kind, index: start });
            start = i + 1;
        }
    }

    let negative = matches!(tokens.first(), Some(Token { kind: TokenKind::Minus, .. }));
    if tokens.iter().skip(usize::from(negative)).any(|t| matches!(t.kind, TokenKind::Minus)) {
        let t = tokens
            .iter()
            .skip(usize::from(negative))
            .find(|t| matches!(t.kind, TokenKind::Minus))
            .unwrap();
        return Err(WordsError::InvalidWord { index: t.index });
    }

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for token in tokens.iter().skip(usize::from(negative)) {
        match token.kind {
            TokenKind::Number => {
                group = group.checked_add(token.value).ok_or(WordsError::Range)?;
            }
            TokenKind::Hundred => {
                group = group.checked_mul(100).ok_or(WordsError::Range)?;
            }
            TokenKind::Scale => {
                let scaled = group.checked_mul(token.value).ok_or(WordsError::Range)?;
                total = total.checked_add(scaled).ok_or(WordsError::Range)?;
                group = 0;
            }
            TokenKind::And | TokenKind::Minus => {}
        }
        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    let signed = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    let original = input.as_bytes();
    let encoded = canonical.as_bytes();
    let common = original.len().min(encoded.len());
    for i in 0..common {
        if original[i] != encoded[i] {
            return Err(WordsError::NonCanonical { index: i });
        }
    }
    if original.len() != encoded.len() {
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(signed)
}

fn encode_group(n: u16) -> String {
    const NAMES: [&str; 20] = [
        "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
        "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
        "seventeen", "eighteen", "nineteen",
    ];
    const DECADES: [&str; 8] = [
        "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
    ];

    let mut result = String::new();
    let rem = n % 100;
    if n >= 100 {
        result.push_str(NAMES[(n / 100) as usize]);
        result.push_str(" hundred");
        if rem != 0 {
            result.push_str(" and ");
        }
    }
    if rem < 20 {
        if rem != 0 || n < 100 {
            result.push_str(NAMES[rem as usize]);
        }
    } else {
        result.push_str(DECADES[(rem / 10 - 2) as usize]);
        if rem % 10 != 0 {
            result.push('-');
            result.push_str(NAMES[(rem % 10) as usize]);
        }
    }
    result
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let scale_names = [
        "quintillion",
        "quadrillion",
        "trillion",
        "billion",
        "million",
        "thousand",
        "",
    ];
    let mut chunks = Vec::new();
    let mut rest = magnitude;
    for (index, name) in scale_names.iter().enumerate() {
        let exponent = 18 - index * 3;
        let base = 10u128.pow(exponent as u32);
        let group = (rest / base) as u16;
        rest %= base;
        if group != 0 {
            chunks.push((*name, group, encode_group(group)));
        }
    }

    let mut result = String::new();
    for (index, (scale, group, phrase)) in chunks.iter().enumerate() {
        if index != 0 {
            let final_units_short = index == chunks.len() - 1 && scale.is_empty() && *group < 100;
            result.push_str(if final_units_short { " and " } else { " " });
        }
        result.push_str(phrase);
        if !scale.is_empty() {
            result.push(' ');
            result.push_str(scale);
        }
    }
    if value < 0 {
        Ok(format!("minus {result}"))
    } else {
        Ok(result)
    }
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    for (i, b) in input.bytes().enumerate() {
        if !(b.is_ascii_lowercase() || b == b' ' || b == b'-') {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    let bytes = input.as_bytes();
    for (i, b) in bytes.iter().enumerate() {
        if (*b == b' ' || *b == b'-')
            && (i == 0
                || i + 1 == bytes.len()
                || !bytes[i - 1].is_ascii_lowercase()
                || !bytes[i + 1].is_ascii_lowercase())
        {
            return Err(WordsError::Separator { index: i });
        }
    }

    let tokens: Vec<(usize, &str)> = input
        .split(|c| c == ' ' || c == '-')
        .scan(0usize, |offset, token| {
            let start = *offset;
            *offset += token.len() + 1;
            Some((start, token))
        })
        .collect();

    for (idx, (position, word)) in tokens.iter().enumerate() {
        if !recognized(word) || (*word == "minus" && idx != 0) {
            return Err(WordsError::InvalidWord { index: *position });
        }
    }

    let negative = tokens.first().is_some_and(|(_, word)| *word == "minus");
    let mut total = 0u128;
    let mut group = 0u128;
    for (_, word) in tokens.iter().skip(usize::from(negative)) {
        match *word {
            "zero" | "and" => {}
            "hundred" => group *= 100,
            _ => {
                if let Some(s) = scale(word) {
                    total += group * s;
                    group = 0;
                } else {
                    group += number(word);
                }
            }
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative { -(magnitude as i128) } else { magnitude as i128 };
    let canonical = words_encode(value)?;
    if canonical != input {
        let left = canonical.as_bytes();
        let right = input.as_bytes();
        let common = left.len().min(right.len());
        let mismatch = (0..common).find(|&i| left[i] != right[i]).unwrap_or(common);
        return Err(WordsError::NonCanonical { index: mismatch });
    }
    Ok(value)
}
