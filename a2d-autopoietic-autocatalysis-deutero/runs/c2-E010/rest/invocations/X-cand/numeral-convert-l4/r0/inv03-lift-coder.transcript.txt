    let mag = value.unsigned_abs();
    if mag > LIMIT {
        return Err(WordsError::Range);
    }
    if mag == 0 {
        return Ok("zero".to_owned());
    }

    let mut n = mag;
    let mut groups = Vec::<(u16, &'static str)>::new();
    let scales = [
        (1_000_000_000_000_000_000u128, "quintillion"),
        (1_000_000_000_000_000u128, "quadrillion"),
        (1_000_000_000_000u128, "trillion"),
        (1_000_000_000u128, "billion"),
        (1_000_000u128, "million"),
        (1_000u128, "thousand"),

        let q = (n / v) as u16;
        n %= v;
        if q != 0 {
            groups.push((q, name));
        }

    if n != 0 {
        groups.push((n as u16, ""));

            output.push_str(name);

    if input.is_empty() {
        return Err(WordsError::Empty);
    }
    let b = input.as_bytes();

    if let Some(i) = b
        .iter()
        .position(|x| !x.is_ascii_lowercase() && *x != b' ' && *x != b'-')
    {
        return Err(WordsError::InvalidByte { index: i });
    }

    for i in 0..b.len() {
        if b[i] == b' ' || b[i] == b'-' {
            if i == 0 || i + 1 == b.len() || !b[i - 1].is_ascii_lowercase() || !b[i + 1].is_ascii_lowercase() {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens = Vec::<(&str, usize)>::new();
    let mut start = 0;
    for i in 0..b.len() {
        if b[i] == b' ' || b[i] == b'-' {
            tokens.push((&input[start..i], start));
            start = i + 1;
        }
    }
    tokens.push((&input[start..], start));

    for (i, (w, pos)) in tokens.iter().enumerate() {
        if *w == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index: *pos });
            }
        } else if *w != "hundred" && *w != "and" && scale(w).is_none() && vocab_value(w).is_none() {
            return Err(WordsError::InvalidWord { index: *pos });
        }

    let mut group = 0u128;

        if *w == "and" {
        } else if *w == "hundred" {
            group *= 100;

            total += group * s;
            group = 0;
        } else if let Some(v) = vocab_value(w) {
            group += v;
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }

            .iter()
            .zip(b.iter())
            .take_while(|(x, y)| x == y)
            .count();
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(result)
}
