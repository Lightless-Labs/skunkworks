- Decode checks run in the specified order: whole-input byte scan, delimiter scan, full dictionary scan, arithmetic, then canonical comparison.
- Delimiters must have lowercase letters on both sides; spaces and hyphens both split tokens.
- A leading `minus` is special; later `minus` is an invalid word. A lone `minus` decodes arithmetically to zero, then fails canonical comparison.
- Arithmetic uses checked operations and tests the `10^18` bound after every arithmetic token, including `and`.
- Canonical output uses `and` inside non-exact hundreds and between a higher group and a units group from 1 to 99; the limit includes `10^18`.
- Solution B uses a different token representation for lookup and arithmetic while preserving the same specified behavior.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen", "seventeen",
    "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [&str; 7] = [
    "quintillion", "quadrillion", "trillion", "billion", "million", "thousand", "",
];

fn append_under_hundred(n: u128, out: &mut String) {
    if n < 20 {
        out.push_str(SMALL[n as usize]);
    } else {
        let tens = (n / 10) as usize - 2;
        out.push_str(TENS[tens]);
        if n % 10 != 0 {
            out.push('-');
            out.push_str(SMALL[(n % 10) as usize]);
        }
    }
}

fn append_group(n: u128, out: &mut String) {
    if n >= 100 {
        out.push_str(SMALL[(n / 100) as usize]);
        out.push_str(" hundred");
        let rem = n % 100;
        if rem != 0 {
            out.push_str(" and ");
            append_under_hundred(rem, out);
        }
    } else {
        append_under_hundred(n, out);
    }
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut groups = [0u16; 7];
    let mut rest = magnitude;
    for i in (0..7).rev() {
        groups[i] = (rest % 1000) as u16;
        rest /= 1000;
    }

    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }

    let nonzero: Vec<usize> = (0..7).filter(|&i| groups[i] != 0).collect();
    for (pos, &i) in nonzero.iter().enumerate() {
        if pos != 0 {
            let is_final_small_units =
                i == 6 && groups[i] < 100 && groups[i] > 0;
            if is_final_small_units {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }
        append_group(groups[i] as u128, &mut out);
        if !SCALES[i].is_empty() {
            out.push(' ');
            out.push_str(SCALES[i]);
        }
    }
    Ok(out)
}

fn lookup(word: &str) -> Option<(u128, TokenKind)> {
    match word {

#[derive(Clone, Copy)]
enum TokenKind {
    Number,
    Hundred,
    Scale,
    And,
    Minus,
}

#[derive(Clone, Copy)]
struct Token {
    value: u128,
    kind: TokenKind,
    index: usize,
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }

    let bytes = input.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_lowercase() && b != b' ' && b != b'-' {
            return Err(WordsError::InvalidByte { index: i });
        }
    }

    for i in 0..bytes.len() {
        if bytes[i] == b' ' || bytes[i] == b'-' {
            if i == 0
                || i + 1 == bytes.len()
                || !bytes[i - 1].is_ascii_lowercase()
                || !bytes[i + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut tokens: Vec<Token> = Vec::new();
    let mut start = 0;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b' ' || bytes[i] == b'-' {
            let word = &input[start..i];
            let (value, kind) = lookup(word).ok_or(WordsError::InvalidWord { index: start })?;
            tokens.push(Token { value, kind, index: start });
            start = i + 1;
        }
    }

    let negative = matches!(tokens.first(), Some(Token { kind: TokenKind::Minus, .. }));
    if tokens.iter().skip(usize::from(negative)).any(|t| matches!(t.kind, TokenKind::Minus)) {
        let t = tokens
            .iter()
            .skip(usize::from(negative))
            .find(|t| matches!(t.kind, TokenKind::Minus))
            .unwrap();
        return Err(WordsError::InvalidWord { index: t.index });
    }

    let mut total: u128 = 0;
    let mut group: u128 = 0;
    for token in tokens.iter().skip(usize::from(negative)) {
        match token.kind {
            TokenKind::Number => {
                group = group.checked_add(token.value).ok_or(WordsError::Range)?;
            }
            TokenKind::Hundred => {
                group = group.checked_mul(100).ok_or(WordsError::Range)?;
            }
            TokenKind::Scale => {
                let scaled = group.checked_mul(token.value).ok_or(WordsError::Range)?;
                total = total.checked_add(scaled).ok_or(WordsError::Range)?;
                group = 0;
            }
            TokenKind::And | TokenKind::Minus => {}
        }
        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total.checked_add(group).ok_or(WordsError::Range)?;
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    let signed = if negative {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };

    let canonical = words_encode(signed)?;
    let original = input.as_bytes();
    let encoded = canonical.as_bytes();
    let common = original.len().min(encoded.len());
    for i in 0..common {
        if original[i] != encoded[i] {
            return Err(WordsError::NonCanonical { index: i });
        }
    }
    if original.len() != encoded.len() {
        return Err(WordsError::NonCanonical { index: common });
    }
    Ok(signed)
}
```

```rust

    Separator { index: usize },
    InvalidWord { index: usize },

}

const LIMIT: u128 = 1_000_000_000_000_000_000;

fn small_word(n: usize) -> &'static str {
    match n {
        0 => "zero",
        1 => "one",
        2 => "two",
        3 => "three",
        4 => "four",
        5 => "five",
        6 => "six",
        7 => "seven",
        8 => "eight",
        9 => "nine",
        10 => "ten",
        11 => "eleven",
        12 => "twelve",
        13 => "thirteen",
        14 => "fourteen",
        15 => "fifteen",
        16 => "sixteen",
        17 => "seventeen",
        18 => "eighteen",
        _ => "nineteen",
    }
}

fn tens_word(n: usize) -> &'static str {
    match n {
        20 => "twenty",
        30 => "thirty",
        40 => "forty",
        50 => "fifty",
        60 => "sixty",
        70 => "seventy",
        80 => "eighty",
        _ => "ninety",
    }
}

fn encode_group(n: u16) -> String {
    let mut s = String::new();
    let n = n as usize;
    if n >= 100 {
        s.push_str(small_word(n / 100));
        s.push_str(" hundred");
        if n % 100 != 0 {
            s.push_str(" and ");
            s.push_str(&encode_group((n % 100) as u16));
        }
    } else if n < 20 {
        s.push_str(small_word(n));
    } else {
        s.push_str(tens_word(n / 10 * 10));
        if n % 10 != 0 {
            s.push('-');
            s.push_str(small_word(n % 10));
        }
    }
    s
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let mag = value.unsigned_abs();
    if mag > LIMIT {
        return Err(WordsError::Range);
    }
    if mag == 0 {
        return Ok(String::from("zero"));
    }

    let mut groups = Vec::new();
    let mut rest = mag;
    while rest > 0 {
        groups.push((rest % 1000) as u16);
        rest /= 1000;
    }
    while groups.len() < 7 {
        groups.push(0);
    }
    groups.reverse();

    let scale = ["quintillion", "quadrillion", "trillion", "billion", "million", "thousand", ""];
    let nonzero: Vec<(usize, u16)> = groups.iter().copied().enumerate().filter(|(_, g)| *g != 0).collect();
    let mut out = String::new();
    if value < 0 {
        out.push_str("minus ");
    }
    for (j, (i, g)) in nonzero.iter().enumerate() {
        if j != 0 {
            if *i == 6 && *g < 100 {
                out.push_str(" and ");
            } else {
                out.push(' ');
            }
        }
        out.push_str(&encode_group(*g));
        if !scale[*i].is_empty() {
            out.push(' ');
            out.push_str(scale[*i]);
        }
    }
    Ok(out)
}

#[derive(Clone, Copy)]
enum Op {
    Add(u128),
    TimesHundred,
    TimesScale(u128),
    Ignore,
    Minus,
}

fn word_op(s: &str) -> Option<Op> {
    Some(match s {
        "zero" => Op::Add(0),
        "one" => Op::Add(1),
        "two" => Op::Add(2),
        "three" => Op::Add(3),
        "four" => Op::Add(4),
        "five" => Op::Add(5),
        "six" => Op::Add(6),
        "seven" => Op::Add(7),
        "eight" => Op::Add(8),
        "nine" => Op::Add(9),
        "ten" => Op::Add(10),
        "eleven" => Op::Add(11),
        "twelve" => Op::Add(12),
        "thirteen" => Op::Add(13),
        "fourteen" => Op::Add(14),
        "fifteen" => Op::Add(15),
        "sixteen" => Op::Add(16),
        "seventeen" => Op::Add(17),
        "eighteen" => Op::Add(18),
        "nineteen" => Op::Add(19),
        "twenty" => Op::Add(20),
        "thirty" => Op::Add(30),
        "forty" => Op::Add(40),
        "fifty" => Op::Add(50),
        "sixty" => Op::Add(60),
        "seventy" => Op::Add(70),
        "eighty" => Op::Add(80),
        "ninety" => Op::Add(90),
        "hundred" => Op::TimesHundred,
        "thousand" => Op::TimesScale(1_000),
        "million" => Op::TimesScale(1_000_000),
        "billion" => Op::TimesScale(1_000_000_000),
        "trillion" => Op::TimesScale(1_000_000_000_000),
        "quadrillion" => Op::TimesScale(1_000_000_000_000_000),
        "quintillion" => Op::TimesScale(1_000_000_000_000_000_000),
        "and" => Op::Ignore,
        "minus" => Op::Minus,
        _ => return None,
    })
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    if input.is_empty() {
        return Err(WordsError::Empty);
    }
    let b = input.as_bytes();
    if let Some((i, _)) = b.iter().enumerate().find(|(_, c)| !(c.is_ascii_lowercase() || **c == b' ' || **c == b'-')) {
        return Err(WordsError::InvalidByte { index: i });
    }
    for i in 0..b.len() {
        if b[i] == b' ' || b[i] == b'-' {
            if i == 0 || i + 1 == b.len() || !b[i - 1].is_ascii_lowercase() || !b[i + 1].is_ascii_lowercase() {
                return Err(WordsError::Separator { index: i });
            }
        }
    }

    let mut ops = Vec::new();
    let mut starts = Vec::new();
    let mut begin = 0;
    for i in 0..=b.len() {
        if i == b.len() || b[i] == b' ' || b[i] == b'-' {
            let op = word_op(&input[begin..i]).ok_or(WordsError::InvalidWord { index: begin })?;
            ops.push(op);
            starts.push(begin);
            begin = i + 1;
        }
    }

    let negative = matches!(ops.first(), Some(Op::Minus));
    for i in usize::from(negative)..ops.len() {
        if matches!(ops[i], Op::Minus) {
            return Err(WordsError::InvalidWord { index: starts[i] });
        }
    }

    let mut total = 0u128;
    let mut group = 0u128;
    for op in ops.iter().skip(usize::from(negative)) {
        match *op {
            Op::Add(v) => group = group.checked_add(v).ok_or(WordsError::Range)?,
            Op::TimesHundred => group = group.checked_mul(100).ok_or(WordsError::Range)?,
            Op::TimesScale(scale) => {
                total = total
                    .checked_add(group.checked_mul(scale).ok_or(WordsError::Range)?)
                    .ok_or(WordsError::Range)?;
                group = 0;
            }
            Op::Ignore | Op::Minus => {}
        }
        if total.checked_add(group).ok_or(WordsError::Range)? > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let mag = total.checked_add(group).ok_or(WordsError::Range)?;
    if mag > LIMIT {
        return Err(WordsError::Range);
    }
    let result = if negative { -(mag as i128) } else { mag as i128 };
    let canonical = words_encode(result)?;
    let a = input.as_bytes();
    let c = canonical.as_bytes();
    let n = a.len().min(c.len());
    for i in 0..n {
        if a[i] != c[i] {
            return Err(WordsError::NonCanonical { index: i });
        }
    }
    if a.len() != c.len() {
        return Err(WordsError::NonCanonical { index: n });
    }
    Ok(result)
}
```
