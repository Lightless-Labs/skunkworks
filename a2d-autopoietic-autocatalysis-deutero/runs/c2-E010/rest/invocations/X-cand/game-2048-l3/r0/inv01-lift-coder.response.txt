- Validate dimensions and cell count first, using checked arithmetic so extreme dimensions return `Shape`.
- Then scan every cell in row-major order; the lowest-index invalid tile takes precedence over any merge overflow.
- Scan each line from its destination edge, processing rows or columns in increasing order; merge equal nonzero inputs greedily once.
- On overflow, return the first overflowing pair in scan order and no partial result.
- For spawning, a full board consumes no RNG draws; otherwise draw twice and apply modulo in `u64`.
- For `play_turn`, spawn only after a changed move; an unchanged move preserves the input RNG.

### Solution A

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction { Up, Left, Down, Right }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Spawn {
    pub index: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SpawnResult {
    pub board: Board,
    pub rng: u64,
    pub spawn: Option<Spawn>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Turn {
    pub board: Board,
    pub rng: u64,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
    pub spawn: Option<Spawn>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }
    let Some(size) = board.width.checked_mul(board.height) else {
        return Err(Error::Shape);
    };
    if board.cells.len() != size {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn index(board: &Board, direction: Direction, line: usize, offset: usize) -> usize {
    match direction {
        Direction::Left => line * board.width + offset,
        Direction::Right => line * board.width + (board.width - 1 - offset),
        Direction::Up => offset * board.width + line,
        Direction::Down => (board.height - 1 - offset) * board.width + line,
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut cells = vec![0u32; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };
    let line_len = match direction {
        Direction::Left | Direction::Right => board.width,
        Direction::Up | Direction::Down => board.height,
    };

    for line in 0..line_count {
        let mut tiles: Vec<(usize, u32)> = Vec::new();
        for offset in 0..line_len {
            let src = index(board, direction, line, offset);
            let value = board.cells[src];
            if value != 0 {
                tiles.push((src, value));
            }
        }

        let mut out_offset = 0;
        let mut i = 0;
        while i < tiles.len() {
            let (src, value) = tiles[i];
            if i + 1 < tiles.len() && tiles[i + 1].1 == value {
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first: src, second: tiles[i + 1].0 });
                }
                let merged_value = value * 2;
                let dst = index(board, direction, line, out_offset);
                cells[dst] = merged_value;
                merges.push(Merge { from: [src, tiles[i + 1].0], to: dst, value: merged_value });
                score += u64::from(merged_value);
                out_offset += 1;
                i += 2;
            } else {
                let dst = index(board, direction, line, out_offset);
                cells[dst] = value;
                out_offset += 1;
                i += 1;
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board { width: board.width, height: board.height, cells },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut result = Vec::new();
    for direction in [Direction::Up, Direction::Left, Direction::Down, Direction::Right] {
        match move_board(board, direction) {
            Ok(moved) if moved.changed => result.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(result)
}

fn next_rng(rng: u64) -> u64 {
    rng.wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let empty: Vec<usize> = board.cells.iter().enumerate()
        .filter_map(|(i, &value)| if value == 0 { Some(i) } else { None })
        .collect();

    if empty.is_empty() {
        return Ok(SpawnResult { board: board.clone(), rng, spawn: None });
    }

    let a = next_rng(rng);
    let index = empty[(a % empty.len() as u64) as usize];
    let b = next_rng(a);
    let value = if b % 10 == 0 { 4 } else { 2 };

    let mut spawned = board.clone();
    spawned.cells[index] = value;
    Ok(SpawnResult {
        board: spawned,
        rng: b,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }
    let spawned = spawn_tile(&moved.board, rng)?;
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: moved.changed,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}
```

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction { Up, Left, Down, Right }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Spawn {
    pub index: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct SpawnResult {
    pub board: Board,
    pub rng: u64,
    pub spawn: Option<Spawn>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Turn {
    pub board: Board,
    pub rng: u64,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
    pub spawn: Option<Spawn>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }
    if board.cells.len() != board.width * board.height {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || value & (value - 1) != 0) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn flat(board: &Board, row: usize, col: usize) -> usize {
    row * board.width + col
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let (lines, length, vertical) = match direction {
        Direction::Up => (board.width, board.height, true),
        Direction::Down => (board.width, board.height, true),
        Direction::Left => (board.height, board.width, false),
        Direction::Right => (board.height, board.width, false),
    };

    for line in 0..lines {
        let mut compact = Vec::new();
        for offset in 0..length {
            let (row, col) = if vertical {
                let row = if direction == Direction::Down { board.height - 1 - offset } else { offset };
                (row, line)
            } else {
                let col = if direction == Direction::Right { board.width - 1 - offset } else { offset };
                (line, col)
            };
            let src = flat(board, row, col);
            let value = board.cells[src];
            if value != 0 {
                compact.push((src, value));
            }
        }

        let mut produced = Vec::new();
        let mut i = 0;
        while i < compact.len() {
            if i + 1 < compact.len() && compact[i].1 == compact[i + 1].1 {
                let (first, value) = compact[i];
                let second = compact[i + 1].0;
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first, second });
                }
                let merged = value * 2;
                let out_offset = produced.len();
                let (row, col) = if vertical {
                    let row = if direction == Direction::Down { board.height - 1 - out_offset } else { out_offset };
                    (row, line)
                } else {
                    let col = if direction == Direction::Right { board.width - 1 - out_offset } else { out_offset };
                    (line, col)
                };
                let dst = flat(board, row, col);
                produced.push(merged);
                merges.push(Merge { from: [first, second], to: dst, value: merged });
                score += u64::from(merged);
                i += 2;
            } else {
                produced.push(compact[i].1);
                i += 1;
            }
        }

        for (out_offset, value) in produced.into_iter().enumerate() {
            let (row, col) = if vertical {
                let row = if direction == Direction::Down { board.height - 1 - out_offset } else { out_offset };
                (row, line)
            } else {
                let col = if direction == Direction::Right { board.width - 1 - out_offset } else { out_offset };
                (line, col)
            };
            cells[flat(board, row, col)] = value;
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board { width: board.width, height: board.height, cells },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut legal = Vec::new();
    for d in [Direction::Up, Direction::Left, Direction::Down, Direction::Right] {
        match move_board(board, d) {
            Ok(result) if result.changed => legal.push(d),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(e) => return Err(e),
        }
    }
    Ok(legal)
}

fn draw(state: u64) -> u64 {
    state.wrapping_mul(6_364_136_223_846_793_005)
        .wrapping_add(1_442_695_040_888_963_407)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let empty_count = board.cells.iter().filter(|&&v| v == 0).count();
    if empty_count == 0 {
        return Ok(SpawnResult { board: board.clone(), rng, spawn: None });
    }

    let a = draw(rng);
    let rank = (a % empty_count as u64) as usize;
    let index = board.cells.iter().enumerate().filter(|(_, v)| **v == 0)
        .nth(rank).map(|(i, _)| i).unwrap();
    let b = draw(a);
    let value = if b % 10 == 0 { 4 } else { 2 };

    let mut cells = board.cells.clone();
    cells[index] = value;
    Ok(SpawnResult {
        board: Board { width: board.width, height: board.height, cells },
        rng: b,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(board, direction)?;
    if !moved.changed {
        return Ok(Turn {
            board: moved.board,
            rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }
    let spawned = spawn_tile(&moved.board, rng)?;
    Ok(Turn {
        board: spawned.board,
        rng: spawned.rng,
        score: moved.score,
        changed: true,
        merges: moved.merges,
        spawn: spawned.spawn,
    })
}
```
