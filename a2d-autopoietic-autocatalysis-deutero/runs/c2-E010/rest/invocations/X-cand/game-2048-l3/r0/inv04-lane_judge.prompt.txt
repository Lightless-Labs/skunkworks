    board: &Board,
    direction: Direction,
    line: usize,
    cells: &mut [u32],
    merges: &mut Vec<Merge>,
    score: &mut u64,
) -> Result<(), Error> {
    let count = match direction {
        Direction::Left | Direction::Right => board.width,
        Direction::Up | Direction::Down => board.height,
    };
    let mut indices = Vec::with_capacity(count);
    for offset in 0..count {
        let position = match direction {
            Direction::Left => line * board.width + offset,
            Direction::Right => line * board.width + (board.width - 1 - offset),
            Direction::Up => offset * board.width + line,
            Direction::Down => (board.height - 1 - offset) * board.width + line,
        };
        indices.push(position);

    let mut write = 0;
    while read < count {
        let first = indices[read];
        let value = board.cells[first];
        if value == 0 {
            read += 1;
            continue;
        }

        let mut next = read + 1;
        while next < count && board.cells[indices[next]] == 0 {
            next += 1;

        if next < count && board.cells[indices[next]] == value {
            if value == 2_147_483_648 {

            read += 1;
        }
        write += 1;
    }

            if value != 0 {

            }

    let mut out = vec![0u32; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    for line in 0..nlines {
        let mut vals: Vec<(u32, usize)> = (0..npos)
            .filter_map(|p| {
                let i = coords(board, direction, line, p);
                let v = board.cells[i];
                if v == 0 { None } else { Some((v, i)) }
            })
            .collect();

        let mut packed = 0usize;
        let mut p = 0usize;
        while p < vals.len() {
            let (v, source) = vals[p];
            if p + 1 < vals.len() && vals[p + 1].0 == v {
                let other = vals[p + 1].1;
                if v == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first: source, second: other });
                }
                let doubled = v * 2;
                let destination = coords(board, direction, line, packed);
                merges.push(Merge { from: [source, other], to: destination, value: doubled });
                score += doubled as u64;
                out[destination] = doubled;
                p += 2;
            } else {
                let destination = coords(board, direction, line, packed);
                out[destination] = v;
                p += 1;
            }
            packed += 1;
        }
    }

    let changed = out != board.cells;
    Ok(MoveResult {
        board: Board { width: board.width, height: board.height, cells: out },
        score: if changed { score } else { 0 },
        changed,
        merges: if changed { merges } else { Vec::new() },
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut legal = Vec::new();
    for d in [Direction::Up, Direction::Left, Direction::Down, Direction::Right] {
        if let Ok(m) = move_board(board, d) {
            if m.changed { legal.push(d); }
        }
    }
    Ok(legal)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let mut count = 0usize;
    for &v in &board.cells {
        if v == 0 { count += 1; }
    }
    if count == 0 {
        return Ok(SpawnResult { board: board.clone(), rng, spawn: None });
    }

    let mut state = rng;
    state = state.wrapping_mul(6_364_136_223_846_793_005).wrapping_add(1_442_695_040_888_963_407);
    let rank = (state % count as u64) as usize;
    let mut seen = 0usize;
    let mut index = 0usize;
    for (i, &v) in board.cells.iter().enumerate() {
        if v == 0 {
            if seen == rank { index = i; break; }
            seen += 1;
        }
    }
    state = state.wrapping_mul(6_364_136_223_846_793_005).wrapping_add(1_442_695_040_888_963_407);
    let value = if state % 10 == 0 { 4 } else { 2 };
    let mut cells = board.cells.clone();
    cells[index] = value;
    Ok(SpawnResult {
        board: Board { width: board.width, height: board.height, cells },
        rng: state,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {
    let moved = move_board(board, direction)?;
    if !moved.changed {
        return Ok(Turn { board: moved.board, rng, score: 0, changed: false, merges: Vec::new(), spawn: None });
    }
    let spawned = spawn_tile(&moved.board, rng)?;
    Ok(Turn { board: spawned.board, rng: spawned.rng, score: moved.score, changed: true, merges: moved.merges, spawn: spawned.spawn })
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    check(board)?;

    let mut output = vec![0u32; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let (lines, length) = match direction {
        Direction::Left | Direction::Right => (board.height, board.width),
        Direction::Up | Direction::Down => (board.width, board.height),
    };

    for line in 0..lines {
        let mut source = Vec::new();
        for offset in 0..length {
            let index = match direction {
                Direction::Left => line * board.width + offset,
                Direction::Right => line * board.width + board.width - 1 - offset,
                Direction::Up => offset * board.width + line,
                Direction::Down => (board.height - 1 - offset) * board.width + line,
            };
            if board.cells[index] != 0 {
                source.push((index, board.cells[index]));
            }
        }

        let mut write_offset = 0;
        let mut source_offset = 0;
        while source_offset < source.len() {
            let (original, value) = source[source_offset];
            let target = match direction {
                Direction::Left => line * board.width + write_offset,
                Direction::Right => line * board.width + board.width - 1 - write_offset,
                Direction::Up => write_offset * board.width + line,
                Direction::Down => (board.height - 1 - write_offset) * board.width + line,
            };

            if source_offset + 1 < source.len() && source[source_offset + 1].1 == value {
                let second = source[source_offset + 1].0;
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: original,
                        second,
                    });
                }
                let merged = value * 2;
                output[target] = merged;
                merges.push(Merge {
                    from: [original, second],
                    to: target,
                    value: merged,
                });
                score += merged as u64;
                source_offset += 2;
            } else {
                output[target] = value;
                source_offset += 1;
            }
            write_offset += 1;
        }
    }

    let changed = output != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells: output,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    check(board)?;
    let mut result = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(movement) if movement.changed => result.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),

    }
    Ok(result)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    check(board)?;
    let count = board.cells.iter().filter(|&&cell| cell == 0).count();
    if count == 0 {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let first_draw = advance(rng);
    let rank = (first_draw % count as u64) as usize;
    let mut seen = 0usize;
    let mut index = 0usize;
    for (position, &cell) in board.cells.iter().enumerate() {
        if cell == 0 {
            if seen == rank {
                index = position;
                break;
            }
            seen += 1;
        }
    }

    let second_draw = advance(first_draw);
    let value = if second_draw % 10 == 0 { 4 } else { 2 };
    let mut cells = board.cells.clone();
    cells[index] = value;

    Ok(SpawnResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        rng: second_draw,
        spawn: Some(Spawn { index, value }),
    })
}

fn check(board: &Board) -> Result<(), Error> {
    if board.width == 0
        || board.height == 0
        || board.width > 8
        || board.height > 8
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0
            && (value > 2_147_483_648 || !value.is_power_of_two())
        {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

}
