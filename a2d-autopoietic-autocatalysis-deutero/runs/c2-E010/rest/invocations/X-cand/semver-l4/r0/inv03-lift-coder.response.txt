- Core fields are validated for leading zeroes and overflow before checking the following separator; I follow that error order.
- In a pre identifier, invalid characters are reported before numeric leading-zero errors, and pre is fully parsed before build.
- `||` separates arms only as a standalone token; empty arms report the separator position, except a trailing arm reports input length.
- Partial, caret, tilde, and hyphen bounds use discrete SemVer precedence, including the `-0` boundary and finite `u64` core limits.
- Default prerelease admission is per arm and requires an explicit prerelease anchor at the candidate’s exact core tuple.
- Package validation is global by phase, and conflict labels are chosen by minimum cardinality, then ASCII lexicographic order.

### Solution A

```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_uint(bytes: &[u8], start: usize) -> Result<u64, VersionError> {
    let mut n = 0u64;
    for &b in bytes {
        n = n.checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(n)
}

fn is_ident_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut pos = start;
    loop {
        let ident_start = pos;
        while pos < end && bytes[pos] != b'.' {
            if !is_ident_byte(bytes[pos]) {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
        if pos == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
        let ident = &s[ident_start..pos];
        if pre && ident.len() > 1 && ident.starts_with('0')
            && ident.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        result.push(ident.to_owned());
        if pos == end {
            break;
        }
        pos += 1;
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut nums = [0u64; 3];
    for field in 0..3 {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }
        if pos == start {
            if pos == bytes.len() || matches!(bytes[pos], b'.' | b'-' | b'+') {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        nums[field] = parse_uint(&bytes[start..pos], start)?;
        if field < 2 {
            if pos == bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            if bytes[pos] != b'.' {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }
    }
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                let pre_start = pos + 1;
                let plus = bytes[pre_start..].iter().position(|&b| b == b'+')
                    .map(|i| pre_start + i);
                let pre_end = plus.unwrap_or(bytes.len());
                pre = parse_identifiers(s, pre_start, pre_end, true)?;
                if let Some(p) = plus {
                    build = parse_identifiers(s, p + 1, bytes.len(), false)?;
                }
            }
            b'+' => build = parse_identifiers(s, pos + 1, bytes.len(), false)?,
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }
    Ok(Version {
        major: nums[0], minor: nums[1], patch: nums[2], pre, build,
    })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let xn = numeric_id(x);
                    let yn = numeric_id(y);
                    let ord = match (xn, yn) {
                        (true, true) => cmp_numeric_id(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if ord != Ordering::Equal { return ord; }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Cmp {
    op: Op,
    v: Version,
    anchor: bool,
}
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op { Eq, Lt, Le, Gt, Ge }

#[derive(Clone, Debug)]
pub struct Range { arms: Vec<Vec<Cmp>> }

fn core(v: &Version) -> (u64, u64, u64) { (v.major, v.minor, v.patch) }

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version { major, minor, patch, pre: Vec::new(), build: Vec::new() }
}
fn endpoint_zero_pre(major: u64, minor: u64, patch: u64) -> Version {
    Version { major, minor, patch, pre: vec!["0".into()], build: Vec::new() }
}
fn cmp(op: Op, v: Version, anchor: bool) -> Cmp { Cmp { op, v, anchor } }

fn operand_full(s: &str, start: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at: start, error })
}

#[derive(Clone, Debug)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
}
fn parse_partial(s: &str, start: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        let v = operand_full(s, start)?;
        return Ok(Partial { nums: vec![v.major, v.minor, v.patch], full: Some(v) });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && fields.iter().all(|f| !matches!(*f, "*" | "x" | "X")) {
        let v = operand_full(s, start)?;
        return Ok(Partial { nums: vec![v.major, v.minor, v.patch], full: Some(v) });
    }
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: start });
    }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for f in fields {
        if f.is_empty() { return Err(RangeError::Syntax { at: start }); }
        if matches!(f, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
            {
                return Err(RangeError::Syntax { at: start });
            }
            let n = parse_uint(f.as_bytes(), start).map_err(|_| RangeError::Syntax { at: start })?;
            nums.push(n);
        }
    }
    Ok(Partial { nums, full: None })
}

fn padded(p: &Partial) -> Version {
    let mut n = [0u64; 3];
    for (i, x) in p.nums.iter().enumerate() { n[i] = *x; }
    let mut v = endpoint(n[0], n[1], n[2]);
    if let Some(full) = &p.full { v = full.clone(); }
    v
}

fn increment_bound(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut n = [v.major, v.minor, v.patch];
    n[field] = n[field].checked_add(1).ok_or(RangeError::Syntax { at })?;
    for x in n.iter_mut().skip(field + 1) { *x = 0; }
    Ok(endpoint_zero_pre(n[0], n[1], n[2]))
}

fn partial_upper(p: &Partial, start: usize) -> Result<Version, RangeError> {
    let v = padded(p);
    let f = if p.nums.len() <= 1 { 0 } else { 1 };
    increment_bound(&v, f, start)
}

fn expand_bare(s: &str, start: usize) -> Result<Vec<Cmp>, RangeError> {
    let p = parse_partial(s, start)?;
    if let Some(v) = p.full { return Ok(vec![cmp(Op::Eq, v.clone(), !v.pre.is_empty())]); }
    if p.nums.is_empty() { return Ok(Vec::new()); }
    let lo = padded(&p);
    let hi = partial_upper(&p, start)?;
    Ok(vec![cmp(Op::Ge, lo, false), cmp(Op::Lt, hi, false)])
}

fn expand_caret(s: &str, start: usize, token_start: usize) -> Result<Vec<Cmp>, RangeError> {
    let p = parse_partial(s, start)?;
    if p.nums.is_empty() { return Err(RangeError::Syntax { at: token_start }); }
    let lo = padded(&p);
    let field = if lo.major > 0 || p.nums.len() == 1 { 0 }
        else if lo.minor > 0 || p.nums.len() == 2 { 1 } else { 2 };
    let hi = increment_bound(&lo, field, token_start)?;
    Ok(vec![cmp(Op::Ge, lo.clone(), !lo.pre.is_empty()), cmp(Op::Lt, hi, false)])
}
fn expand_tilde(s: &str, start: usize, token_start: usize) -> Result<Vec<Cmp>, RangeError> {
    let p = parse_partial(s, start)?;
    if p.nums.is_empty() { return Err(RangeError::Syntax { at: token_start }); }
    let lo = padded(&p);
    let field = if p.nums.len() == 1 { 0 } else { 1 };
    let hi = increment_bound(&lo, field, token_start)?;
    Ok(vec![cmp(Op::Ge, lo.clone(), !lo.pre.is_empty()), cmp(Op::Lt, hi, false)])
}

fn compare_one(v: &Version, c: &Cmp) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.v) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.v) == Ordering::Less,
        Op::Le => compare_versions(v, &c.v) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.v) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.v) != Ordering::Less,
    }
}
fn ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

fn tokenize(s: &str) -> Vec<(usize, &str)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut p = 0;
    while p < b.len() {
        while p < b.len() && ws(b[p]) { p += 1; }
        if p == b.len() { break; }
        let st = p;
        while p < b.len() && !ws(b[p]) { p += 1; }
        out.push((st, &s[st..p]));
    }
    out
}
fn parse_full_cmp(tok: &str, start: usize) -> Result<Cmp, RangeError> {
    let (op, off) = if tok.starts_with(">=") { (Op::Ge, 2) }
        else if tok.starts_with("<=") { (Op::Le, 2) }
        else if tok.starts_with('>') { (Op::Gt, 1) }
        else if tok.starts_with('<') { (Op::Lt, 1) }
        else if tok.starts_with('=') { (Op::Eq, 1) }
        else { (Op::Eq, 0) };
    if off == tok.len() || !tok.as_bytes()[off].is_ascii_digit() {
        return Err(RangeError::Syntax { at: start });
    }
    let v = operand_full(&tok[off..], start + off)?;
    Ok(cmp(op, v.clone(), !v.pre.is_empty()))
}

fn parse_arm(tokens: &[(usize, &str)]) -> Result<Vec<Cmp>, RangeError> {
    if let Some(&(pos, _)) = tokens.iter().find(|(_, t)| *t == "-") {
        if tokens.len() != 3 || tokens[1].1 != "-" {
            return Err(RangeError::Syntax { at: pos });
        }
        let (ls, lt) = tokens[0];
        let (us, ut) = tokens[2];
        let lp = parse_partial(lt, ls)?;
        let up = parse_partial(ut, us)?;
        if lp.nums.is_empty() { return Err(RangeError::Syntax { at: ls }); }
        if up.nums.is_empty() { return Err(RangeError::Syntax { at: us }); }
        let lo = padded(&lp);
        let upper = if let Some(v) = up.full {
            v
        } else {
            partial_upper(&up, us)?
        };
        let mut out = vec![cmp(Op::Ge, lo.clone(), !lo.pre.is_empty())];
        if up.full.is_some() { out.push(cmp(Op::Le, upper.clone(), !upper.pre.is_empty())); }
        else { out.push(cmp(Op::Lt, upper, false)); }
        return Ok(out);
    }
    let mut out = Vec::new();
    for &(st, tok) in tokens {
        if tok.starts_with('^') {
            out.extend(expand_caret(&tok[1..], st + 1, st)?);
        } else if tok.starts_with('~') {
            out.extend(expand_tilde(&tok[1..], st + 1, st)?);
        } else if tok.starts_with(['<', '>', '=']) {
            out.push(parse_full_cmp(tok, st)?);
        } else if tok.starts_with(|c: char| c.is_ascii_digit() || matches!(c, '*' | 'x' | 'X')) {
            out.extend(expand_bare(tok, st)?);
        } else {
            return Err(RangeError::Syntax { at: st });
        }
    }
    Ok(out)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].1 == "||" {
            if current.is_empty() { return Err(RangeError::Empty { at: tokens[i].0 }); }
            arms.push(parse_arm(&current)?);
            current.clear();
            i += 1;
            if i == tokens.len() { return Err(RangeError::Empty { at: s.len() }); }
        } else {
            current.push(tokens[i]);
            i += 1;
        }
    }
    if current.is_empty() { return Err(RangeError::Empty { at: s.len() }); }
    arms.push(parse_arm(&current)?);
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &[Cmp], include: bool) -> bool {
    if !arm.iter().all(|c| compare_one(v, c)) { return false; }
    if include || v.pre.is_empty() { return true; }
    arm.iter().any(|c| c.anchor && c.v.pre.len() > 0 && core(&c.v) == core(v))
}
pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}
pub fn matches(v: &Version, r: &Range) -> bool { matches_with(v, r, false) }

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }
fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let parsed: Vec<Version> = candidates.iter().enumerate().map(|(i, s)| {
        parse_version(s).map_err(|error| CandidateError { index: i, error })
    }).collect::<Result<_, _>>()?;
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) { continue; }
        if let Some(j) = best {
            let ord = compare_versions(&parsed[i], &parsed[j]);
            if (max && ord == Ordering::Greater) || (!max && ord == Ordering::Less) { best = Some(i); }
        } else { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

fn core_successor(v: &Version) -> Option<Version> {
    if v.patch < u64::MAX { Some(endpoint(v.major, v.minor, v.patch + 1)) }
    else if v.minor < u64::MAX { Some(endpoint(v.major, v.minor + 1, 0)) }
    else if v.major < u64::MAX { Some(endpoint(v.major + 1, 0, 0)) }
    else { None }
}

fn arm_bounds(arm: &[Cmp]) -> (Option<(Version, bool)>, Option<(Version, bool)>) {
    let mut lo: Option<(Version, bool)> = None;
    let mut hi: Option<(Version, bool)> = None;
    for c in arm {
        match c.op {
            Op::Eq => {
                update_lower(&mut lo, c.v.clone(), true);
                update_upper(&mut hi, c.v.clone(), true);
            }
            Op::Gt | Op::Ge => update_lower(&mut lo, c.v.clone(), c.op == Op::Ge),
            Op::Lt | Op::Le => update_upper(&mut hi, c.v.clone(), c.op == Op::Le),
        }
    }
    (lo, hi)
}
fn update_lower(dst: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    match dst {
        None => *dst = Some((v, inclusive)),
        Some((old, inc)) => match compare_versions(&v, old) {
            Ordering::Greater => *dst = Some((v, inclusive)),
            Ordering::Equal => *inc &= inclusive,
            _ => {}
        }
    }
}
fn update_upper(dst: &mut Option<(Version, bool)>, v: Version, inclusive: bool) {
    match dst {
        None => *dst = Some((v, inclusive)),
        Some((old, inc)) => match compare_versions(&v, old) {
            Ordering::Less => *dst = Some((v, inclusive)),
            Ordering::Equal => *inc &= inclusive,
            _ => {}
        }
    }
}
fn bounds_have_release(lo: &Option<(Version, bool)>, hi: &Option<(Version, bool)>) -> bool {
    match (lo, hi) {
        (Some((l, li)), Some((h, hi))) => match compare_versions(l, h) {
            Ordering::Greater => false,
            Ordering::Equal => *li && *hi && l.pre.is_empty(),
            Ordering::Less => true,
        },
        _ => true,
    }
}
pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for aa in &a.arms {
        for bb in &b.arms {
            let mut both = aa.clone();
            both.extend(bb.clone());
            let (lo, hi) = arm_bounds(&both);
            if !bounds_have_release(&lo, &hi) { continue; }
            if include_prerelease { return true; }
            let stable_possible = match (&lo, &hi) {
                (Some((l, li)), Some((h, hi))) if compare_versions(l, h) == Ordering::Equal =>
                    *li && *hi && l.pre.is_empty(),
                _ => true,
            };
            if stable_possible { return true; }
            let anchors_a: Vec<_> = aa.iter().filter(|c| c.anchor).map(|c| &c.v).collect();
            let anchors_b: Vec<_> = bb.iter().filter(|c| c.anchor).map(|c| &c.v).collect();
            if anchors_a.iter().any(|x| anchors_b.iter().any(|y| core(x) == core(y))) {
                let tuple = core(anchors_a.iter().find(|x| anchors_b.iter().any(|y| core(x) == core(y))).unwrap());
                let candidate = Version { major: tuple.0, minor: tuple.1, patch: tuple.2, pre: vec!["0".into()], build: Vec::new() };
                if both.iter().all(|c| compare_one(&candidate, c)) { return true; }
            }
        }
    }
    false
}

pub fn canonical(r: &Range) -> String {
    let mut strings = Vec::new();
    for arm in &r.arms {
        let (lo, hi) = arm_bounds(arm);
        if !bounds_have_release(&lo, &hi) { continue; }
        let mut anchors: Vec<(u64,u64,u64)> = arm.iter().filter(|c| c.anchor).map(|c| core(&c.v)).collect();
        anchors.sort();
        anchors.dedup();
        let l = lo.map(|(v, inc)| format!("{}{}{}", if inc { ">=" } else { ">" }, version_text(&v), ""));
        let h = hi.map(|(v, inc)| format!("{}{}{}", if inc { "<=" } else { "<" }, version_text(&v), ""));
        let anchor_text = if anchors.is_empty() { "-".into() } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        strings.push(format!("[{};{};pre={}]", l.unwrap_or_else(|| "*".into()), h.unwrap_or_else(|| "*".into()), anchor_text));
    }
    strings.sort();
    strings.dedup();
    if strings.is_empty() { "EMPTY".into() } else { strings.join(" || ") }
}
fn version_text(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() { s.push('-'); s.push_str(&v.pre.join(".")); }
    s
}

#[derive(Clone, Debug)]
pub enum SetExpr { Range(String), Union(Vec<SetExpr>), Intersection(Vec<SetExpr>) }
#[derive(Clone, Debug)]
pub struct Constraint { pub label: String, pub expr: SetExpr }
#[derive(Clone, Debug)]
pub struct Package { pub name: String, pub candidates: Vec<String>, pub constraints: Vec<Constraint> }
#[derive(Clone, Copy, Debug)]
pub enum Preference { Highest, Lowest, StableFirst }
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection { pub package: String, pub version: String }
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr(
    expr: &SetExpr, package: &str, label: &str, path: &mut Vec<usize>,
    out: &mut Vec<Range>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_owned(), label: label.to_owned(), path: path.clone(), error
            })?;
            out.push(r);
        }
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, x) in xs.iter().enumerate() {
                path.push(i);
                validate_expr(x, package, label, path, out)?;
                path.pop();
            }
        }
    }
    Ok(())
}
fn eval_expr(expr: &SetExpr, version: &Version, include: bool, it: &mut std::slice::Iter<'_, Range>) -> bool {
    match expr {
        SetExpr::Range(_) => matches_with(version, it.next().unwrap(), include),
        SetExpr::Union(xs) => xs.iter().any(|x| eval_expr(x, version, include, it)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval_expr(x, version, include, it)),
    }
}
fn diagnostic(cands: &[Version], constraints: &[Constraint], include: bool) -> Vec<String> {
    if cands.is_empty() { return Vec::new(); }
    let n = constraints.len();
    for size in 1..=n {
        let mut best: Option<Vec<String>> = None;
        let mut chosen = Vec::new();
        fn search(
            start: usize, size: usize, chosen: &mut Vec<usize>, best: &mut Option<Vec<String>>,
            constraints: &[Constraint], cands: &[Version], include: bool
        ) {
            if chosen.len() == size {
                let ok = cands.iter().any(|v| chosen.iter().all(|&i| {
                    let mut dummy = 0;
                    eval_expr(&constraints[i].expr, v, include, &mut [].iter())
                }));
                if !ok {
                    let mut labels: Vec<String> = chosen.iter().map(|&i| constraints[i].label.clone()).collect();
                    labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
                    if best.as_ref().map_or(true, |b| labels.as_slice() < b.as_slice()) { *best = Some(labels); }
                }
                return;
            }
            let need = size - chosen.len();
            for i in start..=constraints.len().saturating_sub(need) {
                chosen.push(i); search(i+1,size,chosen,best,constraints,cands,include); chosen.pop();
            }
        }
        search(0,size,&mut chosen,&mut best,constraints,cands,include);
        if let Some(v) = best { return v; }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package], preference: Preference, include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    for i in 0..packages.len() {
        if packages[..i].iter().any(|p| p.name == packages[i].name) {
            return Err(SelectError::DuplicatePackage { name: packages[i].name.clone() });
        }
    }
    for p in packages {
        for i in 0..p.constraints.len() {
            if p.constraints[..i].iter().any(|c| c.label == p.constraints[i].label) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: p.constraints[i].label.clone() });
            }
        }
    }
    let mut parsed: Vec<Vec<Version>> = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (i, s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index: i, error
            })?);
        }
        parsed.push(vs);
    }
    let mut ranges_by_package: Vec<Vec<Vec<Range>>> = Vec::new();
    for p in packages {
        let mut constraints_ranges = Vec::new();
        for c in &p.constraints {
            let mut ranges = Vec::new();
            validate_expr(&c.expr, &p.name, &c.label, &mut Vec::new(), &mut ranges)?;
            constraints_ranges.push(ranges);
        }
        ranges_by_package.push(constraints_ranges);
    }
    let mut result = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let mut eligible = Vec::new();
        for (i, v) in parsed[pi].iter().enumerate() {
            let accepted = p.constraints.iter().enumerate().all(|(ci, c)| {
                let mut iter = ranges_by_package[pi][ci].iter();
                eval_expr(&c.expr, v, include_prerelease, &mut iter)
            });
            if accepted { eligible.push(i); }
        }
        if eligible.is_empty() {
            let labels = diagnostic(&parsed[pi], &p.constraints, include_prerelease);
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }
        let stable_exists = eligible.iter().any(|&i| parsed[pi][i].pre.is_empty());
        let mut best = eligible[0];
        for &i in eligible.iter().skip(1) {
            let vi = &parsed[pi][i];
            let vb = &parsed[pi][best];
            let better = match preference {
                Preference::Highest => compare_versions(vi, vb) == Ordering::Greater,
                Preference::Lowest => compare_versions(vi, vb) == Ordering::Less,
                Preference::StableFirst => {
                    let si = vi.pre.is_empty();
                    let sb = vb.pre.is_empty();
                    if stable_exists && si != sb { si }
                    else { compare_versions(vi, vb) == Ordering::Greater }
                }
            };
            if better { best = i; }
        }
        result.push(Selection { package: p.name.clone(), version: p.candidates[best].clone() });
    }
    Ok(result)
}
```
