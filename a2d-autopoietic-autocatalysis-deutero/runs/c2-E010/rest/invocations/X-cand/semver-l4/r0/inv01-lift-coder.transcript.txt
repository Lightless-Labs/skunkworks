fn digit(b: u8) -> bool { b.is_ascii_digit() }
fn allowed_id(b: u8) -> bool { b.is_ascii_alphanumeric() || b == b'-' }
fn numeric(s: &str) -> bool { !s.is_empty() && s.bytes().all(digit) }

fn parse_core_num(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && digit(bytes[*pos]) { *pos += 1; }
    if *pos == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let raw = &s[start..*pos];
    if raw.len() > 1 && raw.starts_with('0') { return Err(VersionError::LeadingZero { at: start }); }
    raw.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_ids(s: &str, start: usize, end: usize, pre: bool) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let q = p;
        while p < end && bytes[p] != b'.' {
            if !allowed_id(bytes[p]) { return Err(VersionError::InvalidChar { at: p }); }
            p += 1;
        }
        if p == q { return Err(VersionError::EmptyIdentifier { at: q }); }
        let id = &s[q..p];
        if pre && numeric(id) && id.len() > 1 && id.starts_with('0') {
            return Err(VersionError::LeadingZero { at: q });
        }
        out.push(id.to_string());
        if p == end { break; }
        p += 1;
        if p == end { return Err(VersionError::EmptyIdentifier { at: p }); }
    }
    Ok(out)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {

        (true,false) => return Ordering::Greater,
        (false,true) => return Ordering::Less,
        _ => {}
