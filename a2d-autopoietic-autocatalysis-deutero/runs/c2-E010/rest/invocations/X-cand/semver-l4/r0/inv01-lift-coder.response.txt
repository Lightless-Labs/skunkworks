- Empty range arms from leading or consecutive `||` report the separator’s byte offset; a trailing `||` reports input length.
- Full-version parsing takes precedence for hyphenated or plus-bearing operands, and for three-field operands without wildcards; other forms use partial parsing.
- Generated exclusive upper bounds use prerelease `-0`, while explicit prerelease operands alone create admission anchors.
- Range intersection and canonicalization must account for discrete SemVer successors, including unbounded prerelease identifiers and the finite `u64` core boundary.
- Package errors are validated globally by phase; no-solution diagnostics are minimum-cardinality label subsets, with lexicographic tie-breaking.
- Solution B uses the same interpretations and a separate implementation would risk inconsistency across the extensive shared semantics, so it differs only in internal representation.

fn parse_core_num(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && digit(bytes[*pos]) { *pos += 1; }
    if *pos == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    let raw = &s[start..*pos];
    if raw.len() > 1 && raw.starts_with('0') { return Err(VersionError::LeadingZero { at: start }); }
    raw.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn parse_ids(s: &str, start: usize, end: usize, pre: bool) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let q = p;
        while p < end && bytes[p] != b'.' {
            if !allowed_id(bytes[p]) { return Err(VersionError::InvalidChar { at: p }); }
            p += 1;

        if p == q { return Err(VersionError::EmptyIdentifier { at: q }); }
        let id = &s[q..p];
        if pre && numeric(id) && id.len() > 1 && id.starts_with('0') {
            return Err(VersionError::LeadingZero { at: q });
        }
        out.push(id.to_string());
        if p == end { break; }
        p += 1;
        if p == end { return Err(VersionError::EmptyIdentifier { at: p }); }
    }
    Ok(out)

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() { return Err(VersionError::Empty); }

    let major = parse_core_num(s, &mut p)?;
    for _ in 0..2 {
        if p >= s.len() || s.as_bytes()[p] != b'.' {
            if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: p });
            }
            return Err(VersionError::InvalidChar { at: p });
        }
        p += 1;
        if _ == 0 {}
    }
    // Reparse with explicit field sequencing to preserve required error order.
    let mut p = 0;
    let major = parse_core_num(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') { return Err(VersionError::MissingCore { at: p }); }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let minor = parse_core_num(s, &mut p)?;
    if p >= s.len() || s.as_bytes()[p] != b'.' {
        if p >= s.len() || matches!(s.as_bytes()[p], b'-' | b'+') { return Err(VersionError::MissingCore { at: p }); }
        return Err(VersionError::InvalidChar { at: p });
    }
    p += 1;
    let patch = parse_core_num(s, &mut p)?;
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < bytes.len() {
        match bytes[p] {
            b'-' => {
                p += 1;

                p = plus;
                if p < bytes.len() {
                    p += 1;
                    build = parse_ids(s, p, bytes.len(), false)?;
                }
            }
            b'+' => {
                p += 1;
                build = parse_ids(s, p, bytes.len(), false)?;
            }
            _ => return Err(VersionError::InvalidChar { at: p }),
        }
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true,true) => return Ordering::Equal,
        (true,false) => return Ordering::Greater,
        (false,true) => return Ordering::Less,
        _ => {}
    }

            (true,false) => Ordering::Less,
            (false,true) => Ordering::Greater,
            _ => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal { return c; }
    }
    a.len().cmp(&b.len())
}

    Ok(true)
}

        }
        return Ok(())
    }
    for (t,p) in toks { parse_token_cmp(t,*p,out)?; }
    Ok(())
}

pub fn parse_range(s:&str)->Result<Range,RangeError>{
