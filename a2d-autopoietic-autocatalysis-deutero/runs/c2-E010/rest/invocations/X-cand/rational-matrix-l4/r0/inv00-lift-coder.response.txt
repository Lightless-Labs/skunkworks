- **Parse precedence:** I validate the full byte grammar before checking canonical form, then apply the listed magnitude and canonicality checks in order.
- **Arithmetic overflow:** I reduce the mathematical result before checking whether it fits the `Rational` representation.
- **Matrix elimination:** I follow the specified pivot ordering and bounded scalar-operation sequence, including its temporary-overflow behavior.
- **Empty shapes:** I preserve explicit dimensions and avoid scanning unused rows or columns once elimination has no remaining pivot rows.
- **Solution B:** The requirements are specific enough that I use the same interpretation in both files; B uses an independent widened-integer arithmetic approach.

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let mut n = num;
        let mut d = den;
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128
            || n > i64::MAX as i128
            || d < 1
            || d > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        Self::from_i128(self.num as i128 * b + rhs.num as i128 * a, a * rhs.den as i128)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = self.den as i128 / g;
        let b = rhs.den as i128 / g;
        Self::from_i128(self.num as i128 * b - rhs.num as i128 * a, a * rhs.den as i128)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.den as u128);
        let g2 = gcd_u128(rhs.num.unsigned_abs() as u128, self.den as u128);
        Self::from_i128(
            (self.num as i128 / g1 as i128) * (rhs.num as i128 / g2 as i128),
            (self.den as i128 / g2 as i128) * (rhs.den as i128 / g1 as i128),
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        let g1 = gcd_u128(self.num.unsigned_abs() as u128, rhs.num.unsigned_abs() as u128);
        let g2 = gcd_u128(self.den as u128, rhs.den as u128);
        let n = (self.num as i128 / g1 as i128) * (rhs.den as i128 / g2 as i128);
        let d = (self.den as i128 / g2 as i128) * (rhs.num as i128 / g1 as i128);
        Self::from_i128(n, d)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() { return Err(fail(0, ParseKind::Empty)); }

        let mut i = 0;
        if bytes[0] == b'-' { i = 1; }
        let nstart = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
        if i == nstart {
            return Err(fail(i, if i < bytes.len() { ParseKind::InvalidByte } else { ParseKind::InvalidByte }));
        }
        let nend = i;
        let mut slash = None;
        let mut dstart = 0;
        let mut dend = 0;
        if i < bytes.len() {
            if bytes[i] != b'/' { return Err(fail(i, ParseKind::InvalidByte)); }
            slash = Some(i);
            i += 1;
            dstart = i;
            while i < bytes.len() && bytes[i].is_ascii_digit() { i += 1; }
            if i == dstart { return Err(fail(i, ParseKind::InvalidByte)); }
            dend = i;
            if i < bytes.len() { return Err(fail(i, ParseKind::InvalidByte)); }
        }

        let num_digits = nend - nstart;
        if num_digits >= 2 && bytes[nstart] == b'0' {
            return Err(fail(nstart + 1, ParseKind::NonCanonical));
        }
        if bytes[0] == b'-' && num_digits == 1 && bytes[nstart] == b'0' {
            return Err(fail(0, ParseKind::NonCanonical));
        }
        if slash.is_some() && dend - dstart >= 2 && bytes[dstart] == b'0' {
            return Err(fail(dstart + 1, ParseKind::NonCanonical));
        }

        let negative = bytes[0] == b'-';
        let mut magnitude: u128 = 0;
        let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        for &b in &bytes[nstart..nend] {
            magnitude = magnitude.saturating_mul(10).saturating_add((b - b'0') as u128);
            if magnitude > limit { return Err(fail(0, ParseKind::OutOfRange)); }
        }
        let num = if negative {
            if magnitude == (1u128 << 63) { i64::MIN }
            else { -(magnitude as i64) }
        } else { magnitude as i64 };

        let mut den = 1u64;
        if slash.is_some() {
            den = 0;
            for &b in &bytes[dstart..dend] {
                den = den.saturating_mul(10).saturating_add((b - b'0') as u64);
                if den > i64::MAX as u64 { return Err(fail(dstart, ParseKind::OutOfRange)); }
            }
            if den == 0 { return Err(fail(dstart, ParseKind::ZeroDenominator)); }
            if den == 1 { return Err(fail(slash.unwrap(), ParseKind::NonCanonical)); }
            if num == 0 { return Err(fail(slash.unwrap(), ParseKind::NonCanonical)); }
            if gcd_u128(num.unsigned_abs() as u128, den as u128) > 1 {
                return Err(fail(slash.unwrap(), ParseKind::NonCanonical));
            }
        }
        Ok(Self { num, den: den as i64 })
    }
}

pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }
    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }
    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }
    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows { data.push(self.data[r * self.cols + c]); }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let term = self.data[r * self.cols + k].mul(&rhs.data[k * rhs.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term)
                        .map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&self, candidates: usize) -> Result<(Self, Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        for col in 0..candidates {
            if p == m.rows { break; }
            let mut best: Option<(i64, u64, usize)> = None;
            for r in p..m.rows {
                let v = m.data[r * m.cols + col];
                if v.num == 0 { continue; }
                let key = (v.den, v.num.unsigned_abs(), r);
                if best.map_or(true, |b| key < b) { best = Some(key); }
            }
            let Some((_, _, q)) = best else { continue; };
            if q != p {
                for c in 0..m.cols { m.data.swap(p * m.cols + c, q * m.cols + c); }
                swaps.push((p, q));
            }
            let pivot = m.data[p * m.cols + col];
            for c in 0..m.cols {
                let idx = p * m.cols + c;
                m.data[idx] = m.data[idx].div(&pivot).map_err(|source|
                    MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
            }
            for r in 0..m.rows {
                if r == p { continue; }
                let factor = m.data[r * m.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..m.cols {
                    let product = factor.mul(&m.data[p * m.cols + c]).map_err(|source|
                        MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                    let idx = r * m.cols + c;
                    m.data[idx] = m.data[idx].sub(&product).map_err(|source|
                        MatrixError::Arithmetic { row: r, col: c, step: p, source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok((m, pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let (matrix, pivots, swaps) = self.eliminate(self.cols)?;
        Ok(Reduction { matrix, pivots, swaps })
    }
    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n { data.push(self.data[r * n + c]); }
            for c in 0..n { data.push(Rational { num: (r == c) as i64, den: 1 }); }
        }
        let aug = Self { rows: n, cols: width, data };
        let (red, pivots, _) = aug.eliminate(n)?;
        if pivots.len() < n { return Err(MatrixError::Singular { rank: pivots.len() }); }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            out.extend_from_slice(&red.data[r * width + n..r * width + width]);
        }
        Ok(Self { rows: n, cols: n, data: out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let width = self.cols.checked_add(rhs.cols).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * self.cols..(r + 1) * self.cols]);
            data.extend_from_slice(&rhs.data[r * rhs.cols..(r + 1) * rhs.cols]);
        }
        let aug = Self { rows: self.rows, cols: width, data };
        let (red, pivots, _) = aug.eliminate(self.cols).map_err(SolveError::Matrix)?;
        let rank = pivots.len();

        for r in 0..self.rows {
            let mut coeff_zero = true;
            for c in 0..self.cols {
                if red.data[r * width + c].num != 0 { coeff_zero = false; break; }
            }
            if coeff_zero {
                for c in 0..rhs.cols {
                    if red.data[r * width + self.cols + c].num != 0 {
                        let witness = red.data[r * width..(r + 1) * width].to_vec();
                        return Ok(Solution::None { row: r, rhs_col: c, witness });
                    }
                }
            }
        }

        let out_len = self.cols.checked_mul(rhs.cols)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular = vec![Rational { num: 0, den: 1 }; out_len];
        for (i, &pc) in pivots.iter().enumerate() {
            for c in 0..rhs.cols {
                particular[pc * rhs.cols + c] = red.data[i * width + self.cols + c];
            }
        }
        let particular = Self { rows: self.cols, cols: rhs.cols, data: particular };
        if rank == self.cols {
            return Ok(Solution::Unique { values: particular, pivots });
        }

        let mut free = Vec::new();
        for c in 0..self.cols {
            if !pivots.contains(&c) { free.push(c); }
        }
        let basis_len = self.cols.checked_mul(free.len())
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &fc) in free.iter().enumerate() {
            basis[fc * free.len() + j] = Rational { num: 1, den: 1 };
            for (i, &pc) in pivots.iter().enumerate() {
                let value = red.data[i * width + fc].neg().map_err(|source|
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pc, col: j, step: rank, source,
                    }))?;
                basis[pc * free.len() + j] = value;
            }
        }
        Ok(Solution::Infinite { particular, free, basis: Self { rows: self.cols, cols: free.len(), data: basis }, pivots })
    }
}

            let mut x=Rational{num:0,den:1};
            for k in 0..self.cols{
                let t=self.data[r*self.cols+k].mul(&b.data[k*b.cols+c]).map_err(|source|MatrixError::Arithmetic{row:r,col:c,step:k,source})?;
                x=x.add(&t).map_err(|source|MatrixError::Arithmetic{row:r,col:c,step:k,source})?;
            } d.push(x);
        }}
        Ok(Self{rows:self.rows,cols:b.cols,data:d})
    }
    fn eliminate(&self,candidates:usize)->Result<(Self,Vec<usize>,Vec<(usize,usize)>),MatrixError>{
        let mut m=self.clone();let mut piv=Vec::new();let mut sw=Vec::new();let mut p=0;
        for col in 0..candidates{
            if p==m.rows{break;}
            let mut best:Option<(i64,u64,usize)>=None;
            for r in p..m.rows{let v=m.data[r*m.cols+col];if v.num==0{continue;}let key=(v.den,v.num.unsigned_abs(),r);if best.map_or(true,|x|key<x){best=Some(key);}}
            let Some((_,_,q))=best else{continue;};
            if q!=p{for c in 0..m.cols{m.data.swap(p*m.cols+c,q*m.cols+c);}sw.push((p,q));}
            let pivot=m.data[p*m.cols+col];
            for c in 0..m.cols{let z=p*m.cols+c;m.data[z]=m.data[z].div(&pivot).map_err(|source|MatrixError::Arithmetic{row:p,col:c,step:p,source})?;}
            for r in 0..m.rows{if r==p{continue;}let factor=m.data[r*m.cols+col];if factor.num==0{continue;}
                for c in 0..m.cols{let prod=factor.mul(&m.data[p*m.cols+c]).map_err(|source|MatrixError::Arithmetic{row:r,col:c,step:p,source})?;let z=r*m.cols+c;m.data[z]=m.data[z].sub(&prod).map_err(|source|MatrixError::Arithmetic{row:r,col:c,step:p,source})?;}
            }
            piv.push(col);p+=1;
        } Ok((m,piv,sw))
    }
    pub fn rref(&self)->Result<Reduction,MatrixError>{let(m,p,s)=self.eliminate(self.cols)?;Ok(Reduction{matrix:m,pivots:p,swaps:s})}
    pub fn rank(&self)->Result<usize,MatrixError>{Ok(self.rref()?.rank())}
    pub fn inverse(&self)->Result<Self,MatrixError>{
        if self.rows!=self.cols{return Err(MatrixError::NotSquare{rows:self.rows,cols:self.cols});}
        let n=self.rows;let w=n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;let len=n.checked_mul(w).ok_or(MatrixError::ShapeOverflow)?;
        let mut d=Vec::with_capacity(len);
        for r in 0..n{d.extend_from_slice(&self.data[r*n..(r+1)*n]);for c in 0..n{d.push(Rational{num:(r==c) as i64,den:1});}}
        let(a,p,_)=Self{rows:n,cols:w,data:d}.eliminate(n)?;
        if p.len()<n{return Err(MatrixError::Singular{rank:p.len()});}
        let mut out=Vec::with_capacity(n*n);for r in 0..n{out.extend_from_slice(&a.data[r*w+n..r*w+w]);}
        Ok(Self{rows:n,cols:n,data:out})
    }
    pub fn solve(&self,rhs:&Self)->Result<Solution,SolveError>{
        if self.rows!=rhs.rows{return Err(SolveError::RowMismatch{coefficients:self.rows,rhs:rhs.rows});}
        let w=self.cols.checked_add(rhs.cols).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len=self.rows.checked_mul(w).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut d=Vec::with_capacity(len);
        for r in 0..self.rows{d.extend_from_slice(&self.data[r*self.cols..(r+1)*self.cols]);d.extend_from_slice(&rhs.data[r*rhs.cols..(r+1)*rhs.cols]);}
        let(a,p,_)=Self{rows:self.rows,cols:w,data:d}.eliminate(self.cols).map_err(SolveError::Matrix)?;
        for r in 0..self.rows{let mut z=true;for c in 0..self.cols{if a.data[r*w+c].num!=0{z=false;break;}}if z{for c in 0..rhs.cols{if a.data[r*w+self.cols+c].num!=0{return Ok(Solution::None{row:r,rhs_col:c,witness:a.data[r*w..(r+1)*w].to_vec()});}}}}
        let rank=p.len();
        let count=self.cols.checked_mul(rhs.cols).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut pd=vec![Rational{num:0,den:1};count];
        for(i,&pc)in p.iter().enumerate(){for c in 0..rhs.cols{pd[pc*rhs.cols+c]=a.data[i*w+self.cols+c];}}
        let particular=Self{rows:self.cols,cols:rhs.cols,data:pd};
        if rank==self.cols{return Ok(Solution::Unique{values:particular,pivots:p});}
        let mut free=Vec::new();for c in 0..self.cols{if !p.contains(&c){free.push(c);}}
        let bl=self.cols.checked_mul(free.len()).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut bd=vec![Rational{num:0,den:1};bl];
        for(j,&fc)in free.iter().enumerate(){bd[fc*free.len()+j]=Rational{num:1,den:1};for(i,&pc)in p.iter().enumerate(){bd[pc*free.len()+j]=a.data[i*w+fc].neg().map_err(|source|SolveError::Matrix(MatrixError::Arithmetic{row:pc,col:j,step:rank,source}))?;}}
        Ok(Solution::Infinite{particular,free,basis:Self{rows:self.cols,cols:free.len(),data:bd},pivots:p})
    }
}
#[derive(Clone,Debug,PartialEq,Eq)]
pub struct Reduction{pub matrix:Matrix,pub pivots:Vec<usize>,pub swaps:Vec<(usize,usize)>}
impl Reduction{pub fn rank(&self)->usize{self.pivots.len()}}
#[derive(Clone,Debug,PartialEq,Eq)]
pub enum SolveError{RowMismatch{coefficients:usize,rhs:usize},AugmentedShapeOverflow,Matrix(MatrixError)}
#[derive(Clone,Debug,PartialEq,Eq)]
pub enum Solution{
    Unique{values:Matrix,pivots:Vec<usize>},
    None{row:usize,rhs_col:usize,witness:Vec<Rational>},
    Infinite{particular:Matrix,free:Vec<usize>,basis:Matrix,pivots:Vec<usize>},
}
