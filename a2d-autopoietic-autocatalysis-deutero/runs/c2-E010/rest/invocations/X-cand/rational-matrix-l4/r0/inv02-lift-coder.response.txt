- Parse syntax is validated across the entire UTF-8 byte string before canonicality or range checks; I use the first missing or invalid byte offset specified.
- Arithmetic uses wider intermediates and reduces before checking representable bounds, while matrix elimination uses the required bounded operations in the specified order.
- Pivot selection orders by denominator, then unsigned numerator magnitude, then current row; elimination visits columns and rows in the prescribed order.
- Empty dimensions retain their explicit shapes; algorithms avoid traversing unused rows or columns when there are no stored entries.
- For solving, consistency is checked only after coefficient elimination, and basis negation follows the specified column-then-pivot order.
- Both implementations use the same interpretation of these requirements; Solution B uses an independent arithmetic and elimination implementation.

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 { let r = a % b; a = b; b = r; }
    a
}
fn mag_i64(n: i64) -> u128 { n.unsigned_abs() as u128 }

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 { return Err(RationalError::ZeroDenominator); }
        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 { n = -n; d = -d; }
        if n == 0 { return Ok(Self { num: 0, den: 1 }); }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g; d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }
    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }
    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {

                        self.den as i128 * rhs.den as i128)
    }
    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_wide(self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
                        self.den as i128 * rhs.den as i128)
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {

    }
    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {

        Self::from_wide(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }
    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_wide(-(self.num as i128), self.den as i128)
    }
    fn from_wide(n: i128, d: i128) -> Result<Self, RationalError> {
        if d == 0 { return Err(RationalError::ZeroDenominator); }
        let mut n = n; let mut d = d;
        if d < 0 { n = -n; d = -d; }
        if n == 0 { return Ok(Self { num: 0, den: 1 }); }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g; d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        fn err(at: usize, kind: ParseKind) -> RationalError { RationalError::Parse { at, kind } }
        let b = text.as_bytes();
        if b.is_empty() { return Err(err(0, ParseKind::Empty)); }
        let mut i = 0;
        if b[0] == b'-' { i = 1; }
        let ns = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == ns { return Err(err(i, ParseKind::InvalidByte)); }
        let ne = i;
        let mut slash = None;
        let mut ds = 0;
        let mut de = 0;
        if i < b.len() {
            if b[i] != b'/' { return Err(err(i, ParseKind::InvalidByte)); }
            slash = Some(i); i += 1; ds = i;
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == ds { return Err(err(i, ParseKind::InvalidByte)); }
            de = i;
            if i < b.len() { return Err(err(i, ParseKind::InvalidByte)); }
        }
        let digit_count = ne - ns;
        if digit_count >= 2 && b[ns] == b'0' {
            return Err(err(ns + 1, ParseKind::NonCanonical));
        }
        if b[0] == b'-' && digit_count == 1 && b[ns] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if slash.is_some() && de - ds >= 2 && b[ds] == b'0' {
            return Err(err(ds + 1, ParseKind::NonCanonical));
        }
        let mut nm: u128 = 0;
        for &c in &b[ns..ne] {
            nm = nm.saturating_mul(10).saturating_add((c-b'0') as u128);
            let lim = if b[0] == b'-' { 1u128 << 63 } else { i64::MAX as u128 };
            if nm > lim { return Err(err(0, ParseKind::OutOfRange)); }
        }
        let n = if b[0] == b'-' {
            if nm == 1u128 << 63 { i64::MIN } else { -(nm as i64) }
        } else { nm as i64 };
        let mut d = 1i64;
        if slash.is_some() {
            let mut dm: u128 = 0;
            for &c in &b[ds..de] {
                dm = dm.saturating_mul(10).saturating_add((c-b'0') as u128);
                if dm > i64::MAX as u128 { return Err(err(ds, ParseKind::OutOfRange)); }
            }
            d = dm as i64;
            if d == 0 { return Err(err(ds, ParseKind::ZeroDenominator)); }
            if d == 1 { return Err(err(slash.unwrap(), ParseKind::NonCanonical)); }
            if n == 0 { return Err(err(slash.unwrap(), ParseKind::NonCanonical)); }
            if gcd_u128(mag_i64(n), d as u128) > 1 {
                return Err(err(slash.unwrap(), ParseKind::NonCanonical));
            }
        }
        Ok(Self { num: n, den: d })
    }

pub enum MatrixError {
    ShapeOverflow, DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize }, Singular { rank: usize },
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix { rows: usize, cols: usize, data: Vec<Rational> }

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected { return Err(MatrixError::DataLength { expected, actual: data.len() }); }
        Ok(Self { rows, cols, data })
    }
    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }
    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols { return Err(MatrixError::Index { row, col }); }
        Ok(self.data[row * self.cols + col])
    }
    pub fn transpose(&self) -> Self {
        let mut d = Vec::with_capacity(self.data.len());
        for c in 0..self.cols { for r in 0..self.rows { d.push(self.data[r*self.cols+c]); } }
        Self { rows: self.cols, cols: self.rows, data: d }
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch { left:(self.rows,self.cols), right:(rhs.rows,rhs.cols) });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(len);
        for r in 0..self.rows { for c in 0..rhs.cols {
            let mut acc = Rational { num:0, den:1 };
            for k in 0..self.cols {
                let a = self.data[r*self.cols+k].mul(&rhs.data[k*rhs.cols+c])
                    .map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                acc = acc.add(&a).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
            }
            out.push(acc);
        }}
        Ok(Self { rows:self.rows, cols:rhs.cols, data:out })
    }
    fn eliminate(&mut self, candidates: usize, width: usize) -> Result<(Vec<usize>,Vec<(usize,usize)>),MatrixError> {
        let mut p=0; let mut pivots=Vec::new(); let mut swaps=Vec::new();
        for col in 0..candidates {
            if p == self.rows { break; }
            let mut chosen=None;
            for r in p..self.rows {
                let v=self.data[r*width+col];
                if v.num != 0 {
                    let key=(v.den,mag_i64(v.num),r);
                    if chosen.map_or(true, |(_,k)| key<k) { chosen=Some((r,key)); }
                }
            }
            let Some((q,_))=chosen else { continue };
            if q!=p {
                for c in 0..width { self.data.swap(p*width+c,q*width+c); }
                swaps.push((p,q));
            }
            let pivot=self.data[p*width+col];
            let mut norm=Vec::with_capacity(width);
            for c in 0..width {
                norm.push(self.data[p*width+c].div(&pivot).map_err(|source| MatrixError::Arithmetic {row:p,col:c,step:p,source})?);
            }
            for c in 0..width { self.data[p*width+c]=norm[c]; }
            for r in 0..self.rows {
                if r==p { continue; }
                let factor=self.data[r*width+col];
                if factor.num==0 { continue; }
                for c in 0..width {
                    let prod=factor.mul(&self.data[p*width+c]).map_err(|source| MatrixError::Arithmetic {row:r,col:c,step:p,source})?;
                    let val=self.data[r*width+c].sub(&prod).map_err(|source| MatrixError::Arithmetic {row:r,col:c,step:p,source})?;
                    self.data[r*width+c]=val;
                }
            }
            pivots.push(col); p+=1;
        }
        Ok((pivots,swaps))
    }
    pub fn rref(&self) -> Result<Reduction,MatrixError> {
        let mut m=self.clone(); let (pivots,swaps)=m.eliminate(self.cols,self.cols)?;
        Ok(Reduction {matrix:m,pivots,swaps})
    }
    pub fn rank(&self) -> Result<usize,MatrixError> { Ok(self.rref()?.rank()) }
    pub fn inverse(&self) -> Result<Self,MatrixError> {
        if self.rows!=self.cols { return Err(MatrixError::NotSquare {rows:self.rows,cols:self.cols}); }
        let n=self.rows; let width=n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len=n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut d=Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n { d.push(self.data[r*n+c]); }
            for c in 0..n { d.push(if r==c {Rational{num:1,den:1}} else {Rational{num:0,den:1}}); }
        }
        let mut aug=Self {rows:n,cols:width,data:d};
        let (pivots,_)=aug.eliminate(n,width)?;
        if pivots.len()<n { return Err(MatrixError::Singular {rank:pivots.len()}); }
        let mut out=Vec::with_capacity(len);
        for r in 0..n { for c in 0..n { out.push(aug.data[r*width+n+c]); } }
        Ok(Self {rows:n,cols:n,data:out})
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction { pub matrix: Matrix, pub pivots: Vec<usize>, pub swaps: Vec<(usize,usize)> }
impl Reduction { pub fn rank(&self)->usize { self.pivots.len() } }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError { RowMismatch { coefficients:usize, rhs:usize }, AugmentedShapeOverflow, Matrix(MatrixError) }
#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values:Matrix, pivots:Vec<usize> },
    None { row:usize, rhs_col:usize, witness:Vec<Rational> },
    Infinite { particular:Matrix, free:Vec<usize>, basis:Matrix, pivots:Vec<usize> },
}
impl Matrix {
    pub fn solve(&self, rhs:&Self)->Result<Solution,SolveError> {
        if self.rows!=rhs.rows { return Err(SolveError::RowMismatch {coefficients:self.rows,rhs:rhs.rows}); }
        let n=self.cols; let k=rhs.cols;
        let width=n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len=self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut d=Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..n { d.push(self.data[r*n+c]); }
            for c in 0..k { d.push(rhs.data[r*k+c]); }
        }
        let mut aug=Self {rows:self.rows,cols:width,data:d};
        let (pivots,_)=aug.eliminate(n,width).map_err(SolveError::Matrix)?;
        for r in 0..self.rows {
            let coeff_zero=(0..n).all(|c| aug.data[r*width+c].num==0);
            if coeff_zero {
                if let Some(c)=(0..k).find(|&c| aug.data[r*width+n+c].num!=0) {
                    return Ok(Solution::None {row:r,rhs_col:c,witness:aug.data[r*width..(r+1)*width].to_vec()});
                }
            }
        }
        let mut free=Vec::new(); let mut pi=0;
        for c in 0..n {
            if pi<pivots.len() && pivots[pi]==c { pi+=1; } else { free.push(c); }
        }
        let plen=n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular=vec![Rational{num:0,den:1};plen];
        for (i,&pc) in pivots.iter().enumerate() {
            for c in 0..k { particular[pc*k+c]=aug.data[i*width+n+c]; }
        }
        let particular=Self {rows:n,cols:k,data:particular};
        if pivots.len()==n { return Ok(Solution::Unique {values:particular,pivots}); }
        let blen=n.checked_mul(free.len()).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis=vec![Rational{num:0,den:1};blen];
        for (j,&fc) in free.iter().enumerate() {
            basis[fc*free.len()+j]=Rational{num:1,den:1};
            for (i,&pc) in pivots.iter().enumerate() {
                let v=aug.data[i*width+fc].neg().map_err(|source| SolveError::Matrix(MatrixError::Arithmetic {row:pc,col:j,step:pivots.len(),source}))?;
                basis[pc*free.len()+j]=v;
            }
        }
        Ok(Solution::Infinite {particular,free,basis:Self {rows:n,cols:free.len(),data:basis},pivots})
    }
}

            let pivot=self.data[p*width+col];
            let mut row=Vec::with_capacity(width);
            for c in 0..width{row.push(self.data[p*width+c].div(&pivot).map_err(|source|MatrixError::Arithmetic{row:p,col:c,step:p,source})?)}
            for c in 0..width{self.data[p*width+c]=row[c]}
            for r in 0..self.rows{
                if r==p{continue}
                let factor=self.data[r*width+col];
                if factor.num==0{continue}
                for c in 0..width{
                    let product=factor.mul(&self.data[p*width+c]).map_err(|source|MatrixError::Arithmetic{row:r,col:c,step:p,source})?;
                    self.data[r*width+c]=self.data[r*width+c].sub(&product).map_err(|source|MatrixError::Arithmetic{row:r,col:c,step:p,source})?;
                }
            }
            pivots.push(col);p+=1;
        }
        Ok((pivots,swaps))
    }
    pub fn rref(&self)->Result<Reduction,MatrixError>{
        let mut m=self.clone();let(pivots,swaps)=m.eliminate(self.cols,self.cols)?;
        Ok(Reduction{matrix:m,pivots,swaps})
    }
    pub fn rank(&self)->Result<usize,MatrixError>{Ok(self.rref()?.rank())}
    pub fn inverse(&self)->Result<Self,MatrixError>{
        if self.rows!=self.cols{return Err(MatrixError::NotSquare{rows:self.rows,cols:self.cols})}
        let n=self.rows;let width=n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len=n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut v=Vec::with_capacity(len);
        for r in 0..n{for c in 0..n{v.push(self.data[r*n+c])}for c in 0..n{v.push(if r==c{Rational{num:1,den:1}}else{Rational{num:0,den:1}})}}
        let mut m=Self{rows:n,cols:width,data:v};let(pivots,_)=m.eliminate(n,width)?;
        if pivots.len()<n{return Err(MatrixError::Singular{rank:pivots.len()})}
        let mut out=Vec::with_capacity(len);
        for r in 0..n{for c in 0..n{out.push(m.data[r*width+n+c])}}
        Ok(Self{rows:n,cols:n,data:out})
    }
}
#[derive(Clone,Debug,PartialEq,Eq)]
pub struct Reduction{pub matrix:Matrix,pub pivots:Vec<usize>,pub swaps:Vec<(usize,usize)>}
impl Reduction{pub fn rank(&self)->usize{self.pivots.len()}}

#[derive(Clone,Debug,PartialEq,Eq)]
pub enum SolveError{RowMismatch{coefficients:usize,rhs:usize},AugmentedShapeOverflow,Matrix(MatrixError)}
#[derive(Clone,Debug,PartialEq,Eq)]
pub enum Solution{
    Unique{values:Matrix,pivots:Vec<usize>},
    None{row:usize,rhs_col:usize,witness:Vec<Rational>},
    Infinite{particular:Matrix,free:Vec<usize>,basis:Matrix,pivots:Vec<usize>},
}
impl Matrix{
    pub fn solve(&self,rhs:&Self)->Result<Solution,SolveError>{
        if self.rows!=rhs.rows{return Err(SolveError::RowMismatch{coefficients:self.rows,rhs:rhs.rows})}
        let n=self.cols;let k=rhs.cols;
        let width=n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let total=self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut v=Vec::with_capacity(total);
        for r in 0..self.rows{for c in 0..n{v.push(self.data[r*n+c])}for c in 0..k{v.push(rhs.data[r*k+c])}}
        let mut aug=Self{rows:self.rows,cols:width,data:v};
        let(pivots,_)=aug.eliminate(n,width).map_err(SolveError::Matrix)?;
        for r in 0..self.rows{
            if (0..n).all(|c|aug.data[r*width+c].num==0){
                if let Some(c)=(0..k).find(|&c|aug.data[r*width+n+c].num!=0){
                    return Ok(Solution::None{row:r,rhs_col:c,witness:aug.data[r*width..(r+1)*width].to_vec()})
                }
            }
        }
        let mut free=Vec::new();let mut j=0;
        for c in 0..n{if j<pivots.len()&&pivots[j]==c{j+=1}else{free.push(c)}}
        let plen=n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut pv=vec![Rational{num:0,den:1};plen];
        for(i,&pc)in pivots.iter().enumerate(){for c in 0..k{pv[pc*k+c]=aug.data[i*width+n+c]}}
        let particular=Self{rows:n,cols:k,data:pv};
        if pivots.len()==n{return Ok(Solution::Unique{values:particular,pivots})}
        let blen=n.checked_mul(free.len()).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let f=free.len();let mut bv=vec![Rational{num:0,den:1};blen];
        for(col,&fc)in free.iter().enumerate(){
            bv[fc*f+col]=Rational{num:1,den:1};
            for(i,&pc)in pivots.iter().enumerate(){
                bv[pc*f+col]=aug.data[i*width+fc].neg().map_err(|source|SolveError::Matrix(MatrixError::Arithmetic{row:pc,col,step:pivots.len(),source}))?;
            }
        }
        Ok(Solution::Infinite{particular,free,basis:Self{rows:n,cols:f,data:bv},pivots})
    }
}
