```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1].split('\n').collect()
    }
}

fn validate(text: &str, side: Side) -> Result<(), DiffError> {
    if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(DiffError::CarriageReturn { side, byte });
    }
    if !text.is_empty() && !text.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side });
    }
    Ok(())
}

fn make_script(old: &[&str], new: &[&str]) -> Vec<Edit> {
    let (m, n) = (old.len(), new.len());
    let mut lcs = vec![vec![0usize; n + 1]; m + 1];

    for i in (0..m).rev() {
        for j in (0..n).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::new();
    let (mut i, mut j) = (0, 0);
    while i < m && j < n {
        if old[i] == new[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].to_owned(),
            });
            i += 1;
            j += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            if del > ins || (del == ins && (i + j) % 2 == 0) {
                edits.push(Edit::Delete {
                    old_index: i,
                    line: old[i].to_owned(),
                });
                i += 1;
            } else {
                edits.push(Edit::Insert {
                    new_index: j,
                    line: new[j].to_owned(),
                });
                j += 1;
            }
        }
    }
    while i < m {
        edits.push(Edit::Delete {
            old_index: i,
            line: old[i].to_owned(),
        });
        i += 1;
    }
    while j < n {
        edits.push(Edit::Insert {
            new_index: j,
            line: new[j].to_owned(),
        });
        j += 1;
    }
    edits
}

fn is_change(edit: &Edit) -> bool {
    !matches!(edit, Edit::Keep { .. })
}

fn consumed_before(edits: &[Edit], end: usize) -> (usize, usize) {
    let mut old = 0;
    let mut new = 0;
    for edit in &edits[..end] {
        match edit {
            Edit::Keep { .. } => {
                old += 1;
                new += 1;
            }
            Edit::Delete { .. } => old += 1,
            Edit::Insert { .. } => new += 1,
        }
    }
    (old, new)
}

fn write_hunk(out: &mut String, edits: &[Edit], start: usize, end: usize) {
    let (old_before, new_before) = consumed_before(edits, start);
    let mut old_count = 0;
    let mut new_count = 0;
    for edit in &edits[start..end] {
        match edit {
            Edit::Keep { .. } => {
                old_count += 1;
                new_count += 1;
            }
            Edit::Delete { .. } => old_count += 1,
            Edit::Insert { .. } => new_count += 1,
        }
    }
    let old_start = if old_count > 0 {
        old_before + 1
    } else {
        old_before
    };
    let new_start = if new_count > 0 {
        new_before + 1
    } else {
        new_before
    };

    use std::fmt::Write;
    writeln!(
        out,
        "@@ -{},{} +{},{} @@",
        old_start, old_count, new_start, new_count
    )
    .unwrap();

    for edit in &edits[start..end] {
        match edit {
            Edit::Keep { line, .. } => {
                out.push(' ');
                out.push_str(line);
                out.push('\n');
            }
            Edit::Delete { line, .. } => {
                out.push('-');
                out.push_str(line);
                out.push('\n');
            }
            Edit::Insert { line, .. } => {
                out.push('+');
                out.push_str(line);
                out.push('\n');
            }
        }
    }
}

pub fn unified_diff(old_text: &str, new_text: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }
    validate(old_text, Side::Old)?;
    validate(new_text, Side::New)?;

    let old = split_lines(old_text);
    let new = split_lines(new_text);
    if old.len() > 128 || new.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old.len(),
            new_len: new.len(),
        });
    }

    let edits = make_script(&old, &new);
    if !edits.iter().any(is_change) {
        return Ok(String::new());
    }

    let mut groups = Vec::new();
    let mut pos = 0;
    while pos < edits.len() {
        if !is_change(&edits[pos]) {
            pos += 1;
            continue;
        }
        let start = pos;
        while pos < edits.len() && is_change(&edits[pos]) {
            pos += 1;
        }
        let end = pos;
        let mut left = start;
        let mut right = end;
        while left > 0 && matches!(edits[left - 1], Edit::Keep { .. }) {
            left -= 1;
        }
        while right < edits.len() && matches!(edits[right], Edit::Keep { .. }) {
            right += 1;
        }
        groups.push((start, end, left, right));
    }

    let mut merged: Vec<(usize, usize)> = Vec::new();
    for (start, end, _, _) in groups {
        if let Some(last) = merged.last_mut() {
            let keeps_between = edits[last.1..start]
                .iter()
                .filter(|e| matches!(e, Edit::Keep { .. }))
                .count();
            if keeps_between <= 2 * context {
                last.1 = end;
                continue;
            }
        }
        merged.push((start, end));
    }

    let mut out = String::from("--- before\n+++ after\n");
    for (start, end) in merged {
        let mut hunk_start = start;
        let mut remaining = context;
        while hunk_start > 0 && remaining > 0 {
            if matches!(edits[hunk_start - 1], Edit::Keep { .. }) {
                hunk_start -= 1;
                remaining -= 1;
            } else {
                break;
            }
        }

        let mut hunk_end = end;
        remaining = context;
        while hunk_end < edits.len() && remaining > 0 {
            if matches!(edits[hunk_end], Edit::Keep { .. }) {
                hunk_end += 1;
                remaining -= 1;
            } else {
                break;
            }
        }
        write_hunk(&mut out, &edits, hunk_start, hunk_end);
    }
    Ok(out)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn lines(text: &str) -> Vec<&str> {
    match text.strip_suffix('\n') {
        Some(body) => body.split('\n').collect(),
        None if text.is_empty() => Vec::new(),
        None => unreachable!(),
    }
}

pub fn unified_diff(old_text: &str, new_text: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }
    for (text, side) in [(old_text, Side::Old), (new_text, Side::New)] {
        if let Some(byte) = text.bytes().position(|b| b == b'\r') {
            return Err(DiffError::CarriageReturn { side, byte });
        }
    }
    if !old_text.is_empty() && !old_text.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::Old });
    }
    if !new_text.is_empty() && !new_text.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::New });
    }

    let old = lines(old_text);
    let new = lines(new_text);
    if old.len() > 128 || new.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old.len(),
            new_len: new.len(),
        });
    }
    let (m, n) = (old.len(), new.len());
    let mut table = vec![vec![0usize; n + 1]; m + 1];
    for i in (0..m).rev() {
        for j in (0..n).rev() {
            table[i][j] = if old[i] == new[j] {
                table[i + 1][j + 1] + 1
            } else {
                table[i + 1][j].max(table[i][j + 1])
            };
        }
    }

    let mut script = Vec::new();
    let (mut i, mut j) = (0usize, 0usize);
    while i < m && j < n {
        if old[i] == new[j] {
            script.push(Edit::Keep { old_index: i, new_index: j, line: old[i].to_owned() });
            i += 1;
            j += 1;
        } else if table[i + 1][j] > table[i][j + 1]
            || (table[i + 1][j] == table[i][j + 1] && (i + j) % 2 == 0)
        {
            script.push(Edit::Delete { old_index: i, line: old[i].to_owned() });
            i += 1;
        } else {
            script.push(Edit::Insert { new_index: j, line: new[j].to_owned() });
            j += 1;
        }
    }
    while i < m {
        script.push(Edit::Delete { old_index: i, line: old[i].to_owned() });
        i += 1;
    }
    while j < n {
        script.push(Edit::Insert { new_index: j, line: new[j].to_owned() });
        j += 1;
    }
    if script.iter().all(|e| matches!(e, Edit::Keep { .. })) {
        return Ok(String::new());
    }

    let mut groups = Vec::<(usize, usize)>::new();
    let mut k = 0;
    while k < script.len() {
        if matches!(script[k], Edit::Keep { .. }) {
            k += 1;
        } else {
            let start = k;
            while k < script.len() && !matches!(script[k], Edit::Keep { .. }) {
                k += 1;
            }
            groups.push((start, k));
        }
    }

    let mut combined = Vec::<(usize, usize)>::new();
    for (start, end) in groups {
        if let Some((_, previous_end)) = combined.last_mut() {
            let keep_count = script[*previous_end..start]
                .iter()
                .filter(|e| matches!(e, Edit::Keep { .. }))
                .count();
            if keep_count <= context * 2 {
                *previous_end = end;
                continue;
            }
        }
        combined.push((start, end));
    }

    let mut output = String::from("--- before\n+++ after\n");
    for (start, end) in combined {
        let mut a = start;
        for _ in 0..context {
            if a > 0 && matches!(script[a - 1], Edit::Keep { .. }) {
                a -= 1;
            } else {
                break;
            }
        }
        let mut b = end;
        for _ in 0..context {
            if b < script.len() && matches!(script[b], Edit::Keep { .. }) {
                b += 1;
            } else {
                break;
            }
        }

        let mut old_before = 0;
        let mut new_before = 0;
        for e in &script[..a] {
            match e {
                Edit::Keep { .. } => { old_before += 1; new_before += 1; }
                Edit::Delete { .. } => old_before += 1,
                Edit::Insert { .. } => new_before += 1,
            }
        }
        let (mut old_count, mut new_count) = (0, 0);
        for e in &script[a..b] {
            match e {
                Edit::Keep { .. } => { old_count += 1; new_count += 1; }
                Edit::Delete { .. } => old_count += 1,
                Edit::Insert { .. } => new_count += 1,
            }
        }
        let old_start = if old_count == 0 { old_before } else { old_before + 1 };
        let new_start = if new_count == 0 { new_before } else { new_before + 1 };
        use std::fmt::Write;
        writeln!(output, "@@ -{},{} +{},{} @@", old_start, old_count, new_start, new_count).unwrap();
        for e in &script[a..b] {
            let (prefix, line) = match e {
                Edit::Keep { line, .. } => (' ', line),
                Edit::Delete { line, .. } => ('-', line),
                Edit::Insert { line, .. } => ('+', line),
            };
            output.push(prefix);
            output.push_str(line);
            output.push('\n');
        }
    }
    Ok(output)
}
```
