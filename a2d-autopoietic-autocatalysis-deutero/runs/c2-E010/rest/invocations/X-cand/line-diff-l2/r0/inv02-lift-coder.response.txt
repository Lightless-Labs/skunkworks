```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    ContextTooLarge { context: usize },
    CarriageReturn { side: Side, byte: usize },
    MissingFinalLf { side: Side },
    TooLarge { old_len: usize, new_len: usize },
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1].split('\n').collect()
    }
}

fn validate<'a>(
    old: &'a str,
    new: &'a str,
    context: usize,
) -> Result<(Vec<&'a str>, Vec<&'a str>), DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }
    if let Some(byte) = old.find('\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::Old,
            byte,
        });
    }
    if let Some(byte) = new.find('\r') {
        return Err(DiffError::CarriageReturn {
            side: Side::New,
            byte,
        });
    }
    if !old.is_empty() && !old.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::Old });
    }
    if !new.is_empty() && !new.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::New });
    }

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);
    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }
    Ok((old_lines, new_lines))
}

fn edit_script(old: &[&str], new: &[&str]) -> Vec<Edit> {
    let (m, n) = (old.len(), new.len());
    let mut lcs = vec![vec![0usize; n + 1]; m + 1];

    for i in (0..m).rev() {
        for j in (0..n).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0, 0);
    let mut edits = Vec::new();
    while i < m && j < n {
        if old[i] == new[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].to_owned(),
            });
            i += 1;
            j += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            if delete_score > insert_score
                || (delete_score == insert_score && (i + j) % 2 == 0)
            {
                edits.push(Edit::Delete {
                    old_index: i,
                    line: old[i].to_owned(),
                });
                i += 1;
            } else {
                edits.push(Edit::Insert {
                    new_index: j,
                    line: new[j].to_owned(),
                });
                j += 1;
            }
        }
    }

    while i < m {
        edits.push(Edit::Delete {
            old_index: i,
            line: old[i].to_owned(),
        });
        i += 1;
    }
    while j < n {
        edits.push(Edit::Insert {
            new_index: j,
            line: new[j].to_owned(),
        });
        j += 1;
    }
    edits
}

fn is_change(edit: &Edit) -> bool {
    !matches!(edit, Edit::Keep { .. })
}

fn hunk_header(edits: &[Edit], start: usize, end: usize) -> (usize, usize, usize, usize) {
    let mut old_count = 0;
    let mut new_count = 0;
    for edit in &edits[start..end] {
        match edit {
            Edit::Keep { .. } => {
                old_count += 1;
                new_count += 1;
            }
            Edit::Delete { .. } => old_count += 1,
            Edit::Insert { .. } => new_count += 1,
        }
    }

    let mut old_before = 0;
    let mut new_before = 0;
    for edit in &edits[..start] {
        match edit {
            Edit::Keep { .. } => {
                old_before += 1;
                new_before += 1;
            }
            Edit::Delete { .. } => old_before += 1,
            Edit::Insert { .. } => new_before += 1,
        }
    }

    let old_start = if old_count > 0 {
        old_before + 1
    } else {
        old_before
    };
    let new_start = if new_count > 0 {
        new_before + 1
    } else {
        new_before
    };
    (old_start, old_count, new_start, new_count)
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    let (old_lines, new_lines) = validate(old, new, context)?;
    let edits = edit_script(&old_lines, &new_lines);
    if !edits.iter().any(is_change) {
        return Ok(String::new());
    }

    let mut groups = Vec::new();
    let mut pos = 0;
    while pos < edits.len() {
        if !is_change(&edits[pos]) {
            pos += 1;
            continue;
        }
        let start = pos;
        while pos < edits.len() && is_change(&edits[pos]) {
            pos += 1;
        }
        groups.push((start, pos));
    }

    let merge_gap = context * 2;
    let mut merged: Vec<(usize, usize)> = Vec::new();
    for group in groups {
        if let Some(last) = merged.last_mut() {
            let keeps_between = group.0 - last.1;
            if keeps_between <= merge_gap {
                last.1 = group.1;
                continue;
            }
        }
        merged.push(group);
    }

    let mut output = String::from("--- before\n+++ after\n");
    for (change_start, change_end) in merged {
        let start = change_start.saturating_sub(context);
        let end = (change_end + context).min(edits.len());
        let (old_start, old_count, new_start, new_count) =
            hunk_header(&edits, start, end);
        output.push_str(&format!(
            "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
        ));

        for edit in &edits[start..end] {
            match edit {
                Edit::Keep { line, .. } => {
                    output.push(' ');
                    output.push_str(line);
                    output.push('\n');
                }
                Edit::Delete { line, .. } => {
                    output.push('-');
                    output.push_str(line);
                    output.push('\n');
                }
                Edit::Insert { line, .. } => {
                    output.push('+');
                    output.push_str(line);
                    output.push('\n');
                }
            }
        }
    }
    Ok(output)
}
``` 

```rust
fn script(old: &[&str], new: &[&str]) -> Vec<Edit> {
    let m = old.len();
    let n = new.len();
    let mut lcs = vec![vec![0usize; n + 1]; m + 1];
    for i in (0..m).rev() {
        for j in (0..n).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])

        if old[i] == new[j] {
            out.push(Edit::Keep { old_index: i, new_index: j, line: old[i].to_owned() });
            i += 1;
            j += 1;
        } else if lcs[i + 1][j] > lcs[i][j + 1]
            || (lcs[i + 1][j] == lcs[i][j + 1] && (i + j) % 2 == 0)

        j += 1;
    }
    out
}

fn changed(edit: &Edit) -> bool {
    !matches!(edit, Edit::Keep { .. })
}

fn counts_and_starts(edits: &[Edit], start: usize, end: usize) -> (usize, usize, usize, usize) {
    let mut old_before = 0usize;
    let mut new_before = 0usize;
    for edit in &edits[..start] {
        match edit {
            Edit::Keep { .. } => { old_before += 1; new_before += 1; }
            Edit::Delete { .. } => old_before += 1,
            Edit::Insert { .. } => new_before += 1,
        }
    }

    let mut old_count = 0usize;
    let mut new_count = 0usize;
    for edit in &edits[start..end] {
        match edit {
            Edit::Keep { .. } => { old_count += 1; new_count += 1; }
            Edit::Delete { .. } => old_count += 1,
            Edit::Insert { .. } => new_count += 1,
        }
    }
    let old_start = if old_count == 0 { old_before } else { old_before + 1 };
    let new_start = if new_count == 0 { new_before } else { new_before + 1 };
    (old_start, old_count, new_start, new_count)
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    let (old_lines, new_lines) = validate(old, new, context)?;
    let edits = script(&old_lines, &new_lines);
    if !edits.iter().any(changed) {
        return Ok(String::new());
    }

    let mut groups: Vec<(usize, usize)> = Vec::new();
    let mut i = 0;
    while i < edits.len() {
        if !changed(&edits[i]) {
            i += 1;
            continue;
        }
        let start = i;
        while i < edits.len() && changed(&edits[i]) {
            i += 1;
        }
        groups.push((start, i));
    }

    let mut hunks: Vec<(usize, usize)> = Vec::new();
    for group in groups {
        if let Some(last) = hunks.last_mut() {
            if group.0 - last.1 <= 2 * context {
                last.1 = group.1;
                continue;
            }
        }
        hunks.push(group);
    }

    let mut result = String::from("--- before\n+++ after\n");
    for (change_start, change_end) in hunks {
        let start = change_start.saturating_sub(context);
        let end = (change_end + context).min(edits.len());
        let (os, oc, ns, nc) = counts_and_starts(&edits, start, end);
        result.push_str(&format!("@@ -{os},{oc} +{ns},{nc} @@\n"));
        for edit in &edits[start..end] {
            match edit {
                Edit::Keep { line, .. } => { result.push(' '); result.push_str(line); result.push('\n'); }
                Edit::Delete { line, .. } => { result.push('-'); result.push_str(line); result.push('\n'); }
                Edit::Insert { line, .. } => { result.push('+'); result.push_str(line); result.push('\n'); }
            }
        }
    }
    Ok(result)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {

    MissingFinalLf { side: Side },

fn split_lines(text: &str) -> Vec<&str> {

fn checked<'a>(old: &'a str, new: &'a str, context: usize) -> Result<(Vec<&'a str>, Vec<&'a str>), DiffError> {
    if context > 8 { return Err(DiffError::ContextTooLarge { context }); }
    if let Some(byte) = old.find('\r') { return Err(DiffError::CarriageReturn { side: Side::Old, byte }); }
    if let Some(byte) = new.find('\r') { return Err(DiffError::CarriageReturn { side: Side::New, byte }); }
    if !old.is_empty() && !old.ends_with('\n') { return Err(DiffError::MissingFinalLf { side: Side::Old }); }
    if !new.is_empty() && !new.ends_with('\n') { return Err(DiffError::MissingFinalLf { side: Side::New }); }
    let a = split_lines(old);
    let b = split_lines(new);
    if a.len() > 128 || b.len() > 128 {
        return Err(DiffError::TooLarge { old_len: a.len(), new_len: b.len() });
    }
    Ok((a, b))
}

fn make_script(a: &[&str], b: &[&str]) -> Vec<Edit> {
    let mut table = [[0usize; 129]; 129];
    for i in (0..a.len()).rev() {
        for j in (0..b.len()).rev() {
            table[i][j] = if a[i] == b[j] {
                table[i + 1][j + 1] + 1
            } else {
                table[i + 1][j].max(table[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut out = Vec::new();
    while i < a.len() && j < b.len() {
        if a[i] == b[j] {
            out.push(Edit::Keep { old_index: i, new_index: j, line: a[i].to_owned() });
            i += 1;
            j += 1;
        } else {
            let d = table[i + 1][j];
            let ins = table[i][j + 1];
            if d > ins || (d == ins && (i + j) % 2 == 0) {
                out.push(Edit::Delete { old_index: i, line: a[i].to_owned() });
                i += 1;
            } else {
                out.push(Edit::Insert { new_index: j, line: b[j].to_owned() });
                j += 1;
            }
        }
    }
    while i < a.len() {
        out.push(Edit::Delete { old_index: i, line: a[i].to_owned() });
        i += 1;
    }
    while j < b.len() {
        out.push(Edit::Insert { new_index: j, line: b[j].to_owned() });
        j += 1;
    }
    out
}

fn is_keep(e: &Edit) -> bool { matches!(e, Edit::Keep { .. }) }

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    let (a, b) = checked(old, new, context)?;
    let script = make_script(&a, &b);
    if script.iter().all(is_keep) { return Ok(String::new()); }

    let mut intervals = Vec::<(usize, usize)>::new();
    let mut p = 0;
    while p < script.len() {
        if is_keep(&script[p]) { p += 1; continue; }
        let s = p;
        while p < script.len() && !is_keep(&script[p]) { p += 1; }
        intervals.push((s, p));
    }

    let mut merged = Vec::<(usize, usize)>::new();
    for (s, e) in intervals {
        if let Some(prev) = merged.last_mut() {
            let between = s - prev.1;
            if between <= context * 2 {
                prev.1 = e;
                continue;
            }
        }
        merged.push((s, e));
    }

    let mut output = String::from("--- before\n+++ after\n");
    for (s, e) in merged {
        let lo = s.saturating_sub(context);
        let hi = (e + context).min(script.len());
        let mut old_before = 0usize;
        let mut new_before = 0usize;
        for row in &script[..lo] {
            match row {
                Edit::Keep { .. } => { old_before += 1; new_before += 1; }
                Edit::Delete { .. } => old_before += 1,
                Edit::Insert { .. } => new_before += 1,
            }
        }
        let mut old_count = 0usize;
        let mut new_count = 0usize;
        for row in &script[lo..hi] {
            match row {
                Edit::Keep { .. } => { old_count += 1; new_count += 1; }
                Edit::Delete { .. } => old_count += 1,
                Edit::Insert { .. } => new_count += 1,
            }
        }
        let old_start = if old_count == 0 { old_before } else { old_before + 1 };
        let new_start = if new_count == 0 { new_before } else { new_before + 1 };
        output.push_str(&format!("@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"));
        for row in &script[lo..hi] {
            match row {
                Edit::Keep { line, .. } => { output.push(' '); output.push_str(line); output.push('\n'); }
                Edit::Delete { line, .. } => { output.push('-'); output.push_str(line); output.push('\n'); }
                Edit::Insert { line, .. } => { output.push('+'); output.push_str(line); output.push('\n'); }
            }
        }
    }
    Ok(output)
}
```
