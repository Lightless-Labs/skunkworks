pub fn unified_diff(old_text: &str, new_text: &str, context: usize) -> Result<String, DiffError> {

        return Err(DiffError::ContextTooLarge { context });
    }

    validate(new_text, Side::New)?;

    let old = split_lines(old_text);
    let new = split_lines(new_text);
    if old.len() > 128 || new.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old.len(),
            new_len: new.len(),
        });
    }

    let edits = make_script(&old, &new);
    if !edits.iter().any(is_change) {
        return Ok(String::new());
    }

    let mut groups = Vec::new();
    let mut pos = 0;
    while pos < edits.len() {
        if !is_change(&edits[pos]) {
            pos += 1;
            continue;
        }
        let start = pos;
        while pos < edits.len() && is_change(&edits[pos]) {
            pos += 1;
        }
        let end = pos;
        let mut left = start;
        let mut right = end;
        while left > 0 && matches!(edits[left - 1], Edit::Keep { .. }) {
            left -= 1;
        }
        while right < edits.len() && matches!(edits[right], Edit::Keep { .. }) {
            right += 1;
        }
        groups.push((start, end, left, right));
    }

    let mut merged: Vec<(usize, usize)> = Vec::new();
    for (start, end, _, _) in groups {
        if let Some(last) = merged.last_mut() {
            let keeps_between = edits[last.1..start]
                .iter()
                .filter(|e| matches!(e, Edit::Keep { .. }))
                .count();
            if keeps_between <= 2 * context {
                last.1 = end;
                continue;
            }
        }
        merged.push((start, end));
    }

    let mut out = String::from("--- before\n+++ after\n");
    for (start, end) in merged {
        let mut hunk_start = start;
        let mut remaining = context;
        while hunk_start > 0 && remaining > 0 {
            if matches!(edits[hunk_start - 1], Edit::Keep { .. }) {
                hunk_start -= 1;
                remaining -= 1;
            } else {
                break;
            }
        }

        let mut hunk_end = end;
        remaining = context;
        while hunk_end < edits.len() && remaining > 0 {
            if matches!(edits[hunk_end], Edit::Keep { .. }) {
                hunk_end += 1;
                remaining -= 1;
            } else {
                break;
            }
        }
        write_hunk(&mut out, &edits, hunk_start, hunk_end);
    }
    Ok(out)
}

fn script(a: &[&str], b: &[&str]) -> Vec<Edit> {
    let (m, n) = (a.len(), b.len());
    let mut dp = vec![0usize; (m + 1) * (n + 1)];
    let at = |i: usize, j: usize| i * (n + 1) + j;

    for i in (0..m).rev() {
        for j in (0..n).rev() {
            dp[at(i, j)] = if a[i] == b[j] {
                dp[at(i + 1, j + 1)] + 1
            } else {
                dp[at(i + 1, j)].max(dp[at(i, j + 1)])
            };
        }
    }

    let mut result = Vec::new();
    let (mut i, mut j) = (0, 0);
    while i < m && j < n {
        if a[i] == b[j] {
            result.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: a[i].to_string(),
            });
            i += 1;
            j += 1;
        } else if dp[at(i + 1, j)] > dp[at(i, j + 1)]
            || (dp[at(i + 1, j)] == dp[at(i, j + 1)] && (i + j) % 2 == 0)
        {
            result.push(Edit::Delete {
                old_index: i,
                line: a[i].to_string(),
            });
            i += 1;
        } else {
            result.push(Edit::Insert {
                new_index: j,
                line: b[j].to_string(),
            });
            j += 1;
        }
    }
    while i < m {
        result.push(Edit::Delete { old_index: i, line: a[i].to_string() });
        i += 1;
    }
    while j < n {
        result.push(Edit::Insert { new_index: j, line: b[j].to_string() });
        j += 1;
    }
    result
}

    }
    if !new.is_empty() && !new.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::New });
    }

    let a = split_lines(old);
    let b = split_lines(new);
    if a.len() > 128 || b.len() > 128 {
        return Err(DiffError::TooLarge { old_len: a.len(), new_len: b.len() });
    }

    let rows = script(&a, &b);
    if rows.iter().all(is_keep) {
        return Ok(String::new());
    }

    let mut groups = Vec::new();
    let mut pos = 0;
    while pos < rows.len() {
        if is_keep(&rows[pos]) {
            pos += 1;
            continue;
        }
        let start = pos;
        while pos < rows.len() && !is_keep(&rows[pos]) {
            pos += 1;
        }
        groups.push((start, pos));
    }

    let mut combined: Vec<(usize, usize)> = Vec::new();
    for (start, end) in groups {
        if let Some(prev) = combined.last_mut() {
            if start - prev.1 <= context * 2 {
                prev.1 = end;
                continue;
            }
        }
        combined.push((start, end));
    }

    let mut output = String::from("--- before\n+++ after\n");
    for (start, end) in combined {
        let mut first = start;
        let mut remaining = context;
        while first > 0 && remaining > 0 && is_keep(&rows[first - 1]) {
            first -= 1;
            remaining -= 1;
        }
        let mut last = end;
        remaining = context;
        while last < rows.len() && remaining > 0 && is_keep(&rows[last]) {
            last += 1;
            remaining -= 1;
        }
        render(&mut output, &rows, first, last);
    }
    Ok(output)
}

fn canonical_script(old: &[&str], new: &[&str]) -> Vec<Edit> {
    let m = old.len();
    let n = new.len();
    let width = n + 1;
    let mut lcs = vec![0usize; (m + 1) * (n + 1)];

    for i in (0..m).rev() {
        for j in (0..n).rev() {
            let value = if old[i] == new[j] {
                1 + lcs[(i + 1) * width + j + 1]
            } else {
                lcs[(i + 1) * width + j].max(lcs[i * width + j + 1])
            };
            lcs[i * width + j] = value;
        }
    }

    let mut edits = Vec::with_capacity(m + n);
    let (mut i, mut j) = (0, 0);
    while i < m && j < n {
        if old[i] == new[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].to_owned(),
            });
            i += 1;
            j += 1;
        } else {
            let delete_score = lcs[(i + 1) * width + j];
            let insert_score = lcs[i * width + j + 1];
            if delete_score > insert_score
                || (delete_score == insert_score && (i + j) % 2 == 0)
            {
                edits.push(Edit::Delete {
                    old_index: i,
                    line: old[i].to_owned(),
                });
                i += 1;
            } else {
                edits.push(Edit::Insert {
                    new_index: j,
                    line: new[j].to_owned(),
                });
                j += 1;
            }
        }
    }

    while i < m {
        edits.push(Edit::Delete {
            old_index: i,
            line: old[i].to_owned(),
        });
        i += 1;
    }
    while j < n {
        edits.push(Edit::Insert {
            new_index: j,
            line: new[j].to_owned(),
        });
        j += 1;
    }
    edits
}

fn row_is_change(edit: &Edit) -> bool {
    !matches!(edit, Edit::Keep { .. })
}

fn consumed_before(edits: &[Edit], end: usize) -> (usize, usize) {
    let mut old = 0;
    let mut new = 0;
    for edit in &edits[..end] {
        match edit {
            Edit::Keep { .. } => {
                old += 1;
                new += 1;
            }
            Edit::Delete { .. } => old += 1,
            Edit::Insert { .. } => new += 1,
        }
    }
    (old, new)
}

fn render_hunk(out: &mut String, edits: &[Edit], start: usize, end: usize) {
    let (old_before, new_before) = consumed_before(edits, start);
    let mut old_count = 0usize;
    let mut new_count = 0usize;

    for edit in &edits[start..end] {
        match edit {
            Edit::Keep { .. } => {
                old_count += 1;
                new_count += 1;
            }
            Edit::Delete { .. } => old_count += 1,
            Edit::Insert { .. } => new_count += 1,
        }
    }

    let old_start = if old_count > 0 { old_before + 1 } else { old_before };
    let new_start = if new_count > 0 { new_before + 1 } else { new_before };
    out.push_str(&format!(
        "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
    ));

    for edit in &edits[start..end] {
        let (prefix, line) = match edit {
            Edit::Keep { line, .. } => (' ', line),
            Edit::Delete { line, .. } => ('-', line),
            Edit::Insert { line, .. } => ('+', line),
        };
        out.push(prefix);
        out.push_str(line);
        out.push('\n');
    }
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }
    if let Some(byte) = first_cr(old) {
        return Err(DiffError::CarriageReturn { side: Side::Old, byte });
    }
    if let Some(byte) = first_cr(new) {
        return Err(DiffError::CarriageReturn { side: Side::New, byte });
    }
    if !old.is_empty() && !old.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::Old });
    }
    if !new.is_empty() && !new.ends_with('\n') {
        return Err(DiffError::MissingFinalLf { side: Side::New });
    }

    let old_lines = lines(old);
    let new_lines = lines(new);
    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    let edits = canonical_script(&old_lines, &new_lines);
    if edits.iter().all(|edit| !row_is_change(edit)) {
        return Ok(String::new());
    }

    let mut groups = Vec::<(usize, usize)>::new();
    let mut k = 0;
    while k < edits.len() {
        if row_is_change(&edits[k]) {
            let start = k;
            while k < edits.len() && row_is_change(&edits[k]) {
                k += 1;
            }
            groups.push((start, k));
        } else {
            k += 1;
        }
    }

    let merge_gap = 2 * context;
    let mut merged = Vec::<(usize, usize)>::new();
    for (start, end) in groups {
        if let Some(last) = merged.last_mut() {
            let keep_gap = start - last.1;
            if keep_gap <= merge_gap {
                last.1 = end;
                continue;
            }
        }
        merged.push((start, end));
    }

    let mut out = String::from("--- before\n+++ after\n");
    for (group_start, group_end) in merged {
        let mut start = group_start;
        let mut before = context;
        while start > 0 && before > 0 {
            if matches!(edits[start - 1], Edit::Keep { .. }) {
                start -= 1;
                before -= 1;
            } else {
                break;
            }
        }

        let mut end = group_end;
        let mut after = context;
        while end < edits.len() && after > 0 {
            if matches!(edits[end], Edit::Keep { .. }) {
                end += 1;
                after -= 1;
            } else {
                break;
            }
        }

        render_hunk(&mut out, &edits, start, end);
    }
    Ok(out)
}

        out.push(Edit::Insert {
            new_index: j,
            line: new[j].to_string(),
        });
        j += 1;
    }
    out
}

fn is_keep(edit: &Edit) -> bool {
    matches!(edit, Edit::Keep { .. })
}

pub fn unified_diff(old: &str, new: &str, context: usize) -> Result<String, DiffError> {
    if context > 8 {
        return Err(DiffError::ContextTooLarge { context });
    }
    validate(old, Side::Old)?;
    validate(new, Side::New)?;

    let old_lines = split_lines(old);
    let new_lines = split_lines(new);
    if old_lines.len() > 128 || new_lines.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old_lines.len(),
            new_len: new_lines.len(),
        });
    }

    let script = edits(&old_lines, &new_lines);
    let changed: Vec<usize> = script
        .iter()
        .enumerate()
        .filter_map(|(i, e)| (!is_keep(e)).then_some(i))
        .collect();
    if changed.is_empty() {
        return Ok(String::new());
    }

    let merge_gap = 2 * context;
    let mut groups = Vec::<(usize, usize)>::new();
    let mut start = changed[0];
    let mut end = changed[0];
    for &idx in &changed[1..] {
        let keep_gap = script[end + 1..idx].iter().filter(|e| is_keep(e)).count();
        if keep_gap <= merge_gap {
            end = idx;
        } else {
            groups.push((start, end));
            start = idx;
            end = idx;
        }
    }
    groups.push((start, end));

    let mut hunks = Vec::<(usize, usize)>::new();
    for (gs, ge) in groups {
        let hs = gs.saturating_sub(context);
        let mut he = (ge + context + 1).min(script.len());
        while he > ge + 1 && !is_keep(&script[he - 1]) {
            break;
        }
        if let Some(last) = hunks.last_mut() {
            if hs <= last.1 {
                last.1 = last.1.max(he);
                continue;
            }
        }
        hunks.push((hs, he));
    }

    let mut result = String::from("--- before\n+++ after\n");
    for (hs, he) in hunks {
        let mut old_count = 0usize;
        let mut new_count = 0usize;
        let mut old_before = 0usize;
        let mut new_before = 0usize;
        for e in &script[..hs] {
            match e {
                Edit::Keep { .. } => {
                    old_before += 1;
                    new_before += 1;
                }
                Edit::Delete { .. } => old_before += 1,
                Edit::Insert { .. } => new_before += 1,
            }
        }
        for e in &script[hs..he] {
            match e {
                Edit::Keep { .. } => {
                    old_count += 1;
                    new_count += 1;
                }
                Edit::Delete { .. } => old_count += 1,
                Edit::Insert { .. } => new_count += 1,
            }
        }
        let old_start = if old_count > 0 {
            old_before + 1
        } else {
            old_before
        };
        let new_start = if new_count > 0 {
            new_before + 1
        } else {
            new_before
        };
        result.push_str(&format!(
            "@@ -{old_start},{old_count} +{new_start},{new_count} @@\n"
        ));
        for e in &script[hs..he] {
            match e {
                Edit::Keep { line, .. } => {
                    result.push(' ');
                    result.push_str(line);
                    result.push('\n');
                }
                Edit::Delete { line, .. } => {
                    result.push('-');
                    result.push_str(line);
                    result.push('\n');
                }
                Edit::Insert { line, .. } => {
                    result.push('+');
                    result.push_str(line);
                    result.push('\n');
                }
            }
        }
    }
    Ok(result)
}
