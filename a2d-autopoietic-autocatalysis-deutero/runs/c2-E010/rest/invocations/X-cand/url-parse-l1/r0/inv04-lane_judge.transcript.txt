    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: bytes.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map_or(bytes.len(), |offset| authority_start + offset);
    let authority = &bytes[authority_start..authority_end];

    let mut at_sign = None;
    for (offset, &byte) in authority.iter().enumerate() {
        if byte == b'@' {
            if at_sign.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: authority_start + offset,
                });
            }
            at_sign = Some(offset);
        }
    }

    let (userinfo, host_port_start) = match at_sign {
        Some(offset) => {
            if offset == 0 {
                return Err(ParseError::InvalidUserInfo { at: authority_start });
            }
            (
                Some(input[authority_start..authority_start + offset].to_owned()),
                authority_start + offset + 1,
            )
        }
        None => (None, authority_start),
    };

    let host_port = &bytes[host_port_start..authority_end];
    let host_colon = host_port.iter().position(|&b| b == b':');
    let host_end = host_colon.map_or(authority_end, |offset| host_port_start + offset);

    if host_end == host_port_start {
        return Err(ParseError::InvalidHost { at: host_port_start });
    }

    for (offset, &byte) in bytes[host_port_start..host_end].iter().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_port_start + offset,
            });
        }
    }

    let port = if let Some(offset) = host_colon {
        let port_start = host_port_start + offset + 1;
        if port_start == authority_end {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        let port_bytes = &bytes[port_start..authority_end];
        if let Some(nondigit) = port_bytes.iter().position(|b| !b.is_ascii_digit()) {
            return Err(ParseError::InvalidPort {
                at: port_start + nondigit,
            });

        if port_bytes.len() >= 2 && port_bytes[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_start });

                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u32))

                None => return Err(ParseError::InvalidPort { at: port_start }),
            };
        }
        if !(1..=65535).contains(&value) {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        Some(value as u16)
    } else {
        None
    };

        .iter()
        .position(|&b| b == b'#')
        .map(|offset| authority_end + offset);

    let before_fragment_end = fragment_start.unwrap_or(bytes.len());

        .iter()
        .position(|&b| b == b'?')
        .map(|offset| authority_end + offset);

        input[authority_end..path_end].to_owned()

        userinfo,
        host: input[host_port_start..host_end].to_ascii_lowercase(),
        port,
        path,
        query,
        fragment,
    })

    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let b = input.as_bytes();
    for (i, &x) in b.iter().enumerate() {
        if x < 0x21 || x > 0x7e {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let c = match b.iter().position(|&x| x == b':') {
        Some(c) => c,
        None => return Err(ParseError::InvalidScheme { at: b.len() }),
    };
    if c == 0 || !b[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..c {
        let x = b[i];
        if !x.is_ascii_alphanumeric() && x != b'+' && x != b'-' && x != b'.' {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    let a = c + 1;
    if b.get(a) != Some(&b'/') || b.get(a + 1) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: a });
    }

    let start = a + 2;
    let end = b[start..]
        .iter()
        .position(|&x| x == b'/' || x == b'?' || x == b'#')
        .map_or(b.len(), |p| start + p);

    let mut at = None;
    for (i, &x) in b[start..end].iter().enumerate() {
        if x == b'@' {
            if at.is_some() {
                return Err(ParseError::InvalidUserInfo { at: start + i });
            }
            at = Some(i);
        }
    }

    let (userinfo, hp_start) = match at {
        Some(0) => return Err(ParseError::InvalidUserInfo { at: start }),
        Some(i) => (
            Some(input[start..start + i].to_owned()),
            start + i + 1,
        ),
        None => (None, start),
    };

    let colon_in_hp = b[hp_start..end].iter().position(|&x| x == b':');
    let hp_end = colon_in_hp.map_or(end, |i| hp_start + i);
    if hp_end == hp_start {
        return Err(ParseError::InvalidHost { at: hp_start });
    }

    for (i, &x) in b[hp_start..hp_end].iter().enumerate() {
        if !x.is_ascii_alphanumeric() && x != b'.' && x != b'-' && x != b'_' {
            return Err(ParseError::InvalidHost { at: hp_start + i });
        }
    }

    let port = match colon_in_hp {
        None => None,
        Some(i) => {
            let p = hp_start + i + 1;
            if p == end {
                return Err(ParseError::InvalidPort { at: p });
            }

            for (offset, &x) in b[p..end].iter().enumerate() {
                if !x.is_ascii_digit() {
                    return Err(ParseError::InvalidPort { at: p + offset });
                }
            }
            if end - p >= 2 && b[p] == b'0' {
                return Err(ParseError::InvalidPort { at: p });
            }

            let mut n: u32 = 0;
            for &x in &b[p..end] {
                n = match n.checked_mul(10).and_then(|v| v.checked_add(u32::from(x - b'0'))) {
                    Some(n) if n <= u16::MAX as u32 => n,
                    _ => return Err(ParseError::InvalidPort { at: p }),
                };
            }
            if n == 0 {
                return Err(ParseError::InvalidPort { at: p });
            }
            Some(n as u16)
        }
    };

    let hash = b[end..].iter().position(|&x| x == b'#').map(|i| end + i);
    let frag_bound = hash.unwrap_or(b.len());
    let query = b[end..frag_bound]
        .iter()
        .position(|&x| x == b'?')
        .map(|i| end + i);

    let path_bound = query.unwrap_or(frag_bound);
    let path = if path_bound == end {
        String::from("/")
    } else {
        input[end..path_bound].to_owned()
    };
    let query_value = query.map(|q| input[q + 1..frag_bound].to_owned());
    let fragment = hash.map(|h| input[h + 1..].to_owned());

    Ok(Url {
        scheme: input[..c].to_ascii_lowercase(),
        userinfo,
        host: input[hp_start..hp_end].to_ascii_lowercase(),
        port,
        path,
        query: query_value,
        fragment,
    })
}

    InvalidUserInfo { at: usize },

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(ParseError::Empty);
    }

    for (i, &b) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: bytes.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }

    for (i, &b) in bytes[..colon].iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| authority_start + i)
        .unwrap_or(bytes.len());

    let authority = &input[authority_start..authority_end];
    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, host_start_rel) = if let Some(&at_rel) = at_positions.first() {
        if at_rel == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at_rel,
            });
        }
        (Some(authority[..at_rel].to_owned()), at_rel + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[host_start_rel..];
    let host_start = authority_start + host_start_rel;
    let port_colon_rel = hostport.bytes().position(|b| b == b':');
    let host_end_rel = port_colon_rel.unwrap_or(hostport.len());
    let host = &hostport[..host_end_rel];

    if host.is_empty() {
        return Err(ParseError::InvalidHost { at: host_start });
    }

    for (i, b) in host.bytes().enumerate() {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_start + i,
            });
        }
    }

    let port = if let Some(port_colon_rel) = port_colon_rel {
        let port_start = host_start + port_colon_rel + 1;
        let port_text = &hostport[port_colon_rel + 1..];

        if port_text.is_empty() {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        for (i, b) in port_text.bytes().enumerate() {
            if !b.is_ascii_digit() {
                return Err(ParseError::InvalidPort {
                    at: port_start + i,
                });
            }
        }

        if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        let mut value: u32 = 0;
        for b in port_text.bytes() {
            value = match value
                .checked_mul(10)
                .and_then(|v| v.checked_add((b - b'0') as u32))
            {
                Some(v) => v,
                None => return Err(ParseError::InvalidPort { at: port_start }),
            };
            if value > u16::MAX as u32 {
                return Err(ParseError::InvalidPort { at: port_start });
            }
        }

        if value == 0 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
        Some(value as u16)
    } else {
        None
    };

    let remainder = &input[authority_end..];
    let fragment_rel = remainder.find('#');
    let before_fragment = fragment_rel
        .map(|i| &remainder[..i])
        .unwrap_or(remainder);
    let fragment = fragment_rel.map(|i| remainder[i + 1..].to_owned());

    let query_rel = before_fragment.find('?');
    let path = query_rel
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_rel.map(|i| before_fragment[i + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: host.to_ascii_lowercase(),
        port,
        path: if path.is_empty() {
            "/".to_owned()
        } else {
            path.to_owned()
        },
        query,
        fragment,
    })
}

pub enum ParseError {

    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let b = input.as_bytes();
    for i in 0..b.len() {
        if b[i] < 0x21 || b[i] > 0x7e {
            return Err(ParseError::InvalidByte { at: i });
        }
    }

    let scheme_end = b.iter().position(|&x| x == b':').unwrap_or(b.len());
    if scheme_end == b.len() {
        return Err(ParseError::InvalidScheme { at: b.len() });
    }
    if scheme_end == 0 || !b[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    let mut i = 1;
    while i < scheme_end {
        let x = b[i];
        if !x.is_ascii_alphanumeric() && x != b'+' && x != b'-' && x != b'.' {
            return Err(ParseError::InvalidScheme { at: i });
        }
        i += 1;
    }

    let auth_start = scheme_end + 1;
    if auth_start + 1 >= b.len() || b[auth_start] != b'/' || b[auth_start + 1] != b'/' {
        return Err(ParseError::MissingAuthority { at: auth_start });
    }

    let auth_start = auth_start + 2;
    let mut auth_end = auth_start;
    while auth_end < b.len() && b[auth_end] != b'/' && b[auth_end] != b'?' && b[auth_end] != b'#' {
        auth_end += 1;
    }

    let mut first_at = None;
    let mut j = auth_start;
    while j < auth_end {
        if b[j] == b'@' {
            if first_at.is_some() {
                return Err(ParseError::InvalidUserInfo { at: j });
            }
            first_at = Some(j);
        }
        j += 1;
    }

    let (userinfo, hp_start) = match first_at {
        Some(pos) if pos == auth_start => {
            return Err(ParseError::InvalidUserInfo { at: pos });
        }
        Some(pos) => (
            Some(input[auth_start..pos].to_owned()),
            pos + 1,
        ),
        None => (None, auth_start),
    };

    let mut hp_end = auth_end;
    let mut colon = None;
    j = hp_start;
    while j < auth_end {
        if b[j] == b':' {
            colon = Some(j);
            hp_end = j;
            break;
        }
        j += 1;
    }

    if hp_start == hp_end {
        return Err(ParseError::InvalidHost { at: hp_start });
    }
    j = hp_start;
    while j < hp_end {
        let x = b[j];
        if !x.is_ascii_alphanumeric() && x != b'.' && x != b'-' && x != b'_' {
            return Err(ParseError::InvalidHost { at: j });
        }
        j += 1;
    }

    let port = if let Some(c) = colon {
        let start = c + 1;
        if start == auth_end {
            return Err(ParseError::InvalidPort { at: start });
        }
        j = start;
        while j < auth_end {
            if !b[j].is_ascii_digit() {
                return Err(ParseError::InvalidPort { at: j });
            }
            j += 1;
        }
        if auth_end - start >= 2 && b[start] == b'0' {
            return Err(ParseError::InvalidPort { at: start });
        }
        let mut n: u32 = 0;
        j = start;
        while j < auth_end {
            n = n * 10 + u32::from(b[j] - b'0');
            if n > 65535 {
                return Err(ParseError::InvalidPort { at: start });
            }
            j += 1;
        }
        if n == 0 {
            return Err(ParseError::InvalidPort { at: start });
        }
        Some(n as u16)
    } else {
        None
    };

    let fragment_at = (auth_end..b.len()).find(|&k| b[k] == b'#');
    let component_end = fragment_at.unwrap_or(b.len());
    let query_at = (auth_end..component_end).find(|&k| b[k] == b'?');
    let path_end = query_at.unwrap_or(component_end);

    let path = if auth_end == path_end {
        String::from("/")
    } else {
        input[auth_end..path_end].to_owned()
    };
    let query = query_at.map(|q| input[q + 1..component_end].to_owned());
    let fragment = fragment_at.map(|f| input[f + 1..].to_owned());

    Ok(Url {
        scheme: input[..scheme_end].to_ascii_lowercase(),
        userinfo,
        host: input[hp_start..hp_end].to_ascii_lowercase(),
        port,
        path,
        query,
        fragment,
    })
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    let authority_start = colon + 1;
    if bytes.get(authority_start) != Some(&b'/')
        || bytes.get(authority_start + 1) != Some(&b'/')
    {
        return Err(ParseError::MissingAuthority { at: authority_start });
    }

    let authority_start = authority_start + 2;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map_or(input.len(), |i| authority_start + i);
    let authority = &input[authority_start..authority_end];

    let mut at_positions = authority.match_indices('@').map(|(i, _)| i);
    let first_at = at_positions.next();
    if let Some(second_at) = at_positions.next() {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + second_at,
        });
    }

    let (userinfo, host_port, host_port_start) = match first_at {
        Some(i) if i == 0 => {
            return Err(ParseError::InvalidUserInfo { at: authority_start });
        }
        Some(i) => (
            Some(authority[..i].to_owned()),
            &authority[i + 1..],
            authority_start + i + 1,
        ),
        None => (None, authority, authority_start),
    };

    let port_colon = host_port.find(':');
    let (host, port_text, port_start) = match port_colon {
        Some(i) => (
            &host_port[..i],
            Some(&host_port[i + 1..]),
            host_port_start + i + 1,
        ),
        None => (host_port, None, 0),
    };

    if host.is_empty() {
        return Err(ParseError::InvalidHost { at: host_port_start });
    }
    for (i, byte) in host.bytes().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost { at: host_port_start + i });
        }
    }

    let port = match port_text {
        None => None,
        Some("") => return Err(ParseError::InvalidPort { at: port_start }),
        Some(text) => {
            for (i, byte) in text.bytes().enumerate() {
                if !byte.is_ascii_digit() {
                    return Err(ParseError::InvalidPort { at: port_start + i });
                }
            }
            if text.len() >= 2 && text.as_bytes()[0] == b'0' {
                return Err(ParseError::InvalidPort { at: port_start });
            }
            let mut value: u32 = 0;
            for byte in text.bytes() {
                value = value * 10 + u32::from(byte - b'0');
                if value > 65535 {
                    return Err(ParseError::InvalidPort { at: port_start });
                }
            }
            if value == 0 {
                return Err(ParseError::InvalidPort { at: port_start });
            }
            Some(value as u16)
        }
    };

    let fragment_mark = bytes[authority_end..]
        .iter()
        .position(|&b| b == b'#')
        .map(|i| authority_end + i);
    let components_end = fragment_mark.unwrap_or(input.len());
    let query_mark = bytes[authority_end..components_end]
        .iter()
        .position(|&b| b == b'?')
        .map(|i| authority_end + i);

    let path_end = query_mark.unwrap_or(components_end);
    let path = if authority_end == path_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_mark.map(|q| input[q + 1..components_end].to_owned());
    let fragment = fragment_mark.map(|f| input[f + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: host.to_ascii_lowercase(),
        port,
        path,
        query,
        fragment,
    })
