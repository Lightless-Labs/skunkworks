#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum State {
    Queued,
    Cancelled,
    Completed,
}

#[derive(Debug, Clone, PartialEq, Eq)]
struct Entry {
    job: Job,
    sequence: u64,
    state: State,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    entries: Vec<Entry>,
    next_sequence: u64,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            entries: Vec::new(),
            next_sequence: 0,
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.entries.iter().any(|entry| entry.job.id == job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.wrapping_add(1);
        self.entries.push(Entry {
            job,
            sequence,
            state: State::Queued,
        });
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        let Some(entry) = self.entries.iter_mut().find(|entry| entry.job.id == id) else {
            return Err(Error::Unknown);
        };
        match entry.state {
            State::Queued => entry.state = State::Cancelled,
            State::Cancelled => return Err(Error::Cancelled),
            State::Completed => return Err(Error::Completed),
        }
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let mut best: Option<(usize, u8, u64)> = None;
        for (index, entry) in self.entries.iter().enumerate() {
            if entry.state != State::Queued || entry.job.ready_at > now {
                continue;
            }
            let effective = effective_priority(&entry.job, now, self.aging_ms);
            match best {
                None => best = Some((index, effective, entry.sequence)),
                Some((_, best_priority, best_sequence))
                    if effective > best_priority
                        || (effective == best_priority && entry.sequence < best_sequence) =>
                {
                    best = Some((index, effective, entry.sequence));
                }
                _ => {}
            }
        }

        self.last_time = now;
        let Some((index, effective_priority, _)) = best else {
            return Ok(None);
        };
        self.entries[index].state = State::Completed;
        Ok(Some(Dispatch {
            id: self.entries[index].job.id.clone(),
            effective_priority,
        }))
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|&byte| byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-')
}

fn effective_priority(job: &Job, now: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        return job.priority;
    }
    let gained = (now - job.ready_at) / aging_ms;
    (job.priority as u64 + gained).min(9) as u8
}

        self.entries.push(Entry {

            sequence,
            state: State::Queued,
        });
        self.last_time = now;

    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        let Some(entry) = self.entries.iter_mut().find(|entry| entry.job.id == id) else {
            return Err(Error::Unknown);
        };
        match entry.state {
            State::Queued => entry.state = State::Cancelled,
            State::Cancelled => return Err(Error::Cancelled),
            State::Completed => return Err(Error::Completed),
        }
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }

        let mut selected: Option<(usize, u8, u64)> = None;
        for (index, entry) in self.entries.iter().enumerate() {
            if entry.state != State::Queued || entry.job.ready_at > now {
                continue;
            }
            let priority = effective_priority(&entry.job, now, self.aging_ms);
            if selected
                .map(|(_, best_priority, best_sequence)| {
                    priority > best_priority
                        || (priority == best_priority && entry.sequence < best_sequence)
                })
                .unwrap_or(true)
            {
                selected = Some((index, priority, entry.sequence));
            }
        }

        self.last_time = now;
        match selected {
            None => Ok(None),
            Some((index, effective_priority, _)) => {
                self.entries[index].state = State::Completed;
                Ok(Some(Dispatch {
                    id: self.entries[index].job.id.clone(),
                    effective_priority,
                }))
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    (1..=16).contains(&id.len())
        && id.bytes().all(|byte| {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'-'
        })
}

fn effective_priority(job: &Job, now: u64, aging_ms: u64) -> u8 {
    match aging_ms {
        0 => job.priority,
        interval => {
            let age_steps = (now - job.ready_at) / interval;
            (job.priority as u64 + age_steps).min(9) as u8
        }
    }
}
