    indent: usize,

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn validate_markers(input: &str) -> Result<(), WrapError> {
    let chars: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..chars.len() {
        if chars[i].1 != '~' {
            continue;
        }
        let valid_side = |j: usize| {
            j < chars.len()
                && chars[j].1 != '~'
                && !is_separator(chars[j].1)
        };
        if i == 0 || !valid_side(i - 1) || !valid_side(i + 1) {
            return Err(WrapError::InvalidMarker { index: chars[i].0 });
        }
    }
    Ok(())
}

fn normalize_newlines(input: &str) -> Vec<char> {
    let mut out = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            out.push('\n');
        } else {
            out.push(c);
        }
    }
    out
}

fn parse_word(chars: &[char]) -> Word {
    let mut display = Vec::new();
    let mut breaks = Vec::new();
    for &c in chars {
        if c == '~' {
            breaks.push(display.len());
        } else {
            display.push(c);
        }
    }
    Word {
        chars: display,
        breaks,
    }
}

fn line_string(line: &Line, width: usize, justify: bool, final_line: bool, line_index: usize) -> String {
    let gaps = line.pieces.len().saturating_sub(1);
    let base_width: usize = line.pieces.iter().map(|p| p.width).sum::<usize>() + gaps;
    let extra = if justify && !final_line && gaps > 0 {
        width.saturating_sub(line.indent + base_width)
    } else {
        0
    };

    let mut result = " ".repeat(line.indent);
    let each = if gaps > 0 { extra / gaps } else { 0 };
    let rem = if gaps > 0 { extra % gaps } else { 0 };
    let start = if gaps > 0 { line_index % gaps } else { 0 };

    for (i, piece) in line.pieces.iter().enumerate() {
        result.push_str(&piece.text);
        if i < gaps {
            let bonus = if rem > 0 && (i + gaps - start) % gaps < rem { 1 } else { 0 };
            result.push_str(&" ".repeat(1 + each + bonus));
        }
    }
    result
}

fn flush_line(
    line: &mut Line,
    output: &mut Vec<String>,
    options: &WrapOptions,
    final_line: bool,
    line_index: usize,
) {
    if line.content_width > 0 {
        output.push(line_string(line, options.width, options.justify, final_line, line_index));
        line.pieces.clear();
        line.content_width = 0;
    }
}

fn wrap_paragraph(words: &[Word], options: &WrapOptions, output: &mut Vec<String>) {
    let mut line_index = 0usize;
    let mut line = Line {
        indent: options.first_indent,
        content_width: 0,
        pieces: Vec::new(),
    };

    for word in words {
        let mut offset = 0usize;
        let mut first_piece_of_word = true;

        while offset < word.chars.len() {
            let capacity = options.width - line.indent;
            let available = capacity.saturating_sub(line.content_width);
            let gap = if line.pieces.is_empty() || !first_piece_of_word { 0 } else { 1 };
            let room = available.saturating_sub(gap);

            let suffix_len = word.chars.len() - offset;
            if suffix_len <= room {
                if gap > 0 {
                    line.pieces.push(Piece { text: " ".to_string(), width: 0 });
                    // The gap is represented by the inter-piece joining below instead.
                    line.pieces.pop();
                }
                let text: String = word.chars[offset..].iter().collect();
                if line.pieces.is_empty() {
                    line.pieces.push(Piece { text, width: suffix_len });
                } else {
                    line.pieces.push(Piece { text, width: suffix_len });
                }
                line.content_width += gap + suffix_len;
                first_piece_of_word = false;
                offset = word.chars.len();
                continue;
            }

            let mut usable: Option<(usize, usize)> = None;
            for &br in &word.breaks {
                if br > offset && br < word.chars.len() {
                    let prefix_len = br - offset;
                    if prefix_len + 1 <= room {
                        usable = Some((br, prefix_len));
                    }
                }
            }

            let accepted = if let Some((br, prefix_len)) = usable {
                let unused = room - prefix_len - 1;
                if line.pieces.is_empty() {
                    Some((br, prefix_len))
                } else {
                    let lhs = (unused as u128) * (unused as u128) + options.hyphen_cost as u128;
                    let rhs = (available as u128) * (available as u128);
                    if lhs < rhs { Some((br, prefix_len)) } else { None }
                }
            } else {
                None
            };

            if let Some((br, prefix_len)) = accepted {
                let mut text: String = word.chars[offset..br].iter().collect();
                text.push('-');
                if !line.pieces.is_empty() && first_piece_of_word {
                    // A hyphenated word after prior words contributes one ordinary gap.
                    line.content_width += 1;
                }
                line.content_width += prefix_len + 1;
                line.pieces.push(Piece { text, width: prefix_len + 1 });
                flush_line(&mut line, output, options, false, line_index);
                line_index += 1;
                line.indent = options.rest_indent;
                offset = br;
                first_piece_of_word = false;
                continue;
            }

            if !line.pieces.is_empty() {
                flush_line(&mut line, output, options, false, line_index);
                line_index += 1;
                line.indent = options.rest_indent;
                continue;
            }

            let capacity = options.width - line.indent;
            let suffix_len = word.chars.len() - offset;
            if suffix_len > capacity {
                let end = offset + capacity;
                let text: String = word.chars[offset..end].iter().collect();
                line.pieces.push(Piece { text, width: capacity });
                line.content_width = capacity;
                flush_line(&mut line, output, options, false, line_index);
                line_index += 1;
                line.indent = options.rest_indent;

                // Discard opportunities crossed by a raw split.
                offset = end;
                while offset < word.chars.len() && word.breaks.contains(&offset) {
                    offset += 1;
                }
                first_piece_of_word = false;
            } else {
                let text: String = word.chars[offset..].iter().collect();
                line.content_width = suffix_len;
                line.pieces.push(Piece { text, width: suffix_len });
                offset = word.chars.len();
            }
        }
    }

    if !line.pieces.is_empty() {
        output.push(line_string(&line, options.width, options.justify, true, line_index));
    }
}

fn parse_segment(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Vec::new();
    for &c in chars {
        if is_separator(c) {
            if !current.is_empty() {
                words.push(parse_word(&current));
                current.clear();
            }
        } else {
            current.push(c);
        }
    }
    if !current.is_empty() {
        words.push(parse_word(&current));
    }
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    validate_markers(input)?;

    let normalized = normalize_newlines(input);
    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut i = 0usize;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let run = i - start;
            if run >= 2 {
                let words = parse_segment(&normalized[segment_start..start]);
                wrap_paragraph(&words, options, &mut output);
                output.extend((0..run - 1).map(|_| String::new()));
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }

    let words = parse_segment(&normalized[segment_start..]);
    wrap_paragraph(&words, options, &mut output);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    breaks: Vec<usize>,
}

#[derive(Clone)]
struct Piece {
    text: String,
    width: usize,
}

struct Line {
    indent: usize,
    content_width: usize,
    pieces: Vec<Piece>,
}

fn separator(c: char) -> bool {
    c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00a0}'
}

fn validate_markers(input: &str) -> Result<(), WrapError> {
    let chars: Vec<(usize, char)> = input.char_indices().collect();
    for (i, &(byte, c)) in chars.iter().enumerate() {
        if c != '~' {
            continue;
        }
        let left_ok = i > 0 && chars[i - 1].1 != '~' && !separator(chars[i - 1].1);
        let right_ok = i + 1 < chars.len()
            && chars[i + 1].1 != '~'
            && !separator(chars[i + 1].1);
        if !left_ok || !right_ok {
            return Err(WrapError::InvalidMarker { index: byte });
        }
    }
    Ok(())
}

fn normalized_chars(input: &str) -> Vec<char> {
    let mut chars = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(c) = iter.next() {
        if c == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            chars.push('\n');
        } else {
            chars.push(c);
        }
    }
    chars
}

fn make_word(raw: &[char]) -> Word {
    let mut chars = Vec::new();
    let mut breaks = Vec::new();
    for &c in raw {
        if c == '~' {
            breaks.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    Word { chars, breaks }
}

fn segment_words(raw: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut start = 0;
    for i in 0..=raw.len() {
        if i == raw.len() || separator(raw[i]) {
            if start < i {
                words.push(make_word(&raw[start..i]));
            }
            start = i + 1;
        }
    }
    words
}

fn append_line(line: &Line, output: &mut Vec<String>, options: &WrapOptions, justify: bool, index: usize) {
    let gap_count = line.pieces.len().saturating_sub(1);
    let piece_sum: usize = line.pieces.iter().map(|p| p.width).sum();
    let pre_width = piece_sum + gap_count;
    let extra = if justify && gap_count > 0 {
        options.width - line.indent - pre_width
    } else {
        0
    };

    let mut result = String::new();
    result.extend(std::iter::repeat(' ').take(line.indent));
    let regular = if gap_count > 0 { extra / gap_count } else { 0 };
    let remainder = if gap_count > 0 { extra % gap_count } else { 0 };
    let start_gap = if gap_count > 0 { index % gap_count } else { 0 };

    for (i, piece) in line.pieces.iter().enumerate() {
        result.push_str(&piece.text);
        if i < gap_count {
            let extra_here = if remainder > 0 && (i + gap_count - start_gap) % gap_count < remainder { 1 } else { 0 };
            result.extend(std::iter::repeat(' ').take(1 + regular + extra_here));
        }
    }
    output.push(result);
}

fn emit(
    line: &mut Line,
    output: &mut Vec<String>,
    options: &WrapOptions,
    line_index: usize,
) {
    append_line(line, output, options, options.justify && line.pieces.len() > 1, line_index);
    line.pieces.clear();
    line.content_width = 0;
}

fn layout(words: &[Word], options: &WrapOptions, output: &mut Vec<String>) {
    let mut index = 0usize;
    let mut line = Line { indent: options.first_indent, content_width: 0, pieces: Vec::new() };

    for word in words {
        let mut pos = 0usize;
        let mut continuation = false;
        while pos < word.chars.len() {
            let cap = options.width - line.indent;
            let free = cap - line.content_width;
            let gap = usize::from(!line.pieces.is_empty() && !continuation);
            let room = free.saturating_sub(gap);
            let rest = word.chars.len() - pos;

            if rest <= room {
                line.content_width += gap + rest;
                line.pieces.push(Piece { text: word.chars[pos..].iter().collect(), width: rest });
                pos = word.chars.len();
                continue;
            }

            let mut candidate = None;
            for &point in &word.breaks {
                if point > pos && point < word.chars.len() {
                    let len = point - pos;
                    if len + 1 <= room {
                        candidate = Some((point, len));
                    }
                }
            }

            let use_candidate = match candidate {
                Some((point, len)) if line.pieces.is_empty() => Some((point, len)),
                Some((point, len)) => {
                    let unused = room - len - 1;
                    let lhs = unused as u128 * unused as u128 + options.hyphen_cost as u128;
                    let rhs = free as u128 * free as u128;
                    if lhs < rhs { Some((point, len)) } else { None }
                }
                None => None,
            };

            if let Some((point, len)) = use_candidate {
                let mut text: String = word.chars[pos..point].iter().collect();
                text.push('-');
                if gap == 1 {
                    line.content_width += 1;
                }
                line.content_width += len + 1;
                line.pieces.push(Piece { text, width: len + 1 });
                emit(&mut line, output, options, index);
                index += 1;
                line.indent = options.rest_indent;
                pos = point;
                continuation = true;
                continue;
            }

            if !line.pieces.is_empty() {
                emit(&mut line, output, options, index);
                index += 1;
                line.indent = options.rest_indent;
                continue;
            }

            let cap = options.width - line.indent;
            if rest > cap {
                let end = pos + cap;
                line.content_width = cap;
                line.pieces.push(Piece { text: word.chars[pos..end].iter().collect(), width: cap });
                emit(&mut line, output, options, index);
                index += 1;
                line.indent = options.rest_indent;
                pos = end;
                while word.breaks.contains(&pos) {
                    pos += 1;
                }
                continuation = true;
            } else {
                line.content_width = rest;
                line.pieces.push(Piece { text: word.chars[pos..].iter().collect(), width: rest });
                pos = word.chars.len();
            }
        }
    }

    if !line.pieces.is_empty() {
        append_line(&line, output, options, false, index);
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && c != '\t' && c != '\r' && c != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }
    validate_markers(input)?;

    let chars = normalized_chars(input);
    let mut result = Vec::new();
    let mut start = 0usize;
    let mut cursor = 0usize;

    while cursor < chars.len() {
        if chars[cursor] == '\n' {
            let run_start = cursor;
            while cursor < chars.len() && chars[cursor] == '\n' {
                cursor += 1;
            }
            let run_length = cursor - run_start;
            if run_length >= 2 {
                let words = segment_words(&chars[start..run_start]);
                layout(&words, options, &mut result);
                for _ in 1..run_length {
                    result.push(String::new());
                }
                start = cursor;
            }
        } else {
            cursor += 1;
        }
    }

    let words = segment_words(&chars[start..]);
    layout(&words, options, &mut result);
    Ok(result)
}

fn parse_b(chars: &[char]) -> Vec<WordB> {
    let mut out = Vec::new();
    let mut pos = 0;
    while pos < chars.len() {
        while pos < chars.len() && sep_b(chars[pos]) {
            pos += 1;
        }
        if pos == chars.len() {
            break;
        }
        let mut word = Vec::new();
        let mut markers = Vec::new();
        while pos < chars.len() && !sep_b(chars[pos]) {
            if chars[pos] == '~' {
                markers.push(word.len());
            } else {
                word.push(chars[pos]);
            }
            pos += 1;
        }
        out.push(WordB { chars: word, markers });
    }
    out
}

fn emit_b(line: LineB, options: &WrapOptions, index: usize, final_line: bool) -> String {
    let justify = options.justify && !final_line && line.pieces.len() >= 2;
    let gaps = line.pieces.len().saturating_sub(1);
    let extra = if justify {
        options.width.saturating_sub(line.indent + line.content_width)
    } else {
        0
    };
    let each = if gaps == 0 { 0 } else { extra / gaps };
    let rem = if gaps == 0 { 0 } else { extra % gaps };
    let start = if gaps == 0 { 0 } else { index % gaps };

    let mut out = String::with_capacity(options.width);
    out.extend(std::iter::repeat_n(' ', line.indent));
    for i in 0..line.pieces.len() {
        out.push_str(&line.pieces[i]);
        if i < gaps {
            let bonus = usize::from((i + gaps - start) % gaps < rem);
            out.extend(std::iter::repeat_n(' ', 1 + each + bonus));
        }
    }
    out
}

fn layout_b(words: &[WordB], options: &WrapOptions) -> Vec<String> {
    let mut completed = Vec::<LineB>::new();
    if words.is_empty() {
        return Vec::new();
    }

    let mut para_line = 0usize;
    let mut current = LineB::new(options.first_indent);
    let mut wi = 0usize;

    while wi < words.len() {
        let w = &words[wi];
        let mut off = 0usize;
        loop {
            if off == w.chars.len() {
                wi += 1;
                break;
            }

            let indent = if para_line == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            current.indent = indent;
            let cap = options.width - indent;
            let room = cap.saturating_sub(current.content_width);
            let gap = usize::from(!current.pieces.is_empty());
            let remaining = w.chars.len() - off;

            if remaining + gap <= room {
                current.push(w.chars[off..].iter().collect(), remaining);
                wi += 1;
                break;
            }

            let best = w
                .markers
                .iter()
                .copied()
                .filter(|&m| m > off && m < w.chars.len() && m - off + 1 <= room)
                .max();

            let mut take = false;
            if let Some(m) = best {
                if current.pieces.is_empty() {
                    take = true;
                } else {
                    let unused = cap.saturating_sub(current.content_width + (m - off + 1));
                    let baseline = cap.saturating_sub(current.content_width);
                    let cost = (unused as u128) * (unused as u128)
                        + options.hyphen_cost as u128;
                    let base = (baseline as u128) * (baseline as u128);
                    take = cost < base;
                }
            }

            if take {
                let m = best.unwrap();
                let mut part: String = w.chars[off..m].iter().collect();
                part.push('-');
                current.push(part, m - off + 1);
                completed.push(current);
                para_line += 1;
                current = LineB::new(options.rest_indent);
                off = m;
                continue;
            }

            if !current.pieces.is_empty() {
                completed.push(current);
                para_line += 1;
                current = LineB::new(options.rest_indent);
                continue;
            }

            if remaining > cap {
                current.push(w.chars[off..off + cap].iter().collect(), cap);
                completed.push(current);
                para_line += 1;
                current = LineB::new(options.rest_indent);
                off += cap;
            } else {
                current.push(w.chars[off..].iter().collect(), remaining);
                wi += 1;
                break;
            }
        }
    }

    if !current.pieces.is_empty() {
        completed.push(current);
    }

    let last = completed.len().saturating_sub(1);
    completed
        .into_iter()
        .enumerate()
        .map(|(i, line)| emit_b(line, options, i, i == last))
        .collect()
}

fn normalize_b(input: &str) -> Vec<char> {
    let mut result = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            result.push('\n');
        } else {
            result.push(c);
        }
    }
    result
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    for (byte, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index: byte });
        }
    }

    let original: Vec<char> = input.chars().collect();
    let byte_offsets: Vec<usize> = input.char_indices().map(|(b, _)| b).collect();
    for i in 0..original.len() {
        if original[i] == '~' {
            let left = i > 0 && original[i - 1] != '~' && !sep_b(original[i - 1]);
            let right = i + 1 < original.len()
                && original[i + 1] != '~'
                && !sep_b(original[i + 1]);
            if !left || !right {
                return Err(WrapError::InvalidMarker {
                    index: byte_offsets[i],
                });
            }
        }
    }

    if input.is_empty() {
        return Ok(Vec::new());
    }

    let chars = normalize_b(input);
    let mut output = Vec::new();
    let mut start = 0;
    let mut cursor = 0;
    while cursor < chars.len() {
        if chars[cursor] == '\n' {
            let run_start = cursor;
            while cursor < chars.len() && chars[cursor] == '\n' {
                cursor += 1;
            }
            let run_len = cursor - run_start;
            if run_len >= 2 {
                output.extend(layout_b(&parse_b(&chars[start..run_start]), options));
                output.extend(std::iter::repeat_n(String::new(), run_len - 1));
                start = cursor;
            }
        } else {
            cursor += 1;
        }
    }
    output.extend(layout_b(&parse_b(&chars[start..]), options));
    Ok(output)
}

fn validate(s: &str) -> Result<(), WrapError> {
    let v: Vec<(usize, char)> = s.char_indices().collect();
    for &(index, c) in &v {
        if c.is_control() && c != '\t' && c != '\r' && c != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..v.len() {
        if v[i].1 == '~'
            && (i == 0 || i + 1 == v.len()
                || v[i - 1].1 == '~' || v[i + 1].1 == '~'
                || sep(v[i - 1].1) || sep(v[i + 1].1))
        {
            return Err(WrapError::InvalidMarker { index: v[i].0 });
        }
    }
    Ok(())
}

fn parse_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Vec::new();
    let mut breaks = Vec::new();
    let mut marked = false;

    for &c in chars {
        if sep(c) {
            if !current.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut current),
                    breaks: std::mem::take(&mut breaks),
                });
            }
            marked = false;
        } else if c == '~' {
            marked = true;
        } else {
            if marked {
                breaks.push(current.len());
            }
            current.push(c);
            marked = false;
        }
    }
    if !current.is_empty() {
        words.push(Word { chars: current, breaks });
    }
    words

#[derive(Default)]
struct CurrentLine {
    parts: Vec<String>,
}

impl CurrentLine {
    fn content_len(&self) -> usize {
        self.parts.iter().map(|p| p.chars().count()).sum::<usize>()
            + self.parts.len().saturating_sub(1)
    }

    fn append_word(&mut self, text: String) {
        if !text.is_empty() {
            self.parts.push(text);
        }
    }
}

fn output_line(
    line: &CurrentLine,
    indent: usize,
    width: usize,
    justify: bool,
    final_line: bool,
    line_index: usize,
) -> String {
    let gaps = line.parts.len().saturating_sub(1);
    let content = line.content_len();
    let extra_total = width.saturating_sub(indent + content);
    let each_extra = if gaps == 0 { 0 } else { extra_total / gaps };
    let remainder = if gaps == 0 { 0 } else { extra_total % gaps };
    let start = if gaps == 0 { 0 } else { line_index % gaps };

    let mut out = " ".repeat(indent);
    for (i, part) in line.parts.iter().enumerate() {
        out.push_str(part);
        if i < gaps {
            let spaces = if justify && !final_line && gaps > 0 {
                1 + each_extra + usize::from((i + gaps - start) % gaps < remainder)
            } else {
                1
            };
            out.push_str(&" ".repeat(spaces));
        }
    }
    out
}

fn paragraph(chars: &[char], opt: &WrapOptions, out: &mut Vec<String>) {
    let words = parse_words(chars);
    if words.is_empty() {
        return;
    }

    let mut line = CurrentLine::default();
    let mut first = true;
    let mut line_index = 0usize;

    let emit = |line: &mut CurrentLine, first: &mut bool, line_index: &mut usize, out: &mut Vec<String>, final_line: bool| {
        let indent = if *first { opt.first_indent } else { opt.rest_indent };
        out.push(output_line(line, indent, opt.width, opt.justify, final_line, *line_index));
        *first = false;
        *line_index += 1;
        *line = CurrentLine::default();
    };

    for word in words {
        let mut pos = 0usize;
        while pos < word.chars.len() {
            let indent = if first { opt.first_indent } else { opt.rest_indent };
            let cap = opt.width - indent;
            let line_len = line.content_len();
            let gap = usize::from(!line.parts.is_empty());
            let room = cap.saturating_sub(line_len + gap);
            let remaining = word.chars.len() - pos;

            if remaining <= room {
                if gap != 0 {
                    line.append_word(word.chars[pos..].iter().collect());
                } else {
                    line.append_word(word.chars[pos..].iter().collect());
                }
                break;
            }

            let available = if line.parts.is_empty() { cap } else { room };
            let opportunity = word.breaks.iter().copied()
                .filter(|&b| b > pos && b < word.chars.len())
                .filter(|&b| b - pos + 1 <= available)
                .last();

            let accept = opportunity.and_then(|b| {
                if line.parts.is_empty() {
                    Some(b)
                } else {
                    let used = b - pos + 1;
                    let unused = available - used;
                    let baseline = cap - line_len;
                    let lhs = (unused as u128) * (unused as u128) + opt.hyphen_cost as u128;
                    let rhs = (baseline as u128) * (baseline as u128);
                    (lhs < rhs).then_some(b)
                }
            });

            if let Some(b) = accept {
                let mut fragment: String = word.chars[pos..b].iter().collect();
                fragment.push('-');
                line.append_word(fragment);
                pos = b;
                emit(&mut line, &mut first, &mut line_index, out, false);
                continue;
            }

            if !line.parts.is_empty() {
                emit(&mut line, &mut first, &mut line_index, out, false);
                continue;
            }

            if remaining > cap {
                let end = pos + cap;
                line.append_word(word.chars[pos..end].iter().collect());
                pos = end;
                emit(&mut line, &mut first, &mut line_index, out, false);
            } else {
                line.append_word(word.chars[pos..].iter().collect());
                pos = word.chars.len();
            }
        }
    }

    if !line.parts.is_empty() {
        emit(&mut line, &mut first, &mut line_index, out, true);
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    validate(input)?;

    let mut normalized = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0usize;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let begin = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let count = i - begin;
            if count >= 2 {
                paragraph(&segment, options, &mut result);
                segment.clear();
                for _ in 1..count {
                    result.push(String::new());
                }
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[i]);
            i += 1;
        }
    }
    paragraph(&segment, options, &mut result);
    Ok(result)
}

#[derive(Clone)]
struct Line {
    pieces: Vec<String>,
    gaps: Vec<usize>,
    scalar_width: usize,
    indent: usize,
}

impl Line {
    fn new(indent: usize) -> Self {
        Self {
            pieces: Vec::new(),
            gaps: Vec::new(),
            scalar_width: 0,
            indent,
        }
    }

    fn content_len(&self) -> usize {
        self.scalar_width
    }

    fn add_piece(&mut self, text: String, width: usize) {
        if !self.pieces.is_empty() {
            self.gaps.push(1);
            self.scalar_width += 1;
        }
        self.scalar_width += width;
        self.pieces.push(text);
    }

    fn render(&self, justify_to: Option<(usize, usize)>) -> String {
        let mut out = " ".repeat(self.indent);
        if self.pieces.is_empty() {
            return out;
        }

        if let Some((target, line_index)) = justify_to {
            let extra = target.saturating_sub(self.indent + self.scalar_width);
            let count = self.gaps.len();
            if count >= 1 && extra > 0 {
                let each = extra / count;
                let rem = extra % count;
                let start = line_index % count;
                for (i, piece) in self.pieces.iter().enumerate() {
                    out.push_str(piece);
                    if i < count {
                        let add = each + usize::from((i + count - start) % count < rem);
                        out.push_str(&" ".repeat(1 + add));
                    }
                }
                return out;
            }
        }

        for (i, piece) in self.pieces.iter().enumerate() {
            if i > 0 {
                out.push(' ');
            }
            out.push_str(piece);
        }
        out
    }
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_words(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut i = 0;
    while i < segment.len() {
        while i < segment.len() && is_separator(segment[i]) {
            i += 1;
        }
        if i == segment.len() {
            break;
        }
        let mut chars = Vec::new();
        let mut breaks = Vec::new();
        while i < segment.len() && !is_separator(segment[i]) {
            if segment[i] == '~' {
                breaks.push(chars.len());
            } else {
                chars.push(segment[i]);
            }
            i += 1;
        }
        words.push(Word { chars, breaks });
    }
    words
}

fn layout_paragraph(words: &[Word], options: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }

    let mut lines = Vec::<Line>::new();
    let mut line_index = 0usize;
    let mut line = Line::new(options.first_indent);
    let mut word_index = 0usize;

    while word_index < words.len() {
        let word = &words[word_index];
        let mut offset = 0usize;

        loop {
            if offset >= word.chars.len() {
                word_index += 1;
                break;
            }
            let indent = if line_index == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;
            if line.indent != indent {
                line.indent = indent;
            }
            let current = line.content_len();
            let room = capacity.saturating_sub(current);
            let gap = usize::from(!line.pieces.is_empty());
            let suffix_len = word.chars.len() - offset;

            if suffix_len + gap <= room {
                let text: String = word.chars[offset..].iter().collect();
                line.add_piece(text, suffix_len);
                word_index += 1;
                break;
            }

            let mut usable = None;
            for &at in &word.breaks {
                if at > offset && at < word.chars.len() {
                    let prefix_len = at - offset;
                    if prefix_len + 1 <= room {
                        usable = Some(at);
                    }
                }
            }

            if let Some(at) = usable {
                let prefix_len = at - offset;
                let used = prefix_len + 1;
                let unused = capacity.saturating_sub(current + used);
                let accepted = if line.pieces.is_empty() {
                    true
                } else {
                    let baseline = capacity.saturating_sub(current);
                    let lhs = (unused as u128) * (unused as u128)
                        + (options.hyphen_cost as u128);
                    let rhs = (baseline as u128) * (baseline as u128);
                    lhs < rhs
                };

                if accepted {
                    let mut text: String = word.chars[offset..at].iter().collect();
                    text.push('-');
                    line.add_piece(text, used);
                    lines.push(line);
                    line_index += 1;
                    line = Line::new(options.rest_indent);
                    offset = at;
                    continue;
                }
            }

            if !line.pieces.is_empty() {
                lines.push(line);
                line_index += 1;
                line = Line::new(options.rest_indent);
                continue;
            }

            if suffix_len > capacity {
                let end = offset + capacity;
                let text: String = word.chars[offset..end].iter().collect();
                line.add_piece(text, capacity);
                lines.push(line);
                line_index += 1;
                line = Line::new(options.rest_indent);
                offset = end;
            } else {
                let text: String = word.chars[offset..].iter().collect();
                line.add_piece(text, suffix_len);
                word_index += 1;
                break;
            }
        }
    }

    if !line.pieces.is_empty() {
        lines.push(line);
    }

    let final_index = lines.len().saturating_sub(1);
    lines
        .iter()
        .enumerate()
        .map(|(i, line)| {
            let nonfinal = i != final_index;
            let justify = options.justify && nonfinal && line.pieces.len() >= 2;
            if justify {
                line.render(Some((options.width, i)))
            } else {
                line.render(None)
            }
        })
        .collect()
}

fn normalized_chars(input: &str) -> Vec<char> {
    let mut out = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(c) = iter.next() {
        if c == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            out.push('\n');
        } else {
            out.push(c);
        }
    }
    out
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let chars: Vec<char> = input.chars().collect();
    for i in 0..chars.len() {
        if chars[i] == '~' {
            let valid_side = |j: isize| -> bool {
                if j < 0 || j as usize >= chars.len() {
                    return false;
                }
                let c = chars[j as usize];
                c != '~' && !is_separator(c)
            };
            if !valid_side(i as isize - 1) || !valid_side(i as isize + 1) {
                let byte_index = input
                    .char_indices()
                    .nth(i)
                    .map(|(b, _)| b)
                    .unwrap_or(input.len());
                return Err(WrapError::InvalidMarker { index: byte_index });
            }
        }
    }

    if input.is_empty() {
        return Ok(Vec::new());
    }

    let chars = normalized_chars(input);
    let mut result = Vec::new();
    let mut start = 0usize;
    let mut i = 0usize;

    while i < chars.len() {
        if chars[i] == '\n' {
            let run_start = i;
            while i < chars.len() && chars[i] == '\n' {
                i += 1;
            }
            if i - run_start >= 2 {
                let words = parse_words(&chars[start..run_start]);
                result.extend(layout_paragraph(&words, options));
                for _ in 1..(i - run_start) {
                    result.push(String::new());
                }
                start = i;
            }
        } else {
            i += 1;
        }
    }

    let words = parse_words(&chars[start..]);
    result.extend(layout_paragraph(&words, options));
    Ok(result)
}

#[derive(Clone)]
struct Piece {
    text: String,
    // Scalar offsets in displayed text where a marker preceded the scalar.
    breaks: Vec<usize>,
}

fn make_words(segment: &[char]) -> Vec<Piece> {
    let mut words = Vec::new();
    let mut text = String::new();
    let mut breaks = Vec::new();
    let mut pending_marker = false;

    let flush = |words: &mut Vec<Piece>, text: &mut String, breaks: &mut Vec<usize>| {
        if !text.is_empty() {
            words.push(Piece {
                text: std::mem::take(text),
                breaks: std::mem::take(breaks),
            });
        }
    };

    for &c in segment {
        if is_separator(c) {
            pending_marker = false;
            flush(&mut words, &mut text, &mut breaks);
        } else if c == '~' {
            pending_marker = true;
        } else {
            if pending_marker {
                breaks.push(text.chars().count());
            }
            text.push(c);
            pending_marker = false;
        }
    }
    flush(&mut words, &mut text, &mut breaks);
    words
}

#[derive(Clone)]
struct Line {
    pieces: Vec<String>,
}

impl Line {
    fn new() -> Self {
        Self { pieces: Vec::new() }
    }

    fn len(&self) -> usize {
        self.pieces.iter().map(|s| s.chars().count()).sum::<usize>()
            + self.pieces.len().saturating_sub(1)
    }

    fn push_piece(&mut self, piece: String) {
        if !piece.is_empty() {
            self.pieces.push(piece);
        }
    }
}

fn render_line(line: &Line, indent: usize, width: usize, justify: bool, is_final: bool, line_index: usize) -> String {
    let mut out = " ".repeat(indent);
    let gaps = line.pieces.len().saturating_sub(1);
    let base = if gaps == 0 { 1 } else { 1 + (width - indent - line.len()) / gaps };
    let extra = if gaps == 0 { 0 } else { (width - indent - line.len()) % gaps };
    let start = if gaps == 0 { 0 } else { line_index % gaps };

    for (i, piece) in line.pieces.iter().enumerate() {
        out.push_str(piece);
        if i < gaps {
            let spaces = if justify && !is_final && gaps > 0 {
                base + usize::from((i + gaps - start) % gaps < extra)
            } else {
                1
            };
            out.push_str(&" ".repeat(spaces));
        }
    }
    out
}

fn wrap_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let words = make_words(segment);
    if words.is_empty() {
        return;
    }

    let mut line = Line::new();
    let mut first_line = true;
    let mut line_index = 0usize;

    let emit = |line: &mut Line, first_line: &mut bool, line_index: &mut usize, output: &mut Vec<String>, options: &WrapOptions, is_final: bool| {
        let indent = if *first_line { options.first_indent } else { options.rest_indent };
        output.push(render_line(line, indent, options.width, options.justify, is_final, *line_index));
        *first_line = false;
        *line_index += 1;
        *line = Line::new();
    };

    for word in words {
        let chars: Vec<char> = word.text.chars().collect();
        let mut offset = 0usize;
        let mut local_breaks = word.breaks;

        while offset < chars.len() {
            let indent = if first_line { options.first_indent } else { options.rest_indent };
            let capacity = options.width - indent;
            let current_len = line.len();
            let separator = usize::from(!line.pieces.is_empty());
            let room = capacity.saturating_sub(current_len + separator);
            let suffix_len = chars.len() - offset;

            if suffix_len <= room {
                if separator == 1 {
                    line.push_piece(String::new());
                }
                line.push_piece(chars[offset..].iter().collect());
                break;
            }

            let available = if line.pieces.is_empty() {
                capacity
            } else {
                room
            };

            let candidate = local_breaks
                .iter()
                .copied()
                .filter(|&b| b > offset && b < chars.len())
                .filter_map(|b| {
                    let prefix_len = b - offset;
                    (prefix_len + 1 <= available).then_some((b, prefix_len + 1))
                })
                .last();

            let accepted = if let Some((b, used)) = candidate {
                let unused = available - used;
                if line.pieces.is_empty() {
                    Some((b, used))
                } else {
                    let threshold = capacity.saturating_sub(current_len);
                    let lhs = (unused as u128) * (unused as u128) + options.hyphen_cost as u128;
                    let rhs = (threshold as u128) * (threshold as u128);
                    (lhs < rhs).then_some((b, used))
                }
            } else {
                None
            };

            if let Some((b, _)) = accepted {
                if separator == 1 {
                    line.push_piece(String::new());
                }
                let mut fragment: String = chars[offset..b].iter().collect();
                fragment.push('-');
                line.push_piece(fragment);
                offset = b;
                emit(&mut line, &mut first_line, &mut line_index, output, options, false);
                continue;
            }

            if !line.pieces.is_empty() {
                emit(&mut line, &mut first_line, &mut line_index, output, options, false);
                continue;
            }

            if suffix_len > capacity {
                let end = offset + capacity;
                let raw: String = chars[offset..end].iter().collect();
                line.push_piece(raw);
                offset = end;
                local_breaks.retain(|&b| b > offset);
                emit(&mut line, &mut first_line, &mut line_index, output, options, false);
            } else {
                line.push_piece(chars[offset..].iter().collect());
                offset = chars.len();
            }
        }
    }

    if !line.pieces.is_empty() {
        let indent = if first_line { options.first_indent } else { options.rest_indent };
        let _ = indent;
        let is_final = true;
        emit(&mut line, &mut first_line, &mut line_index, output, options, is_final);
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {

        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    validate_input(input)?;

    let mut normalized = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0usize;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            let run = i - start;
            if run >= 2 {
                wrap_segment(&segment, options, &mut output);
                segment.clear();
                output.extend(std::iter::repeat(String::new()).take(run - 1));
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[i]);
            i += 1;
        }
    }
    wrap_segment(&segment, options, &mut output);
    Ok(output)
}

fn make_words(segment: &[char]) -> Vec<Word> {

    let mut chars = Vec::new();
    let mut breaks = Vec::new();

    let flush = |words: &mut Vec<Word>, chars: &mut Vec<char>, breaks: &mut Vec<usize>| {

        }
    }
    flush(&mut words, &mut chars, &mut breaks);

fn materialize_line(line: &Line, width: usize, justify: bool, is_final: bool, line_index: usize) -> String {
    let mut result = " ".repeat(line.indent);
    let gaps = line.pieces.len().saturating_sub(1);
    let extra = if justify && !is_final && gaps > 0 {
        width.saturating_sub(line.indent + line.content_width)
    } else {
        0
    };

    let each = if gaps > 0 { extra / gaps } else { 0 };
    let rem = if gaps > 0 { extra % gaps } else { 0 };
    let start = if gaps > 0 { line_index % gaps } else { 0 };

    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            result.push(' ');
            if justify && !is_final && gaps > 0 {
                result.push_str(&" ".repeat(each));
                let rel = (i - 1 + gaps - start) % gaps;
                if rel < rem {
                    result.push(' ');
                }
            }
        }
        result.push_str(&piece.text);
    }
    result
}

fn emit_line(line: &mut Line, output: &mut Vec<String>, width: usize, justify: bool, is_final: bool, line_index: usize) {
    if line.content_width > 0 {
        output.push(materialize_line(line, width, justify, is_final, line_index));
        line.pieces.clear();
        line.content_width = 0;
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..original.len() {
        if original[i].1 == '~' {
            let prev_ok = i > 0 && original[i - 1].1 != '~' && !is_separator(original[i - 1].1);
            let next_ok = i + 1 < original.len() && original[i + 1].1 != '~' && !is_separator(original[i + 1].1);
            if !prev_ok || !next_ok {
                return Err(WrapError::InvalidMarker { index: original[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut start = 0;
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                layout_segment(&normalized[start..i], options, &mut output);
                for _ in 1..(j - i) {
                    output.push(String::new());
                }
                start = j;
                i = j;
                continue;
            }
        }
        i += 1;
    }
    layout_segment(&normalized[start..], options, &mut output);
    Ok(output)
}

fn layout_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let words = make_words(segment);
    if words.is_empty() {
        return;
    }

    let mut lines = Vec::<Line>::new();
    let mut line_index = 0usize;
    let mut line = Line {
        indent: options.first_indent,
        pieces: Vec::new(),
        content_width: 0,
    };

    for word in words {
        let mut offset = 0usize;
        while offset < word.chars.len() {
            let cap = options.width - line.indent;
            let current = line.content_width;
            let room_with_gap = cap.saturating_sub(current + usize::from(current > 0));
            let suffix_len = word.chars.len() - offset;

            if suffix_len <= room_with_gap {
                if current > 0 {
                    line.content_width += 1;
                }
                let text: String = word.chars[offset..].iter().collect();
                line.content_width += suffix_len;
                line.pieces.push(Piece { text, is_word: true });
                offset = word.chars.len();
                continue;
            }

            let available = cap.saturating_sub(current);
            let mut opportunity = None;
            for &b in &word.breaks {
                if b > offset && b < word.chars.len() {
                    let plen = b - offset;
                    if plen + 1 <= available {
                        opportunity = Some((b, plen));
                    }
                }
            }

            let accept = if let Some((_, plen)) = opportunity {
                if current == 0 {
                    true
                } else {
                    let unused = available - plen - 1;
                    let lhs = (unused as u128) * (unused as u128) + options.hyphen_cost as u128;
                    let baseline = available as u128;
                    lhs < baseline * baseline
                }
            } else {
                false
            };

            if accept {
                let (b, plen) = opportunity.unwrap();
                if current > 0 {
                    line.content_width += 1;
                }
                let mut text: String = word.chars[offset..b].iter().collect();
                text.push('-');
                line.content_width += plen + 1;
                line.pieces.push(Piece { text, is_word: true });
                lines.push(line);
                line_index += 1;
                line = Line {
                    indent: options.rest_indent,
                    pieces: Vec::new(),
                    content_width: 0,
                };
                offset = b;
                continue;
            }

            if current > 0 {
                lines.push(line);
                line_index += 1;
                line = Line {
                    indent: options.rest_indent,
                    pieces: Vec::new(),
                    content_width: 0,
                };
                continue;
            }

            let cap = options.width - line.indent;
            if suffix_len > cap {
                let end = offset + cap;
                let text: String = word.chars[offset..end].iter().collect();
                line.content_width = cap;
                line.pieces.push(Piece { text, is_word: true });
                lines.push(line);
                line_index += 1;
                line = Line {
                    indent: options.rest_indent,
                    pieces: Vec::new(),
                    content_width: 0,
                };
                offset = end;
            } else {
                let text: String = word.chars[offset..].iter().collect();
                line.content_width += suffix_len;
                line.pieces.push(Piece { text, is_word: true });
                offset = word.chars.len();
            }
        }
    }

    if line.content_width > 0 {
        lines.push(line);
    }

    for idx in 0..lines.len() {
        let final_line = idx + 1 == lines.len();
        output.push(materialize_line(
            &lines[idx],
            options.width,
            options.justify,
            final_line,
            idx,
        ));
    }
}
