pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(base: u8, which: usize) -> Result<(), BaseError> {
    if (2..=36).contains(&base) {
        Ok(())
    } else {
        Err(BaseError::BadBase { which, value: base })
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {

        return Err(BaseError::Empty { operand });
    }

        let expected = radix.to_string();
        if bytes[..hash] != *expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        (hash + 1, &bytes[hash + 1..])
    } else {
        (0, bytes)
    };

    if payload.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut digits = Vec::new();

    // Invalid digits take precedence over every separator and leading-zero error.
    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            let left_is_digit = offset > 0 && payload[offset - 1] != b'_';
            let right_is_digit = offset + 1 < payload.len() && payload[offset + 1] != b'_';
            if !left_is_digit || !right_is_digit {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

// Convert an arbitrary-size digit vector from `from` to `to` using repeated
// long division. The vector is big-endian and contains no separators.
fn convert_digits(digits: &[u8], from: u8, to: u8) -> Vec<u8> {
    let first_nonzero = digits.iter().position(|&d| d != 0);
    let Some(first_nonzero) = first_nonzero else {
        return vec![0];
    };

    let mut work = digits[first_nonzero..].to_vec();
    let mut reversed = Vec::new();

    while !work.is_empty() {
        let mut quotient = Vec::with_capacity(work.len());
        let mut remainder: u16 = 0;
        for &digit in &work {
            let value = remainder * from as u16 + digit as u16;
            let qdigit = (value / to as u16) as u8;
            remainder = value % to as u16;
            if !quotient.is_empty() || qdigit != 0 {
                quotient.push(qdigit);
            }
        }
        reversed.push(remainder as u8);
        work = quotient;
    }

    reversed.reverse();
    reversed
}

fn format_digits(mut digits: Vec<u8>, style: BaseStyle) -> String {
    if digits.is_empty() {
        digits.push(0);
    }
    let padding = style.min_width.saturating_sub(digits.len());
    let total = padding + digits.len();
    let mut out = String::with_capacity(total + if style.group == 0 { 0 } else { total / style.group });

    for position in 0..total {
        if style.group != 0 && position != 0 && (total - position) % style.group == 0 {
            out.push('_');
        }

        let value = if position < padding {
            0
        } else {
            digits[position - padding]
        };
        let ch = match value {
            0..=9 => (b'0' + value) as char,
            _ if style.upper => (b'A' + value - 10) as char,
            _ => (b'a' + value - 10) as char,
        };
        out.push(ch);
    }

    out
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,

        return Err(BaseError::BadStyle);

    let mut left_pos = left_digits.len();
    let mut right_pos = right_digits.len();

    let mut reversed = Vec::with_capacity(left_digits.len().max(right_digits.len()) + 1);

        if left_pos > 0 {
            left_pos -= 1;
            sum += left_digits[left_pos] as u16;

        if right_pos > 0 {
            right_pos -= 1;
            sum += right_digits[right_pos] as u16;
        }
        reversed.push((sum % base as u16) as u8);
        carry = sum / base as u16;
    }

    reversed.reverse();
    Ok(format_digits(reversed, style))
}

pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn check_base(base: u8, which: usize) -> Result<(), BaseError> {
    if base < 2 || base > 36 {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn value_of(b: u8) -> Option<u8> {
    if b.is_ascii_digit() {
        Some(b - b'0')
    } else if b.is_ascii_uppercase() {
        Some(b - b'A' + 10)
    } else if b.is_ascii_lowercase() {
        Some(b - b'a' + 10)
    } else {
        None
    }
}

fn parse(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let mut start = 0;
    if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let spelling = radix.to_string();
        if &bytes[..hash] != spelling.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        start = hash + 1;
    }

    let payload = &bytes[start..];
    if payload.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    // First pass checks every non-separator byte.
    let mut digits = Vec::with_capacity(payload.len());
    for (i, &b) in payload.iter().enumerate() {
        if b == b'_' {
            continue;
        }
        match value_of(b) {
            Some(v) if v < radix => digits.push(v),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: start + i,
                });
            }
        }
    }

    // Second pass enforces separators and records their original byte offsets.
    for (i, &b) in payload.iter().enumerate() {
        if b == b'_' && (i == 0 || i + 1 == payload.len() || payload[i - 1] == b'_' || payload[i + 1] == b'_') {
            return Err(BaseError::Separator {
                operand,
                index: start + i,
            });
        }
    }

    if digits.len() > 1 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: start,
        });
    }

    Ok(digits)
}

// Accumulate a big-endian number into little-endian base-`to` digits.
fn rebase(input: &[u8], from: u8, to: u8) -> Vec<u8> {
    let mut out = vec![0u8];
    for &digit in input {
        let mut carry = digit as u16;
        for x in &mut out {
            let n = *x as u16 * from as u16 + carry;
            *x = (n % to as u16) as u8;
            carry = n / to as u16;
        }
        while carry != 0 {
            out.push((carry % to as u16) as u8);
            carry /= to as u16;
        }
    }
    while out.len() > 1 && out.last() == Some(&0) {
        out.pop();
    }
    out.reverse();
    out
}

fn render(digits: Vec<u8>, style: BaseStyle) -> String {
    let width = digits.len().max(style.min_width);
    let pad = width - digits.len();
    let mut output = String::with_capacity(width + width / style.group.max(1));

    for i in 0..width {
        if style.group > 0 && i > 0 && (width - i) % style.group == 0 {
            output.push('_');
        }
        let d = if i < pad { 0 } else { digits[i - pad] };
        output.push(match d {
            0..=9 => (b'0' + d) as char,
            _ if style.upper => (b'A' + d - 10) as char,
            _ => (b'a' + d - 10) as char,
        });
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    check_base(from, 0)?;
    check_base(to, 1)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }
    let parsed = parse(input, from, 0)?;
    Ok(render(rebase(&parsed, from, to), style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    check_base(base, 0)?;
    if style.group > 8 || style.min_width > 256 {
        return Err(BaseError::BadStyle);
    }

    let a = parse(left, base, 0)?;
    let b = parse(right, base, 1)?;
    let mut sum = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut ia = a.len();
    let mut ib = b.len();
    let mut carry = 0u16;

    while ia > 0 || ib > 0 || carry != 0 {
        let mut n = carry;
        if ia > 0 {
            ia -= 1;
            n += a[ia] as u16;
        }
        if ib > 0 {
            ib -= 1;
            n += b[ib] as u16;
        }
        sum.push((n % base as u16) as u8);
        carry = n / base as u16;
    }
    if sum.is_empty() {
        sum.push(0);
    }
    sum.reverse();
    Ok(render(sum, style))
}

fn mul_add_small(number: &mut Vec<u32>, multiplier: u32, addend: u32) {
    let mut carry = addend as u64;
    for limb in number.iter_mut() {
        let value = (*limb as u64) * multiplier as u64 + carry;
        *limb = value as u32;
        carry = value >> 32;
    }
    while carry != 0 {
        number.push(carry as u32);
        carry >>= 32;
    }
}

fn from_digits(digits: &[u8], base: u8) -> Vec<u32> {
    let mut number = Vec::new();
    for &digit in digits {
        mul_add_small(&mut number, base as u32, digit as u32);
    }
    number
}

fn div_rem_small(number: &mut Vec<u32>, divisor: u32) -> u32 {
    let mut remainder = 0u64;
    for limb in number.iter_mut().rev() {
        let value = (remainder << 32) | *limb as u64;
        *limb = (value / divisor as u64) as u32;
        remainder = value % divisor as u64;
    }
    while number.last() == Some(&0) {
        number.pop();
    }
    remainder as u32
}

fn render(mut number: Vec<u32>, base: u8, style: &BaseStyle) -> String {
    let alphabet = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };

    let mut reversed = Vec::new();
    if number.is_empty() {
        reversed.push(b'0');
    } else {
        while !number.is_empty() {
            reversed.push(alphabet[div_rem_small(&mut number, base as u32) as usize]);
        }
    }
    reversed.reverse();

    if reversed.len() < style.min_width {
        let mut padded = vec![b'0'; style.min_width - reversed.len()];

        reversed = padded;

    let mut output = Vec::with_capacity(reversed.len() + reversed.len() / style.group.max(1));
    for (index, &byte) in reversed.iter().enumerate() {
        if style.group != 0 && index != 0 && (reversed.len() - index) % style.group == 0 {

        output.push(byte);
    }
    String::from_utf8(output).unwrap()

    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },

fn base_check(base: u8, which: usize) -> Result<(), BaseError> {
    if base < 2 || base > 36 {
        Err(BaseError::BadBase { which, value: base })
    } else {
        Ok(())
    }
}

fn value(b: u8) -> Option<u32> {
    match b {
        b'0'..=b'9' => Some((b - b'0') as u32),
        b'A'..=b'Z' => Some((b - b'A' + 10) as u32),
        b'a'..=b'z' => Some((b - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse(s: &str, radix: u8, operand: usize) -> Result<Vec<u32>, BaseError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let start = match bytes.iter().position(|&b| b == b'#') {
        Some(i) => {
            if bytes[..i] != *radix.to_string().as_bytes() {
                return Err(BaseError::InvalidTag { operand, index: 0 });
            }
            i + 1
        }
        None => 0,
    };
    if start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let mut ds = Vec::new();
    for (i, &b) in bytes[start..].iter().enumerate() {
        if b == b'_' {
            continue;
        }
        match value(b) {
            Some(d) if d < radix as u32 => ds.push(d),
            _ => return Err(BaseError::InvalidDigit { operand, index: start + i }),
        }
    }
    for (i, &b) in bytes[start..].iter().enumerate() {
        if b == b'_' && (i == 0 || i + 1 == bytes.len() - start
            || bytes[start + i - 1] == b'_' || bytes[start + i + 1] == b'_')
        {
            return Err(BaseError::Separator { operand, index: start + i });
        }
    }
    if ds.len() > 1 && ds[0] == 0 {
        return Err(BaseError::LeadingZero { operand, index: start });
    }
    Ok(ds)
}

// Little-endian arbitrary precision limbs in base 1,000,000,000.
const LIMB_BASE: u64 = 1_000_000_000;

fn add_digit(number: &mut Vec<u32>, radix: u8, digit: u32) {
    let mut carry = digit as u64;
    for limb in number.iter_mut() {
        let n = *limb as u64 * radix as u64 + carry;
        *limb = (n % LIMB_BASE) as u32;
        carry = n / LIMB_BASE;
    }
    while carry != 0 {
        number.push((carry % LIMB_BASE) as u32);
        carry /= LIMB_BASE;
    }
}

fn number_from_digits(ds: &[u32], radix: u8) -> Vec<u32> {
    let mut n = vec![0];
    for &d in ds {
        add_digit(&mut n, radix, d);
    }
    n
}

fn div_limb_number(n: &mut Vec<u32>, divisor: u8) -> u8 {
    let mut rem = 0u64;
    for limb in n.iter_mut().rev() {
        let cur = rem * LIMB_BASE + *limb as u64;
        *limb = (cur / divisor as u64) as u32;
        rem = cur % divisor as u64;
    }
    while n.len() > 1 && n.last() == Some(&0) {
        n.pop();
    }
    rem as u8
}

fn format(mut n: Vec<u32>, radix: u8, style: BaseStyle) -> String {
    let mut ds = Vec::new();
    if n.len() == 1 && n[0] == 0 {
        ds.push(0);
    } else {
        while !(n.len() == 1 && n[0] == 0) {
            ds.push(div_limb_number(&mut n, radix));
        }
        ds.reverse();
    }

    let width = style.min_width.max(ds.len());
    let zeros = width - ds.len();
    let alpha = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };
    let mut out = String::with_capacity(width + if style.group == 0 { 0 } else { width / style.group });
    for i in 0..width {
        if style.group != 0 && i > 0 && (width - i) % style.group == 0 {
            out.push('_');
        }
        let d = if i < zeros { 0 } else { ds[i - zeros] };
        out.push(alpha[d as usize] as char);
    }
    out
}

fn valid_style(s: &BaseStyle) -> Result<(), BaseError> {
    if s.group > 8 || s.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

pub fn convert_base(input: &str, from: u8, to: u8, style: BaseStyle) -> Result<String, BaseError> {
    base_check(from, 0)?;
    base_check(to, 1)?;
    valid_style(&style)?;
    let ds = parse(input, from, 0)?;
    Ok(format(number_from_digits(&ds, from), to, style))
}

pub fn add_base(left: &str, right: &str, base: u8, style: BaseStyle) -> Result<String, BaseError> {
    base_check(base, 0)?;
    valid_style(&style)?;
    let a = number_from_digits(&parse(left, base, 0)?, base);
    let b = number_from_digits(&parse(right, base, 1)?, base);
    let mut sum = vec![0u32; a.len().max(b.len())];
    let mut carry = 0u64;
    for i in 0..sum.len() {
        let x = a.get(i).copied().unwrap_or(0) as u64
            + b.get(i).copied().unwrap_or(0) as u64 + carry;
        sum[i] = (x % LIMB_BASE) as u32;
        carry = x / LIMB_BASE;
    }
    if carry != 0 {
        sum.push(carry as u32);
    }
    while sum.len() > 1 && sum.last() == Some(&0) {
        sum.pop();
    }
    Ok(format(sum, base, style))
}

fn validated_digits(s: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if s.is_empty() {
        return Err(BaseError::Empty { operand });
    }
    let bytes = s.as_bytes();
    let start = match bytes.iter().position(|&b| b == b'#') {
        None => 0,

            if &s[..i] != radix.to_string() {

            }

        }

    if start == bytes.len() {
        return Err(BaseError::Empty { operand });

    let part = &bytes[start..];
    let mut digit_values = Vec::new();
    for (i, &b) in part.iter().enumerate() {
        if b == b'_' {
            continue;
        }
        match value_of(b) {
            Some(v) if v < radix => digit_values.push(v),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: start + i,
                });
            }
        }
    }

    for (i, &b) in part.iter().enumerate() {
        if b == b'_'
            && (i == 0
                || i + 1 >= part.len()
                || value_of(part[i - 1]).map_or(true, |v| v >= radix)
                || value_of(part[i + 1]).map_or(true, |v| v >= radix))
        {
            return Err(BaseError::Separator {
                operand,
                index: start + i,
            });
        }
    }

    if digit_values.len() >= 2 && digit_values[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: start,
        });
    }
    Ok(digit_values)
}

fn multiply_small(number: &mut Vec<u32>, radix: u32) {
    let mut carry = 0u32;
    for limb in number.iter_mut() {
        let x = *limb * radix + carry;
        *limb = x % 1_000_000_000;
        carry = x / 1_000_000_000;
    }
    if carry != 0 {
        number.push(carry);
    }
}

fn add_small(number: &mut Vec<u32>, value: u32) {
    let mut carry = value;
    let mut i = 0;
    while carry != 0 && i < number.len() {
        let x = number[i] + carry;
        number[i] = x % 1_000_000_000;
        carry = x / 1_000_000_000;
        i += 1;
    }
    if carry != 0 {
        number.push(carry);
    }
}

    for digit in digits {
        multiply_small(&mut n, radix as u32);
        add_small(&mut n, digit as u32);
    }
    while n.len() > 1 && n.last() == Some(&0) {
        n.pop();

    Ok(n)

fn render(mut number: Vec<u32>, radix: u8, style: &BaseStyle) -> String {
    let mut raw = Vec::new();
    if number.len() == 1 && number[0] == 0 {
        raw.push(0u8);
    } else {
        while !(number.len() == 1 && number[0] == 0) {
            raw.push(div_small(&mut number, radix as u32) as u8);
        }
        raw.reverse();
    }

    let mut chars: Vec<char> = raw
        .iter()
        .map(|&d| {
            if d < 10 {
                (b'0' + d) as char
            } else if style.upper {
                (b'A' + d - 10) as char
            } else {
                (b'a' + d - 10) as char
            }
        })
        .collect();

    if chars.len() < style.min_width {
        let mut padded = vec!['0'; style.min_width - chars.len()];
        padded.extend(chars);
        chars = padded;
    }

    let mut output = String::with_capacity(chars.len() + chars.len() / 2);
    for (i, ch) in chars.iter().enumerate() {
        if style.group > 0 && i > 0 && (chars.len() - i) % style.group == 0 {
            output.push('_');
        }
        output.push(*ch);
    }
    output
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let payload_start = if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &bytes[..hash] != expected.as_bytes() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        hash + 1
    } else {
        0
    };

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (i, &b) in payload.iter().enumerate() {
        if b == b'_' {
            continue;
        }
        match digit_value(b) {
            Some(v) if v < radix => digits.push(v),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + i,
                });
            }
        }
    }

    for (i, &b) in payload.iter().enumerate() {
        if b == b'_' {
            let left_ok = i > 0 && payload[i - 1] != b'_';
            let right_ok = i + 1 < payload.len() && payload[i + 1] != b'_';
            if !left_ok || !right_ok {
                return Err(BaseError::Separator {
                    operand,
                    index: payload_start + i,
                });
            }
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

    for d in value.iter_mut() {

        *d = (n % 10) as u8;
        carry = n / 10;

    while carry != 0 {
        value.push((carry % 10) as u8);
        carry /= 10;
    }

    rem as u8
}

fn render(mut decimal: Vec<u8>, radix: u8, style: BaseStyle) -> String {
    let mut digits = Vec::new();
    if decimal.iter().all(|&d| d == 0) {
        digits.push(0);
    } else {
        while decimal.iter().any(|&d| d != 0) {
            digits.push(div_small(&mut decimal, radix));
        }
        digits.reverse();
    }

    let pad = style.min_width.saturating_sub(digits.len());
    let mut out = Vec::with_capacity(digits.len() + pad + digits.len() / style.group.max(1));
    out.extend(std::iter::repeat(0).take(pad));
    out.extend(digits);

    let alphabet = if style.upper {
        b"0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ"
    } else {
        b"0123456789abcdefghijklmnopqrstuvwxyz"
    };
    let count = out.len();
    let mut result = String::with_capacity(count + if style.group == 0 { 0 } else { count / style.group });
    for (i, d) in out.iter().enumerate() {
        if style.group != 0 && i > 0 && (count - i) % style.group == 0 {
            result.push('_');
        }
        result.push(alphabet[*d as usize] as char);
    }
    result
}

fn check_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group > 8 || style.min_width > 256 {
        Err(BaseError::BadStyle)
    } else {
        Ok(())
    }
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    check_style(&style)?;
    let digits = parse_operand(input, from, 0)?;
    Ok(render(to_decimal(&digits, from), to, style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    check_style(&style)?;
    let a = parse_operand(left, base, 0)?;
    let b = parse_operand(right, base, 1)?;
    let mut sum = to_decimal(&a, base);
    let other = to_decimal(&b, base);
    let n = sum.len().max(other.len());
    sum.resize(n, 0);
    let mut carry = 0u8;
    for i in 0..n {
        let x = sum[i] + other.get(i).copied().unwrap_or(0) + carry;
        sum[i] = x % 10;
        carry = x / 10;
    }
    if carry != 0 {
        sum.push(carry);
    }
    Ok(render(sum, base, style))
}
