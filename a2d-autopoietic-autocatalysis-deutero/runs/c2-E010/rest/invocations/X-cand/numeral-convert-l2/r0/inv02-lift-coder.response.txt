- Check radix arguments in order, then validate style, then validate operands left to right.
- The first `#` is the tag boundary; its prefix must exactly match the radix in decimal. Later `#` bytes are invalid digits.
- Check empty input and empty tagged payload before scanning digits; scan all invalid digits before separators, then check leading zeros.
- Error indices are UTF-8 byte offsets in the complete input. Underscores must sit between digits.
- Values and addition use arbitrary-precision digit vectors; zero renders as `0`, with padding and grouping applied afterward.
- Both solutions use the same specified input and output behavior; Solution B uses an independent arithmetic representation.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn validate_base(value: u8, which: usize) -> Result<(), BaseError> {
    if (2..=36).contains(&value) {
        Ok(())
    } else {
        Err(BaseError::BadBase { which, value })
    }
}

fn validate_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group <= 8 && style.min_width <= 256 {
        Ok(())
    } else {
        Err(BaseError::BadStyle)
    }
}

fn digit_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn parse_operand(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if input.is_empty() {
        return Err(BaseError::Empty { operand });
    }

    let bytes = input.as_bytes();
    let mut payload_start = 0usize;
    if let Some(hash) = bytes.iter().position(|&b| b == b'#') {
        let expected = radix.to_string();
        if &input[..hash] != expected.as_str() {
            return Err(BaseError::InvalidTag { operand, index: 0 });
        }
        payload_start = hash + 1;
    }

    if payload_start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let payload = &bytes[payload_start..];
    let mut digits = Vec::new();

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_' {
            continue;
        }
        match digit_value(byte) {
            Some(value) if value < radix => digits.push(value),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: payload_start + offset,
                });
            }
        }
    }

    for (offset, &byte) in payload.iter().enumerate() {
        if byte == b'_'
            && (offset == 0
                || offset + 1 == payload.len()
                || digit_value(payload[offset - 1]).is_none()
                || digit_value(payload[offset + 1]).is_none())
        {
            return Err(BaseError::Separator {
                operand,
                index: payload_start + offset,
            });
        }
    }

    if digits.len() >= 2 && digits[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: payload_start,
        });
    }

    Ok(digits)
}

fn multiply_add_msd(value: &mut Vec<u8>, radix: u8, digit: u8) {
    let mut carry = digit as u16;
    for current in value.iter_mut().rev() {
        let total = (*current as u16) * radix as u16 + carry;
        *current = (total % 10) as u8;
        carry = total / 10;
    }
    while carry != 0 {
        value.insert(0, (carry % 10) as u8);
        carry /= 10;
    }
}

fn parse_value(input: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    let digits = parse_operand(input, radix, operand)?;
    let mut value = vec![0];
    for digit in digits {
        multiply_add_msd(&mut value, radix, digit);
    }
    while value.len() > 1 && value[0] == 0 {
        value.remove(0);
    }
    Ok(value)
}

fn render_value(value: &[u8], radix: u8, style: &BaseStyle) -> String {
    let mut digits = value.to_vec();
    while digits.len() > 1 && digits[0] == 0 {
        digits.remove(0);
    }

    let mut output_digits = Vec::new();
    if digits.len() == 1 && digits[0] == 0 {
        output_digits.push(0u8);
    } else {
        while digits.len() > 1 || digits[0] >= radix {
            let mut remainder = 0u16;
            let mut quotient = Vec::with_capacity(digits.len());
            let mut nonzero_seen = false;
            for &digit in &digits {
                let current = remainder * 10 + digit as u16;
                let qdigit = (current / radix as u16) as u8;
                remainder = current % radix as u16;
                if qdigit != 0 || nonzero_seen {
                    quotient.push(qdigit);
                    nonzero_seen = true;
                }
            }
            output_digits.push(remainder as u8);
            digits = if quotient.is_empty() { vec![0] } else { quotient };
        }
        output_digits.push(digits[0]);
        output_digits.reverse();
    }

    let mut chars = Vec::with_capacity(output_digits.len());
    for digit in output_digits {
        let ch = if digit < 10 {
            (b'0' + digit) as char
        } else if style.upper {
            (b'A' + digit - 10) as char
        } else {
            (b'a' + digit - 10) as char
        };
        chars.push(ch);
    }

    if chars.len() < style.min_width {
        let mut padded = vec!['0'; style.min_width - chars.len()];
        padded.extend(chars);
        chars = padded;
    }

    let mut result = String::with_capacity(chars.len() + chars.len() / 2);
    for (index, ch) in chars.iter().enumerate() {
        if style.group != 0
            && index != 0
            && (chars.len() - index) % style.group == 0
        {
            result.push('_');
        }
        result.push(*ch);
    }
    result
}

fn checked_single(
    input: &str,
    radix: u8,
    operand: usize,
) -> Result<Vec<u8>, BaseError> {
    parse_value(input, radix, operand)
}

fn add_decimal(left: &[u8], right: &[u8]) -> Vec<u8> {
    let mut result = Vec::with_capacity(left.len().max(right.len()) + 1);
    let mut i = left.len();
    let mut j = right.len();
    let mut carry = 0u8;

    while i > 0 || j > 0 || carry > 0 {
        let a = if i > 0 {
            i -= 1;
            left[i]
        } else {
            0
        };
        let b = if j > 0 {
            j -= 1;
            right[j]
        } else {
            0
        };
        let sum = a + b + carry;
        result.push(sum % 10);
        carry = sum / 10;
    }
    result.reverse();
    result
}

fn format_with_validation(
    value: &[u8],
    radix: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_style(&style)?;
    Ok(render_value(value, radix, &style))
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(from, 0)?;
    validate_base(to, 1)?;
    validate_style(&style)?;
    let value = checked_single(input, from, 0)?;
    Ok(render_value(&value, to, &style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    validate_base(base, 0)?;
    validate_style(&style)?;
    let left_value = checked_single(left, base, 0)?;
    let right_value = checked_single(right, base, 1)?;
    let sum = add_decimal(&left_value, &right_value);

    // The decimal digit vector stores most significant digit first.
    // Render it directly through the same radix conversion path.
    let mut result_digits = Vec::new();
    let mut decimal = sum;
    if decimal.len() == 1 && decimal[0] == 0 {
        result_digits.push(0);
    } else {
        while decimal.len() > 1 || decimal[0] >= base {
            let mut remainder = 0u16;
            let mut quotient = Vec::with_capacity(decimal.len());
            let mut started = false;
            for &digit in &decimal {
                let current = remainder * 10 + digit as u16;
                let qdigit = (current / base as u16) as u8;
                remainder = current % base as u16;
                if qdigit != 0 || started {
                    quotient.push(qdigit);
                    started = true;
                }
            }
            result_digits.push(remainder as u8);
            decimal = if quotient.is_empty() { vec![0] } else { quotient };
        }
        result_digits.push(decimal[0]);
        result_digits.reverse();
    }

    let chars: Vec<char> = result_digits
        .iter()
        .map(|&digit| {
            if digit < 10 {
                (b'0' + digit) as char
            } else if style.upper {
                (b'A' + digit - 10) as char
            } else {
                (b'a' + digit - 10) as char
            }
        })
        .collect();

    let mut chars = chars;
    if chars.len() < style.min_width {
        let mut padded = vec!['0'; style.min_width - chars.len()];
        padded.extend(chars);
        chars = padded;
    }

    let mut result = String::with_capacity(chars.len() + chars.len() / 2);
    for (index, ch) in chars.iter().enumerate() {
        if style.group != 0 && index != 0 && (chars.len() - index) % style.group == 0 {
            result.push('_');
        }
        result.push(*ch);
    }
    Ok(result)
}
``` 

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BaseStyle {
    pub upper: bool,
    pub group: usize,
    pub min_width: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BaseError {
    BadBase { which: usize, value: u8 },
    BadStyle,
    Empty { operand: usize },
    InvalidTag { operand: usize, index: usize },
    InvalidDigit { operand: usize, index: usize },
    Separator { operand: usize, index: usize },
    LeadingZero { operand: usize, index: usize },
}

fn check_radix(radix: u8, which: usize) -> Result<(), BaseError> {
    if (2..=36).contains(&radix) {
        Ok(())
    } else {
        Err(BaseError::BadBase {
            which,
            value: radix,
        })
    }
}

fn check_style(style: &BaseStyle) -> Result<(), BaseError> {
    if style.group <= 8 && style.min_width <= 256 {
        Ok(())
    } else {
        Err(BaseError::BadStyle)
    }
}

fn value_of(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'Z' => Some(byte - b'A' + 10),
        b'a'..=b'z' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn validated_digits(s: &str, radix: u8, operand: usize) -> Result<Vec<u8>, BaseError> {
    if s.is_empty() {
        return Err(BaseError::Empty { operand });
    }
    let bytes = s.as_bytes();
    let start = match bytes.iter().position(|&b| b == b'#') {
        None => 0,
        Some(i) => {
            if &s[..i] != radix.to_string() {
                return Err(BaseError::InvalidTag { operand, index: 0 });
            }
            i + 1
        }
    };
    if start == bytes.len() {
        return Err(BaseError::Empty { operand });
    }

    let part = &bytes[start..];
    let mut digit_values = Vec::new();
    for (i, &b) in part.iter().enumerate() {
        if b == b'_' {
            continue;
        }
        match value_of(b) {
            Some(v) if v < radix => digit_values.push(v),
            _ => {
                return Err(BaseError::InvalidDigit {
                    operand,
                    index: start + i,
                });
            }
        }
    }

    for (i, &b) in part.iter().enumerate() {
        if b == b'_'
            && (i == 0
                || i + 1 >= part.len()
                || value_of(part[i - 1]).map_or(true, |v| v >= radix)
                || value_of(part[i + 1]).map_or(true, |v| v >= radix))
        {
            return Err(BaseError::Separator {
                operand,
                index: start + i,
            });
        }
    }

    if digit_values.len() >= 2 && digit_values[0] == 0 {
        return Err(BaseError::LeadingZero {
            operand,
            index: start,
        });
    }
    Ok(digit_values)
}

fn multiply_small(number: &mut Vec<u32>, radix: u32) {
    let mut carry = 0u32;
    for limb in number.iter_mut() {
        let x = *limb * radix + carry;
        *limb = x % 1_000_000_000;
        carry = x / 1_000_000_000;
    }
    if carry != 0 {
        number.push(carry);
    }
}

fn add_small(number: &mut Vec<u32>, value: u32) {
    let mut carry = value;
    let mut i = 0;
    while carry != 0 && i < number.len() {
        let x = number[i] + carry;
        number[i] = x % 1_000_000_000;
        carry = x / 1_000_000_000;
        i += 1;
    }
    if carry != 0 {
        number.push(carry);
    }
}

fn parse_number(s: &str, radix: u8, operand: usize) -> Result<Vec<u32>, BaseError> {
    let digits = validated_digits(s, radix, operand)?;
    let mut n = vec![0u32];
    for digit in digits {
        multiply_small(&mut n, radix as u32);
        add_small(&mut n, digit as u32);
    }
    while n.len() > 1 && n.last() == Some(&0) {
        n.pop();
    }
    Ok(n)
}

fn add_numbers(a: &[u32], b: &[u32]) -> Vec<u32> {
    let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
    let mut carry = 0u64;
    for i in 0..a.len().max(b.len()) {
        let x = *a.get(i).unwrap_or(&0) as u64
            + *b.get(i).unwrap_or(&0) as u64
            + carry;
        out.push((x % 1_000_000_000) as u32);
        carry = x / 1_000_000_000;
    }
    if carry != 0 {
        out.push(carry as u32);
    }
    out
}

fn div_small(number: &mut Vec<u32>, divisor: u32) -> u32 {
    let mut remainder = 0u64;
    for limb in number.iter_mut().rev() {
        let current = remainder * 1_000_000_000 + *limb as u64;
        *limb = (current / divisor as u64) as u32;
        remainder = current % divisor as u64;
    }
    while number.len() > 1 && number.last() == Some(&0) {
        number.pop();
    }
    remainder as u32
}

fn render(mut number: Vec<u32>, radix: u8, style: &BaseStyle) -> String {
    let mut raw = Vec::new();
    if number.len() == 1 && number[0] == 0 {
        raw.push(0u8);
    } else {
        while !(number.len() == 1 && number[0] == 0) {
            raw.push(div_small(&mut number, radix as u32) as u8);
        }
        raw.reverse();
    }

    let mut chars: Vec<char> = raw
        .iter()
        .map(|&d| {
            if d < 10 {
                (b'0' + d) as char
            } else if style.upper {
                (b'A' + d - 10) as char
            } else {
                (b'a' + d - 10) as char
            }
        })
        .collect();

    if chars.len() < style.min_width {
        let mut padded = vec!['0'; style.min_width - chars.len()];
        padded.extend(chars);
        chars = padded;
    }

    let mut output = String::with_capacity(chars.len() + chars.len() / 2);
    for (i, ch) in chars.iter().enumerate() {
        if style.group > 0 && i > 0 && (chars.len() - i) % style.group == 0 {
            output.push('_');
        }
        output.push(*ch);
    }
    output
}

pub fn convert_base(
    input: &str,
    from: u8,
    to: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    check_radix(from, 0)?;
    check_radix(to, 1)?;
    check_style(&style)?;
    let number = parse_number(input, from, 0)?;
    Ok(render(number, to, &style))
}

pub fn add_base(
    left: &str,
    right: &str,
    base: u8,
    style: BaseStyle,
) -> Result<String, BaseError> {
    check_radix(base, 0)?;
    check_style(&style)?;
    let a = parse_number(left, base, 0)?;
    let b = parse_number(right, base, 1)?;
    Ok(render(add_numbers(&a, &b), base, &style))
}
```
