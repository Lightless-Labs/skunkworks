- Validation follows the explicitly stated order, including maintenance gates and vend’s stock-before-payment checks.
- Greedy change is mandatory when exact; otherwise choose the exact combination with the fewest coins, breaking ties lexicographically by descending denominations.
- Tube capacity is checked after change selection and escrow deposit; it does not trigger a search for a different combination.
- Empty refunds and identical reconciliations succeed and each create an event.
- All additions that may exceed a bound are checked without wrapping; audit counters use saturating arithmetic.
- Solution B uses a separate change-search strategy but follows the same specified behavior.

const DENOMS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }
        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            if !valid_id(&product.id) {
                return Err(Error::BadId { id: product.id.clone() });
            }
            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId { id: product.id.clone() });
            }
            seen.push(&product.id);
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id.clone() });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit { id: product.id.clone() });
            }
        }
        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    fn product_index(&self, id: &str) -> Option<usize> {
        self.products.iter().position(|p| p.id == id)

        let seq = self.events.len();
        self.events.push(Event { seq, kind });

        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let index = match coin {
            1 => 0,
            3 => 1,
            4 => 2,
            10 => 3,
            _ => return Err(Error::BadCoin { coin }),
        };
        let new_credit = u32::from(self.credit) + u32::from(coin);
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit = new_credit as u16;
        self.escrow.push(coin);
        self.inserted = self.inserted.saturating_add(u64::from(coin));
        self.log(EventKind::Inserted { coin, credit: self.credit });
        let _ = index;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if self.product_index(id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_string() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_string());
        self.log(EventKind::Selected { id: id.to_string() });
        Ok(())
    }

    fn choose_change(&self, amount: u16) -> Result<[u16; 4], Error> {
        let mut greedy = [0u16; 4];
        let mut remaining = amount;
        for i in (0..4).rev() {
            let take = self.tubes[i].min(remaining / DENOMS[i]);
            greedy[i] = take;
            remaining -= take * DENOMS[i];
        }
        if remaining == 0 {
            return Ok(greedy);
        }

        let mut best: Option<[u16; 4]> = None;
        let max10 = self.tubes[3].min(amount / 10);
        for c10 in 0..=max10 {
            let rem10 = amount - c10 * 10;
            let max4 = self.tubes[2].min(rem10 / 4);
            for c4 in 0..=max4 {
                let rem4 = rem10 - c4 * 4;
                let max3 = self.tubes[1].min(rem4 / 3);
                for c3 in 0..=max3 {
                    let rem3 = rem4 - c3 * 3;
                    if rem3 > self.tubes[0] {
                        continue;
                    }
                    let candidate = [rem3, c3, c4, c10];
                    let total: u32 = candidate.iter().map(|&x| u32::from(x)).sum();
                    let replace = match best {
                        None => true,
                        Some(old) => {
                            let old_total: u32 = old.iter().map(|&x| u32::from(x)).sum();
                            total < old_total || (total == old_total && candidate[3..] > old[3..])
                        }
                    };
                    if replace {
                        best = Some(candidate);
                    }
                }
            }
        }
        best.ok_or(Error::NoChange { amount })
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let index = self.product_index(&id).expect("selected product invariant");
        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid { price, credit: self.credit });
        }
        let amount = self.credit - price;
        let selected_change = self.choose_change(amount)?;
        let mut proposed = self.tubes;
        let mut coins_desc = Vec::new();
        for i in (0..4).rev() {
            proposed[i] -= selected_change[i];
            for _ in 0..selected_change[i] {
                coins_desc.push(DENOMS[i]);
            }
        }
        for &coin in &self.escrow {
            let i = match coin {
                1 => 0,
                3 => 1,
                4 => 2,
                10 => 3,
                _ => unreachable!(),
            };
            proposed[i] += 1;
        }
        for (i, &count) in proposed.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        self.tubes = proposed;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        self.change = self.change.saturating_add(u64::from(amount));
        let receipt = Receipt { id: id.clone(), price, change: coins_desc.clone() };
        self.log(EventKind::Vended { id, price, change: coins_desc });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let credit = self.credit;
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        self.refunded = self.refunded.saturating_add(u64::from(credit));
        self.log(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.cmp(&b.id));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = self.product_index(id).ok_or_else(|| Error::UnknownProduct { id: id.to_string() })?;
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_string() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged { id: id.to_string(), old, new: price });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = self.product_index(id).ok_or_else(|| Error::UnknownProduct { id: id.to_string() })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = u32::from(self.products[index].stock) + u32::from(count);
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_string() });
        }
        self.products[index].stock = new_stock as u16;
        self.log(EventKind::Restocked { id: id.to_string(), added: count, stock: self.products[index].stock });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: value_of(&self.tubes).saturating_add(u64::from(self.credit)),
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let before_value = value_of(&self.tubes);
        let after_value = value_of(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift { before: before_value, after: after_value });
        }
        let before = self.tubes;
        self.tubes = actual;
        let delta = i128::from(after_value) - i128::from(before_value);
        let sum = i128::from(self.adjustment) + delta;
        self.adjustment = sum.clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        self.log(EventKind::Reconciled { before, after: actual });
        Ok(())
    }
}

fn valid_catalog_id(s: &str) -> bool {

fn tube_value(a: &[u16; 4]) -> u64 {
    a.iter().zip(D).map(|(&n, d)| u64::from(n) * u64::from(d)).sum()

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }
        for i in 0..products.len() {
            let p = &products[i];
            if !valid_catalog_id(&p.id) {
                return Err(Error::BadId { id: p.id.clone() });
            }
            if products[..i].iter().any(|old| old.id == p.id) {
                return Err(Error::DuplicateId { id: p.id.clone() });

            if p.price == 0 || p.price > 100 {
                return Err(Error::BadPrice { id: p.id.clone() });

                return Err(Error::StockLimit { id: p.id.clone() });
            }

                return Err(Error::TubeLimit { coin: D[i] });
            }

            items: products,
            state: State {
                credit: 0,
                selected: None,
                escrow: Vec::new(),
                tubes,
                maintenance: false,

            audit_refunded: 0,
            audit_sales: 0,
            audit_change: 0,
            audit_adjustment: 0,

            return Err(Error::MaintenanceActive);

        if !D.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.state.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);

        self.state.credit += coin;
        self.state.escrow.push(coin);

        self.record(EventKind::Inserted { coin, credit: self.state.credit });
        Ok(())

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.state.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if self.find(id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.state.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.state.selected = Some(id.to_owned());
        self.record(EventKind::Selected { id: id.to_owned() });
        Ok(())

    fn exact_search(
        tubes: &[u16; 4],
        target: u16,
        index: usize,
        chosen: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
    ) {
        if index == 4 {
            if target != 0 {
                return;
            }
            let candidate = *chosen;
            let count: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
            let improves = match best {
                None => true,
                Some(old) => {
                    let old_count: u32 = old.iter().map(|&n| u32::from(n)).sum();
                    if count != old_count {
                        count < old_count
                    } else {
                        (0..4).rev().find_map(|i| {
                            if candidate[i] != old[i] {
                                Some(candidate[i] > old[i])
                            } else {
                                None
                            }
                        }).unwrap_or(false)
                    }
                }
            };
            if improves {
                *best = Some(candidate);
            }
            return;
        }
        let denom = D[index];
        let max = tubes[index].min(target / denom);
        for n in 0..=max {
            chosen[index] = n;
            Self::exact_search(tubes, target - n * denom, index + 1, chosen, best);
        }
        chosen[index] = 0;
    }
    fn change_counts(&self, amount: u16) -> Result<[u16; 4], Error> {
        let mut greedy = [0; 4];
        let mut rem = amount;
        for i in (0..4).rev() {
            greedy[i] = self.state.tubes[i].min(rem / D[i]);
            rem -= greedy[i] * D[i];
        }
        if rem == 0 {
            return Ok(greedy);
        }
        let mut best = None;
        Self::exact_search(&self.state.tubes, amount, 0, &mut [0; 4], &mut best);
        best.ok_or(Error::NoChange { amount })
    }
    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.state.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let id = self.state.selected.clone().ok_or(Error::NoSelection)?;
        let idx = self.find(&id).expect("selection is a catalog ID");
        if self.items[idx].stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = self.items[idx].price;
        if self.state.credit < price {
            return Err(Error::Underpaid { price, credit: self.state.credit });
        }
        let amount = self.state.credit - price;
        let chosen = self.change_counts(amount)?;
        let mut new_tubes = self.state.tubes;
        let mut change_coins = Vec::new();
        for i in (0..4).rev() {
            new_tubes[i] -= chosen[i];
            for _ in 0..chosen[i] {
                change_coins.push(D[i]);
            }
        }
        for &coin in &self.state.escrow {
            let i = D.iter().position(|&d| d == coin).expect("escrow denomination invariant");
            new_tubes[i] += 1;
        }
        for i in 0..4 {
            if new_tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: D[i] });
            }
        }
        self.state.tubes = new_tubes;
        self.items[idx].stock -= 1;
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();
        self.audit_sales = self.audit_sales.saturating_add(u64::from(price));
        self.audit_change = self.audit_change.saturating_add(u64::from(amount));
        self.record(EventKind::Vended { id: id.clone(), price, change: change_coins.clone() });
        Ok(Receipt { id, price, change: change_coins })
    }
    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.state.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.state.escrow.iter().rev().copied().collect();
        self.audit_refunded = self.audit_refunded.saturating_add(u64::from(self.state.credit));
        self.state.credit = 0;
        self.state.selected = None;
        self.state.escrow.clear();
        self.record(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }
    pub fn products(&self) -> Vec<Product> {
        let mut out = self.items.clone();
        out.sort_by(|a, b| a.id.cmp(&b.id));
        out
    }
    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.state.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let idx = self.find(id).ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.state.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let idx = self.find(id).ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let stock = u32::from(self.items[idx].stock) + u32::from(count);
        if stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.items[idx].stock = stock as u16;
        self.record(EventKind::Restocked { id: id.to_owned(), added: count, stock: stock as u16 });
        Ok(())
    }
    pub fn events(&self) -> &[Event] {
        &self.log
    }
    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.state.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.state.credit != 0 {
            return Err(Error::Busy);
        }
        self.state.maintenance = enabled;
        self.state.selected = None;
        self.record(EventKind::Maintenance { enabled });
        Ok(())
    }
    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.audit_inserted,
            refunded: self.audit_refunded,
            sales: self.audit_sales,
            change: self.audit_change,
            adjustment: self.audit_adjustment,
            cash: tube_value(&self.state.tubes).saturating_add(u64::from(self.state.credit)),
        }
    }
    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.state.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.state.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: D[i],
                    expected: expected[i],
                    found: self.state.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: D[i] });
            }
        }
        let before_value = tube_value(&self.state.tubes);
        let after_value = tube_value(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift { before: before_value, after: after_value });
        }
        let before = self.state.tubes;
        self.state.tubes = actual;
        let delta = i128::from(after_value) - i128::from(before_value);
        self.audit_adjustment = (i128::from(self.audit_adjustment) + delta)
            .clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        self.record(EventKind::Reconciled { before, after: actual });
        Ok(())
    }
}
