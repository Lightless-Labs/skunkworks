const DENOMS: [u16; 4] = [1, 3, 4, 10];

fn denom_index(coin: u16) -> Option<usize> {
    DENOMS.iter().position(|&d| d == coin)
}

fn is_valid_id(id: &str) -> bool {
    !id.is_empty() && id.len() <= 8 && id.bytes().all(|b| b'a' <= b && b <= b'z')
}

fn product_index(products: &[Product], id: &str) -> Option<usize> {
    products.iter().position(|p| p.id == id)
}

fn push_event(events: &mut Vec<Event>, kind: EventKind) {
    let seq = events.len();
    events.push(Event { seq, kind });
}

fn tube_value(tubes: &[u16; 4]) -> u64 {
    let mut sum = 0u64;
    for i in 0..4 {
        sum += u64::from(tubes[i]) * u64::from(DENOMS[i]);
    }
    sum
}

fn best_exact(tubes: &[u16; 4], amount: u16) -> Option<[u16; 4]> {
    let mut best = None;
    let mut min_coins = u32::MAX;
    for ten in 0..=tubes[3].min(amount / 10) {
        for four in 0..=tubes[2].min((amount - ten * 10) / 4) {
            for three in 0..=tubes[1].min((amount - ten * 10 - four * 4) / 3) {
                let one = amount - ten * 10 - four * 4 - three * 3;
                if one > tubes[0] {
                    continue;
                }
                let candidate = [one, three, four, ten];
                let n = u32::from(one) + u32::from(three) + u32::from(four) + u32::from(ten);
                let lex_better = best.map_or(true, |old: [u16; 4]| {
                    (ten, four, three, one) > (old[3], old[2], old[1], old[0])
                });
                if n < min_coins || (n == min_coins && lex_better) {
                    min_coins = n;
                    best = Some(candidate);
                }
            }
        }
    }
    best
}

fn plan_change(tubes: &[u16; 4], amount: u16) -> Option<[u16; 4]> {
    let mut counts = [0u16; 4];
    let mut left = amount;
    for i in (0..4).rev() {
        counts[i] = tubes[i].min(left / DENOMS[i]);
        left -= counts[i] * DENOMS[i];
    }
    if left == 0 { Some(counts) } else { best_exact(tubes, amount) }
}

fn materialize(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for i in (0..4).rev() {
        coins.extend(std::iter::repeat(DENOMS[i]).take(usize::from(counts[i])));
    }
    coins
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }
        for (i, product) in products.iter().enumerate() {
            if !is_valid_id(&product.id) {
                return Err(Error::BadId { id: product.id.clone() });
            }
            if products[..i].iter().any(|p| p.id == product.id) {
                return Err(Error::DuplicateId { id: product.id.clone() });
            }
            if product.price == 0 || product.price > 100 {
                return Err(Error::BadPrice { id: product.id.clone() });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit { id: product.id.clone() });
            }
        }
        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if denom_index(coin).is_none() {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit += coin;
        self.escrow.push(coin);
        self.inserted = self.inserted.saturating_add(u64::from(coin));
        push_event(&mut self.events, EventKind::Inserted { coin, credit: self.credit });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if product_index(&self.products, id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        push_event(&mut self.events, EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let id = match self.selected.clone() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };
        let idx = product_index(&self.products, &id).expect("selected IDs always belong to catalog");
        let price = self.products[idx].price;
        if self.products[idx].stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < price {
            return Err(Error::Underpaid { price, credit: self.credit });
        }
        let amount = self.credit - price;
        let chosen = plan_change(&self.tubes, amount).ok_or(Error::NoChange { amount })?;
        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= chosen[i];
        }
        for &coin in &self.escrow {
            let i = denom_index(coin).expect("escrow only contains accepted denominations");
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        let coins = materialize(chosen);
        self.products[idx].stock -= 1;
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        self.change = self.change.saturating_add(coins.iter().map(|&c| u64::from(c)).sum());
        push_event(
            &mut self.events,
            EventKind::Vended { id: id.clone(), price, change: coins.clone() },
        );
        Ok(Receipt { id, price, change: coins })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.refunded = self.refunded.saturating_add(u64::from(self.credit));
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        push_event(&mut self.events, EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut result = self.products.clone();
        result.sort_by(|a, b| a.id.cmp(&b.id));
        result
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(idx) = product_index(&self.products, id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[idx].price;
        self.products[idx].price = price;
        push_event(&mut self.events, EventKind::PriceChanged { id: id.to_owned(), old, new: price });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(idx) = product_index(&self.products, id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = u32::from(self.products[idx].stock) + u32::from(count);
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[idx].stock = new_stock as u16;
        push_event(
            &mut self.events,
            EventKind::Restocked { id: id.to_owned(), added: count, stock: new_stock as u16 },
        );
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if self.maintenance == enabled {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        push_event(&mut self.events, EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value(&self.tubes) + u64::from(self.credit),
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let before_value = tube_value(&self.tubes);
        let after_value = tube_value(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift { before: before_value, after: after_value });
        }
        let before = self.tubes;
        let delta = i128::from(after_value) - i128::from(before_value);
        self.adjustment = (i128::from(self.adjustment) + delta)
            .clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        self.tubes = actual;
        push_event(&mut self.events, EventKind::Reconciled { before, after: actual });
        Ok(())
    }
}

    fn change_counts(&self, amount: u16) -> Option<[u16; 4]> {
        let mut counts = [0u16; 4];
        let mut remaining = amount as u32;
        for i in (0..4).rev() {
            let take = (self.tubes[i] as u32).min(remaining / DENOMS[i] as u32);
            counts[i] = take as u16;
            remaining -= take * DENOMS[i] as u32;
        }
        if remaining == 0 {
            return Some(counts);
        }

        let mut best: Option<[u16; 4]> = None;
        fn search(
            i: isize,
            remaining: u32,
            tubes: &[u16; 4],
            counts: &mut [u16; 4],
            best: &mut Option<[u16; 4]>,
        ) {
            if i < 0 {
                if remaining == 0 {
                    let candidate = *counts;
                    let total: u32 = candidate.iter().map(|&x| x as u32).sum();
                    let better = match best {
                        None => true,
                        Some(old) => {
                            let old_total: u32 = old.iter().map(|&x| x as u32).sum();
                            total < old_total || (total == old_total && candidate[3..] > old[3..])
                        }
                    };
                    if better {
                        *best = Some(candidate);
                    }
                }
                return;
            }
            let idx = i as usize;
            let denom = DENOMS[idx] as u32;
            let max = (tubes[idx] as u32).min(remaining / denom);
            for n in 0..=max {
                counts[idx] = n as u16;
                search(i - 1, remaining - n * denom, tubes, counts, best);
            }
            counts[idx] = 0;
        }
        search(3, amount as u32, &self.tubes, &mut counts, &mut best);
        best
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);

            return Err(Error::NoSelection);

        if self.credit < price {
            return Err(Error::Underpaid {

                credit: self.credit,
            });

            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= counts[i];

        for &coin in &self.escrow {
            let i = DENOMS.iter().position(|&d| d == coin).unwrap();
            proposed[i] += 1;

            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        for i in (0..4).rev() {
            for _ in 0..counts[i] {
                change_coins.push(DENOMS[i]);
            }
        }

        let id_for_event = id.clone();

        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(price as u64);
        let change_value = amount as u64;
        self.change = self.change.saturating_add(change_value);

        self.tubes = actual;

        self.adjustment = self.adjustment.saturating_add(delta_i64);

            before,
            after: actual,

    fn counts_for_amount(&self, amount: u16) -> Option<[u16; 4]> {
        let mut best: Option<([u16; 4], u32)> = None;
        let mut c = [0u16; 4];

        fn visit(
            idx: usize,
            remaining: i32,
            tubes: &[u16; 4],
            c: &mut [u16; 4],
            best: &mut Option<([u16; 4], u32)>,
        ) {
            if idx == 4 {
                if remaining != 0 { return; }
                let total: u32 = c.iter().map(|&x| x as u32).sum();
                match best {
                    None => *best = Some((*c, total)),
                    Some((old, old_total)) => {
                        if total < *old_total || (total == *old_total && [c[3], c[2], c[1], c[0]] > [old[3], old[2], old[1], old[0]]) {
                            *best = Some((*c, total));
                        }
                    }
                }
                return;
            }
            let denom = DENOMS[3 - idx] as i32;
            let tube_idx = 3 - idx;
            let max = (tubes[tube_idx] as i32).min(remaining / denom);
            for n in 0..=max {
                c[tube_idx] = n as u16;
                visit(idx + 1, remaining - n * denom, tubes, c, best);
            }
            c[tube_idx] = 0;
        }

        visit(0, amount as i32, &self.tubes, &mut c, &mut best);
        best.map(|x| x.0)
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {

        if p.stock == 0 { return Err(Error::SoldOut { id }); }

            let mut rem = amount as u32;
            let mut c = [0u16; 4];
            for i in (0..4).rev() {
                let n = (self.tubes[i] as u32).min(rem / DENOMS[i] as u32);
                c[i] = n as u16;
                rem -= n * DENOMS[i] as u32;
            }
            if rem == 0 { Some(c) } else { None }
        };

            Some(c) => c,
            None => return Err(Error::NoChange { amount }),

        self.escrow.clear();

        self.change = self.change.saturating_add(amount as u64);
        self.log(EventKind::Vended { id: id.clone(), price, change: change.clone() });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance { return Err(Error::MaintenanceActive); }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.refunded = self.refunded.saturating_add(self.credit as u64);
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        self.products.values().cloned().collect()
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance { return Err(Error::MaintenanceRequired); }
        let Some(p) = self.products.get(id) else { return Err(Error::UnknownProduct { id: id.to_owned() }); };
        if self.credit != 0 { return Err(Error::Busy); }
        if !(1..=100).contains(&price) { return Err(Error::BadPrice { id: id.to_owned() }); }
        let old = p.price;
        self.products.get_mut(id).unwrap().price = price;
        self.log(EventKind::PriceChanged { id: id.to_owned(), old, new: price });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance { return Err(Error::MaintenanceRequired); }
        let Some(p) = self.products.get(id) else { return Err(Error::UnknownProduct { id: id.to_owned() }); };
        if count == 0 { return Err(Error::ZeroRestock); }
        let stock = p.stock as u32 + count as u32;
        if stock > 100 { return Err(Error::StockLimit { id: id.to_owned() }); }
        self.products.get_mut(id).unwrap().stock = stock as u16;
        self.log(EventKind::Restocked { id: id.to_owned(), added: count, stock: stock as u16 });
        Ok(())
    }

    pub fn events(&self) -> &[Event] { &self.events }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance { return Err(Error::AlreadyMode); }
        if self.credit != 0 { return Err(Error::Busy); }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    fn tube_value(t: &[u16; 4]) -> u64 {
        t.iter().zip(DENOMS).map(|(&n, d)| n as u64 * d as u64).sum()
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: Self::tube_value(&self.tubes).saturating_add(self.credit as u64),
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance { return Err(Error::MaintenanceRequired); }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch { coin: DENOMS[i], expected: expected[i], found: self.tubes[i] });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 { return Err(Error::TubeLimit { coin: DENOMS[i] }); }
        }
        let before_value = Self::tube_value(&self.tubes);
        let after_value = Self::tube_value(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift { before: before_value, after: after_value });
        }
        let before = self.tubes;
        self.tubes = actual;
        let delta = after_value as i64 - before_value as i64;
        self.adjustment = self.adjustment.saturating_add(delta);
        self.log(EventKind::Reconciled { before, after: actual });
        Ok(())
    }
}

    fn exact_search(
        tubes: &[u16; 4],
        target: u16,
        index: usize,
        chosen: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
    ) {
        if index == 4 {
            if target != 0 {
                return;
            }
            let candidate = *chosen;
            let count: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
            let improves = match best {
                None => true,
                Some(old) => {
                    let old_count: u32 = old.iter().map(|&n| u32::from(n)).sum();
                    if count != old_count {
                        count < old_count
                    } else {
                        (0..4).rev().find_map(|i| {
                            if candidate[i] != old[i] {
                                Some(candidate[i] > old[i])
                            } else {
                                None
                            }
                        }).unwrap_or(false)
                    }
                }
            };
            if improves {
                *best = Some(candidate);
            }
            return;
        }
        let denom = D[index];
        let max = tubes[index].min(target / denom);
        for n in 0..=max {
            chosen[index] = n;
            Self::exact_search(tubes, target - n * denom, index + 1, chosen, best);
        }
        chosen[index] = 0;
    }

            rem -= greedy[i] * D[i];

const DENOMS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())

fn value_of(counts: &[u16; 4]) -> u64 {
    counts
        .iter()
        .zip(DENOMS)
        .map(|(&count, denom)| u64::from(count) * u64::from(denom))

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        for product in &products {
            if !valid_id(&product.id) {

            }
            if seen.iter().any(|id| *id == product.id) {

            }
            if product.stock > 100 {
                return Err(Error::StockLimit { id: product.id.clone() });
            }
        }

                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,

            inserted: 0,

            change: 0,
            adjustment: 0,

    fn choose_change(&self, amount: u16) -> Result<[u16; 4], Error> {
        let mut greedy = [0u16; 4];
        let mut remaining = amount;
        for i in (0..4).rev() {
            let take = self.tubes[i].min(remaining / DENOMS[i]);
            greedy[i] = take;
            remaining -= take * DENOMS[i];
        }
        if remaining == 0 {
            return Ok(greedy);
        }

        let mut best: Option<[u16; 4]> = None;
        let max10 = self.tubes[3].min(amount / 10);
        for c10 in 0..=max10 {
            let rem10 = amount - c10 * 10;
            let max4 = self.tubes[2].min(rem10 / 4);
            for c4 in 0..=max4 {
                let rem4 = rem10 - c4 * 4;
                let max3 = self.tubes[1].min(rem4 / 3);
                for c3 in 0..=max3 {
                    let rem3 = rem4 - c3 * 3;
                    if rem3 > self.tubes[0] {
                        continue;
                    }
                    let candidate = [rem3, c3, c4, c10];
                    let total: u32 = candidate.iter().map(|&x| u32::from(x)).sum();
                    let replace = match best {
                        None => true,
                        Some(old) => {
                            let old_total: u32 = old.iter().map(|&x| u32::from(x)).sum();
                            total < old_total || (total == old_total && candidate[3..] > old[3..])
                        }
                    };
                    if replace {
                        best = Some(candidate);
                    }
                }
            }
        }
        best.ok_or(Error::NoChange { amount })
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let index = self.product_index(&id).expect("selected product invariant");
        let product = &self.products[index];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid { price, credit: self.credit });
        }
        let amount = self.credit - price;
        let selected_change = self.choose_change(amount)?;
        let mut proposed = self.tubes;
        let mut coins_desc = Vec::new();
        for i in (0..4).rev() {
            proposed[i] -= selected_change[i];
            for _ in 0..selected_change[i] {
                coins_desc.push(DENOMS[i]);
            }
        }
        for &coin in &self.escrow {
            let i = match coin {
                1 => 0,
                3 => 1,
                4 => 2,
                10 => 3,
                _ => unreachable!(),
            };
            proposed[i] += 1;
        }
        for (i, &count) in proposed.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        self.tubes = proposed;
        self.products[index].stock -= 1;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        self.change = self.change.saturating_add(u64::from(amount));
        let receipt = Receipt { id: id.clone(), price, change: coins_desc.clone() };
        self.log(EventKind::Vended { id, price, change: coins_desc });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let credit = self.credit;
        self.escrow.clear();
        self.credit = 0;
        self.selected = None;
        self.refunded = self.refunded.saturating_add(u64::from(credit));
        self.log(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.cmp(&b.id));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = self.product_index(id).ok_or_else(|| Error::UnknownProduct { id: id.to_string() })?;
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_string() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged { id: id.to_string(), old, new: price });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let index = self.product_index(id).ok_or_else(|| Error::UnknownProduct { id: id.to_string() })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = u32::from(self.products[index].stock) + u32::from(count);
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_string() });
        }
        self.products[index].stock = new_stock as u16;
        self.log(EventKind::Restocked { id: id.to_string(), added: count, stock: self.products[index].stock });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: value_of(&self.tubes).saturating_add(u64::from(self.credit)),
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let before_value = value_of(&self.tubes);
        let after_value = value_of(&actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift { before: before_value, after: after_value });
        }
        let before = self.tubes;
        self.tubes = actual;
        let delta = i128::from(after_value) - i128::from(before_value);
        let sum = i128::from(self.adjustment) + delta;
        self.adjustment = sum.clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        self.log(EventKind::Reconciled { before, after: actual });
        Ok(())
    }
}

fn tube_value(tubes: [u16; 4]) -> u64 {
    tubes
        .iter()
        .zip(COINS.iter())
        .map(|(&count, &coin)| u64::from(count) * u64::from(coin))
        .sum()
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    if amount == 0 {
        return Some([0; 4]);
    }

    let mut greedy = [0u16; 4];
    let mut remaining = u32::from(amount);
    for i in (0..4).rev() {
        let take = u32::from(tubes[i]).min(remaining / u32::from(COINS[i]));
        greedy[i] = take as u16;
        remaining -= take * u32::from(COINS[i]);
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let target = usize::from(amount);
    let mut best: Vec<Option<[u16; 4]>> = vec![None; target + 1];
    best[0] = Some([0; 4]);
    for value in 1..=target {
        let mut winner: Option<[u16; 4]> = None;
        for i in 0..4 {
            let coin = usize::from(COINS[i]);
            if value < coin || tubes[i] == 0 {
                continue;
            }
            let Some(mut candidate) = best[value - coin] else {
                continue;
            };
            if candidate[i] >= tubes[i] {
                continue;
            }
            candidate[i] += 1;
            if winner
                .as_ref()
                .map_or(true, |old| better_change(&candidate, old))
            {
                winner = Some(candidate);
            }
        }
        best[value] = winner;
    }
    best[target]
}

fn better_change(candidate: &[u16; 4], old: &[u16; 4]) -> bool {
    let candidate_total: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
    let old_total: u32 = old.iter().map(|&n| u32::from(n)).sum();
    if candidate_total != old_total {
        return candidate_total < old_total;
    }
    for i in (0..4).rev() {
        if candidate[i] != old[i] {
            return candidate[i] > old[i];
        }
    }
    false
}

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;

    for n10 in 0..=tubes[3].min(amount / 10) {
        let rem10 = amount - n10 * 10;
        for n4 in 0..=tubes[2].min(rem10 / 4) {
            let rem4 = rem10 - n4 * 4;
            for n3 in 0..=tubes[1].min(rem4 / 3) {
                let rem3 = rem4 - n3 * 3;
                if rem3 > tubes[0] {
                    continue;
                }
                let candidate = [rem3, n3, n4, n10];
                let count = candidate.iter().map(|&n| u32::from(n)).sum();
                let better = match best {
                    None => true,
                    Some(old) => {
                        count < best_count
                            || (count == best_count
                                && (candidate[3], candidate[2], candidate[1], candidate[0])
                                    > (old[3], old[2], old[1], old[0]))
                    }
                };
                if better {
                    best = Some(candidate);
                    best_count = count;
                }
            }
        }
    }
    best
}

fn choose_change(tubes: &[u16; 4], amount: u16) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / COINS[i]);
        greedy[i] = take;
        remaining -= take * COINS[i];
    }
    if remaining == 0 {
        Some(greedy)
    } else {
        exact_best_change(tubes, amount)
    }
}

fn change_vector(counts: &[u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            result.push(COINS[i]);
        }
    }
    result
}

    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashMap::new();
        for (i, product) in products.iter().enumerate() {
            if !valid_id(&product.id) {

                return Err(Error::StockLimit { id: product.id.clone() });

            if count > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }

        Ok(Self {
            products,
            index: seen,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),

        }

            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        record(&mut self.events, EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let idx = self.index[&id];
        let product = &self.products[idx];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid { price: product.price, credit: self.credit });
        }

        let price = product.price;
        let amount = self.credit - price;
        let selected_counts = choose_change(&self.tubes, amount)
            .ok_or(Error::NoChange { amount })?;
        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= selected_counts[i];
        }
        for &coin in &self.escrow {
            proposed[coin_index(coin).expect("escrow contains only accepted coins")] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }

        let change = change_vector(&selected_counts);
        let product = &mut self.products[idx];
        product.stock -= 1;
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.sales = self.sales.saturating_add(u64::from(price));
        let change_value = total_coin_value(&change);
        self.change = self.change.saturating_add(change_value);
        record(

            EventKind::Vended { id: id.clone(), price, change: change.clone() },

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let refunded_value = u64::from(self.credit);
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.refunded = self.refunded.saturating_add(refunded_value);
        record(&mut self.events, EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));

    }

        }

                    expected: expected[i],
                    found: self.tubes[i],

            if actual[i] > 100 {

        let after_value = value(&actual);

        let before = self.tubes;
        let delta = i128::from(after_value) - i128::from(before_value);
        self.adjustment = (i128::from(self.adjustment) + delta)
            .clamp(i128::from(i64::MIN), i128::from(i64::MAX)) as i64;
        self.tubes = actual;
        record(&mut self.events, EventKind::Reconciled { before, after: actual });
        Ok(())
    }

fn tube_value(tubes: [u16; 4]) -> u64 {
    tubes.iter().zip(COINS).map(|(&n, c)| u64::from(n) * u64::from(c)).sum()
}

fn choose_change_backtracking(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    if amount == 0 {
        return Some([0; 4]);
    }

    let mut greedy = [0u16; 4];
    let mut remaining = u32::from(amount);
    for i in (0..4).rev() {
        let coin = u32::from(COINS[i]);
        let take = u32::from(tubes[i]).min(remaining / coin);
        greedy[i] = take as u16;
        remaining -= take * coin;
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut current = [0u16; 4];
    search_change(
        3,
        u32::from(amount),
        0,
        &mut current,
        tubes,
        &mut best,
    );
    best
}

fn search_change(
    denomination: isize,
    remaining: u32,
    used: u32,
    current: &mut [u16; 4],
    tubes: [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if remaining == 0 {
        if best
            .as_ref()
            .map_or(true, |old| better_change(current, old))
        {
            *best = Some(*current);
        }
        return;
    }
    if denomination < 0 {
        return;
    }
    if let Some(old) = best.as_ref() {
        let old_count: u32 = old.iter().map(|&n| u32::from(n)).sum();
        if used >= old_count {
            return;
        }
    }

    let i = denomination as usize;
    let coin = u32::from(COINS[i]);
    let max_take = u32::from(tubes[i]).min(remaining / coin);
    for take in (0..=max_take).rev() {
        current[i] = take as u16;
        search_change(
            denomination - 1,
            remaining - take * coin,
            used + take,
            current,
            tubes,
            best,
        );
    }
    current[i] = 0;
}

fn better_change(candidate: &[u16; 4], old: &[u16; 4]) -> bool {
    let a: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
    let b: u32 = old.iter().map(|&n| u32::from(n)).sum();
    if a != b {
        return a < b;
    }
    for i in (0..4).rev() {
        if candidate[i] != old[i] {
            return candidate[i] > old[i];
        }
