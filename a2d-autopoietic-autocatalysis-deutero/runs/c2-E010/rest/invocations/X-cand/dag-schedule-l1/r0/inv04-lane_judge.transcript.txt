pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegrees = vec![0usize; n];
    for (id, task) in tasks.iter().enumerate() {
        indegrees[id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(id);
        }
    }

    let mut queue = std::collections::BTreeSet::new();
    for id in 0..n {
        if indegrees[id] == 0 {
            queue.insert(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut left = indegrees;
    while let Some(id) = queue.pop_first() {
        order.push(id);
        for &successor in &successors[id] {
            left[successor] -= 1;
            if left[successor] == 0 {
                queue.insert(successor);
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut start_times = vec![0u64; n];
    let mut finish_times = vec![0u64; n];
    let mut makespan = 0u64;

    for &id in &order {
        let mut start = 0u64;
        for &pred in &tasks[id].deps {
            start = start.max(finish_times[pred]);
        }
        let finish = match start.checked_add(tasks[id].duration) {

            None => return Err(Error::Overflow { task: id }),

            start: start_times[id],
            finish: finish_times[id],

    Ok(Plan {
        order,
        times,
        makespan,
    })

            children[dep].push(id);
        }

            was_removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !was_removed[id]).collect(),
        });

            starts[child] = starts[child].max(finish);
        }
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })

                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    let mut available = BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut times = vec![Timing { start: 0, finish: 0 }; n];
    let mut makespan = 0u64;
    for &id in &order {
        let start = tasks[id]
            .deps
            .iter()
            .map(|&dep| times[dep].finish)
            .max()
            .unwrap_or(0);
        let finish = start
            .checked_add(tasks[id].duration)
            .ok_or(Error::Overflow { task: id })?;
        times[id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })
}

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        // A vector keeps validation deterministic and preserves left-to-right checks.
        let mut seen = Vec::with_capacity(task.deps.len());
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen.contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen.push(value);
        }
    }

    // Store outgoing edges and indegrees.
    let mut outgoing: Vec<Vec<usize>> = (0..n).map(|_| Vec::new()).collect();
    let mut remaining_predecessors = vec![0usize; n];
    for (id, task) in tasks.iter().enumerate() {
        remaining_predecessors[id] = task.deps.len();
        for &predecessor in &task.deps {
            outgoing[predecessor].push(id);
        }
    }

    let mut ready = BinaryHeap::<Reverse<usize>>::new();
    for id in 0..n {
        if remaining_predecessors[id] == 0 {
            ready.push(Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(Reverse(id)) = ready.pop() {
        order.push(id);
        for &child in &outgoing[id] {
            remaining_predecessors[child] -= 1;
            if remaining_predecessors[child] == 0 {
                ready.push(Reverse(child));
            }
        }
    }

    if order.len() < n {
        let mut was_removed = vec![false; n];
        for &id in &order {
            was_removed[id] = true;
        }
        let mut blocked = Vec::new();
        for id in 0..n {
            if !was_removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut times = Vec::with_capacity(n);
    for _ in 0..n {
        times.push(Timing { start: 0, finish: 0 });
    }

    let mut makespan = 0u64;
    for &id in &order {
        let mut start = 0u64;
        for &predecessor in &tasks[id].deps {
            start = start.max(times[predecessor].finish);
        }
        let finish = match start.checked_add(tasks[id].duration) {
            Some(value) => value,
            None => return Err(Error::Overflow { task: id }),
        };
        times[id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })
}

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    let mut children = vec![Vec::new(); n];
    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    for (child, task) in tasks.iter().enumerate() {
        for &parent in &task.deps {
            children[parent].push(child);
        }
    }

    // A sorted vector serves as the ready set; always remove its smallest ID.
    let mut ready = Vec::new();
    for id in 0..n {
        if indegree[id] == 0 {
            ready.push(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while !ready.is_empty() {
        let id = ready.remove(0);
        order.push(id);

        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let insertion = ready.binary_search(&child).unwrap_or_else(|i| i);
                ready.insert(insertion, child);
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut times = vec![Timing { start: 0, finish: 0 }; n];
    let mut makespan = 0u64;

    for &id in &order {
        let mut start = 0u64;
        for &parent in &tasks[id].deps {
            start = start.max(times[parent].finish);
        }
        let finish = start
            .checked_add(tasks[id].duration)
            .ok_or(Error::Overflow { task: id })?;
        times[id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    Ok(Plan { order, times, makespan })
}
