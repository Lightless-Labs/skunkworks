- Validate every dependency before checking for cycles or doing time arithmetic.
- Report the first invalid entry in task and list order; out-of-range precedes self-reference, which precedes duplicate.
- Use a min-heap Kahn traversal, so the smallest currently available task is always removed next.
- Include every unremoved task in a cycle error, even descendants blocked by the cycle.
- Allow exact `u64::MAX`; only checked addition beyond it is overflow.
- Empty input has an empty plan and makespan zero.

### Solution A

```rust
use std::cmp::Reverse;
use std::collections::BinaryHeap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Plan {
    pub order: Vec<usize>,
    pub times: Vec<Timing>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency { task: usize, position: usize, value: usize },
    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },
    Cycle { blocked: Vec<usize> },
    Overflow { task: usize },
}

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let n = tasks.len();

    // Validate all dependency entries before any graph processing.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::with_capacity(task.deps.len());
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    let mut available = BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut times = vec![Timing { start: 0, finish: 0 }; n];
    let mut makespan = 0u64;
    for &id in &order {
        let start = tasks[id]
            .deps
            .iter()
            .map(|&dep| times[dep].finish)
            .max()
            .unwrap_or(0);
        let finish = start
            .checked_add(tasks[id].duration)
            .ok_or(Error::Overflow { task: id })?;
        times[id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })
}
```

### Solution B

```rust
use std::cmp::Reverse;
use std::collections::BinaryHeap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Timing {
    pub start: u64,
    pub finish: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Plan {
    pub order: Vec<usize>,
    pub times: Vec<Timing>,
    pub makespan: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency { task: usize, position: usize, value: usize },
    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },
    Cycle { blocked: Vec<usize> },
    Overflow { task: usize },
}

pub fn earliest(tasks: &[Task]) -> Result<Plan, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        // A vector keeps validation deterministic and preserves left-to-right checks.
        let mut seen = Vec::with_capacity(task.deps.len());
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen.contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen.push(value);
        }
    }

    // Store outgoing edges and indegrees.
    let mut outgoing: Vec<Vec<usize>> = (0..n).map(|_| Vec::new()).collect();
    let mut remaining_predecessors = vec![0usize; n];
    for (id, task) in tasks.iter().enumerate() {
        remaining_predecessors[id] = task.deps.len();
        for &predecessor in &task.deps {
            outgoing[predecessor].push(id);
        }
    }

    let mut ready = BinaryHeap::<Reverse<usize>>::new();
    for id in 0..n {
        if remaining_predecessors[id] == 0 {
            ready.push(Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(Reverse(id)) = ready.pop() {
        order.push(id);
        for &child in &outgoing[id] {
            remaining_predecessors[child] -= 1;
            if remaining_predecessors[child] == 0 {
                ready.push(Reverse(child));
            }
        }
    }

    if order.len() < n {
        let mut was_removed = vec![false; n];
        for &id in &order {
            was_removed[id] = true;
        }
        let mut blocked = Vec::new();
        for id in 0..n {
            if !was_removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut times = Vec::with_capacity(n);
    for _ in 0..n {
        times.push(Timing { start: 0, finish: 0 });
    }

    let mut makespan = 0u64;
    for &id in &order {
        let mut start = 0u64;
        for &predecessor in &tasks[id].deps {
            start = start.max(times[predecessor].finish);
        }
        let finish = match start.checked_add(tasks[id].duration) {
            Some(value) => value,
            None => return Err(Error::Overflow { task: id }),
        };
        times[id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    Ok(Plan {
        order,
        times,
        makespan,
    })
}
```
