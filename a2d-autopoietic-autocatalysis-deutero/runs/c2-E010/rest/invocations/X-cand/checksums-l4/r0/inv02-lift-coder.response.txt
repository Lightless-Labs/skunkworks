- Span validation follows the supplied order: reversed, bounds, then overlap at each span.
- Empty spans are valid and advance the ordering boundary, including repeated positions.
- Adler and Fletcher use exact modulo reductions; Fletcher-32 carries an odd byte across span boundaries and pads only at the end.
- Snapshot and manifest tokens require exact byte lengths, separators, and lowercase hexadecimal; validation precedes numeric comparisons.
- CRC combination uses the specified zero-byte transformation for all 64 bits of the right length.
- Both files use the same contract interpretation; Solution B is independently structured around shared streaming accumulators.

impl Acc {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff, a: 1, b: 0,
            f16a: 0, f16b: 0, f32a: 0, f32b: 0,
            pending: None, c16: match variant { Crc16::Xmodem => 0, _ => 0xffff },
            variant, len: 0,
        }
    }

    fn byte(&mut self, x: u8) {
        self.crc = crc_step(self.crc, x);
        self.a = (self.a + x as u32) % 65521;
        self.b = (self.b + self.a) % 65521;

        self.f16a = (self.f16a + x as u32) % 255;
        self.f16b = (self.f16b + self.f16a) % 255;

        if let Some(high) = self.pending.take() {
            let w = ((high as u32) << 8) | x as u32;
            self.f32a = (self.f32a + w) % 65535;
            self.f32b = (self.f32b + self.f32a) % 65535;
        } else {
            self.pending = Some(x);
        }

        match self.variant {
            Crc16::Xmodem | Crc16::False => {
                self.c16 ^= (x as u16) << 8;
                for _ in 0..8 {
                    let top = self.c16 & 0x8000 != 0;
                    self.c16 = self.c16.wrapping_shl(1);
                    if top { self.c16 ^= 0x1021; }
                }

                    self.c16 >>= 1;
                    if low { self.c16 ^= 0x8408; }
                }
            }
        }
        self.len += 1;
    }

    fn finish_f32(&self) -> u32 {

            (a, b)
        } else {
            (self.f32a, self.f32b)
        };
        (b << 16) | a
    }

        Hashes { crc32: self.crc ^ 0xffff_ffff, adler32: (self.b << 16) | self.a }
    }

    }

            return Err(SnapshotError::Shape);
        }

        let mut adler = 0u32;

        }
        for i in 11..19 {
            let d = hex_digit(bytes[i]).ok_or(SnapshotError::Hex { offset: i })?;
            adler = (adler << 4) | d;
        }
        Self::from_seed(crc, adler)
    }

fn hex_digit(b: u8) -> Option<u32> {
    match b {
        b'0'..=b'9' => Some((b - b'0') as u32),
        b'a'..=b'f' => Some((b - b'a' + 10) as u32),
        _ => None,
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 { return left; }
    fn zero_byte(mut r: u32) -> u32 {
        for _ in 0..8 {
            r = (r >> 1) ^ if r & 1 != 0 { 0xedb88320 } else { 0 };
        }
        r
    }
    let mut basis = [0u32; 32];
    for i in 0..32 { basis[i] = zero_byte(1u32 << i); }
    let mut value = left;
    let mut n = right_len;
    while n != 0 {
        if n & 1 != 0 {
            let mut transformed = 0;
            for i in 0..32 { if value & (1u32 << i) != 0 { transformed ^= basis[i]; } }
            value = transformed;
        }
        let mut squared = [0u32; 32];
        for i in 0..32 {
            let mut transformed = 0;
            for j in 0..32 { if basis[i] & (1u32 << j) != 0 { transformed ^= basis[j]; } }
            squared[i] = transformed;
        }
        basis = squared;
        n >>= 1;
    }
    value ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total.checked_add(part.len).ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 { return Err(CombineError::EmptyCrc { part: i }); }
    }
    let mut crc = 0u32;
    for part in parts { crc = crc32_combine(crc, part.crc32, part.len); }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut acc = Acc::new(Crc16::Xmodem);
    for span in spans { for &x in &data[span.start..span.end] { acc.byte(x); } }
    let h = acc.hashes();
    let mut out = String::with_capacity(48);
    write!(&mut out, "len={:016x};crc={:08x};adler={:08x}", acc.len, h.crc32, h.adler32).unwrap();
    Ok(out)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48 || &bytes[0..4] != b"len=" || &bytes[20..25] != b";crc=" || &bytes[33..40] != b";adler=" {
        return Err(VerifyError::Shape);
    }
    let ranges = [(4usize, 20usize), (25, 33), (40, 48)];
    let mut nums = [0u64; 3];
    for (field, (start, end)) in ranges.iter().enumerate() {
        for i in *start..*end {
            let d = hex_digit(bytes[i]).ok_or(VerifyError::Hex { offset: i })? as u64;
            nums[field] = (nums[field] << 4) | d;
        }
    }
    let mut acc = Acc::new(Crc16::Xmodem);
    for span in spans { for &x in &data[span.start..span.end] { acc.byte(x); } }
    if nums[0] != acc.len { return Err(VerifyError::Length); }
    let h = acc.hashes();
    if nums[1] as u32 != h.crc32 {
        return Err(VerifyError::Mismatch { field: 0, expected: nums[1] as u32, actual: h.crc32 });
    }
    if nums[2] as u32 != h.adler32 {
        return Err(VerifyError::Mismatch { field: 1, expected: nums[2] as u32, actual: h.adler32 });
    }
    Ok(())
}

    if right_len==0 { return left; }
    fn z(mut x:u32)->u32 { for _ in 0..8 { let bit=x&1; x>>=1; if bit!=0 {x^=0xedb88320;} } x }

        if n&1!=0 { let mut q=0; for i in 0..32 {if v&(1u32<<i)!=0 {q^=cols[i];}} v=q; }
        let old=cols; let mut next=[0u32;32];
        for i in 0..32 { let mut q=0; for j in 0..32 {if old[i]&(1u32<<j)!=0 {q^=old[j];}} next[i]=q; }
        cols=next; n>>=1;
    }
    v^right
}
pub fn combine_parts(parts:&[Part])->Result<Part,CombineError> {
    let mut len=0u64;
    for (i,p) in parts.iter().enumerate() { len=len.checked_add(p.len).ok_or(CombineError::LengthOverflow{part:i})?; }
    for (i,p) in parts.iter().enumerate() { if p.len==0 && p.crc32!=0 {return Err(CombineError::EmptyCrc{part:i});} }
    let mut c=0;
    for p in parts { c=crc32_combine(c,p.crc32,p.len); }
    Ok(Part{crc32:c,len})
}
pub fn manifest(data:&[u8],spans:&[Span])->Result<String,SpanError> {
    let s=run(data,spans,Crc16::Xmodem)?;
    let h=s.base();
    let mut out=String::with_capacity(48);
    write!(&mut out,"len={:016x};crc={:08x};adler={:08x}",s.size,h.crc32,h.adler32).unwrap();
    Ok(out)
}
pub fn verify(data:&[u8],spans:&[Span],token:&str)->Result<(),VerifyError> {
    validate(data.len(),spans).map_err(VerifyError::Span)?;
    let x=token.as_bytes();
    if x.len()!=48 || &x[0..4]!=b"len=" || &x[20..25]!=b";crc=" || &x[33..40]!=b";adler=" {return Err(VerifyError::Shape);}
    let mut v=[0u64;3];
    for &(lo,hi,k) in &[(4usize,20usize,0usize),(25,33,1),(40,48,2)] {
        for i in lo..hi {v[k]=(v[k]<<4)|digit(x[i]).ok_or(VerifyError::Hex{offset:i})? as u64;}
    }
    let s=run(data,spans,Crc16::Xmodem).map_err(VerifyError::Span)?;
    if v[0]!=s.size {return Err(VerifyError::Length);}
    let h=s.base();
    if v[1] as u32!=h.crc32 {return Err(VerifyError::Mismatch{field:0,expected:v[1] as u32,actual:h.crc32});}
    if v[2] as u32!=h.adler32 {return Err(VerifyError::Mismatch{field:1,expected:v[2] as u32,actual:h.adler32});}
    Ok(())
}
