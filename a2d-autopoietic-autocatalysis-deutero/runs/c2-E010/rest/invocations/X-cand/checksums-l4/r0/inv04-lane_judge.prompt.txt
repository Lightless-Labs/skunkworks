fn update_crc(crc: &mut u32, data: &[u8]) {
    for &byte in data {
        *crc ^= byte as u32;
        for _ in 0..8 {
            let low = *crc & 1;
            *crc >>= 1;
            if low != 0 {

        }
    }

fn update_adler(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {

    }

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.end,
            });
        }
        if i > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: i });
        }
        previous_end = span.end;
    }
    Ok(())
}

        }
    }
}

fn crc16_update(reg: &mut u16, data: &[u8], variant: Crc16) {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                *reg ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = *reg & 0x8000 != 0;
                    *reg <<= 1;
                    if top {
                        *reg ^= 0x1021;
                    }
                }
            }
        }
        Crc16::X25 => {
            for &byte in data {
                *reg ^= byte as u16;
                for _ in 0..8 {
                    let low = *reg & 1 != 0;
                    *reg >>= 1;
                    if low {
                        *reg ^= 0x8408;
                    }
                }
            }
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut reg = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    crc16_update(&mut reg, data, variant);
    if variant == Crc16::X25 {
        reg ^ 0xffff
    } else {
        reg
    }
}

        let word = (byte as u32) << 8;
        f32a = (f32a + word) % 65535;
        f32b = (f32b + f32a) % 65535;
    }
    if variant == Crc16::X25 {
        c16 ^= 0xffff;
    }
    Ok(Multi {
        base: stream.checksum(),
        fletcher16: ((f16b as u16) << 8) | f16a as u16,
        fletcher32: (f32b << 16) | f32a,
        crc16: c16,
    })
}

fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut i = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[i];
        }
        vector >>= 1;
        i += 1;
    }
    sum
}

fn matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 {
        square[i] = matrix_times(matrix, matrix[i]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }
    let mut even = [0u32; 32];
    matrix_square(&mut even, &odd);
    matrix_square(&mut odd, &even);
    let mut len = right_len;
    let mut crc = left;
    loop {
        matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
        matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }
    let mut crc = 0;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let (hashes, len) = compute_selected(data, spans);
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, hashes.crc32, hashes.adler32
    ))
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex(bytes: &[u8]) -> Option<u32> {
    let mut value = 0u32;
    for &byte in bytes {
        value = (value << 4) | hex_value(byte)?;
    }
    Some(value)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }
    for range in [4..20, 25..33, 40..48] {
        for i in range {
            if hex_value(bytes[i]).is_none() {
                return Err(VerifyError::Hex { offset: i });
            }
        }
    }
    let len = u64::from_str_radix(std::str::from_utf8(&bytes[4..20]).unwrap(), 16).unwrap();
    let expected_crc = parse_hex(&bytes[25..33]).unwrap();
    let expected_adler = parse_hex(&bytes[40..48]).unwrap();
    let (actual, actual_len) = compute_selected(data, spans);
    if len != actual_len {
        return Err(VerifyError::Length);
    }
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

struct Acc {
    crc: u32,
    a: u32,
    b: u32,
    f16a: u32,
    f16b: u32,
    f32a: u32,
    f32b: u32,
    pending: Option<u8>,
    c16: u16,
    variant: Crc16,
    len: u64,
}

impl Acc {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff, a: 1, b: 0,
            f16a: 0, f16b: 0, f32a: 0, f32b: 0,
            pending: None, c16: match variant { Crc16::Xmodem => 0, _ => 0xffff },
            variant, len: 0,
        }
    }

    fn byte(&mut self, x: u8) {
        self.crc = crc_step(self.crc, x);
        self.a = (self.a + x as u32) % 65521;
        self.b = (self.b + self.a) % 65521;

        self.f16a = (self.f16a + x as u32) % 255;
        self.f16b = (self.f16b + self.f16a) % 255;

        if let Some(high) = self.pending.take() {
            let w = ((high as u32) << 8) | x as u32;
            self.f32a = (self.f32a + w) % 65535;
            self.f32b = (self.f32b + self.f32a) % 65535;
        } else {
            self.pending = Some(x);
        }

        match self.variant {
            Crc16::Xmodem | Crc16::False => {
                self.c16 ^= (x as u16) << 8;
                for _ in 0..8 {
                    let top = self.c16 & 0x8000 != 0;
                    self.c16 = self.c16.wrapping_shl(1);
                    if top { self.c16 ^= 0x1021; }
                }
            }
            Crc16::X25 => {
                self.c16 ^= x as u16;
                for _ in 0..8 {
                    let low = self.c16 & 1 != 0;
                    self.c16 >>= 1;
                    if low { self.c16 ^= 0x8408; }
                }
            }
        }
        self.len += 1;
    }

    fn finish_f32(&self) -> u32 {
        let (a, b) = if let Some(high) = self.pending {
            let a = (self.f32a + ((high as u32) << 8)) % 65535;
            let b = (self.f32b + a) % 65535;
            (a, b)
        } else {
            (self.f32a, self.f32b)
        };
        (b << 16) | a
    }

    fn hashes(&self) -> Hashes {
        Hashes { crc32: self.crc ^ 0xffff_ffff, adler32: (self.b << 16) | self.a }
    }
}

impl Stream {
    pub fn new() -> Self { Self { crc_raw: 0xffff_ffff, a: 1, b: 0 } }

    pub fn checksum(&self) -> Hashes {
        Hashes { crc32: self.crc_raw ^ 0xffff_ffff, adler32: (self.b << 16) | self.a }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 { return Err(SnapshotError::Adler { component: 0 }); }
        if b >= 65521 { return Err(SnapshotError::Adler { component: 1 }); }
        Ok(Self { crc_raw: crc32 ^ 0xffff_ffff, a, b })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &x in data {
            self.crc_raw = crc_step(self.crc_raw, x);
            self.a = (self.a + x as u32) % 65521;
            self.b = (self.b + self.a) % 65521;
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans { self.update(&data[span.start..span.end]); }
        Ok(())
    }

    pub fn reset(&mut self) { *self = Self::new(); }

    pub fn snapshot(&self) -> String {
        let h = self.checksum();
        format!("c{:08x}:a{:08x}", h.crc32, h.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let mut crc = 0u32;
        let mut adler = 0u32;
        for i in 1..9 {
            let d = hex_digit(bytes[i]).ok_or(SnapshotError::Hex { offset: i })?;
            crc = (crc << 4) | d;
        }
        for i in 11..19 {
            let d = hex_digit(bytes[i]).ok_or(SnapshotError::Hex { offset: i })?;
            adler = (adler << 4) | d;
        }
        Self::from_seed(crc, adler)
    }
}

fn hex_digit(b: u8) -> Option<u32> {

        b'0'..=b'9' => Some((b - b'0') as u32),
        b'a'..=b'f' => Some((b - b'a' + 10) as u32),
        _ => None,
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 { return left; }
    fn zero_byte(mut r: u32) -> u32 {
        for _ in 0..8 {
            r = (r >> 1) ^ if r & 1 != 0 { 0xedb88320 } else { 0 };
        }
        r
    }
    let mut basis = [0u32; 32];
    for i in 0..32 { basis[i] = zero_byte(1u32 << i); }
    let mut value = left;
    let mut n = right_len;
    while n != 0 {
        if n & 1 != 0 {
            let mut transformed = 0;
            for i in 0..32 { if value & (1u32 << i) != 0 { transformed ^= basis[i]; } }
            value = transformed;
        }
        let mut squared = [0u32; 32];
        for i in 0..32 {
            let mut transformed = 0;
            for j in 0..32 { if basis[i] & (1u32 << j) != 0 { transformed ^= basis[j]; } }
            squared[i] = transformed;
        }
        basis = squared;
        n >>= 1;
    }
    value ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total.checked_add(part.len).ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 { return Err(CombineError::EmptyCrc { part: i }); }
    }
    let mut crc = 0u32;
    for part in parts { crc = crc32_combine(crc, part.crc32, part.len); }
    Ok(Part { crc32: crc, len: total })
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48 || &bytes[0..4] != b"len=" || &bytes[20..25] != b";crc=" || &bytes[33..40] != b";adler=" {
        return Err(VerifyError::Shape);
    }
    let ranges = [(4usize, 20usize), (25, 33), (40, 48)];
    let mut nums = [0u64; 3];
    for (field, (start, end)) in ranges.iter().enumerate() {
        for i in *start..*end {
            let d = hex_digit(bytes[i]).ok_or(VerifyError::Hex { offset: i })? as u64;
            nums[field] = (nums[field] << 4) | d;
        }
    }
    let mut acc = Acc::new(Crc16::Xmodem);
    for span in spans { for &x in &data[span.start..span.end] { acc.byte(x); } }
    if nums[0] != acc.len { return Err(VerifyError::Length); }
    let h = acc.hashes();
    if nums[1] as u32 != h.crc32 {
        return Err(VerifyError::Mismatch { field: 0, expected: nums[1] as u32, actual: h.crc32 });
    }
    if nums[2] as u32 != h.adler32 {
        return Err(VerifyError::Mismatch { field: 1, expected: nums[2] as u32, actual: h.adler32 });
    }
    Ok(())
}

    crc: u32, aa: u32, ab: u32, fa: u32, fb: u32, wa: u32, wb: u32,

impl State {

            c16: if v == Crc16::Xmodem { 0 } else { !0 }, variant: v, size: 0 }

        self.crc = crc_byte(self.crc, x);
        self.aa = (self.aa + x as u32) % 65521;
        self.ab = (self.ab + self.aa) % 65521;
        self.fa = (self.fa + x as u32) % 255;
        self.fb = (self.fb + self.fa) % 255;
        if let Some(h) = self.high.take() {
            self.wa = (self.wa + ((h as u32) << 8 | x as u32)) % 65535;
            self.wb = (self.wb + self.wa) % 65535;
        } else { self.high = Some(x); }
        match self.variant {

                self.c16 ^= (x as u16) << 8;

                    self.c16 >>= 1;
                    if t { self.c16 ^= 0x8408; }
                }
            }
        }
        self.size += 1;
    }

            None => (self.wa, self.wb),

fn crc_step(mut state: u32, byte: u8) -> u32 {
    state ^= byte as u32;
    for _ in 0..8 {
        state = (state >> 1) ^ if state & 1 != 0 { 0xedb8_8320 } else { 0 };

    state ^ 0xffff_ffff

fn check_all(data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
    let mut boundary = None;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds { span: index, offset: span.start });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds { span: index, offset: span.end });
        }
        if let Some(previous_end) = boundary {
            if span.start < previous_end {
                return Err(SpanError::Overlap { span: index });
            }
        }
        boundary = Some(span.end);
    }
    Ok(())
}

struct F32 {
    a: u32,
    b: u32,
    high: Option<u8>,
}

impl F32 {
    fn new() -> Self { Self { a: 0, b: 0, high: None } }
    fn word(&mut self, word: u16) {
        self.a = (self.a + word as u32) % 65535;
        self.b = (self.b + self.a) % 65535;
    }
    fn byte(&mut self, byte: u8) {
        if let Some(high) = self.high.take() {
            self.word(u16::from_be_bytes([high, byte]));
        } else {
            self.high = Some(byte);
        }
    }
    fn finish(mut self) -> u32 {
        if let Some(high) = self.high.take() {
            self.word((high as u16) << 8);
        }
        (self.b << 16) | self.a
    }
}

fn crc16_raw(state: u16, data: &[u8], variant: Crc16) -> u16 {
    let mut crc = state;
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    crc = (crc << 1) ^ if crc & 0x8000 != 0 { 0x1021 } else { 0 };
                }
            }
            Crc16::X25 => {
                crc ^= byte as u16;
                for _ in 0..8 {
                    crc = (crc >> 1) ^ if crc & 1 != 0 { 0x8408 } else { 0 };
                }
            }
        }
    }
    crc
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let initial = match variant { Crc16::Xmodem => 0, _ => 0xffff };
    let raw = crc16_raw(initial, data, variant);
    match variant { Crc16::X25 => raw ^ 0xffff, _ => raw }
}

pub fn multi_checksums(data: &[u8], spans: &[Span], variant: Crc16) -> Result<Multi, SpanError> {
    check_all(data, spans)?;
    let mut base = Stream::new();
    let mut f16a = 0u32;
    let mut f16b = 0u32;
    let mut f32 = F32::new();
    let mut c16 = match variant { Crc16::Xmodem => 0, _ => 0xffff };

    for span in spans {
        let bytes = &data[span.start..span.end];
        base.update(bytes);
        for &byte in bytes {
            f16a = (f16a + byte as u32) % 255;
            f16b = (f16b + f16a) % 255;
            f32.byte(byte);
        }
        c16 = crc16_raw(c16, bytes, variant);
    }

    Ok(Multi {
        base: base.checksum(),
        fletcher16: ((f16b as u16) << 8) | f16a as u16,
        fletcher32: f32.finish(),
        crc16: if variant == Crc16::X25 { c16 ^ 0xffff } else { c16 },
    })
}

fn gf_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut result = 0;
    let mut bit = 0;
    while vector != 0 {
        if vector & 1 != 0 { result ^= matrix[bit]; }
        vector >>= 1;
        bit += 1;
    }
    result
}

fn gf_square(out: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 { out[i] = gf_times(matrix, matrix[i]); }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 { return left; }

    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }
    let mut even = [0u32; 32];
    gf_square(&mut even, &odd);
    gf_square(&mut odd, &even);

    let mut crc = left;
    let mut len = right_len;
    loop {
        gf_square(&mut even, &odd);
        if len & 1 != 0 { crc = gf_times(&even, crc); }
        len >>= 1;
        if len == 0 { break; }
        gf_square(&mut odd, &even);
        if len & 1 != 0 { crc = gf_times(&odd, crc); }
        len >>= 1;
        if len == 0 { break; }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        match total.checked_add(part.len) {
            Some(next) => total = next,
            None => return Err(CombineError::LengthOverflow { part: index }),
        }
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }
    let mut crc = 0;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn logical_len(spans: &[Span]) -> u64 {
    spans.iter().fold(0u64, |sum, span| sum + (span.end - span.start) as u64)
}

}

    check_all(data, spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut len_value = 0u64;
    let mut crc_value = 0u32;
    let mut adler_value = 0u32;
    for offset in 4..20 {
        len_value = (len_value << 4) | hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })? as u64;
    }
    for offset in 25..33 {
        crc_value = (crc_value << 4) | hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })? as u32;
    }
    for offset in 40..48 {
        adler_value = (adler_value << 4) | hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })? as u32;
    }

    let actual_len = logical_len(spans);
    if len_value != actual_len { return Err(VerifyError::Length); }
    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if crc_value != actual.crc32 {
        return Err(VerifyError::Mismatch { field: 0, expected: crc_value, actual: actual.crc32 });
    }
    if adler_value != actual.adler32 {
        return Err(VerifyError::Mismatch { field: 1, expected: adler_value, actual: actual.adler32 });
    }
    Ok(())
}

        format!("c{:08x}:a{:08x}", h.crc32, h.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        for &i in &[1usize, 2, 3, 4, 5, 6, 7, 8, 11, 12, 13, 14, 15, 16, 17, 18] {
            if hex_value(bytes[i]).is_none() {
                return Err(SnapshotError::Hex { offset: i });
            }
        }
        let crc = parse_hex(&bytes[1..9]);
        let adler = parse_hex(&bytes[11..19]);
        Self::from_seed(crc, adler)
    }

fn update_base(stream: &mut Stream, data: &[u8]) {
    for &byte in data {
        stream.crc_raw ^= byte as u32;
        for _ in 0..8 {

        stream.adler_a = (stream.adler_a + byte as u32) % 65521;
        stream.adler_b = (stream.adler_b + stream.adler_a) % 65521;

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {

    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds { span: i, offset: span.start });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds { span: i, offset: span.end });
        }
        if i > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: i });
        }
        previous_end = span.end;
    }
    Ok(())

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u64;
    let mut b = 0u64;
    let mut i = 0usize;
    while i + 1 < data.len() {
        let word = ((data[i] as u64) << 8) | data[i + 1] as u64;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
        i += 2;
    }
    if i < data.len() {
        let word = (data[i] as u64) << 8;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
    }
    ((b << 16) | a) as u32
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u64;
    let mut b = 0u64;
    let mut pending: Option<u8> = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                let word = ((high as u64) << 8) | byte as u64;
                a = (a + word) % 65535;
                b = (b + a) % 65535;
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        let word = (high as u64) << 8;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
    }

}

fn crc16_update(mut reg: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                reg ^= (byte as u16) << 8;
                for _ in 0..8 {
                    reg = if reg & 0x8000 != 0 {
                        (reg << 1) ^ 0x1021
                    } else {
                        reg << 1
                    };
                }
            }
            reg
        }
        Crc16::X25 => {
            for &byte in data {
                reg ^= byte as u16;
                for _ in 0..8 {
                    reg = if reg & 1 != 0 {
                        (reg >> 1) ^ 0x8408
                    } else {
                        reg >> 1
                    };
                }
            }
            reg ^ 0xffff
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let init = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,

    crc16_update(init, data, variant)

struct AllState {
    base: Stream,
    f16_a: u32,
    f16_b: u32,
    f32_a: u32,
    f32_b: u32,
    f32_pending: Option<u8>,
    crc16: u16,
    variant: Crc16,
}

impl AllState {
    fn new(variant: Crc16) -> Self {
        Self {
            base: Stream::new(),
            f16_a: 0,
            f16_b: 0,
            f32_a: 0,
            f32_b: 0,
            f32_pending: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            variant,
        }
    }

    fn update(&mut self, data: &[u8]) {
        self.base.update(data);
        self.fletcher_bytes(data);
        self.crc16 = crc16_update(self.crc16, data, self.variant);
    }

    fn fletcher_bytes(&mut self, data: &[u8]) {
        for &byte in data {
            self.f16_a = (self.f16_a + byte as u32) % 255;
            self.f16_b = (self.f16_b + self.f16_a) % 255;
            if let Some(high) = self.f32_pending.take() {
                let word = ((high as u32) << 8) | byte as u32;
                self.f32_a = (self.f32_a + word) % 65535;
                self.f32_b = (self.f32_b + self.f32_a) % 65535;
            } else {
                self.f32_pending = Some(byte);
            }
        }
    }

    fn finish(mut self) -> Multi {
        if let Some(high) = self.f32_pending.take() {
            let word = (high as u32) << 8;
            self.f32_a = (self.f32_a + word) % 65535;
            self.f32_b = (self.f32_b + self.f32_a) % 65535;
        }
        Multi {
            base: self.base.checksum(),
            fletcher16: ((self.f16_b << 8) | self.f16_a) as u16,
            fletcher32: (self.f32_b << 16) | self.f32_a,
            crc16: self.crc16,
        }
    }
}

pub fn multi_checksums(data: &[u8], spans: &[Span], variant: Crc16) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = AllState::new(variant);
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.finish())
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 {
        square[i] = gf2_matrix_times(matrix, matrix[i]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for slot in odd.iter_mut().skip(1) {
        *slot = row;
        row <<= 1;
    }
    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut len = right_len;
    let mut crc = left;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total.checked_add(part.len).ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }
    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex(bytes: &[u8]) -> u32 {
    bytes.iter().fold(0u32, |value, &byte| (value << 4) | hex_value(byte).unwrap())
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut len = 0u64;
    for span in spans {
        let selected_len = (span.end - span.start) as u64;
        len += selected_len;
        stream.update(&data[span.start..span.end]);
    }
    let h = stream.checksum();

}

}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for range in [4usize..20, 25..33, 40..48] {
        for offset in range {
            if hex_value(bytes[offset]).is_none() {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let expected_len = parse_hex_u64(&bytes[4..20]);
    let expected_crc = parse_hex(&bytes[25..33]);
    let expected_adler = parse_hex(&bytes[40..48]);

    let mut stream = Stream::new();
    let mut selected_len = 0u64;
    for span in spans {
        selected_len += (span.end - span.start) as u64;
        stream.update(&data[span.start..span.end]);
    }
    if expected_len != selected_len {
        return Err(VerifyError::Length);
    }
    let actual = stream.checksum();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 { return left; }
    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut bit = 1;
    for slot in odd.iter_mut().skip(1) {
        *slot = bit;
        bit <<= 1;
    }
    let mut even = [0u32; 32];
    matrix_square(&mut even, &odd);
    matrix_square(&mut odd, &even);
    let mut len = right_len;
    let mut crc = left;
    loop {
        matrix_square(&mut even, &odd);
        if len & 1 != 0 { crc = matrix_times(&even, crc); }
        len >>= 1;
        if len == 0 { break; }
        matrix_square(&mut odd, &even);
        if len & 1 != 0 { crc = matrix_times(&odd, crc); }
        len >>= 1;
        if len == 0 { break; }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut len = 0u64;
    for (i, part) in parts.iter().enumerate() {
        len = len.checked_add(part.len).ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }
    let mut crc = 0;
    for part in parts { crc = crc32_combine(crc, part.crc32, part.len); }
    Ok(Part { crc32: crc, len })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let (h, len) = selected_hashes(data, spans);
    Ok(format!("len={:016x};crc={:08x};adler={:08x}", len, h.crc32, h.adler32))
}

fn hex_value(x: u8) -> Option<u32> {
    match x {
        b'0'..=b'9' => Some((x - b'0') as u32),
        b'a'..=b'f' => Some((x - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex(bytes: &[u8]) -> u32 {
    bytes.iter().fold(0, |n, x| (n << 4) | hex_value(*x).unwrap())
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let b = token.as_bytes();
    if b.len() != 48 || &b[0..4] != b"len=" || &b[20..25] != b";crc=" || &b[33..40] != b";adler=" {
        return Err(VerifyError::Shape);
    }
    for range in [4..20, 25..33, 40..48] {
        for i in range {
            if hex_value(b[i]).is_none() { return Err(VerifyError::Hex { offset: i }); }
        }
    }
    let mut len = 0u64;
    for &x in &b[4..20] { len = (len << 4) | hex_value(x).unwrap() as u64; }
    let crc = parse_hex(&b[25..33]);
    let adler = parse_hex(&b[40..48]);
    let (actual, actual_len) = selected_hashes(data, spans);
    if len != actual_len { return Err(VerifyError::Length); }
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch { field: 0, expected: crc, actual: actual.crc32 });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch { field: 1, expected: adler, actual: actual.adler32 });
    }
    Ok(())
}
