- Span validation checks each span in order: reversed, bounds, then overlap with the previous span’s end.
- Empty spans are valid, still advance the ordering boundary, and contribute no bytes.
- Fletcher-32 carries an unmatched high byte across span boundaries and pads only once at the end.
- Snapshot and manifest tokens require exact byte lengths, separators, and lowercase hexadecimal; all digits are checked before value comparisons.
- Adler state uses canonical residues below 65521; Fletcher reductions use modulo 255 or 65535, with zero represented as zero.
- CRC combination uses GF(2) matrix exponentiation over all 64 bits of the right length; part lengths are fully validated before empty-part CRCs.

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u64;
    let mut b = 0u64;
    let mut i = 0usize;
    while i + 1 < data.len() {
        let word = ((data[i] as u64) << 8) | data[i + 1] as u64;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
        i += 2;
    }
    if i < data.len() {
        let word = (data[i] as u64) << 8;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
    }
    ((b << 16) | a) as u32
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut a = 0u64;
    let mut b = 0u64;
    let mut pending: Option<u8> = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                let word = ((high as u64) << 8) | byte as u64;
                a = (a + word) % 65535;
                b = (b + a) % 65535;
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        let word = (high as u64) << 8;
        a = (a + word) % 65535;
        b = (b + a) % 65535;
    }
    Ok(((b << 16) | a) as u32)
}

fn crc16_update(mut reg: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                reg ^= (byte as u16) << 8;
                for _ in 0..8 {
                    reg = if reg & 0x8000 != 0 {
                        (reg << 1) ^ 0x1021
                    } else {
                        reg << 1
                    };
                }
            }
            reg
        }
        Crc16::X25 => {
            for &byte in data {
                reg ^= byte as u16;
                for _ in 0..8 {
                    reg = if reg & 1 != 0 {
                        (reg >> 1) ^ 0x8408
                    } else {
                        reg >> 1
                    };
                }
            }
            reg ^ 0xffff
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let init = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    crc16_update(init, data, variant)
}

struct AllState {
    base: Stream,
    f16_a: u32,
    f16_b: u32,
    f32_a: u32,
    f32_b: u32,
    f32_pending: Option<u8>,
    crc16: u16,
    variant: Crc16,
}

impl AllState {
    fn new(variant: Crc16) -> Self {
        Self {
            base: Stream::new(),
            f16_a: 0,
            f16_b: 0,
            f32_a: 0,
            f32_b: 0,
            f32_pending: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            variant,
        }
    }

    fn update(&mut self, data: &[u8]) {
        self.base.update(data);
        self.fletcher_bytes(data);
        self.crc16 = crc16_update(self.crc16, data, self.variant);
    }

    fn fletcher_bytes(&mut self, data: &[u8]) {
        for &byte in data {
            self.f16_a = (self.f16_a + byte as u32) % 255;
            self.f16_b = (self.f16_b + self.f16_a) % 255;
            if let Some(high) = self.f32_pending.take() {
                let word = ((high as u32) << 8) | byte as u32;
                self.f32_a = (self.f32_a + word) % 65535;
                self.f32_b = (self.f32_b + self.f32_a) % 65535;
            } else {
                self.f32_pending = Some(byte);
            }
        }
    }

    fn finish(mut self) -> Multi {
        if let Some(high) = self.f32_pending.take() {
            let word = (high as u32) << 8;
            self.f32_a = (self.f32_a + word) % 65535;
            self.f32_b = (self.f32_b + self.f32_a) % 65535;
        }
        Multi {
            base: self.base.checksum(),
            fletcher16: ((self.f16_b << 8) | self.f16_a) as u16,
            fletcher32: (self.f32_b << 16) | self.f32_a,
            crc16: self.crc16,
        }
    }
}

pub fn multi_checksums(data: &[u8], spans: &[Span], variant: Crc16) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = AllState::new(variant);
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.finish())
}

fn gf2_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0u32;
    let mut index = 0usize;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn gf2_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 {
        square[i] = gf2_matrix_times(matrix, matrix[i]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    let mut odd = [0u32; 32];
    let mut even = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for slot in odd.iter_mut().skip(1) {
        *slot = row;
        row <<= 1;
    }
    gf2_matrix_square(&mut even, &odd);
    gf2_matrix_square(&mut odd, &even);

    let mut len = right_len;
    let mut crc = left;
    loop {
        gf2_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
        gf2_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = gf2_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total.checked_add(part.len).ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }
    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn hex_value(byte: u8) -> Option<u32> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as u32),
        b'a'..=b'f' => Some((byte - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse_hex(bytes: &[u8]) -> u32 {
    bytes.iter().fold(0u32, |value, &byte| (value << 4) | hex_value(byte).unwrap())
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut len = 0u64;
    for span in spans {
        let selected_len = (span.end - span.start) as u64;
        len += selected_len;
        stream.update(&data[span.start..span.end]);
    }
    let h = stream.checksum();
    Ok(format!("len={:016x};crc={:08x};adler={:08x}", len, h.crc32, h.adler32))
}

fn parse_hex_u64(bytes: &[u8]) -> u64 {
    bytes.iter().fold(0u64, |value, &byte| (value << 4) | hex_value(byte).unwrap() as u64)
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;

    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    for range in [4usize..20, 25..33, 40..48] {
        for offset in range {
            if hex_value(bytes[offset]).is_none() {
                return Err(VerifyError::Hex { offset });
            }
        }
    }

    let expected_len = parse_hex_u64(&bytes[4..20]);
    let expected_crc = parse_hex(&bytes[25..33]);
    let expected_adler = parse_hex(&bytes[40..48]);

    let mut stream = Stream::new();
    let mut selected_len = 0u64;
    for span in spans {
        selected_len += (span.end - span.start) as u64;
        stream.update(&data[span.start..span.end]);
    }
    if expected_len != selected_len {
        return Err(VerifyError::Length);
    }
    let actual = stream.checksum();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

    match x {
        b'0'..=b'9' => Some((x - b'0') as u32),
        b'a'..=b'f' => Some((x - b'a' + 10) as u32),
        _ => None,
    }
}

fn parse8(bytes: &[u8]) -> u32 {
    bytes.iter().fold(0, |v, &x| (v << 4) | digit(x).unwrap())
}

fn crc16_start(v: Crc16) -> u16 {
    match v {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    }
}

fn crc16_add(mut r: u16, data: &[u8], v: Crc16) -> u16 {
    match v {
        Crc16::Xmodem | Crc16::False => {
            for &x in data {
                r ^= (x as u16) << 8;
                for _ in 0..8 {
                    let top = r & 0x8000;
                    r = r.wrapping_shl(1) ^ (0x1021 & (0u16.wrapping_sub((top >> 15) as u16)));
                }
            }
            r
        }
        Crc16::X25 => {
            for &x in data {
                r ^= x as u16;
                for _ in 0..8 {
                    let low = r & 1;
                    r = (r >> 1) ^ (0x8408 & (0u16.wrapping_sub(low)));
                }
            }
            r ^ 0xffff
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    crc16_add(crc16_start(variant), data, variant)
}

struct Combined {
    stream: Stream,
    f16a: u32,
    f16b: u32,
    f32a: u32,
    f32b: u32,
    high: Option<u8>,
    c16: u16,
    variant: Crc16,
}

impl Combined {
    fn new(variant: Crc16) -> Self {
        Self {
            stream: Stream::new(),
            f16a: 0,
            f16b: 0,
            f32a: 0,
            f32b: 0,
            high: None,
            c16: crc16_start(variant),
            variant,
        }
    }

    fn update(&mut self, data: &[u8]) {
        self.stream.update(data);
        self.c16 = crc16_add(self.c16, data, self.variant);
        for &x in data {
            self.f16a = (self.f16a + x as u32) % 255;
            self.f16b = (self.f16b + self.f16a) % 255;
            if let Some(h) = self.high.take() {
                let w = ((h as u32) << 8) | x as u32;
                self.f32a = (self.f32a + w) % 65535;
                self.f32b = (self.f32b + self.f32a) % 65535;
            } else {
                self.high = Some(x);
            }
        }
    }

    fn result(mut self) -> Multi {
        if let Some(h) = self.high {
            self.f32a = (self.f32a + ((h as u32) << 8)) % 65535;
            self.f32b = (self.f32b + self.f32a) % 65535;
        }
        Multi {
            base: self.stream.checksum(),
            fletcher16: ((self.f16b << 8) | self.f16a) as u16,
            fletcher32: (self.f32b << 16) | self.f32a,
            crc16: if self.variant == Crc16::X25 { self.c16 } else { self.c16 },
        }
    }
}

pub fn multi_checksums(data: &[u8], spans: &[Span], variant: Crc16) -> Result<Multi, SpanError> {
    validate(data.len(), spans)?;
    let mut c = Combined::new(variant);
    for s in spans {
        c.update(&data[s.start..s.end]);
    }
    Ok(c.result())
}

fn matrix_times(m: &[u32; 32], mut v: u32) -> u32 {
    let mut out = 0;
    let mut i = 0;
    while v != 0 {
        if v & 1 != 0 {
            out ^= m[i];
        }
        v >>= 1;
        i += 1;
    }
    out
}

fn matrix_square(out: &mut [u32; 32], m: &[u32; 32]) {
    for i in 0..32 {
        out[i] = matrix_times(m, m[i]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    let mut op = [0u32; 32];
    op[0] = 0xedb8_8320;
    let mut bit = 1u32;
    for x in op.iter_mut().skip(1) {
        *x = bit;
        bit <<= 1;
    }
    let mut power = op;
    let mut result = left;
    let mut n = right_len;
    while n != 0 {
        if n & 1 != 0 {
            result = matrix_times(&power, result);
        }
        n >>= 1;
        if n != 0 {
            let mut squared = [0u32; 32];
            matrix_square(&mut squared, &power);
            power = squared;
        }
    }
    result ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, p) in parts.iter().enumerate() {
        total = total.checked_add(p.len).ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, p) in parts.iter().enumerate() {
        if p.len == 0 && p.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }
    let mut crc = 0;
    for p in parts {
        crc = crc32_combine(crc, p.crc32, p.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate(data.len(), spans)?;
    let mut c = Stream::new();
    let mut len = 0u64;
    for s in spans {
        len += (s.end - s.start) as u64;
        c.update(&data[s.start..s.end]);
    }
    let h = c.checksum();
    Ok(format!("len={:016x};crc={:08x};adler={:08x}", len, h.crc32, h.adler32))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate(data.len(), spans).map_err(VerifyError::Span)?;
    let t = token.as_bytes();
    if t.len() != 48 || &t[0..4] != b"len=" || &t[20..25] != b";crc=" || &t[33..40] != b";adler=" {
        return Err(VerifyError::Shape);
    }
    for r in [4usize..20, 25..33, 40..48] {
        for i in r {
            if digit(t[i]).is_none() {
                return Err(VerifyError::Hex { offset: i });
            }
        }
    }
    let len = t[4..20].iter().fold(0u64, |v, &x| (v << 4) | digit(x).unwrap() as u64);
    let crc = parse8(&t[25..33]);
    let adler = parse8(&t[40..48]);
    let mut c = Stream::new();
    let mut selected = 0u64;
    for s in spans {
        selected += (s.end - s.start) as u64;
        c.update(&data[s.start..s.end]);
    }
    if len != selected {
        return Err(VerifyError::Length);
    }
    let actual = c.checksum();
    if crc != actual.crc32 {
        return Err(VerifyError::Mismatch { field: 0, expected: crc, actual: actual.crc32 });
    }
    if adler != actual.adler32 {
        return Err(VerifyError::Mismatch { field: 1, expected: adler, actual: actual.adler32 });
    }
    Ok(())
}
