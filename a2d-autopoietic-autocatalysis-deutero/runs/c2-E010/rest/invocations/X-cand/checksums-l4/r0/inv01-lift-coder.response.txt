- Spans are validated in supplied order: reversed, bounds, then overlap for each span; endpoints equal to `data.len()` are valid.
- Empty spans contribute no bytes but still advance the ordering boundary; repeated empty spans at one position are valid.
- Stream seeds are finalized checksums; empty updates preserve them, while reset returns to the standard empty state.
- Snapshot and manifest formats require exact byte lengths, separators, lowercase hex, and leading zeroes; all digits are checked before numeric comparisons.
- Fletcher-32 carries an odd high byte across span boundaries and pads once at the end; arithmetic reduces each step to avoid overflow.
- CRC combination uses logarithmic GF(2) matrix exponentiation over all 64 length bits; zero-length parts are identities after all lengths validate.

#[derive(Debug, Clone)]
struct Fletcher32State {
    a: u32,
    b: u32,
    pending_high: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending_high: None,
        }
    }

    fn push(&mut self, byte: u8) {
        if let Some(high) = self.pending_high.take() {
            self.word(u16::from_be_bytes([high, byte]));
        } else {
            self.pending_high = Some(byte);
        }
    }

    fn word(&mut self, word: u16) {
        self.a = (self.a + word as u32) % 65535;
        self.b = (self.b + self.a) % 65535;
    }

    fn finish(mut self) -> u32 {
        if let Some(high) = self.pending_high.take() {
            self.word((high as u16) << 8);
        }
        (self.b << 16) | self.a
    }
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = Fletcher32State::new();
    for &byte in data {
        state.push(byte);
    }
    state.finish()
}

fn crc16_update(mut state: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                state ^= (byte as u16) << 8;
                for _ in 0..8 {
                    state = (state << 1) ^ if state & 0x8000 != 0 { 0x1021 } else { 0 };
                }
            }
            state
        }
        Crc16::X25 => {
            for &byte in data {
                state ^= byte as u16;
                for _ in 0..8 {
                    state = (state >> 1) ^ if state & 1 != 0 { 0x8408 } else { 0 };
                }
            }
            state
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem => crc16_update(0, data, variant),
        Crc16::False | Crc16::X25 => crc16_update(0xffff, data, variant),
    }
    .pipe(|result| if variant == Crc16::X25 { result ^ 0xffff } else { result })
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut base = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32 = Fletcher32State::new();
    let mut c16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        base.update(selected);
        for &byte in selected {
            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;
            f32.push(byte);
        }
        c16 = crc16_update(c16, selected, variant);
    }

    let crc16 = if variant == Crc16::X25 {
        c16 ^ 0xffff
    } else {
        c16
    };

    Ok(Multi {
        base: base.checksum(),
        fletcher16: ((f16_b as u16) << 8) | f16_a as u16,
        fletcher32: f32.finish(),
        crc16,
    })
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }

    fn matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
        let mut sum = 0u32;
        let mut index = 0usize;
        while vector != 0 {
            if vector & 1 != 0 {
                sum ^= matrix[index];
            }
            vector >>= 1;
            index += 1;
        }
        sum
    }

    fn matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
        for index in 0..32 {
            square[index] = matrix_times(matrix, matrix[index]);
        }
    }

    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for slot in odd.iter_mut().skip(1) {
        *slot = row;
        row <<= 1;
    }

    let mut even = [0u32; 32];
    matrix_square(&mut even, &odd);
    matrix_square(&mut odd, &even);

    let mut len = right_len;
    let mut crc = left;
    loop {
        matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
        matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: index })?;
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }

    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn selected_len(spans: &[Span]) -> usize {
    spans.iter().map(|span| span.end - span.start).sum()
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data.len(), spans)?;
    let base = checksums_spans(data, spans)?;
    let len = selected_len(spans) as u64;
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, base.crc32, base.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data.len(), spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut values = [0u64; 3];
    for (field, range) in [(0, 4..20), (1, 25..33), (2, 40..48)] {
        for offset in range {
            let digit = hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })?;
            values[field] = (values[field] << 4) | digit as u64;
        }
    }

    let expected_len = values[0];
    let actual_len = selected_len(spans) as u64;
    if expected_len != actual_len {
        return Err(VerifyError::Length);
    }

    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    let expected_crc = values[1] as u32;
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    let expected_adler = values[2] as u32;
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

    let initial = match variant { Crc16::Xmodem => 0, _ => 0xffff };
    let raw = crc16_raw(initial, data, variant);
    match variant { Crc16::X25 => raw ^ 0xffff, _ => raw }
}

pub fn multi_checksums(data: &[u8], spans: &[Span], variant: Crc16) -> Result<Multi, SpanError> {
    check_all(data, spans)?;
    let mut base = Stream::new();
    let mut f16a = 0u32;
    let mut f16b = 0u32;
    let mut f32 = F32::new();
    let mut c16 = match variant { Crc16::Xmodem => 0, _ => 0xffff };

    for span in spans {
        let bytes = &data[span.start..span.end];
        base.update(bytes);
        for &byte in bytes {
            f16a = (f16a + byte as u32) % 255;
            f16b = (f16b + f16a) % 255;
            f32.byte(byte);
        }
        c16 = crc16_raw(c16, bytes, variant);
    }

    Ok(Multi {
        base: base.checksum(),
        fletcher16: ((f16b as u16) << 8) | f16a as u16,
        fletcher32: f32.finish(),
        crc16: if variant == Crc16::X25 { c16 ^ 0xffff } else { c16 },
    })
}

fn gf_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut result = 0;
    let mut bit = 0;
    while vector != 0 {
        if vector & 1 != 0 { result ^= matrix[bit]; }
        vector >>= 1;
        bit += 1;
    }
    result
}

fn gf_square(out: &mut [u32; 32], matrix: &[u32; 32]) {
    for i in 0..32 { out[i] = gf_times(matrix, matrix[i]); }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 { return left; }

    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }
    let mut even = [0u32; 32];
    gf_square(&mut even, &odd);
    gf_square(&mut odd, &even);

    let mut crc = left;
    let mut len = right_len;
    loop {
        gf_square(&mut even, &odd);
        if len & 1 != 0 { crc = gf_times(&even, crc); }
        len >>= 1;
        if len == 0 { break; }
        gf_square(&mut odd, &even);
        if len & 1 != 0 { crc = gf_times(&odd, crc); }
        len >>= 1;
        if len == 0 { break; }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (index, part) in parts.iter().enumerate() {
        match total.checked_add(part.len) {
            Some(next) => total = next,
            None => return Err(CombineError::LengthOverflow { part: index }),
        }
    }
    for (index, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: index });
        }
    }
    let mut crc = 0;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

fn logical_len(spans: &[Span]) -> u64 {
    spans.iter().fold(0u64, |sum, span| sum + (span.end - span.start) as u64)
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    check_all(data, spans)?;
    let base = checksums_spans(data, spans)?;
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        logical_len(spans), base.crc32, base.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    check_all(data, spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }

    let mut len_value = 0u64;
    let mut crc_value = 0u32;
    let mut adler_value = 0u32;
    for offset in 4..20 {
        len_value = (len_value << 4) | hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })? as u64;
    }
    for offset in 25..33 {
        crc_value = (crc_value << 4) | hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })? as u32;
    }
    for offset in 40..48 {
        adler_value = (adler_value << 4) | hex_value(bytes[offset]).ok_or(VerifyError::Hex { offset })? as u32;
    }

    let actual_len = logical_len(spans);
    if len_value != actual_len { return Err(VerifyError::Length); }
    let actual = checksums_spans(data, spans).map_err(VerifyError::Span)?;
    if crc_value != actual.crc32 {
        return Err(VerifyError::Mismatch { field: 0, expected: crc_value, actual: actual.crc32 });
    }
    if adler_value != actual.adler32 {
        return Err(VerifyError::Mismatch { field: 1, expected: adler_value, actual: actual.adler32 });
    }
    Ok(())
}
