pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    match options.width {
        0 => return Err(WrapError::WidthZero),
        width if width > 256 => return Err(WrapError::WidthTooLarge { width }),
        _ => {}
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (byte_index, scalar) in input.char_indices() {
        if scalar.is_control() && !matches!(scalar, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index: byte_index });
        }
    }

    let mut normalized = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(ch);
        }
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut index = 0;
    while index < normalized.len() {
        if normalized[index] != '\n' {
            segment.push(normalized[index]);
            index += 1;
            continue;
        }

        let start = index;
        while index < normalized.len() && normalized[index] == '\n' {
            index += 1;
        }

        if index - start == 1 {
            segment.push('\n');
        } else {
            layout(&segment, options, &mut result);
            segment.clear();
            for _ in 1..(index - start) {
                result.push(String::new());
            }
        }
    }
    layout(&segment, options, &mut result);
    Ok(result)
}

fn separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn layout(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<&[char]> = Vec::new();
    let mut start = None;
    for (i, &ch) in segment.iter().enumerate() {
        if separator(ch) {
            if let Some(begin) = start.take() {
                words.push(&segment[begin..i]);
            }
        } else if start.is_none() {
            start = Some(i);
        }
    }
    if let Some(begin) = start {
        words.push(&segment[begin..]);
    }
    if words.is_empty() {
        return;
    }

    let mut line = Vec::new();
    let mut line_number = 0usize;
    let mut wi = 0usize;
    let mut offset = 0usize;

    while wi < words.len() {
        let indent = if line_number == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        let cap = options.width - indent;
        let tail_len = words[wi].len() - offset;

        if tail_len <= cap {
            let separator_len = usize::from(!line.is_empty());
            if line.len() + separator_len + tail_len <= cap {
                if separator_len != 0 {
                    line.push(' ');
                }
                line.extend_from_slice(&words[wi][offset..]);
                wi += 1;
                offset = 0;
            } else {
                push(&line, indent, output);
                line.clear();
                line_number += 1;
            }
        } else if line.is_empty() {
            line.extend_from_slice(&words[wi][offset..offset + cap]);
            offset += cap;
            push(&line, indent, output);
            line.clear();
            line_number += 1;
            if offset == words[wi].len() {
                wi += 1;
                offset = 0;
            }
        } else {
            let available = cap - line.len() - 1;
            if available < 2 {
                push(&line, indent, output);
                line.clear();
                line_number += 1;
            } else {
                line.push(' ');
                line.extend_from_slice(&words[wi][offset..offset + available]);
                offset += available;
                push(&line, indent, output);
                line.clear();
                line_number += 1;
                if offset == words[wi].len() {
                    wi += 1;
                    offset = 0;
                }
            }
        }
    }

    if !line.is_empty() {
        let indent = if line_number == 0 {
            options.first_indent
        } else {
            options.rest_indent
        };
        push(&line, indent, output);
    }
}

fn push(content: &[char], indent: usize, output: &mut Vec<String>) {
    let mut value = String::new();
    for _ in 0..indent {
        value.push(' ');
    }
    for &ch in content {
        value.push(ch);
    }
    output.push(value);
}

    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

fn is_separator_b(c: char) -> bool {
    c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00A0}'
}

fn append_indent_b(line: &mut String, line_no: usize, options: &WrapOptions) {
    let indent = if line_no == 0 {
        options.first_indent
    } else {
        options.rest_indent
    };
    for _ in 0..indent {
        line.push(' ');
    }
}

fn finish_line_b(
    line: &mut String,
    line_no: &mut usize,
    options: &WrapOptions,
    output: &mut Vec<String>,
) {
    output.push(std::mem::take(line));
    *line_no += 1;
    append_indent_b(line, *line_no, options);
}

fn layout_segment_b(chars: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut line_no = 0usize;
    let mut line = String::new();
    append_indent_b(&mut line, line_no, options);
    let mut content = 0usize;

    let mut pos = 0usize;
    while pos < chars.len() {
        while pos < chars.len() && is_separator_b(chars[pos]) {
            pos += 1;
        }
        if pos == chars.len() {
            break;
        }

        let begin = pos;
        while pos < chars.len() && !is_separator_b(chars[pos]) {
            pos += 1;
        }
        let end = pos;
        let mut word_pos = begin;

        while word_pos < end {
            let indent = if line_no == 0 {
                options.first_indent
            } else {
                options.rest_indent
            };
            let cap = options.width - indent;
            let remaining = end - word_pos;

            if remaining <= cap {
                if content + remaining + usize::from(content != 0) <= cap {
                    if content != 0 {
                        line.push(' ');
                        content += 1;
                    }
                    for &ch in &chars[word_pos..end] {
                        line.push(ch);
                    }
                    content += remaining;
                    word_pos = end;
                } else {
                    finish_line_b(&mut line, &mut line_no, options, output);
                    content = 0;
                }
            } else if content == 0 {
                for &ch in &chars[word_pos..word_pos + cap] {
                    line.push(ch);
                }
                word_pos += cap;
                finish_line_b(&mut line, &mut line_no, options, output);
                content = 0;
            } else {
                let available = cap - content - 1;
                if available >= 2 {
                    line.push(' ');
                    for &ch in &chars[word_pos..word_pos + available] {
                        line.push(ch);
                    }
                    content += 1 + available;
                    word_pos += available;
                    finish_line_b(&mut line, &mut line_no, options, output);
                    content = 0;
                } else {
                    finish_line_b(&mut line, &mut line_no, options, output);
                    content = 0;
                }
            }
        }
    }

    if content > 0 {
        output.push(line);
    }
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut chars = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(ch) = iter.next() {
        if ch == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            chars.push('\n');
        } else {
            chars.push(ch);
        }
    }

    if chars.is_empty() {
        return Ok(Vec::new());
    }

    let mut output = Vec::new();
    let mut start = 0usize;
    let mut scan = 0usize;
    while scan < chars.len() {
        if chars[scan] == '\n' {
            let run_start = scan;
            while scan < chars.len() && chars[scan] == '\n' {
                scan += 1;
            }
            let run_len = scan - run_start;
            if run_len >= 2 {
                layout_segment_b(&chars[start..run_start], options, &mut output);
                for _ in 1..run_len {
                    output.push(String::new());
                }
                start = scan;
            }
        } else {
            scan += 1;
        }
    }
    layout_segment_b(&chars[start..], options, &mut output);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && !matches!(ch, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let mut normalized = Vec::with_capacity(input.len());
    let mut chars = input.chars().peekable();
    while let Some(ch) = chars.next() {
        match ch {
            '\r' => {
                if chars.peek() == Some(&'\n') {
                    chars.next();
                }
                normalized.push('\n');
            }
            _ => normalized.push(ch),
        }
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0;
    while i < normalized.len() {
        if normalized[i] != '\n' {
            segment.push(normalized[i]);
            i += 1;
            continue;
        }

        let mut end = i + 1;
        while end < normalized.len() && normalized[end] == '\n' {
            end += 1;
        }
        if end - i >= 2 {
            layout(&segment, options, &mut result);
            segment.clear();
            for _ in 1..(end - i) {
                result.push(String::new());
            }
            i = end;
        } else {
            segment.push('\n');
            i += 1;
        }
    }
    layout(&segment, options, &mut result);
    Ok(result)
}

fn separator(ch: char) -> bool {
    matches!(ch, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn layout(segment: &[char], options: &WrapOptions, result: &mut Vec<String>) {
    let mut words = Vec::<Vec<char>>::new();
    let mut current = Vec::<char>::new();
    for &ch in segment {
        if separator(ch) {
            if !current.is_empty() {
                words.push(std::mem::take(&mut current));
            }
        } else {
            current.push(ch);
        }
    }
    if !current.is_empty() {
        words.push(current);
    }

    let mut line = String::new();
    let mut len = 0usize;
    let mut is_first = true;

    for word in words {
        let mut start = 0;
        while start < word.len() {
            let indent = if is_first {
                options.first_indent
            } else {
                options.rest_indent
            };
            let cap = options.width - indent;
            let remaining = word.len() - start;

            if remaining <= cap {
                let separator_len = usize::from(len > 0);
                if len + separator_len + remaining <= cap {
                    if separator_len == 1 {
                        line.push(' ');
                        len += 1;
                    }
                    for &ch in &word[start..] {
                        line.push(ch);
                    }
                    len += remaining;
                    start = word.len();
                } else {
                    push_line(result, &mut line, &mut len, &mut is_first, indent);
                }
            } else if len == 0 {
                for &ch in &word[start..start + cap] {
                    line.push(ch);
                }
                len = cap;
                start += cap;
                push_line(result, &mut line, &mut len, &mut is_first, indent);
            } else {
                let available = cap - len - 1;
                if available >= 2 {
                    line.push(' ');
                    for &ch in &word[start..start + available] {
                        line.push(ch);
                    }
                    len += available + 1;
                    start += available;
                }
                push_line(result, &mut line, &mut len, &mut is_first, indent);
            }
        }
    }

    if !line.is_empty() {
        let indent = if is_first {
            options.first_indent
        } else {
            options.rest_indent
        };
        let mut full = " ".repeat(indent);
        full.push_str(&line);
        result.push(full);
    }
}

fn push_line(
    result: &mut Vec<String>,
    line: &mut String,
    len: &mut usize,
    is_first: &mut bool,
    indent: usize,
) {
    let mut full = " ".repeat(indent);
    full.push_str(line);
    result.push(full);
    line.clear();
    *len = 0;
    *is_first = false;
}
