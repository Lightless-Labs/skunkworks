fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            if hi.is_none() || lo.is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hi.unwrap() << 4) | lo.unwrap());
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = origins.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    for &b in text.as_bytes() {
        let base = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let extra = match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        };
        if base || extra {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
    }
    out
}

fn codec_to_parse(e: CodecError, base: usize) -> ParseError {
    match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

        input[authority_end..path_end].to_owned()
    };

    if let Some(ref raw) = userinfo {
        validate_component(Component::UserInfo, raw, rest_start)?;
    }
    if path != "/" || path_end != authority_end {
        validate_component(Component::Path, &path, authority_end)?;
    }
    if let Some(ref raw) = query {
        let q_at = path_end + 1;
        validate_component(Component::Query, raw, q_at)?;
    }
    if let Some(ref raw) = fragment {
        let hash_at = before_fragment_end;
        validate_component(Component::Fragment, raw, hash_at + 1)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn parse_port(s: &str, at: usize) -> Result<u16, ParseError> {
    if s.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    let b = s.as_bytes();
    for (i, &c) in b.iter().enumerate() {
        if !c.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: at + i });
        }
    }
    if b.len() >= 2 && b[0] == b'0' {
        return Err(ParseError::InvalidPort { at });
    }
    let mut value: u32 = 0;
    for &c in b {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((c - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at });
    }
    Ok(value as u16)
}

pub enum PathError {
    NotAbsolute,
    BadEscape { at: usize },
    InvalidUtf8 { at: usize },
}

fn codec_to_path(e: CodecError) -> PathError {
    match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(codec_to_path)?;

    let mut stack: Vec<&str> = Vec::new();
    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut terminal_navigation = false;
    for (i, raw) in segments.iter().enumerate() {
        let decoded = decode_component(Component::Path, raw).map_err(codec_to_path)?;
        let nav = decoded == "." || decoded == "..";
        if i + 1 == segments.len() {
            terminal_navigation = nav;
        }
        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(raw);
        }
    }
    if terminal_navigation {
        stack.push("");
    }
    Ok(format!("/{}", stack.join("/")))
}

pub enum ResolveError {
    Base(ParseError),
    Reference(ParseError),
    Path(PathError),
}

fn scan_reference(input: &str) -> Result<(), ParseError> {
    for (i, &b) in input.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())
}

fn parse_reference_parts(reference: &str) -> (String, Option<String>, Option<String>) {
    let bytes = reference.as_bytes();
    let mut frag_at = reference.len();
    let mut fragment = None;
    if let Some(i) = bytes.iter().position(|&b| b == b'#') {
        frag_at = i;
        fragment = Some(reference[i + 1..].to_owned());
    }
    let mut query_at = frag_at;
    let mut query = None;
    if let Some(i) = bytes[..frag_at].iter().position(|&b| b == b'?') {
        query_at = i;
        query = Some(reference[i + 1..frag_at].to_owned());
    }
    (reference[..query_at].to_owned(), query, fragment)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    scan_reference(reference).map_err(ResolveError::Reference)?;

    if !reference.is_empty() {
        let bytes = reference.as_bytes();
        let first_delim = bytes
            .iter()
            .position(|&b| matches!(b, b'/' | b'?' | b'#'))
            .unwrap_or(bytes.len());
        if bytes[..first_delim].contains(&b':') {
            let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
            u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
            return Ok(u);
        }
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    let (ref_path, ref_query, ref_fragment) = parse_reference_parts(reference);
    let (path, query) = if ref_path.is_empty() {
        let q = if ref_query.is_none() {
            base_url.query.clone()
        } else {
            ref_query
        };
        (base_path, q)
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path
        } else {
            match base_path.rfind('/') {
                Some(i) => format!("{}{}", &base_path[..=i], ref_path),
                None => format!("/{}", ref_path),
            }
        };
        (merged, ref_query)
    };

    let mut constructed = format!("{}://", base_url.scheme);
    if let Some(userinfo) = &base_url.userinfo {
        constructed.push_str(userinfo);
        constructed.push('@');
    }
    if base_url.ipv6 {
        constructed.push('[');
        constructed.push_str(&base_url.host);
        constructed.push(']');
    } else {
        constructed.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        constructed.push(':');
        constructed.push_str(&port.to_string());
    }
    constructed.push_str(&path);
    if let Some(q) = &query {
        constructed.push('?');
        constructed.push_str(q);
    }
    if let Some(f) = &ref_fragment {
        constructed.push('#');
        constructed.push_str(f);
    }
    let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
    u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
    Ok(u)
}

fn form_decode(s: &str) -> String {
    s.replace('+', " ")
}

fn codec_to_query(e: CodecError, base: usize) -> QueryError {
    match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut map = QueryMap { entries: Vec::new() };
    let mut start = 0;
    while start <= input.len() {
        let end = input[start..]
            .find('&')
            .map(|i| start + i)
            .unwrap_or(input.len());
        if end > start {
            let field = &input[start..end];
            let eq = field.find('=');
            let (raw_key, raw_value, value_start) = match eq {
                Some(i) => (&field[..i], &field[i + 1..], start + i + 1),
                None => (field, "", end),
            };
            let key_plus = form_decode(raw_key);
            let key = decode_component(Component::Query, &key_plus)
                .map_err(|e| codec_to_query(e, start))?;
            if key.is_empty() {
                return Err(QueryError::EmptyKey { at: start });
            }
            let value_plus = form_decode(raw_value);
            let value = decode_component(Component::Query, &value_plus)
                .map_err(|e| codec_to_query(e, value_start))?;
            if let Some((_, values)) = map.entries.iter_mut().find(|(k, _)| *k == key) {
                values.push(value);
            } else {
                map.entries.push((key, vec![value]));
            }
        }
        if end == input.len() {
            break;
        }
        start = end + 1;
    }
    Ok(map)
}

fn form_encode(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b == b' ' {
            out.push('+');
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(&str, Vec<&str>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, acc)) = merged.iter_mut().find(|(k, _)| *k == key.as_str()) {
            acc.extend(values.iter().map(String::as_str));
        } else {
            merged.push((key.as_str(), values.iter().map(String::as_str).collect()));
        }
    }
    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = form_encode(key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, form_encode(value)));
        }
    }
    fields.join("&")
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let u = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&u.path).map_err(CanonicalError::Path)?;
    let query_map = match &u.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", u.scheme);
    if let Some(raw) = &u.userinfo {
        let decoded = decode_component(Component::UserInfo, raw)
            .map_err(|e| CanonicalError::Parse(codec_to_parse(e, 0)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if u.ipv6 {
        let addr = Ipv6Addr::from_str(&u.host).map_err(|_| {
            CanonicalError::Parse(ParseError::InvalidIpv6 { at: 0 })
        })?;
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&u.host);
    }
    let keep_port = match (u.scheme.as_str(), u.port) {
        ("http", Some(80)) | ("https", Some(443)) => false,
        (_, Some(_)) => true,
        _ => false,
    };
    if keep_port {
        out.push(':');
        out.push_str(&u.port.unwrap().to_string());
    }

    let segments: Vec<&str> = path.split('/').collect();
    let mut encoded_segments = Vec::with_capacity(segments.len());
    for segment in segments {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(codec_to_path(e)))?;
        encoded_segments.push(encode_unreserved(&decoded));
    }
    out.push_str(&encoded_segments.join("/"));

    if let Some(map) = query_map {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(raw) = &u.fragment {
        let decoded = decode_component(Component::Fragment, raw)
            .map_err(|e| CanonicalError::Parse(codec_to_parse(e, 0)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn encode_unreserved(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::new();
    for &b in text.as_bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn global_check(bytes: &[u8]) -> Result<(), ParseError> {
    for (i, &b) in bytes.iter().enumerate() {
        if b < 0x21 || b > 0x7e {
            return Err(ParseError::InvalidByte { at: i });
        }
    }
    Ok(())

fn hv(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}
fn map_codec(e: CodecError, offset: usize) -> ParseError {
    match e {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut origin = Vec::with_capacity(input.len());
    let s = input.as_bytes();
    let mut i = 0;
    while i < s.len() {
        if s[i] == b'%' {
            if i.checked_add(2).filter(|&n| n < s.len()).is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            let h = hv(s[i + 1]);
            let l = hv(s[i + 2]);
            if h.is_none() || l.is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            bytes.push(h.unwrap() * 16 + l.unwrap());
            origin.push(i);
            i += 3;
        } else {
            bytes.push(s[i]);
            origin.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&bytes) {
        Ok(t) => Ok(t.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origin.get(e.valid_up_to()).copied().unwrap_or(s.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut result = String::new();
    for b in text.bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => b == b'/' || b == b'?',
                Component::Fragment => b == b'/' || b == b'?' || b == b'#',
            };
        if safe {
            result.push(b as char);
        } else {
            result.push('%');
            result.push(HEX[(b >> 4) as usize] as char);
            result.push(HEX[(b & 15) as usize] as char);
        }
    }
    result
}

fn parse_port(s: &str, at: usize) -> Result<u16, ParseError> {
    if s.is_empty() {
        return Err(ParseError::InvalidPort { at });
    }
    if let Some((i, _)) = s.bytes().enumerate().find(|(_, b)| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: at + i });
    }
    if s.len() > 1 && s.starts_with('0') {
        return Err(ParseError::InvalidPort { at });
    }
    let mut n = 0u32;
    for b in s.bytes() {
        n = n
            .checked_mul(10)
            .and_then(|x| x.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at })?;
    }
    if n == 0 || n > 65535 {
        Err(ParseError::InvalidPort { at })
    } else {
        Ok(n as u16)
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    global_check(input.as_bytes())?;
    let b = input.as_bytes();
    let c = match b.iter().position(|&x| x == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: b.len() }),
    };
    if c == 0 || !b[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..c {
        if !(b[i].is_ascii_alphanumeric() || matches!(b[i], b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if b.get(c + 1) != Some(&b'/') || b.get(c + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: c + 1 });
    }
    let scheme = input[..c].to_ascii_lowercase();
    let auth_start = c + 3;
    let auth_end = b[auth_start..]
        .iter()
        .position(|x| matches!(x, b'/' | b'?' | b'#'))
        .map(|x| auth_start + x)
        .unwrap_or(b.len());
    let a = &input[auth_start..auth_end];

    let first_at = a.find('@');
    let userinfo;
    let hp_start;
    if let Some(at) = first_at {
        if let Some(second) = a[at + 1..].find('@') {
            return Err(ParseError::InvalidUserInfo {
                at: auth_start + at + 1 + second,
            });
        }
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        userinfo = Some(a[..at].to_owned());
        hp_start = at + 1;
    } else {
        userinfo = None;
        hp_start = 0;
    }
    let hp = &a[hp_start..];
    let host_at = auth_start + hp_start;
    let (host, port, ipv6) = if hp.starts_with('[') {
        let close = hp.find(']').ok_or(ParseError::InvalidIpv6 { at: host_at })?;
        let inner = &hp[1..close];
        if Ipv6Addr::from_str(inner).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_at });
        }
        let rest = &hp[close + 1..];
        let port = if rest.is_empty() {
            None
        } else if rest.starts_with(':') {
            Some(parse_port(&rest[1..], host_at + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_at + close + 1,
            });
        };
        (inner.to_ascii_lowercase(), port, true)
    } else {
        let split = hp.find(':');
        let (h, p, p_at) = match split {
            Some(i) => (&hp[..i], Some(&hp[i + 1..]), host_at + i + 1),
            None => (hp, None, host_at + hp.len()),
        };
        if h.is_empty() {
            return Err(ParseError::InvalidHost { at: host_at });
        }
        for (i, ch) in h.bytes().enumerate() {
            if !(ch.is_ascii_alphanumeric() || matches!(ch, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_at + i });
            }
        }
        let port = match p {
            Some(s) => Some(parse_port(s, p_at)?),
            None => None,
        };
        (h.to_ascii_lowercase(), port, false)
    };

    let tail = &input[auth_end..];
    let fpos = tail.find('#');
    let pre_f = fpos.map(|i| &tail[..i]).unwrap_or(tail);
    let frag = fpos.map(|i| tail[i + 1..].to_owned());
    let qpos = pre_f.find('?');
    let raw_path = qpos.map(|i| &pre_f[..i]).unwrap_or(pre_f);
    let query = qpos.map(|i| pre_f[i + 1..].to_owned());
    let path = if raw_path.is_empty() { "/".to_string() } else { raw_path.to_string() };

    if let Some(u) = userinfo.as_ref() {
        decode_component(Component::UserInfo, u).map_err(|e| map_codec(e, auth_start))?;
    }
    if !raw_path.is_empty() {
        decode_component(Component::Path, raw_path).map_err(|e| map_codec(e, auth_end))?;
    }
    if let Some(q) = query.as_ref() {
        decode_component(Component::Query, q).map_err(|e| map_codec(e, auth_end + qpos.unwrap() + 1))?;
    }
    if let Some(f) = frag.as_ref() {
        decode_component(Component::Fragment, f).map_err(|e| map_codec(e, auth_end + fpos.unwrap() + 1))?;
    }

    Ok(Url { scheme, userinfo, host, port, path, query, fragment: frag, ipv6 })
}

fn codec_to_path(e: CodecError) -> PathError {
    match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(codec_to_path)?;
    let mut stack: Vec<&str> = Vec::new();
    let mut last_navigation = false;
    for segment in path[1..].split('/') {
        let decoded = decode_component(Component::Path, segment).map_err(codec_to_path)?;
        match decoded.as_str() {
            "." => last_navigation = true,
            ".." => {
                last_navigation = true;
                stack.pop();
            }
            _ => {
                last_navigation = false;
                stack.push(segment);
            }
        }
    }
    if path == "/" {
        return Ok("/".to_owned());
    }
    if last_navigation {
        stack.push("");
    }
    Ok(format!("/{}", stack.join("/")))
}

fn codec_to_query(e: CodecError, offset: usize) -> QueryError {
    match e {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: offset + at },
    }
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut start = 0;
    for field in input.split('&') {
        let at = start;
        start += field.len() + 1;
        if field.is_empty() {
            continue;
        }
        let eq = field.find('=');
        let (key_raw, value_raw) = eq.map(|i| (&field[..i], &field[i + 1..])).unwrap_or((field, ""));
        let key_prepared = key_raw.replace('+', " ");
        let key = decode_component(Component::Query, &key_prepared)
            .map_err(|e| codec_to_query(e, at))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at });
        }
        let value_at = at + key_raw.len() + usize::from(eq.is_some());
        let value_prepared = value_raw.replace('+', " ");
        let value = decode_component(Component::Query, &value_prepared)
            .map_err(|e| codec_to_query(e, value_at))?;
        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| k == &key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }
    Ok(QueryMap { entries })
}

fn form_encode(input: &str) -> String {
    let mut output = String::new();
    const H: &[u8; 16] = b"0123456789ABCDEF";
    for b in input.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            output.push(b as char);
        } else if b == b' ' {
            output.push('+');
        } else {
            output.push('%');
            output.push(H[(b >> 4) as usize] as char);
            output.push(H[(b & 15) as usize] as char);
        }
    }
    output
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, vals) in &map.entries {
        if key.is_empty() || vals.is_empty() { continue; }
        if let Some((_, found)) = merged.iter_mut().find(|(k, _)| k == key) {
            found.extend(vals.iter().cloned());
        } else {
            merged.push((key.clone(), vals.clone()));
        }
    }
    let mut fields = Vec::new();
    for (key, vals) in merged {
        let key = form_encode(&key);
        for val in vals {
            fields.push(format!("{}={}", key, form_encode(&val)));
        }
    }
    fields.join("&")
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    global_check(reference.as_bytes()).map_err(ResolveError::Reference)?;

    let b = reference.as_bytes();
    let cutoff = b.iter().position(|&x| matches!(x, b'/' | b'?' | b'#')).unwrap_or(b.len());
    if b[..cutoff].contains(&b':') {
        let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&absolute).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    let fpos = reference.find('#');
    let pre_f = fpos.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = fpos.map(|i| reference[i + 1..].to_string());
    let qpos = pre_f.find('?');
    let ref_path = qpos.map(|i| &pre_f[..i]).unwrap_or(pre_f);
    let ref_query = qpos.map(|i| pre_f[i + 1..].to_string());

    let (chosen_path, chosen_query) = if ref_path.is_empty() {
        (base_path, if qpos.is_some() { ref_query } else { base_url.query.clone() })
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_string()
        } else {
            let cut = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_path[..cut], ref_path)
        };
        (merged, ref_query)
    };

    let mut raw = format!("{}://", base_url.scheme);
    if let Some(info) = &base_url.userinfo {
        raw.push_str(info);
        raw.push('@');
    }
    if base_url.ipv6 {
        raw.push('[');
        raw.push_str(&base_url.host);
        raw.push(']');
    } else {
        raw.push_str(&base_url.host);
    }
    if let Some(p) = base_url.port {
        raw.push(':');
        raw.push_str(&p.to_string());
    }
    raw.push_str(&chosen_path);
    if let Some(q) = chosen_query {
        raw.push('?');
        raw.push_str(&q);
    }
    if let Some(f) = fragment {
        raw.push('#');
        raw.push_str(&f);
    }
    let mut u = parse_url(&raw).map_err(ResolveError::Reference)?;
    u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
    Ok(u)
}

fn encode_unreserved(input: &str) -> String {
    let mut out = String::new();
    const H: &[u8; 16] = b"0123456789ABCDEF";
    for b in input.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(H[(b >> 4) as usize] as char);
            out.push(H[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let u = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&u.path).map_err(CanonicalError::Path)?;
    let query = match &u.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", u.scheme);
    if let Some(info) = &u.userinfo {
        let decoded = decode_component(Component::UserInfo, info)
            .map_err(|e| CanonicalError::Parse(map_codec(e, 0)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if u.ipv6 {
        let address = Ipv6Addr::from_str(&u.host).expect("parsed IPv6");
        out.push('[');
        out.push_str(&address.to_string());
        out.push(']');
    } else {
        out.push_str(&u.host);
    }
    if let Some(port) = u.port {
        if !((u.scheme == "http" && port == 80) || (u.scheme == "https" && port == 443)) {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let encoded_segments: Vec<String> = path
        .split('/')
        .map(|segment| {
            let decoded = decode_component(Component::Path, segment).expect("normalized path validated");
            encode_unreserved(&decoded)
        })
        .collect();
    out.push_str(&encoded_segments.join("/"));
    if let Some(q) = query {
        out.push('?');
        out.push_str(&serialise_query(&q));
    }
    if let Some(fragment) = &u.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| CanonicalError::Parse(map_codec(e, 0)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn scan_global(input: &str) -> Result<(), ParseError> {
    for (at, b) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ParseError::InvalidByte { at });
        }
    }
    Ok(())
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let (Some(h), Some(l)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) else {
                return Err(CodecError::BadEscape { at: i });
            };
            out.push((h << 4) | l);
            offsets.push(i);
            i += 3;
        } else {
            let b = bytes[i];
            out.push(b);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&out) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: offsets.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for b in text.bytes() {
        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let extra = match component {
            Component::UserInfo => b == b':',
            Component::Path => b == b'/',
            Component::Query => matches!(b, b'/' | b'?'),
            Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
        };
        if unreserved || extra {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn parse_port(port: &str, start: usize) -> Result<Option<u16>, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    for (i, b) in port.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: start + i });
        }
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value: u32 = 0;
    for b in port.bytes() {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::InvalidPort { at: start })?;
    }
    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(Some(value as u16))
}

    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    scan_global(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let tail_rel = bytes[auth_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'));
    let auth_end = tail_rel.map(|x| auth_start + x).unwrap_or(input.len());
    let authority = &input[auth_start..auth_end];

    let ats: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| if b == b'@' { Some(i) } else { None })
        .collect();
    if ats.len() > 1 {
        return Err(ParseError::InvalidUserInfo {
            at: auth_start + ats[1],
        });
    }
    let (userinfo, host_start, hostport) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (
            Some(authority[..at].to_owned()),
            auth_start + at + 1,
            &authority[at + 1..],
        )
    } else {
        (None, auth_start, authority)
    };

    let (host, port, ipv6) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: host_start }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: host_start });
        }
        let after = &hostport[close + 1..];
        let port = if after.is_empty() {
            None
        } else if after.as_bytes()[0] == b':' {
            parse_port(&after[1..], host_start + close + 2)?
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        };
        (interior.to_ascii_lowercase(), port, true)
    } else {
        let split = hostport.as_bytes().iter().position(|&b| b == b':');
        let (raw_host, raw_port, port_start) = match split {
            Some(i) => (
                &hostport[..i],
                Some(&hostport[i + 1..]),
                host_start + i + 1,
            ),
            None => (hostport, None, host_start + hostport.len()),
        };
        if raw_host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (i, b) in raw_host.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_start + i });
            }
        }
        let port = match raw_port {
            Some(p) => parse_port(p, port_start)?,
            None => None,
        };
        (raw_host.to_ascii_lowercase(), port, false)
    };

    let tail = &input[auth_end..];
    let fragment_pos = tail.find('#');
    let before_fragment = fragment_pos.map(|i| &tail[..i]).unwrap_or(tail);
    let fragment = fragment_pos.map(|i| tail[i + 1..].to_owned());
    let query_pos = before_fragment.find('?');
    let path_raw = query_pos
        .map(|i| &before_fragment[..i])
        .unwrap_or(before_fragment);
    let query = query_pos.map(|i| before_fragment[i + 1..].to_owned());
    let path = if path_raw.is_empty() {
        "/".to_owned()
    } else {
        path_raw.to_owned()
    };

    if let Some(ref u) = userinfo {
        decode_component(Component::UserInfo, u)
            .map_err(|e| codec_parse_error(e, auth_start))?;
    }
    if !path_raw.is_empty() {
        decode_component(Component::Path, path_raw)
            .map_err(|e| codec_parse_error(e, auth_end))?;
    }
    if let Some(q) = query.as_ref() {
        let q_start = auth_end + query_pos.unwrap() + 1;
        decode_component(Component::Query, q)
            .map_err(|e| codec_parse_error(e, q_start))?;
    }
    if let Some(f) = fragment.as_ref() {
        let f_start = auth_end + fragment_pos.unwrap() + 1;
        decode_component(Component::Fragment, f)
            .map_err(|e| codec_parse_error(e, f_start))?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn codec_path_error(err: CodecError) -> PathError {
    match err {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    }
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(codec_path_error)?;

    let mut segments: Vec<&str> = Vec::new();
    let mut start = 1;
    let mut last_nav = false;
    for (i, part) in path[1..].split('/').enumerate() {
        let _ = i;
        let decoded = decode_component(Component::Path, part).map_err(codec_path_error)?;
        if decoded == "." {
            last_nav = true;
        } else if decoded == ".." {
            last_nav = true;
            if !segments.is_empty() {
                segments.pop();
            }
        } else {
            last_nav = false;
            segments.push(part);
        }
        start += part.len() + 1;
    }
    let _ = start;
    if path == "/" {
        return Ok("/".to_owned());
    }
    if last_nav {
        segments.push("");
    }
    Ok(format!("/{}", segments.join("/")))
}

fn query_codec_error(err: CodecError, base: usize) -> QueryError {
    match err {
        CodecError::BadEscape { at } => QueryError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: base + at },
    }
}

fn plus_to_space(s: &str) -> String {
    s.replace('+', " ")
}

pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut offset = 0usize;
    for field in input.split('&') {
        let field_start = offset;
        offset += field.len() + 1;
        if field.is_empty() {
            continue;
        }
        let (raw_key, raw_value) = match field.find('=') {
            Some(i) => (&field[..i], &field[i + 1..]),
            None => (field, ""),
        };
        let key_input = plus_to_space(raw_key);
        let key = decode_component(Component::Query, &key_input)
            .map_err(|e| query_codec_error(e, field_start))?;
        if key.is_empty() {
            return Err(QueryError::EmptyKey { at: field_start });
        }
        let value_base = field_start + raw_key.len() + usize::from(field.as_bytes().contains(&b'='));
        let value_input = plus_to_space(raw_value);
        let value = decode_component(Component::Query, &value_input)
            .map_err(|e| query_codec_error(e, value_base))?;

        if let Some((_, values)) = entries.iter_mut().find(|(k, _)| *k == key) {
            values.push(value);
        } else {
            entries.push((key, vec![value]));
        }
    }
    Ok(QueryMap { entries })
}

fn encode_form(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for b in text.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else if b == b' ' {
            out.push('+');
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (key, values) in &map.entries {
        if key.is_empty() || values.is_empty() {
            continue;
        }
        if let Some((_, existing)) = merged.iter_mut().find(|(k, _)| k == key) {
            existing.extend(values.iter().cloned());
        } else {
            merged.push((key.clone(), values.clone()));
        }
    }
    let mut fields = Vec::new();
    for (key, values) in merged {
        let encoded_key = encode_form(&key);
        for value in values {
            fields.push(format!("{}={}", encoded_key, encode_form(&value)));
        }
    }
    fields.join("&")
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    scan_global(reference).map_err(ResolveError::Reference)?;

    let bytes = reference.as_bytes();
    let first_struct = bytes
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(bytes.len());
    if bytes[..first_struct].contains(&b':') {
        let mut url = parse_url(reference).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }
    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
        url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
        return Ok(url);
    }

    let fragment_pos = reference.find('#');
    let before_fragment = fragment_pos.map(|i| &reference[..i]).unwrap_or(reference);
    let fragment = fragment_pos.map(|i| reference[i + 1..].to_owned());
    let query_pos = before_fragment.find('?');
    let ref_path = query_pos.map(|i| &before_fragment[..i]).unwrap_or(before_fragment);
    let ref_query = query_pos.map(|i| before_fragment[i + 1..].to_owned());

    let (path, query) = if ref_path.is_empty() {
        (base_path, if query_pos.is_none() { base_url.query.clone() } else { ref_query })
    } else {
        let merged_path = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| i + 1).unwrap_or(0);
            format!("{}{}", &base_path[..slash], ref_path)
        };
        (merged_path, ref_query)
    };

    let mut absolute = format!("{}://", base_url.scheme);
    if let Some(u) = &base_url.userinfo {
        absolute.push_str(u);
        absolute.push('@');
    }
    if base_url.ipv6 {
        absolute.push('[');
        absolute.push_str(&base_url.host);
        absolute.push(']');
    } else {
        absolute.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        absolute.push(':');
        absolute.push_str(&port.to_string());
    }
    absolute.push_str(&path);
    if let Some(q) = &query {
        absolute.push('?');
        absolute.push_str(q);
    }
    if let Some(f) = &fragment {
        absolute.push('#');
        absolute.push_str(f);
    }
    let mut url = parse_url(&absolute).map_err(ResolveError::Reference)?;
    url.path = normalise_path(&url.path).map_err(ResolveError::Path)?;
    Ok(url)
}

pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let url = parse_url(input).map_err(CanonicalError::Parse)?;
    let path = normalise_path(&url.path).map_err(CanonicalError::Path)?;
    let query = match &url.query {
        Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?),
        None => None,
    };

    let mut out = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        let decoded = decode_component(Component::UserInfo, userinfo)
            .map_err(|e| CanonicalError::Parse(codec_parse_error(e, 0)))?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if url.ipv6 {
        let addr = Ipv6Addr::from_str(&url.host).expect("validated IPv6 address");
        out.push('[');
        out.push_str(&addr.to_string());
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        if !((url.scheme == "http" && port == 80) || (url.scheme == "https" && port == 443)) {
            out.push(':');
            out.push_str(&port.to_string());
        }
    }

    let mut canonical_segments = Vec::new();
    for segment in path.split('/') {
        let decoded = decode_component(Component::Path, segment)
            .map_err(|e| CanonicalError::Path(codec_path_error(e)))?;
        canonical_segments.push(encode_unreserved(&decoded));
    }
    out.push_str(&canonical_segments.join("/"));

    if let Some(map) = query {
        out.push('?');
        out.push_str(&serialise_query(&map));
    }
    if let Some(fragment) = &url.fragment {
        let decoded = decode_component(Component::Fragment, fragment)
            .map_err(|e| CanonicalError::Parse(codec_parse_error(e, 0)))?;
        out.push('#');
        out.push_str(&encode_component(Component::Fragment, &decoded));
    }
    Ok(out)
}

fn encode_unreserved(text: &str) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let mut out = String::with_capacity(text.len());
    for b in text.bytes() {
        if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}
fn codec_decode(input: &str) -> Result<(String, Vec<usize>), CodecError> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex(bytes[i + 1]);
            let lo = hex(bytes[i + 2]);
            if hi.is_none() || lo.is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            out.push((hi.unwrap() << 4) | lo.unwrap());
            origins.push(i);
            i += 3;
        } else {
            out.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }
    match String::from_utf8(out) {
        Ok(s) => Ok((s, origins)),
        Err(e) => {
            let at = e.utf8_error().valid_up_to();
            Err(CodecError::InvalidUtf8 { at: origins.get(at).copied().unwrap_or(input.len()) })
        }
    }
}
pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    codec_decode(input).map(|x| x.0)
}
pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::new();
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe { out.push(b as char); }
        else { out.push('%'); out.push(char::from(b"0123456789ABCDEF"[(b >> 4) as usize])); out.push(char::from(b"0123456789ABCDEF"[(b & 15) as usize])); }
    }
    out
}
fn invalid_raw_byte(input: &str) -> Option<usize> {
    input.as_bytes().iter().position(|&b| !(0x21..=0x7e).contains(&b))
}
fn parse_port(port: &str, start: usize) -> Result<u16, ParseError> {
    if port.is_empty() { return Err(ParseError::InvalidPort { at: start }); }
    if let Some(i) = port.as_bytes().iter().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: start + i });
    }
    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut n: u32 = 0;
    for &b in port.as_bytes() {
        n = n.saturating_mul(10).saturating_add((b - b'0') as u32);
        if n > 65535 { return Err(ParseError::InvalidPort { at: start }); }
    }
    if n == 0 { return Err(ParseError::InvalidPort { at: start }); }
    Ok(n as u16)
}
fn validate_component(s: &str, base: usize, comp: Component) -> Result<(), ParseError> {
    match codec_decode(s) {
        Ok(_) => Ok(()),
        Err(CodecError::BadEscape { at }) => Err(ParseError::BadEscape { at: base + at }),
        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 { at: base + at }),
    }
}
pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() { return Err(ParseError::Empty); }
    if let Some(at) = invalid_raw_byte(input) { return Err(ParseError::InvalidByte { at }); }
    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') { Some(i) => i, None => return Err(ParseError::InvalidScheme { at: input.len() }) };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() { return Err(ParseError::InvalidScheme { at: 0 }); }
    for i in 1..colon {
        if !(bytes[i].is_ascii_alphanumeric() || matches!(bytes[i], b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }
    let auth_start = colon + 3;
    let auth_end = bytes[auth_start..].iter().position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|i| auth_start + i).unwrap_or(bytes.len());
    let authority = &input[auth_start..auth_end];
    let ats: Vec<usize> = authority.bytes().enumerate().filter_map(|(i,b)| if b == b'@' { Some(i) } else { None }).collect();
    if ats.len() > 1 { return Err(ParseError::InvalidUserInfo { at: auth_start + ats[1] }); }
    let (userinfo, hostpart_start) = if let Some(&a) = ats.first() {
        if a == 0 { return Err(ParseError::InvalidUserInfo { at: auth_start }); }
        (Some(authority[..a].to_string()), a + 1)
    } else { (None, 0) };
    let host_start = auth_start + hostpart_start;
    let hostpart = &authority[hostpart_start..];
    let mut ipv6 = false;
    let (host, port) = if hostpart.starts_with('[') {
        let open = host_start;
        let close_rel = match hostpart.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &hostpart[1..close_rel];
        if interior.contains('%') || Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        let suffix = &hostpart[close_rel + 1..];
        let p = if suffix.is_empty() { None }
        else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], host_start + close_rel + 2)?)
        } else {
            return Err(ParseError::InvalidHost { at: host_start + close_rel + 1 });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), p)
    } else {
        let split = hostpart.find(':');
        let (h, p) = if let Some(i) = split { (&hostpart[..i], Some(parse_port(&hostpart[i+1..], host_start+i+1)?)) } else { (hostpart, None) };
        if h.is_empty() { return Err(ParseError::InvalidHost { at: host_start }); }
        for (i, b) in h.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_start + i });
            }
        }
        (h.to_ascii_lowercase(), p)
    };
    let rest = &input[auth_end..];
    let frag_rel = rest.find('#');
    let pre_frag_end = frag_rel.unwrap_or(rest.len());
    let before_frag = &rest[..pre_frag_end];
    let query_rel = before_frag.find('?');
    let path_end = query_rel.unwrap_or(before_frag.len());
    let raw_path = &before_frag[..path_end];
    let query = query_rel.map(|q| before_frag[q+1..].to_string());
    let fragment = frag_rel.map(|f| rest[f+1..].to_string());
    let path = if raw_path.is_empty() { "/".to_string() } else { raw_path.to_string() };
    if let Some(ref u) = userinfo { validate_component(u, auth_start, Component::UserInfo)?; }
    if !raw_path.is_empty() { validate_component(raw_path, auth_end, Component::Path)?; }
    if let Some(q) = query_rel { validate_component(&before_frag[q+1..], auth_end + q + 1, Component::Query)?; }
    if let Some(f) = frag_rel { validate_component(&rest[f+1..], auth_end + f + 1, Component::Fragment)?; }
    Ok(Url { scheme: input[..colon].to_ascii_lowercase(), userinfo, host, port, path, query, fragment, ipv6 })
}
pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') { return Err(PathError::NotAbsolute); }
    let decoded = decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;
    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded[1..].split('/').collect();
    let last_nav = decoded_segments.last().map(|s| *s == "." || *s == "..").unwrap_or(false);
    let mut stack: Vec<&str> = Vec::new();
    for (raw, dec) in raw_segments.iter().zip(decoded_segments.iter()) {
        if *dec == "." { continue; }
        if *dec == ".." { stack.pop(); } else { stack.push(*raw); }
    }
    if last_nav { stack.push(""); }
    Ok(format!("/{}", stack.join("/")))
}
fn global_ref_scan(s: &str) -> Result<(), ParseError> {
    if let Some(at) = invalid_raw_byte(s) { Err(ParseError::InvalidByte { at }) } else { Ok(()) }
}
fn append_query_fragment(s: &mut String, q: Option<&str>, f: Option<&str>) {
    if let Some(q) = q { s.push('?'); s.push_str(q); }
    if let Some(f) = f { s.push('#'); s.push_str(f); }
}
fn normalized_url(mut u: Url) -> Result<Url, PathError> {
    u.path = normalise_path(&u.path)?;
    Ok(u)
}
pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalized_url(base_url).map_err(ResolveError::Path)?;
    global_ref_scan(reference).map_err(ResolveError::Reference)?;
    let rb = reference.as_bytes();
    let first_delim = rb.iter().position(|&b| matches!(b, b'/' | b'?' | b'#')).unwrap_or(rb.len());
    if rb[..first_delim].contains(&b':') {
        let u = parse_url(reference).map_err(ResolveError::Reference)?;
        return normalized_url(u).map_err(ResolveError::Path);
    }
    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let u = parse_url(&constructed).map_err(ResolveError::Reference)?;
        return normalized_url(u).map_err(ResolveError::Path);
    }
    let frag = reference.find('#');
    let pre_frag_end = frag.unwrap_or(reference.len());
    let before_frag = &reference[..pre_frag_end];
    let query = before_frag.find('?');
    let path_end = query.unwrap_or(before_frag.len());
    let rpath = &before_frag[..path_end];
    let rq = query.map(|i| &before_frag[i+1..]);
    let rf = frag.map(|i| &reference[i+1..]);
    let (selected_path, selected_query) = if rpath.is_empty() {
        (base_url.path.clone(), if rq.is_some() { rq.map(str::to_string) } else { base_url.query.clone() })
    } else {
        let p = if rpath.starts_with('/') { rpath.to_string() } else {
            let slash = base_url.path.rfind('/').unwrap_or(0);
            format!("{}{}", &base_url.path[..=slash], rpath)
        };
        (p, rq.map(str::to_string))
    };
    let mut constructed = format!("{}://", base_url.scheme);
    if let Some(ui) = &base_url.userinfo { constructed.push_str(ui); constructed.push('@'); }
    if base_url.ipv6 { constructed.push('['); constructed.push_str(&base_url.host); constructed.push(']'); }
    else { constructed.push_str(&base_url.host); }
    if let Some(p) = base_url.port { constructed.push(':'); constructed.push_str(&p.to_string()); }
    constructed.push_str(&selected_path);
    append_query_fragment(&mut constructed, selected_query.as_deref(), rf);
    let u = parse_url(&constructed).map_err(ResolveError::Reference)?;
    normalized_url(u).map_err(ResolveError::Path)
}
fn form_encode(s: &str) -> String {
    let mut out = String::new();
    for &b in s.as_bytes() {
        if b == b' ' { out.push('+'); }
        else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') { out.push(b as char); }
        else { out.push('%'); out.push(char::from(b"0123456789ABCDEF"[(b >> 4) as usize])); out.push(char::from(b"0123456789ABCDEF"[(b & 15) as usize])); }
    }
    out
}
fn form_decode(s: &str) -> Result<String, CodecError> {
    let replaced = s.replace('+', " ");
    decode_component(Component::Query, &replaced)
}
pub fn parse_query(input: &str) -> Result<QueryMap, QueryError> {
    let mut entries: Vec<(String, Vec<String>)> = Vec::new();
    let mut offset = 0usize;
    for field in input.split('&') {
        let start = offset;
        offset += field.len() + 1;
        if field.is_empty() { continue; }
        let eq = field.find('=');
        let raw_key = eq.map(|i| &field[..i]).unwrap_or(field);
        let raw_value = eq.map(|i| &field[i+1..]).unwrap_or("");
        let key = form_decode(raw_key).map_err(|e| match e {
            CodecError::BadEscape { at } => QueryError::BadEscape { at: start + at },
            CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: start + at },
        })?;
        if key.is_empty() { return Err(QueryError::EmptyKey { at: start }); }
        let value = form_decode(raw_value).map_err(|e| match e {
            CodecError::BadEscape { at } => QueryError::BadEscape { at: start + eq.unwrap_or(field.len()) + if eq.is_some() { 1 } else { 0 } + at },
            CodecError::InvalidUtf8 { at } => QueryError::InvalidUtf8 { at: start + eq.unwrap_or(field.len()) + if eq.is_some() { 1 } else { 0 } + at },
        })?;
        if let Some((_, vals)) = entries.iter_mut().find(|(k, _)| *k == key) { vals.push(value); }
        else { entries.push((key, vec![value])); }
    }
    Ok(QueryMap { entries })
}
pub fn serialise_query(map: &QueryMap) -> String {
    let mut merged: Vec<(String, Vec<String>)> = Vec::new();
    for (k, values) in &map.entries {
        if k.is_empty() || values.is_empty() { continue; }
        if let Some((_, vals)) = merged.iter_mut().find(|(key, _)| key == k) { vals.extend(values.iter().cloned()); }
        else { merged.push((k.clone(), values.clone())); }
    }
    let mut fields = Vec::new();
    for (k, vals) in merged {
        for v in vals { fields.push(format!("{}={}", form_encode(&k), form_encode(&v))); }
    }
    fields.join("&")
}
fn ipv6_canonical(host: &str) -> String {
    Ipv6Addr::from_str(host).map(|x| x.to_string()).unwrap_or_else(|_| host.to_string())
}
pub fn canonical_url(input: &str) -> Result<String, CanonicalError> {
    let mut u = parse_url(input).map_err(CanonicalError::Parse)?;
    u.path = normalise_path(&u.path).map_err(CanonicalError::Path)?;
    let qmap = match &u.query { Some(q) => Some(parse_query(q).map_err(CanonicalError::Query)?), None => None };
    let mut out = format!("{}://", u.scheme);
    if let Some(ui) = &u.userinfo {
        let decoded = decode_component(Component::UserInfo, ui).map_err(|e| match e {
            CodecError::BadEscape { at } => CanonicalError::Parse(ParseError::BadEscape { at }),
            CodecError::InvalidUtf8 { at } => CanonicalError::Parse(ParseError::InvalidUtf8 { at }),
        })?;
        out.push_str(&encode_component(Component::UserInfo, &decoded));
        out.push('@');
    }
    if u.ipv6 { out.push('['); out.push_str(&ipv6_canonical(&u.host)); out.push(']'); }
    else { out.push_str(&u.host); }
    if let Some(p) = u.port {
        if !((u.scheme == "http" && p == 80) || (u.scheme == "https" && p == 443)) {
            out.push(':'); out.push_str(&p.to_string());
        }
    }
    let segs: Vec<String> = u.path.split('/').map(|s| {
        let d = decode_component(Component::Path, s).unwrap_or_else(|_| s.to_string());
        let mut encoded = String::new();
        for &b in d.as_bytes() {
            if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') { encoded.push(b as char); }
            else { encoded.push('%'); encoded.push(char::from(b"0123456789ABCDEF"[(b >> 4) as usize])); encoded.push(char::from(b"0123456789ABCDEF"[(b & 15) as usize])); }
        }
        encoded
    }).collect();
    out.push_str(&segs.join("/"));
    if let Some(q) = qmap { out.push('?'); out.push_str(&serialise_query(&q)); }
    if let Some(f) = &u.fragment {
        let d = decode_component(Component::Fragment, f).map_err(|e| match e {
            CodecError::BadEscape { at } => CanonicalError::Parse(ParseError::BadEscape { at }),
            CodecError::InvalidUtf8 { at } => CanonicalError::Parse(ParseError::InvalidUtf8 { at }),
        })?;
        out.push('#'); out.push_str(&encode_component(Component::Fragment, &d));
    }
    Ok(out)
}
