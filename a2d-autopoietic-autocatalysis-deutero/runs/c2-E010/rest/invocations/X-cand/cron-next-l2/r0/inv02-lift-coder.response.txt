- Empty items and invalid characters are checked before numeric parsing; field count is checked first.
- Item structure is checked before any numbers; numeric base errors precede step errors.
- Offsets are UTF-8 byte offsets within a field; leading zeros precede overflow and bounds checks.
- A stepped single number extends to the field maximum; an unstepped number selects only itself.
- Day-of-month and day-of-week selections are conjunctive; only exact `*` in either day field gets the equality flag.
- `next_fire` scans actual supported minutes strictly after the input and validates date components in the specified order.

### Solution A

```rust
use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [Vec<u8>; 5],
    dom_exact_star: bool,
    dow_exact_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn field_info(index: usize) -> (Field, u32, u32) {
    match index {
        0 => (Field::Minute, 0, 59),
        1 => (Field::Hour, 0, 23),
        2 => (Field::DayOfMonth, 1, 31),
        3 => (Field::Month, 1, 12),
        _ => (Field::DayOfWeek, 0, 6),
    }
}

fn err(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidCharacter { field, offset },
        1 => ParseError::EmptyItem { field, offset },
        2 => ParseError::InvalidSyntax { field, offset },
        3 => ParseError::LeadingZero { field, offset },
        4 => ParseError::NumberOverflow { field, offset },
        5 => ParseError::OutOfRange { field, offset },
        6 => ParseError::ReversedRange { field, offset },
        _ => ParseError::BadStep { field, offset },
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    min: u32,
    max: u32,
    step: bool,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(err(field, start, 3));
    }
    let mut value = 0u32;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or_else(|| err(field, start, 4))?;
    }
    if step {
        if value < 1 || value > max - min + 1 {
            return Err(err(field, start, 7));
        }
    } else if value < min || value > max {
        return Err(err(field, start, 5));
    }
    Ok(value)
}

fn parse_field(text: &str, index: usize) -> Result<Vec<u8>, ParseError> {
    let (field, min, max) = field_info(index);
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(err(field, i, 0));
        }
    }

    let mut selected = BTreeSet::new();
    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| item_start + p)
            .unwrap_or(bytes.len());
        if item_start == item_end {
            return Err(err(field, item_start, 1));
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        let (base_end, step_range) = match slash_positions.as_slice() {
            [] => (item.len(), None),
            [p] => (*p, Some((*p + 1, item.len()))),
            _ => return Err(err(field, item_start, 2)),
        };
        if base_end == 0 || step_range.is_some_and(|(s, e)| s == e) {
            return Err(err(field, item_start, 2));
        }

        let base = &item[..base_end];
        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        enum Base {
            Star,
            Single(usize, usize),
            Range(usize, usize, usize, usize),
        }

        let parsed_base = if is_star && hyphens.is_empty() {
            Base::Star
        } else if !is_star {
            match hyphens.as_slice() {
                [] if !base.is_empty() => Base::Single(0, base.len()),
                [p] if *p > 0 && *p + 1 < base.len() => {
                    Base::Range(0, *p, *p + 1, base.len())
                }
                _ => return Err(err(field, item_start, 2)),
            }
        } else {
            return Err(err(field, item_start, 2));
        };

        let step = if let Some((s, e)) = step_range {
            if item[s..e].contains(&b'/') || item[s..e].contains(&b'-') {
                return Err(err(field, item_start, 2));
            }
            Some(parse_number(item, s, e, field, 1, max - min + 1, true)?)
        } else {
            None
        };

        let (start, end) = match parsed_base {
            Base::Star => (min, max),
            Base::Single(s, e) => {
                let n = parse_number(item, s, e, field, min, max, false)?;
                (n, if step.is_some() { max } else { n })
            }
            Base::Range(s1, e1, s2, e2) => {
                let a = parse_number(item, s1, e1, field, min, max, false)?;
                let b = parse_number(item, s2, e2, field, min, max, false)?;
                if a > b {
                    return Err(err(field, s2, 6));
                }
                (a, b)
            }
        };

        let stride = step.unwrap_or(1);
        let mut n = start;
        while n <= end {
            selected.insert(n as u8);
            match n.checked_add(stride) {
                Some(next) => n = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(selected.into_iter().collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }
    let mut parsed = Vec::with_capacity(5);
    for (i, piece) in pieces.iter().enumerate() {
        parsed.push(parse_field(piece, i)?);
    }
    Ok(Cron {
        values: parsed.try_into().ok().unwrap(),
        dom_exact_star: pieces[2] == "*",
        dow_exact_star: pieces[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.values[i].clone()
    }

    fn contains(&self, i: usize, n: u8) -> bool {
        self.values[i].binary_search(&n).is_ok()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        let weekday = weekday(at.year, at.month, at.day);
        Ok(self.contains(0, at.minute)
            && self.contains(1, at.hour)
            && self.contains(2, at.day)
            && self.contains(3, at.month)
            && self.contains(4, weekday))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut current = after;
        loop {
            current = advance_minute(current);
            if self.matches(current)? {
                return Ok(Some(current));
            }
            if current.year == 2399
                && current.month == 12
                && current.day == 31
                && current.hour == 23
                && current.minute == 59
            {
                return Ok(None);
            }
        }
    }
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_days(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_days(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut w = 6u8;
    let mut y = 2000u16;
    let mut m = 1u8;
    let mut d = 1u8;
    while y < year || m < month || d < day {
        d += 1;
        if d > month_days(y, m) {
            d = 1;
            m += 1;
            if m > 12 {
                m = 1;
                y += 1;
            }
        }
        w = (w + 1) % 7;
    }
    w
}

fn advance_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
        return at;
    }
    at.minute = 0;
    if at.hour < 23 {
        at.hour += 1;
        return at;
    }
    at.hour = 0;
    if at.day < month_days(at.year, at.month) {
        at.day += 1;
        return at;
    }
    at.day = 1;
    if at.month < 12 {
        at.month += 1;
        return at;
    }
    at.month = 1;
    at.year += 1;
    at
}
```

### Solution B

```rust
use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    bits: [u64; 5],
    dom_exact_star: bool,
    dow_exact_star: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

fn info(i: usize) -> (Field, u32, u32) {
    match i {
        0 => (Field::Minute, 0, 59),
        1 => (Field::Hour, 0, 23),
        2 => (Field::DayOfMonth, 1, 31),
        3 => (Field::Month, 1, 12),
        _ => (Field::DayOfWeek, 0, 6),
    }
}

fn parse_field(src: &str, i: usize) -> Result<Vec<u8>, ParseError> {
    let (field, lo, hi) = info(i);
    let b = src.as_bytes();
    if let Some((p, _)) = b.iter().enumerate().find(|(_, c)| {
        !c.is_ascii_digit() && !matches!(**c, b'*' | b'-' | b'/' | b',')
    }) {
        return Err(ParseError::InvalidCharacter { field, offset: p });
    }
    let mut set = BTreeSet::new();
    let mut start = 0;
    while start <= b.len() {
        let end = b[start..].iter().position(|x| *x == b',').map(|x| x + start).unwrap_or(b.len());
        if start == end {
            return Err(ParseError::EmptyItem { field, offset: start });
        }
        let item = &b[start..end];
        let slash: Vec<_> = item.iter().enumerate().filter(|(_, x)| **x == b'/').map(|(p, _)| p).collect();
        if slash.len() > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }
        let base_end = slash.first().copied().unwrap_or(item.len());
        if base_end == 0 || base_end == item.len() && !slash.is_empty() {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }
        let base = &item[..base_end];
        let hy: Vec<_> = base.iter().enumerate().filter(|(_, x)| **x == b'-').map(|(p, _)| p).collect();
        let kind = if base == b"*" && hy.is_empty() {
            0
        } else if hy.is_empty() && !base.is_empty() {
            1
        } else if hy.len() == 1 && hy[0] > 0 && hy[0] + 1 < base.len() {
            2
        } else {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        };
        let parse_num = |s: usize, e: usize, is_step: bool| -> Result<u32, ParseError> {
            if e - s > 1 && item[s] == b'0' {
                return Err(ParseError::LeadingZero { field, offset: start + s });
            }
            let mut n = 0u32;
            for &c in &item[s..e] {
                n = n.checked_mul(10).and_then(|v| v.checked_add((c - b'0') as u32))
                    .ok_or(ParseError::NumberOverflow { field, offset: start + s })?;
            }
            if is_step {
                if !(1..=hi - lo + 1).contains(&n) {
                    return Err(ParseError::BadStep { field, offset: start + s });
                }
            } else if n < lo || n > hi {
                return Err(ParseError::OutOfRange { field, offset: start + s });
            }
            Ok(n)
        };
        let (a, z) = if kind == 0 {
            (lo, hi)
        } else if kind == 1 {
            let n = parse_num(0, base.len(), false)?;
            (n, if slash.is_empty() { n } else { hi })
        } else {
            let h = hy[0];
            let a = parse_num(0, h, false)?;
            let z = parse_num(h + 1, base.len(), false)?;
            if a > z {
                return Err(ParseError::ReversedRange { field, offset: start + h + 1 });
            }
            (a, z)
        };
        let step = if let Some(&s) = slash.first() {
            Some(parse_num(s + 1, item.len(), true)?)
        } else {
            None
        };
        let stride = step.unwrap_or(1);
        let mut n = a;
        while n <= z {
            set.insert(n as u8);
            n += stride;
        }
        if end == b.len() { break; }
        start = end + 1;
    }
    Ok(set.into_iter().collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<_> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let fields = [
        parse_field(parts[0], 0)?,
        parse_field(parts[1], 1)?,
        parse_field(parts[2], 2)?,
        parse_field(parts[3], 3)?,
        parse_field(parts[4], 4)?,
    ];
    let mut bits = [0u64; 5];
    for i in 0..5 {
        for &n in &fields[i] {
            bits[i] |= 1u64 << n;
        }
    }
    Ok(Cron {
        bits,
        dom_exact_star: parts[2] == "*",
        dow_exact_star: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = match field { Field::Minute => 0, Field::Hour => 1, Field::DayOfMonth => 2, Field::Month => 3, Field::DayOfWeek => 4 };
        (0..64).filter(|n| self.bits[i] & (1u64 << n) != 0).map(|n| n as u8).collect()
    }
    fn has(&self, i: usize, n: u8) -> bool { self.bits[i] & (1u64 << n) != 0 }
    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        let wd = weekday(at.year, at.month, at.day);
        Ok(self.has(0, at.minute) && self.has(1, at.hour) && self.has(2, at.day)
            && self.has(3, at.month) && self.has(4, wd))
    }
    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut t = after;
        while !(t.year == 2399 && t.month == 12 && t.day == 31 && t.hour == 23 && t.minute == 59) {
            t = next_minute(t);
            if self.matches(t)? { return Ok(Some(t)); }
        }
        Ok(None)
    }
}
fn leap(y: u16) -> bool { y % 4 == 0 && (y % 100 != 0 || y % 400 == 0) }
fn mdays(y: u16, m: u8) -> u8 { match m { 2 => if leap(y) { 29 } else { 28 }, 4|6|9|11 => 30, _ => 31 } }
fn validate(t: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&t.year) { return Err(DateError::InvalidDateTime { component: DatePart::Year }); }
    if !(1..=12).contains(&t.month) { return Err(DateError::InvalidDateTime { component: DatePart::Month }); }
    if t.day == 0 || t.day > mdays(t.year, t.month) { return Err(DateError::InvalidDateTime { component: DatePart::Day }); }
    if t.hour > 23 { return Err(DateError::InvalidDateTime { component: DatePart::Hour }); }
    if t.minute > 59 { return Err(DateError::InvalidDateTime { component: DatePart::Minute }); }
    Ok(())
}
fn weekday(y: u16, m: u8, d: u8) -> u8 {
    let mut w = 6;
    let mut yy = 2000;
    let mut mm = 1;
    let mut dd = 1;
    while yy < y || mm < m || dd < d {
        dd += 1;
        if dd > mdays(yy, mm) { dd = 1; mm += 1; if mm > 12 { mm = 1; yy += 1; } }
        w = (w + 1) % 7;
    }
    w
}
fn next_minute(mut t: CivilDateTime) -> CivilDateTime {
    if t.minute < 59 { t.minute += 1; return t; }
    t.minute = 0;
    if t.hour < 23 { t.hour += 1; return t; }
    t.hour = 0;
    if t.day < mdays(t.year, t.month) { t.day += 1; return t; }
    t.day = 1;
    if t.month < 12 { t.month += 1; return t; }
    t.month = 1; t.year += 1; t
}
```
