fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    if amount == 0 {
        return Some([0; 4]);
    }

    let mut greedy = [0u16; 4];
    let mut rem = amount;
    for i in (0..4).rev() {
        let take = tubes[i].min(rem / DENOMS[i]);
        greedy[i] = take;
        rem -= take * DENOMS[i];
    }
    if rem == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_total = u32::MAX;

    fn search(
        i: isize,
        rem: u16,
        tubes: [u16; 4],
        current: &mut [u16; 4],
        total: u32,
        best: &mut Option<[u16; 4]>,
        best_total: &mut u32,
    ) {
        if i < 0 {
            if rem == 0
                && (total < *best_total
                    || (total == *best_total
                        && best.map_or(true, |b| {
                            for j in (0..4).rev() {
                                if current[j] != b[j] {
                                    return current[j] > b[j];
                                }
                            }
                            false
                        })))
            {
                *best_total = total;
                *best = Some(*current);
            }
            return;
        }

        let idx = i as usize;
        let denom = DENOMS[idx];
        let max = tubes[idx].min(rem / denom);
        for count in 0..=max {
            let next_total = total + u32::from(count);
            if next_total > *best_total {
                continue;
            }
            current[idx] = count;
            search(
                i - 1,
                rem - count * denom,
                tubes,
                current,
                next_total,
                best,
                best_total,
            );
        }
        current[idx] = 0;
    }

    search(
        3,
        amount,
        tubes,
        &mut [0; 4],
        0,
        &mut best,
        &mut best_total,
    );
    best
}

fn exact_minimum_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut best: Option<([u16; 4], u16)> = None;

    for count10 in 0..=tubes[3].min(amount / 10) {
        for count4 in 0..=tubes[2].min(amount / 4) {
            for count3 in 0..=tubes[1].min(amount / 3) {
                let partial = count10 * 10 + count4 * 4 + count3 * 3;
                if partial > amount {
                    continue;
                }
                let count1 = amount - partial;
                if count1 > tubes[0] {
                    continue;
                }

                let candidate = [count1, count3, count4, count10];
                let total = count1 + count3 + count4 + count10;
                let replace = match best {
                    None => true,
                    Some((old, old_total)) => {
                        total < old_total
                            || (total == old_total
                                && (candidate[3], candidate[2], candidate[1])
                                    > (old[3], old[2], old[1]))
                    }
                };
                if replace {
                    best = Some((candidate, total));
                }
            }
        }
    }

        for &coin in &self.escrow {
            let index = DENOMINATIONS.iter().position(|&d| d == coin).unwrap();
            proposed[index] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit {
                    coin: DENOMINATIONS[i],
                });
            }
        }

        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let mut result = Vec::with_capacity(self.escrow.len());
        for &coin in self.escrow.iter().rev() {
            result.push(coin);
        }
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(result)
    }
}

    let bytes = id.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 8
        && bytes.iter().all(|byte| (b'a'..=b'z').contains(byte))
}

fn find_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut remaining = amount;
    let mut greedy = Vec::new();
    for &coin in &DESCENDING {
        let index = DENOMINATIONS.iter().position(|&d| d == coin).unwrap();
        let take = std::cmp::min(tubes[index], remaining / coin);
        for _ in 0..take {
            greedy.push(coin);
        }
        remaining -= take * coin;
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<Vec<u16>> = None;
    let mut counts = [0u16; 4];
    let mut value = 0u16;
    for ones in 0..=tubes[0] {
        counts[0] = ones;
        value = ones;
        if value > amount {
            break;
        }
        for threes in 0..=tubes[1] {
            counts[1] = threes;
            value = ones + 3 * threes;
            if value > amount {
                break;
            }
            for fours in 0..=tubes[2] {
                counts[2] = fours;
                value = ones + 3 * threes + 4 * fours;
                if value > amount {
                    break;
                }
                for tens in 0..=tubes[3] {
                    value = ones + 3 * threes + 4 * fours + 10 * tens;
                    if value > amount {
                        break;
                    }
                    if value == amount {
                        let mut candidate = Vec::new();
                        for &coin in &DESCENDING {
                            let index =
                                DENOMINATIONS.iter().position(|&d| d == coin).unwrap();
                            for _ in 0..counts[index] {
                                candidate.push(coin);
                            }
                        }
                        let replace = match &best {
                            None => true,
                            Some(old) => {
                                candidate.len() < old.len()
                                    || (candidate.len() == old.len()
                                        && candidate.as_slice() > old.as_slice())
                            }
                        };
                        if replace {
                            best = Some(candidate);
                        }
                    }
                }
            }
        }
    }
    best
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMINATIONS.iter().position(|&d| d == coin)
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let descending = [10u16, 4, 3, 1];
    let mut remaining = amount;
    let mut greedy = Vec::new();
    for &coin in &descending {
        let i = denomination_index(coin).unwrap();
        let take = tubes[i].min(remaining / coin);
        greedy.extend(std::iter::repeat_n(coin, take as usize));
        remaining -= take * coin;
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<Vec<u16>> = None;
    search_change(
        0,
        amount,
        tubes,
        &descending,
        &mut Vec::new(),
        &mut best,
    );
    best
}

fn search_change(
    index: usize,
    remaining: u16,
    tubes: &[u16; 4],
    descending: &[u16; 4],
    current: &mut Vec<u16>,
    best: &mut Option<Vec<u16>>,
) {
    if index == descending.len() {
        if remaining == 0 {
            let replace = match best {
                None => true,
                Some(old) => {
                    current.len() < old.len()
                        || (current.len() == old.len() && current.as_slice() > old.as_slice())
                }
            };
            if replace {
                *best = Some(current.clone());
            }
        }
        return;
    }

    if best.as_ref().is_some_and(|old| current.len() >= old.len()) {
        return;
    }

    let coin = descending[index];
    let tube_index = denomination_index(coin).unwrap();
    let max_take = tubes[tube_index].min(remaining / coin);
    for take in (0..=max_take).rev() {
        let next_remaining = remaining - take * coin;
        current.extend(std::iter::repeat_n(coin, take as usize));
        search_change(
            index + 1,
            next_remaining,
            tubes,
            descending,
            current,
            best,
        );
        current.truncate(current.len() - take as usize);
    }
}

fn best_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    let mut greedy = [0u16; 4];
    let mut left = amount;
    for i in (0..4).rev() {
        greedy[i] = tubes[i].min(left / COINS[i]);
        left -= greedy[i] * COINS[i];
    }
    if left == 0 {
        return Some(expand(greedy));
    }

    let mut table: Vec<Option<[u16; 4]>> = vec![None; amount as usize + 1];
    table[0] = Some([0; 4]);

    for i in (0..4).rev() {
        let denom = COINS[i];
        let available = tubes[i];
        let previous = table.clone();
        for value in 0..=amount {
            let Some(base) = previous[value as usize] else {
                continue;
            };
            let max = available.min((amount - value) / denom);
            for n in 1..=max {
                let new_value = value + n * denom;
                let mut candidate = base;
                candidate[i] += n;
                let slot = &mut table[new_value as usize];
                let replace = match slot {
                    None => true,
                    Some(old) => {
                        let candidate_total: u32 =
                            candidate.iter().map(|&c| c as u32).sum();
                        let old_total: u32 = old.iter().map(|&c| c as u32).sum();
                        candidate_total < old_total
                            || (candidate_total == old_total
                                && lex_better(&candidate, old))
                    }
                };
                if replace {
                    *slot = Some(candidate);
                }
            }
        }
    }

    table[amount as usize].map(expand)
}

fn lex_better(a: &[u16; 4], b: &[u16; 4]) -> bool {
    for i in (0..4).rev() {
        if a[i] != b[i] {
            return a[i] > b[i];
        }
    }
    false
}

fn expand(counts: [u16; 4]) -> Vec<u16> {
    let mut coins = Vec::new();
    for i in (0..4).rev() {
        coins.extend(std::iter::repeat_n(COINS[i], counts[i] as usize));
    }
    coins
}

fn dynamic_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remainder = amount;
    for i in (0..4).rev() {
        let count = tubes[i].min(remainder / DENOMS[i]);
        greedy[i] = count;
        remainder -= count * DENOMS[i];
    }
    if remainder == 0 {
        return Some(greedy);
    }

    // For each value, retain the best coin-count vector seen so far.
    let mut dp: Vec<Option<[u16; 4]>> = vec![None; usize::from(amount) + 1];
    dp[0] = Some([0; 4]);

    for i in 0..4 {
        let denomination = DENOMS[i] as usize;
        let limit = usize::from(tubes[i]);
        let mut next = vec![None; usize::from(amount) + 1];
        for value in 0..=usize::from(amount) {
            if let Some(base) = dp[value] {
                let max = limit.min((usize::from(amount) - value) / denomination);
                for count in 0..=max {
                    let new_value = value + count * denomination;
                    let mut candidate = base;
                    candidate[i] += count as u16;
                    let old_is_better = next[new_value].map_or(false, |old: [u16; 4]| {
                        let candidate_total: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
                        let old_total: u32 = old.iter().map(|&n| u32::from(n)).sum();
                        candidate_total > old_total
                            || (candidate_total == old_total
                                && (0..4).rev().find_map(|j| {
                                    if candidate[j] == old[j] {
                                        None
                                    } else {
                                        Some(candidate[j] < old[j])
                                    }
                                }) == Some(true))
                    });
                    if !old_is_better {
                        next[new_value] = Some(candidate);
                    }
                }
            }
        }
        dp = next;
    }

    dp[usize::from(amount)]
}

    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid = !product.id.is_empty()
                && product.id.len() <= 8
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }

            if seen.iter().any(|id| *id == product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);

            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }

        let next = self.credit.checked_add(coin).ok_or(Error::CreditLimit)?;
        if next > 100 {
            return Err(Error::CreditLimit);
        }

        self.escrow.push(coin);
        self.credit = next;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let product = self
            .products
            .iter()
            .find(|product| product.id == id)
            .ok_or_else(|| Error::UnknownProduct { id: id.to_owned() })?;

        if self.selected.as_deref() == Some(product.id.as_str()) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(product.id.clone());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = self.selected.as_ref().ok_or(Error::NoSelection)?;
        let product = self
            .products
            .iter()
            .find(|product| product.id == *selected_id)
            .expect("selected product is always in the catalog");

        let id = product.id.clone();
        let price = product.price;
        let credit = self.credit;

        if credit < price {
            return Err(Error::Underpaid { price, credit });
        }

        let amount = credit - price;
        let change = choose_change(amount, self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for (i, denom) in DENOMS.iter().enumerate() {
            let count = change.iter().filter(|coin| *coin == denom).count() as u16;
            proposed[i] -= count;
        }

        for coin in &self.escrow {
            let i = DENOMS.iter().position(|d| d == coin).unwrap();
            let next = proposed[i] + 1;
            if next > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
            proposed[i] = next;
        }

        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}

fn choose_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut greedy_counts = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / DENOMS[i]);
        greedy_counts[i] = take;
        remaining -= take * DENOMS[i];
    }
    if remaining == 0 {
        return Some(counts_to_descending(greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    search_change(0, amount, tubes, [0; 4], &mut best);
    best.map(counts_to_descending)
}

fn search_change(
    index: usize,
    remaining: u16,
    tubes: [u16; 4],
    counts: [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index == 4 {
        if remaining != 0 {
            return;
        }
        let total: u32 = counts.iter().map(|&n| n as u32).sum();
        if best
            .as_ref()
            .map(|old| {
                let old_total: u32 = old.iter().map(|&n| n as u32).sum();
                total < old_total || (total == old_total && lex_greater_desc(&counts, old))
            })
            .unwrap_or(true)
        {
            *best = Some(counts);
        }
        return;
    }

    let denom = DENOMS[3 - index];
    let tube_index = 3 - index;
    let max_take = tubes[tube_index].min(remaining / denom);
    for take in 0..=max_take {
        let mut next_counts = counts;
        next_counts[tube_index] = take;
        search_change(
            index + 1,
            remaining - take * denom,
            tubes,
            next_counts,
            best,
        );
    }
}

fn lex_greater_desc(a: &[u16; 4], b: &[u16; 4]) -> bool {
    for i in (0..4).rev() {
        if a[i] != b[i] {
            return a[i] > b[i];
        }
    }
    false
}

fn counts_to_descending(counts: [u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            result.push(DENOMS[i]);
        }
    }
    result
}

use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: HashMap<String, u16>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut catalog = HashMap::new();
        for product in products {
            let valid = (1..=8).contains(&product.id.len())
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid {
                return Err(Error::BadId { id: product.id });
            }
            if catalog.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id });
            }
            catalog.insert(product.id, product.price);
        }

        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            products: catalog,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if self.credit + coin > 100 {
            return Err(Error::CreditLimit);
        }
        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(_) = self.products.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let price = self.products[id.as_str()];
        if self.credit < price {
            return Err(Error::Underpaid { price, credit: self.credit });
        }

        let amount = self.credit - price;
        let mut counts = [0u16; 4];
        let mut remainder = amount;
        for i in (0..4).rev() {
            let take = self.tubes[i].min(remainder / DENOMS[i]);
            counts[i] = take;
            remainder -= take * DENOMS[i];
        }

        if remainder != 0 {
            counts = match best_change(amount, self.tubes) {
                Some(c) => c,
                None => return Err(Error::NoChange { amount }),
            };
        }

        let mut proposed = self.tubes;
        let mut change = Vec::new();
        for i in 0..4 {
            proposed[i] -= counts[i];
        }
        for i in (0..4).rev() {
            for _ in 0..counts[i] {
                change.push(DENOMS[i]);
            }
        }

        for &coin in &self.escrow {
            let i = denomination_index(coin).expect("escrow only contains accepted coins");
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        let receipt = Receipt {
            id: id.clone(),
            price,
            change,
        };
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(coins)
    }
}

fn denomination_index(coin: u16) -> Option<usize> {
    DENOMS.iter().position(|&d| d == coin)
}

fn best_change(amount: u16, tubes: [u16; 4]) -> Option<[u16; 4]> {
    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u16::MAX;

    for n10 in 0..=tubes[3].min(amount / 10) {
        for n4 in 0..=tubes[2].min(amount / 4) {
            for n3 in 0..=tubes[1].min(amount / 3) {
                let used = n10 * 10 + n4 * 4 + n3 * 3;
                if used > amount {
                    continue;
                }
                let rem = amount - used;
                if rem > tubes[0] {
                    continue;
                }
                let candidate = [rem, n3, n4, n10];
                let count = rem + n3 + n4 + n10;
                if best.is_none()
                    || count < best_count
                    || (count == best_count && candidate[3..] > best.unwrap()[3..])
                {
                    best = Some(candidate);
                    best_count = count;
                }
            }
        }
    }
    best
}
