pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = match input {
        "@hourly" => Some("0 * * * *"),
        "@daily" => Some("0 0 * * *"),
        "@weekly" => Some("0 0 * * 0"),
        "@monthly" => Some("0 0 1 * *"),
        "@yearly" => Some("0 0 1 1 *"),
        _ => None,
    };
    if input.as_bytes().first() == Some(&b'@') {
        if let Some(s) = expanded { return parse_normal(s); }
        return Err(ParseError::UnknownSpecial);
    }
    parse_normal(input)
}

fn parse_normal(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 { return Err(ParseError::FieldCount { found: pieces.len() }); }
    let fields = [Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek];
    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|i| vec![false; (field_max(fields[i]) - field_min(fields[i]) + 1) as usize]);
    let mut flags = [false; 5];
    for i in 0..5 {
        let text = pieces[i];
        let f = fields[i];
        if text == "*" { flags[i] = true; }
        for (off, b) in text.bytes().enumerate() {
            if !(b.is_ascii_digit() || b.is_ascii_alphabetic() || matches!(b, b'*' | b'-' | b'/' | b',')) {
                return Err(ParseError::InvalidCharacter { field: f, offset: off });
            }
        }
        let mut pos = 0;
        for item in text.split(',') {
            if item.is_empty() { return Err(ParseError::EmptyItem { field: f, offset: pos }); }
            parse_item(item, pos, f, &mut sets[i])?;
            pos += item.len() + 1;
        }
    }
    Ok(Cron { sets, unrestricted_dom: flags[2], unrestricted_dow: flags[4] })
}

fn parse_item(item: &str, offset: usize, f: Field, set: &mut [bool]) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes.iter().enumerate().filter_map(|(i,b)| if *b == b'/' {Some(i)} else {None}).collect();
    let (base, step_part) = match slash_positions.as_slice() {
        [] => (item, None),
        [p] => (&item[..*p], Some(&item[p+1..])),
        _ => return Err(ParseError::InvalidSyntax { field: f, offset }),
    };
    if base.is_empty() || step_part == Some("") {
        return Err(ParseError::InvalidSyntax { field: f, offset });
    }
    let hyphens: Vec<usize> = base.bytes().enumerate().filter_map(|(i,b)| if b == b'-' {Some(i)} else {None}).collect();
    let is_star = base == "*";
    let (left, right) = match hyphens.as_slice() {
        [] if !is_star => (base, None),
        [] => ("*", None),
        [p] if !is_star => (&base[..*p], Some(&base[p+1..])),
        _ => return Err(ParseError::InvalidSyntax { field: f, offset }),
    };
    if left.is_empty() || right == Some("") || (is_star && right.is_some()) {
        return Err(ParseError::InvalidSyntax { field: f, offset });
    }
    let lval = if left == "*" { None } else { Some(parse_atom(left, offset, f)?) };
    let rval = if let Some(r) = right {
        Some(parse_atom(r, offset + (r.as_ptr() as usize - item.as_ptr() as usize), f)?)
    } else { None };
    if let (Some(a), Some(b)) = (lval, rval) {
        if a > b { return Err(ParseError::ReversedRange { field: f, offset: offset + (right.unwrap().as_ptr() as usize - item.as_ptr() as usize) }); }
    }
    let step = if let Some(s) = step_part {
        let so = offset + (s.as_ptr() as usize - item.as_ptr() as usize);
        let n = parse_decimal(s, so, f, false)?;
        let width = (field_max(f) - field_min(f) + 1) as u32;
        if n == 0 || n > width { return Err(ParseError::BadStep { field: f, offset: so }); }
        n
    } else { 1 };
    let min = field_min(f) as u32;
    let max = field_max(f) as u32;
    let (start, end) = if is_star { (min, max) } else if let Some(a) = lval {
        (a, if let Some(b) = rval { b } else if step_part.is_some() { max } else { a })
    } else { (min, max) };
    let mut v = start;
    while v <= end {
        set[(v - min) as usize] = true;
        match v.checked_add(step) { Some(n) => v = n, None => break }
    }
    Ok(())
}

fn parse_atom(s: &str, offset: usize, f: Field) -> Result<u32, ParseError> {
    if s.bytes().all(|b| b.is_ascii_digit()) {
        return parse_decimal(s, offset, f, true);
    }
    let upper = s.to_ascii_uppercase();
    let names: &[&str] = match f {
        Field::Month => &["JAN","FEB","MAR","APR","MAY","JUN","JUL","AUG","SEP","OCT","NOV","DEC"],
        Field::DayOfWeek => &["SUN","MON","TUE","WED","THU","FRI","SAT"],
        _ => return Err(ParseError::InvalidSyntax { field: f, offset: offset - (offset % 1) }),
    };
    names.iter().position(|n| *n == upper).map(|i| {
        if f == Field::Month { i as u32 + 1 } else { i as u32 }
    }).ok_or(ParseError::InvalidSyntax { field: f, offset })
}

fn parse_decimal(s: &str, offset: usize, f: Field, domain: bool) -> Result<u32, ParseError> {
    let b = s.as_bytes();
    if b.is_empty() || !b.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field: f, offset });
    }
    if b.len() > 1 && b[0] == b'0' {
        return Err(ParseError::LeadingZero { field: f, offset });
    }
    let mut n = 0u32;
    for c in b {
        n = n.checked_mul(10).and_then(|x| x.checked_add((c - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field: f, offset })?;
    }
    if domain && (n < field_min(f) as u32 || n > field_max(f) as u32) {
        return Err(ParseError::OutOfRange { field: f, offset });
    }
    Ok(n)
}

fn field_index(f: Field) -> usize {
    match f { Field::Minute => 0, Field::Hour => 1, Field::DayOfMonth => 2, Field::Month => 3, Field::DayOfWeek => 4 }
}
fn field_min(f: Field) -> u8 { match f { Field::Minute | Field::Hour | Field::DayOfWeek => 0, Field::DayOfMonth | Field::Month => 1 } }
fn field_max(f: Field) -> u8 { match f { Field::Minute => 59, Field::Hour => 23, Field::DayOfMonth => 31, Field::Month => 12, Field::DayOfWeek => 6 } }

fn validate(x: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&x.year) { return Err(DateError::InvalidDateTime { component: DatePart::Year }); }
    if !(1..=12).contains(&x.month) { return Err(DateError::InvalidDateTime { component: DatePart::Month }); }
    if x.day == 0 || x.day > days_in_month(x.year, x.month) { return Err(DateError::InvalidDateTime { component: DatePart::Day }); }
    if x.hour > 23 { return Err(DateError::InvalidDateTime { component: DatePart::Hour }); }
    if x.minute > 59 { return Err(DateError::InvalidDateTime { component: DatePart::Minute }); }
    Ok(())
}
fn leap(y: u16) -> bool { y % 4 == 0 && (y % 100 != 0 || y % 400 == 0) }
fn days_in_month(y: u16, m: u8) -> u8 {
    match m { 2 => if leap(y) {29} else {28}, 4|6|9|11 => 30, _ => 31 }
}
fn weekday(y: u16, m: u8, d: u8) -> u8 {
    let mut days = 0u32;
    for year in 2000..y { days += if leap(year) {366} else {365}; }
    for month in 1..m { days += days_in_month(y, month) as u32; }
    days += (d - 1) as u32;
    ((6 + days) % 7) as u8
}
fn next_date(y: u16, m: u8, d: u8) -> (u16,u8,u8) {
    if d < days_in_month(y,m) { (y,m,d+1) }
    else if m < 12 { (y,m+1,1) }
    else { (y+1,1,1) }
}
fn previous_date(y: u16, m: u8, d: u8) -> (u16,u8,u8) {
    if d > 1 { (y,m,d-1) }
    else if m > 1 { (y,m-1,days_in_month(y,m-1)) }
    else { (y-1,12,31) }
}

fn parse_part(s:&str,o:usize,f:Field,bits:&mut u64)->Result<(),ParseError>{

    let (a,z)=if hy.is_empty(){(base,None)}else{let i=hy[0];(&base[..i],Some(&base[i+1..]))};
