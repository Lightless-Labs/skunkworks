- Field splitting uses only literal ASCII spaces and retains empty pieces; field count is checked first.
- Whole-field character validation precedes item parsing; items are then checked left to right, with syntax checked before numeric values.
- Numeric bounds are inclusive; weekday 7 is invalid, and steps may be as large as the full field width.
- A day field is unrestricted only when its original text is exactly `*`; when both day fields are restricted, the last actual day of the month requires both memberships.
- Window bounds and cap are checked in the specified order; windows are half-open, and `truncated` requires finding an extra match.
- Search advances by dates and selected times, including across the full supported range, rather than scanning every minute.

### Solution A

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    sets: [Vec<bool>; 5],
    unrestricted_dom: bool,
    unrestricted_dow: bool,
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = field_index(field);
        self.sets[i]
            .iter()
            .enumerate()
            .filter_map(|(n, yes)| if *yes { Some((n + field_min(field)) as u8) } else { None })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.has(Field::Minute, at.minute)
            || !self.has(Field::Hour, at.hour)
            || !self.has(Field::Month, at.month)
        {
            return false;
        }
        let dom = self.has(Field::DayOfMonth, at.day);
        let dow = self.has(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == days_in_month(at.year, at.month) {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }

    fn has(&self, f: Field, value: u8) -> bool {
        let min = field_min(f) as usize;
        self.sets[field_index(f)]
            .get(value as usize - min)
            .copied()
            .unwrap_or(false)
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        let mut date = (after.year, after.month, after.day);
        let max_date = (2399u16, 12u8, 31u8);
        loop {
            if self.has(Field::Month, date.1) {
                let dt = CivilDateTime { year: date.0, month: date.1, day: date.2, hour: 0, minute: 0 };
                if self.day_matches(dt) {
                    for h in self.allowed(Field::Hour) {
                        for m in self.allowed(Field::Minute) {
                            let candidate = CivilDateTime { hour: h, minute: m, ..dt };
                            if candidate > after {
                                return Ok(Some(candidate));
                            }
                        }
                    }
                }
            }
            if date == max_date { break; }
            date = next_date(date.0, date.1, date.2);
        }
        Ok(None)
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        let mut date = (before.year, before.month, before.day);
        loop {
            if self.has(Field::Month, date.1) {
                let dt = CivilDateTime { year: date.0, month: date.1, day: date.2, hour: 0, minute: 0 };
                if self.day_matches(dt) {
                    for h in self.allowed(Field::Hour).into_iter().rev() {
                        for m in self.allowed(Field::Minute).into_iter().rev() {
                            let candidate = CivilDateTime { hour: h, minute: m, ..dt };
                            if candidate < before {
                                return Ok(Some(candidate));
                            }
                        }
                    }
                }
            }
            if date == (2000, 1, 1) { break; }
            date = previous_date(date.0, date.1, date.2);
        }
        Ok(None)
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e { DateError::InvalidDateTime { component } => WindowError::InvalidStart { component } })?;
        validate(end).map_err(|e| match e { DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component } })?;
        if start > end { return Err(WindowError::ReversedWindow); }
        if cap == 0 { return Err(WindowError::ZeroCap); }
        let mut fires = Vec::new();
        let mut cursor = start;
        while let Some(hit) = self.next_fire(cursor).expect("validated cursor") {
            if hit >= end { break; }
            if fires.len() == cap {
                return Ok(FireWindow { fires, truncated: true });
            }
            fires.push(hit);
            cursor = hit;
        }
        Ok(FireWindow { fires, truncated: false })
    }

    pub fn canonical(&self) -> String {
        let fields = [
            self.format_field(Field::Minute, false),
            self.format_field(Field::Hour, false),
            self.format_field(Field::DayOfMonth, self.unrestricted_dom),
            self.format_field(Field::Month, false),
            self.format_field(Field::DayOfWeek, self.unrestricted_dow),
        ];
        fields.join(" ")
    }

    fn format_field(&self, f: Field, unrestricted: bool) -> String {
        if unrestricted { return "*".to_string(); }
        let vals = self.allowed(f);
        let width = field_max(f) - field_min(f) + 1;
        if vals.len() == width as usize {
            if matches!(f, Field::DayOfMonth | Field::DayOfWeek) { return "*/1".to_string(); }
            return "*".to_string();
        }
        vals.iter().map(|v| v.to_string()).collect::<Vec<_>>().join(",")
    }

    fn day_matches(&self, dt: CivilDateTime) -> bool {
        let dom = self.has(Field::DayOfMonth, dt.day);
        let dow = self.has(Field::DayOfWeek, weekday(dt.year, dt.month, dt.day));
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) if dt.day == days_in_month(dt.year, dt.month) => dom && dow,
            (false, false) => dom || dow,
        }
    }
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = match input {
        "@hourly" => Some("0 * * * *"),
        "@daily" => Some("0 0 * * *"),
        "@weekly" => Some("0 0 * * 0"),
        "@monthly" => Some("0 0 1 * *"),
        "@yearly" => Some("0 0 1 1 *"),
        _ => None,
    };
    if input.as_bytes().first() == Some(&b'@') {
        if let Some(s) = expanded { return parse_normal(s); }
        return Err(ParseError::UnknownSpecial);
    }
    parse_normal(input)
}

fn parse_normal(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 { return Err(ParseError::FieldCount { found: pieces.len() }); }
    let fields = [Field::Minute, Field::Hour, Field::DayOfMonth, Field::Month, Field::DayOfWeek];
    let mut sets: [Vec<bool>; 5] = std::array::from_fn(|i| vec![false; (field_max(fields[i]) - field_min(fields[i]) + 1) as usize]);
    let mut flags = [false; 5];
    for i in 0..5 {
        let text = pieces[i];
        let f = fields[i];
        if text == "*" { flags[i] = true; }
        for (off, b) in text.bytes().enumerate() {
            if !(b.is_ascii_digit() || b.is_ascii_alphabetic() || matches!(b, b'*' | b'-' | b'/' | b',')) {
                return Err(ParseError::InvalidCharacter { field: f, offset: off });
            }
        }
        let mut pos = 0;
        for item in text.split(',') {
            if item.is_empty() { return Err(ParseError::EmptyItem { field: f, offset: pos }); }
            parse_item(item, pos, f, &mut sets[i])?;
            pos += item.len() + 1;
        }
    }
    Ok(Cron { sets, unrestricted_dom: flags[2], unrestricted_dow: flags[4] })
}

fn parse_item(item: &str, offset: usize, f: Field, set: &mut [bool]) -> Result<(), ParseError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes.iter().enumerate().filter_map(|(i,b)| if *b == b'/' {Some(i)} else {None}).collect();
    let (base, step_part) = match slash_positions.as_slice() {
        [] => (item, None),
        [p] => (&item[..*p], Some(&item[p+1..])),
        _ => return Err(ParseError::InvalidSyntax { field: f, offset }),
    };
    if base.is_empty() || step_part == Some("") {
        return Err(ParseError::InvalidSyntax { field: f, offset });
    }
    let hyphens: Vec<usize> = base.bytes().enumerate().filter_map(|(i,b)| if b == b'-' {Some(i)} else {None}).collect();
    let is_star = base == "*";
    let (left, right) = match hyphens.as_slice() {
        [] if !is_star => (base, None),
        [] => ("*", None),
        [p] if !is_star => (&base[..*p], Some(&base[p+1..])),
        _ => return Err(ParseError::InvalidSyntax { field: f, offset }),
    };
    if left.is_empty() || right == Some("") || (is_star && right.is_some()) {
        return Err(ParseError::InvalidSyntax { field: f, offset });
    }
    let lval = if left == "*" { None } else { Some(parse_atom(left, offset, f)?) };
    let rval = if let Some(r) = right {
        Some(parse_atom(r, offset + (r.as_ptr() as usize - item.as_ptr() as usize), f)?)
    } else { None };
    if let (Some(a), Some(b)) = (lval, rval) {
        if a > b { return Err(ParseError::ReversedRange { field: f, offset: offset + (right.unwrap().as_ptr() as usize - item.as_ptr() as usize) }); }
    }
    let step = if let Some(s) = step_part {
        let so = offset + (s.as_ptr() as usize - item.as_ptr() as usize);
        let n = parse_decimal(s, so, f, false)?;
        let width = (field_max(f) - field_min(f) + 1) as u32;
        if n == 0 || n > width { return Err(ParseError::BadStep { field: f, offset: so }); }
        n
    } else { 1 };
    let min = field_min(f) as u32;
    let max = field_max(f) as u32;
    let (start, end) = if is_star { (min, max) } else if let Some(a) = lval {
        (a, if let Some(b) = rval { b } else if step_part.is_some() { max } else { a })
    } else { (min, max) };
    let mut v = start;
    while v <= end {
        set[(v - min) as usize] = true;
        match v.checked_add(step) { Some(n) => v = n, None => break }
    }
    Ok(())
}

fn parse_atom(s: &str, offset: usize, f: Field) -> Result<u32, ParseError> {
    if s.bytes().all(|b| b.is_ascii_digit()) {
        return parse_decimal(s, offset, f, true);
    }
    let upper = s.to_ascii_uppercase();
    let names: &[&str] = match f {
        Field::Month => &["JAN","FEB","MAR","APR","MAY","JUN","JUL","AUG","SEP","OCT","NOV","DEC"],
        Field::DayOfWeek => &["SUN","MON","TUE","WED","THU","FRI","SAT"],
        _ => return Err(ParseError::InvalidSyntax { field: f, offset: offset - (offset % 1) }),
    };
    names.iter().position(|n| *n == upper).map(|i| {
        if f == Field::Month { i as u32 + 1 } else { i as u32 }
    }).ok_or(ParseError::InvalidSyntax { field: f, offset })
}

fn parse_decimal(s: &str, offset: usize, f: Field, domain: bool) -> Result<u32, ParseError> {
    let b = s.as_bytes();
    if b.is_empty() || !b.iter().all(u8::is_ascii_digit) {
        return Err(ParseError::InvalidSyntax { field: f, offset });
    }
    if b.len() > 1 && b[0] == b'0' {
        return Err(ParseError::LeadingZero { field: f, offset });
    }
    let mut n = 0u32;
    for c in b {
        n = n.checked_mul(10).and_then(|x| x.checked_add((c - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field: f, offset })?;
    }
    if domain && (n < field_min(f) as u32 || n > field_max(f) as u32) {
        return Err(ParseError::OutOfRange { field: f, offset });
    }
    Ok(n)
}

fn field_index(f: Field) -> usize {
    match f { Field::Minute => 0, Field::Hour => 1, Field::DayOfMonth => 2, Field::Month => 3, Field::DayOfWeek => 4 }
}
fn field_min(f: Field) -> u8 { match f { Field::Minute | Field::Hour | Field::DayOfWeek => 0, Field::DayOfMonth | Field::Month => 1 } }
fn field_max(f: Field) -> u8 { match f { Field::Minute => 59, Field::Hour => 23, Field::DayOfMonth => 31, Field::Month => 12, Field::DayOfWeek => 6 } }

fn validate(x: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&x.year) { return Err(DateError::InvalidDateTime { component: DatePart::Year }); }
    if !(1..=12).contains(&x.month) { return Err(DateError::InvalidDateTime { component: DatePart::Month }); }
    if x.day == 0 || x.day > days_in_month(x.year, x.month) { return Err(DateError::InvalidDateTime { component: DatePart::Day }); }
    if x.hour > 23 { return Err(DateError::InvalidDateTime { component: DatePart::Hour }); }
    if x.minute > 59 { return Err(DateError::InvalidDateTime { component: DatePart::Minute }); }
    Ok(())
}
fn leap(y: u16) -> bool { y % 4 == 0 && (y % 100 != 0 || y % 400 == 0) }
fn days_in_month(y: u16, m: u8) -> u8 {
    match m { 2 => if leap(y) {29} else {28}, 4|6|9|11 => 30, _ => 31 }
}
fn weekday(y: u16, m: u8, d: u8) -> u8 {
    let mut days = 0u32;
    for year in 2000..y { days += if leap(year) {366} else {365}; }
    for month in 1..m { days += days_in_month(y, month) as u32; }
    days += (d - 1) as u32;
    ((6 + days) % 7) as u8
}
fn next_date(y: u16, m: u8, d: u8) -> (u16,u8,u8) {
    if d < days_in_month(y,m) { (y,m,d+1) }
    else if m < 12 { (y,m+1,1) }
    else { (y+1,1,1) }
}
fn previous_date(y: u16, m: u8, d: u8) -> (u16,u8,u8) {
    if d > 1 { (y,m,d-1) }
    else if m > 1 { (y,m-1,days_in_month(y,m-1)) }
    else { (y-1,12,31) }
}

}
fn parse_plain(s:&str)->Result<Cron,ParseError>{
    let p:Vec<_>=s.split(' ').collect();if p.len()!=5{return Err(ParseError::FieldCount{found:p.len()})}
    let fs=[Field::Minute,Field::Hour,Field::DayOfMonth,Field::Month,Field::DayOfWeek];
    let mut c=Cron{bits:[0;5],unrestricted_dom:p[2]=="*",unrestricted_dow:p[4]=="*"};
    for i in 0..5{
        let f=fs[i];
        for (o,b) in p[i].bytes().enumerate(){if !(b.is_ascii_alphanumeric()||matches!(b,b'*'|b'-'|b'/'|b',')){return Err(ParseError::InvalidCharacter{field:f,offset:o})}}
        let mut off=0;
        for it in p[i].split(','){if it.is_empty(){return Err(ParseError::EmptyItem{field:f,offset:off})} parse_part(it,off,f,&mut c.bits[i])?;off+=it.len()+1}
    }
    Ok(c)
}
fn parse_part(s:&str,o:usize,f:Field,bits:&mut u64)->Result<(),ParseError>{
    let sl:Vec<_>=s.match_indices('/').map(|(i,_)|i).collect();
    if sl.len()>1{return Err(ParseError::InvalidSyntax{field:f,offset:o})}
    let (base,step)=if sl.is_empty(){(s,None)}else{(&s[..sl[0]],Some(&s[sl[0]+1..]))};
    if base.is_empty()||step==Some(""){return Err(ParseError::InvalidSyntax{field:f,offset:o})}
    let hy:Vec<_>=base.match_indices('-').map(|(i,_)|i).collect();
    if hy.len()>1{return Err(ParseError::InvalidSyntax{field:f,offset:o})}
    let star=base=="*";
    let (a,z)=if hy.is_empty(){(base,None)}else{let i=hy[0];(&base[..i],Some(&base[i+1..]))};
    if a.is_empty()||z==Some("")||(star&&z.is_some()){return Err(ParseError::InvalidSyntax{field:f,offset:o})}
    let av=if star{None}else{Some(atom(a,o,f)?)};
    let zo=z.map(|v|atom(v,o+(v.as_ptr() as usize-s.as_ptr() as usize),f)).transpose()?;
    if let(Some(x),Some(y))=(av,zo){if x>y{return Err(ParseError::ReversedRange{field:f,offset:o+(z.unwrap().as_ptr() as usize-s.as_ptr() as usize)})}}
    let st=if let Some(v)=step{let so=o+(v.as_ptr() as usize-s.as_ptr() as usize);let n=decimal(v,so,f,false)?;if n==0||n>(max(f)-min(f)+1) as u32{return Err(ParseError::BadStep{field:f,offset:so})}n}else{1};
    let lo=min(f) as u32;let hi=max(f) as u32;
    let (start,end)=if star{(lo,hi)}else if let Some(x)=av{(x,if let Some(y)=zo{y}else if step.is_some(){hi}else{x})}else{(lo,hi)};
    let mut x=start;while x<=end{*bits|=1u64<<(x-lo);match x.checked_add(st){Some(n)=>x=n,None=>break}}
    Ok(())
}
fn atom(s:&str,o:usize,f:Field)->Result<u32,ParseError>{
    if s.bytes().all(|b|b.is_ascii_digit()){return decimal(s,o,f,true)}
    let u=s.to_ascii_uppercase();
    let names:&[&str]=match f{Field::Month=>&["JAN","FEB","MAR","APR","MAY","JUN","JUL","AUG","SEP","OCT","NOV","DEC"],Field::DayOfWeek=>&["SUN","MON","TUE","WED","THU","FRI","SAT"],_=>return Err(ParseError::InvalidSyntax{field:f,offset:o})};
    names.iter().position(|n|*n==u).map(|i|if f==Field::Month{i as u32+1}else{i as u32}).ok_or(ParseError::InvalidSyntax{field:f,offset:o})
}
fn decimal(s:&str,o:usize,f:Field,domain:bool)->Result<u32,ParseError>{
    let x=s.as_bytes();if x.is_empty()||!x.iter().all(u8::is_ascii_digit){return Err(ParseError::InvalidSyntax{field:f,offset:o})}
    if x.len()>1&&x[0]==b'0'{return Err(ParseError::LeadingZero{field:f,offset:o})}
    let mut n=0u32;for b in x{n=n.checked_mul(10).and_then(|v|v.checked_add((b-b'0') as u32)).ok_or(ParseError::NumberOverflow{field:f,offset:o})?}
    if domain&&(n<min(f) as u32||n>max(f) as u32){return Err(ParseError::OutOfRange{field:f,offset:o})}Ok(n)
}
fn idx(f:Field)->usize{match f{Field::Minute=>0,Field::Hour=>1,Field::DayOfMonth=>2,Field::Month=>3,Field::DayOfWeek=>4}}
fn min(f:Field)->u8{match f{Field::Minute|Field::Hour|Field::DayOfWeek=>0,Field::DayOfMonth|Field::Month=>1}}
fn max(f:Field)->u8{match f{Field::Minute=>59,Field::Hour=>23,Field::DayOfMonth=>31,Field::Month=>12,Field::DayOfWeek=>6}}
fn validate(x:CivilDateTime)->Result<(),DateError>{
    if !(2000..=2399).contains(&x.year){return Err(DateError::InvalidDateTime{component:DatePart::Year})}
    if !(1..=12).contains(&x.month){return Err(DateError::InvalidDateTime{component:DatePart::Month})}
    if x.day==0||x.day>dim(x.year,x.month){return Err(DateError::InvalidDateTime{component:DatePart::Day})}
    if x.hour>23{return Err(DateError::InvalidDateTime{component:DatePart::Hour})}
    if x.minute>59{return Err(DateError::InvalidDateTime{component:DatePart::Minute})}Ok(())
}
fn leap(y:u16)->bool{y%4==0&&(y%100!=0||y%400==0)}
fn dim(y:u16,m:u8)->u8{match m{2=>if leap(y){29}else{28},4|6|9|11=>30,_=>31}}
fn weekday(y:u16,m:u8,d:u8)->u8{let mut n=0u32;for yy in 2000..y{n+=if leap(yy){366}else{365}}for mm in 1..m{n+=dim(y,mm) as u32}((6+n+(d as u32)-1)%7) as u8}
