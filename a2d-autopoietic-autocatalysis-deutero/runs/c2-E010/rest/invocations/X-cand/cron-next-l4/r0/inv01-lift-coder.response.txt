- Field splitting uses only literal spaces and retains empty fields; lexical validation scans the whole field before item parsing.
- Syntax is validated before numeric values; base numbers are checked before step numbers, and leading zeroes precede overflow and range checks.
- Only an exact `*` makes a day field unrestricted; `*/1` remains restricted, and two restricted day fields use OR except on the month’s actual last day, where they use AND.
- Date arguments are validated component by component before schedule or window checks; windows are half-open, and zero cap is an error after date and ordering validation.
- Search advances through calendar dates and selected times rather than scanning every minute; it respects the inclusive 2000–2399 range and strict next/previous boundaries.
- Canonical output uses ascending selected values, with `*` for full non-day domains and `*/1` for a full restricted day domain.

fn parse_number(s: &str, field: Field, offset: usize) -> Result<u32, ParseError> {

fn parse_field(
    text: &str,
    field: Field,
) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit()
            || b.is_ascii_alphabetic()
            || matches!(b, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let mut selected = vec![false; (bounds(field).1 + 1) as usize];
    let (min, max) = bounds(field);
    let width = max - min + 1;

    let mut start = 0;
    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map(|p| start + p)
            .unwrap_or(bytes.len());
        if end == start {
            return Err(ParseError::EmptyItem { field, offset: start });
        }
        let item = &text[start..end];
        let slash_positions: Vec<usize> = item.match_indices('/').map(|(p, _)| p).collect();
        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }
        let (base, step_text, step_offset) = if let Some(&slash) = slash_positions.first() {
            (&item[..slash], Some(&item[slash + 1..]), start + slash + 1)
        } else {
            (item, None, 0)
        };

        let hyphens: Vec<usize> = base.match_indices('-').map(|(p, _)| p).collect();
        if base == "*" {
            if !hyphens.is_empty() {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
        } else if hyphens.len() > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            if base[..h].is_empty() || base[h + 1..].is_empty() {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
            if base[..h] == "*" || base[h + 1..] == "*" {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
            let _ = endpoint(&base[..h], field, start)?;
            let _ = endpoint(&base[h + 1..], field, start + h + 1)?;
        } else {
            let _ = endpoint(base, field, start)?;
        }
        if let Some(st) = step_text {
            if st.is_empty() {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
            if !st.bytes().all(|b| b.is_ascii_digit()) {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
        }

        let (a, b, a_num, b_num, is_star) = if base == "*" {
            (min, max, false, false, true)
        } else if let Some(h) = hyphens.first().copied() {
            let (a, an) = endpoint(&base[..h], field, start)?;
            let (b, bn) = endpoint(&base[h + 1..], field, start + h + 1)?;
            (a, b, an, bn, false)
        } else {
            let (a, an) = endpoint(base, field, start)?;
            (a, a, an, an, false)
        };

        if a_num {
            if a < min || a > max {
                return Err(ParseError::OutOfRange { field, offset: start });
            }
        }
        if b_num && hyphens.len() == 1 {
            let h = hyphens[0];
            if b < min || b > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: start + h + 1,
                });
            }
        }
        if hyphens.len() == 1 && a > b {
            let h = hyphens[0];
            return Err(ParseError::ReversedRange {
                field,
                offset: start + h + 1,
            });
        }

        let step = if let Some(st) = step_text {
            let v = parse_number(st, field, step_offset)?;
            if v == 0 || v > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: step_offset,
                });
            }
            v
        } else {
            1
        };

        let (lo, hi) = if step_text.is_some() {
            if is_star || hyphens.len() == 1 {
                (a, b)
            } else {
                (a, max)
            }
        } else {
            (a, b)
        };
        let mut v = lo;
        while v <= hi {
            selected[v as usize] = true;
            match v.checked_add(step) {
                Some(next) => v = next,
                None => break,
            }
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
        if start > bytes.len() {
            break;
        }
    }

    let values = (min..=max)
        .filter(|&v| selected[v as usize])
        .map(|v| v as u8)
        .collect();
    Ok((values, text == "*"))
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = match input {
        "@hourly" => Some("0 * * * *"),
        "@daily" => Some("0 0 * * *"),
        "@weekly" => Some("0 0 * * 0"),
        "@monthly" => Some("0 0 1 * *"),
        "@yearly" => Some("0 0 1 1 *"),
        _ => None,
    };
    if input.as_bytes().first() == Some(&b'@') {
        if let Some(s) = expanded {
            return parse_cron_fields(s);
        }
        return Err(ParseError::UnknownSpecial);
    }
    parse_cron_fields(input)
}

fn parse_cron_fields(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }
    let mut sets: [Vec<u8>; 5] = std::array::from_fn(|_| Vec::new());
    let mut flags = [false; 5];
    for i in 0..5 {
        let (v, unrestricted) = parse_field(pieces[i], FIELDS[i])?;
        sets[i] = v;
        flags[i] = unrestricted;
    }
    Ok(Cron {
        sets,
        unrestricted_dom: flags[2],
        unrestricted_dow: flags[4],
    })
}

    fn matches_valid(&self, at: CivilDateTime) -> bool {

                let dom = contains(&self.sets[2], date.2);
                let dow = contains(&self.sets[4], weekday(date.0, date.1, date.2));

                    (true, true) => true,
                    (true, false) => dow,
                    (false, true) => dom,
                    (false, false) => {
                        if date.2 == month_len(date.0, date.1) {
                            dom && dow
                        } else {
                            dom || dow
                        }
                    }
                };
                if day_ok {
                    let min_hour = if first_date { after.hour } else { 0 };
                    for &hour in &self.sets[1] {
                        if hour < min_hour {
                            continue;
                        }
                        let min_minute = if first_date && hour == after.hour {
                            after.minute.saturating_add(1)
                        } else {
                            0
                        };

                            if minute < min_minute {
                                continue;
                            }
                            return Ok(Some(CivilDateTime {
                                year: date.0,
                                month: date.1,
                                day: date.2,
                                hour,
                                minute,
                            }));
                        }
                    }
                }
            }
            match next_date(date) {
                Some(d) => {
                    date = d;
                    first_date = false;
                }
                None => return Ok(None),
            }
        }
    }

        let mut date = (before.year, before.month, before.day);
        let mut first_date = true;
        loop {
            if contains(&self.sets[3], date.1) {
                let dom = contains(&self.sets[2], date.2);
                let dow = contains(&self.sets[4], weekday(date.0, date.1, date.2));
                let day_ok = match (self.unrestricted_dom, self.unrestricted_dow) {

                    (true, false) => dow,
                    (false, true) => dom,
                    (false, false) => {
                        if date.2 == month_len(date.0, date.1) { dom && dow } else { dom || dow }
                    }
                };
                if day_ok {
                    for &hour in self.sets[1].iter().rev() {
                        if first_date && hour > before.hour {
                            continue;
                        }
                        let max_minute = if first_date && hour == before.hour {
                            before.minute.saturating_sub(1)
                        } else {
                            59
                        };
                        for &minute in self.sets[0].iter().rev() {
                            if minute > max_minute {
                                continue;
                            }
                            return Ok(Some(CivilDateTime { year: date.0, month: date.1, day: date.2, hour, minute }));
                        }
                    }
                }
            }
            match prev_date(date) {
                Some(d) => { date = d; first_date = false; }
                None => return Ok(None),
            }
        }
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| WindowError::InvalidStart { component: component(e) })?;
        validate(end).map_err(|e| WindowError::InvalidEnd { component: component(e) })?;
        if start > end {
            return Err(WindowError::ReversedWindow);
        }
        if cap == 0 {
            return Err(WindowError::ZeroCap);
        }
        let mut fires = Vec::new();
        let mut cursor = start;
        loop {
            match self.next_inclusive(cursor) {
                Some(t) if t < end => {
                    if fires.len() == cap {
                        return Ok(FireWindow { fires, truncated: true });
                    }
                    fires.push(t);
                    if let Some(n) = next_minute(t) {
                        cursor = n;
                    } else {
                        break;
                    }
                }
                _ => break,
            }
        }
        Ok(FireWindow { fires, truncated: false })
    }

    fn next_inclusive(&self, at: CivilDateTime) -> Option<CivilDateTime> {
        if self.matches_valid(at) {
            Some(at)
        } else {
            self.next_fire(at).ok().flatten()
        }
    }

    pub fn canonical(&self) -> String {
        let mut out = Vec::new();
        for i in 0..5 {
            let field = FIELDS[i];
            let (min, max) = bounds(field);
            let full = self.sets[i].len() == (max - min + 1) as usize;
            let text = if i == 2 && self.unrestricted_dom {
                "*".to_string()
            } else if i == 4 && self.unrestricted_dow {
                "*".to_string()
            } else if full && i != 2 && i != 4 {
                "*".to_string()
            } else if full {
                "*/1".to_string()
            } else {
                self.sets[i].iter().map(|v| v.to_string()).collect::<Vec<_>>().join(",")
            };
            out.push(text);
        }
        out.join(" ")
    }

fn contains(v: &[u8], x: u8) -> bool {
    v.binary_search(&x).is_ok()
}

fn leap(y: u16) -> bool {
    y % 4 == 0 && (y % 100 != 0 || y % 400 == 0)
}

fn month_len(y: u16, m: u8) -> u8 {
    match m {
        2 => if leap(y) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(y: u16, m: u8, d: u8) -> u8 {
    let mut days = 0u32;
    for year in 2000..y {
        days += if leap(year) { 366 } else { 365 };
    }
    for month in 1..m {
        days += month_len(y, month) as u32;
    }
    days += d as u32 - 1;
    ((days + 6) % 7) as u8
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_len(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn next_date((y, m, d): (u16, u8, u8)) -> Option<(u16, u8, u8)> {
    if d < month_len(y, m) {
        Some((y, m, d + 1))
    } else if m < 12 {
        Some((y, m + 1, 1))
    } else if y < 2399 {
        Some((y + 1, 1, 1))
    } else {
        None
    }
}

fn prev_date((y, m, d): (u16, u8, u8)) -> Option<(u16, u8, u8)> {
    if d > 1 {
        Some((y, m, d - 1))
    } else if m > 1 {
        Some((y, m - 1, month_len(y, m - 1)))
    } else if y > 2000 {
        Some((y - 1, 12, 31))
    } else {
        None
    }
}

fn next_minute(t: CivilDateTime) -> Option<CivilDateTime> {
    if t.minute < 59 {
        Some(CivilDateTime { minute: t.minute + 1, ..t })
    } else if t.hour < 23 {
        Some(CivilDateTime { hour: t.hour + 1, minute: 0, ..t })
    } else if let Some((y, m, d)) = next_date((t.year, t.month, t.day)) {
        Some(CivilDateTime { year: y, month: m, day: d, hour: 0, minute: 0 })
    } else {
        None
    }
}

fn component(e: DateError) -> DatePart {
    match e {
        DateError::InvalidDateTime { component } => component,
    }
}

pub fn parse_cron(input:&str)->Result<Cron,ParseError>{
    if input.as_bytes().first()==Some(&b'@'){
        let s=match input {"@hourly"=>"0 * * * *","@daily"=>"0 0 * * *","@weekly"=>"0 0 * * 0","@monthly"=>"0 0 1 * *","@yearly"=>"0 0 1 1 *",_=>return Err(ParseError::UnknownSpecial)};
        return parse_fields(s);
    }
    parse_fields(input)
}
fn parse_fields(s:&str)->Result<Cron,ParseError>{
    let v:Vec<&str>=s.split(' ').collect();
    if v.len()!=5{return Err(ParseError::FieldCount{found:v.len()});}
    let mut sets:[Vec<u8>;5]=std::array::from_fn(|_|Vec::new());
    let mut flags=[false;5];
    for i in 0..5 {let(x,u)=field_parse(v[i],FIELDS[i])?;sets[i]=x;flags[i]=u;}
    Ok(Cron{sets,unrestricted_dom:flags[2],unrestricted_dow:flags[4]})
}
fn contains(v:&[u8],x:u8)->bool{v.binary_search(&x).is_ok()}
fn leap(y:u16)->bool{y%4==0&&(y%100!=0||y%400==0)}
fn ml(y:u16,m:u8)->u8{match m{2=>if leap(y){29}else{28},4|6|9|11=>30,_=>31}}
fn wd(y:u16,m:u8,d:u8)->u8{

    for mm in 1..m{n+=ml(y,mm) as u32;} n+=d as u32-1; ((n+6)%7) as u8
}
fn validate(t:CivilDateTime)->Result<(),DateError>{
    if !(2000..=2399).contains(&t.year){return Err(DateError::InvalidDateTime{component:DatePart::Year});}
    if !(1..=12).contains(&t.month){return Err(DateError::InvalidDateTime{component:DatePart::Month});}
    if t.day==0||t.day>ml(t.year,t.month){return Err(DateError::InvalidDateTime{component:DatePart::Day});}
    if t.hour>23{return Err(DateError::InvalidDateTime{component:DatePart::Hour});}
    if t.minute>59{return Err(DateError::InvalidDateTime{component:DatePart::Minute});}
    Ok(())
}
fn day_ok(c:&Cron,y:u16,m:u8,d:u8)->bool{
    let a=contains(&c.sets[2],d);let b=contains(&c.sets[4],wd(y,m,d));
    match(c.unrestricted_dom,c.unrestricted_dow){(true,true)=>true,(true,false)=>b,(false,true)=>a,(false,false)=>if d==ml(y,m){a&&b}else{a||b}}
}
fn nd((y,m,d):(u16,u8,u8))->Option<(u16,u8,u8)>{if d<ml(y,m){Some((y,m,d+1))}else if m<12{Some((y,m+1,1))}else if y<2399{Some((y+1,1,1))}else{None}}
