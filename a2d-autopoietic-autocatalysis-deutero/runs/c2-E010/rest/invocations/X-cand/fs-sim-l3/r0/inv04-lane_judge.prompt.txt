#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self { entries, cwd: "/".to_string() }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn validate_normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts: Vec<String> = if path.starts_with('/') {
            Vec::new()
        } else if self.cwd == "/" {
            Vec::new()
        } else {
            self.cwd[1..].split('/').map(str::to_string).collect()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_string()),
            }
        }
        Ok(Self::canonical(&parts))
    }

    fn parent(path: &str) -> String {
        match path.rfind('/') {
            Some(0) | None => "/".to_string(),
            Some(i) => path[..i].to_string(),
        }
    }

    fn basename(path: &str) -> &str {
        path.rsplit('/').next().unwrap_or("")
    }

    fn resolve_dir(&self, path: &str) -> Result<(), Error> {
        if path == "/" {
            return Ok(());
        }
        let mut current = String::new();
        let parts: Vec<&str> = path[1..].split('/').collect();
        for (i, part) in parts.iter().enumerate() {
            current.push('/');
            current.push_str(part);
            match self.entries.get(&current) {
                None => return Err(Error::Missing(current)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => {
                    if i + 1 < parts.len() {
                        return Err(Error::NotDir(current));
                    }
                    return Err(Error::NotDir(current));
                }
            }
        }
        Ok(())
    }

    fn resolve_parent(&self, path: &str) -> Result<(), Error> {
        let parent = Self::parent(path);
        self.resolve_dir(&parent)
    }

    fn is_within(path: &str, ancestor: &str) -> bool {
        if ancestor == "/" {
            return true;
        }
        path == ancestor
            || path.strip_prefix(ancestor).is_some_and(|rest| rest.starts_with('/'))
    }

    fn children(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter_map(|candidate| {
                if candidate == "/" || Self::parent(candidate) != path {
                    None
                } else {
                    Some(candidate.clone())
                }
            })
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.validate_normalize(path)?;
        self.resolve_parent(&path)?;
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.validate_normalize(path)?;
        self.resolve_parent(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => return Err(Error::IsDir(path)),
            Some(Entry::File(_)) | None => {}
        }
        self.entries.insert(path, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.validate_normalize(path)?;
        self.resolve_parent(&path)?;
        match self.entries.get(&path) {
            None => Err(Error::Missing(path)),
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.validate_normalize(path)?;
        self.resolve_dir(&path)?;
        let mut result = Vec::new();
        for child in self.children(&path) {
            let kind = match self.entries.get(&child).expect("child key exists") {
                Entry::Dir => Kind::Dir,
                Entry::File(_) => Kind::File,
            };
            result.push((Self::basename(&child).to_string(), kind));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.validate_normalize(path)?;
        self.resolve_dir(&path)?;
        self.cwd = path;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from = self.validate_normalize(from)?;
        let to = self.validate_normalize(to)?;
        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&from)?;
        let source = self.entries.get(&from).cloned().ok_or_else(|| Error::Missing(from.clone()))?;

        if matches!(source, Entry::Dir) && Self::is_within(&self.cwd, &from) {
            return Err(Error::Busy(from));
        }

        self.resolve_parent(&to)?;
        if matches!(self.entries.get(&to), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &to)
        {
            return Err(Error::Busy(to));
        }

        if from == to {
            return Ok(());
        }

        if matches!(source, Entry::Dir) && Self::is_within(&to, &from) {
            return Err(Error::Cycle(to));
        }

        let destination = self.entries.get(&to).cloned();
        if destination.is_some() && !overwrite {
            return Err(Error::Exists(to));
        }

        if let Some(dest) = &destination {
            match (&source, dest) {
                (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(to)),
                (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(to)),
                (Entry::Dir, Entry::Dir) if !self.children(&to).is_empty() => {
                    return Err(Error::NotEmpty(to));
                }
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self.entries
            .iter()
            .filter_map(|(key, entry)| {
                if Self::is_within(key, &from) {
                    let suffix = &key[from.len()..];
                    Some((format!("{to}{suffix}"), entry.clone()))
                } else {
                    None
                }
            })
            .collect();

        if destination.is_some() {
            self.entries.retain(|key, _| !Self::is_within(key, &to));
        }
        self.entries.retain(|key, _| !Self::is_within(key, &from));
        for (key, entry) in moving {
            self.entries.insert(key, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.validate_normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }
        self.resolve_parent(&path)?;
        let entry = self.entries.get(&path).ok_or_else(|| Error::Missing(path.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &path) {
            return Err(Error::Busy(path));
        }

        if matches!(entry, Entry::Dir) && !recursive && !self.children(&path).is_empty() {
            return Err(Error::NotEmpty(path));
        }

        self.entries.retain(|key, _| !Self::is_within(key, &path));
        Ok(())
    }
}

pub enum Kind { Dir, File }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Node {
    Dir,
    File(Vec<u8>),
}

impl Node {
    fn kind(&self) -> Kind {
        match self {
            Node::Dir => Kind::Dir,
            Node::File(_) => Kind::File,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Fs {
    nodes: std::collections::BTreeMap<String, Node>,
    cwd: Vec<String>,
}

impl Default for Fs {
    fn default() -> Self { Self::new() }
}

impl Fs {
    pub fn new() -> Self {
        let mut nodes = std::collections::BTreeMap::new();
        nodes.insert("/".to_owned(), Node::Dir);
        Self { nodes, cwd: Vec::new() }
    }

    fn path_from(parts: &[String]) -> String {
        if parts.is_empty() { "/".to_owned() }
        else { format!("/{}", parts.join("/")) }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() { return Err(Error::InvalidPath(0)); }

        let mut parts = if path.starts_with('/') { Vec::new() } else { self.cwd.clone() };
        for c in path.split('/') {
            match c {
                "" | "." => {}
                ".." => { parts.pop(); }
                x => parts.push(x.to_owned()),
            }
        }
        Ok(Self::path_from(&parts))
    }

    fn comps(path: &str) -> Vec<&str> {
        if path == "/" { Vec::new() } else { path[1..].split('/').collect() }
    }

    fn parent(path: &str) -> String {
        path.rfind('/').map(|i| if i == 0 { "/".into() } else { path[..i].into() }).unwrap_or_else(|| "/".into())
    }

    fn resolve_parent(&self, path: &str) -> Result<(), Error> {
        let parent = Self::parent(path);
        let mut prefix = String::new();
        for c in Self::comps(&parent) {
            prefix.push('/');
            prefix.push_str(c);
            match self.nodes.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Node::Dir) => {}
                Some(Node::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn within(dir: &str, path: &str) -> bool {
        dir == "/" || path == dir || path.strip_prefix(dir).is_some_and(|s| s.starts_with('/'))
    }

    fn children(&self, path: &str) -> bool {
        self.nodes.keys().any(|p| p != path && Self::within(path, p))
    }

    fn erase_tree(&mut self, path: &str) {
        self.nodes.retain(|p, _| p != path && !Self::within(path, p));
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let p = self.normalize(path)?;
        if p == "/" { return Err(Error::Exists(p)); }
        self.resolve_parent(&p)?;
        let parent = Self::parent(&p);
        if !matches!(self.nodes.get(&parent), Some(Node::Dir)) {
            return match self.nodes.get(&parent) {
                Some(_) => Err(Error::NotDir(parent)),
                None => Err(Error::Missing(parent)),
            };
        }
        if self.nodes.contains_key(&p) { return Err(Error::Exists(p)); }
        self.nodes.insert(p, Node::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let p = self.normalize(path)?;
        if p == "/" { return Err(Error::IsDir(p)); }
        self.resolve_parent(&p)?;
        let parent = Self::parent(&p);
        if !matches!(self.nodes.get(&parent), Some(Node::Dir)) {
            return match self.nodes.get(&parent) {
                Some(_) => Err(Error::NotDir(parent)),
                None => Err(Error::Missing(parent)),
            };
        }
        match self.nodes.get_mut(&p) {
            Some(Node::Dir) => Err(Error::IsDir(p)),
            Some(Node::File(bytes)) => { *bytes = data.to_vec(); Ok(()) }
            None => { self.nodes.insert(p, Node::File(data.to_vec())); Ok(()) }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let p = self.normalize(path)?;
        if p == "/" { return Err(Error::IsDir(p)); }
        self.resolve_parent(&p)?;
        match self.nodes.get(&p) {
            None => Err(Error::Missing(p)),
            Some(Node::Dir) => Err(Error::IsDir(p)),
            Some(Node::File(v)) => Ok(v.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let p = self.normalize(path)?;
        self.resolve_parent(&p)?;
        match self.nodes.get(&p) {
            None => return Err(Error::Missing(p)),
            Some(Node::File(_)) => return Err(Error::NotDir(p)),
            Some(Node::Dir) => {}
        }
        let mut out = Vec::new();
        for (q, n) in &self.nodes {
            if q == &p { continue; }
            let name = if p == "/" {
                match q.strip_prefix('/') { Some(x) => x, None => continue }
            } else {
                match q.strip_prefix(&(p.clone() + "/")) { Some(x) => x, None => continue }
            };
            if !name.is_empty() && !name.contains('/') {
                out.push((name.to_owned(), n.kind()));
            }
        }
        out.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(out)
    }

    pub fn cwd(&self) -> String { Self::path_from(&self.cwd) }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let p = self.normalize(path)?;
        self.resolve_parent(&p)?;
        match self.nodes.get(&p) {
            None => Err(Error::Missing(p)),
            Some(Node::File(_)) => Err(Error::NotDir(p)),
            Some(Node::Dir) => {
                self.cwd = Self::comps(&p).into_iter().map(str::to_owned).collect();
                Ok(())
            }
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let s = self.normalize(from)?;
        let d = self.normalize(to)?;
        if s == "/" || d == "/" { return Err(Error::Root); }

        self.resolve_parent(&s)?;
        let source = match self.nodes.get(&s) {
            None => return Err(Error::Missing(s)),
            Some(n) => n.clone(),
        };
        if matches!(source, Node::Dir) && Self::within(&s, &self.cwd()) {
            return Err(Error::Busy(s));
        }

        self.resolve_parent(&d)?;
        let dp = Self::parent(&d);
        match self.nodes.get(&dp) {
            None => return Err(Error::Missing(dp)),
            Some(Node::File(_)) => return Err(Error::NotDir(dp)),
            Some(Node::Dir) => {}
        }
        if matches!(self.nodes.get(&d), Some(Node::Dir)) && Self::within(&d, &self.cwd()) {
            return Err(Error::Busy(d));
        }

                (Node::Dir, Node::File(_)) => return Err(Error::NotDir(d)),
                (Node::File(_), Node::Dir) => return Err(Error::IsDir(d)),
                (Node::Dir, Node::Dir) if self.children(&d) => return Err(Error::NotEmpty(d)),

            .map(|(p, n)| (p.clone(), n.clone()))
            .collect();
        if dest.is_some() { self.erase_tree(&d); }
        self.erase_tree(&s);
        for (old, node) in moving {
            let new = if old == s { d.clone() } else { format!("{}{}", d, &old[s.len()..]) };
            self.nodes.insert(new, node);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let p = self.normalize(path)?;
        if p == "/" { return Err(Error::Root); }
        self.resolve_parent(&p)?;
        match self.nodes.get(&p) {
            None => return Err(Error::Missing(p)),
            Some(_) => {}
        }
        if matches!(self.nodes.get(&p), Some(Node::Dir)) && Self::within(&p, &self.cwd()) {
            return Err(Error::Busy(p));
        }
        if matches!(self.nodes.get(&p), Some(Node::Dir)) && self.children(&p) && !recursive {
            return Err(Error::NotEmpty(p));
        }
        if recursive { self.erase_tree(&p); } else { self.nodes.remove(&p); }
        Ok(())
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn validate_and_normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts: Vec<&str> = if path.starts_with('/') {
            Vec::new()
        } else if self.cwd == "/" {
            Vec::new()
        } else {
            self.cwd[1..].split('/').collect()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other),
            }
        }

        if parts.is_empty() {
            Ok("/".to_owned())
        } else {
            let mut result = String::new();
            for part in parts {
                result.push('/');
                result.push_str(part);
            }
            Ok(result)
        }
    }

    fn parent(path: &str) -> (&str, &str) {
        if path == "/" {
            return ("/", "");
        }
        match path.rfind('/') {
            Some(0) => ("/", &path[1..]),
            Some(i) => (&path[..i], &path[i + 1..]),
            None => ("/", path),
        }
    }

    fn ancestors(path: &str) -> Vec<String> {
        let mut result = vec!["/".to_owned()];
        if path == "/" {
            return result;
        }
        let mut current = String::new();
        for component in path[1..].split('/') {
            current.push('/');
            current.push_str(component);
            result.push(current.clone());
        }
        result
    }

    fn resolve_parent(&self, path: &str) -> Result<(), Error> {
        let (parent, _) = Self::parent(path);
        for prefix in Self::ancestors(parent) {
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn resolve_entry(&self, path: &str) -> Result<&Entry, Error> {
        let components = Self::ancestors(path);
        for (index, prefix) in components.iter().enumerate() {
            match self.entries.get(prefix) {
                None => return Err(Error::Missing(prefix.clone())),
                Some(entry) if index + 1 < components.len() && !matches!(entry, Entry::Dir) => {
                    return Err(Error::NotDir(prefix.clone()));
                }
                Some(entry) => {
                    if index + 1 == components.len() {
                        return Ok(entry);
                    }
                }
            }
        }
        self.entries.get("/").ok_or_else(|| Error::Missing("/".to_owned()))
    }

    fn contains_path(dir: &str, path: &str) -> bool {
        if dir == "/" {
            return true;
        }
        path == dir || path.strip_prefix(dir).is_some_and(|tail| tail.starts_with('/'))
    }

    fn subtree_keys(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|key| Self::contains_path(path, key))
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.validate_and_normalize(path)?;
        self.resolve_parent(&canonical)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }
        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.validate_and_normalize(path)?;
        self.resolve_parent(&canonical)?;
        match self.entries.get(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.validate_and_normalize(path)?;
        match self.resolve_entry(&canonical)? {
            Entry::Dir => Err(Error::IsDir(canonical)),
            Entry::File(bytes) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.validate_and_normalize(path)?;
        match self.resolve_entry(&canonical)? {
            Entry::File(_) => return Err(Error::NotDir(canonical)),
            Entry::Dir => {}
        }
        let mut children = Vec::new();
        for (key, entry) in &self.entries {
            if key == "/" {
                continue;
            }
            let (parent, name) = Self::parent(key);
            if parent == canonical {
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                children.push((name.to_owned(), kind));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.validate_and_normalize(path)?;
        match self.resolve_entry(&canonical)? {
            Entry::Dir => {
                self.cwd = canonical;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(canonical)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.validate_and_normalize(from)?;
        let destination = self.validate_and_normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&source)?;
        let source_entry = match self.entries.get(&source) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(source.clone())),
        };

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        self.resolve_parent(&destination)?;

        if let Some(Entry::Dir) = self.entries.get(&destination) {
            if Self::contains_path(&destination, &self.cwd) {
                return Err(Error::Busy(destination));
            }
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let dest_entry = self.entries.get(&destination).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref dest) = dest_entry {
            match (&source_entry, dest) {
                (Entry::Dir, Entry::File(_)) => {
                    return Err(Error::NotDir(destination));
                }
                (Entry::File(_), Entry::Dir) => {
                    return Err(Error::IsDir(destination));
                }
                (Entry::Dir, Entry::Dir) => {
                    if self
                        .entries
                        .keys()
                        .any(|key| key != &destination && Self::contains_path(&destination, key))
                    {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let source_keys = self.subtree_keys(&source);
        let mut moved = Vec::with_capacity(source_keys.len());
        for old in &source_keys {
            let suffix = &old[source.len()..];
            moved.push((format!("{destination}{suffix}"), self.entries[old].clone()));
        }

        for key in &source_keys {
            self.entries.remove(key);
        }
        if dest_entry.is_some() {
            for key in self.subtree_keys(&destination) {
                self.entries.remove(&key);
            }
        }
        for (key, entry) in moved {
            self.entries.insert(key, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.validate_and_normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        self.resolve_parent(&canonical)?;
        let entry = match self.entries.get(&canonical) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(canonical)),
        };

        if matches!(entry, Entry::Dir) && Self::contains_path(&canonical, &self.cwd) {
            return Err(Error::Busy(canonical));
        }

        let keys = self.subtree_keys(&canonical);
        if matches!(entry, Entry::Dir) && keys.len() > 1 && !recursive {
            return Err(Error::NotEmpty(canonical));
        }
        for key in keys {
            self.entries.remove(&key);
        }
        Ok(())
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let absolute = path.starts_with('/');
        let mut parts: Vec<String> = if absolute {
            Vec::new()
        } else {
            Self::components(&self.cwd)
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other.to_string()),
            }
        }

        Ok(Self::join_components(&parts))
    }

    fn components(path: &str) -> Vec<String> {
        path.split('/')
            .filter(|part| !part.is_empty())
            .map(str::to_string)
            .collect()
    }

    fn join_components(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            let mut out = String::new();
            for part in parts {
                out.push('/');
                out.push_str(part);
            }
            out
        }
    }

    fn parent_and_name(path: &str) -> (String, String) {
        match path.rfind('/') {
            Some(0) => ("/".to_string(), path[1..].to_string()),
            Some(i) => (path[..i].to_string(), path[i + 1..].to_string()),
            None => ("/".to_string(), path.to_string()),
        }
    }

    fn resolve_existing(&self, path: &str) -> Result<&Entry, Error> {
        if path == "/" {
            return Ok(self.entries.get("/").expect("root exists"));
        }

        let parts = Self::components(path);
        let mut prefix = String::new();
        for (i, part) in parts.iter().enumerate() {
            prefix.push('/');
            prefix.push_str(part);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(entry) if i + 1 < parts.len() && entry.kind() != Kind::Dir => {
                    return Err(Error::NotDir(prefix));
                }
                Some(entry) if i + 1 == parts.len() => return Ok(entry),
                Some(_) => {}
            }
        }
        Err(Error::Missing(path.to_string()))
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent_and_name(path);
        match self.resolve_existing(&parent)? {
            Entry::Dir => Ok((parent, name)),
            _ => Err(Error::NotDir(parent)),
        }
    }

    fn exists(&self, path: &str) -> bool {
        self.entries.contains_key(path)
    }

    fn contains_path(directory: &str, path: &str) -> bool {
        if directory == "/" {
            return path.starts_with('/');
        }
        path == directory
            || path
                .strip_prefix(directory)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn subtree_has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && Self::contains_path(path, candidate))

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Exists(canonical));
        }
        let (parent, _) = self.resolve_parent(&canonical)?;
        if self.exists(&canonical) {
            return Err(Error::Exists(canonical));
        }
        self.entries.insert(canonical, Entry::Dir);
        let _ = parent;
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::IsDir(canonical));
        }
        self.resolve_parent(&canonical)?;
        match self.entries.get_mut(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

        match self.resolve_existing(&canonical)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir => Err(Error::IsDir(canonical)),

        let canonical = self.normalize(path)?;
        match self.resolve_existing(&canonical)? {
            Entry::File(_) => return Err(Error::NotDir(canonical)),
            Entry::Dir => {}
        }

        let prefix = if canonical == "/" {
            "/".to_string()
        } else {
            format!("{}/", canonical)
        };
        let mut children = Vec::new();
        for (candidate, entry) in &self.entries {
            if let Some(rest) = candidate.strip_prefix(&prefix) {
                if !rest.is_empty() && !rest.contains('/') {
                    children.push((rest.to_string(), entry.kind()));
                }
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        match self.resolve_existing(&canonical)? {
            Entry::Dir => {
                self.cwd = canonical;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(canonical)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (source_parent, _) = Self::parent_and_name(&source);
        self.resolve_existing(&source_parent)?;
        let source_entry = match self.entries.get(&source) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(source.clone())),
        };

        if source_entry.kind() == Kind::Dir && Self::contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        let (destination_parent, _) = Self::parent_and_name(&destination);
        self.resolve_parent(&destination)?;

        let destination_entry = self.entries.get(&destination).cloned();
        if matches!(destination_entry, Some(Entry::Dir))
            && Self::contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if source_entry.kind() == Kind::Dir && Self::contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if overwrite {
            if let Some(ref dest_entry) = destination_entry {
                match (source_entry.kind(), dest_entry.kind()) {
                    (Kind::Dir, Kind::File) => return Err(Error::NotDir(destination)),
                    (Kind::File, Kind::Dir) => return Err(Error::IsDir(destination)),
                    (Kind::Dir, Kind::Dir) if self.subtree_has_children(&destination) => {
                        return Err(Error::NotEmpty(destination));
                    }
                    _ => {}
                }
            }
        }

        let _ = destination_parent;

        let mut moved = Vec::new();
        for (old_path, entry) in &self.entries {
            if old_path == &source || Self::contains_path(&source, old_path) {
                let suffix = &old_path[source.len()..];
                moved.push((format!("{}{}", destination, suffix), entry.clone()));
            }
        }

        if destination_entry.is_some() {
            let keys: Vec<String> = self
                .entries
                .keys()
                .filter(|candidate| {
                    *candidate == &destination || Self::contains_path(&destination, candidate)
                })
                .cloned()
                .collect();
            for key in keys {
                self.entries.remove(&key);
            }
        }

        let source_keys: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| *candidate == &source || Self::contains_path(&source, candidate))
            .cloned()
            .collect();
        for key in source_keys {
            self.entries.remove(&key);
        }

        for (new_path, entry) in moved {
            self.entries.insert(new_path, entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, _) = Self::parent_and_name(&canonical);
        self.resolve_existing(&parent)?;
        let entry = match self.entries.get(&canonical) {
            Some(entry) => entry,
            None => return Err(Error::Missing(canonical)),
        };

        if entry.kind() == Kind::Dir && Self::contains_path(&canonical, &self.cwd) {
            return Err(Error::Busy(canonical));
        }

        let nonempty = entry.kind() == Kind::Dir && self.subtree_has_children(&canonical);
        if nonempty && !recursive {
            return Err(Error::NotEmpty(canonical));
        }

        let keys: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| *candidate == &canonical || Self::contains_path(&canonical, candidate))
            .cloned()
            .collect();
        for key in keys {
            self.entries.remove(&key);
        }
        Ok(())
    }
}

    fn default() -> Self {
        Self::new()
    }
}

pub struct Fs {

            return "/".to_string();
        }

        for part in parts {
            path.push('/');
            path.push_str(part);
        }
        path
    }

        if let Some(index) = input.as_bytes().iter().position(|byte| *byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if input.is_empty() {
            return Err(Error::InvalidPath(0));

            Vec::new()
        } else {
            self.cwd_parts.clone()
        };
        for item in input.split('/') {
            match item {
                "" | "." => {}
                ".." => {

                }

            }
        }
        Ok(Self::from_parts(&parts))
    }

        let slash = path.rfind('/').unwrap_or(0);
        if slash == 0 {
            ("/", &path[1..])
        } else {
            (&path[..slash], &path[slash + 1..])
        }
    }

    fn path_prefixes(path: &str) -> Vec<String> {
        if path == "/" {
            return vec!["/".to_string()];
        }
        let mut prefixes = vec!["/".to_string()];
        let mut prefix = String::new();
        for part in path[1..].split('/') {
            prefix.push('/');
            prefix.push_str(part);
            prefixes.push(prefix.clone());
        }
        prefixes
    }

    fn check_parent(&self, path: &str) -> Result<(), Error> {
        let (parent, _) = Self::parent_and_name(path);
        for prefix in Self::path_prefixes(parent) {
            match self.nodes.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Node::Directory) => {}
                Some(Node::Bytes(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn lookup(&self, path: &str) -> Result<&Node, Error> {
        let prefixes = Self::path_prefixes(path);
        for (i, prefix) in prefixes.iter().enumerate() {
            match self.nodes.get(prefix) {
                None => return Err(Error::Missing(prefix.clone())),
                Some(node) if i + 1 < prefixes.len() && !matches!(node, Node::Directory) => {
                    return Err(Error::NotDir(prefix.clone()));
                }
                Some(node) if i + 1 == prefixes.len() => return Ok(node),
                Some(_) => {}
            }
        }
        self.nodes.get("/").ok_or_else(|| Error::Missing("/".to_string()))
    }

    fn contains(parent: &str, child: &str) -> bool {
        parent == "/"
            || child == parent
            || child
                .strip_prefix(parent)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn keys_in_tree(&self, path: &str) -> Vec<String> {
        self.nodes
            .keys()
            .filter(|key| Self::contains(path, key))
            .cloned()
            .collect()
    }

    pub fn new_unused() -> Self {
        Self::new()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        self.check_parent(&path)?;
        if self.nodes.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.nodes.insert(path, Node::Directory);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        self.check_parent(&path)?;
        match self.nodes.get(&path) {
            Some(Node::Directory) => Err(Error::IsDir(path)),
            Some(Node::Bytes(_)) | None => {
                self.nodes.insert(path, Node::Bytes(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        match self.lookup(&path)? {
            Node::Directory => Err(Error::IsDir(path)),
            Node::Bytes(data) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        match self.lookup(&path)? {
            Node::Bytes(_) => return Err(Error::NotDir(path)),
            Node::Directory => {}
        }

        let mut result = Vec::new();
        for (entry_path, node) in &self.nodes {
            if entry_path == "/" {
                continue;
            }
            let (parent, basename) = Self::parent_and_name(entry_path);
            if parent == path {
                result.push((
                    basename.to_string(),
                    match node {
                        Node::Directory => Kind::Dir,
                        Node::Bytes(_) => Kind::File,
                    },
                ));
            }
        }
        result.sort_by(|left, right| left.0.cmp(&right.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd_path()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        match self.lookup(&path)? {
            Node::Directory => {
                self.cwd_parts = if path == "/" {
                    Vec::new()
                } else {
                    path[1..].split('/').map(str::to_string).collect()
                };
                Ok(())
            }
            Node::Bytes(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        self.check_parent(&source)?;
        let source_entry = match self.nodes.get(&source) {
            Some(node) => node.clone(),
            None => return Err(Error::Missing(source.clone())),
        };
        if matches!(source_entry, Node::Directory) && Self::contains(&source, &self.cwd_path()) {
            return Err(Error::Busy(source));
        }

        self.check_parent(&destination)?;
        if matches!(self.nodes.get(&destination), Some(Node::Directory))
            && Self::contains(&destination, &self.cwd_path())
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }
        if matches!(source_entry, Node::Directory) && Self::contains(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let target_entry = self.nodes.get(&destination).cloned();
        if target_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }
        if let Some(ref target) = target_entry {
            match (&source_entry, target) {
                (Node::Directory, Node::Bytes(_)) => return Err(Error::NotDir(destination)),
                (Node::Bytes(_), Node::Directory) => return Err(Error::IsDir(destination)),
                (Node::Directory, Node::Directory) => {
                    if self
                        .nodes
                        .keys()
                        .any(|key| key != &destination && Self::contains(&destination, key))
                    {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let old_keys = self.keys_in_tree(&source);
        let moved: Vec<(String, Node)> = old_keys
            .iter()
            .map(|old| {
                (
                    format!("{}{}", destination, &old[source.len()..]),
                    self.nodes[old].clone(),
                )
            })
            .collect();

        for old in old_keys {
            self.nodes.remove(&old);
        }
        if target_entry.is_some() {
            for key in self.keys_in_tree(&destination) {
                self.nodes.remove(&key);
            }
        }
        for (new, node) in moved {
            self.nodes.insert(new, node);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }
        self.check_parent(&path)?;
        let node = match self.nodes.get(&path) {
            Some(node) => node.clone(),
            None => return Err(Error::Missing(path.clone())),
        };

        if matches!(node, Node::Directory) && Self::contains(&path, &self.cwd_path()) {
            return Err(Error::Busy(path));
        }
        let keys = self.keys_in_tree(&path);
        if matches!(node, Node::Directory) && keys.len() > 1 && !recursive {
            return Err(Error::NotEmpty(path));
        }
        for key in keys {
            self.nodes.remove(&key);
        }
        Ok(())
    }
}
