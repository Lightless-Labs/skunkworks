- Error precedence follows the stated check order: normalize first, then resolve parents, then check final entries.
- A directory “containing cwd” includes the directory itself and uses whole path components.
- `..` is applied lexically before lookup, so nonexistent and file components can be canceled.
- Rename to the same canonical path succeeds only after source and destination checks; a source containing cwd is still `Busy`.
- Empty files and empty directories are valid; trailing slashes impose no type requirement.
- Both solutions use `BTreeMap` for deterministic child ordering and canonical absolute paths as keys.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn validate_and_normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts: Vec<&str> = if path.starts_with('/') {
            Vec::new()
        } else if self.cwd == "/" {
            Vec::new()
        } else {
            self.cwd[1..].split('/').collect()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other),
            }
        }

        if parts.is_empty() {
            Ok("/".to_owned())
        } else {
            let mut result = String::new();
            for part in parts {
                result.push('/');
                result.push_str(part);
            }
            Ok(result)
        }
    }

    fn parent(path: &str) -> (&str, &str) {
        if path == "/" {
            return ("/", "");
        }
        match path.rfind('/') {
            Some(0) => ("/", &path[1..]),
            Some(i) => (&path[..i], &path[i + 1..]),
            None => ("/", path),
        }
    }

    fn ancestors(path: &str) -> Vec<String> {
        let mut result = vec!["/".to_owned()];
        if path == "/" {
            return result;
        }
        let mut current = String::new();
        for component in path[1..].split('/') {
            current.push('/');
            current.push_str(component);
            result.push(current.clone());
        }
        result
    }

    fn resolve_parent(&self, path: &str) -> Result<(), Error> {
        let (parent, _) = Self::parent(path);
        for prefix in Self::ancestors(parent) {
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn resolve_entry(&self, path: &str) -> Result<&Entry, Error> {
        let components = Self::ancestors(path);
        for (index, prefix) in components.iter().enumerate() {
            match self.entries.get(prefix) {
                None => return Err(Error::Missing(prefix.clone())),
                Some(entry) if index + 1 < components.len() && !matches!(entry, Entry::Dir) => {
                    return Err(Error::NotDir(prefix.clone()));
                }
                Some(entry) => {
                    if index + 1 == components.len() {
                        return Ok(entry);
                    }
                }
            }
        }
        self.entries.get("/").ok_or_else(|| Error::Missing("/".to_owned()))
    }

    fn contains_path(dir: &str, path: &str) -> bool {
        if dir == "/" {
            return true;
        }
        path == dir || path.strip_prefix(dir).is_some_and(|tail| tail.starts_with('/'))
    }

    fn subtree_keys(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|key| Self::contains_path(path, key))
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.validate_and_normalize(path)?;
        self.resolve_parent(&canonical)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }
        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.validate_and_normalize(path)?;
        self.resolve_parent(&canonical)?;
        match self.entries.get(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.validate_and_normalize(path)?;
        match self.resolve_entry(&canonical)? {
            Entry::Dir => Err(Error::IsDir(canonical)),
            Entry::File(bytes) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.validate_and_normalize(path)?;
        match self.resolve_entry(&canonical)? {
            Entry::File(_) => return Err(Error::NotDir(canonical)),
            Entry::Dir => {}
        }
        let mut children = Vec::new();
        for (key, entry) in &self.entries {
            if key == "/" {
                continue;
            }
            let (parent, name) = Self::parent(key);
            if parent == canonical {
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                children.push((name.to_owned(), kind));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.validate_and_normalize(path)?;
        match self.resolve_entry(&canonical)? {
            Entry::Dir => {
                self.cwd = canonical;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(canonical)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.validate_and_normalize(from)?;
        let destination = self.validate_and_normalize(to)?;

        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&source)?;
        let source_entry = match self.entries.get(&source) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(source.clone())),
        };

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        self.resolve_parent(&destination)?;

        if let Some(Entry::Dir) = self.entries.get(&destination) {
            if Self::contains_path(&destination, &self.cwd) {
                return Err(Error::Busy(destination));
            }
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let dest_entry = self.entries.get(&destination).cloned();
        if dest_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(ref dest) = dest_entry {
            match (&source_entry, dest) {
                (Entry::Dir, Entry::File(_)) => {
                    return Err(Error::NotDir(destination));
                }
                (Entry::File(_), Entry::Dir) => {
                    return Err(Error::IsDir(destination));
                }
                (Entry::Dir, Entry::Dir) => {
                    if self
                        .entries
                        .keys()
                        .any(|key| key != &destination && Self::contains_path(&destination, key))
                    {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let source_keys = self.subtree_keys(&source);
        let mut moved = Vec::with_capacity(source_keys.len());
        for old in &source_keys {
            let suffix = &old[source.len()..];
            moved.push((format!("{destination}{suffix}"), self.entries[old].clone()));
        }

        for key in &source_keys {
            self.entries.remove(key);
        }
        if dest_entry.is_some() {
            for key in self.subtree_keys(&destination) {
                self.entries.remove(&key);
            }
        }
        for (key, entry) in moved {
            self.entries.insert(key, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.validate_and_normalize(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        self.resolve_parent(&canonical)?;
        let entry = match self.entries.get(&canonical) {
            Some(entry) => entry.clone(),
            None => return Err(Error::Missing(canonical)),
        };

        if matches!(entry, Entry::Dir) && Self::contains_path(&canonical, &self.cwd) {
            return Err(Error::Busy(canonical));
        }

        let keys = self.subtree_keys(&canonical);
        if matches!(entry, Entry::Dir) && keys.len() > 1 && !recursive {
            return Err(Error::NotEmpty(canonical));
        }
        for key in keys {
            self.entries.remove(&key);
        }
        Ok(())
    }
}
```

### Solution B

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Clone)]
enum Node {
    Directory,
    Bytes(Vec<u8>),
}

pub struct Fs {
    nodes: BTreeMap<String, Node>,
    cwd_parts: Vec<String>,
}

impl Fs {
    pub fn new() -> Self {
        let mut nodes = BTreeMap::new();
        nodes.insert("/".to_string(), Node::Directory);
        Self {
            nodes,
            cwd_parts: Vec::new(),
        }
    }

    fn cwd_path(&self) -> String {
        Self::from_parts(&self.cwd_parts)
    }

    fn from_parts(parts: &[String]) -> String {
        if parts.is_empty() {
            return "/".to_string();
        }
        let mut path = String::new();
        for part in parts {
            path.push('/');
            path.push_str(part);
        }
        path
    }

    fn normalize(&self, input: &str) -> Result<String, Error> {
        if let Some(index) = input.as_bytes().iter().position(|byte| *byte == 0) {
            return Err(Error::InvalidPath(index));
        }
        if input.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts = if input.starts_with('/') {
            Vec::new()
        } else {
            self.cwd_parts.clone()
        };
        for item in input.split('/') {
            match item {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                name => parts.push(name.to_string()),
            }
        }
        Ok(Self::from_parts(&parts))
    }

    fn parent_and_name(path: &str) -> (&str, &str) {
        let slash = path.rfind('/').unwrap_or(0);
        if slash == 0 {
            ("/", &path[1..])
        } else {
            (&path[..slash], &path[slash + 1..])
        }
    }

    fn path_prefixes(path: &str) -> Vec<String> {
        if path == "/" {
            return vec!["/".to_string()];
        }
        let mut prefixes = vec!["/".to_string()];
        let mut prefix = String::new();
        for part in path[1..].split('/') {
            prefix.push('/');
            prefix.push_str(part);
            prefixes.push(prefix.clone());
        }
        prefixes
    }

    fn check_parent(&self, path: &str) -> Result<(), Error> {
        let (parent, _) = Self::parent_and_name(path);
        for prefix in Self::path_prefixes(parent) {
            match self.nodes.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Node::Directory) => {}
                Some(Node::Bytes(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn lookup(&self, path: &str) -> Result<&Node, Error> {
        let prefixes = Self::path_prefixes(path);
        for (i, prefix) in prefixes.iter().enumerate() {
            match self.nodes.get(prefix) {
                None => return Err(Error::Missing(prefix.clone())),
                Some(node) if i + 1 < prefixes.len() && !matches!(node, Node::Directory) => {
                    return Err(Error::NotDir(prefix.clone()));
                }
                Some(node) if i + 1 == prefixes.len() => return Ok(node),
                Some(_) => {}
            }
        }
        self.nodes.get("/").ok_or_else(|| Error::Missing("/".to_string()))
    }

    fn contains(parent: &str, child: &str) -> bool {
        parent == "/"
            || child == parent
            || child
                .strip_prefix(parent)
                .is_some_and(|suffix| suffix.starts_with('/'))
    }

    fn keys_in_tree(&self, path: &str) -> Vec<String> {
        self.nodes
            .keys()
            .filter(|key| Self::contains(path, key))
            .cloned()
            .collect()
    }

    pub fn new_unused() -> Self {
        Self::new()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        self.check_parent(&path)?;
        if self.nodes.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        self.nodes.insert(path, Node::Directory);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        self.check_parent(&path)?;
        match self.nodes.get(&path) {
            Some(Node::Directory) => Err(Error::IsDir(path)),
            Some(Node::Bytes(_)) | None => {
                self.nodes.insert(path, Node::Bytes(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        match self.lookup(&path)? {
            Node::Directory => Err(Error::IsDir(path)),
            Node::Bytes(data) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        match self.lookup(&path)? {
            Node::Bytes(_) => return Err(Error::NotDir(path)),
            Node::Directory => {}
        }

        let mut result = Vec::new();
        for (entry_path, node) in &self.nodes {
            if entry_path == "/" {
                continue;
            }
            let (parent, basename) = Self::parent_and_name(entry_path);
            if parent == path {
                result.push((
                    basename.to_string(),
                    match node {
                        Node::Directory => Kind::Dir,
                        Node::Bytes(_) => Kind::File,
                    },
                ));
            }
        }
        result.sort_by(|left, right| left.0.cmp(&right.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd_path()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        match self.lookup(&path)? {
            Node::Directory => {
                self.cwd_parts = if path == "/" {
                    Vec::new()
                } else {
                    path[1..].split('/').map(str::to_string).collect()
                };
                Ok(())
            }
            Node::Bytes(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.normalize(from)?;
        let destination = self.normalize(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        self.check_parent(&source)?;
        let source_entry = match self.nodes.get(&source) {
            Some(node) => node.clone(),
            None => return Err(Error::Missing(source.clone())),
        };
        if matches!(source_entry, Node::Directory) && Self::contains(&source, &self.cwd_path()) {
            return Err(Error::Busy(source));
        }

        self.check_parent(&destination)?;
        if matches!(self.nodes.get(&destination), Some(Node::Directory))
            && Self::contains(&destination, &self.cwd_path())
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }
        if matches!(source_entry, Node::Directory) && Self::contains(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let target_entry = self.nodes.get(&destination).cloned();
        if target_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }
        if let Some(ref target) = target_entry {
            match (&source_entry, target) {
                (Node::Directory, Node::Bytes(_)) => return Err(Error::NotDir(destination)),
                (Node::Bytes(_), Node::Directory) => return Err(Error::IsDir(destination)),
                (Node::Directory, Node::Directory) => {
                    if self
                        .nodes
                        .keys()
                        .any(|key| key != &destination && Self::contains(&destination, key))
                    {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                _ => {}
            }
        }

        let old_keys = self.keys_in_tree(&source);
        let moved: Vec<(String, Node)> = old_keys
            .iter()
            .map(|old| {
                (
                    format!("{}{}", destination, &old[source.len()..]),
                    self.nodes[old].clone(),
                )
            })
            .collect();

        for old in old_keys {
            self.nodes.remove(&old);
        }
        if target_entry.is_some() {
            for key in self.keys_in_tree(&destination) {
                self.nodes.remove(&key);
            }
        }
        for (new, node) in moved {
            self.nodes.insert(new, node);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }
        self.check_parent(&path)?;
        let node = match self.nodes.get(&path) {
            Some(node) => node.clone(),
            None => return Err(Error::Missing(path.clone())),
        };

        if matches!(node, Node::Directory) && Self::contains(&path, &self.cwd_path()) {
            return Err(Error::Busy(path));
        }
        let keys = self.keys_in_tree(&path);
        if matches!(node, Node::Directory) && keys.len() > 1 && !recursive {
            return Err(Error::NotEmpty(path));
        }
        for key in keys {
            self.nodes.remove(&key);
        }
        Ok(())
    }
}
```
