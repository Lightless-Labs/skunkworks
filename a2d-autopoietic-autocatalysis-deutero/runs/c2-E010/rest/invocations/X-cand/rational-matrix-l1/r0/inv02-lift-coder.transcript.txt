#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn checked_result(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        let nn = i64::try_from(n).map_err(|_| RationalError::Overflow)?;
        let dd = i64::try_from(d).map_err(|_| RationalError::Overflow)?;
        if dd <= 0 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: nn, den: dd })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::checked_result(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::checked_result(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::checked_result(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::checked_result(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::checked_result(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::checked_result(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut slash = None;
        for (i, &c) in b.iter().enumerate() {
            if c == b'/' {
                if slash.is_some() {
                    return Err(err(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
            } else if c.is_ascii_digit() || (i == 0 && c == b'-') {
            } else {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }

        let ne = slash.unwrap_or(b.len());
        let ns = usize::from(b[0] == b'-');
        if ne == ns {
            return Err(err(b.len(), ParseKind::InvalidByte));
        }
        let ds = slash.map(|x| x + 1);
        if ds == Some(b.len()) {
            return Err(err(b.len(), ParseKind::InvalidByte));
        }

        if ne - ns > 1 && b[ns] == b'0' {
            return Err(err(ns + 1, ParseKind::NonCanonical));
        }
        if b[0] == b'-' && ne == 2 && b[1] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let Some(start) = ds {
            if b.len() - start > 1 && b[start] == b'0' {
                return Err(err(start + 1, ParseKind::NonCanonical));
            }
        }

        let negative = b[0] == b'-';
        let lim = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut nm = 0u128;
        for &c in &b[ns..ne] {
            let digit = (c - b'0') as u128;
            if nm > (lim - digit) / 10 {
                return Err(err(0, ParseKind::OutOfRange));
            }
            nm = nm * 10 + digit;
        }
        let n = if negative {
            if nm == 1u128 << 63 { i64::MIN } else { -(nm as i64) }
        } else {
            nm as i64
        };

        let Some(start) = ds else {
            return Ok(Self { num: n, den: 1 });
        };
        let mut dm = 0u128;
        for &c in &b[start..] {
            let digit = (c - b'0') as u128;
            if dm > (i64::MAX as u128 - digit) / 10 {
                return Err(err(start, ParseKind::OutOfRange));
            }
            dm = dm * 10 + digit;
        }
        if dm == 0 {
            return Err(err(start, ParseKind::ZeroDenominator));
        }
        if dm == 1 || n == 0 {
            return Err(err(start - 1, ParseKind::NonCanonical));
        }
        if gcd_u128(n.unsigned_abs() as u128, dm) != 1 {
            return Err(err(start - 1, ParseKind::NonCanonical));
        }
        Ok(Self { num: n, den: dm as i64 })
    }
