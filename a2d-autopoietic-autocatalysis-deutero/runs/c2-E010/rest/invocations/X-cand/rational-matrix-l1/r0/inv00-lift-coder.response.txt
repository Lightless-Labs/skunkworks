### Solution A

```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_wide(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }

    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let divisor = gcd(n.unsigned_abs(), d as u128) as i128;
    let n = n / divisor;
    let d = d / divisor;

    if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }

    Ok(Rational {
        num: n as i64,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }

        let mut n = num as i128;
        let mut d = den as i128;
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let divisor = gcd(n.unsigned_abs(), d as u128) as i128;
        n /= divisor;
        d /= divisor;

        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128
            + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_wide(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128
            - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        from_wide(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_wide(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        from_wide(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        from_wide(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }
        let num_end = i;

        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            i += 1;
            let start = i;
            den_start = Some(start);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == start {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let negative = bytes[0] == b'-';
        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(parse_error(num_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_end - num_start == 1 && bytes[num_start] == b'0' {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }
        if let (Some(start), Some(end)) = (den_start, den_end) {
            if end - start >= 2 && bytes[start] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let num_limit: u128 = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let mut num_mag = 0u128;
        for &b in &bytes[num_start..num_end] {
            num_mag = num_mag * 10 + (b - b'0') as u128;
            if num_mag > num_limit {
                return Err(parse_error(0, ParseKind::OutOfRange));
            }
        }

        let mut den_mag = 1u128;
        if let (Some(start), Some(end)) = (den_start, den_end) {
            den_mag = 0;
            for &b in &bytes[start..end] {
                den_mag = den_mag * 10 + (b - b'0') as u128;
                if den_mag > i64::MAX as u128 {
                    return Err(parse_error(start, ParseKind::OutOfRange));
                }
            }
            if den_mag == 0 {
                return Err(parse_error(start, ParseKind::ZeroDenominator));
            }
            if den_mag == 1 {
                return Err(parse_error(start - 1, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_error(start - 1, ParseKind::NonCanonical));
            }
            if gcd(num_mag, den_mag) > 1 {
                return Err(parse_error(start - 1, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };
        Ok(Self {
            num: signed_num,
            den: den_mag as i64,
        })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```

### Solution B

```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        (a, b) = (b, a % b);
    }
    a
}

fn make(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    let n = i64::try_from(n).map_err(|_| RationalError::Overflow)?;
    let d = i64::try_from(d).map_err(|_| RationalError::Overflow)?;
    Ok(Rational { num: n, den: d })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            Err(RationalError::ZeroDenominator)
        } else {
            make(num as i128, den as i128)
        }
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        make(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        make(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        make(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        make(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        make(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut p = usize::from(b[0] == b'-');
        let ns = p;
        while p < b.len() && b[p].is_ascii_digit() { p += 1; }
        if p == ns { return Err(err(p, ParseKind::InvalidByte)); }
        let ne = p;

        let (ds, de) = if p < b.len() {
            if b[p] != b'/' { return Err(err(p, ParseKind::InvalidByte)); }
            p += 1;
            let start = p;
            while p < b.len() && b[p].is_ascii_digit() { p += 1; }
            if p == start { return Err(err(p, ParseKind::InvalidByte)); }
            if p < b.len() { return Err(err(p, ParseKind::InvalidByte)); }
            (Some(start), Some(p))
        } else {
            (None, None)
        };

        let negative = b[0] == b'-';
        if ne - ns > 1 && b[ns] == b'0' {
            return Err(err(ns + 1, ParseKind::NonCanonical));
        }
        if negative && ne - ns == 1 && b[ns] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let (Some(s), Some(e)) = (ds, de) {
            if e - s > 1 && b[s] == b'0' {
                return Err(err(s + 1, ParseKind::NonCanonical));
            }
        }

        let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut nm = 0u128;
        for &c in &b[ns..ne] {
            nm = nm * 10 + (c - b'0') as u128;
            if nm > limit { return Err(err(0, ParseKind::OutOfRange)); }
        }

        let mut dm = 1u128;
        if let (Some(s), Some(e)) = (ds, de) {
            dm = 0;
            for &c in &b[s..e] {
                dm = dm * 10 + (c - b'0') as u128;
                if dm > i64::MAX as u128 { return Err(err(s, ParseKind::OutOfRange)); }
            }
            if dm == 0 { return Err(err(s, ParseKind::ZeroDenominator)); }
            if dm == 1 { return Err(err(s - 1, ParseKind::NonCanonical)); }
            if nm == 0 { return Err(err(s - 1, ParseKind::NonCanonical)); }
            if gcd(nm, dm) != 1 { return Err(err(s - 1, ParseKind::NonCanonical)); }
        }

        let n = if negative {
            if nm == 1u128 << 63 { i64::MIN } else { -(nm as i64) }
        } else {
            nm as i64
        };
        Ok(Rational { num: n, den: dm as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self.den {
            1 => write!(f, "{}", self.num),
            _ => write!(f, "{}/{}", self.num, self.den),
        }
    }
}
```
