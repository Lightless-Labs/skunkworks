- Syntax is validated across the entire input before any canonicality or range checks.
- Canonicality checks follow R11–R16 exactly, including their specified error offsets.
- `new` and arithmetic normalize in wider-than-`i64` integer space before checking representable bounds.
- A denominator is always positive; zero is represented only as `0/1`.
- Parsing arbitrarily long digit tokens is bounded in work per byte and reports range errors at the required token offsets.
- Solution B uses an independent `i128`-based arithmetic implementation; both take the same reading of the stated requirements.

### Solution A

```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let mut n = num;
        let mut d = den;
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }

        let mut i = 0;
        if b[0] == b'-' { i = 1; }
        let nstart = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == nstart {
            return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
        }

        let slash = if i < b.len() && b[i] == b'/' {
            let s = i;
            i += 1;
            let dstart = i;
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == dstart {
                return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
            }
            Some((s, dstart, i))
        } else {
            None
        };

        if i < b.len() {
            return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
        }

        let nend = slash.map_or(b.len(), |x| x.0);
        let n_digits_start = usize::from(b[0] == b'-');
        let n_digits = nend - n_digits_start;
        if n_digits >= 2 && b[n_digits_start] == b'0' {
            return Err(RationalError::Parse { at: n_digits_start + 1, kind: ParseKind::NonCanonical });
        }
        if b[0] == b'-' && n_digits == 1 && b[1] == b'0' {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::NonCanonical });
        }
        if let Some((_, ds, de)) = slash {
            if de - ds >= 2 && b[ds] == b'0' {
                return Err(RationalError::Parse { at: ds + 1, kind: ParseKind::NonCanonical });
            }
        }

        let negative = b[0] == b'-';
        let nlimit: u128 = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut nmag = 0u128;
        for &c in &b[n_digits_start..nend] {
            nmag = nmag * 10 + (c - b'0') as u128;
            if nmag > nlimit {
                return Err(RationalError::Parse { at: 0, kind: ParseKind::OutOfRange });
            }
        }

        let (dmag, slash_at) = if let Some((s, ds, de)) = slash {
            let mut d = 0u128;
            for &c in &b[ds..de] {
                d = d * 10 + (c - b'0') as u128;
                if d > i64::MAX as u128 {
                    return Err(RationalError::Parse { at: ds, kind: ParseKind::OutOfRange });
                }
            }
            (d, Some(s))
        } else {
            (1, None)
        };

        if let Some((_, ds, de)) = slash {
            if dmag == 0 {
                return Err(RationalError::Parse { at: ds, kind: ParseKind::ZeroDenominator });
            }
            if dmag == 1 {
                return Err(RationalError::Parse { at: slash_at.unwrap(), kind: ParseKind::NonCanonical });
            }
            if nmag == 0 {
                return Err(RationalError::Parse { at: slash_at.unwrap(), kind: ParseKind::NonCanonical });
            }
            let g = gcd_u128(nmag, dmag);
            if g > 1 {
                return Err(RationalError::Parse { at: slash_at.unwrap(), kind: ParseKind::NonCanonical });
            }
        }

        let num = if negative {
            if nmag == (1u128 << 63) { i64::MIN } else { -(nmag as i64) }
        } else {
            nmag as i64
        };
        Ok(Self { num, den: dmag as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```

### Solution B

```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn make(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 { return Err(RationalError::ZeroDenominator); }
        Self::make(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let (a, b, c, d) = (self.num as i128, self.den as i128, rhs.num as i128, rhs.den as i128);
        Self::make(a * d + c * b, b * d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let (a, b, c, d) = (self.num as i128, self.den as i128, rhs.num as i128, rhs.den as i128);
        Self::make(a * d - c * b, b * d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::make(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        Self::make(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::make(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let s = text.as_bytes();
        if s.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }

        let mut p = if s[0] == b'-' { 1 } else { 0 };
        let ns = p;
        while p < s.len() && s[p].is_ascii_digit() { p += 1; }
        if p == ns {
            return Err(RationalError::Parse { at: p, kind: ParseKind::InvalidByte });
        }
        let ne = p;
        let mut frac = None;
        if p < s.len() && s[p] == b'/' {
            let slash = p;
            p += 1;
            let ds = p;
            while p < s.len() && s[p].is_ascii_digit() { p += 1; }
            if p == ds {
                return Err(RationalError::Parse { at: p, kind: ParseKind::InvalidByte });
            }
            frac = Some((slash, ds, p));
        }
        if p < s.len() {
            return Err(RationalError::Parse { at: p, kind: ParseKind::InvalidByte });
        }

        let nstart = usize::from(s[0] == b'-');
        let ndigits = ne - nstart;
        if ndigits > 1 && s[nstart] == b'0' {
            return Err(RationalError::Parse { at: nstart + 1, kind: ParseKind::NonCanonical });
        }
        if s[0] == b'-' && ndigits == 1 && s[1] == b'0' {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::NonCanonical });
        }
        if let Some((_, ds, de)) = frac {
            if de - ds > 1 && s[ds] == b'0' {
                return Err(RationalError::Parse { at: ds + 1, kind: ParseKind::NonCanonical });
            }
        }

        let neg = s[0] == b'-';
        let nlim = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        let mut nv = 0u128;
        for &c in &s[nstart..ne] {
            nv = nv * 10 + (c - b'0') as u128;
            if nv > nlim {
                return Err(RationalError::Parse { at: 0, kind: ParseKind::OutOfRange });
            }
        }

        let (dv, slash) = if let Some((sl, ds, de)) = frac {
            let mut v = 0u128;
            for &c in &s[ds..de] {
                v = v * 10 + (c - b'0') as u128;
                if v > i64::MAX as u128 {
                    return Err(RationalError::Parse { at: ds, kind: ParseKind::OutOfRange });
                }
            }
            (v, Some(sl))
        } else {
            (1, None)
        };

        if let Some((_, ds, _)) = frac {
            let sl = slash.unwrap();
            if dv == 0 {
                return Err(RationalError::Parse { at: ds, kind: ParseKind::ZeroDenominator });
            }
            if dv == 1 || nv == 0 || gcd_u128(nv, dv) > 1 {
                return Err(RationalError::Parse { at: sl, kind: ParseKind::NonCanonical });
            }
        }

        let n = if neg {
            if nv == 1u128 << 63 { i64::MIN } else { -(nv as i64) }
        } else {
            nv as i64
        };
        Ok(Self { num: n, den: dv as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 { write!(f, "{}", self.num) } else { write!(f, "{}/{}", self.num, self.den) }
    }
}
```
