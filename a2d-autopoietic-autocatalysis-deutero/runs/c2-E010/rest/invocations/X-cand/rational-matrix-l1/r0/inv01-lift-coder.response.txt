- Parsing validates the entire byte grammar before checking canonicality or range; syntax errors take precedence.
- Canonicality and magnitude errors follow R11–R16 in their specified order, including the explicit-fraction checks.
- Negative magnitude `2^63` is valid only for the numerator; denominators are limited to `i64::MAX`.
- Arithmetic reduces the mathematical result before checking representable bounds, using wider intermediates.
- Both implementations use `i128` intermediates, which safely hold products and cross-products of `i64` inputs.

### Solution A

```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128(mut num: i128, mut den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    if den < 0 {
        num = -num;
        den = -den;
    }
    if num == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd(num.unsigned_abs(), den as u128) as i128;
    num /= g;
    den /= g;
    if num < i64::MIN as i128
        || num > i64::MAX as i128
        || den < 1
        || den > i64::MAX as i128
    {
        return Err(RationalError::Overflow);
    }
    Ok(Rational {
        num: num as i64,
        den: den as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_err(0, ParseKind::Empty));
        }

        let mut i = 0;
        if bytes[i] == b'-' {
            i += 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }
        let num_end = i;
        let mut den_start = None;
        let mut den_end = None;
        if i < bytes.len() && bytes[i] == b'/' {
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(parse_err(i, ParseKind::InvalidByte));
            }
            den_end = Some(i);
        }
        if i < bytes.len() {
            return Err(parse_err(i, ParseKind::InvalidByte));
        }

        let neg = bytes[0] == b'-';
        let num_digits = &bytes[num_start..num_end];
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(parse_err(if neg { 2 } else { 1 }, ParseKind::NonCanonical));
        }
        if neg && num_digits == b"0" {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        if let (Some(ds), Some(de)) = (den_start, den_end) {
            let digits = &bytes[ds..de];
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(parse_err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let max_num = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        let num_mag = parse_magnitude(num_digits, max_num)
            .ok_or_else(|| parse_err(0, ParseKind::OutOfRange))?;

        let den_mag = if let (Some(ds), Some(de)) = (den_start, den_end) {
            Some(
                parse_magnitude(&bytes[ds..de], i64::MAX as u128)
                    .ok_or_else(|| parse_err(ds, ParseKind::OutOfRange))?,
            )
        } else {
            None
        };

        if let (Some(ds), Some(d)) = (den_start, den_mag) {
            if d == 0 {
                return Err(parse_err(ds, ParseKind::ZeroDenominator));
            }
        }

        if let (Some(ds), Some(d)) = (den_start, den_mag) {
            if d == 1 {
                return Err(parse_err(ds - 1, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(parse_err(ds - 1, ParseKind::NonCanonical));
            }
            if gcd(num_mag, d) > 1 {
                return Err(parse_err(ds - 1, ParseKind::NonCanonical));
            }
            let signed_num = if neg {
                -(num_mag as i128)
            } else {
                num_mag as i128
            };
            return from_i128(signed_num, d as i128);
        }

        let signed_num = if neg {
            -(num_mag as i128)
        } else {
            num_mag as i128
        };
        from_i128(signed_num, 1)
    }
}

fn parse_magnitude(digits: &[u8], limit: u128) -> Option<u128> {
    let mut value = 0u128;
    for &b in digits {
        let d = (b - b'0') as u128;
        if value > (limit.saturating_sub(d)) / 10 {
            return None;
        }
        value = value * 10 + d;
        if value > limit {
            return None;
        }
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```

### Solution B

```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut x: u128, mut y: u128) -> u128 {
    while y != 0 {
        let rem = x % y;
        x = y;
        y = rem;
    }
    x
}

fn build(n: i128, d: i128) -> Result<Rational, RationalError> {
    if d == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let (mut n, mut d) = if d < 0 { (-n, -d) } else { (n, d) };
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let factor = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= factor;
    d /= factor;
    let n64 = i64::try_from(n).map_err(|_| RationalError::Overflow)?;
    let d64 = i64::try_from(d).map_err(|_| RationalError::Overflow)?;
    if d64 <= 0 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n64, den: d64 })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            Err(RationalError::ZeroDenominator)
        } else {
            build(num as i128, den as i128)
        }
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        build(a * d + c * b, b * d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        build(a * d - c * b, b * d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        build(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        build(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        build(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let s = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if s.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut pos = usize::from(s[0] == b'-');
        let negative = pos == 1;
        let n0 = pos;
        while pos < s.len() && s[pos].is_ascii_digit() {
            pos += 1;
        }
        if pos == n0 {
            return Err(err(pos, ParseKind::InvalidByte));
        }
        let n1 = pos;

        let mut dspan = None;
        if pos < s.len() && s[pos] == b'/' {
            pos += 1;
            let d0 = pos;
            while pos < s.len() && s[pos].is_ascii_digit() {
                pos += 1;
            }
            if pos == d0 {
                return Err(err(pos, ParseKind::InvalidByte));
            }
            dspan = Some((d0, pos));
        }
        if pos != s.len() {
            return Err(err(pos, ParseKind::InvalidByte));
        }

        let nd = &s[n0..n1];
        if nd.len() > 1 && nd[0] == b'0' {
            return Err(err(if negative { 2 } else { 1 }, ParseKind::NonCanonical));
        }
        if negative && nd == b"0" {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let Some((d0, d1)) = dspan {
            if d1 - d0 > 1 && s[d0] == b'0' {
                return Err(err(d0 + 1, ParseKind::NonCanonical));
            }
        }

        let nm = magnitude(nd, if negative { 1u128 << 63 } else { i64::MAX as u128 })
            .ok_or_else(|| err(0, ParseKind::OutOfRange))?;
        let dm = if let Some((d0, d1)) = dspan {
            Some(
                magnitude(&s[d0..d1], i64::MAX as u128)
                    .ok_or_else(|| err(d0, ParseKind::OutOfRange))?,
            )
        } else {
            None
        };

        if let (Some((d0, _)), Some(dv)) = (dspan, dm) {
            if dv == 0 {
                return Err(err(d0, ParseKind::ZeroDenominator));
            }
        }

        if let (Some((d0, _)), Some(dv)) = (dspan, dm) {
            let slash = d0 - 1;
            if dv == 1 || nm == 0 || gcd_u128(nm, dv) != 1 {
                return Err(err(slash, ParseKind::NonCanonical));
            }
            let numerator = if negative { -(nm as i128) } else { nm as i128 };
            build(numerator, dv as i128)
        } else {
            let numerator = if negative { -(nm as i128) } else { nm as i128 };
            build(numerator, 1)
        }
    }
}

fn magnitude(digits: &[u8], limit: u128) -> Option<u128> {
    let mut acc = 0u128;
    for &byte in digits {
        let digit = (byte - b'0') as u128;
        if acc > limit / 10 || (acc == limit / 10 && digit > limit % 10) {
            return None;
        }
        acc = acc * 10 + digit;
    }
    Some(acc)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self.den {
            1 => write!(f, "{}", self.num),
            den => write!(f, "{}/{}", self.num, den),
        }
    }
}
```
