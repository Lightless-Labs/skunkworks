- Syntax validation scans the entire UTF-8 byte string before any canonicality or magnitude checks; missing terminal digits report `text.len()`.
- Canonicality checks run in the exact order listed, so earlier numerator errors beat later denominator errors.
- Fractions with denominator `1`, explicit zero fractions, leading zeroes, or a common factor are rejected by parsing; bare integers remain valid.
- `new` and arithmetic normalize the mathematical result before checking bounds, including inputs involving `i64::MIN`.
- Arithmetic uses `i128` intermediates; products of bounded `i64` values and their cross-products fit safely.
- I interpret “cost on the largest sizes allowed” as requiring linear scanning of input, without parsing arbitrarily long digit strings into integers.

### Solution A

```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }

        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;

        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: ParseKind::Empty });
        }

        let mut slash = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
                }
                slash = Some(i);
                continue;
            }
            if b.is_ascii_digit() {
                continue;
            }
            if b == b'-' && i == 0 {
                continue;
            }
            return Err(RationalError::Parse { at: i, kind: ParseKind::InvalidByte });
        }

        let num_end = slash.unwrap_or(bytes.len());
        let num_start = if bytes.first() == Some(&b'-') { 1 } else { 0 };
        if num_start == num_end {
            return Err(RationalError::Parse { at: bytes.len(), kind: ParseKind::InvalidByte });
        }
        if let Some(s) = slash {
            if s + 1 == bytes.len() {
                return Err(RationalError::Parse { at: bytes.len(), kind: ParseKind::InvalidByte });
            }
        }

        let parse_err = |at, kind| RationalError::Parse { at, kind };

        if num_end - num_start >= 2 && bytes[num_start] == b'0' {
            return Err(parse_err(num_start + 1, ParseKind::NonCanonical));
        }
        if bytes.first() == Some(&b'-') && num_end == 2 && bytes[1] == b'0' {
            return Err(parse_err(0, ParseKind::NonCanonical));
        }

        let den_start = slash.map(|s| s + 1);
        if let Some(ds) = den_start {
            if bytes.len() - ds >= 2 && bytes[ds] == b'0' {
                return Err(parse_err(ds + 1, ParseKind::NonCanonical));
            }
        }

        let negative = bytes[0] == b'-';
        let mut mag: u128 = 0;
        for &b in &bytes[num_start..num_end] {
            mag = mag * 10 + (b - b'0') as u128;
            let limit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
            if mag > limit {
                return Err(parse_err(0, ParseKind::OutOfRange));
            }
        }
        let n = if negative {
            if mag == (1u128 << 63) { i64::MIN } else { -(mag as i64) }
        } else {
            mag as i64
        };

        let Some(ds) = den_start else {
            return Ok(Self { num: n, den: 1 });
        };
        let mut dmag: u128 = 0;
        for &b in &bytes[ds..] {
            dmag = dmag * 10 + (b - b'0') as u128;
            if dmag > i64::MAX as u128 {
                return Err(parse_err(ds, ParseKind::OutOfRange));
            }
        }
        if dmag == 0 {
            return Err(parse_err(ds, ParseKind::ZeroDenominator));
        }
        let d = dmag as i64;
        let slash_at = ds - 1;
        if d == 1 || n == 0 {
            return Err(parse_err(slash_at, ParseKind::NonCanonical));
        }
        if gcd_u128(n.unsigned_abs() as u128, d as u128) > 1 {
            return Err(parse_err(slash_at, ParseKind::NonCanonical));
        }
        Ok(Self { num: n, den: d })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```

```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn checked_result(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        let nn = i64::try_from(n).map_err(|_| RationalError::Overflow)?;
        let dd = i64::try_from(d).map_err(|_| RationalError::Overflow)?;
        if dd <= 0 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: nn, den: dd })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::checked_result(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::checked_result(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::checked_result(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::checked_result(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        Self::checked_result(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::checked_result(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() {
            return Err(err(0, ParseKind::Empty));
        }

        let mut slash = None;
        for (i, &c) in b.iter().enumerate() {
            if c == b'/' {
                if slash.is_some() {
                    return Err(err(i, ParseKind::InvalidByte));
                }
                slash = Some(i);
            } else if c.is_ascii_digit() || (i == 0 && c == b'-') {
            } else {
                return Err(err(i, ParseKind::InvalidByte));
            }
        }

        let ne = slash.unwrap_or(b.len());
        let ns = usize::from(b[0] == b'-');
        if ne == ns {
            return Err(err(b.len(), ParseKind::InvalidByte));
        }
        let ds = slash.map(|x| x + 1);
        if ds == Some(b.len()) {
            return Err(err(b.len(), ParseKind::InvalidByte));
        }

        if ne - ns > 1 && b[ns] == b'0' {
            return Err(err(ns + 1, ParseKind::NonCanonical));
        }
        if b[0] == b'-' && ne == 2 && b[1] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let Some(start) = ds {
            if b.len() - start > 1 && b[start] == b'0' {
                return Err(err(start + 1, ParseKind::NonCanonical));
            }
        }

        let negative = b[0] == b'-';
        let lim = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let mut nm = 0u128;
        for &c in &b[ns..ne] {
            let digit = (c - b'0') as u128;
            if nm > (lim - digit) / 10 {
                return Err(err(0, ParseKind::OutOfRange));
            }
            nm = nm * 10 + digit;
        }
        let n = if negative {
            if nm == 1u128 << 63 { i64::MIN } else { -(nm as i64) }
        } else {
            nm as i64
        };

        let Some(start) = ds else {
            return Ok(Self { num: n, den: 1 });
        };
        let mut dm = 0u128;
        for &c in &b[start..] {
            let digit = (c - b'0') as u128;
            if dm > (i64::MAX as u128 - digit) / 10 {
                return Err(err(start, ParseKind::OutOfRange));
            }
            dm = dm * 10 + digit;
        }
        if dm == 0 {
            return Err(err(start, ParseKind::ZeroDenominator));
        }
        if dm == 1 || n == 0 {
            return Err(err(start - 1, ParseKind::NonCanonical));
        }
        if gcd_u128(n.unsigned_abs() as u128, dm) != 1 {
            return Err(err(start - 1, ParseKind::NonCanonical));
        }
        Ok(Self { num: n, den: dm as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
