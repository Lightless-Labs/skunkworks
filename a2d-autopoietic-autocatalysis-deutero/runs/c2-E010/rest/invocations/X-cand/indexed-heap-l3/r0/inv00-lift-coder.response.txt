- Key validation trims only ASCII spaces, then reports the first invalid byte at its offset in the original input.
- Keys are ASCII-lowercased; values remain unchanged and rank by UTF-8 byte length.
- Ties use insertion age; successful updates retain age, while new pushes and merged keys get fresh ages.
- `Reject` reports the first collision in the source’s pre-merge rank order and leaves both queues unchanged.
- Since no capacity API is specified, the queue has no capacity setting to preserve.
- Both implementations use checked-by-construction rank comparisons without priority arithmetic or numeric age counters.

### Solution A

```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    next_age: u128,
}

fn canonical_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();
    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for (i, &b) in bytes[start..end].iter().enumerate() {
        let valid = if i == 0 {
            b.is_ascii_alphabetic()
        } else {
            b.is_ascii_alphabetic() || b.is_ascii_digit() || b == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + i });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    fn cmp_entries(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item.priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn rebuild_indices(&mut self) {
        self.indices.clear();
        for (i, e) in self.entries.iter().enumerate() {
            self.indices.insert(e.item.key.clone(), i);
        }
    }

    fn sort_entries(&mut self) {
        self.entries.sort_by(Self::cmp_entries);
        self.rebuild_indices();
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            for (i, e) in self.entries.iter_mut().enumerate() {
                e.age = i as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonical_key(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }
        let age = self.fresh_age();
        self.entries.push(Entry {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age,
        });
        self.sort_entries();
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.first().map(|e| e.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.entries.is_empty() {
            return None;
        }
        let entry = self.entries.remove(0);
        self.rebuild_indices();
        Some(entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonical_key(key)?;
        Ok(self.indices.get(&key).map(|&i| self.entries[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonical_key(key)?;
        let &i = self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.entries[i].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[i].item.priority = priority;
        self.sort_entries();
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonical_key(key)?;
        let &i = self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.entries[i].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[i].item.priority = priority;
        self.sort_entries();
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonical_key(key)?;
        let &i = self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let previous = std::mem::replace(&mut self.entries[i].item.value, value.to_owned());
        self.sort_entries();
        Ok(previous)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonical_key(key)?;
        let i = *self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let entry = self.entries.remove(i);
        self.rebuild_indices();
        Ok(entry.item)
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let mut source = other.entries.clone();
        source.sort_by(Self::cmp_entries);

        if policy == CollisionPolicy::Reject {
            for entry in &source {
                if self.indices.contains_key(&entry.item.key) {
                    return Err(Error::Collision {
                        key: entry.item.key.clone(),
                    });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
        };

        for entry in source {
            if let Some(&i) = self.indices.get(&entry.item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[i].item.priority = entry.item.priority;
                        self.entries[i].item.value = entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                self.entries.push(Entry { item: entry.item, age });
                report.moved += 1;
            }
        }

        self.sort_entries();
        other.entries.clear();
        other.indices.clear();
        Ok(report)
    }
}
```

### Solution B

```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
}

struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    by_key: HashMap<String, Entry>,
    next_age: u128,
}

fn canonical_key(input: &str) -> Result<String, Error> {
    let b = input.as_bytes();
    let (mut start, mut end) = (0, b.len());
    while start < end && b[start] == b' ' {
        start += 1;
    }
    while end > start && b[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }
    for i in start..end {
        let c = b[i];
        let valid = if i == start {
            c.is_ascii_alphabetic()
        } else {
            c.is_ascii_alphabetic() || c.is_ascii_digit() || c == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: i });
        }
    }
    Ok(input[start..end].to_ascii_lowercase())
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            by_key: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.by_key.len()
    }

    pub fn is_empty(&self) -> bool {
        self.by_key.is_empty()
    }

    fn ordered(&self) -> Vec<&Entry> {
        let mut v: Vec<&Entry> = self.by_key.values().collect();
        v.sort_by(|a, b| {
            a.item
                .priority
                .cmp(&b.item.priority)
                .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
                .then_with(|| a.age.cmp(&b.age))
        });
        v
    }

    fn age_for_insert(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut ordered: Vec<_> = self.by_key.values().map(|e| (e.age, e.item.key.clone())).collect();
            ordered.sort_by_key(|x| x.0);
            for (n, (_, key)) in ordered.into_iter().enumerate() {
                if let Some(e) = self.by_key.get_mut(&key) {
                    e.age = n as u128;
                }
            }
            self.next_age = self.by_key.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonical_key(key)?;
        if self.by_key.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }
        let age = self.age_for_insert();
        self.by_key.insert(
            key.clone(),
            Entry {
                item: Item {
                    key,
                    priority,
                    value: value.to_owned(),
                },
                age,
            },
        );
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.ordered().first().map(|e| e.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.ordered().first()?.item.key.clone();
        self.by_key.remove(&key).map(|e| e.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonical_key(key)?;
        Ok(self.by_key.get(&key).map(|e| e.item.clone()))
    }

    fn update_priority(
        &mut self,
        key: &str,
        priority: i64,
        decrease: bool,
    ) -> Result<i64, Error> {
        let key = canonical_key(key)?;
        let e = self.by_key.get_mut(&key).ok_or_else(|| Error::Missing { key })?;
        let current = e.item.priority;
        if (decrease && priority >= current) || (!decrease && priority <= current) {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        e.item.priority = priority;
        Ok(current)
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        self.update_priority(key, priority, true)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        self.update_priority(key, priority, false)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonical_key(key)?;
        let e = self.by_key.get_mut(&key).ok_or_else(|| Error::Missing { key })?;
        Ok(std::mem::replace(&mut e.item.value, value.to_owned()))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonical_key(key)?;
        self.by_key
            .remove(&key)
            .map(|e| e.item)
            .ok_or(Error::Missing { key })
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source: Vec<(String, Entry)> = other
            .ordered()
            .into_iter()
            .map(|e| (e.item.key.clone(), Entry { item: e.item.clone(), age: e.age }))
            .collect();

        if policy == CollisionPolicy::Reject {
            for (key, _) in &source {
                if self.by_key.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let mut report = MergeReport { moved: 0, replaced: 0, discarded: 0 };
        for (key, entry) in source {
            if let Some(receiver) = self.by_key.get_mut(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        receiver.item.priority = entry.item.priority;
                        receiver.item.value = entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.age_for_insert();
                self.by_key.insert(
                    key,
                    Entry { item: entry.item, age },
                );
                report.moved += 1;
            }
        }
        other.by_key.clear();
        Ok(report)
    }
}
```
