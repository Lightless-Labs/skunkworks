#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    heap: Vec<Entry>,
    positions: HashMap<String, usize>,
    next_age: u128,
}

fn canonical_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();
    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for (i, &b) in bytes[start..end].iter().enumerate() {
        let valid = if i == 0 {
            b.is_ascii_alphabetic()
        } else {
            b.is_ascii_alphabetic() || b.is_ascii_digit() || b == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + i });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            heap: Vec::new(),
            positions: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.heap.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    fn better(a: &Entry, b: &Entry) -> bool {
        a.item.priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
            .is_lt()
    }

    fn swap_slots(&mut self, a: usize, b: usize) {
        self.heap.swap(a, b);
        self.positions.insert(self.heap[a].item.key.clone(), a);
        self.positions.insert(self.heap[b].item.key.clone(), b);
    }

    fn sift_up(&mut self, mut i: usize) {
        while i > 0 {
            let p = (i - 1) / 2;
            if !Self::better(&self.heap[i], &self.heap[p]) {
                break;
            }
            self.swap_slots(i, p);
            i = p;
        }
    }

    fn sift_down(&mut self, mut i: usize) {
        loop {
            let left = i * 2 + 1;
            if left >= self.heap.len() {
                break;
            }
            let right = left + 1;
            let mut best = left;
            if right < self.heap.len() && Self::better(&self.heap[right], &self.heap[left]) {
                best = right;
            }
            if !Self::better(&self.heap[best], &self.heap[i]) {
                break;
            }
            self.swap_slots(i, best);
            i = best;
        }
    }

        if i > 0 && Self::better(&self.heap[i], &self.heap[(i - 1) / 2]) {
            self.sift_up(i);
        } else {
            self.sift_down(i);
        }
    }

        let last = self.heap.len() - 1;
        self.heap.swap(i, last);
        let removed = self.heap.pop().unwrap();

        if i < self.heap.len() {
            self.positions.insert(self.heap[i].item.key.clone(), i);
            self.repair(i);
        }
        removed
    }

    fn renumber_ages(&mut self) {
        let mut order: Vec<usize> = (0..self.heap.len()).collect();
        order.sort_by_key(|&i| self.heap[i].age);
        for (age, i) in order.into_iter().enumerate() {
            self.heap[i].age = age as u128;
        }
        self.next_age = self.heap.len() as u128;
    }

        let key = canonical_key(key)?;
        if self.positions.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }
        if self.next_age == u128::MAX {
            self.renumber_ages();
        }
        let i = self.heap.len();
        self.heap.push(Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        });
        self.next_age += 1;
        self.positions.insert(key, i);
        self.sift_up(i);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.heap.first().map(|e| e.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.heap.is_empty() {
            None
        } else {
            Some(self.remove_at(0).item)
        }
    }

        let key = canonical_key(key)?;
        Ok(self.positions.get(&key).map(|&i| self.heap[i].item.clone()))
    }

        let key = canonical_key(key)?;
        let i = *self.positions.get(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.heap[i].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.heap[i].item.priority = priority;
        self.repair(i);
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonical_key(key)?;
        let i = *self.positions.get(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.heap[i].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.heap[i].item.priority = priority;
        self.repair(i);
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonical_key(key)?;
        let i = *self.positions.get(&key).ok_or_else(|| Error::Missing { key })?;
        let old = std::mem::replace(&mut self.heap[i].item.value, value.to_owned());
        self.repair(i);
        Ok(old)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonical_key(key)?;
        let i = *self.positions.get(&key).ok_or_else(|| Error::Missing { key })?;
        Ok(self.remove_at(i).item)
    }

        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let mut source: Vec<Entry> = other.heap.iter().cloned().collect();
        source.sort_by(|a, b| {
            a.item
                .priority
                .cmp(&b.item.priority)
                .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
                .then_with(|| a.age.cmp(&b.age))
        });

            for entry in &source {
                if self.positions.contains_key(&entry.item.key) {
                    return Err(Error::Collision {
                        key: entry.item.key.clone(),
                    });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
        };

        for entry in source {
            if let Some(&i) = self.positions.get(&entry.item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.heap[i].item.priority = entry.item.priority;
                        self.heap[i].item.value = entry.item.value;
                        self.repair(i);
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                if self.next_age == u128::MAX {
                    self.renumber_ages();
                }
                let i = self.heap.len();
                self.heap.push(Entry {
                    item: entry.item,
                    age: self.next_age,
                });
                self.next_age += 1;
                let key = self.heap[i].item.key.clone();
                self.positions.insert(key, i);
                self.sift_up(i);
                report.moved += 1;
            }
        }

        other.heap.clear();
        other.positions.clear();
        other.next_age = 0;
        Ok(report)
    }
}

    pub replaced: usize,
    pub discarded: usize,

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn canonical_key(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }
        if start == end {
            return Err(Error::EmptyKey);
        }

        for (offset, &byte) in bytes[start..end].iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic()
            } else {
                byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + offset });
            }
        }

        Ok(input[start..end].to_ascii_lowercase())
    }

    fn compare(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn swap_slots(&mut self, a: usize, b: usize) {
        self.entries.swap(a, b);
        self.indices.insert(self.entries[a].item.key.clone(), a);
        self.indices.insert(self.entries[b].item.key.clone(), b);
    }

    fn sift_up(&mut self, mut at: usize) {
        while at > 0 {
            let parent = (at - 1) / 2;
            if Self::compare(&self.entries[at], &self.entries[parent]).is_lt() {
                self.swap_slots(at, parent);
                at = parent;
            } else {
                break;
            }
        }
    }

    fn sift_down(&mut self, mut at: usize) {
        loop {
            let left = at * 2 + 1;
            if left >= self.entries.len() {
                break;
            }
            let right = left + 1;
            let best = if right < self.entries.len()
                && Self::compare(&self.entries[right], &self.entries[left]).is_lt()
            {
                right
            } else {
                left
            };
            if Self::compare(&self.entries[best], &self.entries[at]).is_lt() {
                self.swap_slots(at, best);
                at = best;
            } else {
                break;
            }
        }
    }

    fn fix_at(&mut self, at: usize) {
        if at > 0 && Self::compare(&self.entries[at], &self.entries[(at - 1) / 2]).is_lt() {
            self.sift_up(at);
        } else {
            self.sift_down(at);
        }
    }

    fn take_at(&mut self, at: usize) -> Entry {
        let removed = self.entries.swap_remove(at);
        self.indices.remove(&removed.item.key);
        if at < self.entries.len() {
            self.indices.insert(self.entries[at].item.key.clone(), at);
            self.fix_at(at);
        }
        removed
    }

    fn ranked_entries(&self) -> Vec<Entry> {
        let mut ranked = self.entries.clone();
        ranked.sort_by(Self::compare);
        ranked
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {

            return Err(Error::Duplicate { key });

        let age = self.next_age;
        self.next_age = self.next_age.wrapping_add(1);
        let at = self.entries.len();

            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),

            age,

    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.entries.is_empty() {
            None
        } else {
            Some(self.take_at(0).item)
        }
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = Self::canonical_key(key)?;
        Ok(self.indices.get(&key).map(|&at| self.entries[at].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&at) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[at].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[at].item.priority = priority;
        self.fix_at(at);
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&at) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[at].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[at].item.priority = priority;
        self.fix_at(at);
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&at) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        let previous = std::mem::replace(&mut self.entries[at].item.value, value.to_owned());
        self.fix_at(at);
        Ok(previous)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = Self::canonical_key(key)?;
        let Some(&at) = self.indices.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(self.take_at(at).item)
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source = other.ranked_entries();
        if policy == CollisionPolicy::Reject {
            for entry in &source {
                if self.indices.contains_key(&entry.item.key) {
                    return Err(Error::Collision {
                        key: entry.item.key.clone(),
                    });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
        };

        for entry in source {
            if let Some(&at) = self.indices.get(&entry.item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[at].item.priority = entry.item.priority;
                        self.entries[at].item.value = entry.item.value;
                        self.fix_at(at);
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.next_age;
                self.next_age = self.next_age.wrapping_add(1);
                let at = self.entries.len();
                let key = entry.item.key.clone();
                self.entries.push(Entry {
                    item: entry.item,
                    age,
                });
                self.indices.insert(key, at);
                self.sift_up(at);
                report.moved += 1;
            }
        }

        other.entries.clear();
        other.indices.clear();
        other.next_age = 0;
        Ok(report)
    }
}

impl Default for IndexedHeap {
    fn default() -> Self {
        Self::new()
    }
}

fn canonicalize(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }
    Ok(input[start..end].to_ascii_lowercase())
}

impl IndexedHeap {
    fn rank_cmp(a: &Entry, b: &Entry) -> std::cmp::Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    pub fn new() -> Self {
        Self::default()
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn swap_entries(&mut self, a: usize, b: usize) {
        self.entries.swap(a, b);
        self.positions.insert(self.entries[a].item.key.clone(), a);
        self.positions.insert(self.entries[b].item.key.clone(), b);
    }

    fn sift_up(&mut self, mut i: usize) {
        while i > 0 {
            let p = (i - 1) / 2;
            if Self::rank_cmp(&self.entries[i], &self.entries[p]) != std::cmp::Ordering::Less {
                return;
            }
            self.swap_entries(i, p);
            i = p;
        }
    }

    fn sift_down(&mut self, mut i: usize) {
        loop {
            let l = i * 2 + 1;
            if l >= self.entries.len() {
                return;
            }
            let r = l + 1;
            let best = if r < self.entries.len()
                && Self::rank_cmp(&self.entries[r], &self.entries[l]) == std::cmp::Ordering::Less
            {
                r
            } else {
                l
            };
            if Self::rank_cmp(&self.entries[best], &self.entries[i])
                != std::cmp::Ordering::Less
            {
                return;
            }
            self.swap_entries(i, best);
            i = best;
        }
    }

    fn repair(&mut self, i: usize) {
        if i > 0
            && Self::rank_cmp(&self.entries[i], &self.entries[(i - 1) / 2])
                == std::cmp::Ordering::Less
        {
            self.sift_up(i);
        } else {
            self.sift_down(i);
        }
    }

    fn remove_at(&mut self, i: usize) -> Item {
        let removed = self.entries.swap_remove(i);
        self.positions.remove(&removed.item.key);
        if i < self.entries.len() {
            self.positions.insert(self.entries[i].item.key.clone(), i);
            self.repair(i);
        }
        removed.item
    }

    fn compact_ages(&mut self) {
        let mut order: Vec<usize> = (0..self.entries.len()).collect();
        order.sort_by_key(|&i| self.entries[i].age);
        for (new_age, i) in order.into_iter().enumerate() {
            self.entries[i].age = new_age as u128;
        }
        self.next_age = self.entries.len() as u128;
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            self.compact_ages();
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonicalize(key)?;
        if self.positions.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }
        let age = self.fresh_age();
        let i = self.entries.len();
        self.entries.push(Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age,
        });
        self.positions.insert(key, i);
        self.sift_up(i);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.first().map(|e| e.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.entries.is_empty() {
            None
        } else {
            Some(self.remove_at(0))
        }
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        Ok(self.positions.get(&key).map(|&i| self.entries[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let Some(&i) = self.positions.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[i].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {

                requested: priority,

        }
        self.entries[i].item.priority = priority;
        self.repair(i);

        let key = canonicalize(key)?;
        let Some(&i) = self.positions.get(&key) else {
            return Err(Error::Missing { key });
        };
        let current = self.entries[i].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[i].item.priority = priority;
        self.repair(i);
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize(key)?;
        let Some(&i) = self.positions.get(&key) else {
            return Err(Error::Missing { key });
        };
        let old = std::mem::replace(&mut self.entries[i].item.value, value.to_owned());
        self.repair(i);
        Ok(old)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize(key)?;
        let Some(&i) = self.positions.get(&key) else {
            return Err(Error::Missing { key });
        };
        Ok(self.remove_at(i))
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let mut order: Vec<usize> = (0..other.entries.len()).collect();
        order.sort_by(|&a, &b| Self::rank_cmp(&other.entries[a], &other.entries[b]));

        if policy == CollisionPolicy::Reject {
            for &i in &order {
                let key = &other.entries[i].item.key;
                if self.positions.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
        };

        for i in order {
            let source = other.entries[i].clone();
            if let Some(&receiver_i) = self.positions.get(&source.item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[receiver_i].item.priority = source.item.priority;
                        self.entries[receiver_i].item.value = source.item.value;
                        self.repair(receiver_i);
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                let new_i = self.entries.len();
                self.entries.push(Entry {
                    item: source.item,
                    age,
                });
                self.positions.insert(self.entries[new_i].item.key.clone(), new_i);
                self.sift_up(new_i);
                report.moved += 1;
            }
        }

        other.entries.clear();
        other.positions.clear();
        other.next_age = 0;
        Ok(report)
    }
}

    let mut start = 0;

fn before(a: &Entry, b: &Entry) -> bool {
    a.item.priority < b.item.priority
        || (a.item.priority == b.item.priority
            && (a.item.value.len() < b.item.value.len()
                || (a.item.value.len() == b.item.value.len() && a.age < b.age)))
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self::default()
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn swap_entries(&mut self, a: usize, b: usize) {
        self.entries.swap(a, b);
        self.positions
            .insert(self.entries[a].item.key.clone(), a);
        self.positions
            .insert(self.entries[b].item.key.clone(), b);
    }

    fn sift_up(&mut self, mut index: usize) {
        while index > 0 {
            let parent = (index - 1) / 2;
            if !before(&self.entries[index], &self.entries[parent]) {
                break;
            }
            self.swap_entries(index, parent);
            index = parent;
        }
    }

    fn sift_down(&mut self, mut index: usize) {
        let len = self.entries.len();
        loop {
            let left = index * 2 + 1;
            if left >= len {
                break;
            }
            let right = left + 1;
            let best = if right < len && before(&self.entries[right], &self.entries[left]) {
                right
            } else {
                left
            };
            if !before(&self.entries[best], &self.entries[index]) {
                break;
            }
            self.swap_entries(index, best);
            index = best;
        }
    }

    fn repair(&mut self, index: usize) {
        if index > 0 && before(&self.entries[index], &self.entries[(index - 1) / 2]) {
            self.sift_up(index);
        } else {
            self.sift_down(index);
        }
    }

    fn remove_at(&mut self, index: usize) -> Item {
        let removed = self.entries.swap_remove(index);
        self.positions.remove(&removed.item.key);
        if index < self.entries.len() {
            self.positions
                .insert(self.entries[index].item.key.clone(), index);
            self.repair(index);
        }
        removed.item
    }

        order.sort_by_key(|&i| self.entries[i].age);

        }
        self.next_age = self.entries.len() as u128;
    }

        self.next_age += 1;

        source_order.sort_by(|&a, &b| {
            if before(&other.entries[a], &other.entries[b]) {
                std::cmp::Ordering::Less
            } else if before(&other.entries[b], &other.entries[a]) {
                std::cmp::Ordering::Greater
            } else {
                std::cmp::Ordering::Equal
            }
        });

        if policy == CollisionPolicy::Reject {
            for &index in &source_order {
                let key = &other.entries[index].item.key;
                if self.positions.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
        };

        for &index in &source_order {
            let source = &other.entries[index];
            if let Some(&receiver_index) = self.positions.get(&source.item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[receiver_index].item.priority = source.item.priority;
                        self.entries[receiver_index].item.value = source.item.value.clone();
                        self.repair(receiver_index);
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                let new_index = self.entries.len();
                self.entries.push(Entry {
                    item: source.item.clone(),
                    age,
                });
                self.positions
                    .insert(source.item.key.clone(), new_index);
                self.sift_up(new_index);
                report.moved += 1;
            }
        }

        other.entries.clear();
        other.positions.clear();
        other.next_age = 0;

        Ok(report)
    }
}
