- Validation scans the whole template before binding lookup, so later syntax errors take precedence.
- For an unclosed expression, the opening brace is reported even if another opening brace appears later.
- A nested opening brace is checked before the expression’s operator or variable content.
- Empty comma-separated entries are errors; a present empty value still contributes a field.
- Only a leading supported operator is special; all other punctuation fails name validation.
- Valid percent triplets are preserved with uppercase hex digits; all other value bytes are encoded.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    names: Vec<(usize, usize)>,
}

#[derive(Debug, Clone)]
enum Part {
    Literal(usize, usize),
    Expr(Expression),
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut content_start = open + 1;
    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    match bytes[content_start] {
        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => {}
    }

    let mut names = Vec::new();
    let mut seen: Vec<(usize, usize)> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = (spec_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let mut i = spec_start;
        if !is_name_start(bytes[i]) {
            return Err(Error::InvalidName { pos: i });
        }
        i += 1;
        while i < spec_end && is_name_continue(bytes[i]) {
            i += 1;
        }
        if i != spec_end {
            return Err(Error::InvalidName { pos: i });
        }

        if seen.iter().any(|&(a, b)| template[a..b] == template[spec_start..spec_end]) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }

        seen.push((spec_start, spec_end));
        names.push((spec_start, spec_end));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        names,
    })
}

fn validate(template: &str) -> Result<Vec<Part>, Error> {
    let bytes = template.as_bytes();
    let mut parts = Vec::new();
    let mut i = 0;
    let mut literal_start = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                if literal_start < i {
                    parts.push(Part::Literal(literal_start, i));
                }
                let close = match (i + 1..bytes.len()).find(|&j| bytes[j] == b'}') {
                    Some(j) => j,
                    None => return Err(Error::Unclosed { pos: i }),
                };
                let expr = parse_expression(template, i, close)?;
                i = close + 1;
                literal_start = i;
                parts.push(Part::Expr(expr));
            }
            _ => i += 1,
        }
    }

    if literal_start < bytes.len() {
        parts.push(Part::Literal(literal_start, bytes.len()));
    }
    Ok(parts)
}

fn encode_value(value: &str, out: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];

        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
            i += 1;
        }
    }
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(candidate, _)| *candidate == name)
        .map(|(_, value)| *value)
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parts = validate(template)?;
    let mut out = String::new();

    for part in parts {
        match part {
            Part::Literal(start, end) => out.push_str(&template[start..end]),
            Part::Expr(expr) => {
                let _ = (expr.start, expr.end);
                let mut first = true;
                for (start, end) in expr.names {
                    if let Some(value) = lookup(&template[start..end], vars) {
                        if !first {
                            out.push(',');
                        }
                        first = false;
                        encode_value(value, &mut out);
                    }
                }
            }
        }
    }

    Ok(out)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

fn valid_start(byte: u8) -> bool {
    byte == b'_' || byte.is_ascii_alphabetic()
}

fn valid_continue(byte: u8) -> bool {
    valid_start(byte) || byte.is_ascii_digit()
}

fn validate_expression(source: &str, open: usize, close: usize) -> Result<(), Error> {
    let bytes = source.as_bytes();
    if let Some(position) = (open + 1..close).find(|&p| bytes[p] == b'{') {
        return Err(Error::NestedOpen { pos: position });
    }

    let mut cursor = open + 1;
    if cursor == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(bytes[cursor], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator { pos: cursor });
    }

    let mut previous_names: Vec<(usize, usize)> = Vec::new();
    loop {
        let end = (cursor..close)
            .find(|&p| bytes[p] == b',')
            .unwrap_or(close);

        if cursor == end {
            return Err(Error::EmptyVariable { pos: cursor });
        }
        if !valid_start(bytes[cursor]) {
            return Err(Error::InvalidName { pos: cursor });
        }

        let mut p = cursor + 1;
        while p < end && valid_continue(bytes[p]) {
            p += 1;
        }
        if p != end {
            return Err(Error::InvalidName { pos: p });
        }

        for &(a, b) in &previous_names {
            if source[a..b] == source[cursor..end] {
                return Err(Error::DuplicateVariable { pos: cursor });
            }
        }
        previous_names.push((cursor, end));

        if end == close {
            return Ok(());
        }
        cursor = end + 1;
    }
}

fn encode(source: &str, output: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = source.as_bytes();
    let mut p = 0;

    while p < bytes.len() {
        let byte = bytes[p];
        if byte == b'%' && p + 2 < bytes.len()
            && bytes[p + 1].is_ascii_hexdigit()
            && bytes[p + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[p + 1] as char).to_ascii_uppercase());
            output.push((bytes[p + 2] as char).to_ascii_uppercase());
            p += 3;
        } else if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
            p += 1;
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 15) as usize] as char);
            p += 1;
        }
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut pos = 0;
    let mut literal = 0;

    while pos < bytes.len() {
        if bytes[pos] == b'}' {
            return Err(Error::UnexpectedClose { pos });
        }
        if bytes[pos] != b'{' {
            pos += 1;
            continue;
        }

        let close = match (pos + 1..bytes.len()).find(|&p| bytes[p] == b'}') {
            Some(close) => close,
            None => return Err(Error::Unclosed { pos }),
        };
        validate_expression(template, pos, close)?;

        if literal < pos {
            output.push_str(&template[literal..pos]);
        }

        let mut spec = pos + 1;
        let mut first_field = true;
        loop {
            let end = (spec..close)
                .find(|&p| bytes[p] == b',')
                .unwrap_or(close);
            let name = &template[spec..end];

            let value = vars.iter().rev()
                .find(|(candidate, _)| *candidate == name)
                .map(|(_, value)| *value);

            if let Some(value) = value {
                if !first_field {
                    output.push(',');
                }
                first_field = false;
                encode(value, &mut output);
            }

            if end == close {
                break;
            }
            spec = end + 1;
        }

        pos = close + 1;
        literal = pos;
    }

    if literal < bytes.len() {
        output.push_str(&template[literal..]);
    }
    Ok(output)
}
```
