#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: open + 1 + offset });
                }

                validate_expression(bytes, open + 1, close)?;
                expressions.push((open, close));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    for (open, close) in expressions {
        output.push_str(&template[cursor..open]);
        expand_expression(template, open + 1, close, vars, &mut output);
        cursor = close + 1;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}

fn validate_expression(bytes: &[u8], start: usize, close: usize) -> Result<(), Error> {
    let mut content_start = start;
    if content_start < close && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator { pos: content_start });
    }

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut seen: Vec<&[u8]> = Vec::new();
    let mut spec_start = content_start;
    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name = &bytes[spec_start..spec_end];
        for (index, &byte) in name.iter().enumerate() {
            let valid = if index == 0 {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName { pos: spec_start + index });
            }
        }

        if seen.iter().any(|existing| *existing == name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(())
}

fn expand_expression(
    template: &str,
    start: usize,
    close: usize,
    vars: &[(&str, &str)],
    output: &mut String,
) {
    let bytes = template.as_bytes();
    let mut spec_start = start;
    let mut first_field = true;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);
        let name = &template[spec_start..spec_end];

        if let Some(value) = vars.iter().rev().find_map(|(key, value)| (*key == name).then_some(*value)) {
            if !first_field {
                output.push(',');
            }
            encode_value(value, output);
            first_field = false;
        }

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }
}

fn encode_value(value: &str, output: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(HEX[high] as char);
                output.push(HEX[low] as char);
                i += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }
}

fn hex_value(byte: u8) -> Option<usize> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as usize),
        b'a'..=b'f' => Some((byte - b'a' + 10) as usize),
        b'A'..=b'F' => Some((byte - b'A' + 10) as usize),
        _ => None,
    }
}
