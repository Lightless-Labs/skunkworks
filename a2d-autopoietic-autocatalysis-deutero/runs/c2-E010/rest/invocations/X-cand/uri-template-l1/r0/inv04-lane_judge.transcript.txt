    Expr(Expression),

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut content_start = open + 1;
    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    match bytes[content_start] {
        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&' => {
            return Err(Error::UnsupportedOperator { pos: content_start });
        }
        _ => {}
    }

    let mut names = Vec::new();
    let mut seen: Vec<(usize, usize)> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = (spec_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let mut i = spec_start;
        if !is_name_start(bytes[i]) {
            return Err(Error::InvalidName { pos: i });
        }
        i += 1;
        while i < spec_end && is_name_continue(bytes[i]) {
            i += 1;
        }
        if i != spec_end {
            return Err(Error::InvalidName { pos: i });
        }

        if seen.iter().any(|&(a, b)| template[a..b] == template[spec_start..spec_end]) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }

        seen.push((spec_start, spec_end));
        names.push((spec_start, spec_end));

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        names,
    })
}

    }
    Ok(parts)

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

fn valid_start(byte: u8) -> bool {
    byte == b'_' || byte.is_ascii_alphabetic()
}

fn valid_continue(byte: u8) -> bool {
    valid_start(byte) || byte.is_ascii_digit()
}

fn validate_expression(source: &str, open: usize, close: usize) -> Result<(), Error> {
    let bytes = source.as_bytes();
    if let Some(position) = (open + 1..close).find(|&p| bytes[p] == b'{') {
        return Err(Error::NestedOpen { pos: position });
    }

    let mut cursor = open + 1;
    if cursor == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(bytes[cursor], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator { pos: cursor });
    }

    let mut previous_names: Vec<(usize, usize)> = Vec::new();
    loop {
        let end = (cursor..close)
            .find(|&p| bytes[p] == b',')
            .unwrap_or(close);

        if cursor == end {
            return Err(Error::EmptyVariable { pos: cursor });
        }
        if !valid_start(bytes[cursor]) {
            return Err(Error::InvalidName { pos: cursor });
        }

        let mut p = cursor + 1;
        while p < end && valid_continue(bytes[p]) {
            p += 1;
        }
        if p != end {
            return Err(Error::InvalidName { pos: p });
        }

        for &(a, b) in &previous_names {
            if source[a..b] == source[cursor..end] {
                return Err(Error::DuplicateVariable { pos: cursor });
            }
        }
        previous_names.push((cursor, end));

        if end == close {
            return Ok(());
        }
        cursor = end + 1;
    }
}

        let byte = bytes[p];
        if byte == b'%' && p + 2 < bytes.len()

            output.push('%');
            output.push((bytes[p + 1] as char).to_ascii_uppercase());
            output.push((bytes[p + 2] as char).to_ascii_uppercase());

            output.push(byte as char);
            p += 1;
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 15) as usize] as char);
            p += 1;
        }
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut pos = 0;
    let mut literal = 0;

    while pos < bytes.len() {
        if bytes[pos] == b'}' {
            return Err(Error::UnexpectedClose { pos });
        }
        if bytes[pos] != b'{' {
            pos += 1;
            continue;
        }

        let close = match (pos + 1..bytes.len()).find(|&p| bytes[p] == b'}') {
            Some(close) => close,
            None => return Err(Error::Unclosed { pos }),
        };
        validate_expression(template, pos, close)?;

        if literal < pos {
            output.push_str(&template[literal..pos]);
        }

        let mut spec = pos + 1;
        let mut first_field = true;
        loop {
            let end = (spec..close)
                .find(|&p| bytes[p] == b',')
                .unwrap_or(close);
            let name = &template[spec..end];

            let value = vars.iter().rev()
                .find(|(candidate, _)| *candidate == name)
                .map(|(_, value)| *value);

            if let Some(value) = value {
                if !first_field {
                    output.push(',');
                }
                first_field = false;
                encode(value, &mut output);
            }

            if end == close {
                break;
            }
            spec = end + 1;
        }

        pos = close + 1;
        literal = pos;
    }

    if literal < bytes.len() {
        output.push_str(&template[literal..]);
    }
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => i + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: open + 1 + offset });
                }

                validate_expression(bytes, open + 1, close)?;
                expressions.push((open, close));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    for (open, close) in expressions {
        output.push_str(&template[cursor..open]);
        expand_expression(template, open + 1, close, vars, &mut output);
        cursor = close + 1;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}

fn validate_expression(bytes: &[u8], start: usize, close: usize) -> Result<(), Error> {
    let mut content_start = start;
    if content_start < close && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator { pos: content_start });
    }

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut seen: Vec<&[u8]> = Vec::new();
    let mut spec_start = content_start;
    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let name = &bytes[spec_start..spec_end];
        for (index, &byte) in name.iter().enumerate() {
            let valid = if index == 0 {
                byte.is_ascii_alphabetic() || byte == b'_'
            } else {
                byte.is_ascii_alphanumeric() || byte == b'_'
            };
            if !valid {
                return Err(Error::InvalidName { pos: spec_start + index });
            }
        }

        if seen.iter().any(|existing| *existing == name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(())
}

fn expand_expression(
    template: &str,
    start: usize,
    close: usize,
    vars: &[(&str, &str)],
    output: &mut String,
) {
    let bytes = template.as_bytes();
    let mut spec_start = start;
    let mut first_field = true;

    loop {
        let spec_end = bytes[spec_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| spec_start + offset)
            .unwrap_or(close);
        let name = &template[spec_start..spec_end];

        if let Some(value) = vars.iter().rev().find_map(|(key, value)| (*key == name).then_some(*value)) {
            if !first_field {
                output.push(',');
            }
            encode_value(value, output);
            first_field = false;
        }

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }
}

fn encode_value(value: &str, output: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(HEX[high] as char);
                output.push(HEX[low] as char);
                i += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }
}

fn hex_value(byte: u8) -> Option<usize> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as usize),
        b'a'..=b'f' => Some((byte - b'a' + 10) as usize),
        b'A'..=b'F' => Some((byte - b'A' + 10) as usize),
        _ => None,
    }
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    start: usize,
    end: usize,
    names: Vec<(usize, usize)>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut content_start = open + 1;
    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator { pos: content_start });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut item_start = content_start;

    loop {
        let item_end = (item_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if item_start == item_end {
            return Err(Error::EmptyVariable { pos: item_start });
        }

        let mut invalid_at = None;
        for i in item_start..item_end {
            let valid = if i == item_start {
                is_name_start(bytes[i])
            } else {
                is_name_continue(bytes[i])
            };
            if !valid {
                invalid_at = Some(i);
                break;
            }
        }

        if let Some(pos) = invalid_at {
            return Err(Error::InvalidName { pos });
        }

        let name = &template[item_start..item_end];
        if seen.iter().any(|&previous| previous == name) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        seen.push(name);
        names.push((item_start, item_end));

        if item_end == close {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close,
        names,
    })
}

fn validate(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match (open + 1..bytes.len()).find(|&j| bytes[j] == b'}') {
                    Some(close) => close,
                    None => return Err(Error::Unclosed { pos: open }),
                };
                expressions.push(parse_expression(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~') {
            out.push(b as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0F) as usize] as char);
            i += 1;
        }
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let mut out = String::with_capacity(template.len());
    let mut cursor = 0;

    for expression in expressions {
        out.push_str(&template[cursor..expression.start]);
        let mut first = true;

        for &(name_start, name_end) in &expression.names {
            let name = &template[name_start..name_end];
            if let Some(value) = lookup(name, vars) {
                if !first {
                    out.push(',');
                }
                encode_value(value, &mut out);
                first = false;
            }
        }

        cursor = expression.end + 1;
    }

    out.push_str(&template[cursor..]);
    Ok(out)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug, Clone)]
struct Expression {
    open: usize,
    close: usize,
    names: Vec<(usize, usize)>,
}

fn name_byte_valid(byte: u8, first: bool) -> bool {
    if first {
        byte.is_ascii_alphabetic() || byte == b'_'
    } else {
        byte.is_ascii_alphanumeric() || byte == b'_'
    }
}

fn parse(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    if let Some(pos) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos });
    }

    let mut begin = open + 1;
    if begin == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    if matches!(bytes[begin], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        return Err(Error::UnsupportedOperator { pos: begin });
    }

    let mut names = Vec::new();
    let mut previous_names: Vec<&str> = Vec::new();

    while begin <= close {
        let finish = (begin..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if begin == finish {
            return Err(Error::EmptyVariable { pos: begin });
        }

        for index in begin..finish {
            if !name_byte_valid(bytes[index], index == begin) {
                return Err(Error::InvalidName { pos: index });
            }
        }

        let name = &template[begin..finish];
        if previous_names.contains(&name) {
            return Err(Error::DuplicateVariable { pos: begin });
        }
        previous_names.push(name);
        names.push((begin, finish));

        if finish == close {
            break;
        }
        begin = finish + 1;
    }

    Ok(Expression { open, close, names })
}

fn validate_all(template: &str) -> Result<Vec<Expression>, Error> {
    let bytes = template.as_bytes();
    let mut result = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        if bytes[pos] == b'}' {
            return Err(Error::UnexpectedClose { pos });
        }
        if bytes[pos] != b'{' {
            pos += 1;
            continue;
        }

        let open = pos;
        let Some(close) = (open + 1..bytes.len()).find(|&i| bytes[i] == b'}') else {
            return Err(Error::Unclosed { pos: open });
        };

        result.push(parse(template, open, close)?);
        pos = close + 1;
    }

    Ok(result)
}

fn binding_value<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    let mut found = None;
    for &(binding_name, value) in vars {
        if binding_name == name {
            found = Some(value);
        }
    }
    found
}

fn percent_encode(value: &str, destination: &mut String) {
    const HEX: &[u8] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut offset = 0;

    while offset < bytes.len() {
        let byte = bytes[offset];
        if byte == b'%'
            && offset + 2 < bytes.len()
            && bytes[offset + 1].is_ascii_hexdigit()
            && bytes[offset + 2].is_ascii_hexdigit()
        {
            destination.push('%');
            destination.push((bytes[offset + 1] as char).to_ascii_uppercase());
            destination.push((bytes[offset + 2] as char).to_ascii_uppercase());
            offset += 3;
            continue;
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            destination.push(byte as char);
        } else {
            destination.push('%');
            destination.push(HEX[(byte / 16) as usize] as char);
            destination.push(HEX[(byte % 16) as usize] as char);
        }
        offset += 1;
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parsed = validate_all(template)?;
    let mut output = String::new();
    let mut copied_through = 0;

    for expression in parsed {
        output.push_str(&template[copied_through..expression.open]);
        let mut emitted = false;

        for (start, end) in expression.names {
            let name = &template[start..end];
            if let Some(value) = binding_value(name, vars) {
                if emitted {
                    output.push(',');
                }
                percent_encode(value, &mut output);
                emitted = true;
            }
        }

        copied_through = expression.close + 1;
    }

    output.push_str(&template[copied_through..]);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression {
    vars: Vec<String>,
}

#[derive(Debug)]
enum Part {
    Literal(String),
    Expression(Expression),
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let mut parts = Vec::new();
    let mut cursor = 0;

    while cursor < template.len() {
        let remaining = &template[cursor..];
        let next_open = remaining.find('{');
        let next_close = remaining.find('}');

        match (next_open, next_close) {
            (None, None) => {
                parts.push(Part::Literal(remaining.to_owned()));
                cursor = template.len();
            }
            (None, Some(close)) => {
                return Err(Error::UnexpectedClose {
                    pos: cursor + close,
                });
            }
            (Some(open), Some(close)) if close < open => {
                return Err(Error::UnexpectedClose {
                    pos: cursor + close,
                });
            }
            (Some(open), _) => {
                let open_pos = cursor + open;
                if open > 0 {
                    parts.push(Part::Literal(template[cursor..open_pos].to_owned()));
                }

                let content_start = open_pos + 1;
                let close_rel = match template[content_start..].find('}') {
                    Some(pos) => pos,
                    None => return Err(Error::Unclosed { pos: open_pos }),
                };
                let close_pos = content_start + close_rel;

                if let Some(inner_rel) = template[content_start..close_pos].find('{') {
                    return Err(Error::NestedOpen {
                        pos: content_start + inner_rel,
                    });
                }

                let content = &template[content_start..close_pos];
                let mut name_start = content_start;
                if let Some(first) = content.as_bytes().first().copied() {
                    if matches!(first, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                }

                if content.is_empty() {
                    return Err(Error::EmptyExpression { pos: close_pos });
                }

                let mut names = Vec::new();
                let mut seen = std::collections::HashSet::new();
                for spec in content.split(',') {
                    let spec_pos = name_start;
                    if spec.is_empty() {
                        return Err(Error::EmptyVariable { pos: spec_pos });
                    }

                    let bytes = spec.as_bytes();
                    for (i, &byte) in bytes.iter().enumerate() {
                        let valid = if i == 0 {
                            byte.is_ascii_alphabetic() || byte == b'_'
                        } else {
                            byte.is_ascii_alphanumeric() || byte == b'_'
                        };
                        if !valid {
                            return Err(Error::InvalidName { pos: spec_pos + i });
                        }
                    }

                    if !seen.insert(spec.to_owned()) {
                        return Err(Error::DuplicateVariable { pos: spec_pos });
                    }
                    names.push(spec.to_owned());
                    name_start += spec.len() + 1;
                }

                parts.push(Part::Expression(Expression { vars: names }));
                cursor = close_pos + 1;
            }
        }
    }

    let mut output = String::new();
    for part in parts {
        match part {
            Part::Literal(text) => output.push_str(&text),
            Part::Expression(expression) => {
                let mut first = true;
                for name in expression.vars {
                    if let Some(value) = vars
                        .iter()
                        .rev()
                        .find_map(|(key, value)| (*key == name).then_some(*value))
                    {
                        if !first {
                            output.push(',');
                        }
                        first = false;
                        encode_value(value, &mut output);
                    }
                }
            }
        }
    }
    Ok(output)
}

fn encode_value(value: &str, output: &mut String) {
    let bytes = value.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(upper_hex(high));
                output.push(upper_hex(low));
                i += 3;
                continue;
            }
        }

        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~') {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(upper_hex(byte >> 4));
            output.push(upper_hex(byte & 0x0f));
        }
        i += 1;
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn upper_hex(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
