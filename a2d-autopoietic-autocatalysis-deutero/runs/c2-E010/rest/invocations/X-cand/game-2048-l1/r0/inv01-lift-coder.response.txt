### Solution A

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    if row.len() > 32 {
        return Err(Error::RowTooLong { len: row.len() });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let indices: Box<dyn Iterator<Item = usize>> = match side {
        Side::Left => Box::new(0..row.len()),
        Side::Right => Box::new((0..row.len()).rev()),
    };

    let tiles: Vec<(usize, u32)> = indices
        .filter_map(|index| (row[index] != 0).then_some((index, row[index])))
        .collect();

    let mut outputs = Vec::with_capacity(row.len());
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut i = 0;

    while i < tiles.len() {
        if i + 1 < tiles.len() && tiles[i].1 == tiles[i + 1].1 {
            let first = tiles[i];
            let second = tiles[i + 1];
            let value = first.1.checked_add(second.1).ok_or(Error::MergeOverflow {
                first: first.0,
                second: second.0,
            })?;
            if value > 2_147_483_648 {
                return Err(Error::MergeOverflow {
                    first: first.0,
                    second: second.0,
                });
            }

            let to = match side {
                Side::Left => outputs.len(),
                Side::Right => row.len() - 1 - outputs.len(),
            };
            outputs.push(value);
            merges.push(Merge {
                from: [first.0, second.0],
                to,
                value,
            });
            score += u64::from(value);
            i += 2;
        } else {
            outputs.push(tiles[i].1);
            i += 1;
        }
    }

    let mut cells = vec![0; row.len()];
    for (offset, value) in outputs.into_iter().enumerate() {
        let index = match side {
            Side::Left => offset,
            Side::Right => row.len() - 1 - offset,
        };
        cells[index] = value;
    }

    let changed = cells != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
```

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Side {
    Left,
    Right,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    RowTooLong { len: usize },
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct RowResult {
    pub cells: Vec<u32>,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

pub fn slide_row(row: &[u32], side: Side) -> Result<RowResult, Error> {
    let len = row.len();
    if len > 32 {
        return Err(Error::RowTooLong { len });
    }

    for (index, &value) in row.iter().enumerate() {
        if value != 0 && !(2..=2_147_483_648).contains(&value) {
            return Err(Error::InvalidTile { index, value });
        }
        if value != 0 && value & (value - 1) != 0 {
            return Err(Error::InvalidTile { index, value });
        }
    }

    let mut cells = vec![0; len];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let mut input_cursor = match side {
        Side::Left => 0,
        Side::Right => len,
    };
    let mut output_count = 0usize;

    loop {
        let next_index = match side {
            Side::Left => {
                let mut found = None;
                while input_cursor < len {
                    let index = input_cursor;
                    input_cursor += 1;
                    if row[index] != 0 {
                        found = Some(index);
                        break;
                    }
                }
                found
            }
            Side::Right => {
                let mut found = None;
                while input_cursor > 0 {
                    input_cursor -= 1;
                    let index = input_cursor;
                    if row[index] != 0 {
                        found = Some(index);
                        break;
                    }
                }
                found
            }
        };

        let Some(first) = next_index else {
            break;
        };

        let mut pair = None;
        let saved_cursor = input_cursor;
        let second_index = match side {
            Side::Left => {
                let mut found = None;
                while input_cursor < len {
                    let index = input_cursor;
                    input_cursor += 1;
                    if row[index] != 0 {
                        found = Some(index);
                        break;
                    }
                }
                found
            }
            Side::Right => {
                let mut found = None;
                while input_cursor > 0 {
                    input_cursor -= 1;
                    let index = input_cursor;
                    if row[index] != 0 {
                        found = Some(index);
                        break;
                    }
                }
                found
            }
        };

        if let Some(second) = second_index {
            if row[first] == row[second] {
                let value = row[first].checked_add(row[second]).ok_or(
                    Error::MergeOverflow {
                        first,
                        second,
                    },
                )?;
                if value > 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first,
                        second,
                    });
                }
                let to = match side {
                    Side::Left => output_count,
                    Side::Right => len - 1 - output_count,
                };
                cells[to] = value;
                merges.push(Merge {
                    from: [first, second],
                    to,
                    value,
                });
                score += u64::from(value);
                output_count += 1;
                continue;
            } else {
                input_cursor = saved_cursor;
            }
        }

        let to = match side {
            Side::Left => output_count,
            Side::Right => len - 1 - output_count,
        };
        cells[to] = row[first];
        output_count += 1;
    }

    let changed = cells.as_slice() != row;
    Ok(RowResult {
        cells,
        score,
        changed,
        merges,
    })
}
```
