fn add_idle(idle: &mut Vec<(u64, u64)>, start: u64, end: u64) {
    if start >= end {
        return;
    }
    if let Some(last) = idle.last_mut() {
        if last.1 == start {
            last.1 = end;
            return;
        }
    }
    idle.push((start, end));
}

    if count == 0 {
        0

        (total * 100 + count as u64 / 2) / count as u64

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &q) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&q) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &length) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&length)
            } else {
                length <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }

        .iter()

            arrival: job.arrival as u64,
            bursts: job.bursts.clone(),
            burst_pos: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
            cpu_total: job
                .bursts
                .iter()

                .map(|&x| x as u64)
                .sum(),
            io_total: job
                .bursts
                .iter()
                .skip(1)
                .step_by(2)
                .map(|&x| x as u64)
                .sum(),
        })
        .collect();

    let mut events: Vec<Event> = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| Event {
            time: job.arrival as u64,
            kind: 0,
            index,
        })
        .collect();

        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let mut last_dispatch: Option<u32> = None;
    let mut completed = 0usize;

    while completed < jobs.len() {
        events.sort_by_key(|e| (e.time, e.kind, e.index));

        let mut event_pos = 0usize;

            let event = events[event_pos];
            if event.kind == 0 {
                if !admitted[event.index] {
                    admitted[event.index] = true;
                    ready[procs[event.index].level].push_back(event.index);
                }
            } else {
                let p = &mut procs[event.index];
                p.level = event.kind as usize; // replaced below by stored wake level
                ready[p.level].push_back(event.index);
            }
            event_pos += 1;
        }
        events.drain(0..event_pos);

        let selected_level = ready.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if events.is_empty() {
                break;
            }
            let next = events[0].time;
            if next > now {
                add_idle(&mut idle, now, next);
                now = next;
                last_dispatch = None;
            }
            continue;
        };

        let index = ready[level].pop_front().unwrap();
        let id = procs[index].id;
        if let Some(previous) = last_dispatch {
            if previous != id {
                now += switch_cost as u64;
            }
        }

        events.sort_by_key(|e| (e.time, e.kind, e.index));
        let mut event_pos = 0usize;

            let event = events[event_pos];
            if event.kind == 0 {
                if !admitted[event.index] {
                    admitted[event.index] = true;
                    ready[procs[event.index].level].push_back(event.index);
                }
            } else {
                let p = &mut procs[event.index];
                p.level = event.kind as usize;
                ready[p.level].push_back(event.index);
            }
            event_pos += 1;
        }
        events.drain(0..event_pos);

        let quantum = quanta[level];
        let consumed = procs[index].remaining.min(quantum);
        now += consumed as u64;
        procs[index].remaining -= consumed;

        let burst_finished = procs[index].remaining == 0;
        let job_finished = burst_finished && procs[index].burst_pos + 1 == procs[index].bursts.len();
        runs.push(Run {
            id,
            start: run_start,
            end: now,
            finished: job_finished,
            level,

        if burst_finished && !job_finished {

            let next_cpu_pos = io_pos + 1;

            procs[index].burst_pos = next_cpu_pos;

                kind: (promotion_level + 1) as u8,
                index,

                    admitted[event.index] = true;
                    ready[procs[event.index].level].push_back(event.index);

            } else {
                let p = &mut procs[event.index];
                p.level = (event.kind - 1) as usize;
                ready[p.level].push_back(event.index);
            }
            event_pos += 1;

            procs[index].completion = Some(now);
            completed += 1;

            procs[index].level = (level + 1).min(quanta.len() - 1);
            ready[procs[index].level].push_back(index);

        .iter()

            let completion = p.completion.unwrap_or(now);

                id: p.id,
                completion,
                turnaround,
                waiting: turnaround - p.cpu_total - p.io_total,

        })

    } else {
        stats.iter().map(|s| s.completion).max().unwrap_or(0)

        idle,
        stats,
        mean_waiting_x100: mean_x100(waiting_total, jobs.len()),
        mean_turnaround_x100: mean_x100(turnaround_total, jobs.len()),
        end,
    })
}

                    position: pos,

        .map(|(i, job)| Event {
            at: job.arrival as u64,
            kind: 0,
            index: i,
            wake_level: 0,
        })
        .collect();
    let mut now = 0u64;
    let mut last: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut done = 0usize;

    while done < jobs.len() {
        events.sort_by_key(|e| (e.at, e.kind, e.index));
        let mut consumed_events = 0;
        while consumed_events < events.len() && events[consumed_events].at <= now {
            let event = events[consumed_events];
            if event.kind == 0 {
                queues[ps[event.index].level].push_back(event.index);
            } else {
                ps[event.index].level = event.wake_level;
                queues[event.wake_level].push_back(event.index);
            }
            consumed_events += 1;
        }
        events.drain(0..consumed_events);

        let Some(level) = queues.iter().position(|q| !q.is_empty()) else {
            if events.is_empty() {
                break;
            }
            let next = events[0].at;
            if next > now {
                push_idle(&mut idle, now, next);
                now = next;
                last = None;
            }
            continue;
        };

        let i = queues[level].pop_front().unwrap();
        let id = ps[i].job.id;
        if last.is_some_and(|old| old != id) {
            now += switch_cost as u64;
        }

        events.sort_by_key(|e| (e.at, e.kind, e.index));
        let mut n = 0;
        while n < events.len() && events[n].at <= now {
            let ev = events[n];
            if ev.kind == 0 {
                queues[ps[ev.index].level].push_back(ev.index);
            } else {
                ps[ev.index].level = ev.wake_level;
                queues[ev.wake_level].push_back(ev.index);
            }
            n += 1;
        }
        events.drain(0..n);

        let start = now;
        let ran = ps[i].remaining.min(quanta[level]);
        now += ran as u64;
        ps[i].remaining -= ran;
        let burst_done = ps[i].remaining == 0;
        let finished = burst_done && ps[i].burst + 1 == ps[i].job.bursts.len();
        runs.push(Run {
            id,
            start,
            end: now,
            finished,
            level,
        });

        if burst_done && !finished {
            let delay = ps[i].job.bursts[ps[i].burst + 1] as u64;
            ps[i].burst += 2;
            ps[i].remaining = ps[i].job.bursts[ps[i].burst];
            events.push(Event {
                at: now + delay,
                kind: 1,
                index: i,
                wake_level: level.saturating_sub(1),
            });
        }

        events.sort_by_key(|e| (e.at, e.kind, e.index));
        let mut n = 0;
        while n < events.len() && events[n].at <= now {
            let ev = events[n];
            if ev.kind == 0 {
                queues[ps[ev.index].level].push_back(ev.index);
            } else {
                ps[ev.index].level = ev.wake_level;
                queues[ev.wake_level].push_back(ev.index);
            }
            n += 1;
        }
        events.drain(0..n);

        if finished {
            ps[i].completion = Some(now);
            done += 1;
        } else if !burst_done {
            ps[i].level = (level + 1).min(quanta.len() - 1);
            queues[ps[i].level].push_back(i);
        }
        last = Some(id);
    }

    let stats: Vec<Stat> = ps
        .iter()
        .map(|p| {
            let completion = p.completion.unwrap_or(now);
            let turnaround = completion - p.job.arrival as u64;
            Stat {
                id: p.job.id,
                completion,
                turnaround,
                waiting: turnaround - p.cpu - p.io,
            }
        })
        .collect();
    let total_waiting: u64 = stats.iter().map(|x| x.waiting).sum();
    let total_turnaround: u64 = stats.iter().map(|x| x.turnaround).sum();
    let count = jobs.len() as u64;
    let mean_waiting_x100 = if count == 0 {
        0
    } else {
        (total_waiting * 100 + count / 2) / count
    };
    let mean_turnaround_x100 = if count == 0 {
        0
    } else {
        (total_turnaround * 100 + count / 2) / count
    };
    let end = stats.iter().map(|x| x.completion).max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}
