const ADLER_MOD: u32 = 65_521;

const fn make_crc_table() -> [u32; 256] {
    let mut table = [0u32; 256];
    let mut i = 0;
    while i < 256 {
        let mut c = i as u32;
        let mut bit = 0;
        while bit < 8 {
            c = if c & 1 != 0 {
                (c >> 1) ^ 0xedb8_8320
            } else {
                c >> 1
            };
            bit += 1;
        }
        table[i] = c;
        i += 1;
    }
    table
}

const CRC_TABLE: [u32; 256] = make_crc_table();

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    a: u32,
    b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: !0,
            a: 1,
            b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= ADLER_MOD {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: !crc32,
            a,
            b,
        })
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: !self.crc,
            adler32: (self.b << 16) | self.a,
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            self.crc = (self.crc >> 8)
                ^ CRC_TABLE[((self.crc ^ byte as u32) & 0xff) as usize];
            self.a = ((self.a as u64 + byte as u64) % ADLER_MOD as u64) as u32;
            self.b = ((self.b as u64 + self.a as u64) % ADLER_MOD as u64) as u32;
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data.len(), spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let checksums = self.checksum();
        format!("c{:08x}:a{:08x}", checksums.crc32, checksums.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut numbers = [0u32; 2];
        for (offset, &byte) in bytes.iter().enumerate() {
            let slot = if (1..=8).contains(&offset) {
                0
            } else if (11..=18).contains(&offset) {
                1
            } else {
                continue;
            };
            let nibble = match byte {
                b'0'..=b'9' => (byte - b'0') as u32,
                b'a'..=b'f' => (byte - b'a' + 10) as u32,
                _ => return Err(SnapshotError::Hex { offset }),
            };
            numbers[slot] = (numbers[slot] << 4) | nibble;
        }
        Self::from_seed(numbers[0], numbers[1])
    }
}

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = None;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        for endpoint in [span.start, span.end] {
            if endpoint > data_len {
                return Err(SpanError::OutOfBounds {
                    span: index,
                    offset: endpoint,
                });
            }
        }
        if let Some(end) = previous_end {
            if span.start < end {
                return Err(SpanError::Overlap { span: index });
            }
        }
        previous_end = Some(span.end);
    }
    Ok(())
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = !0u32;
    for &byte in data {
        crc = (crc >> 8) ^ CRC_TABLE[((crc ^ byte as u32) & 0xff) as usize];
    }
    !crc
}

    let mut b = 0u32;

        a = ((a as u64 + byte as u64) % ADLER_MOD as u64) as u32;
        b = ((b as u64 + a as u64) % ADLER_MOD as u64) as u32;

        push_hex8(&mut out, h.adler32);

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc: !crc32,
            a,
            b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        for &byte in data {
            let mut x = self.crc ^ byte as u32;
            for _ in 0..8 {
                x = (x >> 1) ^ (0xedb8_8320 & (0u32.wrapping_sub(x & 1)));
            }
            self.crc = x;

            let mut a = self.a as u64 + byte as u64;
            if a >= 65521 {
                a -= 65521;
            }
            let mut b = self.b as u64 + a;
            if b >= 65521 {
                b -= 65521;
            }
            self.a = a as u32;
            self.b = b as u32;
        }
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        let mut previous_end = 0usize;
        for (index, span) in spans.iter().enumerate() {
            if span.start > span.end {
                return Err(SpanError::Reversed { span: index });
            }
            if span.start > data.len() {
                return Err(SpanError::OutOfBounds {
                    span: index,
                    offset: span.start,
                });
            }
            if span.end > data.len() {
                return Err(SpanError::OutOfBounds {
                    span: index,
                    offset: span.end,
                });
            }
            if index != 0 && span.start < previous_end {
                return Err(SpanError::Overlap { span: index });
            }
            previous_end = span.end;
        }

        for span in spans {
            for &byte in &data[span.start..span.end] {
                self.update(std::slice::from_ref(&byte));
            }
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        self.crc = 0xffff_ffff;
        self.a = 1;
        self.b = 0;
    }

    pub fn snapshot(&self) -> String {
        let value = self.checksum();
        let mut result = String::with_capacity(19);
        use std::fmt::Write;
        write!(&mut result, "c{:08x}:a{:08x}", value.crc32, value.adler32)
            .expect("writing to String cannot fail");
        result
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut values = [0u32; 2];
        for (group, range) in [(0, 1..9), (1, 11..19)] {
            let mut value = 0u32;
            for offset in range {
                let digit = match bytes[offset] {
                    b'0'..=b'9' => (bytes[offset] - b'0') as u32,
                    b'a'..=b'f' => (bytes[offset] - b'a' + 10) as u32,
                    _ => return Err(SnapshotError::Hex { offset }),
                };
                value = (value << 4) | digit;
            }
            values[group] = value;
        }
        Self::from_seed(values[0], values[1])
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffffu32;
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let mask = 0u32.wrapping_sub(crc & 1);
            crc = (crc >> 1) ^ (0xedb8_8320 & mask);
        }
    }
    !crc
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = ((a as u64 + byte as u64) % 65521) as u32;
        b = ((b as u64 + a as u64) % 65521) as u32;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index != 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }

    let mut crc = 0xffff_ffffu32;
    let mut a = 1u32;
    let mut b = 0u32;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            crc ^= byte as u32;
            for _ in 0..8 {
                crc = (crc >> 1) ^ (0xedb8_8320 & (0u32.wrapping_sub(crc & 1)));
            }
            a = ((a as u64 + byte as u64) % 65521) as u32;
            b = ((b as u64 + a as u64) % 65521) as u32;
        }
    }
    Ok(Hashes {
        crc32: !crc,
        adler32: (b << 16) | a,
    })
}
