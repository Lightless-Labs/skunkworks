    crc: u32,
    a: u32,
    b: u32,
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }

        let a = adler32 & 0xffff;
        let b = adler32 >> 16;

            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }

            crc: crc32 ^ 0xffff_ffff,
            a,
            b,
        })

fn validate_spans(data_len: usize, spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data_len {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

fn crc32_update(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc
}

fn adler_update(a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *a = (*a + byte as u32) % 65521;
        *b = (*b + *a) % 65521;
    }

pub fn fletcher16(data: &[u8]) -> u16 {
    let mut a = 0u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b << 8) | a) as u16
}

#[derive(Debug, Clone, Copy)]
struct Fletcher32State {
    a: u32,
    b: u32,
    pending: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending: None,
        }
    }

    fn update(&mut self, data: &[u8]) {
        for &byte in data {
            if let Some(high) = self.pending.take() {
                let word = ((high as u32) << 8) | byte as u32;
                self.add_word(word);
            } else {
                self.pending = Some(byte);
            }
        }
    }

    fn add_word(&mut self, word: u32) {
        self.a = (self.a + word) % 65535;
        self.b = (self.b + self.a) % 65535;
    }

    fn finish(mut self) -> u32 {
        if let Some(high) = self.pending.take() {
            self.add_word((high as u32) << 8);
        }
        (self.b << 16) | self.a
    }
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = Fletcher32State::new();
    state.update(data);
    state.finish()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = Fletcher32State::new();
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.finish())
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut crc: u16 = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                crc ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = crc & 0x8000;
                    crc = crc.wrapping_shl(1);
                    if top != 0 {
                        crc ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                crc ^= byte as u16;
                for _ in 0..8 {
                    let low = crc & 1;
                    crc >>= 1;
                    if low != 0 {
                        crc ^= 0x8408;
                    }
                }
            }
        }
    }

    if variant == Crc16::X25 {
        crc ^ 0xffff
    } else {
        crc
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;

    let mut base = Stream::new();
    let mut f16_a = 0u32;
    let mut f16_b = 0u32;
    let mut f32 = Fletcher32State::new();
    let mut c16 = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        base.update(selected);
        f32.update(selected);

        for &byte in selected {
            f16_a = (f16_a + byte as u32) % 255;
            f16_b = (f16_b + f16_a) % 255;

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    c16 ^= (byte as u16) << 8;

                        c16 = c16.wrapping_shl(1);

                            c16 ^= 0x1021;
                        }
                    }
                }
                Crc16::X25 => {

                        c16 >>= 1;
                        if low != 0 {
                            c16 ^= 0x8408;
                        }
                    }
                }
            }
        }
    }

    if variant == Crc16::X25 {
        c16 ^= 0xffff;
    }

        base: base.checksum(),

        fletcher32: f32.finish(),
        crc16: c16,
    })
}

        }

        Self::from_seed(crc, adler)
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn validate_spans(data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

#[derive(Clone)]
struct Accumulator {
    crc: u32,
    a: u32,
    b: u32,
    f16a: u32,
    f16b: u32,
    f32a: u32,
    f32b: u32,
    pending: Option<u8>,
    crc16: u16,
    variant: Crc16,
}

impl Accumulator {
    fn new(variant: Crc16) -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
            f16a: 0,
            f16b: 0,
            f32a: 0,
            f32b: 0,
            pending: None,
            crc16: match variant {
                Crc16::Xmodem => 0,
                Crc16::False | Crc16::X25 => 0xffff,
            },
            variant,
        }
    }

    fn push(&mut self, byte: u8) {
        self.crc ^= byte as u32;
        for _ in 0..8 {
            let low = self.crc & 1;
            self.crc >>= 1;
            if low != 0 {
                self.crc ^= 0xedb8_8320;
            }
        }

        self.a = (self.a + byte as u32) % 65521;
        self.b = (self.b + self.a) % 65521;

        self.f16a = (self.f16a + byte as u32) % 255;
        self.f16b = (self.f16b + self.f16a) % 255;

        if let Some(high) = self.pending.take() {
            self.push_f32_word(((high as u16) << 8) | byte as u16);
        } else {
            self.pending = Some(byte);
        }

        match self.variant {
            Crc16::Xmodem | Crc16::False => {
                self.crc16 ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = self.crc16 & 0x8000 != 0;
                    self.crc16 = self.crc16.wrapping_shl(1);
                    if top {
                        self.crc16 ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                self.crc16 ^= byte as u16;
                for _ in 0..8 {
                    let low = self.crc16 & 1 != 0;
                    self.crc16 >>= 1;
                    if low {
                        self.crc16 ^= 0x8408;
                    }
                }
            }
        }
    }

    fn push_f32_word(&mut self, word: u16) {
        self.f32a = (self.f32a + word as u32) % 65535;
        self.f32b = (self.f32b + self.f32a) % 65535;
    }

    fn finish_f32(&mut self) {
        if let Some(high) = self.pending.take() {
            self.push_f32_word((high as u16) << 8);
        }
    }

    fn result(&mut self) -> Multi {
        self.finish_f32();
        Multi {
            base: Hashes {
                crc32: self.crc ^ 0xffff_ffff,
                adler32: (self.b << 16) | self.a,
            },
            fletcher16: ((self.f16b as u16) << 8) | self.f16a as u16,
            fletcher32: (self.f32b << 16) | self.f32a,
            crc16: match self.variant {
                Crc16::X25 => self.crc16 ^ 0xffff,
                _ => self.crc16,
            },
        }
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffffu32;
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let low = crc & 1;
            crc >>= 1;
            if low != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }

        b = (b + a) % 255;

    let mut b = 0u32;
    let mut pending = None;
    for &byte in data {
        if let Some(high) = pending.take() {
            let word = ((high as u16) << 8) | byte as u16;
            a = (a + word as u32) % 65535;
            b = (b + a) % 65535;
        } else {
            pending = Some(byte);
        }
    }
    if let Some(high) = pending {
        let word = (high as u16) << 8;
        a = (a + word as u32) % 65535;
        b = (b + a) % 65535;

    }

pub struct Stream {
    raw_crc: u32,
    adler_a: u32,
    adler_b: u32,

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }

impl Stream {
    pub fn new() -> Self {
        Self {
            raw_crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: !self.raw_crc,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let adler_a = adler32 & 0xffff;
        let adler_b = adler32 >> 16;
        if adler_a >= 65521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if adler_b >= 65521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            raw_crc: !crc32,
            adler_a,
            adler_b,
        })
    }

        for &x in data {
            let mut c = self.raw_crc ^ x as u32;
            for _ in 0..8 {
                c = (c >> 1) ^ (0xedb8_8320 & (0u32.wrapping_sub(c & 1)));
            }
            self.raw_crc = c;

            self.adler_a += x as u32;
            if self.adler_a >= 65521 {
                self.adler_a -= 65521;
            }
            self.adler_b += self.adler_a;
            self.adler_b %= 65521;
        }
    }

        &mut self,
        data: &[u8],
        spans: &[Span],

        validate_spans(data.len(), spans)?;
        for s in spans {
            self.update(&data[s.start..s.end]);
        }
        Ok(())
    }

        self.raw_crc = 0xffff_ffff;
        self.adler_a = 1;
        self.adler_b = 0;
    }

    pub fn snapshot(&self) -> String {
        let h = self.checksum();
        format!("c{:08x}:a{:08x}", h.crc32, h.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let b = token.as_bytes();
        if b.len() != 19 || b[0] != b'c' || b[9] != b':' || b[10] != b'a' {
            return Err(SnapshotError::Shape);
        }

        let mut values = [0u32; 2];
        for offset in 1..=18 {
            if offset == 9 || offset == 10 {
                continue;
            }
            let ch = b[offset];
            let d = match ch {
                b'0'..=b'9' => (ch - b'0') as u32,
                b'a'..=b'f' => (ch - b'a' + 10) as u32,
                _ => return Err(SnapshotError::Hex { offset }),
            };
            let which = if offset < 9 { 0 } else { 1 };
            values[which] = (values[which] << 4) | d;
        }
        Self::from_seed(values[0], values[1])
    }

    b: u32,

            a: 0,
            b: 0,
            high: None,
        }

    fn byte(&mut self, x: u8) {
        match self.high.take() {
            None => self.high = Some(x),
            Some(h) => {
                let w = ((h as u32) << 8) | x as u32;
                self.a = (self.a + w) % 65535;
                self.b = (self.b + self.a) % 65535;
            }
        }
    }

    fn bytes(&mut self, data: &[u8]) {
        for &x in data {
            self.byte(x);
        }
    }

    fn result(mut self) -> u32 {
        if let Some(h) = self.high {
            let w = (h as u32) << 8;
            self.a = (self.a + w) % 65535;
            self.b = (self.b + self.a) % 65535;
        }
        (self.b << 16) | self.a
    }
}

                    r >>= 1;
                    if low != 0 {
                        r ^= 0x8408;
                    }
                }
            }
        }
    }

    data: &[u8],
    spans: &[Span],
    variant: Crc16,

    let mut fb = 0u32;

    for s in spans {
        let slice = &data[s.start..s.end];
        base.update(slice);
        f32.bytes(slice);
        for &x in slice {
            fa = (fa + x as u32) % 255;
            fb = (fb + fa) % 255;
            match variant {
                Crc16::Xmodem | Crc16::False => {
                    c16 ^= (x as u16) << 8;

                        c16 >>= 1;

                        }
                    }
                }
            }
        }
    }

    if variant == Crc16::X25 {
        c16 ^= 0xffff;
    }

    Ok(Multi {
        base: base.checksum(),
        fletcher16: ((fb << 8) | fa) as u16,

    })

#[derive(Debug, Clone, Copy)]
struct Fletcher32State {
    a: u32,
    b: u32,
    pending: Option<u8>,
}

impl Fletcher32State {
    fn new() -> Self {
        Self {
            a: 0,
            b: 0,
            pending: None,
        }
    }

    fn update(&mut self, data: &[u8]) {
        let mut index = 0;
        if let Some(high) = self.pending.take() {
            if let Some(&low) = data.first() {
                self.word(((high as u16) << 8) | low as u16);
                index = 1;
            } else {
                self.pending = Some(high);
                return;
            }
        }
        while index + 1 < data.len() {
            self.word(((data[index] as u16) << 8) | data[index + 1] as u16);
            index += 2;
        }
        if index < data.len() {
            self.pending = Some(data[index]);
        }
    }

    fn word(&mut self, word: u16) {
        self.a = (self.a + word as u32) % 65_535;
        self.b = (self.b + self.a) % 65_535;
    }

    fn finish(mut self) -> u32 {
        if let Some(high) = self.pending.take() {
            self.word((high as u16) << 8);
        }
        (self.b << 16) | self.a
    }
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut state = Fletcher32State::new();
    state.update(data);
    state.finish()
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut state = Fletcher32State::new();
    for span in spans {
        state.update(&data[span.start..span.end]);
    }
    Ok(state.finish())
}

fn crc16_update(mut register: u16, data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem | Crc16::False => {
            for &byte in data {
                register ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = register & 0x8000;
                    register = register.wrapping_shl(1);
                    if top != 0 {
                        register ^= 0x1021;
                    }
                }
            }
            register
        }
        Crc16::X25 => {
            for &byte in data {
                register ^= byte as u16;
                for _ in 0..8 {
                    let low = register & 1;
                    register >>= 1;
                    if low != 0 {
                        register ^= 0x8408;
                    }
                }
            }
            register
        }
    }
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    match variant {
        Crc16::Xmodem => crc16_update(0, data, variant),
        Crc16::False => crc16_update(0xffff, data, variant),
        Crc16::X25 => crc16_update(0xffff, data, variant) ^ 0xffff,
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data.len(), spans)?;
    let mut stream = Stream::new();
    let mut f16_a = 0;
    let mut f16_b = 0;
    let mut f32 = Fletcher32State::new();
    let mut crc16_register = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        let selected = &data[span.start..span.end];
        stream.update(selected);
        (f16_a, f16_b) = fletcher16_update(f16_a, f16_b, selected);
        f32.update(selected);
        crc16_register = crc16_update(crc16_register, selected, variant);
    }

    let crc16 = if variant == Crc16::X25 {
        crc16_register ^ 0xffff
    } else {
        crc16_register
    };

    Ok(Multi {
        base: stream.checksum(),
        fletcher16: ((f16_b << 8) | f16_a) as u16,
        fletcher32: f32.finish(),
        crc16,
    })
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SnapshotError {

}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Crc16 {

#[derive(Debug, Clone, Copy, PartialEq, Eq)]

#[derive(Debug, Clone)]
pub struct Stream {
    crc: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        match (a < 65521, b < 65521) {
            (false, _) => Err(SnapshotError::Adler { component: 0 }),
            (true, false) => Err(SnapshotError::Adler { component: 1 }),
            (true, true) => Ok(Self {
                crc: crc32 ^ 0xffff_ffff,
                adler_a: a,
                adler_b: b,
            }),
        }
    }

    pub fn update(&mut self, data: &[u8]) {
        update_crc_adler(&mut self.crc, &mut self.adler_a, &mut self.adler_b, data);
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn update_spans(
        &mut self,
        data: &[u8],
        spans: &[Span],
    ) -> Result<(), SpanError> {
        validate_spans(data, spans)?;
        for span in spans {
            update_crc_adler(
                &mut self.crc,
                &mut self.adler_a,
                &mut self.adler_b,
                &data[span.start..span.end],
            );
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        self.crc = 0xffff_ffff;
        self.adler_a = 1;
        self.adler_b = 0;
    }

    pub fn snapshot(&self) -> String {
        let checksums = self.checksum();
        let mut output = String::with_capacity(19);
        output.push('c');
        push_hex8(&mut output, checksums.crc32);
        output.push(':');
        output.push('a');
        push_hex8(&mut output, checksums.adler32);
        output
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19
            || bytes[0] != b'c'
            || bytes[9] != b':'
            || bytes[10] != b'a'
        {
            return Err(SnapshotError::Shape);
        }

        let mut values = [0u32; 2];
        for (slot, range) in [(0usize, 1usize..9usize), (1usize, 11usize..19usize)] {
            let mut value = 0u32;
            for offset in range {
                let digit = parse_hex(bytes[offset])
                    .ok_or(SnapshotError::Hex { offset })?;
                value = (value << 4) | digit as u32;
            }
            values[slot] = value;
        }
        Self::from_seed(values[0], values[1])
    }
}

impl Default for Stream {
    fn default() -> Self {
        Self::new()
    }
}

fn update_crc_adler(crc: &mut u32, a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *crc ^= byte as u32;
        for _ in 0..8 {
            let low = *crc & 1;
            *crc >>= 1;
            if low != 0 {
                *crc ^= 0xedb8_8320;
            }
        }
        *a = (*a + byte as u32) % 65521;
        *b = (*b + *a) % 65521;
    }
}

fn parse_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        _ => None,
    }
}

fn push_hex8(output: &mut String, value: u32) {
    const DIGITS: &[u8; 16] = b"0123456789abcdef";
    for shift in (0..8).rev() {
        output.push(DIGITS[((value >> (shift * 4)) & 0xf) as usize] as char);
    }
}

fn validate_spans(data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index != 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffffu32;
    for &byte in data {
        crc ^= byte as u32;
        for _ in 0..8 {
            let mask = 0u32.wrapping_sub(crc & 1);
            crc = (crc >> 1) ^ (0xedb8_8320 & mask);
        }
    }
    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (1u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 65521;
        b = (b + a) % 65521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data, spans)?;
    let mut stream = Stream::new();
    stream.update_spans(data, spans)?;
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

fn fletcher32_push(a: &mut u32, b: &mut u32, word: u16) {
    *a = (*a + word as u32) % 65535;
    *b = (*b + *a) % 65535;
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let mut a = 0u32;
    let mut b = 0u32;
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = u16::from_be_bytes([pair[0], pair[1]]);
        fletcher32_push(&mut a, &mut b, word);
    }
    if let Some(&last) = chunks.remainder().first() {
        fletcher32_push(&mut a, &mut b, (last as u16) << 8);
    }
    (b << 16) | a
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data, spans)?;
    let mut a = 0u32;
    let mut b = 0u32;
    let mut pending_high = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            match pending_high.take() {
                Some(high) => fletcher32_push(
                    &mut a,
                    &mut b,
                    u16::from_be_bytes([high, byte]),
                ),
                None => pending_high = Some(byte),
            }
        }
    }
    if let Some(high) = pending_high {
        fletcher32_push(&mut a, &mut b, (high as u16) << 8);
    }
    Ok((b << 16) | a)
}

fn crc16_update(mut register: u16, data: &[u8], variant: Crc16) -> u16 {
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                register ^= (byte as u16) << 8;
                for _ in 0..8 {
                    let top = register & 0x8000 != 0;
                    register = register.wrapping_shl(1);
                    if top {
                        register ^= 0x1021;
                    }
                }
            }
            Crc16::X25 => {
                register ^= byte as u16;
                for _ in 0..8 {
                    let low = register & 1 != 0;
                    register >>= 1;
                    if low {
                        register ^= 0x8408;
                    }
                }
            }
        }
    }
    register
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let initial = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };
    let register = crc16_update(initial, data, variant);
    match variant {
        Crc16::X25 => register ^ 0xffff,
        _ => register,
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data, spans)?;
    let mut crc = 0xffff_ffffu32;
    let (mut adler_a, mut adler_b) = (1u32, 0u32);
    let (mut f16a, mut f16b) = (0u32, 0u32);
    let (mut f32a, mut f32b) = (0u32, 0u32);
    let mut pending_high = None;
    let mut crc16_register = match variant {
        Crc16::Xmodem => 0,
        Crc16::False | Crc16::X25 => 0xffff,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            crc ^= byte as u32;
            for _ in 0..8 {
                let mask = 0u32.wrapping_sub(crc & 1);
                crc = (crc >> 1) ^ (0xedb8_8320 & mask);
            }

            adler_a = (adler_a + byte as u32) % 65521;
            adler_b = (adler_b + adler_a) % 65521;

            f16a = (f16a + byte as u32) % 255;
            f16b = (f16b + f16a) % 255;

            if let Some(high) = pending_high.take() {
                fletcher32_push(
                    &mut f32a,
                    &mut f32b,
                    u16::from_be_bytes([high, byte]),
                );
            } else {
                pending_high = Some(byte);
            }

            crc16_register = crc16_update(crc16_register, &[byte], variant);
        }
    }

    if let Some(high) = pending_high {
        fletcher32_push(&mut f32a, &mut f32b, (high as u16) << 8);
    }

    Ok(Multi {
        base: Hashes {
            crc32: crc ^ 0xffff_ffff,
            adler32: (adler_b << 16) | adler_a,
        },
        fletcher16: ((f16b as u16) << 8) | f16a as u16,
        fletcher32: (f32b << 16) | f32a,
        crc16: match variant {
            Crc16::X25 => crc16_register ^ 0xffff,
            _ => crc16_register,
        },
    })
}
