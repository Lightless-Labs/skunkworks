fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            if hi.is_none() || lo.is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hi.unwrap() << 4) | lo.unwrap());
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = origins.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn codec_to_parse(err: CodecError, offset: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },

fn parse_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {
    if port_text.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    let bytes = port_text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: port_start + i });
        }
    }
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for &b in bytes {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u32);
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

fn parse_authority(
    authority: &str,
    authority_start: usize,
) -> Result<(Option<String>, String, Option<u16>, bool), ParseError> {
    let bytes = authority.as_bytes();
    let ats: Vec<usize> = bytes.iter().enumerate().filter_map(|(i, &b)| (b == b'@').then_some(i)).collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo { at: authority_start + ats[1] });
    }

    let (userinfo, hostport, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: authority_start + at });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            authority_start + at + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    if hostport.starts_with('[') {
        let open = hostport_start;
        let close_rel = hostport.as_bytes().iter().position(|&b| b == b']');
        let close_rel = match close_rel {
            Some(v) => v,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &hostport[1..close_rel];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        let suffix = &hostport[close_rel + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_start + close_rel + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close_rel + 1,
            });
        };
        return Ok((userinfo, interior.to_ascii_lowercase(), port, true));
    }

    let colon = hostport.as_bytes().iter().position(|&b| b == b':');
    let (host, port) = match colon {
        Some(i) => (&hostport[..i], Some((&hostport[i + 1..], hostport_start + i + 1))),
        None => (hostport, None),
    };
    if host.is_empty() {
        return Err(ParseError::InvalidHost { at: hostport_start });
    }
    for (i, &b) in host.as_bytes().iter().enumerate() {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost { at: hostport_start + i });
        }
    }
    let parsed_port = match port {
        Some((s, at)) => Some(parse_port(s, at)?),
        None => None,
    };
    Ok((userinfo, host.to_ascii_lowercase(), parsed_port, false))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    validate_global_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphabetic() || b.is_ascii_digit() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let rest = &input[authority_start..];
    let authority_len = rest
        .as_bytes()
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(rest.len());
    let authority = &rest[..authority_len];
    let after_authority_at = authority_start + authority_len;
    let after_authority = &input[after_authority_at..];

    let (userinfo, host, port, ipv6) = parse_authority(authority, authority_start)?;

    let fragment_rel = after_authority.as_bytes().iter().position(|&b| b == b'#');
    let before_fragment = match fragment_rel {
        Some(i) => &after_authority[..i],
        None => after_authority,
    };
    let fragment = fragment_rel.map(|i| after_authority[i + 1..].to_owned());

    let query_rel = before_fragment.as_bytes().iter().position(|&b| b == b'?');
    let path = match query_rel {
        Some(i) => &before_fragment[..i],
        None => before_fragment,
    };
    let query = query_rel.map(|i| before_fragment[i + 1..].to_owned());
    let path_offset = after_authority_at;
    let query_offset = query_rel.map(|i| after_authority_at + i + 1);
    let fragment_offset = fragment_rel.map(|i| after_authority_at + i + 1);

    if let Some(ref u) = userinfo {
        validate_url_component(u, authority_start, Component::UserInfo)?;
    }
    if !path.is_empty() {
        validate_url_component(path, path_offset, Component::Path)?;
    }
    if let Some(ref q) = query {
        validate_url_component(q, query_offset.unwrap(), Component::Query)?;
    }
    if let Some(ref f) = fragment {
        validate_url_component(f, fragment_offset.unwrap(), Component::Fragment)?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path: if path.is_empty() { "/".to_owned() } else { path.to_owned() },
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded_path = decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let mut raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded_path[1..].split('/').collect();
    let terminal_navigation = decoded_segments
        .last()
        .map(|s| *s == "." || *s == "..")
        .unwrap_or(false);

    let mut stack: Vec<&str> = Vec::new();
    for (raw, decoded) in raw_segments.drain(..).zip(decoded_segments.into_iter()) {
        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(raw);
        }
    }
    if terminal_navigation {
        stack.push("");
    }
    Ok(format!("/{}", stack.join("/")))
}

fn raw_url_for_parts(url: &Url, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut out = String::new();
    out.push_str(&url.scheme);
    out.push_str("://");
    if let Some(u) = &url.userinfo {
        out.push_str(u);
        out.push('@');
    }
    if url.ipv6 {
        out.push('[');
        out.push_str(&url.host);
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        out.push(':');
        out.push_str(&port.to_string());
    }
    out.push_str(path);
    if let Some(q) = query {
        out.push('?');
        out.push_str(q);
    }
    if let Some(f) = fragment {
        out.push('#');
        out.push_str(f);
    }
    out
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_i = reference.as_bytes().iter().position(|&b| b == b'#');
    let before_fragment = match fragment_i {
        Some(i) => &reference[..i],
        None => reference,
    };
    let query_i = before_fragment.as_bytes().iter().position(|&b| b == b'?');
    let path = match query_i {
        Some(i) => &before_fragment[..i],
        None => before_fragment,
    };
    let query = query_i.map(|i| &before_fragment[i + 1..]);
    let fragment = fragment_i.map(|i| &reference[i + 1..]);
    (path, query, fragment)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    for (i, &b) in reference.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ResolveError::Reference(ParseError::InvalidByte { at: i }));
        }
    }

    let first_structural = reference
        .as_bytes()
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference.as_bytes()[..first_structural].contains(&b':') {
        let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    let (ref_path, ref_query, ref_fragment) = split_reference(reference);
    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_path.as_str().to_owned(),
            if ref_query.is_some() {
                ref_query
            } else {
                base_url.query.as_deref()
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
            format!("{slash}{ref_path}")
        };
        (merged, ref_query)
    };

    let constructed = raw_url_for_parts(
        &base_url,
        &selected_path,
        selected_query,
        ref_fragment,
    );
    let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
    u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
    Ok(u)
}

fn global_scan(input: &str) -> Result<(), usize> {
    for (i, &b) in input.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(i);

    }
    Ok(())

fn is_scheme_first(b: u8) -> bool {
    b.is_ascii_alphabetic()
}

fn is_scheme_rest(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')
}

fn parse_port(raw: &str, start: usize) -> Result<Option<u16>, ParseError> {
    if raw.is_empty() {
        return Err(ParseError::InvalidPort { at: start });
    }
    let bytes = raw.as_bytes();
    if let Some(i) = bytes.iter().position(|b| !b.is_ascii_digit()) {
        return Err(ParseError::InvalidPort { at: start + i });
    }
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: start });
    }
    let mut value: u32 = 0;
    for &b in bytes {
        value = match value.checked_mul(10).and_then(|v| v.checked_add((b - b'0') as u32)) {
            Some(v) => v,
            None => return Err(ParseError::InvalidPort { at: start }),
        };
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: start });
    }
    Ok(Some(value as u16))
}

fn validate_host(raw: &str, start: usize) -> Result<String, ParseError> {
    if raw.is_empty() {
        return Err(ParseError::InvalidHost { at: start });
    }
    for (i, &b) in raw.as_bytes().iter().enumerate() {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost { at: start + i });
        }
    }
    Ok(raw.to_ascii_lowercase())
}

fn codec_to_parse(err: CodecError, base: usize) -> ParseError {
    match err {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    global_scan(input).map_err(|at| ParseError::InvalidByte { at })?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !is_scheme_first(bytes[0]) {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        if !is_scheme_rest(bytes[i]) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let auth_start = colon + 3;
    let tail = &input[auth_start..];
    let auth_len = tail
        .as_bytes()
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(tail.len());
    let authority = &tail[..auth_len];
    let auth_end = auth_start + auth_len;

    let ats: Vec<usize> = authority
        .as_bytes()
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| if b == b'@' { Some(i) } else { None })
        .collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo { at: auth_start + ats[1] });
    }
    let (userinfo, hostport, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: auth_start });
        }
        (
            Some(authority[..at].to_string()),
            &authority[at + 1..],
            auth_start + at + 1,
        )
    } else {
        (None, authority, auth_start)
    };

    let mut ipv6 = false;
    let (host, port) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes().iter().position(|&b| b == b']') {
            Some(i) => i,
            None => return Err(ParseError::InvalidIpv6 { at: hostport_start }),
        };
        let interior = &hostport[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: hostport_start });
        }
        let suffix = &hostport[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            parse_port(&suffix[1..], hostport_start + close + 2)?
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close + 1,
            });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), parsed_port)
    } else {
        let split = hostport.as_bytes().iter().position(|&b| b == b':');
        let (raw_host, parsed_port) = if let Some(i) = split {
            (
                &hostport[..i],
                parse_port(&hostport[i + 1..], hostport_start + i + 1)?,
            )
        } else {
            (hostport, None)
        };
        (validate_host(raw_host, hostport_start)?, parsed_port)
    };

    let rest = &input[auth_end..];
    let fragment_pos = rest.as_bytes().iter().position(|&b| b == b'#');
    let before_fragment = fragment_pos.map_or(rest, |i| &rest[..i]);
    let fragment = fragment_pos.map(|i| rest[i + 1..].to_string());
    let query_pos = before_fragment.as_bytes().iter().position(|&b| b == b'?');
    let path_raw = query_pos.map_or(before_fragment, |i| &before_fragment[..i]);
    let query = query_pos.map(|i| before_fragment[i + 1..].to_string());
    let path = if path_raw.is_empty() { "/".to_string() } else { path_raw.to_string() };

    let mut component_start = auth_end + path_raw.len();
    if let Some(q) = query.as_ref() {
        decode_component(Component::Query, q)
            .map_err(|e| codec_to_parse(e, component_start + 1))?;
        component_start += 1 + q.len();
    }
    if let Some(f) = fragment.as_ref() {
        decode_component(Component::Fragment, f)
            .map_err(|e| codec_to_parse(e, component_start + 1))?;
    }
    if let Some(u) = userinfo.as_ref() {
        decode_component(Component::UserInfo, u)
            .map_err(|e| codec_to_parse(e, auth_start))?;
    }
    if !path_raw.is_empty() {
        decode_component(Component::Path, path_raw)
            .map_err(|e| codec_to_parse(e, auth_end))?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(h), Some(l)) => {
                    decoded.push((h << 4) | l);
                    origins.push(i);
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_string()),
        Err(e) => Err(CodecError::InvalidUtf8 {
            at: origins.get(e.valid_up_to()).copied().unwrap_or(input.len()),
        }),
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let segments: Vec<&str> = path[1..].split('/').collect();
    let mut decoded_segments = Vec::with_capacity(segments.len());
    for segment in &segments {
        let decoded = decode_component(Component::Path, segment).map_err(|e| match e {
            CodecError::BadEscape { at } => PathError::BadEscape { at: at + 1 },
            CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at: at + 1 },
        })?;
        decoded_segments.push(decoded);
    }
    let terminal_nav = decoded_segments
        .last()
        .map_or(false, |s| s == "." || s == "..");
    let mut stack: Vec<&str> = Vec::new();
    for (raw, decoded) in segments.iter().zip(decoded_segments.iter()) {
        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(raw);
        }
    }
    if terminal_nav {
        stack.push("");
    }
    let joined = stack.join("/");
    if joined.is_empty() {
        Ok("/".to_string())
    } else {
        Ok(format!("/{joined}"))
    }
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let mut base_url = parse_url(base).map_err(ResolveError::Base)?;
    base_url.path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;
    global_scan(reference)
        .map_err(|at| ResolveError::Reference(ParseError::InvalidByte { at }))?;

    let rb = reference.as_bytes();
    let first_boundary = rb
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(rb.len());
    if rb[..first_boundary].contains(&b':') {
        let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&absolute).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    let fragment_pos = reference.find('#');
    let before_fragment = fragment_pos.map_or(reference, |i| &reference[..i]);
    let fragment = fragment_pos.map(|i| reference[i + 1..].to_string());
    let query_pos = before_fragment.find('?');
    let ref_path = query_pos.map_or(before_fragment, |i| &before_fragment[..i]);
    let ref_query = query_pos.map(|i| before_fragment[i + 1..].to_string());

    let (selected_path, selected_query) = if ref_path.is_empty() {
        (base_url.path.clone(), if query_pos.is_none() { base_url.query.clone() } else { ref_query })
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_string()
        } else {
            let slash = base_url.path.rfind('/').map(|i| &base_url.path[..=i]).unwrap_or("/");
            format!("{slash}{ref_path}")
        };
        (merged, ref_query)
    };

    let mut absolute = format!("{}://", base_url.scheme);
    if let Some(ui) = base_url.userinfo.as_ref() {
        absolute.push_str(ui);
        absolute.push('@');
    }
    if base_url.ipv6 {
        absolute.push('[');
        absolute.push_str(&base_url.host);
        absolute.push(']');
    } else {
        absolute.push_str(&base_url.host);
    }
    if let Some(port) = base_url.port {
        absolute.push(':');
        absolute.push_str(&port.to_string());
    }
    absolute.push_str(&selected_path);
    if let Some(q) = selected_query.as_ref() {
        absolute.push('?');
        absolute.push_str(q);
    }
    if let Some(f) = fragment.as_ref() {
        absolute.push('#');
        absolute.push_str(f);
    }

    let mut u = parse_url(&absolute).map_err(ResolveError::Reference)?;
    u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
    Ok(u)
}

    Ok(Url{scheme,userinfo:ui,host:host_name,port:pn,path,query,fragment:frag,ipv6})
}
pub fn decode_component(_component: Component, s: &str) -> Result<String, CodecError> {
    let b=s.as_bytes(); let mut out=Vec::with_capacity(b.len()); let mut origin=Vec::with_capacity(b.len()); let mut i=0;
    while i<b.len() {
        if b[i]==b'%' {
            if i+2>=b.len() {return Err(CodecError::BadEscape{at:i});}
            match (hv(b[i+1]),hv(b[i+2])) { (Some(a),Some(z))=>{out.push(a*16+z);origin.push(i);}, _=>return Err(CodecError::BadEscape{at:i}) }
            i+=3;
        } else {out.push(b[i]);origin.push(i);i+=1;}
    }
    match std::str::from_utf8(&out) {Ok(x)=>Ok(x.to_owned()),Err(e)=>Err(CodecError::InvalidUtf8{at:origin.get(e.valid_up_to()).copied().unwrap_or(s.len())})}
}
pub fn encode_component(c: Component, s: &str) -> String {

        let safe=b.is_ascii_alphanumeric()||matches!(b,b'-'|b'.'|b'_'|b'~')||match c {Component::UserInfo=>b==b':',Component::Path=>b==b'/',Component::Query=>matches!(b,b'/'|b'?'),Component::Fragment=>matches!(b,b'/'|b'?'|b'#')};

    }
    out
}
pub fn normalise_path(s: &str) -> Result<String, PathError> {
    if !s.starts_with('/') {return Err(PathError::NotAbsolute);}
    decode_component(Component::Path,s).map_err(|e|match e{CodecError::BadEscape{at}=>PathError::BadEscape{at},CodecError::InvalidUtf8{at}=>PathError::InvalidUtf8{at}})?;
    let seg:Vec<&str>=s[1..].split('/').collect();
    let mut stack:Vec<&str>=Vec::new(); let mut last_nav=false;
    for raw in &seg {
        let d=decode_component(Component::Path,raw).unwrap();
        if d=="." {last_nav=raw==seg.last().unwrap();continue}
        if d==".." {stack.pop();last_nav=raw==seg.last().unwrap();continue}
        last_nav=false;stack.push(raw);
    }
    if last_nav {stack.push("");}
    if stack.is_empty(){Ok("/".to_string())}else{Ok(format!("/{}",stack.join("/")))}
}
pub fn resolve_url(base:&str, reference:&str)->Result<Url,ResolveError>{

fn parse_port(s: &str, colon_abs: usize) -> Result<u16, ParseError> {
    if s.is_empty() { return Err(ParseError::InvalidPort { at: colon_abs + 1 }); }
    for (i, b) in s.bytes().enumerate() {
        if !b.is_ascii_digit() { return Err(ParseError::InvalidPort { at: colon_abs + 1 + i }); }
    }

        if n > 65535 { return Err(ParseError::InvalidPort { at: colon_abs + 1 }); }
    }

    Ok(n as u16)

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() { return Err(ParseError::Empty); }
    global_scan(input)?;
    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i, None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !is_alpha(bytes[0]) { return Err(ParseError::InvalidScheme { at: 0 }); }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphanumeric() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }
    let auth_start = colon + 3;
    let tail = &input[auth_start..];
    let auth_len = tail.find(&['/', '?', '#'][..]).unwrap_or(tail.len());
    let auth = &tail[..auth_len];
    let auth_end = auth_start + auth_len;

    let ats: Vec<usize> = auth.bytes().enumerate().filter_map(|(i,b)| if b == b'@' {Some(i)} else {None}).collect();
    if ats.len() >= 2 { return Err(ParseError::InvalidUserInfo { at: auth_start + ats[1] }); }
    let (userinfo, hostpart, host_start) = if let Some(&i) = ats.first() {
        if i == 0 { return Err(ParseError::InvalidUserInfo { at: auth_start }); }
        (Some(auth[..i].to_string()), &auth[i+1..], auth_start+i+1)
    } else { (None, auth, auth_start) };

    let mut ipv6 = false;
    let (host, port) = if hostpart.starts_with('[') {
        let open = host_start;
        let close = match hostpart.find(']') {
            Some(i) => i, None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let inner = &hostpart[1..close];
        if inner.contains('%') || inner.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        ipv6 = true;
        let suffix = &hostpart[close+1..];
        let host = inner.to_ascii_lowercase();
        let port = if suffix.is_empty() { None } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], host_start + close + 1)?)
        } else { return Err(ParseError::InvalidHost { at: host_start + close + 1 }); };
        (host, port)
    } else {
        let split = hostpart.find(':');
        let (h, p) = match split { Some(i) => (&hostpart[..i], Some((i, &hostpart[i+1..]))), None => (hostpart, None) };
        if h.is_empty() { return Err(ParseError::InvalidHost { at: host_start }); }
        for (i,b) in h.bytes().enumerate() {
            if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: host_start+i });
            }
        }
        let port = match p { Some((i,p)) => Some(parse_port(p, host_start+i)?), None => None };
        (h.to_ascii_lowercase(), port)
    };

    let rest_start = auth_end;
    let rest = &input[rest_start..];
    let frag_pos = rest.find('#');
    let before_frag = frag_pos.map(|i| &rest[..i]).unwrap_or(rest);
    let fragment = frag_pos.map(|i| rest[i+1..].to_string());
    let query_pos = before_frag.find('?');
    let path_raw = query_pos.map(|i| &before_frag[..i]).unwrap_or(before_frag);
    let query = query_pos.map(|i| before_frag[i+1..].to_string());
    let path = if path_raw.is_empty() { "/".to_string() } else { path_raw.to_string() };

    if let Some(u) = &userinfo { validate_component(u, auth_start)?; }
    if !path_raw.is_empty() { validate_component(path_raw, rest_start)?; }
    if let Some(q) = &query {
        let qbase = rest_start + query_pos.unwrap() + 1;
        validate_component(q, qbase)?;
    }
    if let Some(f) = &fragment {
        let fbase = rest_start + frag_pos.unwrap() + 1;
        validate_component(f, fbase)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(), userinfo, host, port, path, query, fragment, ipv6
    })
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() || !is_hex(bytes[i+1]) || !is_hex(bytes[i+2]) {
                return Err(CodecError::BadEscape { at: i });
            }
            out.push(hex(bytes[i+1]) * 16 + hex(bytes[i+2]));
            origins.push(i);
            i += 3;
        } else {
            out.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }
    match std::str::from_utf8(&out) {
        Ok(s) => Ok(s.to_string()),
        Err(e) => Err(CodecError::InvalidUtf8 { at: origins[e.valid_up_to()] }),
    }
}
pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::new();
    for b in text.bytes() {
        let safe = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe { out.push(b as char); }
        else {
            const HEX: &[u8;16] = b"0123456789ABCDEF";
            out.push('%'); out.push(HEX[(b >> 4) as usize] as char); out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}
pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') { return Err(PathError::NotAbsolute); }
    let decoded = decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;
    let raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded[1..].split('/').collect();
    let terminal_nav = decoded_segments.last().map_or(false, |s| *s == "." || *s == "..");
    let mut stack: Vec<&str> = Vec::new();
    for (raw, dec) in raw_segments.iter().zip(decoded_segments.iter()) {
        if *dec == "." { continue; }
        if *dec == ".." { stack.pop(); }
        else { stack.push(*raw); }
    }
    if terminal_nav { stack.push(""); }
    let mut result = String::from("/");
    result.push_str(&stack.join("/"));
    Ok(result)
}
fn path_error_map(e: PathError) -> ResolveError { ResolveError::Path(e) }
fn normalize_url(mut u: Url) -> Result<Url, ResolveError> {
    u.path = normalise_path(&u.path).map_err(path_error_map)?;
    Ok(u)
}
fn has_absolute_scheme(reference: &str) -> bool {
    let stop = reference.find(&['/', '?', '#'][..]).unwrap_or(reference.len());
    reference[..stop].contains(':')
}
pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_url = normalize_url(base_url)?;
    global_scan(reference).map_err(ResolveError::Reference)?;
    if reference.starts_with("//") {
        let full = format!("{}:{}", base_url.scheme, reference);
        return normalize_url(parse_url(&full).map_err(ResolveError::Reference)?);
    }
    if has_absolute_scheme(reference) {
        return normalize_url(parse_url(reference).map_err(ResolveError::Reference)?);
    }

    let frag_pos = reference.find('#');
    let before_frag = frag_pos.map(|i| &reference[..i]).unwrap_or(reference);
    let frag = frag_pos.map(|i| reference[i+1..].to_string());
    let query_pos = before_frag.find('?');
    let ref_path = query_pos.map(|i| &before_frag[..i]).unwrap_or(before_frag);
    let ref_query = query_pos.map(|i| before_frag[i+1..].to_string());

    let (path, query) = if ref_path.is_empty() {
        (base_url.path.clone(), if query_pos.is_some() { ref_query } else { base_url.query.clone() })
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_string()
        } else {
            let slash = base_url.path.rfind('/').map(|i| i+1).unwrap_or(0);
            format!("{}{}", &base_url.path[..slash], ref_path)
        };
        (merged, ref_query)
    };
    let userinfo = base_url.userinfo.clone();
    let host = if base_url.ipv6 { format!("[{}]", base_url.host) } else { base_url.host.clone() };
    let mut full = format!("{}://", base_url.scheme);
    if let Some(u) = &userinfo { full.push_str(u); full.push('@'); }
    full.push_str(&host);
    if let Some(p) = base_url.port { full.push(':'); full.push_str(&p.to_string()); }
    full.push_str(&path);
    if let Some(q) = &query { full.push('?'); full.push_str(q); }
    if let Some(f) = &frag { full.push('#'); full.push_str(f); }
    normalize_url(parse_url(&full).map_err(ResolveError::Reference)?)
}

        None => (None, auth, start),
    };
    let (host, port, ipv6) = if hp.starts_with('[') {
        let close = match hp.find(']') { Some(i)=>i, None=>return Err(ParseError::InvalidIpv6 { at: hp_at }) };
        let inside = &hp[1..close];
        if inside.contains('%') || inside.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 { at: hp_at });
        }
        let suffix = &hp[close+1..];
        let port = if suffix.is_empty() { None } else if suffix.as_bytes()[0] == b':' {
            Some(valid_port(&suffix[1..], hp_at+close+2)?)
        } else { return Err(ParseError::InvalidHost { at: hp_at+close+1 }); };
        (inside.to_ascii_lowercase(), port, true)
    } else {
        let colon = hp.find(':');
        let (h, p, pstart) = match colon {
            Some(i) => (&hp[..i], Some(&hp[i+1..]), hp_at+i+1),
            None => (hp, None, 0),
        };
        if h.is_empty() { return Err(ParseError::InvalidHost { at: hp_at }); }
        if let Some(i) = h.bytes().position(|x| !(x.is_ascii_alphanumeric() || matches!(x,b'.'|b'-'|b'_'))) {
            return Err(ParseError::InvalidHost { at: hp_at+i });
        }
        let port = match p { Some(s) => Some(valid_port(s,pstart)?), None=>None };
        (h.to_ascii_lowercase(), port, false)
    };
    let rest = &input[auth_at..];
    let fp = rest.find('#');
    let pre = fp.map(|i| &rest[..i]).unwrap_or(rest);
    let fragment = fp.map(|i| rest[i+1..].to_string());
    let qp = pre.find('?');
    let p_raw = qp.map(|i| &pre[..i]).unwrap_or(pre);
    let query = qp.map(|i| pre[i+1..].to_string());
    let path = if p_raw.is_empty() { "/".into() } else { p_raw.into() };
    if let Some(u) = &userinfo { component_check(u,start)?; }
    if !p_raw.is_empty() { component_check(p_raw,auth_at)?; }
    if let Some(q) = &query { component_check(q,auth_at+qp.unwrap()+1)?; }
    if let Some(f) = &fragment { component_check(f,auth_at+fp.unwrap()+1)?; }
    Ok(Url { scheme: input[..ci].to_ascii_lowercase(), userinfo, host, port, path, query, fragment, ipv6 })
}
pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let mut out = Vec::with_capacity(input.len());
    let mut src = Vec::with_capacity(input.len());
    let b = input.as_bytes();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'%' {
            if i+2 >= b.len() || !is_hex(b[i+1]) || !is_hex(b[i+2]) { return Err(CodecError::BadEscape { at:i }); }
            out.push(hex(b[i+1])*16+hex(b[i+2]));
            src.push(i);
            i+=3;
        } else {
            out.push(b[i]); src.push(i); i+=1;
        }
    }
    match std::str::from_utf8(&out) {
        Ok(s)=>Ok(s.to_owned()),
        Err(e)=>Err(CodecError::InvalidUtf8 { at: src[e.valid_up_to()] }),
    }
}
pub fn encode_component(component: Component, text: &str) -> String {
    let mut out=String::new();
    const H:&[u8;16]=b"0123456789ABCDEF";
    for b in text.bytes() {
        let safe=b.is_ascii_alphanumeric()||matches!(b,b'-'|b'.'|b'_'|b'~')||
            match component { Component::UserInfo=>b==b':', Component::Path=>b==b'/', Component::Query=>matches!(b,b'/'|b'?'), Component::Fragment=>matches!(b,b'/'|b'?'|b'#') };
        if safe { out.push(b as char); } else { out.push('%'); out.push(H[(b>>4) as usize] as char); out.push(H[(b&15) as usize] as char); }
    }
    out
}
pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') { return Err(PathError::NotAbsolute); }
    let decoded=decode_component(Component::Path,path).map_err(|e|match e { CodecError::BadEscape{at}=>PathError::BadEscape{at}, CodecError::InvalidUtf8{at}=>PathError::InvalidUtf8{at} })?;
    let raw:Vec<&str>=path[1..].split('/').collect();
    let dec:Vec<&str>=decoded[1..].split('/').collect();
    let terminal=dec.last().map_or(false,|x|*x=="."||*x=="..");
    let mut st:Vec<&str>=Vec::new();
    for (r,d) in raw.iter().zip(dec.iter()) {
        match *d { "."=>{}, ".."=>{st.pop();}, _=>st.push(*r) }
    }
    if terminal { st.push(""); }
    Ok(format!("/{}",st.join("/")))
}
fn normalized(mut u:Url)->Result<Url,ResolveError>{u.path=normalise_path(&u.path).map_err(ResolveError::Path)?;Ok(u)}
fn absolute_marker(s:&str)->bool {
    let end=s.find(&['/','?','#'][..]).unwrap_or(s.len());
    s[..end].contains(':')
}
pub fn resolve_url(base:&str,reference:&str)->Result<Url,ResolveError>{
    let b=normalized(parse_url(base).map_err(ResolveError::Base)?)?;
    scan_global(reference).map_err(ResolveError::Reference)?;
    if reference.starts_with("//") {
        return normalized(parse_url(&format!("{}:{}",b.scheme,reference)).map_err(ResolveError::Reference)?);
    }
    if absolute_marker(reference) {
        return normalized(parse_url(reference).map_err(ResolveError::Reference)?);
    }
    let fp=reference.find('#');
    let before=fp.map(|i|&reference[..i]).unwrap_or(reference);
    let frag=fp.map(|i|reference[i+1..].to_string());
    let qp=before.find('?');
    let rp=qp.map(|i|&before[..i]).unwrap_or(before);
    let rq=qp.map(|i|before[i+1..].to_string());
    let (path,query)=if rp.is_empty(){
        (b.path.clone(),if qp.is_some(){rq}else{b.query.clone()})
    }else{
        let p=if rp.starts_with('/') {rp.to_string()} else {
            let i=b.path.rfind('/').unwrap_or(0)+1;
            format!("{}{}",&b.path[..i],rp)
        };
        (p,rq)
    };
    let host=if b.ipv6{format!("[{}]",b.host)}else{b.host.clone()};
    let mut raw=format!("{}://",b.scheme);
    if let Some(u)=&b.userinfo{raw.push_str(u);raw.push('@');}
    raw.push_str(&host);
    if let Some(p)=b.port{raw.push(':');raw.push_str(&p.to_string());}
    raw.push_str(&path);
    if let Some(q)=&query{raw.push('?');raw.push_str(q);}
    if let Some(f)=&frag{raw.push('#');raw.push_str(f);}
    normalized(parse_url(&raw).map_err(ResolveError::Reference)?)
}
