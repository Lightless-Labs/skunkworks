fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut origins = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            if hi.is_none() || lo.is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            decoded.push((hi.unwrap() << 4) | lo.unwrap());
            origins.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(s) => Ok(s.to_owned()),
        Err(e) => {
            let at = origins.get(e.valid_up_to()).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut out = String::with_capacity(text.len());
    for &b in text.as_bytes() {
        let safe = b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => b == b':',
                Component::Path => b == b'/',
                Component::Query => matches!(b, b'/' | b'?'),
                Component::Fragment => matches!(b, b'/' | b'?' | b'#'),
            };
        if safe {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
    }
    out
}

fn parse_port(port_text: &str, port_start: usize) -> Result<u16, ParseError> {
    if port_text.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    let bytes = port_text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: port_start + i });
        }
    }
    if bytes.len() >= 2 && bytes[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for &b in bytes {
        value = value.saturating_mul(10).saturating_add((b - b'0') as u32);
        if value > 65535 {
            return Err(ParseError::InvalidPort { at: port_start });
        }
    }
    if value == 0 {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

fn parse_authority(
    authority: &str,
    authority_start: usize,
) -> Result<(Option<String>, String, Option<u16>, bool), ParseError> {
    let bytes = authority.as_bytes();
    let ats: Vec<usize> = bytes.iter().enumerate().filter_map(|(i, &b)| (b == b'@').then_some(i)).collect();
    if ats.len() >= 2 {
        return Err(ParseError::InvalidUserInfo { at: authority_start + ats[1] });
    }

    let (userinfo, hostport, hostport_start) = if let Some(&at) = ats.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo { at: authority_start + at });
        }
        (
            Some(authority[..at].to_owned()),
            &authority[at + 1..],
            authority_start + at + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    if hostport.starts_with('[') {
        let open = hostport_start;
        let close_rel = hostport.as_bytes().iter().position(|&b| b == b']');
        let close_rel = match close_rel {
            Some(v) => v,
            None => return Err(ParseError::InvalidIpv6 { at: open }),
        };
        let interior = &hostport[1..close_rel];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open });
        }
        let suffix = &hostport[close_rel + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(&suffix[1..], hostport_start + close_rel + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_start + close_rel + 1,
            });
        };
        return Ok((userinfo, interior.to_ascii_lowercase(), port, true));
    }

    let colon = hostport.as_bytes().iter().position(|&b| b == b':');
    let (host, port) = match colon {
        Some(i) => (&hostport[..i], Some((&hostport[i + 1..], hostport_start + i + 1))),
        None => (hostport, None),
    };
    if host.is_empty() {
        return Err(ParseError::InvalidHost { at: hostport_start });
    }
    for (i, &b) in host.as_bytes().iter().enumerate() {
        if !(b.is_ascii_alphanumeric() || matches!(b, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost { at: hostport_start + i });
        }
    }
    let parsed_port = match port {
        Some((s, at)) => Some(parse_port(s, at)?),
        None => None,
    };
    Ok((userinfo, host.to_ascii_lowercase(), parsed_port, false))
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    validate_global_bytes(input)?;

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(i) => i,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for i in 1..colon {
        let b = bytes[i];
        if !(b.is_ascii_alphabetic() || b.is_ascii_digit() || matches!(b, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: i });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let rest = &input[authority_start..];
    let authority_len = rest
        .as_bytes()
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(rest.len());
    let authority = &rest[..authority_len];
    let after_authority_at = authority_start + authority_len;
    let after_authority = &input[after_authority_at..];

    let (userinfo, host, port, ipv6) = parse_authority(authority, authority_start)?;

    let fragment_rel = after_authority.as_bytes().iter().position(|&b| b == b'#');
    let before_fragment = match fragment_rel {
        Some(i) => &after_authority[..i],
        None => after_authority,
    };
    let fragment = fragment_rel.map(|i| after_authority[i + 1..].to_owned());

    let query_rel = before_fragment.as_bytes().iter().position(|&b| b == b'?');
    let path = match query_rel {
        Some(i) => &before_fragment[..i],
        None => before_fragment,
    };
    let query = query_rel.map(|i| before_fragment[i + 1..].to_owned());
    let path_offset = after_authority_at;
    let query_offset = query_rel.map(|i| after_authority_at + i + 1);
    let fragment_offset = fragment_rel.map(|i| after_authority_at + i + 1);

    if let Some(ref u) = userinfo {
        validate_url_component(u, authority_start, Component::UserInfo)?;
    }
    if !path.is_empty() {
        validate_url_component(path, path_offset, Component::Path)?;
    }
    if let Some(ref q) = query {
        validate_url_component(q, query_offset.unwrap(), Component::Query)?;
    }
    if let Some(ref f) = fragment {
        validate_url_component(f, fragment_offset.unwrap(), Component::Fragment)?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path: if path.is_empty() { "/".to_owned() } else { path.to_owned() },
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded_path = decode_component(Component::Path, path).map_err(|e| match e {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let mut raw_segments: Vec<&str> = path[1..].split('/').collect();
    let decoded_segments: Vec<&str> = decoded_path[1..].split('/').collect();
    let terminal_navigation = decoded_segments
        .last()
        .map(|s| *s == "." || *s == "..")
        .unwrap_or(false);

    let mut stack: Vec<&str> = Vec::new();
    for (raw, decoded) in raw_segments.drain(..).zip(decoded_segments.into_iter()) {
        if decoded == "." {
            continue;
        } else if decoded == ".." {
            stack.pop();
        } else {
            stack.push(raw);
        }
    }
    if terminal_navigation {
        stack.push("");
    }
    Ok(format!("/{}", stack.join("/")))
}

fn raw_url_for_parts(url: &Url, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut out = String::new();
    out.push_str(&url.scheme);
    out.push_str("://");
    if let Some(u) = &url.userinfo {
        out.push_str(u);
        out.push('@');
    }
    if url.ipv6 {
        out.push('[');
        out.push_str(&url.host);
        out.push(']');
    } else {
        out.push_str(&url.host);
    }
    if let Some(port) = url.port {
        out.push(':');
        out.push_str(&port.to_string());
    }
    out.push_str(path);
    if let Some(q) = query {
        out.push('?');
        out.push_str(q);
    }
    if let Some(f) = fragment {
        out.push('#');
        out.push_str(f);
    }
    out
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let fragment_i = reference.as_bytes().iter().position(|&b| b == b'#');
    let before_fragment = match fragment_i {
        Some(i) => &reference[..i],
        None => reference,
    };
    let query_i = before_fragment.as_bytes().iter().position(|&b| b == b'?');
    let path = match query_i {
        Some(i) => &before_fragment[..i],
        None => before_fragment,
    };
    let query = query_i.map(|i| &before_fragment[i + 1..]);
    let fragment = fragment_i.map(|i| &reference[i + 1..]);
    (path, query, fragment)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let base_path = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    for (i, &b) in reference.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&b) {
            return Err(ResolveError::Reference(ParseError::InvalidByte { at: i }));
        }
    }

    let first_structural = reference
        .as_bytes()
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference.as_bytes()[..first_structural].contains(&b':') {
        let mut u = parse_url(reference).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    if reference.starts_with("//") {
        let constructed = format!("{}:{}", base_url.scheme, reference);
        let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
        u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
        return Ok(u);
    }

    let (ref_path, ref_query, ref_fragment) = split_reference(reference);
    let (selected_path, selected_query) = if ref_path.is_empty() {
        (
            base_path.as_str().to_owned(),
            if ref_query.is_some() {
                ref_query
            } else {
                base_url.query.as_deref()
            },
        )
    } else {
        let merged = if ref_path.starts_with('/') {
            ref_path.to_owned()
        } else {
            let slash = base_path.rfind('/').map(|i| &base_path[..=i]).unwrap_or("/");
            format!("{slash}{ref_path}")
        };
        (merged, ref_query)
    };

    let constructed = raw_url_for_parts(
        &base_url,
        &selected_path,
        selected_query,
        ref_fragment,
    );
    let mut u = parse_url(&constructed).map_err(ResolveError::Reference)?;
    u.path = normalise_path(&u.path).map_err(ResolveError::Path)?;
    Ok(u)
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(component: Component, input: &str) -> Result<String, CodecError> {
    let _ = component;
    let mut decoded = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let bytes = input.as_bytes();
    let mut i = 0usize;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i.checked_add(2).filter(|&j| j < bytes.len()).is_none() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = match hex_value(bytes[i + 1]) {
                Some(v) => v,
                None => return Err(CodecError::BadEscape { at: i }),
            };
            let lo = match hex_value(bytes[i + 2]) {
                Some(v) => v,
                None => return Err(CodecError::BadEscape { at: i }),
            };
            decoded.push((hi << 4) | lo);
            positions.push(i);
            i += 3;
        } else {
            decoded.push(bytes[i]);
            positions.push(i);
            i += 1;
        }
    }

    std::str::from_utf8(&decoded)
        .map(str::to_owned)
        .map_err(|error| CodecError::InvalidUtf8 {
            at: positions.get(error.valid_up_to()).copied().unwrap_or(input.len()),
        })
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    for &byte in text.as_bytes() {
        let allowed_extra = match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => byte == b'/' || byte == b'?',
            Component::Fragment => byte == b'/' || byte == b'?' || byte == b'#',
        };
        if byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || allowed_extra
        {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

fn global_scan(input: &str) -> Result<(), ParseError> {
    for (offset, &byte) in input.as_bytes().iter().enumerate() {
        if byte < 0x21 || byte > 0x7e {
            return Err(ParseError::InvalidByte { at: offset });
        }
    }
    Ok(())
}

fn component_parse_error(error: CodecError, offset: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: offset + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: offset + at },
    }
}

fn validate_component(value: &str, offset: usize, component: Component) -> Result<(), ParseError> {
    decode_component(component, value)
        .map(|_| ())
        .map_err(|error| component_parse_error(error, offset))
}

fn parse_port(text: &str, offset: usize) -> Result<u16, ParseError> {
    if text.is_empty() {
        return Err(ParseError::InvalidPort { at: offset });
    }
    for (i, b) in text.bytes().enumerate() {
        if !b.is_ascii_digit() {
            return Err(ParseError::InvalidPort { at: offset + i });
        }
    }
    let digits = text.as_bytes();
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(ParseError::InvalidPort { at: offset });
    }
    let mut number = 0u32;
    for &digit in digits {
        number = match number.checked_mul(10).and_then(|n| n.checked_add((digit - b'0') as u32)) {
            Some(value) if value <= 65535 => value,
            _ => return Err(ParseError::InvalidPort { at: offset }),
        };
    }
    if number == 0 {
        return Err(ParseError::InvalidPort { at: offset });
    }
    Ok(number as u16)
}

fn parse_authority(
    authority: &str,
    start: usize,
) -> Result<(Option<String>, String, Option<u16>, bool), ParseError> {
    let mut at = None;
    for (i, byte) in authority.bytes().enumerate() {
        if byte == b'@' {
            if at.is_some() {
                return Err(ParseError::InvalidUserInfo { at: start + i });
            }
            at = Some(i);
        }
    }

    let (userinfo, remainder, remainder_start) = match at {
        Some(i) if i == 0 => return Err(ParseError::InvalidUserInfo { at: start }),
        Some(i) => (Some(authority[..i].to_owned()), &authority[i + 1..], start + i + 1),
        None => (None, authority, start),
    };

    if remainder.starts_with('[') {
        let open_at = remainder_start;
        let Some(close) = remainder.bytes().position(|b| b == b']') else {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        };
        let address = &remainder[1..close];
        if Ipv6Addr::from_str(address).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        }
        let tail = &remainder[close + 1..];
        let port = if tail.is_empty() {
            None
        } else if tail.starts_with(':') {
            Some(parse_port(&tail[1..], remainder_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: remainder_start + close + 1,
            });
        };
        Ok((userinfo, address.to_ascii_lowercase(), port, true))
    } else {
        let colon = remainder.bytes().position(|b| b == b':');
        let (host, port) = match colon {
            Some(i) => (&remainder[..i], Some((&remainder[i + 1..], remainder_start + i + 1))),
            None => (remainder, None),
        };
        if host.is_empty() {
            return Err(ParseError::InvalidHost { at: remainder_start });
        }
        for (i, byte) in host.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost { at: remainder_start + i });
            }
        }
        let port = match port {
            Some((text, offset)) => Some(parse_port(text, offset)?),
            None => None,
        };
        Ok((userinfo, host.to_ascii_lowercase(), port, false))
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }
    global_scan(input)?;

    let bytes = input.as_bytes();
    let Some(colon) = bytes.iter().position(|&b| b == b':') else {
        return Err(ParseError::InvalidScheme { at: input.len() });
    };
    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for index in 1..colon {
        let byte = bytes[index];
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at: index });
        }
    }
    let scheme = input[..colon].to_ascii_lowercase();

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let rest = &input[authority_start..];
    let authority_len = rest
        .bytes()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .unwrap_or(rest.len());
    let authority = &rest[..authority_len];
    let component_start = authority_start + authority_len;
    let component_text = &input[component_start..];

    let (userinfo, host, port, ipv6) = parse_authority(authority, authority_start)?;

    let fragment_offset = component_text.bytes().position(|b| b == b'#');
    let before_fragment = fragment_offset.map(|i| &component_text[..i]).unwrap_or(component_text);
    let fragment = fragment_offset.map(|i| &component_text[i + 1..].to_owned());

    let query_offset = before_fragment.bytes().position(|b| b == b'?');
    let path = query_offset.map(|i| &before_fragment[..i]).unwrap_or(before_fragment);
    let query = query_offset.map(|i| &before_fragment[i + 1..].to_owned());

    if let Some(ref value) = userinfo {
        validate_component(value, authority_start, Component::UserInfo)?;
    }
    if !path.is_empty() {
        validate_component(path, component_start, Component::Path)?;
    }
    if let Some(ref value) = query {
        validate_component(
            value,
            component_start + query_offset.unwrap() + 1,
            Component::Query,
        )?;
    }
    if let Some(ref value) = fragment {
        validate_component(
            value,
            component_start + fragment_offset.unwrap() + 1,
            Component::Fragment,
        )?;
    }

    Ok(Url {
        scheme,
        userinfo,
        host,
        port,
        path: if path.is_empty() { "/".to_owned() } else { path.to_owned() },
        query,
        fragment,
        ipv6,
    })
}

pub fn normalise_path(path: &str) -> Result<String, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::NotAbsolute);
    }
    let decoded = decode_component(Component::Path, path).map_err(|error| match error {
        CodecError::BadEscape { at } => PathError::BadEscape { at },
        CodecError::InvalidUtf8 { at } => PathError::InvalidUtf8 { at },
    })?;

    let raw: Vec<&str> = path[1..].split('/').collect();
    let logical: Vec<&str> = decoded[1..].split('/').collect();
    let terminal_navigation = logical.last().is_some_and(|segment| *segment == "." || *segment == "..");

    let mut stack: Vec<&str> = Vec::new();
    for (raw_segment, decoded_segment) in raw.into_iter().zip(logical) {
        match decoded_segment {
            "." => {}
            ".." => {
                stack.pop();
            }
            _ => stack.push(raw_segment),
        }
    }
    if terminal_navigation {
        stack.push("");
    }
    Ok(format!("/{}", stack.join("/")))
}

fn construct_url(url: &Url, path: &str, query: Option<&str>, fragment: Option<&str>) -> String {
    let mut text = format!("{}://", url.scheme);
    if let Some(userinfo) = &url.userinfo {
        text.push_str(userinfo);
        text.push('@');
    }
    if url.ipv6 {
        text.push('[');
        text.push_str(&url.host);
        text.push(']');
    } else {
        text.push_str(&url.host);
    }
    if let Some(port) = url.port {
        text.push(':');
        text.push_str(&port.to_string());
    }
    text.push_str(path);
    if let Some(query) = query {
        text.push('?');
        text.push_str(query);
    }
    if let Some(fragment) = fragment {
        text.push('#');
        text.push_str(fragment);
    }
    text
}

fn split_reference(reference: &str) -> (&str, Option<&str>, Option<&str>) {
    let hash = reference.bytes().position(|b| b == b'#');
    let before_hash = hash.map(|i| &reference[..i]).unwrap_or(reference);
    let question = before_hash.bytes().position(|b| b == b'?');
    let path = question.map(|i| &before_hash[..i]).unwrap_or(before_hash);
    let query = question.map(|i| &before_hash[i + 1..]);
    let fragment = hash.map(|i| &reference[i + 1..]);
    (path, query, fragment)
}

pub fn resolve_url(base: &str, reference: &str) -> Result<Url, ResolveError> {
    let base_url = parse_url(base).map_err(ResolveError::Base)?;
    let normalized_base = normalise_path(&base_url.path).map_err(ResolveError::Path)?;

    for (offset, &byte) in reference.as_bytes().iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ResolveError::Reference(ParseError::InvalidByte { at: offset }));
        }
    }

    let boundary = reference
        .bytes()
        .position(|byte| matches!(byte, b'/' | b'?' | b'#'))
        .unwrap_or(reference.len());
    if reference.as_bytes()[..boundary].contains(&b':') {
        let mut result = parse_url(reference).map_err(ResolveError::Reference)?;
        result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
        return Ok(result);
    }

    if reference.starts_with("//") {
        let absolute = format!("{}:{}", base_url.scheme, reference);
        let mut result = parse_url(&absolute).map_err(ResolveError::Reference)?;
        result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
        return Ok(result);
    }

    let (reference_path, reference_query, reference_fragment) = split_reference(reference);
    let (path, query) = if reference_path.is_empty() {
        (
            normalized_base,
            reference_query.or(base_url.query.as_deref()),
        )
    } else {
        let path = if reference_path.starts_with('/') {
            reference_path.to_owned()
        } else {
            let prefix = normalized_base
                .rfind('/')
                .map(|index| &normalized_base[..=index])
                .unwrap_or("/");
            format!("{prefix}{reference_path}")
        };
        (path, reference_query)
    };

    let absolute = construct_url(&base_url, &path, query, reference_fragment);
    let mut result = parse_url(&absolute).map_err(ResolveError::Reference)?;
    result.path = normalise_path(&result.path).map_err(ResolveError::Path)?;
    Ok(result)
}
