use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind { Dir, File, Link }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize), Missing(String), NotDir(String), IsDir(String), Exists(String),
    Root, Busy(String), Cycle(String), NotEmpty(String), InvalidTarget(usize),
    Loop(String), Depth(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self { Entry::Dir => Kind::Dir, Entry::File(_) => Kind::File, Entry::Link(_) => Kind::Link }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self { entries, cwd: "/".to_string() }
    }

    fn components(path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) { return Err(Error::InvalidPath(i)); }
        if path.is_empty() { return Err(Error::InvalidPath(0)); }
        Ok(path.split('/').filter(|s| !s.is_empty() && *s != "." && *s != "..").map(str::to_string).collect())
    }

    fn normalize_parts(base: &[String], path: &str) -> Vec<String> {
        let mut out = if path.starts_with('/') { Vec::new() } else { base.to_vec() };
        for c in path.split('/') {
            match c {
                "" | "." => {}
                ".." => { out.pop(); }
                _ => out.push(c.to_string()),
            }
        }
        out
    }

    fn canon(parts: &[String]) -> String {
        if parts.is_empty() { "/".to_string() } else { format!("/{}", parts.join("/")) }
    }

    fn parent(path: &str) -> (String, String) {
        let mut parts: Vec<&str> = path.split('/').filter(|s| !s.is_empty()).collect();
        let name = parts.pop().unwrap_or("").to_string();
        let p = if parts.is_empty() { "/".to_string() } else { format!("/{}", parts.join("/")) };
        (p, name)
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" { return true; }
        path == dir || path.strip_prefix(dir).map_or(false, |rest| rest.starts_with('/'))
    }

    fn resolve(&self, initial: Vec<String>, allow_missing_final: bool) -> Result<String, Error> {
        let mut todo = initial;
        let mut visited = HashSet::new();
        let mut expansions = 0usize;
        loop {
            let mut built: Vec<String> = Vec::new();
            let mut restarted = false;
            for idx in 0..todo.len() {
                built.push(todo[idx].clone());
                let p = Self::canon(&built);
                let entry = self.entries.get(&p).ok_or_else(|| {
                    if allow_missing_final && idx + 1 == todo.len() { Error::Missing(p.clone()) } else { Error::Missing(p.clone()) }
                });
                let entry = match entry {
                    Ok(e) => e,
                    Err(e) => return Err(e),
                };
                if idx + 1 < todo.len() && !matches!(entry, Entry::Dir | Entry::Link(_)) {
                    return Err(Error::NotDir(p));
                }
                if let Entry::Link(target) = entry {
                    if !visited.insert(p.clone()) { return Err(Error::Loop(p)); }
                    if expansions >= 8 { return Err(Error::Depth(p)); }
                    expansions += 1;
                    let mut target_parts = Self::normalize_parts(&built[..built.len()-1], target);
                    target_parts.extend_from_slice(&todo[idx+1..]);
                    todo = target_parts;
                    restarted = true;
                    break;
                }
            }
            if !restarted { return Ok(Self::canon(&todo)); }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent(path);
        let parent_parts = Self::components(&parent)?;
        let physical = self.resolve(parent_parts, false)?;
        match self.entries.get(&physical) {
            None => Err(Error::Missing(physical)),
            Some(Entry::Dir) => Ok((physical, name)),
            Some(_) => Err(Error::NotDir(physical)),
        }
    }

    fn direct_children(&self, path: &str) -> Vec<(String, Kind)> {
        let prefix = if path == "/" { "/".to_string() } else { format!("{}/", path) };
        self.entries.iter().filter_map(|(p, e)| {
            if p == path { return None; }
            let rest = p.strip_prefix(&prefix)?;
            if !rest.is_empty() && !rest.contains('/') { Some((rest.to_string(), e.kind())) } else { None }
        }).collect()
    }

    fn subtree_paths(&self, path: &str) -> Vec<String> {
        self.entries.keys().filter(|p| Self::is_within(p, path)).cloned().collect()
    }

    fn dir_size(&self, path: &str) -> u64 {
        self.subtree_paths(path).iter().filter_map(|p| match self.entries.get(p) {
            Some(Entry::File(b)) => Some(b.len() as u64),
            Some(Entry::Link(t)) => Some(t.len() as u64),
            _ => None,
        }).fold(0u64, u64::saturating_add)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = Self::components(path)?;
        let canonical = Self::canon(&Self::normalize_parts(&Self::components(&self.cwd)?, path));
        if canonical == "/" { return Err(Error::Exists("/".to_string())); }
        let (parent_path, name) = Self::parent(&canonical);
        let parent = self.resolve(Self::components(&parent_path)?, false)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            None => return Err(Error::Missing(parent)),
            Some(_) => return Err(Error::NotDir(parent)),
        }
        let dest = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        if self.entries.contains_key(&dest) { return Err(Error::Exists(dest)); }
        self.entries.insert(dest, Entry::Dir);
        let _ = parts;
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = Self::components(path)?;
        let normalized = Self::normalize_parts(&Self::components(&self.cwd)?, path);
        let target = self.resolve(normalized, true);
        let dest = match target {
            Ok(p) => p,
            Err(Error::Missing(p)) => p,
            Err(e) => return Err(e),
        };
        if dest == "/" { return Err(Error::IsDir("/".to_string())); }
        let (parent, _) = Self::parent(&dest);
        let parent = self.resolve(Self::components(&parent)?, false)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            None => return Err(Error::Missing(parent)),
            Some(_) => return Err(Error::NotDir(parent)),
        }
        match self.entries.get(&dest) {
            Some(Entry::Dir) => return Err(Error::IsDir(dest)),
            Some(Entry::File(_)) => { self.entries.insert(dest, Entry::File(data.to_vec())); }
            Some(Entry::Link(_)) => return Err(Error::Loop(dest)),
            None => { self.entries.insert(dest, Entry::File(data.to_vec())); }
        }
        let _ = parts;
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let normalized = Self::normalize_parts(&Self::components(&self.cwd)?, path);
        let p = self.resolve(normalized, false)?;
        match self.entries.get(&p) {
            Some(Entry::File(b)) => Ok(b.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(p)),
            Some(Entry::Link(_)) => Err(Error::Loop(p)),
            None => Err(Error::Missing(p)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let normalized = Self::normalize_parts(&Self::components(&self.cwd)?, path);
        let p = self.resolve(normalized, false)?;
        match self.entries.get(&p) {
            Some(Entry::Dir) => Ok(self.direct_children(&p)),
            Some(_) => Err(Error::NotDir(p)),
            None => Err(Error::Missing(p)),
        }
    }

    pub fn cwd(&self) -> String { self.cwd.clone() }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = Self::normalize_parts(&Self::components(&self.cwd)?, path);
        let p = self.resolve(normalized, false)?;
        match self.entries.get(&p) {
            Some(Entry::Dir) => { self.cwd = p; Ok(()) }
            _ => Err(Error::NotDir(p)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let from_parts = Self::components(from)?;
        let to_parts = Self::components(to)?;
        let src = Self::canon(&Self::normalize_parts(&Self::components(&self.cwd)?, from));
        let dst = Self::canon(&Self::normalize_parts(&Self::components(&self.cwd)?, to));
        if src == "/" || dst == "/" { return Err(Error::Root); }
        let (sp, sn) = Self::parent(&src);
        let sp = self.resolve(Self::components(&sp)?, false)?;
        match self.entries.get(&sp) {
            Some(Entry::Dir) => {}
            None => return Err(Error::Missing(sp)),
            Some(_) => return Err(Error::NotDir(sp)),
        }
        let source = if sp == "/" { format!("/{}", sn) } else { format!("{}/{}", sp, sn) };
        let ent = self.entries.get(&source).cloned().ok_or_else(|| Error::Missing(source.clone()))?;
        if matches!(ent, Entry::Dir) && Self::is_within(&self.cwd, &source) { return Err(Error::Busy(source)); }
        let (dp, dn) = Self::parent(&dst);
        let dp = self.resolve(Self::components(&dp)?, false)?;
        match self.entries.get(&dp) {
            Some(Entry::Dir) => {}
            None => return Err(Error::Missing(dp)),
            Some(_) => return Err(Error::NotDir(dp)),
        }
        let dest = if dp == "/" { format!("/{}", dn) } else { format!("{}/{}", dp, dn) };
        if matches!(self.entries.get(&dest), Some(Entry::Dir)) && Self::is_within(&self.cwd, &dest) { return Err(Error::Busy(dest)); }
        if source == dest { return Ok(()); }
        if matches!(ent, Entry::Dir) && Self::is_within(&dest, &source) { return Err(Error::Cycle(dest)); }
        if self.entries.contains_key(&dest) && !overwrite { return Err(Error::Exists(dest)); }
        if let Some(dent) = self.entries.get(&dest) {
            match (&ent, dent) {
                (Entry::Dir, Entry::Dir) => {
                    if !self.direct_children(&dest).is_empty() { return Err(Error::NotEmpty(dest)); }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(dest)),
                (_, Entry::Dir) => return Err(Error::IsDir(dest)),
                _ => {}
            }
        }
        let moved: Vec<(String, Entry)> = self.subtree_paths(&source).into_iter().filter_map(|p| {
            self.entries.get(&p).cloned().map(|e| (p, e))
        }).collect();
        let old_dest = if self.entries.contains_key(&dest) { self.subtree_paths(&dest) } else { vec![] };
        for p in old_dest { self.entries.remove(&p); }
        for (p, _) in &moved { self.entries.remove(p); }
        for (p, e) in moved {
            let suffix = if p == source { String::new() } else { p[source.len()..].to_string() };
            self.entries.insert(format!("{}{}", dest, suffix), e);
        }
        let _ = (from_parts, to_parts);
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let _parts = Self::components(path)?;
        let canonical = Self::canon(&Self::normalize_parts(&Self::components(&self.cwd)?, path));
        if canonical == "/" { return Err(Error::Root); }
        let (parent, name) = Self::parent(&canonical);
        let parent = self.resolve(Self::components(&parent)?, false)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            None => return Err(Error::Missing(canonical)),
            Some(_) => return Err(Error::NotDir(parent)),
        }
        let target = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        let entry = self.entries.get(&target).ok_or_else(|| Error::Missing(target.clone()))?;
        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &target) { return Err(Error::Busy(target)); }
        if matches!(entry, Entry::Dir) && !recursive && !self.direct_children(&target).is_empty() { return Err(Error::NotEmpty(target)); }
        for p in self.subtree_paths(&target) { self.entries.remove(&p); }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {

        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) { return Err(Error::InvalidTarget(i)); }
        let canonical = Self::canon(&Self::normalize_parts(&Self::components(&self.cwd)?, link));
        if canonical == "/" { return Err(Error::Exists("/".to_string())); }
        let (parent, name) = Self::parent(&canonical);
        let parent = self.resolve(Self::components(&parent)?, false)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => {}
            None => return Err(Error::Missing(parent)),
            Some(_) => return Err(Error::NotDir(parent)),
        }
        let dest = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        if self.entries.contains_key(&dest) { return Err(Error::Exists(dest)); }
        self.entries.insert(dest, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = Self::normalize_parts(&Self::components(&self.cwd)?, path);
        if normalized.is_empty() { return Err(Error::InvalidTarget(0)); }
        let mut parent_parts = normalized.clone();
        parent_parts.pop();
        let parent = self.resolve(parent_parts, false)?;
        let dest = if parent == "/" { format!("/{}", normalized.last().unwrap()) } else { format!("{}/{}", parent, normalized.last().unwrap()) };
        match self.entries.get(&dest) {
            Some(Entry::Link(t)) => Ok(t.clone()),
            None => Err(Error::Missing(dest)),
            _ => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let normalized = Self::normalize_parts(&Self::components(&self.cwd)?, path);
        let root = self.resolve(normalized, false)?;
        if !self.entries.contains_key(&root) { return Err(Error::Missing(root)); }
        let mut rows: Vec<(String, u64)> = self.subtree_paths(&root).into_iter().map(|p| {
            let size = match self.entries.get(&p) {
                Some(Entry::File(b)) => b.len() as u64,
                Some(Entry::Link(t)) => t.len() as u64,
                Some(Entry::Dir) => self.dir_size(&p),
                None => 0,
            };
            (p, size)
        }).collect();
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind { Dir, File, Link }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn components(path: &str) -> Result<Vec<String>, Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    let mut out = Vec::new();
    for part in path.split('/') {
        match part {
            "" | "." => {}
            ".." => { out.pop(); }
            _ => out.push(part.to_owned()),
        }
    }
    Ok(out)
}

fn path_of(parts: &[String]) -> String {
    if parts.is_empty() { "/".to_owned() } else { format!("/{}", parts.join("/")) }
}

fn split_path(path: &str) -> Vec<String> {
    if path == "/" { Vec::new() }
    else { path[1..].split('/').map(str::to_owned).collect() }
}

fn is_within(path: &str, dir: &str) -> bool {
    if dir == "/" { return true; }
    path == dir || path.strip_prefix(dir).is_some_and(|s| s.starts_with('/'))
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self { entries, cwd: "/".to_owned() }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        let mut parts = if path.starts_with('/') { Vec::new() } else { split_path(&self.cwd) };
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() { return Err(Error::InvalidPath(0)); }
        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => { parts.pop(); }
                _ => parts.push(part.to_owned()),
            }
        }
        Ok(path_of(&parts))
    }

    fn resolve(&self, initial: &str, allow_missing_final: bool) -> Result<String, Error> {
        let mut todo = split_path(initial);
        let mut built: Vec<String> = Vec::new();
        let mut visited = HashSet::new();
        let mut expansions = 0usize;
        let mut index = 0usize;

        while index < todo.len() {
            let part = todo[index].clone();
            built.push(part);
            let current = path_of(&built);
            let final_component = index + 1 == todo.len();

            let entry = match self.entries.get(&current) {
                Some(e) => e,
                None if final_component && allow_missing_final => return Ok(current),
                None => return Err(Error::Missing(current)),
            };

            if let Entry::Link(target) = entry {
                if !visited.insert(current.clone()) {
                    return Err(Error::Loop(current));
                }
                if expansions >= 8 {
                    return Err(Error::Depth(current));
                }
                expansions += 1;

                let mut target_parts = if target.starts_with('/') {
                    Vec::new()
                } else {
                    built[..built.len() - 1].to_vec()
                };
                for component in target.split('/') {
                    match component {
                        "" | "." => {}
                        ".." => { target_parts.pop(); }
                        _ => target_parts.push(component.to_owned()),
                    }
                }
                target_parts.extend(todo[index + 1..].iter().cloned());
                let expanded = path_of(&target_parts);
                todo = split_path(&expanded);
                built.clear();
                index = 0;
                continue;
            }

            if !final_component && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(current));
            }
            index += 1;
        }
        Ok(path_of(&built))
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let mut parts = split_path(path);
        let name = parts.pop().unwrap_or_default();
        let parent = path_of(&parts);
        let resolved_parent = self.resolve(&parent, false)?;
        if !matches!(self.entries.get(&resolved_parent), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved_parent));
        }
        Ok((resolved_parent, name))
    }

    fn child_paths(&self, path: &str) -> Vec<String> {
        self.entries.keys()
            .filter(|p| *p != path && is_within(p, path))
            .filter(|p| {
                let remainder = if path == "/" { &p[1..] } else { &p[path.len() + 1..] };
                !remainder.is_empty() && !remainder.contains('/')
            })
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let (parent, name) = self.resolve_parent(&canonical)?;
        let resolved = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries.insert(resolved, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, true)?;
        if matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::IsDir(resolved));
        }
        if !self.entries.contains_key(&resolved) {
            let (parent, _) = self.resolve_parent(&resolved)?;
            let _ = parent;
        }
        self.entries.insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        let mut result = Vec::new();
        for child in self.child_paths(&resolved) {
            let basename = child.rsplit('/').next().unwrap_or("").to_owned();
            if let Some(entry) = self.entries.get(&child) {
                result.push((basename, entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String { self.cwd.clone() }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, false)?;
        if !matches!(self.entries.get(&resolved), Some(Entry::Dir)) {
            return Err(Error::NotDir(resolved));
        }
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let src = self.normalize(from)?;
        let dst = self.normalize(to)?;
        if src == "/" || dst == "/" { return Err(Error::Root); }

        let (src_parent, src_name) = self.resolve_parent(&src)?;
        let source = if src_parent == "/" { format!("/{}", src_name) } else { format!("{}/{}", src_parent, src_name) };
        let source_entry = self.entries.get(&source).cloned().ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && is_within(&self.cwd, &source) {
            return Err(Error::Busy(source));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&dst)?;
        let destination = if dst_parent == "/" { format!("/{}", dst_name) } else { format!("{}/{}", dst_parent, dst_name) };

        if matches!(self.entries.get(&destination), Some(Entry::Dir)) && is_within(&self.cwd, &destination) {
            return Err(Error::Busy(destination));
        }
        if source == destination { return Ok(()); }
        if matches!(source_entry, Entry::Dir) && is_within(&destination, &source) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }
        if let Some(ref dest_entry) = destination_entry {
            match (&source_entry, dest_entry) {
                (Entry::Dir, Entry::Dir) => {
                    if self.child_paths(&destination).len() > 0 {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        let moving: Vec<(String, Entry)> = self.entries.iter()
            .filter(|(p, _)| is_within(p, &source))
            .map(|(p, e)| (p.clone(), e.clone()))
            .collect();
        for (p, _) in &moving { self.entries.remove(p); }
        if destination_entry.is_some() { self.entries.remove(&destination); }
        for (old, entry) in moving {
            let suffix = if old == source { String::new() } else { old[source.len()..].to_owned() };
            self.entries.insert(format!("{}{}", destination, suffix), entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" { return Err(Error::Root); }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let resolved = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        let entry = self.entries.get(&resolved).cloned().ok_or_else(|| Error::Missing(resolved.clone()))?;

        if matches!(entry, Entry::Dir) && is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }
        if matches!(entry, Entry::Dir) && !self.child_paths(&resolved).is_empty() && !recursive {
            return Err(Error::NotEmpty(resolved));
        }
        if matches!(entry, Entry::Dir) && recursive {
            let remove: Vec<String> = self.entries.keys()
                .filter(|p| is_within(p, &resolved))
                .cloned().collect();
            for p in remove { self.entries.remove(&p); }
        } else {
            self.entries.remove(&resolved);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical_link = self.normalize(link)?;
        if target.is_empty() { return Err(Error::InvalidTarget(0)); }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        if canonical_link == "/" { return Err(Error::Exists("/".to_owned())); }
        let (parent, name) = self.resolve_parent(&canonical_link)?;
        let resolved = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        if self.entries.contains_key(&resolved) { return Err(Error::Exists(resolved)); }
        self.entries.insert(resolved, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" { return Err(Error::InvalidTarget(0)); }
        let (parent, name) = self.resolve_parent(&canonical)?;
        let resolved = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.normalize(path)?;
        let resolved = self.resolve(&canonical, false)?;
        let root_entry = self.entries.get(&resolved).ok_or_else(|| Error::Missing(resolved.clone()))?;
        let paths: Vec<String> = if matches!(root_entry, Entry::Dir) {
            let mut v = vec![resolved.clone()];
            v.extend(self.entries.keys().filter(|p| *p != &resolved && is_within(p, &resolved)).cloned());
            v
        } else {
            vec![resolved.clone()]
        };
        let mut rows = Vec::new();
        for p in paths {
            let size = match self.entries.get(&p) {
                Some(Entry::File(bytes)) => bytes.len() as u64,
                Some(Entry::Link(target)) => target.len() as u64,
                Some(Entry::Dir) => self.entries.iter()
                    .filter(|(q, _)| *q != &p && is_within(q, &p))
                    .map(|(_, e)| match e {
                        Entry::File(bytes) => bytes.len() as u64,
                        Entry::Link(target) => target.len() as u64,
                        Entry::Dir => 0,
                    })
                    .sum(),
                None => 0,
            };
            rows.push((p, size));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}

fn absolute(parts: &[String]) -> String {

    else { format!("/{}", parts.join("/")) }
}

fn inside(path: &str, root: &str) -> bool {
    root == "/" || path == root || path.strip_prefix(root).is_some_and(|rest| rest.starts_with('/'))
}

pub struct Fs { map: BTreeMap<String, Entry>, cwd: String }

impl Fs {
    pub fn new() -> Self {
        Self { map: BTreeMap::from([("/".into(), Entry::Dir)]), cwd: "/".into() }
    }

    fn normalize(&self, input: &str) -> Result<String, Error> {
        if let Some(i) = input.as_bytes().iter().position(|b| *b == 0) { return Err(Error::InvalidPath(i)); }
        if input.is_empty() { return Err(Error::InvalidPath(0)); }
        let mut p = if input.starts_with('/') { Vec::new() } else { parts_of(&self.cwd) };
        for c in input.split('/') {
            if c.is_empty() || c == "." { continue; }
            if c == ".." { p.pop(); } else { p.push(c.into()); }
        }
        Ok(absolute(&p))
    }

    fn resolve(&self, path: &str, missing_final_ok: bool) -> Result<String, Error> {
        let mut queue = parts_of(path);
        let mut prefix = Vec::<String>::new();
        let mut seen = HashSet::new();
        let mut count = 0;
        let mut i = 0;
        while i < queue.len() {
            prefix.push(queue[i].clone());
            let current = absolute(&prefix);
            let final_one = i + 1 == queue.len();
            let Some(entry) = self.map.get(&current) else {
                if final_one && missing_final_ok { return Ok(current); }
                return Err(Error::Missing(current));
            };
            if let Entry::Link(target) = entry {
                if !seen.insert(current.clone()) { return Err(Error::Loop(current)); }
                if count == 8 { return Err(Error::Depth(current)); }
                count += 1;
                let mut exp = if target.starts_with('/') { Vec::new() } else { prefix[..prefix.len()-1].to_vec() };
                for c in target.split('/') {
                    if c.is_empty() || c == "." { continue; }
                    if c == ".." { exp.pop(); } else { exp.push(c.into()); }
                }
                exp.extend(queue[i+1..].iter().cloned());
                queue = parts_of(&absolute(&exp));
                prefix.clear();
                i = 0;
                continue;
            }
            if !final_one && !matches!(entry, Entry::Dir) { return Err(Error::NotDir(current)); }
            i += 1;
        }
        Ok(absolute(&prefix))
    }

    fn parent(&self, path: &str) -> Result<(String, String), Error> {
        let mut p = parts_of(path);
        let name = p.pop().unwrap_or_default();
        let dir = self.resolve(&absolute(&p), false)?;
        if !matches!(self.map.get(&dir), Some(Entry::Dir)) { return Err(Error::NotDir(dir)); }
        Ok((dir, name))
    }

    fn joined(parent: &str, name: &str) -> String {
        if parent == "/" { format!("/{name}") } else { format!("{parent}/{name}") }
    }

    fn descendants(&self, path: &str) -> Vec<String> {
        self.map.keys().filter(|p| *p != path && inside(p, path)).cloned().collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let p = self.normalize(path)?;
        let (d, n) = self.parent(&p)?;
        let full = Self::joined(&d, &n);
        if self.map.contains_key(&full) { return Err(Error::Exists(full)); }
        self.map.insert(full, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let p = self.normalize(path)?;
        let full = self.resolve(&p, true)?;
        if matches!(self.map.get(&full), Some(Entry::Dir)) { return Err(Error::IsDir(full)); }
        if !self.map.contains_key(&full) { self.parent(&full)?; }
        self.map.insert(full, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let p = self.normalize(path)?;
        let full = self.resolve(&p, false)?;
        match self.map.get(&full) {
            Some(Entry::File(v)) => Ok(v.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(full)),
            _ => Err(Error::Missing(full)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let p = self.normalize(path)?;
        let full = self.resolve(&p, false)?;
        if !matches!(self.map.get(&full), Some(Entry::Dir)) { return Err(Error::NotDir(full)); }
        let mut out = Vec::new();
        for child in self.map.keys().filter(|x| **x != full && inside(x, &full)) {
            let rest = if full == "/" { &child[1..] } else { &child[full.len()+1..] };
            if !rest.contains('/') {
                let kind = match self.map.get(child).unwrap() {
                    Entry::Dir => Kind::Dir, Entry::File(_) => Kind::File, Entry::Link(_) => Kind::Link,
                };
                out.push((rest.to_owned(), kind));
            }
        }
        out.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(out)
    }

    pub fn cwd(&self) -> String { self.cwd.clone() }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let p = self.normalize(path)?;
        let full = self.resolve(&p, false)?;
        if !matches!(self.map.get(&full), Some(Entry::Dir)) { return Err(Error::NotDir(full)); }
        self.cwd = full;
        Ok(())
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let a = self.normalize(from)?;
        let b = self.normalize(to)?;
        if a == "/" || b == "/" { return Err(Error::Root); }
        let (ap, an) = self.parent(&a)?;
        let src = Self::joined(&ap, &an);
        let src_entry = self.map.get(&src).cloned().ok_or_else(|| Error::Missing(src.clone()))?;
        if matches!(src_entry, Entry::Dir) && inside(&self.cwd, &src) { return Err(Error::Busy(src)); }
        let (bp, bn) = self.parent(&b)?;
        let dst = Self::joined(&bp, &bn);
        if matches!(self.map.get(&dst), Some(Entry::Dir)) && inside(&self.cwd, &dst) { return Err(Error::Busy(dst)); }
        if src == dst { return Ok(()); }
        if matches!(src_entry, Entry::Dir) && inside(&dst, &src) { return Err(Error::Cycle(dst)); }
        let dst_entry = self.map.get(&dst).cloned();
        if dst_entry.is_some() && !overwrite { return Err(Error::Exists(dst)); }
        if let Some(ref d) = dst_entry {
            match (&src_entry, d) {
                (Entry::Dir, Entry::Dir) if !self.descendants(&dst).is_empty() => return Err(Error::NotEmpty(dst)),
                (Entry::Dir, Entry::Dir) => {}
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }
        let moved: Vec<_> = self.map.iter().filter(|(p, _)| inside(p, &src)).map(|(p,e)| (p.clone(),e.clone())).collect();
        for (p, _) in &moved { self.map.remove(p); }
        if dst_entry.is_some() { self.map.remove(&dst); }
        for (old, e) in moved {
            let suffix = &old[src.len()..];
            self.map.insert(format!("{dst}{suffix}"), e);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let p = self.normalize(path)?;
        if p == "/" { return Err(Error::Root); }
        let (parent, name) = self.parent(&p)?;
        let full = Self::joined(&parent, &name);
        let entry = self.map.get(&full).cloned().ok_or_else(|| Error::Missing(full.clone()))?;
        if matches!(entry, Entry::Dir) && inside(&self.cwd, &full) { return Err(Error::Busy(full)); }
        let children = self.descendants(&full);
        if matches!(entry, Entry::Dir) && !children.is_empty() && !recursive { return Err(Error::NotEmpty(full)); }
        if matches!(entry, Entry::Dir) && recursive {
            for child in children { self.map.remove(&child); }
        }
        self.map.remove(&full);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let p = self.normalize(link)?;
        if target.is_empty() { return Err(Error::InvalidTarget(0)); }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) { return Err(Error::InvalidTarget(i)); }
        if p == "/" { return Err(Error::Exists("/".into())); }
        let (parent, name) = self.parent(&p)?;
        let full = Self::joined(&parent, &name);
        if self.map.contains_key(&full) { return Err(Error::Exists(full)); }
        self.map.insert(full, Entry::Link(target.into()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let p = self.normalize(path)?;
        if p == "/" { return Err(Error::InvalidTarget(0)); }
        let (parent, name) = self.parent(&p)?;
        let full = Self::joined(&parent, &name);
        match self.map.get(&full) {
            Some(Entry::Link(t)) => Ok(t.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(full)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let p = self.normalize(path)?;
        let root = self.resolve(&p, false)?;
        let entry = self.map.get(&root).ok_or_else(|| Error::Missing(root.clone()))?;
        let mut paths = vec![root.clone()];
        if matches!(entry, Entry::Dir) { paths.extend(self.descendants(&root)); }
        let mut out = Vec::new();
        for p in paths {
            let size = match self.map.get(&p) {
                Some(Entry::File(b)) => b.len() as u64,
                Some(Entry::Link(t)) => t.len() as u64,
                Some(Entry::Dir) => self.map.iter()
                    .filter(|(q, _)| *q != &p && inside(q, &p))
                    .map(|(_, e)| match e {
                        Entry::File(b) => b.len() as u64,
                        Entry::Link(t) => t.len() as u64,
                        Entry::Dir => 0,
                    }).sum(),
                None => 0,
            };
            out.push((p, size));
        }
        out.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(out)
    }
}

use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind { Dir, File, Link }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize), Missing(String), NotDir(String), IsDir(String), Exists(String),
    Root, Busy(String), Cycle(String), NotEmpty(String), InvalidTarget(usize),
    Loop(String), Depth(String),
}

#[derive(Clone)]
enum Entry { Dir, File(Vec<u8>), Link(String) }

impl Entry {
    fn kind(&self) -> Kind {
        match self { Entry::Dir => Kind::Dir, Entry::File(_) => Kind::File, Entry::Link(_) => Kind::Link }
    }
}

pub struct Fs { entries: BTreeMap<String, Entry>, cwd: String }

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".into(), Entry::Dir);
        Fs { entries, cwd: "/".into() }
    }

    fn validate(s: &str) -> Result<(), Error> {
        if let Some(i) = s.as_bytes().iter().position(|b| *b == 0) { return Err(Error::InvalidPath(i)); }
        if s.is_empty() { return Err(Error::InvalidPath(0)); }
        Ok(())
    }

    fn parts(s: &str) -> Vec<String> {
        s.split('/').filter(|x| !x.is_empty()).map(|x| x.to_string()).collect()
    }

    fn normalize(base: &[String], s: &str) -> Vec<String> {
        let mut out = if s.starts_with('/') { Vec::new() } else { base.to_vec() };
        for x in s.split('/') {
            match x { "" | "." => {}, ".." => { out.pop(); }, _ => out.push(x.to_string()) }
        }
        out
    }

    fn canon(p: &[String]) -> String {
        if p.is_empty() { "/".into() } else { format!("/{}", p.join("/")) }
    }

    fn split_parent(p: &str) -> (String, String) {
        let i = p.rfind('/').unwrap_or(0);
        let name = p[i + 1..].to_string();
        let parent = if i == 0 { "/".into() } else { p[..i].to_string() };
        (parent, name)
    }

    fn within(p: &str, d: &str) -> bool {
        d == "/" || p == d || p.strip_prefix(d).map_or(false, |r| r.starts_with('/'))
    }

    fn resolve(&self, mut parts: Vec<String>) -> Result<String, Error> {
        let mut seen = HashSet::new();
        let mut n = 0;
        loop {
            let mut built = Vec::<String>::new();
            let mut restart = None;
            for i in 0..parts.len() {
                built.push(parts[i].clone());
                let full = Self::canon(&built);
                let ent = self.entries.get(&full).ok_or_else(|| Error::Missing(full.clone()))?;
                if i + 1 < parts.len() && !matches!(ent, Entry::Dir | Entry::Link(_)) {
                    return Err(Error::NotDir(full));
                }
                if let Entry::Link(t) = ent {
                    if !seen.insert(full.clone()) { return Err(Error::Loop(full)); }
                    if n == 8 { return Err(Error::Depth(full)); }
                    n += 1;
                    let mut target = Self::normalize(&built[..built.len()-1], t);
                    target.extend_from_slice(&parts[i+1..]);
                    restart = Some(target);
                    break;
                }
            }
            match restart { Some(p) => parts = p, None => return Ok(Self::canon(&parts)) }
        }
    }

    fn resolve_parent(&self, parts: &[String]) -> Result<String, Error> {
        let p = Self::canon(parts);
        let (parent, _) = Self::split_parent(&p);
        let pp = Self::parts(&parent);
        let resolved = self.resolve(pp)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => Ok(resolved),
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    fn children(&self, p: &str) -> Vec<(String, Kind)> {
        let prefix = if p == "/" { "/".to_string() } else { format!("{}/", p) };
        self.entries.iter().filter_map(|(q, e)| {
            let r = q.strip_prefix(&prefix)?;
            if r.is_empty() || r.contains('/') { None } else { Some((r.to_string(), e.kind())) }
        }).collect()
    }

    fn descendants(&self, p: &str) -> Vec<String> {
        self.entries.keys().filter(|q| Self::within(q, p)).cloned().collect()
    }

    fn dir_size(&self, p: &str) -> u64 {
        self.descendants(p).iter().filter_map(|q| match self.entries.get(q) {
            Some(Entry::File(b)) => Some(b.len() as u64),
            Some(Entry::Link(t)) => Some(t.len() as u64),
            _ => None,
        }).fold(0, u64::saturating_add)
    }

    fn resolve_input(&self, s: &str) -> Result<Vec<String>, Error> {
        Self::validate(s)?;
        Ok(Self::normalize(&Self::parts(&self.cwd), s))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = self.resolve_input(path)?;
        let full = Self::canon(&parts);
        if full == "/" { return Err(Error::Exists(full)); }
        let (raw_parent, name) = Self::split_parent(&full);
        let parent = self.resolve(Self::parts(&raw_parent))?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => (),
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        let dest = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        if self.entries.contains_key(&dest) { return Err(Error::Exists(dest)); }
        self.entries.insert(dest, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = self.resolve_input(path)?;
        let p = match self.resolve(parts) {
            Ok(p) => p,
            Err(Error::Missing(p)) => p,
            Err(e) => return Err(e),
        };
        if p == "/" { return Err(Error::IsDir(p)); }
        let (parent_raw, _) = Self::split_parent(&p);
        let parent = self.resolve(Self::parts(&parent_raw))?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => (),
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        match self.entries.get(&p) {
            Some(Entry::Dir) => Err(Error::IsDir(p)),
            Some(Entry::File(_)) => { self.entries.insert(p, Entry::File(data.to_vec())); Ok(()) }
            Some(Entry::Link(_)) => Err(Error::Loop(p)),
            None => { self.entries.insert(p, Entry::File(data.to_vec())); Ok(()) }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = self.resolve_input(path)?;
        let p = self.resolve(parts)?;
        match self.entries.get(&p) {
            Some(Entry::File(b)) => Ok(b.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(p)),
            Some(Entry::Link(_)) => Err(Error::Loop(p)),
            None => Err(Error::Missing(p)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = self.resolve_input(path)?;
        let p = self.resolve(parts)?;
        match self.entries.get(&p) {
            Some(Entry::Dir) => Ok(self.children(&p)),
            Some(_) => Err(Error::NotDir(p)),
            None => Err(Error::Missing(p)),
        }
    }

    pub fn cwd(&self) -> String { self.cwd.clone() }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = self.resolve_input(path)?;
        let p = self.resolve(parts)?;
        match self.entries.get(&p) {
            Some(Entry::Dir) => { self.cwd = p; Ok(()) }
            _ => Err(Error::NotDir(p)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        Self::validate(from)?;
        Self::validate(to)?;
        let base = Self::parts(&self.cwd);
        let src_raw = Self::canon(&Self::normalize(&base, from));
        let dst_raw = Self::canon(&Self::normalize(&base, to));
        if src_raw == "/" || dst_raw == "/" { return Err(Error::Root); }

        let (spp, sn) = Self::split_parent(&src_raw);
        let sp = self.resolve(Self::parts(&spp))?;
        match self.entries.get(&sp) {
            Some(Entry::Dir) => (),
            Some(_) => return Err(Error::NotDir(sp)),
            None => return Err(Error::Missing(sp)),
        }
        let src = if sp == "/" { format!("/{}", sn) } else { format!("{}/{}", sp, sn) };
        let ent = self.entries.get(&src).cloned().ok_or_else(|| Error::Missing(src.clone()))?;
        if matches!(ent, Entry::Dir) && Self::within(&self.cwd, &src) { return Err(Error::Busy(src)); }

        let (dpp, dn) = Self::split_parent(&dst_raw);
        let dp = self.resolve(Self::parts(&dpp))?;
        match self.entries.get(&dp) {
            Some(Entry::Dir) => (),
            Some(_) => return Err(Error::NotDir(dp)),
            None => return Err(Error::Missing(dp)),
        }
        let dst = if dp == "/" { format!("/{}", dn) } else { format!("{}/{}", dp, dn) };
        if matches!(self.entries.get(&dst), Some(Entry::Dir)) && Self::within(&self.cwd, &dst) { return Err(Error::Busy(dst)); }
        if src == dst { return Ok(()); }
        if matches!(ent, Entry::Dir) && Self::within(&dst, &src) { return Err(Error::Cycle(dst)); }
        if self.entries.contains_key(&dst) && !overwrite { return Err(Error::Exists(dst)); }
        if let Some(dent) = self.entries.get(&dst) {
            match (&ent, dent) {
                (Entry::Dir, Entry::Dir) if !self.children(&dst).is_empty() => return Err(Error::NotEmpty(dst)),
                (Entry::Dir, Entry::Dir) => (),
                (Entry::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Entry::Dir) => return Err(Error::IsDir(dst)),
                _ => (),
            }
        }

        let moved: Vec<(String, Entry)> = self.descendants(&src).into_iter().filter_map(|p| self.entries.get(&p).cloned().map(|e| (p, e))).collect();
        let old_dest = if self.entries.contains_key(&dst) { self.descendants(&dst) } else { Vec::new() };
        for p in old_dest { self.entries.remove(&p); }
        for (p, _) in &moved { self.entries.remove(p); }
        for (p, e) in moved {
            let suffix = &p[src.len()..];
            self.entries.insert(format!("{}{}", dst, suffix), e);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let parts = self.resolve_input(path)?;
        let canonical = Self::canon(&parts);
        if canonical == "/" { return Err(Error::Root); }
        let (pp, name) = Self::split_parent(&canonical);
        let parent = self.resolve(Self::parts(&pp))?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => (),
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(canonical)),
        }
        let target = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        let ent = self.entries.get(&target).ok_or_else(|| Error::Missing(target.clone()))?;
        if matches!(ent, Entry::Dir) && Self::within(&self.cwd, &target) { return Err(Error::Busy(target)); }
        if matches!(ent, Entry::Dir) && !recursive && !self.children(&target).is_empty() { return Err(Error::NotEmpty(target)); }
        for p in self.descendants(&target) { self.entries.remove(&p); }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        Self::validate(link)?;
        let canonical = Self::canon(&Self::normalize(&Self::parts(&self.cwd), link));
        if target.is_empty() { return Err(Error::InvalidTarget(0)); }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) { return Err(Error::InvalidTarget(i)); }
        if canonical == "/" { return Err(Error::Exists(canonical)); }
        let (pp, name) = Self::split_parent(&canonical);
        let parent = self.resolve(Self::parts(&pp))?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => (),
            Some(_) => return Err(Error::NotDir(parent)),
            None => return Err(Error::Missing(parent)),
        }
        let dest = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        if self.entries.contains_key(&dest) { return Err(Error::Exists(dest)); }
        self.entries.insert(dest, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let parts = self.resolve_input(path)?;
        if parts.is_empty() { return Err(Error::InvalidTarget(0)); }
        let mut parent_parts = parts.clone();
        parent_parts.pop();
        let parent = self.resolve(parent_parts)?;
        let name = parts.last().unwrap();
        let p = if parent == "/" { format!("/{}", name) } else { format!("{}/{}", parent, name) };
        match self.entries.get(&p) {
            Some(Entry::Link(t)) => Ok(t.clone()),
            None => Err(Error::Missing(p)),
            _ => Err(Error::InvalidTarget(0)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let parts = self.resolve_input(path)?;
        let root = self.resolve(parts)?;
        if !self.entries.contains_key(&root) { return Err(Error::Missing(root)); }
        let mut rows: Vec<(String, u64)> = self.descendants(&root).into_iter().map(|p| {
            let n = match self.entries.get(&p) {
                Some(Entry::File(b)) => b.len() as u64,
                Some(Entry::Link(t)) => t.len() as u64,
                Some(Entry::Dir) => self.dir_size(&p),
                None => 0,
            };
            (p, n)
        }).collect();
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone)]
enum Node {
    Directory,
    Bytes(Vec<u8>),
    Symbol(String),
}

impl Node {
    fn kind(&self) -> Kind {
        match self {
            Node::Directory => Kind::Dir,
            Node::Bytes(_) => Kind::File,
            Node::Symbol(_) => Kind::Link,
        }
    }
}

pub struct Fs {
    nodes: BTreeMap<String, Node>,
    cwd: Vec<String>,
}

fn canon(v: &[String]) -> String {
    if v.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", v.join("/"))
    }
}

fn comps(path: &str) -> Vec<String> {
    if path == "/" {
        Vec::new()
    } else {
        path[1..].split('/').map(|s| s.to_string()).collect()
    }
}

fn lex(mut v: Vec<String>, path: &str) -> Vec<String> {
    for x in path.split('/') {
        if x.is_empty() || x == "." {
            continue;
        }
        if x == ".." {
            v.pop();
        } else {
            v.push(x.to_string());
        }
    }
    v
}

fn checked(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn within(path: &[String], base: &[String]) -> bool {
    path.len() >= base.len() && path[..base.len()] == *base
}

impl Fs {
    pub fn new() -> Self {
        let mut nodes = BTreeMap::new();
        nodes.insert("/".to_string(), Node::Directory);
        Self {
            nodes,
            cwd: Vec::new(),
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        checked(path)?;
        let start = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };
        Ok(lex(start, path))
    }

    fn resolve(&self, initial: Vec<String>, create_final: bool) -> Result<Vec<String>, Error> {
        let mut pending = initial;
        let mut physical = Vec::<String>::new();
        let mut visited = HashSet::<String>::new();
        let mut count = 0usize;
        loop {
            if pending.is_empty() {
                return Ok(physical);
            }
            let part = pending.remove(0);
            let mut candidate = physical.clone();
            candidate.push(part);
            let candidate_path = canon(&candidate);
            match self.nodes.get(&candidate_path) {
                None => {
                    if create_final && pending.is_empty() {
                        let parent = canon(&physical);
                        match self.nodes.get(&parent) {
                            Some(Node::Directory) => return Ok(candidate),
                            Some(_) => return Err(Error::NotDir(parent)),
                            None => return Err(Error::Missing(parent)),
                        }
                    }
                    return Err(Error::Missing(candidate_path));
                }
                Some(Node::Directory) => physical.push(candidate.last().unwrap().clone()),
                Some(Node::Bytes(_)) => {
                    if !pending.is_empty() {
                        return Err(Error::NotDir(candidate_path));
                    }
                    physical.push(candidate.last().unwrap().clone());
                }
                Some(Node::Symbol(target)) => {
                    if !visited.insert(candidate_path.clone()) {
                        return Err(Error::Loop(candidate_path));
                    }
                    if count == 8 {
                        return Err(Error::Depth(candidate_path));
                    }
                    count += 1;
                    let base = if target.starts_with('/') {
                        Vec::new()
                    } else {
                        physical.clone()
                    };
                    let target_parts = lex(base, target);
                    pending = lex(target_parts, &pending.join("/"));
                    physical.clear();
                }
            }
        }
    }

    fn parent(&self, normalized: &[String]) -> Result<(Vec<String>, String), Error> {
        let mut p = normalized.to_vec();
        let name = p.pop().unwrap_or_default();
        let physical = self.resolve(p, false)?;
        let path = canon(&physical);
        match self.nodes.get(&path) {
            Some(Node::Directory) => Ok((physical, name)),
            Some(_) => Err(Error::NotDir(path)),
            None => Err(Error::Missing(path)),
        }
    }

    fn full(parent: &[String], name: &str) -> Vec<String> {
        let mut result = parent.to_vec();
        result.push(name.to_string());
        result
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let (parent, name) = self.parent(&normalized)?;
        let final_path = canon(&Self::full(&parent, &name));
        if self.nodes.contains_key(&final_path) {
            return Err(Error::Exists(final_path));
        }
        self.nodes.insert(final_path, Node::Directory);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        let resolved = self.resolve(normalized, true)?;
        let p = canon(&resolved);
        match self.nodes.get_mut(&p) {
            Some(Node::Directory) => Err(Error::IsDir(p)),
            Some(Node::Bytes(bytes)) => {
                bytes.clear();
                bytes.extend_from_slice(data);
                Ok(())
            }
            Some(Node::Symbol(_)) => unreachable!(),
            None => {
                self.nodes.insert(p, Node::Bytes(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let p = canon(&self.resolve(self.normalize(path)?, false)?);
        match self.nodes.get(&p) {
            Some(Node::Bytes(b)) => Ok(b.clone()),
            Some(Node::Directory) => Err(Error::IsDir(p)),
            _ => Err(Error::Missing(p)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let p = canon(&self.resolve(self.normalize(path)?, false)?);
        match self.nodes.get(&p) {
            Some(Node::Directory) => {
                let prefix = if p == "/" { "/" } else { "" };
                let _ = prefix;
                let mut out = Vec::new();
                for (full, node) in &self.nodes {
                    if full == &p {
                        continue;
                    }
                    let par = full.rsplit_once('/').map(|x| x.0).unwrap_or("");
                    let par = if par.is_empty() { "/" } else { par };
                    if par == p {
                        out.push((full.rsplit('/').next().unwrap_or("").to_string(), node.kind()));
                    }
                }
                out.sort_by(|a, b| a.0.cmp(&b.0));
                Ok(out)
            }
            _ => Err(Error::NotDir(p)),
        }
    }

    pub fn cwd(&self) -> String {
        canon(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let resolved = self.resolve(self.normalize(path)?, false)?;
        let p = canon(&resolved);
        match self.nodes.get(&p) {
            Some(Node::Directory) => {
                self.cwd = resolved;
                Ok(())
            }
            _ => Err(Error::NotDir(p)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let src_norm = self.normalize(from)?;
        let dst_norm = self.normalize(to)?;
        if src_norm.is_empty() || dst_norm.is_empty() {
            return Err(Error::Root);
        }
        let (sp, sn) = self.parent(&src_norm)?;
        let srcv = Self::full(&sp, &sn);
        let src = canon(&srcv);
        let srcnode = self.nodes.get(&src).cloned().ok_or_else(|| Error::Missing(src.clone()))?;
        if matches!(srcnode, Node::Directory) && within(&self.cwd, &srcv) {
            return Err(Error::Busy(src));
        }

        let (dp, dn) = self.parent(&dst_norm)?;
        let dstv = Self::full(&dp, &dn);
        let dst = canon(&dstv);
        if self.nodes.get(&dst).is_some_and(|n| matches!(n, Node::Directory) && within(&self.cwd, &dstv)) {
            return Err(Error::Busy(dst));
        }
        if srcv == dstv {
            return Ok(());
        }
        if matches!(srcnode, Node::Directory) && within(&dstv, &srcv) {
            return Err(Error::Cycle(dst));
        }
        let destnode = self.nodes.get(&dst).cloned();
        if destnode.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }
        if let Some(n) = &destnode {
            match (srcnode.kind(), n.kind()) {
                (Kind::Dir, Kind::Dir) => {
                    let prefix = format!("{}/", dst);
                    if self.nodes.keys().any(|p| p.starts_with(&prefix)) {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Kind::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Kind::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let prefix = format!("{}/", src);
        let moved: Vec<(String, Node)> = self.nodes.iter()
            .filter(|(p, _)| *p == &src || p.starts_with(&prefix))
            .map(|(p, n)| (p.clone(), n.clone())).collect();
        for (p, _) in &moved { self.nodes.remove(p); }
        if destnode.is_some() { self.nodes.remove(&dst); }
        for (old, node) in moved {
            self.nodes.insert(format!("{}{}", dst, &old[src.len()..]), node);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let normalized = self.normalize(path)?;
        if normalized.is_empty() {
            return Err(Error::Root);
        }
        let (parent, name) = self.parent(&normalized)?;
        let targetv = Self::full(&parent, &name);
        let target = canon(&targetv);
        let node = self.nodes.get(&target).cloned().ok_or_else(|| Error::Missing(target.clone()))?;
        if matches!(node, Node::Directory) && within(&self.cwd, &targetv) {
            return Err(Error::Busy(target));
        }
        let prefix = format!("{}/", target);
        let has_children = self.nodes.keys().any(|p| p.starts_with(&prefix));
        if matches!(node, Node::Directory) && has_children && !recursive {
            return Err(Error::NotEmpty(target));
        }
        let doomed: Vec<String> = self.nodes.keys()
            .filter(|p| *p == &target || (recursive && p.starts_with(&prefix)))
            .cloned().collect();
        for p in doomed { self.nodes.remove(&p); }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link_norm = self.normalize(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (parent, name) = self.parent(&link_norm)?;
        let link_path = canon(&Self::full(&parent, &name));
        if self.nodes.contains_key(&link_path) {
            return Err(Error::Exists(link_path));
        }
        self.nodes.insert(link_path, Node::Symbol(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let normalized = self.normalize(path)?;
        let (parent, name) = self.parent(&normalized)?;
        let p = canon(&Self::full(&parent, &name));
        match self.nodes.get(&p) {
            Some(Node::Symbol(t)) => Ok(t.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(p)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let resolved = canon(&self.resolve(self.normalize(path)?, false)?);
        if !self.nodes.contains_key(&resolved) {
            return Err(Error::Missing(resolved));
        }
        let prefix = if resolved == "/" { "/" } else { "" };
        let _ = prefix;
        let mut rows = Vec::new();
        for (full, node) in &self.nodes {
            if full != &resolved && !full.strip_prefix(&resolved).is_some_and(|r| r.starts_with('/')) {
                continue;
            }
            let size = match node {
                Node::Bytes(v) => v.len() as u64,
                Node::Symbol(t) => t.len() as u64,
                Node::Directory => {
                    let dir_prefix = if full == "/" { "/" } else { "" };
                    let dir_prefix = if full == "/" { "/".to_string() } else { format!("{}/", full) };
                    self.nodes.iter()
                        .filter(|(child, _)| child.starts_with(&dir_prefix))
                        .map(|(_, e)| match e {
                            Node::Bytes(v) => v.len() as u64,
                            Node::Symbol(t) => t.len() as u64,
                            Node::Directory => 0,
                        })
                        .fold(0u64, u64::saturating_add)
                }
            };
            rows.push((full.clone(), size));
        }
        rows.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(rows)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
