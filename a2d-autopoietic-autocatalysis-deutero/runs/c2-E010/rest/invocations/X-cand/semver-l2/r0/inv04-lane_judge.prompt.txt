        } else {
            let mut q = p;
            while q < bytes.len() && bytes[q] != b'.' {
                q += 1;
            }
            q
        };
        if end == p {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
        for i in p..end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }
        if pre && end - p > 1 && bytes[p] == b'0'
            && bytes[p..end].iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: p });
        }
        ids.push(s[p..end].to_string());
        if end == bytes.len() {
            break;
        }
        if bytes[end] == b'+' && pre {
            break;
        }
        p = end + 1;
        if p == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(ids)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let (major, p1) = parse_digits(s, 0)?;
    if p1 >= s.len() || s.as_bytes()[p1] != b'.' {
        if p1 >= s.len() || matches!(s.as_bytes()[p1], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p1 });
        }
        return Err(VersionError::InvalidChar { at: p1 });
    }
    let (minor, p2) = parse_digits(s, p1 + 1)?;
    if p2 >= s.len() || s.as_bytes()[p2] != b'.' {
        if p2 >= s.len() || matches!(s.as_bytes()[p2], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p2 });
        }
        return Err(VersionError::InvalidChar { at: p2 });
    }
    let (patch, mut p) = parse_digits(s, p2 + 1)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() {
        match s.as_bytes()[p] {
            b'-' => {
                p += 1;
                pre = parse_identifiers(s, p, true)?;
                if p < s.len() {
                    while p < s.len() && s.as_bytes()[p] != b'+' {
                        p += 1;
                    }
                    if p < s.len() {
                        p += 1;
                        build = parse_identifiers(s, p, false)?;
                    }
                }
            }
            b'+' => {
                p += 1;
                build = parse_identifiers(s, p, false)?;
            }
            _ => return Err(VersionError::InvalidChar { at: p }),
        }
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric(x);
        let yn = numeric(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Copy, Debug)]
enum Op { Eq, Lt, Le, Gt, Ge }

impl Op {
    fn test(self, c: Ordering) -> bool {
        match self {
            Op::Eq => c == Ordering::Equal,
            Op::Lt => c == Ordering::Less,
            Op::Le => c != Ordering::Greater,
            Op::Gt => c == Ordering::Greater,
            Op::Ge => c != Ordering::Less,
        }
    }
}

fn check_version(checks: &mut Vec<Check>, anchors: &mut Vec<(u64,u64,u64)>, op: Op, v: Version, explicit: bool) {
    if explicit && !v.pre.is_empty() {
        let t = (v.major, v.minor, v.patch);
        if !anchors.contains(&t) { anchors.push(t); }
    }
    checks.push(Check { op, version: v });
}

fn operand_version(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.is_empty() { return Err(syntax(at)); }
    let has_sign = s.bytes().any(|b| b == b'-' || b == b'+');
    let pieces: Vec<&str> = s.split('.').collect();
    let has_wild = pieces.iter().any(|p| matches!(*p, "*" | "x" | "X"));
    if has_sign || (pieces.len() >= 3 && !has_wild) {
        let v = operand_version(s, at)?;
        return Ok(Partial { fields: vec![Some(v.major),Some(v.minor),Some(v.patch)], full: Some(v), k: 3 });
    }
    if pieces.len() > 3 { return Err(syntax(at)); }
    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    let mut k = 0;
    for p in pieces {
        if p.is_empty() { return Err(syntax(at)); }
        if matches!(p, "*" | "x" | "X") {
            wildcard_seen = true;
            fields.push(None);
        } else {
            if wildcard_seen || !p.bytes().all(|b| b.is_ascii_digit()) || (p.len() > 1 && p.starts_with('0')) {
                return Err(syntax(at));
            }
            let n = p.parse::<u64>().map_err(|_| syntax(at))?;
            k += 1;
            fields.push(Some(n));
        }
    }
    while fields.len() < 3 { fields.push(None); }
    Ok(Partial { fields, full: None, k })
}

fn lower_version(p: &Partial) -> Version {
    if let Some(v) = &p.full { return v.clone(); }
    Version {
        major: p.fields[0].unwrap_or(0),
        minor: p.fields[1].unwrap_or(0),
        patch: p.fields[2].unwrap_or(0),
        pre: Vec::new(), build: Vec::new()
    }
}

fn upper_increment(p: &Partial, index: usize, at: usize) -> Result<Version, RangeError> {
    let mut vals = [p.fields[0].unwrap_or(0), p.fields[1].unwrap_or(0), p.fields[2].unwrap_or(0)];
    vals[index] = vals[index].checked_add(1).ok_or_else(|| syntax(at))?;
    for x in vals.iter_mut().skip(index + 1) { *x = 0; }
    Ok(Version { major: vals[0], minor: vals[1], patch: vals[2], pre: vec!["0".into()], build: Vec::new() })
}

fn add_partial_bare(p: &Partial, at: usize, checks: &mut Vec<Check>, anchors: &mut Vec<(u64,u64,u64)>) -> Result<(), RangeError> {
    if p.k == 0 { return Ok(()); }
    if p.full.is_some() {
        check_version(checks, anchors, Op::Eq, lower_version(p), true);
    } else {
        check_version(checks, anchors, Op::Ge, lower_version(p), false);
        let idx = if p.k == 1 {0} else {1};
        let upper = upper_increment(p, idx, at)?;
        check_version(checks, anchors, Op::Lt, upper, false);
    }
    Ok(())
}

fn expand_caret_tilde(
    p: &Partial, at: usize, caret: bool,
    checks: &mut Vec<Check>, anchors: &mut Vec<(u64,u64,u64)>
) -> Result<(), RangeError> {
    if p.k == 0 { return Err(syntax(at)); }
    let lower = lower_version(p);
    check_version(checks, anchors, Op::Ge, lower.clone(), p.full.as_ref().map_or(false, |v| !v.pre.is_empty()));
    let idx = if caret {
        if lower.major > 0 || p.k == 1 { 0 }
        else if lower.minor > 0 || p.k == 2 { 1 }
        else { 2 }
    } else if p.k == 1 { 0 } else { 1 };
    let upper = upper_increment(p, idx, at)?;
    check_version(checks, anchors, Op::Lt, upper, false);
    Ok(())
}

fn parse_arm(tokens: &[(String, usize)], arm_start: usize) -> Result<Arm, RangeError> {
    if tokens.is_empty() { return Err(RangeError::Empty { at: arm_start }); }
    let mut minus = None;
    for (i,(t,at)) in tokens.iter().enumerate() {
        if t == "-" { minus = Some((i,*at)); break; }
    }
    let mut checks = Vec::new();
    let mut anchors = Vec::new();
    if let Some((mi, mat)) = minus {
        if mi != 1 || tokens.len() != 3 { return Err(syntax(mat)); }
        let (lo, loa) = (&tokens[0].0, tokens[0].1);
        let (hi, hia) = (&tokens[2].0, tokens[2].1);
        let lp = parse_partial(lo, loa)?;
        let hp = parse_partial(hi, hia)?;
        if lp.k == 0 { return Err(syntax(loa)); }
        if hp.k == 0 { return Err(syntax(hia)); }
        let lv = lower_version(&lp);
        check_version(&mut checks, &mut anchors, Op::Ge, lv, lp.full.as_ref().map_or(false, |v| !v.pre.is_empty()));
        if hp.full.is_some() {
            check_version(&mut checks, &mut anchors, Op::Le, lower_version(&hp), hp.full.as_ref().map_or(false, |v| !v.pre.is_empty()));
        } else {
            let idx = if hp.k == 1 {0} else {1};
            let upper = upper_increment(&hp, idx, hia)?;
            check_version(&mut checks, &mut anchors, Op::Lt, upper, false);
        }
    } else {
        for (token, at) in tokens {
            let bytes = token.as_bytes();
            if bytes.is_empty() { return Err(syntax(*at)); }
            match bytes[0] {
                b'^' | b'~' => {
                    let p = parse_partial(&token[1..], at + 1)?;
                    expand_caret_tilde(&p, at + 1, bytes[0] == b'^', &mut checks, &mut anchors)?;
                }
                b'<' | b'>' | b'=' => {
                    let (op, n) = if token.starts_with("<=") { (Op::Le,2) }
                        else if token.starts_with(">=") { (Op::Ge,2) }
                        else if token.starts_with("<") { (Op::Lt,1) }
                        else if token.starts_with(">") { (Op::Gt,1) }
                        else { (Op::Eq,1) };
                    let operand = &token[n..];
                    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() { return Err(syntax(*at)); }
                    let v = operand_version(operand, at+n)?;
                    check_version(&mut checks, &mut anchors, op, v, true);
                }
                b'0'..=b'9' | b'*' | b'x' | b'X' => {
                    let p = parse_partial(token, *at)?;
                    add_partial_bare(&p, *at, &mut checks, &mut anchors)?;
                }
                _ => return Err(syntax(*at)),
            }
        }
    }
    Ok(Arm { checks, anchors })
}

fn is_ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut tokens: Vec<(String, usize)> = Vec::new();
    let mut arms = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) { i += 1; }
        if i >= bytes.len() { break; }
        let start = i;
        while i < bytes.len() && !is_ws(bytes[i]) { i += 1; }
        let tok = &s[start..i];
        if tok == "||" {
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            arms.push(parse_arm(&tokens, tokens[0].1)?);
            tokens.clear();
            let mut j = i;
            while j < bytes.len() && is_ws(bytes[j]) { j += 1; }
            if j == bytes.len() {
                return Err(RangeError::Empty { at: bytes.len() });
            }
            i = j;
        } else {
            tokens.push((tok.to_string(), start));
        }
    }
    if tokens.is_empty() {
        if arms.is_empty() { return Err(RangeError::Empty { at: 0 }); }
        return Err(RangeError::Empty { at: bytes.len() });
    }
    arms.push(parse_arm(&tokens, tokens[0].1)?);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let tuple = (v.major, v.minor, v.patch);
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty() && !arm.anchors.contains(&tuple) { return false; }
        arm.checks.iter().all(|c| c.op.test(compare_versions(v, &c.version)))
    })
}

pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(candidates: &[String], r: &Range, want_max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        match parse_version(s) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    let mut best: Option<usize> = None;
    for i in 0..candidates.len() {
        if !matches(&parsed[i], r) { continue; }
        if let Some(b) = best {
            let c = compare_versions(&parsed[i], &parsed[b]);
            if (want_max && c == Ordering::Greater) || (!want_max && c == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

fn parse_core_num(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return if is_sep_missing(bytes.get(start).copied()) {
            Err(VersionError::MissingCore { at: start })
        } else {
            Err(VersionError::InvalidChar { at: start })
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut n = 0u64;
    for b in digits.bytes() {
        n = n.checked_mul(10).and_then(|x| x.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(n)
}

fn parse_ids(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            let b = bytes[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..p];
        if prerelease
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {

        }
    }

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core_num(s, &mut p)?;
    if s.as_bytes().get(p) != Some(&b'.') {
        return if is_sep_missing(s.as_bytes().get(p).copied()) {
            Err(VersionError::MissingCore { at: p })
        } else {
            Err(VersionError::InvalidChar { at: p })
        };
    }
    p += 1;
    let minor = parse_core_num(s, &mut p)?;
    if s.as_bytes().get(p) != Some(&b'.') {
        return if is_sep_missing(s.as_bytes().get(p).copied()) {
            Err(VersionError::MissingCore { at: p })
        } else {
            Err(VersionError::InvalidChar { at: p })
        };
    }
    p += 1;
    let patch = parse_core_num(s, &mut p)?;
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < bytes.len() && bytes[p] == b'-' {
        p += 1;
        let start = p;
        let end = bytes[p..].iter().position(|&b| b == b'+').map(|x| p + x).unwrap_or(bytes.len());
        pre = parse_ids(s, start, end, true)?;
        p = end;
    }
    if p < bytes.len() && bytes[p] == b'+' {
        p += 1;
        build = parse_ids(s, p, bytes.len(), false)?;
        p = bytes.len();
    }
    if p != bytes.len() {
        return Err(VersionError::InvalidChar { at: p });
    }
    Ok(Version { major, minor, patch, pre, build })
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| {
        match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {

                    let yn = numeric_id(y);

                        (false, true) => Ordering::Greater,

        }

    Ok(Partial { fields, k, full: None })

fn bump(t: (u64,u64,u64), field: usize, at: usize) -> Result<(u64,u64,u64), RangeError> {

    *val = val.checked_add(1).ok_or(RangeError::Syntax { at })?;
    match field { 0 => {x.1=0;x.2=0}, 1 => x.2=0, _ => {} }

}

fn push_partial_arm(tokens: &[Token<'_>], out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    for t in tokens {
        let txt = t.text;
        if txt == "-" || txt.contains(',') || txt.contains('(') || txt.contains(')') || txt == "&" {
            return Err(RangeError::Syntax { at: t.start });
        }
        if let Some(rest) = txt.strip_prefix('^').or_else(|| txt.strip_prefix('~')) {
            let opbyte = txt.as_bytes()[0];
            let op_at = t.start;
            let operand_at = t.start + 1;
            let p = parse_partial(rest, operand_at)?;
            if p.k == 0 { return Err(RangeError::Syntax { at: op_at }); }
            let lower = p.full.clone().unwrap_or_else(|| stable(padded(&p)));
            let lo_anchor = p.full.as_ref().map_or(false, |v| !v.pre.is_empty());
            let field = if opbyte == b'^' {
                if p.fields[0] > 0 || p.k == 1 { 0 }
                else if p.fields.get(1).copied().unwrap_or(0) > 0 || p.k == 2 { 1 }
                else { 2 }
            } else if p.k == 1 { 0 } else { 1 };
            let upper = zero_pre(bump(padded(&p), field, operand_at)?);
            out.push(cmp(Op::Ge, lower, lo_anchor));
            out.push(cmp(Op::Lt, upper, false));
            continue;
        }
        let (op, operand, operand_at) = if txt.starts_with(">=") { (Some(Op::Ge), &txt[2..], t.start+2) }
            else if txt.starts_with("<=") { (Some(Op::Le), &txt[2..], t.start+2) }
            else if txt.starts_with('>') { (Some(Op::Gt), &txt[1..], t.start+1) }
            else if txt.starts_with('<') { (Some(Op::Lt), &txt[1..], t.start+1) }
            else if txt.starts_with('=') { (Some(Op::Eq), &txt[1..], t.start+1) }
            else { (None, txt, t.start) };
        if let Some(op) = op {
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at: t.start });
            }
            let v = full_or_error(operand, operand_at)?;
            out.push(cmp(op, v.clone(), !v.pre.is_empty()));
            continue;
        }
        let first = txt.as_bytes()[0];
        if !(first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X')) {
            return Err(RangeError::Syntax { at: t.start });
        }
        let p = parse_partial(txt, t.start)?;
        if let Some(v) = p.full {
            out.push(cmp(Op::Eq, v.clone(), !v.pre.is_empty()));
        } else if p.k == 1 {
            let lo = padded(&p);
            out.push(cmp(Op::Ge, stable(lo), false));
            out.push(cmp(Op::Lt, zero_pre(bump(lo, 0, t.start)?), false));
        } else if p.k == 2 {
            let lo = padded(&p);
            out.push(cmp(Op::Ge, stable(lo), false));
            out.push(cmp(Op::Lt, zero_pre(bump(lo, 1, t.start)?), false));
        }
    }
    Ok(())
}

fn parse_hyphen(tokens: &[Token<'_>], out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    let mins: Vec<usize> = tokens.iter().enumerate().filter(|(_,t)| t.text == "-").map(|(i,_)| i).collect();
    if !mins.is_empty() && (tokens.len() != 3 || mins != vec![1]) {
        return Err(RangeError::Syntax { at: tokens[mins[0]].start });
    }
    if mins.is_empty() { return push_partial_arm(tokens, out); }
    let lo_t = &tokens[0];
    let hi_t = &tokens[2];
    let lo = parse_partial(lo_t.text, lo_t.start)?;
    let hi = parse_partial(hi_t.text, hi_t.start)?;
    if lo.k == 0 { return Err(RangeError::Syntax { at: lo_t.start }); }
    if hi.k == 0 { return Err(RangeError::Syntax { at: hi_t.start }); }
    let lov = lo.full.clone().unwrap_or_else(|| stable(padded(&lo)));
    let lo_anchor = lo.full.as_ref().map_or(false, |v| !v.pre.is_empty());
    out.push(cmp(Op::Ge, lov, lo_anchor));
    if let Some(v) = hi.full {
        out.push(cmp(Op::Le, v.clone(), !v.pre.is_empty()));
    } else if hi.k == 1 {
        out.push(cmp(Op::Lt, zero_pre(bump(padded(&hi),0,hi_t.start)?), false));
    } else if hi.k == 2 {
        out.push(cmp(Op::Lt, zero_pre(bump(padded(&hi),1,hi_t.start)?), false));
    } else {
        out.push(cmp(Op::Le, stable(padded(&hi)), false));
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s, 0);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut arm_start = 0usize;
    let mut prev_sep: Option<usize> = None;
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: prev_sep.unwrap_or(tokens[i].start) });
            }
            let mut comps = Vec::new();
            parse_hyphen(&current, &mut comps)?;
            arms.push(comps);
            current.clear();
            prev_sep = Some(tokens[i].start);
            arm_start = i + 1;
        } else {
            current.push(tokens[i].clone());
        }
        i += 1;
    }
    if current.is_empty() {
        let at = if let Some(sep) = prev_sep { s.len() } else { 0 };
        return Err(RangeError::Empty { at });
    }
    let mut comps = Vec::new();
    parse_hyphen(&current, &mut comps)?;
    arms.push(comps);
    let _ = arm_start;
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &[Comparator]) -> bool {
    let admitted = arm.iter().any(|c| c.anchor && !c.version.pre.is_empty() && core(&c.version) == core(v));
    if !v.pre.is_empty() && !admitted { return false; }
    arm.iter().all(|c| {
        let o = compare_versions(v, &c.version);
        match c.op { Op::Eq => o == Ordering::Equal, Op::Lt => o == Ordering::Less,
            Op::Le => o != Ordering::Greater, Op::Gt => o == Ordering::Greater,
            Op::Ge => o != Ordering::Less }
    })
}
pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|a| arm_matches(v,a))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError { pub index: usize, pub error: VersionError }

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}
fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index, error })?);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let o = compare_versions(v, &parsed[j]);
            if (max && o == Ordering::Greater) || (!max && o == Ordering::Less) { best = Some(i); }
        } else { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let c = (a.major,a.minor,a.patch).cmp(&(b.major,b.minor,b.patch));
    if c != Ordering::Equal { return c; }
    if a.pre.is_empty() { return if b.pre.is_empty(){Ordering::Equal}else{Ordering::Greater}; }
    if b.pre.is_empty() { return Ordering::Less; }
    for (x,y) in a.pre.iter().zip(&b.pre) {

        let c = match (nx,ny) {
            (true,false)=>Ordering::Less,(false,true)=>Ordering::Greater,

        };

    }

        if is_pre
            && i - ident_start > 1
            && bytes[ident_start..i].iter().all(u8::is_ascii_digit)
            && bytes[ident_start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }
        out.push(s[ident_start..i].to_owned());
        if i == end {
            break;
        }
        ident_start = i + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let (major, mut i) = parse_digits(s, 0)?;
    if byte(s, i) != Some(b'.') {
        return Err(if missing_start(byte(s, i)) {
            VersionError::MissingCore { at: i }
        } else {
            VersionError::InvalidChar { at: i }
        });
    }
    i += 1;
    let (minor, j) = parse_digits(s, i)?;
    i = j;
    if byte(s, i) != Some(b'.') {
        return Err(if missing_start(byte(s, i)) {
            VersionError::MissingCore { at: i }
        } else {
            VersionError::InvalidChar { at: i }
        });
    }
    i += 1;
    let (patch, j) = parse_digits(s, i)?;
    i = j;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    match byte(s, i) {
        None => {}
        Some(b'-') => {
            let pre_start = i + 1;
            let plus = s.as_bytes()[pre_start..]
                .iter()
                .position(|&b| b == b'+')
                .map(|p| pre_start + p);
            let pre_end = plus.unwrap_or(s.len());
            pre = parse_identifiers(s, pre_start, pre_end, true)?;
            if let Some(p) = plus {
                build = parse_identifiers(s, p + 1, s.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, i + 1, s.len(), false)?;
        }
        Some(_) => return Err(VersionError::InvalidChar { at: i }),
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_ident(a: &str, b: &str) -> Ordering {
    let an = is_numeric(a);
    let bn = is_numeric(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let c = cmp_ident(x, y);
                if c != Ordering::Equal {
                    return c;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Bound {
    op: Op,
    version: Version,
}

#[derive(Clone, Debug)]
struct Arm {
    bounds: Vec<Bound>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn cmp_ok(v: &Version, b: &Bound) -> bool {
    match b.op {
        Op::Eq => compare_versions(v, &b.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &b.version) == Ordering::Less,
        Op::Le => compare_versions(v, &b.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &b.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &b.version) != Ordering::Less,
    }
}

fn anchor(v: &Version) -> Option<(u64, u64, u64)> {
    if v.pre.is_empty() {
        None
    } else {
        Some((v.major, v.minor, v.patch))
    }
}

fn full(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn next_version(
    mut v: Version,
    field: usize,
    at: usize,
) -> Result<Version, RangeError> {
    let n = match field {
        0 => &mut v.major,
        1 => &mut v.minor,
        _ => &mut v.patch,
    };
    *n = n.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field == 0 {
        v.minor = 0;
        v.patch = 0;
    } else if field == 1 {
        v.patch = 0;
    }
    v.pre = vec!["0".to_owned()];
    v.build.clear();
    Ok(v)
}

#[derive(Clone, Debug)]
struct Partial {
    version: Version,
    k: usize,
    full: bool,
    any: bool,
}

fn partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.contains('-') || s.contains('+') {
        let v = full(s, at)?;
        return Ok(Partial { version: v, k: 3, full: true, any: false });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.len() >= 3 && fields.iter().all(|f| !matches!(*f, "*" | "x" | "X")) {
        let v = full(s, at)?;
        return Ok(Partial { version: v, k: 3, full: true, any: false });
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen
                || !f.bytes().all(|b| b.is_ascii_digit())
                || (f.len() > 1 && f.starts_with('0'))
                || f.parse::<u64>().is_err()
            {
                return Err(RangeError::Syntax { at });
            }
            nums.push(f.parse::<u64>().unwrap());
        }
    }
    let k = nums.len();
    let mut v = Version { major: 0, minor: 0, patch: 0, pre: Vec::new(), build: Vec::new() };
    if k > 0 { v.major = nums[0]; }
    if k > 1 { v.minor = nums[1]; }
    if k > 2 { v.patch = nums[2]; }
    Ok(Partial { version: v, k, full: false, any: k == 0 })
}

fn add_bound(arm: &mut Arm, op: Op, v: Version, explicit_anchor: bool) {
    if explicit_anchor {
        if let Some(a) = anchor(&v) {
            arm.anchors.push(a);
        }
    }
    arm.bounds.push(Bound { op, version: v });
}

fn expand_partial(arm: &mut Arm, p: Partial, at: usize) -> Result<(), RangeError> {
    if p.any { return Ok(()); }
    if p.full {
        add_bound(arm, Op::Eq, p.version.clone(), true);
        return Ok(());
    }
    if p.k == 3 {
        add_bound(arm, Op::Eq, p.version, false);
        return Ok(());
    }
    add_bound(arm, Op::Ge, p.version.clone(), false);
    let upper = next_version(p.version, if p.k == 1 { 0 } else { 1 }, at)?;
    add_bound(arm, Op::Lt, upper, false);
    Ok(())
}

fn parse_comparator(tok: &str, start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, n) = if tok.starts_with(">=") { (Op::Ge, 2) }
        else if tok.starts_with("<=") { (Op::Le, 2) }
        else if tok.starts_with('>') { (Op::Gt, 1) }
        else if tok.starts_with('<') { (Op::Lt, 1) }
        else if tok.starts_with('=') { (Op::Eq, 1) }
        else { return Err(RangeError::Syntax { at: start }); };
    let operand = &tok[n..];
    let at = start + n;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: start });
    }
    let v = full(operand, at)?;
    add_bound(arm, op, v.clone(), true);
    Ok(())
}

fn parse_arm(s: &str, base: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let toks: Vec<(&str, usize)> = token_positions(s, base);
    let hyphens: Vec<usize> = toks.iter().enumerate().filter(|(_, (t, _))| *t == "-").map(|(i, _)| i).collect();
    if !hyphens.is_empty() {
        let h = hyphens[0];
        if h != 1 || toks.len() != 3 {
            return Err(RangeError::Syntax { at: toks[h].1 });
        }
        let (ls, la) = toks[0];
        let (us, ua) = toks[2];
        let l = partial(ls, la)?;
        let u = partial(us, ua)?;
        if l.any { return Err(RangeError::Syntax { at: la }); }
        if u.any { return Err(RangeError::Syntax { at: ua }); }
        if let Some(a) = anchor(&l.version) { arm.anchors.push(a); }
        add_bound(arm, Op::Ge, l.version.clone(), false);
        if u.full {
            add_bound(arm, Op::Le, u.version.clone(), true);
        } else {
            let upper = if u.k == 1 { next_version(u.version, 0, ua)? }
                else if u.k == 2 { next_version(u.version, 1, ua)? }
                else { u.version.clone() };
            add_bound(arm, if u.k == 3 { Op::Le } else { Op::Lt }, upper, false);
        }
        return Ok(());
    }

    for (tok, start) in toks {
        if tok.starts_with('^') || tok.starts_with('~') {
            let caret = tok.starts_with('^');
            let operand = &tok[1..];
            let at = start + 1;
            let p = partial(operand, at)?;
            if p.any { return Err(RangeError::Syntax { at: start }); }
            if let Some(a) = anchor(&p.version) { arm.anchors.push(a); }
            add_bound(arm, Op::Ge, p.version.clone(), false);
            let field = if caret {
                if p.version.major > 0 || p.k == 1 { 0 }
                else if p.version.minor > 0 || p.k == 2 { 1 }
                else { 2 }
            } else if p.k == 1 { 0 } else { 1 };
            let upper = next_version(p.version, field, at)?;
            add_bound(arm, Op::Lt, upper, false);
        } else if tok.starts_with('<') || tok.starts_with('>') || tok.starts_with('=') {
            parse_comparator(tok, start, arm)?;
        } else if tok.starts_with(',') || tok.starts_with('(') || tok.starts_with(')') || tok == "&" {
            return Err(RangeError::Syntax { at: start });
        } else if tok.starts_with('*') || tok.starts_with('x') || tok.starts_with('X') || tok.as_bytes().first().is_some_and(u8::is_ascii_digit) {
            let p = partial(tok, start)?;
            expand_partial(arm, p, start)?;
        } else {
            return Err(RangeError::Syntax { at: start });
        }
    }
    Ok(())
}

fn token_positions(s: &str, base: usize) -> Vec<(&str, usize)> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < b.len() {
        while i < b.len() && matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') { i += 1; }
        if i == b.len() { break; }
        let start = i;
        while i < b.len() && !matches!(b[i], b' ' | b'\t' | b'\r' | b'\n') { i += 1; }
        out.push((&s[start..i], base + start));
    }
    out
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let all = token_positions(s, 0);
    if all.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut start = 0;
    let mut i = 0;
    while i < all.len() {
        if all[i].0 == "||" {
            if i == start {
                return Err(RangeError::Empty { at: all[i].1 });
            }
            let first = all[start].1;
            let last = if i + 1 < all.len() { all[i + 1].1 } else { s.len() };
            let arm_text = &s[first..all[i].1];
            let mut arm = Arm { bounds: Vec::new(), anchors: Vec::new() };
            parse_arm(arm_text, first, &mut arm)?;
            arms.push(arm);
            start = i + 1;
            if start == all.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
            let _ = last;
        } else if all[i].0.contains("||") {
            // A glued separator remains an operand and is diagnosed by arm parsing.
        }
        i += 1;
    }
    let first = all[start].1;
    let mut arm = Arm { bounds: Vec::new(), anchors: Vec::new() };
    parse_arm(&s[first..], first, &mut arm)?;
    arms.push(arm);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        arm.bounds.iter().all(|b| cmp_ok(v, b))
            && (v.pre.is_empty() || arm.anchors.contains(&(v.major, v.minor, v.patch)))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(candidates: &[String], r: &Range, want_max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        match parse_version(s) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let c = compare_versions(v, &parsed[j]);
            if (want_max && c == Ordering::Greater) || (!want_max && c == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}
