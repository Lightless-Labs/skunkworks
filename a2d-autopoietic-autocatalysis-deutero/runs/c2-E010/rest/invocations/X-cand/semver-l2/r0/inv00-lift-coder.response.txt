- **Error order:** I validate each core number before its separator, then scan prerelease completely before build metadata.
- **Range whitespace:** Only ASCII space, tab, CR, and LF separate tokens; `||` must be its own token.
- **Partial bounds:** Missing fields are zero for lower bounds; generated upper bounds use prerelease `0`.
- **Prerelease admission:** An arm admits candidates only from explicit prerelease operands with the same core tuple.
- **Hyphen ranges:** I require exactly three tokens and parse both endpoints before checking whether either is all-wildcard.
- **Ties and overflow:** Extrema keep the earliest precedence tie; incrementing a `u64` bound overflows as `Syntax` at that operand.

fn parse_digits(s: &str, pos: usize) -> Result<(u64, usize), VersionError> {
    let bytes = s.as_bytes();
    let start = pos;
    let mut end = pos;
    while end < bytes.len() && bytes[end].is_ascii_digit() {
        end += 1;
    }
    if end == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let n = s[start..end]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    Ok((n, end))
}

fn parse_identifiers(
    s: &str,
    start: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut ids = Vec::new();
    let mut p = start;
    loop {
        let end = if pre {
            let mut q = p;
            while q < bytes.len() && bytes[q] != b'.' && bytes[q] != b'+' {
                q += 1;
            }
            q
        } else {
            let mut q = p;
            while q < bytes.len() && bytes[q] != b'.' {
                q += 1;
            }
            q
        };
        if end == p {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
        for i in p..end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }
        if pre && end - p > 1 && bytes[p] == b'0'
            && bytes[p..end].iter().all(u8::is_ascii_digit)
        {
            return Err(VersionError::LeadingZero { at: p });
        }
        ids.push(s[p..end].to_string());
        if end == bytes.len() {
            break;
        }
        if bytes[end] == b'+' && pre {
            break;
        }
        p = end + 1;
        if p == bytes.len() {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(ids)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let (major, p1) = parse_digits(s, 0)?;
    if p1 >= s.len() || s.as_bytes()[p1] != b'.' {
        if p1 >= s.len() || matches!(s.as_bytes()[p1], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p1 });
        }
        return Err(VersionError::InvalidChar { at: p1 });
    }
    let (minor, p2) = parse_digits(s, p1 + 1)?;
    if p2 >= s.len() || s.as_bytes()[p2] != b'.' {
        if p2 >= s.len() || matches!(s.as_bytes()[p2], b'-' | b'+') {
            return Err(VersionError::MissingCore { at: p2 });
        }
        return Err(VersionError::InvalidChar { at: p2 });
    }
    let (patch, mut p) = parse_digits(s, p2 + 1)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < s.len() {
        match s.as_bytes()[p] {
            b'-' => {
                p += 1;
                pre = parse_identifiers(s, p, true)?;
                if p < s.len() {
                    while p < s.len() && s.as_bytes()[p] != b'+' {
                        p += 1;
                    }
                    if p < s.len() {
                        p += 1;
                        build = parse_identifiers(s, p, false)?;
                    }
                }
            }
            b'+' => {
                p += 1;
                build = parse_identifiers(s, p, false)?;
            }
            _ => return Err(VersionError::InvalidChar { at: p }),
        }
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric(x);
        let yn = numeric(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

impl Op {

        match self {
            Op::Eq => c == Ordering::Equal,
            Op::Lt => c == Ordering::Less,
            Op::Le => c != Ordering::Greater,
            Op::Gt => c == Ordering::Greater,
            Op::Ge => c != Ordering::Less,
        }
    }
}

fn check_version(checks: &mut Vec<Check>, anchors: &mut Vec<(u64,u64,u64)>, op: Op, v: Version, explicit: bool) {
    if explicit && !v.pre.is_empty() {
        let t = (v.major, v.minor, v.patch);
        if !anchors.contains(&t) { anchors.push(t); }
    }
    checks.push(Check { op, version: v });

#[derive(Clone)]

    fields: Vec<Option<u64>>,
    full: Option<Version>,
    k: usize,
}

fn operand_version(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    if s.is_empty() { return Err(syntax(at)); }
    let has_sign = s.bytes().any(|b| b == b'-' || b == b'+');
    let pieces: Vec<&str> = s.split('.').collect();
    let has_wild = pieces.iter().any(|p| matches!(*p, "*" | "x" | "X"));
    if has_sign || (pieces.len() >= 3 && !has_wild) {
        let v = operand_version(s, at)?;
        return Ok(Partial { fields: vec![Some(v.major),Some(v.minor),Some(v.patch)], full: Some(v), k: 3 });
    }
    if pieces.len() > 3 { return Err(syntax(at)); }
    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    let mut k = 0;
    for p in pieces {
        if p.is_empty() { return Err(syntax(at)); }
        if matches!(p, "*" | "x" | "X") {
            wildcard_seen = true;
            fields.push(None);
        } else {
            if wildcard_seen || !p.bytes().all(|b| b.is_ascii_digit()) || (p.len() > 1 && p.starts_with('0')) {
                return Err(syntax(at));
            }
            let n = p.parse::<u64>().map_err(|_| syntax(at))?;
            k += 1;
            fields.push(Some(n));
        }
    }
    while fields.len() < 3 { fields.push(None); }
    Ok(Partial { fields, full: None, k })
}

fn lower_version(p: &Partial) -> Version {
    if let Some(v) = &p.full { return v.clone(); }
    Version {
        major: p.fields[0].unwrap_or(0),
        minor: p.fields[1].unwrap_or(0),
        patch: p.fields[2].unwrap_or(0),
        pre: Vec::new(), build: Vec::new()
    }
}

fn upper_increment(p: &Partial, index: usize, at: usize) -> Result<Version, RangeError> {
    let mut vals = [p.fields[0].unwrap_or(0), p.fields[1].unwrap_or(0), p.fields[2].unwrap_or(0)];
    vals[index] = vals[index].checked_add(1).ok_or_else(|| syntax(at))?;
    for x in vals.iter_mut().skip(index + 1) { *x = 0; }
    Ok(Version { major: vals[0], minor: vals[1], patch: vals[2], pre: vec!["0".into()], build: Vec::new() })
}

fn add_partial_bare(p: &Partial, at: usize, checks: &mut Vec<Check>, anchors: &mut Vec<(u64,u64,u64)>) -> Result<(), RangeError> {
    if p.k == 0 { return Ok(()); }
    if p.full.is_some() {
        check_version(checks, anchors, Op::Eq, lower_version(p), true);
    } else {
        check_version(checks, anchors, Op::Ge, lower_version(p), false);
        let idx = if p.k == 1 {0} else {1};
        let upper = upper_increment(p, idx, at)?;
        check_version(checks, anchors, Op::Lt, upper, false);
    }
    Ok(())
}

fn expand_caret_tilde(
    p: &Partial, at: usize, caret: bool,
    checks: &mut Vec<Check>, anchors: &mut Vec<(u64,u64,u64)>
) -> Result<(), RangeError> {
    if p.k == 0 { return Err(syntax(at)); }
    let lower = lower_version(p);
    check_version(checks, anchors, Op::Ge, lower.clone(), p.full.as_ref().map_or(false, |v| !v.pre.is_empty()));
    let idx = if caret {
        if lower.major > 0 || p.k == 1 { 0 }
        else if lower.minor > 0 || p.k == 2 { 1 }
        else { 2 }
    } else if p.k == 1 { 0 } else { 1 };
    let upper = upper_increment(p, idx, at)?;
    check_version(checks, anchors, Op::Lt, upper, false);
    Ok(())
}

fn parse_arm(tokens: &[(String, usize)], arm_start: usize) -> Result<Arm, RangeError> {
    if tokens.is_empty() { return Err(RangeError::Empty { at: arm_start }); }
    let mut minus = None;
    for (i,(t,at)) in tokens.iter().enumerate() {
        if t == "-" { minus = Some((i,*at)); break; }
    }
    let mut checks = Vec::new();
    let mut anchors = Vec::new();
    if let Some((mi, mat)) = minus {
        if mi != 1 || tokens.len() != 3 { return Err(syntax(mat)); }
        let (lo, loa) = (&tokens[0].0, tokens[0].1);
        let (hi, hia) = (&tokens[2].0, tokens[2].1);
        let lp = parse_partial(lo, loa)?;
        let hp = parse_partial(hi, hia)?;
        if lp.k == 0 { return Err(syntax(loa)); }
        if hp.k == 0 { return Err(syntax(hia)); }
        let lv = lower_version(&lp);
        check_version(&mut checks, &mut anchors, Op::Ge, lv, lp.full.as_ref().map_or(false, |v| !v.pre.is_empty()));
        if hp.full.is_some() {
            check_version(&mut checks, &mut anchors, Op::Le, lower_version(&hp), hp.full.as_ref().map_or(false, |v| !v.pre.is_empty()));
        } else {
            let idx = if hp.k == 1 {0} else {1};
            let upper = upper_increment(&hp, idx, hia)?;
            check_version(&mut checks, &mut anchors, Op::Lt, upper, false);
        }
    } else {
        for (token, at) in tokens {
            let bytes = token.as_bytes();
            if bytes.is_empty() { return Err(syntax(*at)); }
            match bytes[0] {
                b'^' | b'~' => {
                    let p = parse_partial(&token[1..], at + 1)?;
                    expand_caret_tilde(&p, at + 1, bytes[0] == b'^', &mut checks, &mut anchors)?;
                }
                b'<' | b'>' | b'=' => {
                    let (op, n) = if token.starts_with("<=") { (Op::Le,2) }
                        else if token.starts_with(">=") { (Op::Ge,2) }
                        else if token.starts_with("<") { (Op::Lt,1) }
                        else if token.starts_with(">") { (Op::Gt,1) }
                        else { (Op::Eq,1) };
                    let operand = &token[n..];
                    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() { return Err(syntax(*at)); }
                    let v = operand_version(operand, at+n)?;
                    check_version(&mut checks, &mut anchors, op, v, true);
                }
                b'0'..=b'9' | b'*' | b'x' | b'X' => {
                    let p = parse_partial(token, *at)?;
                    add_partial_bare(&p, *at, &mut checks, &mut anchors)?;
                }
                _ => return Err(syntax(*at)),
            }
        }
    }
    Ok(Arm { checks, anchors })
}

fn is_ws(b: u8) -> bool { matches!(b, b' ' | b'\t' | b'\r' | b'\n') }

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut tokens: Vec<(String, usize)> = Vec::new();
    let mut arms = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) { i += 1; }
        if i >= bytes.len() { break; }
        let start = i;
        while i < bytes.len() && !is_ws(bytes[i]) { i += 1; }
        let tok = &s[start..i];
        if tok == "||" {
            if tokens.is_empty() {
                return Err(RangeError::Empty { at: start });
            }
            arms.push(parse_arm(&tokens, tokens[0].1)?);
            tokens.clear();
            let mut j = i;
            while j < bytes.len() && is_ws(bytes[j]) { j += 1; }
            if j == bytes.len() {
                return Err(RangeError::Empty { at: bytes.len() });
            }
            i = j;
        } else {
            tokens.push((tok.to_string(), start));
        }
    }
    if tokens.is_empty() {
        if arms.is_empty() { return Err(RangeError::Empty { at: 0 }); }
        return Err(RangeError::Empty { at: bytes.len() });
    }
    arms.push(parse_arm(&tokens, tokens[0].1)?);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let tuple = (v.major, v.minor, v.patch);
    r.arms.iter().any(|arm| {
        if !v.pre.is_empty() && !arm.anchors.contains(&tuple) { return false; }
        arm.checks.iter().all(|c| c.op.test(compare_versions(v, &c.version)))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(candidates: &[String], r: &Range, want_max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        match parse_version(s) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    let mut best: Option<usize> = None;
    for i in 0..candidates.len() {
        if !matches(&parsed[i], r) { continue; }
        if let Some(b) = best {
            let c = compare_versions(&parsed[i], &parsed[b]);
            if (want_max && c == Ordering::Greater) || (!want_max && c == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

}
fn bump(p:&Part,i:usize,at:usize)->Result<Version,RangeError>{
    let mut a=[p.v.major,p.v.minor,p.v.patch];
    a[i]=a[i].checked_add(1).ok_or_else(||syn(at))?;
    for x in a.iter_mut().skip(i+1){*x=0;}
    Ok(Version{major:a[0],minor:a[1],patch:a[2],pre:vec!["0".into()],build:Vec::new()})
}
fn lower(p:&Part)->Version{p.v.clone()}
fn expand(t:&[(String,usize)],arm_start:usize)->Result<Arm,RangeError>{
    if t.is_empty(){return Err(RangeError::Empty{at:arm_start});}
    let mut a=Arm{tests:Vec::new(),anchors:Vec::new()};
    if let Some((i,(_,m)))=t.iter().enumerate().find(|(_,x)|x.0=="-"){
        if i!=1||t.len()!=3{return Err(syn(*m));}
        let lo=parse_part(&t[0].0,t[0].1)?;
        let hi=parse_part(&t[2].0,t[2].1)?;
        if lo.k==0{return Err(syn(t[0].1));} if hi.k==0{return Err(syn(t[2].1));}
        put(&mut a,Op::Ge,lower(&lo),lo.full&&!lo.v.pre.is_empty());
        if hi.full{put(&mut a,Op::Le,lower(&hi),!hi.v.pre.is_empty());}
        else{let u=bump(&hi,if hi.k==1{0}else{1},t[2].1)?;put(&mut a,Op::Lt,u,false);}
        return Ok(a);
    }
    for(tok,at)in t{
        let b=tok.as_bytes();
        match b.first().copied(){
            Some(b'^')|Some(b'~')=>{
                let p=parse_part(&tok[1..],at+1)?;
                if p.k==0{return Err(syn(*at+1));}
                put(&mut a,Op::Ge,lower(&p),p.full&&!p.v.pre.is_empty());
                let i=if b[0]==b'~'{if p.k==1{0}else{1}}else if p.v.major>0||p.k==1{0}else if p.v.minor>0||p.k==2{1}else{2};
                let u=bump(&p,i,*at+1)?;put(&mut a,Op::Lt,u,false);
            }
            Some(b'<')|Some(b'>')|Some(b'=')=>{
                let(op,n)=if tok.starts_with("<="){(Op::Le,2)}else if tok.starts_with(">="){(Op::Ge,2)}else if tok.starts_with('<'){(Op::Lt,1)}else if tok.starts_with('>'){(Op::Gt,1)}else{(Op::Eq,1)};
                if tok.len()==n||!tok.as_bytes()[n].is_ascii_digit(){return Err(syn(*at));}
                let v=parse_version(&tok[n..]).map_err(|error|RangeError::Version{at:*at+n,error})?;
                put(&mut a,op,v,true);
            }
            Some(b'0'..=b'9')|Some(b'*')|Some(b'x')|Some(b'X')=>{
                let p=parse_part(tok,*at)?;
                if p.k==0{return Ok(a);}
                if p.full{put(&mut a,Op::Eq,p.v,true);}
                else{
                    put(&mut a,Op::Ge,lower(&p),false);
                    let u=bump(&p,if p.k==1{0}else{1},*at)?;put(&mut a,Op::Lt,u,false);
                }
            }
            _=>return Err(syn(*at))
        }
    }
    Ok(a)
}
fn ws(b:u8)->bool{matches!(b,b' '|b'\t'|b'\r'|b'\n')}
pub fn parse_range(s:&str)->Result<Range,RangeError>{
    let b=s.as_bytes();let mut i=0;let mut ts=Vec::new();let mut arms=Vec::new();
    while i<b.len(){
        while i<b.len()&&ws(b[i]){i+=1;}
        if i==b.len(){break;}
        let st=i;while i<b.len()&&!ws(b[i]){i+=1;}
        let z=&s[st..i];
        if z=="||"{
            if ts.is_empty(){return Err(RangeError::Empty{at:st});}
            arms.push(expand(&ts,ts[0].1)?);ts.clear();
            let mut q=i;while q<b.len()&&ws(b[q]){q+=1;}
            if q==b.len(){return Err(RangeError::Empty{at:b.len()});}
            i=q;
        }else{ts.push((z.to_owned(),st));}
    }
    if ts.is_empty(){return Err(RangeError::Empty{at:if arms.is_empty(){0}else{b.len()}});}
    arms.push(expand(&ts,ts[0].1)?);Ok(Range{arms})
}
pub fn matches(v:&Version,r:&Range)->bool{
    let t=(v.major,v.minor,v.patch);
    r.arms.iter().any(|a|(v.pre.is_empty()||a.anchors.contains(&t))&&a.tests.iter().all(|(o,x)|o.ok(compare_versions(v,x))))
}
#[derive(Clone,Debug,PartialEq,Eq)]
pub struct CandidateError{pub index:usize,pub error:VersionError}
fn extreme(c:&[String],r:&Range,max:bool)->Result<Option<String>,CandidateError>{
    let mut vs=Vec::new();
    for(i,s)in c.iter().enumerate(){vs.push(parse_version(s).map_err(|error|CandidateError{index:i,error})?);}
    let mut best=None;
    for i in 0..c.len(){if !matches(&vs[i],r){continue;}if let Some(j)=best{let o=compare_versions(&vs[i],&vs[j]);if(max&&o==Ordering::Greater)||(!max&&o==Ordering::Less){best=Some(i);}}else{best=Some(i);}}
    Ok(best.map(|i|c[i].clone()))
}
pub fn max_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extreme(candidates,r,true)}
pub fn min_satisfying(candidates:&[String],r:&Range)->Result<Option<String>,CandidateError>{extreme(candidates,r,false)}
