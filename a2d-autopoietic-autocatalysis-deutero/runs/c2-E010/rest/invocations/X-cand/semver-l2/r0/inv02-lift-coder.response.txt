- Core numeric errors are checked before their following separators; identifier character errors are checked before numeric prerelease leading zeros.
- A prerelease is admitted per arm only by an explicit prerelease operand with the candidate’s exact core tuple.
- Partial operands expand to comparator bounds; generated bounds never create admission anchors.
- A hyphen range is recognized before parsing other tokens in its arm, and malformed shape reports the first standalone `-`.
- Increment overflow is a syntax error at the operand being expanded, with no carry.
- Extrema parse every candidate before selecting and preserve the earliest input on precedence ties.

fn parse_core_num(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return if is_sep_missing(bytes.get(start).copied()) {
            Err(VersionError::MissingCore { at: start })
        } else {
            Err(VersionError::InvalidChar { at: start })
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut n = 0u64;
    for b in digits.bytes() {
        n = n.checked_mul(10).and_then(|x| x.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(n)
}

fn parse_ids(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {

            let b = bytes[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }

            return Err(VersionError::EmptyIdentifier { at: id_start });

        if p == end {

        }
    }
    Ok(out)

pub fn parse_version(s: &str) -> Result<Version, VersionError> {

        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = parse_core_num(s, &mut p)?;
    if s.as_bytes().get(p) != Some(&b'.') {
        return if is_sep_missing(s.as_bytes().get(p).copied()) {
            Err(VersionError::MissingCore { at: p })
        } else {
            Err(VersionError::InvalidChar { at: p })
        };
    }
    p += 1;
    let minor = parse_core_num(s, &mut p)?;
    if s.as_bytes().get(p) != Some(&b'.') {
        return if is_sep_missing(s.as_bytes().get(p).copied()) {
            Err(VersionError::MissingCore { at: p })
        } else {
            Err(VersionError::InvalidChar { at: p })
        };
    }
    p += 1;
    let patch = parse_core_num(s, &mut p)?;
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < bytes.len() && bytes[p] == b'-' {
        p += 1;
        let start = p;
        let end = bytes[p..].iter().position(|&b| b == b'+').map(|x| p + x).unwrap_or(bytes.len());
        pre = parse_ids(s, start, end, true)?;
        p = end;
    }
    if p < bytes.len() && bytes[p] == b'+' {
        p += 1;
        build = parse_ids(s, p, bytes.len(), false)?;
        p = bytes.len();
    }
    if p != bytes.len() {
        return Err(VersionError::InvalidChar { at: p });
    }
    Ok(Version { major, minor, patch, pre, build })
}

            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,

                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };

                    }
                }

            }
        }

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let b = s.as_bytes();
    if s.is_empty() { return Err(RangeError::Syntax { at }); }
    let contains_suffix = s.contains('-') || s.contains('+');
    let parts: Vec<&str> = s.split('.').collect();
    let has_wild = parts.iter().any(|x| matches!(*x, "*" | "x" | "X"));
    if contains_suffix || (parts.len() >= 3 && !has_wild) {
        let v = full_or_error(s, at)?;
        return Ok(Partial { fields: vec![v.major, v.minor, v.patch], k: 3, full: Some(v) });
    }
    if parts.len() > 3 { return Err(RangeError::Syntax { at }); }
    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    for part in parts {
        if part.is_empty() { return Err(RangeError::Syntax { at }); }
        if matches!(part, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !part.bytes().all(|c| c.is_ascii_digit()) ||
                (part.len() > 1 && part.as_bytes()[0] == b'0') {
                return Err(RangeError::Syntax { at });
            }
            let mut n = 0u64;
            for c in part.bytes() {
                n = n.checked_mul(10).and_then(|x| x.checked_add((c - b'0') as u64))
                    .ok_or(RangeError::Syntax { at })?;
            }
            fields.push(n);
        }
    }
    let k = fields.len();
    let _ = b;
    Ok(Partial { fields, k, full: None })
}

     p.fields.get(2).copied().unwrap_or(0))
}

fn bump(t: (u64,u64,u64), field: usize, at: usize) -> Result<(u64,u64,u64), RangeError> {

}

fn push_partial_arm(tokens: &[Token<'_>], out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    for t in tokens {
        let txt = t.text;
        if txt == "-" || txt.contains(',') || txt.contains('(') || txt.contains(')') || txt == "&" {
            return Err(RangeError::Syntax { at: t.start });
        }
        if let Some(rest) = txt.strip_prefix('^').or_else(|| txt.strip_prefix('~')) {
            let opbyte = txt.as_bytes()[0];
            let op_at = t.start;
            let operand_at = t.start + 1;
            let p = parse_partial(rest, operand_at)?;
            if p.k == 0 { return Err(RangeError::Syntax { at: op_at }); }
            let lower = p.full.clone().unwrap_or_else(|| stable(padded(&p)));
            let lo_anchor = p.full.as_ref().map_or(false, |v| !v.pre.is_empty());
            let field = if opbyte == b'^' {
                if p.fields[0] > 0 || p.k == 1 { 0 }
                else if p.fields.get(1).copied().unwrap_or(0) > 0 || p.k == 2 { 1 }
                else { 2 }
            } else if p.k == 1 { 0 } else { 1 };
            let upper = zero_pre(bump(padded(&p), field, operand_at)?);
            out.push(cmp(Op::Ge, lower, lo_anchor));
            out.push(cmp(Op::Lt, upper, false));
            continue;
        }
        let (op, operand, operand_at) = if txt.starts_with(">=") { (Some(Op::Ge), &txt[2..], t.start+2) }
            else if txt.starts_with("<=") { (Some(Op::Le), &txt[2..], t.start+2) }
            else if txt.starts_with('>') { (Some(Op::Gt), &txt[1..], t.start+1) }
            else if txt.starts_with('<') { (Some(Op::Lt), &txt[1..], t.start+1) }
            else if txt.starts_with('=') { (Some(Op::Eq), &txt[1..], t.start+1) }
            else { (None, txt, t.start) };
        if let Some(op) = op {
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at: t.start });
            }
            let v = full_or_error(operand, operand_at)?;
            out.push(cmp(op, v.clone(), !v.pre.is_empty()));
            continue;
        }
        let first = txt.as_bytes()[0];
        if !(first.is_ascii_digit() || matches!(first, b'*' | b'x' | b'X')) {
            return Err(RangeError::Syntax { at: t.start });
        }
        let p = parse_partial(txt, t.start)?;
        if let Some(v) = p.full {
            out.push(cmp(Op::Eq, v.clone(), !v.pre.is_empty()));
        } else if p.k == 1 {
            let lo = padded(&p);
            out.push(cmp(Op::Ge, stable(lo), false));
            out.push(cmp(Op::Lt, zero_pre(bump(lo, 0, t.start)?), false));
        } else if p.k == 2 {
            let lo = padded(&p);
            out.push(cmp(Op::Ge, stable(lo), false));
            out.push(cmp(Op::Lt, zero_pre(bump(lo, 1, t.start)?), false));
        }
    }
    Ok(())
}

fn parse_hyphen(tokens: &[Token<'_>], out: &mut Vec<Comparator>) -> Result<(), RangeError> {
    let mins: Vec<usize> = tokens.iter().enumerate().filter(|(_,t)| t.text == "-").map(|(i,_)| i).collect();
    if !mins.is_empty() && (tokens.len() != 3 || mins != vec![1]) {
        return Err(RangeError::Syntax { at: tokens[mins[0]].start });
    }
    if mins.is_empty() { return push_partial_arm(tokens, out); }
    let lo_t = &tokens[0];
    let hi_t = &tokens[2];
    let lo = parse_partial(lo_t.text, lo_t.start)?;
    let hi = parse_partial(hi_t.text, hi_t.start)?;
    if lo.k == 0 { return Err(RangeError::Syntax { at: lo_t.start }); }
    if hi.k == 0 { return Err(RangeError::Syntax { at: hi_t.start }); }
    let lov = lo.full.clone().unwrap_or_else(|| stable(padded(&lo)));
    let lo_anchor = lo.full.as_ref().map_or(false, |v| !v.pre.is_empty());
    out.push(cmp(Op::Ge, lov, lo_anchor));
    if let Some(v) = hi.full {
        out.push(cmp(Op::Le, v.clone(), !v.pre.is_empty()));
    } else if hi.k == 1 {
        out.push(cmp(Op::Lt, zero_pre(bump(padded(&hi),0,hi_t.start)?), false));
    } else if hi.k == 2 {
        out.push(cmp(Op::Lt, zero_pre(bump(padded(&hi),1,hi_t.start)?), false));
    } else {
        out.push(cmp(Op::Le, stable(padded(&hi)), false));
    }
    Ok(())
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s, 0);
    if tokens.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut arm_start = 0usize;
    let mut prev_sep: Option<usize> = None;
    let mut i = 0;
    while i < tokens.len() {
        if tokens[i].text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: prev_sep.unwrap_or(tokens[i].start) });
            }
            let mut comps = Vec::new();
            parse_hyphen(&current, &mut comps)?;
            arms.push(comps);
            current.clear();
            prev_sep = Some(tokens[i].start);
            arm_start = i + 1;
        } else {
            current.push(tokens[i].clone());
        }
        i += 1;
    }
    if current.is_empty() {
        let at = if let Some(sep) = prev_sep { s.len() } else { 0 };
        return Err(RangeError::Empty { at });
    }
    let mut comps = Vec::new();
    parse_hyphen(&current, &mut comps)?;
    arms.push(comps);
    let _ = arm_start;
    Ok(Range { arms })
}

fn arm_matches(v: &Version, arm: &[Comparator]) -> bool {
    let admitted = arm.iter().any(|c| c.anchor && !c.version.pre.is_empty() && core(&c.version) == core(v));
    if !v.pre.is_empty() && !admitted { return false; }
    arm.iter().all(|c| {
        let o = compare_versions(v, &c.version);
        match c.op { Op::Eq => o == Ordering::Equal, Op::Lt => o == Ordering::Less,
            Op::Le => o != Ordering::Greater, Op::Gt => o == Ordering::Greater,
            Op::Ge => o != Ordering::Less }
    })
}
pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|a| arm_matches(v,a))
}

pub struct CandidateError { pub index: usize, pub error: VersionError }

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}
pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}
fn extrema(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index, error })?);
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) { continue; }
        if let Some(j) = best {
            let o = compare_versions(v, &parsed[j]);
            if (max && o == Ordering::Greater) || (!max && o == Ordering::Less) { best = Some(i); }
        } else { best = Some(i); }
    }
    Ok(best.map(|i| candidates[i].clone()))
}
