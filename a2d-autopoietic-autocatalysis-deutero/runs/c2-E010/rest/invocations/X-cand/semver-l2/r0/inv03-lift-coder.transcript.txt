#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {

    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },

fn parse_field(s: &str, at: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let begin = *at;
    while *at < bytes.len() && bytes[*at].is_ascii_digit() {
        *at += 1;
    }
    if begin == *at {
        return Err(if begin == bytes.len() || matches!(bytes[begin], b'.' | b'-' | b'+') {
            VersionError::MissingCore { at: begin }
        } else {
            VersionError::InvalidChar { at: begin }
        });
    }
    let part = &s[begin..*at];
    if part.len() > 1 && part.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: begin });
    }
    match part.parse::<u64>() {
        Ok(n) => Ok(n),
        Err(_) => Err(VersionError::Overflow { at: begin }),
    }
}

fn parse_ids(s: &str, from: usize, to: usize, pre: bool) -> Result<Vec<String>, VersionError> {
    let b = s.as_bytes();
    let mut ids = Vec::new();
    let mut start = from;
    while start <= to {
        let mut end = start;
        while end < to && b[end] != b'.' {
            if !(b[end].is_ascii_alphanumeric() || b[end] == b'-') {
                return Err(VersionError::InvalidChar { at: end });
            }
            end += 1;
        }
        if start == end {
            return Err(VersionError::EmptyIdentifier { at: start });

            return Err(VersionError::LeadingZero { at: start });
        }
        ids.push(id.to_owned());
        if end == to {
            break;
        }
        start = end + 1;
    }
    Ok(ids)
}

                VersionError::InvalidChar { at: p }
            });

        p += 1;
        *slot = parse_field(s, &mut p)?;

    }
