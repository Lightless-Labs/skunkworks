pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Dependency validation has precedence over every other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }
            if value == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    let mut ready = BinaryHeap::new();
    for id in 0..n {
        if indegree[id] == 0 {
            ready.push(Reverse(id));
        }
    }
    let mut order = Vec::with_capacity(n);
    while let Some(Reverse(id)) = ready.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                ready.push(Reverse(child));
            }
        }
    }
    if order.len() != n {
        let removed = {
            let mut r = vec![false; n];
            for &id in &order { r[id] = true; }
            r
        };
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = children[id].iter().map(|&c| tails[c]).max().unwrap_or(0);
        tails[id] = tasks[id].duration.checked_add(max_child)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running: BinaryHeap<Reverse<(u64, usize, usize)>> = BinaryHeap::new();
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut completed_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all positive-duration jobs due now, ordered by task ID.
        let mut due = Vec::new();
        while let Some(Reverse((finish, task, worker))) = running.peek().copied() {
            if finish != now { break; }
            running.pop();
            due.push((task, worker));
        }
        due.sort_unstable_by_key(|&(task, _)| task);
        for (task, worker) in due {
            completed[task] = true;
            completed_count += 1;
            makespan = makespan.max(now);
            events.push(Event { time: now, task, worker, kind: EventKind::Finish });
        }

        // Admit one task at a time, reconsidering after every start.
        loop {
            let worker = (0..workers).find(|&w| worker_available[w] <= now);
            let Some(worker) = worker else { break };
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now
                    || !tasks[id].deps.iter().all(|&d| completed[d])
                {
                    continue;
                }
                if best.map_or(true, |b| {
                    tails[id] > tails[b]
                        || (tails[id] == tails[b] && tasks[id].priority > tasks[b].priority)
                        || (tails[id] == tails[b] && tasks[id].priority == tasks[b].priority && id < b)
                }) {
                    best = Some(id);
                }
            }
            let Some(task) = best else { break };

            let finish = now.checked_add(tasks[task].duration)
                .ok_or(Error::Overflow { task, stage: Stage::Finish })?;
            let available = finish.checked_add(turnaround)
                .ok_or(Error::Overflow { task, stage: Stage::Turnaround })?;
            started[task] = true;
            worker_available[worker] = available;
            runs[task] = Some(Run { start: now, finish, worker });
            events.push(Event { time: now, task, worker, kind: EventKind::Start });

            if finish == now {
                completed[task] = true;
                completed_count += 1;
                makespan = makespan.max(now);
                events.push(Event { time: now, task, worker, kind: EventKind::Finish });
            } else {
                running.push(Reverse((finish, task, worker)));
            }
        }

        if completed_count == n { break; }

        let mut next: Option<u64> = None;
        if let Some(Reverse((finish, _, _))) = running.peek().copied() {
            if finish > now { next = Some(finish); }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |x| x.min(tasks[id].release)));
            }
        }
        for &available in &worker_available {
            if available > now {
                next = Some(next.map_or(available, |x| x.min(available)));
            }
        }
        let Some(next_time) = next else {
            // No future event can make progress. This is unreachable for a
            // valid acyclic graph, but avoids relying on that fact for safety.
            break;
        };
        now = next_time;
    }

    let runs = runs.into_iter().map(|r| r.expect("acyclic schedule must start every task")).collect();
    Ok(Schedule { runs, makespan, events })
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency entry before checking any other condition.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }
            if value == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    // Build successor lists and indegrees.
    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(task_id);
        }
    }

    // Kahn order, always selecting the smallest currently available ID.
    let mut available = std::collections::BinaryHeap::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }
    let mut order = Vec::with_capacity(n);
    let mut remaining_indegree = indegree.clone();
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &successors[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for id in 0..n {
            if remaining_indegree[id] != 0 {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    // Bottom levels, computed in reverse topological order.
    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_successor_tail = successors[id]
            .iter()
            .map(|&child| tail[child])
            .max()
            .unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut completed_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all jobs finishing now, in task ID order.
        running.sort_unstable_by_key(|&(finish, task_id, _)| (finish, task_id));
        let mut i = 0;
        while i < running.len() {
            if running[i].0 == now {
                let (_, task_id, worker_id) = running.remove(i);
                completed[task_id] = true;
                completed_count += 1;
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
            } else {
                i += 1;
            }
        }

        // Repeatedly start one eligible task, reevaluating after each admission.
        loop {
            let worker_id = match worker_available.iter().position(|&t| t <= now) {
                Some(id) => id,
                None => break,
            };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if !tasks[id].deps.iter().all(|&pred| completed[pred]) {
                    continue;
                }
                let better = match best {
                    None => true,
                    Some(current) => {
                        tail[id] > tail[current]
                            || (tail[id] == tail[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority && id < current)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }

            let task_id = match best {
                Some(id) => id,
                None => break,
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker_id] = available_at;
            runs[task_id] = Some(Run { start: now, finish, worker: worker_id });
            events.push(Event {
                time: now,
                task: task_id,
                worker: worker_id,
                kind: EventKind::Start,
            });
            running.push((finish, task_id, worker_id));

            if finish > makespan {
                makespan = finish;
            }

            // Zero-duration work completes immediately before another selection.
            if finish == now {
                running.pop();
                completed[task_id] = true;
                completed_count += 1;
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker: worker_id,
                    kind: EventKind::Finish,
                });
            }
        }

        if completed_count == n {
            break;
        }

        // Move to the earliest strictly later event that can change eligibility.
        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |v| v.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                let release = tasks[id].release;
                next_time = Some(next_time.map_or(release, |v| v.min(release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next_time = Some(next_time.map_or(available_at, |v| v.min(available_at)));
            }
        }

        now = next_time.expect("unfinished tasks always have a future event");
    }

    let runs = runs.into_iter().map(|run| run.expect("each task starts once")).collect();
    Ok(Schedule { runs, makespan, events })
}

#[derive(Debug, Clone, PartialEq, Eq)]

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency { task: usize, position: usize, value: usize },
    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },
    Cycle { blocked: Vec<usize> },
    Workers { count: usize },
    Overflow { task: usize, stage: Stage },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }
            if value == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
        }
    }
    if workers == 0 || workers > 64 {
        return Err(Error::Workers { count: workers });
    }

    let mut indegree: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut children = vec![Vec::<usize>::new(); n];
    for (id, task) in tasks.iter().enumerate() {
        for &dep in &task.deps {
            children[dep].push(id);
        }
    }

    // Sorted pending vector supplies the smallest available Kahn node.
    let mut pending: Vec<usize> = (0..n).filter(|&id| indegree[id] == 0).collect();
    let mut order = Vec::with_capacity(n);
    while !pending.is_empty() {
        let id = pending.remove(0);
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let at = pending.binary_search(&child).unwrap_or_else(|p| p);
                pending.insert(at, child);
            }
        }
    }
    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order { removed[id] = true; }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let mut downstream = 0u64;
        for &child in &children[id] {
            downstream = downstream.max(tails[child]);
        }
        tails[id] = tasks[id].duration.checked_add(downstream)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_free = vec![0u64; workers];
    let mut running: Vec<(u64, usize, usize)> = Vec::new();
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut count_done = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while count_done < n {
        running.sort_unstable_by_key(|&(finish, task, _)| (finish, task));
        let due_count = running.partition_point(|&(finish, _, _)| finish == now);
        let due: Vec<_> = running.drain(..due_count).collect();
        for (_, task, worker) in due {
            completed[task] = true;
            count_done += 1;
            makespan = makespan.max(now);
            events.push(Event { time: now, task, worker, kind: EventKind::Finish });
        }

        loop {
            let Some(worker) = (0..workers).find(|&w| worker_free[w] <= now) else { break };
            let mut selected = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now
                    || tasks[id].deps.iter().any(|&dep| !completed[dep])
                {
                    continue;
                }
                selected = match selected {
                    None => Some(id),
                    Some(old) if tails[id] > tails[old]
                        || (tails[id] == tails[old] && tasks[id].priority > tasks[old].priority)
                        || (tails[id] == tails[old] && tasks[id].priority == tasks[old].priority && id < old)
                        => Some(id),
                    other => other,
                };
            }
            let Some(task) = selected else { break };

            let finish = now.checked_add(tasks[task].duration)
                .ok_or(Error::Overflow { task, stage: Stage::Finish })?;
            let free = finish.checked_add(turnaround)
                .ok_or(Error::Overflow { task, stage: Stage::Turnaround })?;
            started[task] = true;
            worker_free[worker] = free;
            runs[task] = Some(Run { start: now, finish, worker });
            events.push(Event { time: now, task, worker, kind: EventKind::Start });

            if finish == now {
                completed[task] = true;
                count_done += 1;
                makespan = makespan.max(now);
                events.push(Event { time: now, task, worker, kind: EventKind::Finish });
            } else {
                running.push((finish, task, worker));
            }
        }

        if count_done == n { break; }

        let mut next: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next = Some(next.map_or(finish, |x| x.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |x| x.min(tasks[id].release)));
            }
        }
        for &free in &worker_free {
            if free > now {
                next = Some(next.map_or(free, |x| x.min(free)));
            }
        }
        let Some(next_time) = next else { break };
        now = next_time;
    }

    let runs = runs.into_iter().map(|r| r.expect("acyclic schedule must start every task")).collect();
    Ok(Schedule { runs, makespan, events })
}

        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }
            if value == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
        }
    }

    if workers == 0 || workers > 64 {
        return Err(Error::Workers { count: workers });
    }

    let mut successors = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (id, task) in tasks.iter().enumerate() {
        indegree[id] = task.deps.len();
        for &dep in &task.deps {
            successors[dep].push(id);
        }
    }

    let mut indeg = indegree.clone();
    let mut topo = Vec::with_capacity(n);
    let mut queued = vec![false; n];
    for id in 0..n {
        if indeg[id] == 0 {
            queued[id] = true;
        }
    }

    // Repeated minimum selection gives the required deterministic Kahn traversal.
    loop {
        let next = (0..n).find(|&id| queued[id]);
        let id = match next {
            Some(id) => id,
            None => break,
        };
        queued[id] = false;
        topo.push(id);
        for &child in &successors[id] {
            indeg[child] -= 1;
            if indeg[child] == 0 {
                queued[child] = true;
            }
        }
    }

    if topo.len() != n {
        let blocked = (0..n).filter(|&id| indeg[id] != 0).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut tail = vec![0u64; n];
    for &id in topo.iter().rev() {
        let mut longest_child = 0u64;
        for &child in &successors[id] {
            longest_child = longest_child.max(tail[child]);
        }
        tail[id] = tasks[id]
            .duration
            .checked_add(longest_child)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let mut runs = vec![Run { start: 0, finish: 0, worker: 0 }; n];
    let mut started = vec![false; n];
    let mut done = vec![false; n];
    let mut worker_free_at = vec![0u64; workers];
    let mut running: Vec<(u64, usize, usize)> = Vec::new();

    let mut now = 0u64;
    let mut complete_count = 0usize;
    let mut makespan = 0u64;

    while complete_count < n {
        running.sort_unstable_by_key(|entry| (entry.0, entry.1));
        let mut index = 0;
        while index < running.len() {
            if running[index].0 == now {
                let (_, id, worker) = running.remove(index);
                done[id] = true;
                complete_count += 1;
                events.push(Event { time: now, task: id, worker, kind: EventKind::Finish });
            } else {
                index += 1;
            }
        }

        loop {
            let worker = match worker_free_at.iter().position(|&free_at| free_at <= now) {
                Some(worker) => worker,
                None => break,
            };
            let candidate = (0..n)
                .filter(|&id| {
                    !started[id]
                        && tasks[id].release <= now
                        && tasks[id].deps.iter().all(|&dep| done[dep])
                })
                .min_by(|&a, &b| {
                    tail[b]
                        .cmp(&tail[a])
                        .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                        .then_with(|| a.cmp(&b))
                });

            let id = match candidate {
                Some(id) => id,
                None => break,
            };
            let finish = now.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let worker_free = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Turnaround,
            })?;

            started[id] = true;
            worker_free_at[worker] = worker_free;
            runs[id] = Run { start: now, finish, worker };
            events.push(Event { time: now, task: id, worker, kind: EventKind::Start });
            makespan = makespan.max(finish);

            if finish == now {
                done[id] = true;
                complete_count += 1;
                events.push(Event { time: now, task: id, worker, kind: EventKind::Finish });
            } else {
                running.push((finish, id, worker));
            }
        }

        if complete_count == n {
            break;
        }

        let mut next = None;
        let mut consider = |time: u64| {
            if time > now {
                next = Some(next.map_or(time, |old: u64| old.min(time)));
            }
        };
        for &(finish, _, _) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &free_at in &worker_free_at {
            consider(free_at);
        }
        now = next.expect("a nonterminal schedule has a future event");
    }

    Ok(Schedule { runs, makespan, events })
}

    let mut completed = vec![false; n];

        for id in 0..n {

                blocked.push(id);
            }

        return Err(Error::Cycle { blocked });

            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;

        task: usize,
        worker: usize,
        finish: u64,

            worker_available[worker] = available;
            runs[id] = Some(Run { start: now, finish, worker });
            events.push(Event { time: now, task: id, worker, kind: EventKind::Start });

            if finish == now {
                completed[id] = true;
                done += 1;
                makespan = makespan.max(finish);
                events.push(Event { time: now, task: id, worker, kind: EventKind::Finish });
            } else {
                running.push(Running { task: id, worker, finish });
            }
        }

        if done == n {
            break;
        }

        let mut next = u64::MAX;
        let mut found = false;
        for r in &running {
            if r.finish > now && (!found || r.finish < next) {
                next = r.finish;
                found = true;
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now && (!found || tasks[id].release < next) {
                next = tasks[id].release;
                found = true;
            }
        }
        for &available in &worker_available {
            if available > now && (!found || available < next) {
                next = available;
                found = true;
            }
        }
        if found {
            now = next;
        } else {
            break;
        }
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();
    Ok(Schedule { runs, makespan, events })
}

pub enum Error {
    BadDependency { task: usize, position: usize, value: usize },
    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },
    Cycle { blocked: Vec<usize> },
    Workers { count: usize },
    Overflow { task: usize, stage: Stage },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Min-heap behavior via Reverse.
    let mut ready = std::collections::BinaryHeap::new();
    for id in 0..n {
        if indegree[id] == 0 {
            ready.push(std::cmp::Reverse(id));
        }
    }

    let mut topo = Vec::with_capacity(n);
    let mut remaining_indegree = indegree.clone();
    while let Some(std::cmp::Reverse(id)) = ready.pop() {
        topo.push(id);
        for &child in &children[id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                ready.push(std::cmp::Reverse(child));
            }
        }
    }

    if topo.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &id in &topo {
            removed[id] = true;
        }
        for id in 0..n {
            if !removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut tail = vec![0u64; n];
    for &id in topo.iter().rev() {
        let mut max_successor_tail = 0u64;
        for &child in &children[id] {
            max_successor_tail = max_successor_tail.max(tail[child]);
        }
        tail[id] = tasks[id]
            .duration
            .checked_add(max_successor_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::with_capacity(n.saturating_mul(2));
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut now = 0u64;
    let mut finished_count = 0usize;
    let mut makespan = 0u64;

    while finished_count < n {
        // Complete all positive-duration tasks finishing now, in task ID order.
        running.sort_unstable_by_key(|&(finish, task, _)| (finish, task));
        let mut i = 0;
        while i < running.len() {
            if running[i].0 == now {
                let (_, task, worker) = running.remove(i);
                completed[task] = true;
                finished_count += 1;
                makespan = makespan.max(now);
                events.push(Event {
                    time: now,
                    task,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                i += 1;
            }
        }

        loop {
            let available_worker = worker_available
                .iter()
                .position(|&available_at| available_at <= now);
            let Some(worker) = available_worker else {
                break;
            };

            let mut selected: Option<usize> = None;
            for task_id in 0..n {
                if started[task_id]
                    || tasks[task_id].release > now
                    || !tasks[task_id].deps.iter().all(|&pred| completed[pred])
                {
                    continue;
                }
                let better = match selected {
                    None => true,
                    Some(current) => {
                        tail[task_id] > tail[current]
                            || (tail[task_id] == tail[current]
                                && (tasks[task_id].priority > tasks[current].priority
                                    || (tasks[task_id].priority == tasks[current].priority
                                        && task_id < current)))
                    }
                };
                if better {
                    selected = Some(task_id);
                }
            }

            let Some(task_id) = selected else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task: task_id,
                worker,
                kind: EventKind::Start,
            });

            if tasks[task_id].duration == 0 {
                completed[task_id] = true;
                finished_count += 1;
                makespan = makespan.max(now);
                events.push(Event {
                    time: now,
                    task: task_id,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                running.push((finish, task_id, worker));
            }
        }

        if finished_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next_time = Some(next_time.map_or(finish, |t| t.min(finish)));
            }
        }
        for task_id in 0..n {
            if !started[task_id] && tasks[task_id].release > now {
                let t = tasks[task_id].release;
                next_time = Some(next_time.map_or(t, |cur| cur.min(t)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next_time = Some(next_time.map_or(available_at, |cur| cur.min(available_at)));
            }
        }

        match next_time {
            Some(t) => now = t,
            None => break,
        }
    }

    let runs = runs
        .into_iter()
        .map(|run| run.expect("acyclic tasks are all scheduled"))
        .collect();

    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}

    DuplicateDependency { task: usize, position: usize, value: usize },

    Workers { count: usize },
    Overflow { task: usize, stage: Stage },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            } else if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            } else if !seen.insert(value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if workers == 0 || workers > 64 {
        return Err(Error::Workers { count: workers });
    }

    let mut successors = vec![Vec::<usize>::new(); n];
    let mut degree = vec![0usize; n];
    for (id, task) in tasks.iter().enumerate() {
        degree[id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(id);
        }
    }

    let mut zeroes = Vec::new();
    for id in 0..n {
        if degree[id] == 0 {
            zeroes.push(id);
        }
    }
    let mut order = Vec::with_capacity(n);
    while !zeroes.is_empty() {
        zeroes.sort_unstable();
        let id = zeroes.remove(0);
        order.push(id);
        for &child in &successors[id] {
            degree[child] -= 1;
            if degree[child] == 0 {
                zeroes.push(child);
            }
        }
    }

    if order.len() != n {
        let mut in_order = vec![false; n];
        for &id in &order {
            in_order[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !in_order[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut done = vec![false; n];
    let mut began = vec![false; n];
    let mut worker_free = vec![0u64; workers];
    let mut active = Vec::<(u64, usize, usize)>::new();
    let mut now = 0u64;
    let mut count_done = 0usize;
    let mut makespan = 0u64;

    while count_done < n {
        active.sort_unstable_by_key(|&(finish, id, _)| (finish, id));
        let mut index = 0;
        while index < active.len() {
            if active[index].0 == now {
                let (_, id, worker) = active.remove(index);
                done[id] = true;
                count_done += 1;
                makespan = makespan.max(now);
                events.push(Event {
                    time: now,
                    task: id,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                index += 1;
            }
        }

        loop {
            let Some(worker) = worker_free.iter().position(|&t| t <= now) else {
                break;
            };
            let mut candidate: Option<usize> = None;
            for id in 0..n {
                if began[id] || tasks[id].release > now {
                    continue;
                }
                if tasks[id].deps.iter().any(|&p| !done[p]) {
                    continue;
                }
                if candidate.map_or(true, |old| {
                    (tails[id], tasks[id].priority, std::cmp::Reverse(id))
                        > (tails[old], tasks[old].priority, std::cmp::Reverse(old))
                }) {
                    candidate = Some(id);
                }
            }
            let Some(id) = candidate else {
                break;
            };

            let finish = now.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let free_at = finish.checked_add(turnaround).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Turnaround,
            })?;

            began[id] = true;
            worker_free[worker] = free_at;
            runs[id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            events.push(Event {
                time: now,
                task: id,
                worker,
                kind: EventKind::Start,
            });

            if finish == now {
                done[id] = true;
                count_done += 1;
                makespan = makespan.max(now);
                events.push(Event {
                    time: now,
                    task: id,
                    worker,
                    kind: EventKind::Finish,
                });
            } else {
                active.push((finish, id, worker));
            }
        }

        if count_done == n {
            break;
        }

        let mut next = None;
        let mut consider = |time: u64| {
            if time > now && next.map_or(true, |current| time < current) {
                next = Some(time);
            }
        };
        for &(finish, _, _) in &active {
            consider(finish);
        }
        for id in 0..n {
            if !began[id] {
                consider(tasks[id].release);
            }
        }
        for &free_at in &worker_free {
            consider(free_at);
        }

        if let Some(time) = next {
            now = time;
        } else {
            break;
        }
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();
    Ok(Schedule {
        runs,
        makespan,
        events,
    })
}
