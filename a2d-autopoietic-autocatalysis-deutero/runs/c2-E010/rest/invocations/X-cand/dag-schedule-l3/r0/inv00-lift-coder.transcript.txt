    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::HashSet::new();
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }
            if value == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if !seen.insert(value) {
                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
        }
    }

    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    let mut ready = std::collections::BTreeSet::new();
    for id in 0..n {
        if indegree[id] == 0 {
            ready.insert(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut remaining = indegree;
    while let Some(id) = ready.pop_first() {
        order.push(id);
        for &child in &children[id] {
            remaining[child] -= 1;
            if remaining[child] == 0 {
                ready.insert(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        for id in 0..n {
            if remaining[id] != 0 {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_child = children[id].iter().map(|&child| tails[child]).max().unwrap_or(0);
        tails[id] = tasks[id].duration.checked_add(max_child)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        worker: usize,
        finish: u64,
    }

    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut events = Vec::new();
    let mut completed = vec![false; n];
    let mut started = vec![false; n];
    let mut worker_available = vec![0u64; workers];
    let mut running: Vec<Running> = Vec::new();
    let mut now = 0u64;
    let mut done = 0usize;
    let mut makespan = 0u64;

    while done < n {
        let mut finishing: Vec<Running> = running.iter().copied().filter(|r| r.finish == now).collect();
        finishing.sort_by_key(|r| r.task);
        if !finishing.is_empty() {
            for r in &finishing {
                completed[r.task] = true;
                done += 1;
                makespan = makespan.max(r.finish);
                events.push(Event { time: now, task: r.task, worker: r.worker, kind: EventKind::Finish });
            }
            running.retain(|r| r.finish != now);
        }

        loop {
            let worker = (0..workers).find(|&w| worker_available[w] <= now);
            let Some(worker) = worker else { break };

            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if tasks[id].deps.iter().any(|&d| !completed[d]) {
                    continue;
                }
                let better = match best {
                    None => true,
                    Some(b) => tails[id] > tails[b]
                        || (tails[id] == tails[b] && tasks[id].priority > tasks[b].priority)
                        || (tails[id] == tails[b] && tasks[id].priority == tasks[b].priority && id < b),
                };
                if better {
                    best = Some(id);
                }
            }

            let Some(id) = best else { break };
            let finish = now.checked_add(tasks[id].duration)
                .ok_or(Error::Overflow { task: id, stage: Stage::Finish })?;
            let available = finish.checked_add(turnaround)
                .ok_or(Error::Overflow { task: id, stage: Stage::Turnaround })?;

            started[id] = true;
            worker_available[worker] = available;
            runs[id] = Some(Run { start: now, finish, worker });
            events.push(Event { time: now, task: id, worker, kind: EventKind::Start });

            if finish == now {
                completed[id] = true;
                done += 1;
                makespan = makespan.max(finish);
                events.push(Event { time: now, task: id, worker, kind: EventKind::Finish });
            } else {
                running.push(Running { task: id, worker, finish });
            }
        }

        if done == n {
            break;
        }

        let mut next = u64::MAX;
        let mut found = false;
        for r in &running {
            if r.finish > now && (!found || r.finish < next) {
                next = r.finish;
                found = true;
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now && (!found || tasks[id].release < next) {
                next = tasks[id].release;
                found = true;
            }
        }
        for &available in &worker_available {
            if available > now && (!found || available < next) {
                next = available;
                found = true;
            }
        }
        if found {
            now = next;
        } else {
            break;
        }
    }

    let runs = runs.into_iter().map(Option::unwrap).collect();
    Ok(Schedule { runs, makespan, events })
}

pub struct Task { pub duration: u64, pub deps: Vec<usize>, pub release: u64, pub priority: i32 }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run { pub start: u64, pub finish: u64, pub worker: usize }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EventKind { Start, Finish }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event { pub time: u64, pub task: usize, pub worker: usize, pub kind: EventKind }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule { pub runs: Vec<Run>, pub makespan: u64, pub events: Vec<Event> }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage { Tail, Finish, Turnaround }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency { task: usize, position: usize, value: usize },
    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },
    Cycle { blocked: Vec<usize> },
    Workers { count: usize },
    Overflow { task: usize, stage: Stage },
}

pub fn schedule(tasks: &[Task], workers: usize, turnaround: u64) -> Result<Schedule, Error> {
    use std::cmp::Reverse;
    use std::collections::{BTreeSet, BinaryHeap, HashSet};

    let n = tasks.len();
    for (t, task) in tasks.iter().enumerate() {
        let mut seen = HashSet::new();
        for (p, &d) in task.deps.iter().enumerate() {
            if d >= n {
                return Err(Error::BadDependency { task: t, position: p, value: d });
            } else if d == t {
                return Err(Error::SelfDependency { task: t, position: p });
            } else if !seen.insert(d) {
                return Err(Error::DuplicateDependency { task: t, position: p, value: d });
            }
        }
    }
    if workers == 0 || workers > 64 {
        return Err(Error::Workers { count: workers });
    }

    let mut indegree: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut successors = vec![Vec::new(); n];
    for (t, task) in tasks.iter().enumerate() {
        for &d in &task.deps {
            successors[d].push(t);
        }
    }
    let mut available = BinaryHeap::new();
    for t in 0..n {
        if indegree[t] == 0 { available.push(Reverse(t)); }
    }
    let mut topo = Vec::with_capacity(n);
    while let Some(Reverse(t)) = available.pop() {
        topo.push(t);
        for &s in &successors[t] {
            indegree[s] -= 1;
            if indegree[s] == 0 { available.push(Reverse(s)); }
        }
    }
    if topo.len() != n {
        let removed: HashSet<usize> = topo.iter().copied().collect();
        let blocked = (0..n).filter(|t| !removed.contains(t)).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut tail = vec![0u64; n];
    for &t in topo.iter().rev() {
        let longest = successors[t].iter().map(|&s| tail[s]).max().unwrap_or(0);
        tail[t] = tasks[t].duration.checked_add(longest)
            .ok_or(Error::Overflow { task: t, stage: Stage::Tail })?;
    }

    #[derive(Clone, Copy)]
    struct Active { task: usize, worker: usize, finish: u64 }

    let mut now = 0u64;
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut worker_free = vec![0u64; workers];
    let mut active = Vec::<Active>::new();
    let mut runs = vec![None; n];
    let mut events = Vec::new();
    let mut finished_count = 0usize;
    let mut makespan = 0u64;

    while finished_count < n {
        let mut at_time: Vec<Active> = active.iter().copied().filter(|a| a.finish == now).collect();
        at_time.sort_by_key(|a| a.task);
        for a in &at_time {
            completed[a.task] = true;
            finished_count += 1;
            makespan = makespan.max(now);
            events.push(Event { time: now, task: a.task, worker: a.worker, kind: EventKind::Finish });
        }
        active.retain(|a| a.finish != now);

        loop {
            let worker = match worker_free.iter().position(|&t| t <= now) {
                Some(w) => w,
                None => break,
            };
            let candidate = (0..n)
                .filter(|&t| !started[t] && tasks[t].release <= now
                    && tasks[t].deps.iter().all(|&d| completed[d]))
                .max_by(|&a, &b| {
                    tail[a].cmp(&tail[b])
                        .then_with(|| tasks[a].priority.cmp(&tasks[b].priority))
                        .then_with(|| b.cmp(&a))
                });
            let Some(t) = candidate else { break };

            let finish = now.checked_add(tasks[t].duration)
                .ok_or(Error::Overflow { task: t, stage: Stage::Finish })?;
            let free = finish.checked_add(turnaround)
                .ok_or(Error::Overflow { task: t, stage: Stage::Turnaround })?;
            started[t] = true;
            worker_free[worker] = free;
            runs[t] = Some(Run { start: now, finish, worker });
            events.push(Event { time: now, task: t, worker, kind: EventKind::Start });
            if finish == now {
                completed[t] = true;
                finished_count += 1;
                makespan = makespan.max(now);
                events.push(Event { time: now, task: t, worker, kind: EventKind::Finish });
            } else {
                active.push(Active { task: t, worker, finish });
            }
        }

        if finished_count == n { break; }

        let next_finish = active.iter().filter_map(|a| (a.finish > now).then_some(a.finish)).min();
        let next_release = (0..n).filter_map(|t| {
            (!started[t] && tasks[t].release > now).then_some(tasks[t].release)
        }).min();
        let next_worker = worker_free.iter().filter_map(|&t| (t > now).then_some(t)).min();
        let next = [next_finish, next_release, next_worker].into_iter().flatten().min();
        match next {
            Some(t) => now = t,
            None => break,
        }
    }

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        events,
    })
}
