- NUL validation takes precedence over every other path error; offsets are UTF-8 byte offsets.
- Empty interior components point to their terminating slash; trailing empty components point to the final slash.
- Canonical paths are unchanged because normalization is forbidden; `/` is the only single-component exception.
- Parent traversal errors precede final-entry checks; creation requires the immediate parent to exist and be a directory.
- Directories at a file-operation target produce `IsDir` for read/write and `NotDir` for `ls`; `mkdir` reports `Exists`.
- Names sort by Rust `String` ordering, and empty file data is valid.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn components(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut result = Vec::new();
        let mut start = 1;

        for i in 1..bytes.len() {
            if bytes[i] == b'/' {
                if i == start {
                    return Err(Error::InvalidPath(i));
                }
                let component = &path[start..i];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                result.push(component);
                start = i + 1;
            }
        }

        if start == bytes.len() {
            return Err(Error::InvalidPath(bytes.len() - 1));
        }

        let component = &path[start..];
        if component == "." || component == ".." {
            return Err(Error::InvalidPath(start));
        }
        result.push(component);
        Ok(result)
    }

    fn canonical_prefix(parts: &[&str]) -> String {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            let mut path = String::new();
            for part in parts {
                path.push('/');
                path.push_str(part);
            }
            path
        }
    }

    fn find<'a>(root: &'a Entry, parts: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = root;
        let mut prefix = Vec::new();

        for part in parts {
            match current {
                Entry::Dir(children) => {
                    prefix.push(*part);
                    current = children
                        .get(*part)
                        .ok_or_else(|| Error::Missing(Self::canonical_prefix(&prefix)))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                }
            }
        }

        Ok(current)
    }

    fn find_parent_mut<'a>(
        root: &'a mut Entry,
        parts: &[&str],
    ) -> Result<(&'a mut BTreeMap<String, Entry>, String), Error> {
        let parent_parts = &parts[..parts.len() - 1];
        let mut current = root;
        let mut prefix = Vec::new();

        for part in parent_parts {
            match current {
                Entry::Dir(children) => {
                    prefix.push(*part);
                    current = children
                        .get_mut(*part)
                        .ok_or_else(|| Error::Missing(Self::canonical_prefix(&prefix)))?;
                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                }
            }
        }

        match current {
            Entry::Dir(children) => Ok((children, Self::canonical_prefix(parts))),
            Entry::File(_) => Err(Error::NotDir(Self::canonical_prefix(parent_parts))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = Self::components(path)?;
        if parts.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let (parent, canonical) = Self::find_parent_mut(&mut self.root, &parts)?;
        let name = parts[parts.len() - 1];
        if parent.contains_key(name) {
            return Err(Error::Exists(canonical));
        }

        parent.insert(name.to_owned(), Entry::Dir(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = Self::components(path)?;
        if parts.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let (parent, canonical) = Self::find_parent_mut(&mut self.root, &parts)?;
        let name = parts[parts.len() - 1];
        match parent.get_mut(name) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name.to_owned(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = Self::components(path)?;
        match Self::find(&self.root, &parts)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical_prefix(&parts))),
            Entry::File(contents) => Ok(contents.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = Self::components(path)?;
        match Self::find(&self.root, &parts)? {
            Entry::File(_) => Err(Error::NotDir(Self::canonical_prefix(&parts))),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}
```

### Solution B

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

enum Node {
    Directory(BTreeMap<String, Node>),
    RegularFile(Vec<u8>),
}

pub struct Fs {
    root: Node,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Node::Directory(BTreeMap::new()),
        }
    }

    fn parse(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(offset) = path.as_bytes().iter().position(|byte| *byte == 0) {
            return Err(Error::InvalidPath(offset));
        }
        if path.is_empty() || path.as_bytes()[0] != b'/' {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut components = Vec::new();
        let mut component_start = 1usize;
        for (offset, byte) in path.bytes().enumerate().skip(1) {
            if byte == b'/' {
                if offset == component_start {
                    return Err(Error::InvalidPath(offset));
                }
                let component = &path[component_start..offset];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(component_start));
                }
                components.push(component);
                component_start = offset + 1;
            }
        }

        if component_start == path.len() {
            return Err(Error::InvalidPath(path.len() - 1));
        }
        let final_component = &path[component_start..];
        if final_component == "." || final_component == ".." {
            return Err(Error::InvalidPath(component_start));
        }
        components.push(final_component);
        Ok(components)
    }

    fn path_for(components: &[&str]) -> String {
        let mut result = String::new();
        if components.is_empty() {
            result.push('/');
        } else {
            for component in components {
                result.push('/');
                result.push_str(component);
            }
        }
        result
    }

    fn lookup<'a>(root: &'a Node, components: &[&str]) -> Result<&'a Node, Error> {
        let mut node = root;
        let mut prefix = Vec::new();

        for component in components {
            match node {
                Node::RegularFile(_) => {
                    return Err(Error::NotDir(Self::path_for(&prefix)));
                }
                Node::Directory(children) => {
                    prefix.push(*component);
                    node = match children.get(*component) {
                        Some(found) => found,
                        None => return Err(Error::Missing(Self::path_for(&prefix))),
                    };
                }
            }
        }
        Ok(node)
    }

    fn parent_mut<'a>(
        root: &'a mut Node,
        components: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Node>, Error> {
        let parents = &components[..components.len() - 1];
        let mut node = root;
        let mut prefix = Vec::new();

        for component in parents {
            match node {
                Node::RegularFile(_) => {
                    return Err(Error::NotDir(Self::path_for(&prefix)));
                }
                Node::Directory(children) => {
                    prefix.push(*component);
                    node = match children.get_mut(*component) {
                        Some(found) => found,
                        None => return Err(Error::Missing(Self::path_for(&prefix))),
                    };
                }
            }
        }

        match node {
            Node::Directory(children) => Ok(children),
            Node::RegularFile(_) => Err(Error::NotDir(Self::path_for(parents))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::parse(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let parent = Self::parent_mut(&mut self.root, &components)?;
        let name = components[components.len() - 1];
        if parent.contains_key(name) {
            return Err(Error::Exists(Self::path_for(&components)));
        }
        parent.insert(name.to_owned(), Node::Directory(BTreeMap::new()));
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::parse(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let parent = Self::parent_mut(&mut self.root, &components)?;
        let name = components[components.len() - 1];
        match parent.get_mut(name) {
            Some(Node::Directory(_)) => Err(Error::IsDir(Self::path_for(&components))),
            Some(Node::RegularFile(contents)) => {
                *contents = data.to_vec();
                Ok(())
            }
            None => {
                parent.insert(name.to_owned(), Node::RegularFile(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::parse(path)?;
        match Self::lookup(&self.root, &components)? {
            Node::Directory(_) => Err(Error::IsDir(Self::path_for(&components))),
            Node::RegularFile(bytes) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::parse(path)?;
        match Self::lookup(&self.root, &components)? {
            Node::RegularFile(_) => Err(Error::NotDir(Self::path_for(&components))),
            Node::Directory(children) => Ok(children
                .iter()
                .map(|(name, node)| {
                    let kind = match node {
                        Node::Directory(_) => Kind::Dir,
                        Node::RegularFile(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}
```
