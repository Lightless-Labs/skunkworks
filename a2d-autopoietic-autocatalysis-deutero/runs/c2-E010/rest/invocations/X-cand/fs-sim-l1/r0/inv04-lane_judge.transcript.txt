    File,

    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),

    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

}

        }

        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut result = Vec::new();

                    return Err(Error::InvalidPath(i));
                }

                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                result.push(component);
                start = i + 1;
            }
        }

        if start == bytes.len() {
            return Err(Error::InvalidPath(bytes.len() - 1));
        }

        let component = &path[start..];
        if component == "." || component == ".." {
            return Err(Error::InvalidPath(start));
        }
        result.push(component);
        Ok(result)
    }

                path.push_str(part);
            }

                }
                Entry::File(_) => {
                    return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                }

                    return Err(Error::NotDir(Self::canonical_prefix(&prefix)));
                }
            }
        }

            Entry::Dir(children) => Ok((children, Self::canonical_prefix(parts))),
            Entry::File(_) => Err(Error::NotDir(Self::canonical_prefix(parent_parts))),
        }
    }

            Some(Entry::File(contents)) => {

                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(name.to_owned(), Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

        match Self::find(&self.root, &parts)? {
            Entry::Dir(_) => Err(Error::IsDir(Self::canonical_prefix(&parts))),
            Entry::File(contents) => Ok(contents.clone()),

                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }

    File(Vec<u8>),

        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self { entries }

            return Err(Error::InvalidPath(i));
        }

            return Err(Error::InvalidPath(0));
        }

            return Ok(Vec::new());
        }

        let root = self.entries.get("/").expect("root entry invariant");
        let mut current = root;

        for component in components {
            if !matches!(current, Entry::Dir) {
                return Err(Error::NotDir(Self::path_for(&prefix)));
            }
            prefix.push(component.clone());
            current = self
                .entries
                .get(&Self::path_for(&prefix))
                .ok_or_else(|| Error::Missing(Self::path_for(&prefix)))?;
        }
        Ok(current)
    }

    fn parent(&self, components: &[String]) -> Result<String, Error> {
        let parent_components = &components[..components.len() - 1];
        match self.lookup(parent_components)? {
            Entry::Dir => Ok(Self::path_for(parent_components)),
            Entry::File(_) => Err(Error::NotDir(Self::path_for(parent_components))),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let components = Self::canonicalize(path)?;
        if components.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let parent = self.parent(&components)?;
        let canonical = Self::path_for(&components);
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }

        let _ = parent;
        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let components = Self::canonicalize(path)?;
        if components.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        self.parent(&components)?;
        let canonical = Self::path_for(&components);
        if matches!(self.entries.get(&canonical), Some(Entry::Dir)) {
            return Err(Error::IsDir(canonical));
        }

        self.entries.insert(canonical, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let components = Self::canonicalize(path)?;
        match self.lookup(&components)? {
            Entry::File(data) => Ok(data.clone()),
            Entry::Dir => Err(Error::IsDir(Self::path_for(&components))),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let components = Self::canonicalize(path)?;
        match self.lookup(&components)? {
            Entry::File(_) => return Err(Error::NotDir(Self::path_for(&components))),
            Entry::Dir => {}
        }

        let parent = Self::path_for(&components);
        let mut result = Vec::new();
        for (candidate, entry) in &self.entries {
            if candidate == "/" {
                continue;
            }
            let Some((candidate_parent, basename)) = candidate.rsplit_once('/') else {
                continue;
            };
            let candidate_parent = if candidate_parent.is_empty() {
                "/"
            } else {
                candidate_parent
            };
            if candidate_parent == parent {
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                result.push((basename.to_owned(), kind));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }
}

    fn default() -> Self {
        Self::new()
    }

                    return Err(Error::InvalidPath(offset));

                component_start = offset + 1;

        match self.lookup(&parts)? {
            Node::File(data) => Ok(data.clone()),
            Node::Dir(_) => Err(Error::IsDir(Self::path_for(&parts))),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = Self::canonicalize(path)?;
        let node = self.lookup(&parts)?;
        let Node::Dir(children) = node else {
            return Err(Error::NotDir(Self::path_for(&parts)));
        };

        Ok(children
            .iter()
            .map(|(name, node)| {
                let kind = match node {
                    Node::Dir(_) => Kind::Dir,
                    Node::File(_) => Kind::File,
                };
                (name.clone(), kind)
            })
            .collect())
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
