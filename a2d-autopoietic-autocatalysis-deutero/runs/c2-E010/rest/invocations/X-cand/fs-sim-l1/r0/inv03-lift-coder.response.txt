- NUL detection takes precedence over all other path errors; offsets are UTF-8 byte offsets.
- Empty components report the terminating slash, including the final slash for trailing empties.
- Parent traversal errors precede final-entry checks; errors name canonical absolute prefixes.
- `write` creates only a missing final entry; all other creation requires an existing directory parent.
- Root has special final-entry behavior: `mkdir` returns `Exists("/")`, while `read` and `write` return `IsDir("/")`.
- Both versions use owned results and sorted immediate children; Solution B uses a flat map instead of a tree.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn parse(path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut parts = Vec::new();
        let mut start = 1;

        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                if i == start {
                    return Err(Error::InvalidPath(i));
                }
                let component = &path[start..i];
                if component == "." || component == ".." {
                    return Err(Error::InvalidPath(start));
                }
                parts.push(component.to_owned());
                start = i + 1;
            }
        }
        Ok(parts)
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            return "/".to_owned();
        }
        let mut result = String::new();
        for part in parts {
            result.push('/');
            result.push_str(part);
        }
        result
    }

    fn find<'a>(root: &'a Entry, parts: &[String]) -> Option<&'a Entry> {
        let mut current = root;
        for part in parts {
            match current {
                Entry::Dir(children) => current = children.get(part)?,
                Entry::File(_) => return None,
            }
        }
        Some(current)
    }

    fn parent<'a>(&'a self, parts: &[String]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        let mut prefix = Vec::new();
        for part in &parts[..parts.len() - 1] {
            if let Entry::File(_) = current {
                return Err(Error::NotDir(Self::canonical(&prefix)));
            }
            prefix.push(part.clone());
            match current {
                Entry::Dir(children) => match children.get(part) {
                    Some(entry) => current = entry,
                    None => return Err(Error::Missing(Self::canonical(&prefix))),
                },
                Entry::File(_) => unreachable!(),
            }
        }
        Ok(current)
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = Self::parse(path)?;
        if parts.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let parent = self.parent(&parts)?;
        let parent_path = Self::canonical(&parts[..parts.len() - 1]);
        let final_path = Self::canonical(&parts);

        match parent {
            Entry::File(_) => return Err(Error::NotDir(parent_path)),
            Entry::Dir(children) if children.contains_key(parts.last().unwrap()) => {
                return Err(Error::Exists(final_path));
            }
            Entry::Dir(_) => {}
        }

        let mut current = &mut self.root;
        for part in &parts[..parts.len() - 1] {
            match current {
                Entry::Dir(children) => current = children.get_mut(part).unwrap(),
                Entry::File(_) => unreachable!(),
            }
        }
        match current {
            Entry::Dir(children) => {
                children.insert(parts.last().unwrap().clone(), Entry::Dir(BTreeMap::new()));
                Ok(())
            }
            Entry::File(_) => unreachable!(),
        }
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = Self::parse(path)?;
        if parts.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let parent = self.parent(&parts)?;
        let parent_path = Self::canonical(&parts[..parts.len() - 1]);
        let final_path = Self::canonical(&parts);

        match parent {
            Entry::File(_) => return Err(Error::NotDir(parent_path)),
            Entry::Dir(children) => {
                if matches!(children.get(parts.last().unwrap()), Some(Entry::Dir(_))) {
                    return Err(Error::IsDir(final_path));
                }
            }
        }

        let mut current = &mut self.root;
        for part in &parts[..parts.len() - 1] {
            match current {
                Entry::Dir(children) => current = children.get_mut(part).unwrap(),
                Entry::File(_) => unreachable!(),
            }
        }
        match current {
            Entry::Dir(children) => {
                children.insert(parts.last().unwrap().clone(), Entry::File(data.to_vec()));
                Ok(())
            }
            Entry::File(_) => unreachable!(),
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = Self::parse(path)?;
        let path = Self::canonical(&parts);
        match Self::find(&self.root, &parts) {
            None => Err(Error::Missing(path)),
            Some(Entry::Dir(_)) => Err(Error::IsDir(path)),
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = Self::parse(path)?;
        let path = Self::canonical(&parts);
        match Self::find(&self.root, &parts) {
            None => Err(Error::Missing(path)),
            Some(Entry::File(_)) => Err(Error::NotDir(path)),
            Some(Entry::Dir(children)) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```

### Solution B

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Node {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Node>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Node::Dir);
        Self { entries }
    }

    fn parse(path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let bytes = path.as_bytes();
        let mut result = Vec::new();
        let mut start = 1;
        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                if start == i {
                    return Err(Error::InvalidPath(i));
                }
                let name = &path[start..i];
                if name == "." || name == ".." {
                    return Err(Error::InvalidPath(start));
                }
                result.push(name.to_owned());
                start = i + 1;
            }
        }
        Ok(result)
    }

    fn path(parts: &[String]) -> String {
        if parts.is_empty() {
            return "/".to_owned();
        }
        format!("/{}", parts.join("/"))
    }

    fn parent_path(parts: &[String]) -> String {
        Self::path(&parts[..parts.len() - 1])
    }

    fn check_parent(&self, parts: &[String]) -> Result<String, Error> {
        let mut prefix = Vec::new();
        let mut current = "/".to_owned();

        for part in &parts[..parts.len() - 1] {
            match self.entries.get(&current) {
                Some(Node::Dir) => {}
                Some(Node::File(_)) => return Err(Error::NotDir(current)),
                None => return Err(Error::Missing(current)),
            }
            prefix.push(part.clone());
            current = Self::path(&prefix);
            if !self.entries.contains_key(&current) {
                return Err(Error::Missing(current));
            }
        }

        match self.entries.get(&current) {
            Some(Node::Dir) => Ok(current),
            Some(Node::File(_)) => Err(Error::NotDir(current)),
            None => Err(Error::Missing(current)),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = Self::parse(path)?;
        if parts.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }
        let parent = self.check_parent(&parts)?;
        let full = Self::path(&parts);
        if self.entries.contains_key(&full) {
            return Err(Error::Exists(full));
        }
        self.entries.insert(full, Node::Dir);
        let _ = parent;
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = Self::parse(path)?;
        if parts.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }
        self.check_parent(&parts)?;
        let full = Self::path(&parts);
        match self.entries.get(&full) {
            Some(Node::Dir) => return Err(Error::IsDir(full)),
            Some(Node::File(_)) | None => {}
        }
        self.entries.insert(full, Node::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = Self::parse(path)?;
        let full = Self::path(&parts);
        match self.entries.get(&full) {
            Some(Node::File(bytes)) => Ok(bytes.clone()),
            Some(Node::Dir) => Err(Error::IsDir(full)),
            None => {
                let mut prefix = Vec::new();
                for part in &parts {
                    prefix.push(part.clone());
                    let current = Self::path(&prefix);
                    match self.entries.get(&current) {
                        Some(Node::File(_)) => return Err(Error::NotDir(current)),
                        Some(Node::Dir) => {}
                        None => return Err(Error::Missing(current)),
                    }
                }
                Err(Error::Missing(full))
            }
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = Self::parse(path)?;
        let full = Self::path(&parts);

        match self.entries.get(&full) {
            Some(Node::File(_)) => return Err(Error::NotDir(full)),
            Some(Node::Dir) => {}
            None => {
                let mut prefix = Vec::new();
                for part in &parts {
                    prefix.push(part.clone());
                    let current = Self::path(&prefix);
                    match self.entries.get(&current) {
                        Some(Node::File(_)) => return Err(Error::NotDir(current)),
                        Some(Node::Dir) => {}
                        Some(_) => {}
                        None => return Err(Error::Missing(current)),
                    }
                }
                return Err(Error::Missing(full));
            }
        }

        let prefix = if full == "/" {
            "/".to_owned()
        } else {
            format!("{}/", full)
        };
        let mut children = Vec::new();

        for (candidate, node) in &self.entries {
            if let Some(rest) = candidate.strip_prefix(&prefix) {
                if !rest.is_empty() && !rest.contains('/') {
                    let kind = match node {
                        Node::Dir => Kind::Dir,
                        Node::File(_) => Kind::File,
                    };
                    children.push((rest.to_owned(), kind));
                }
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
