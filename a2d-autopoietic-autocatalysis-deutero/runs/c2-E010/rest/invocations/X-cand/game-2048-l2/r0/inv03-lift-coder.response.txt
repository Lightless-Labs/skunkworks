- Validate dimensions and cell count first, then scan every tile in flat-index order before attempting any move.
- Process lines by increasing row or column, even for Right and Down; scan each line from its destination edge.
- Ignore zeroes when pairing tiles; each input tile participates in at most one merge.
- Report overflow for the first overflowing pair in processing order, with original indices in scan order.
- Treat all-zero and one-cell boards as valid; `legal_moves` skips overflowing directions and includes only changed moves.
- Both solutions use checked shape arithmetic and support dimensions up to 8×8.

### Solution A

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction { Up, Left, Down, Right }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }
    let size = match board.width.checked_mul(board.height) {
        Some(n) => n,
        None => return Err(Error::Shape),
    };
    if board.cells.len() != size {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn line_indices(board: &Board, direction: Direction, line: usize) -> Vec<usize> {
    match direction {
        Direction::Left => (0..board.width).map(|x| line * board.width + x).collect(),
        Direction::Right => (0..board.width).rev().map(|x| line * board.width + x).collect(),
        Direction::Up => (0..board.height).map(|y| y * board.width + line).collect(),
        Direction::Down => (0..board.height).rev().map(|y| y * board.width + line).collect(),
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line in 0..line_count {
        let indices = line_indices(board, direction, line);
        let mut inputs: Vec<(u32, usize)> = indices
            .into_iter()
            .filter_map(|i| if board.cells[i] == 0 { None } else { Some((board.cells[i], i)) })
            .collect();

        let mut outputs: Vec<(u32, Option<[usize; 2]>)> = Vec::new();
        let mut i = 0;
        while i < inputs.len() {
            if i + 1 < inputs.len() && inputs[i].0 == inputs[i + 1].0 {
                let value = inputs[i].0;
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: inputs[i].1,
                        second: inputs[i + 1].1,
                    });
                }
                let doubled = value * 2;
                outputs.push((doubled, Some([inputs[i].1, inputs[i + 1].1])));
                score += u64::from(doubled);
                i += 2;
            } else {
                outputs.push((inputs[i].0, None));
                i += 1;
            }
        }

        for (position, (value, pair)) in outputs.into_iter().enumerate() {
            let to = indices[position];
            cells[to] = value;
            if let Some(from) = pair {
                merges.push(Merge { from, to, value });
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board { width: board.width, height: board.height, cells },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut result = Vec::new();
    for direction in [Direction::Up, Direction::Left, Direction::Down, Direction::Right] {
        match move_board(board, direction) {
            Ok(moved) if moved.changed => result.push(direction),
            Ok(_) | Err(Error::MergeOverflow { .. }) => {}
            Err(error) => return Err(error),
        }
    }
    Ok(result)
}
```

### Solution B

```rust
#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum Direction { Up, Left, Down, Right }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Shape,
    InvalidTile { index: usize, value: u32 },
    MergeOverflow { first: usize, second: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Board {
    pub width: usize,
    pub height: usize,
    pub cells: Vec<u32>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Merge {
    pub from: [usize; 2],
    pub to: usize,
    pub value: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct MoveResult {
    pub board: Board,
    pub score: u64,
    pub changed: bool,
    pub merges: Vec<Merge>,
}

fn validate(board: &Board) -> Result<(), Error> {
    if board.width == 0 || board.width > 8 || board.height == 0 || board.height > 8 {
        return Err(Error::Shape);
    }
    let expected = board.width * board.height;
    if board.cells.len() != expected {
        return Err(Error::Shape);
    }
    for index in 0..expected {
        let value = board.cells[index];
        if value != 0 && (value < 2 || value > 2_147_483_648 || (value & (value - 1)) != 0) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn line_count(board: &Board, direction: Direction) -> usize {
    match direction {
        Direction::Up | Direction::Down => board.width,
        Direction::Left | Direction::Right => board.height,
    }
}

fn line_length(board: &Board, direction: Direction) -> usize {
    match direction {
        Direction::Up | Direction::Down => board.height,
        Direction::Left | Direction::Right => board.width,
    }
}

fn flat_index(board: &Board, direction: Direction, line: usize, offset: usize) -> usize {
    match direction {
        Direction::Left => line * board.width + offset,
        Direction::Right => line * board.width + (board.width - 1 - offset),
        Direction::Up => offset * board.width + line,
        Direction::Down => (board.height - 1 - offset) * board.width + line,
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut result_cells = vec![0u32; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    for line in 0..line_count(board, direction) {
        let len = line_length(board, direction);
        let mut values = Vec::with_capacity(len);
        for offset in 0..len {
            let index = flat_index(board, direction, line, offset);
            let value = board.cells[index];
            if value != 0 {
                values.push((value, index));
            }
        }

        let mut out = 0usize;
        let mut cursor = 0usize;
        while cursor < values.len() {
            let (value, original) = values[cursor];
            let destination = flat_index(board, direction, line, out);
            if cursor + 1 < values.len() && values[cursor + 1].0 == value {
                let second = values[cursor + 1].1;
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow { first: original, second });
                }
                let merged_value = value * 2;
                result_cells[destination] = merged_value;
                merges.push(Merge { from: [original, second], to: destination, value: merged_value });
                score += merged_value as u64;
                cursor += 2;
            } else {
                result_cells[destination] = value;
                cursor += 1;
            }
            out += 1;
        }
    }

    let changed = result_cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells: result_cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut legal = Vec::with_capacity(4);
    for direction in [Direction::Up, Direction::Left, Direction::Down, Direction::Right] {
        if let Ok(movement) = move_board(board, direction) {
            if movement.changed {
                legal.push(direction);
            }
        }
    }
    Ok(legal)
}
```
