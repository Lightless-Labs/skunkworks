fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width) || !(1..=8).contains(&board.height) {
        return Err(Error::Shape);
    }
    let expected = match board.width.checked_mul(board.height) {

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {

    let mut cells = vec![0u32; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let horizontal = matches!(direction, Direction::Left | Direction::Right);
    let line_count = if horizontal { board.height } else { board.width };
    let line_len = if horizontal { board.width } else { board.height };
    let reverse = matches!(direction, Direction::Right | Direction::Down);

    for line in 0..line_count {
        let mut tiles: Vec<(usize, u32)> = Vec::new();
        for offset in 0..line_len {
            let position = if reverse { line_len - 1 - offset } else { offset };
            let index = if horizontal {
                line * board.width + position
            } else {
                position * board.width + line
            };
            let value = board.cells[index];
            if value != 0 {
                tiles.push((index, value));
            }

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut output = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;
    let rows = matches!(direction, Direction::Left | Direction::Right);
    let reversed = matches!(direction, Direction::Right | Direction::Down);
    let count = if rows { board.height } else { board.width };
    let length = if rows { board.width } else { board.height };

    for line in 0..count {
        let index_at = |step: usize| {
            let position = if reversed { length - 1 - step } else { step };
            if rows {
                line * board.width + position
            } else {
                position * board.width + line
            }
        };

        let mut input = Vec::new();
        for step in 0..length {
            let index = index_at(step);
            if board.cells[index] != 0 {
                input.push((index, board.cells[index]));
            }
        }

        let mut results = Vec::new();
        let mut cursor = 0;
        while cursor < input.len() {
            if cursor + 1 < input.len() && input[cursor].1 == input[cursor + 1].1 {
                let value = input[cursor].1;
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: input[cursor].0,
                        second: input[cursor + 1].0,
                    });
                }
                let merged = value * 2;
                results.push((merged, Some([input[cursor].0, input[cursor + 1].0])));
                score += u64::from(merged);
                cursor += 2;
            } else {
                results.push((input[cursor].1, None));
                cursor += 1;

                    from,
                    to: destination,
                    value,

            }
        }

        board: Board {
            width: board.width,
            height: board.height,
            cells: output,
        },
        score,
        changed,
        merges,
    })
}
