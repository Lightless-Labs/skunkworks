- Checks follow the explicitly stated precedence, including snapshot validation before tail replay and ID validation before Store lookups.
- Empty appends validate ID and expectation but create no stream or record; empty projection ingests are no-ops.
- IDs are validated as bytes, with the 24-byte length error reported only if no earlier invalid byte exists.
- Snapshot structure permits any otherwise-valid state, including unreachable Open states; tags use Rust’s specified wrapping and cast behavior.
- Versions and global positions use checked progression; stored history and projection journals retain every committed record.
- Both implementations use standard-library ordered maps; Solution B uses a separate replay-and-rebuild approach for Store state.

fn apply_event(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    if account.version == u64::MAX {
        return Err(ApplyError::VersionOverflow);
    }

    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.status = Status::Open;
            account.limit = *limit;
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.status = Status::Closed;
        }
    }

    account.version += 1;
    Ok(())
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    let mut account = initial_account();
    for (index, event) in events.iter().enumerate() {
        if let Err(kind) = apply_event(&mut account, event) {
            return Err(ReplayError { index, kind });
        }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {

    records: Vec<Record>,

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 {
            return Err(SnapshotError::ZeroInterval);
        }
        Ok(Self { snapshot_every, streams: BTreeMap::new(), records: Vec::new() })
    }

    pub fn append(&mut self, id: &str, expected: Expected, events: &[Event]) -> Result<Account, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;

        let existing = self.streams.get(id);
        let actual = existing.map(|s| s.account.version);
        let matches = match &expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(v) => actual == Some(*v),
        };
        if !matches {
            return Err(StoreError::Conflict { expected, actual });
        }

        let mut account = existing.map(|s| s.account.clone()).unwrap_or_else(initial_account);
        if events.is_empty() {
            return Ok(account);
        }

        let mut accepted = Vec::with_capacity(events.len());
        let mut new_snapshot = existing.and_then(|s| s.snapshot.clone());
        let mut latest_snapshot_version = new_snapshot.as_ref().map(|s| s.account.version).unwrap_or(0);

        for (index, event) in events.iter().enumerate() {
            if let Err(kind) = apply_event(&mut account, event) {
                return Err(StoreError::Rejected(ReplayError { index, kind }));
            }
            accepted.push(event.clone());
            if account.version % self.snapshot_every == 0 {
                new_snapshot = Some(Snapshot { tag: snapshot_tag(&account), account: account.clone() });
                latest_snapshot_version = account.version;
            }
        }

        let mut new_records = Vec::with_capacity(events.len());
        let start_position = self.records.len() as u64;
        for (offset, event) in accepted.iter().enumerate() {
            let position = start_position + offset as u64 + 1;
            let version = existing.map(|s| s.account.version).unwrap_or(0) + offset as u64 + 1;
            new_records.push(Record { position, stream: id.to_owned(), version, event: event.clone() });
        }

        let stream = self.streams.entry(id.to_owned()).or_insert_with(|| Stream {
            account: initial_account(),
            events: Vec::new(),
            snapshot: None,
        });
        stream.account = account.clone();
        stream.events.extend(accepted);
        stream.snapshot = new_snapshot;
        let _ = latest_snapshot_version;
        self.records.extend(new_records);
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|s| s.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|s| {
            s.events.iter().enumerate()
                .filter(|(i, _)| (*i as u64) + 1 > after)
                .map(|(_, e)| e.clone())
                .collect()
        }))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).and_then(|s| s.snapshot.clone()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|s| {
            let version = s.snapshot.as_ref().map(|v| v.account.version).unwrap_or(0);
            s.events.iter().skip(version as usize).cloned().collect()
        }))
    }

    pub fn records(&self, after: u64) -> Vec<Record> {
        self.records.iter().filter(|r| r.position > after).cloned().collect()
    }
}

pub struct Totals { pub streams: u64, pub open: u64, pub closed: u64, pub balance: i128 }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ProjectionError {
    InvalidId { index: usize, byte: usize }, ZeroPosition { index: usize },
    DuplicateMismatch { index: usize, position: u64 },
    Gap { index: usize, expected: u64, got: u64 },
    StreamVersion { index: usize, expected: u64, got: u64 },
    Rejected { index: usize, kind: ApplyError }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Projection {
    position: u64,
    journal: BTreeMap<u64, Record>,
    accounts: BTreeMap<String, Account>,
}

impl Projection {
    pub fn new() -> Self {
        Self { position: 0, journal: BTreeMap::new(), accounts: BTreeMap::new() }
    }

    pub fn ingest(&mut self, records: &[Record]) -> Result<usize, ProjectionError> {
        let mut provisional = self.clone();
        let mut accepted = 0;

        for (index, record) in records.iter().enumerate() {
            if let Err(byte) = validate_id(&record.stream) {
                return Err(ProjectionError::InvalidId { index, byte });
            }
            if record.position == 0 {
                return Err(ProjectionError::ZeroPosition { index });
            }

            if record.position <= provisional.position {
                if provisional.journal.get(&record.position) != Some(record) {
                    return Err(ProjectionError::DuplicateMismatch { index, position: record.position });
                }
                continue;
            }

            let expected_position = provisional.position + 1;
            if record.position != expected_position {
                return Err(ProjectionError::Gap {
                    index,
                    expected: expected_position,
                    got: record.position,
                });
            }

            let current_version = provisional.accounts.get(&record.stream)
                .map(|a| a.version).unwrap_or(0);
            let expected_version = current_version + 1;
            if record.version != expected_version {
                return Err(ProjectionError::StreamVersion {
                    index,
                    expected: expected_version,
                    got: record.version,
                });
            }

            let mut account = provisional.accounts.get(&record.stream)
                .cloned().unwrap_or_else(initial_account);
            if let Err(kind) = apply_event(&mut account, &record.event) {
                return Err(ProjectionError::Rejected { index, kind });
            }

            provisional.accounts.insert(record.stream.clone(), account);
            provisional.journal.insert(record.position, record.clone());
            provisional.position = record.position;
            accepted += 1;
        }

        *self = provisional;
        Ok(accepted)
    }

    pub fn account(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.accounts.get(id).cloned())
    }

    pub fn position(&self) -> u64 { self.position }

    pub fn totals(&self) -> Totals {
        let mut totals = Totals { streams: self.accounts.len() as u64, open: 0, closed: 0, balance: 0 };
        for account in self.accounts.values() {
            match account.status {
                Status::Open => totals.open += 1,
                Status::Closed => totals.closed += 1,
                Status::New => {}
            }
            totals.balance += account.balance as i128;
        }
        totals
    }
}

        let old = self.data.get(id);

            Expected::Missing => actual.is_none(),
            Expected::Exact(v) => actual == Some(*v),

            stream: id.to_owned(),

            account: initial_account(),
            events: Vec::new(),
            snapshot: None,

        entry.account = new_account.clone();

impl Projection {
    pub fn new() -> Self {
        Self { cursor: 0, journal: Vec::new(), accounts: BTreeMap::new() }
    }

    pub fn ingest(&mut self, records: &[Record]) -> Result<usize, ProjectionError> {
        let mut work = self.clone();
        let mut added = 0usize;
        for (index, record) in records.iter().enumerate() {
            if let Err(byte) = valid_id(&record.stream) {
                return Err(ProjectionError::InvalidId { index, byte });
            }
            if record.position == 0 {
                return Err(ProjectionError::ZeroPosition { index });
            }
            if record.position <= work.cursor {
                let previous = &work.journal[(record.position - 1) as usize];
                if previous != record {
                    return Err(ProjectionError::DuplicateMismatch { index, position: record.position });
                }
                continue;
            }
            let expected_position = work.cursor + 1;
            if record.position != expected_position {
                return Err(ProjectionError::Gap { index, expected: expected_position, got: record.position });
            }

            let current_version = work.accounts.get(&record.stream).map(|a| a.version).unwrap_or(0);
            let expected_version = current_version + 1;
            if record.version != expected_version {
                return Err(ProjectionError::StreamVersion { index, expected: expected_version, got: record.version });
            }
            let mut account = work.accounts.get(&record.stream).cloned().unwrap_or_else(initial_account);
            if let Err(kind) = apply_event(&mut account, &record.event) {
                return Err(ProjectionError::Rejected { index, kind });
            }
            work.accounts.insert(record.stream.clone(), account);
            work.journal.push(record.clone());
            work.cursor = record.position;
            added += 1;
        }
        *self = work;
        Ok(added)
    }

    pub fn account(&self, id: &str) -> Result<Option<Account>, StoreError> {
        valid_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.accounts.get(id).cloned())
    }

    pub fn position(&self) -> u64 { self.cursor }

    pub fn totals(&self) -> Totals {
        let mut t = Totals { streams: self.accounts.len() as u64, open: 0, closed: 0, balance: 0 };
        for a in self.accounts.values() {
            match a.status { Status::Open => t.open += 1, Status::Closed => t.closed += 1, Status::New => {} }
            t.balance += a.balance as i128;
        }
        t
    }
}
