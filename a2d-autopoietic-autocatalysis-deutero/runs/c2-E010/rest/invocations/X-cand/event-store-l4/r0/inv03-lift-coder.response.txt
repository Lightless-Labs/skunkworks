- Error precedence follows the numbered check order, including validation before lookup and version overflow before event checks during rebuild.
- Invalid IDs scan bytes first; length overflow is reported at byte 24 unless an earlier byte is invalid.
- Empty appends still check ID and expectation but never create streams or consume positions.
- Snapshots capture every reached positive interval boundary, including an intermediate state inside a batch.
- Projection duplicates are checked against the complete journal and exact matches skip version and event validation.
- Both versions use the same interpretations; I’m confident in the contract and provide independent implementations.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Status { New, Open, Closed }
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Account { pub status: Status, pub balance: i64, pub limit: i64, pub version: u64 }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { limit: i64 }, Deposit { amount: i64 },
    Withdraw { amount: i64, floor: i64 }, Close { expected_version: u64 }
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    VersionOverflow, InvalidLimit, AlreadyOpened, NotOpened, Closed,
    InvalidAmount, LimitExceeded, InvalidFloor, InsufficientFunds,
    FloorViolation, CloseVersion { expected: u64, actual: u64 }, NonZeroBalance
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ReplayError { pub index: usize, pub kind: ApplyError }

fn initial() -> Account {
    Account { status: Status::New, balance: 0, limit: 0, version: 0 }
}

fn step(a: &mut Account, e: &Event) -> Result<(), ApplyError> {
    if a.version == u64::MAX { return Err(ApplyError::VersionOverflow); }
    match e {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) { return Err(ApplyError::InvalidLimit); }
            if a.status != Status::New { return Err(ApplyError::AlreadyOpened); }
            a.status = Status::Open;
            a.limit = *limit;
        }
        Event::Deposit { amount } => {
            match a.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 { return Err(ApplyError::InvalidAmount); }
            if *amount > a.limit - a.balance { return Err(ApplyError::LimitExceeded); }
            a.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match a.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 { return Err(ApplyError::InvalidAmount); }
            if *floor < 0 || *floor > a.limit { return Err(ApplyError::InvalidFloor); }
            if *amount > a.balance { return Err(ApplyError::InsufficientFunds); }
            if a.balance - *amount < *floor { return Err(ApplyError::FloorViolation); }
            a.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match a.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != a.version {
                return Err(ApplyError::CloseVersion { expected: *expected_version, actual: a.version });
            }
            if a.balance != 0 { return Err(ApplyError::NonZeroBalance); }
            a.status = Status::Closed;
        }
    }
    a.version += 1;
    Ok(())
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    let mut a = initial();
    for (index, e) in events.iter().enumerate() {
        step(&mut a, e).map_err(|kind| ReplayError { index, kind })?;
    }
    Ok(a)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Expected { Any, Missing, Exact(u64) }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StoreError {
    InvalidId { byte: usize },
    Conflict { expected: Expected, actual: Option<u64> },
    Rejected(ReplayError)
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Snapshot { pub account: Account, pub tag: u64 }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SnapshotError { ZeroInterval, InvalidVersion, InvalidState, TagMismatch }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RestoreError { Snapshot(SnapshotError), Tail(ReplayError) }

pub fn snapshot_tag(account: &Account) -> u64 {
    let status_code = match account.status { Status::New => 0, Status::Open => 0xA5, Status::Closed => 0x5A };
    account.version.rotate_left(7)
        ^ (account.limit as u64).wrapping_mul(31)
        ^ (account.balance as u64).rotate_left(17)
        ^ status_code
}

fn validate_snapshot(s: &Snapshot) -> Result<(), SnapshotError> {
    let a = &s.account;
    if a.version == 0 { return Err(SnapshotError::InvalidVersion); }
    if a.status == Status::New
        || !(1..=1_000_000_000).contains(&a.limit)
        || a.balance < 0 || a.balance > a.limit
        || (a.status == Status::Closed && (a.balance != 0 || a.version < 2)) {
        return Err(SnapshotError::InvalidState);
    }
    if s.tag != snapshot_tag(a) { return Err(SnapshotError::TagMismatch); }
    Ok(())
}

pub fn rebuild(snapshot: Option<&Snapshot>, tail: &[Event]) -> Result<Account, RestoreError> {
    let mut a = match snapshot {
        None => return replay(tail).map_err(RestoreError::Tail),
        Some(s) => {
            validate_snapshot(s).map_err(RestoreError::Snapshot)?;
            s.account.clone()
        }
    };
    for (index, e) in tail.iter().enumerate() {
        step(&mut a, e).map_err(|kind| RestoreError::Tail(ReplayError { index, kind }))?;
    }
    Ok(a)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Record { pub position: u64, pub stream: String, pub version: u64, pub event: Event }

#[derive(Debug, Clone, PartialEq, Eq)]
struct Stream {
    account: Account,
    events: Vec<Event>,
    snapshot: Option<Snapshot>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    snapshot_every: u64,
    streams: BTreeMap<String, Stream>,
    records: Vec<Record>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SnapshotErrorDummy { Never }

fn validate_id(id: &str) -> Result<(), usize> {
    let b = id.as_bytes();
    if b.is_empty() { return Err(0); }
    if !(b[0].is_ascii_lowercase()) { return Err(0); }
    for (i, c) in b.iter().enumerate().skip(1) {
        if !(c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'-') { return Err(i); }
    }
    if b.len() > 24 { return Err(24); }
    Ok(())
}

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 { return Err(SnapshotError::ZeroInterval); }
        Ok(Self { snapshot_every, streams: BTreeMap::new(), records: Vec::new() })
    }

    pub fn append(&mut self, id: &str, expected: Expected, events: &[Event]) -> Result<Account, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        let prior = self.streams.get(id);
        let actual = prior.map(|s| s.account.version);
        let matches = match expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(v) => actual == Some(v),
        };
        if !matches { return Err(StoreError::Conflict { expected, actual }); }
        let mut account = prior.map(|s| s.account.clone()).unwrap_or_else(initial);
        if events.is_empty() { return Ok(account); }

        let mut applied = Vec::with_capacity(events.len());
        let mut captured = None;
        for (index, event) in events.iter().enumerate() {
            step(&mut account, event)
                .map_err(|kind| StoreError::Rejected(ReplayError { index, kind }))?;
            applied.push((event.clone(), account.clone()));
            if account.version % self.snapshot_every == 0 {
                captured = Some(Snapshot { tag: snapshot_tag(&account), account: account.clone() });
            }
        }

        let stream = self.streams.entry(id.to_owned()).or_insert_with(|| Stream {
            account: initial(), events: Vec::new(), snapshot: None,
        });
        for (event, state) in applied {
            stream.events.push(event.clone());
            let position = (self.records.len() as u64) + 1;
            let version = state.version;
            self.records.push(Record { position, stream: id.to_owned(), version, event });
        }
        stream.account = account.clone();
        if let Some(s) = captured { stream.snapshot = Some(s); }
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|s| s.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|s| s.events.iter().enumerate()
            .filter(|(i, _)| (*i as u64) + 1 > after)
            .map(|(_, e)| e.clone()).collect()))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).and_then(|s| s.snapshot.clone()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|s| {
            let v = s.snapshot.as_ref().map(|x| x.account.version).unwrap_or(0) as usize;
            s.events.iter().skip(v).cloned().collect()
        }))
    }

    pub fn records(&self, after: u64) -> Vec<Record> {
        self.records.iter().filter(|r| r.position > after).cloned().collect()
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Totals { pub streams: u64, pub open: u64, pub closed: u64, pub balance: i128 }
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ProjectionError {
    InvalidId { index: usize, byte: usize }, ZeroPosition { index: usize },
    DuplicateMismatch { index: usize, position: u64 },
    Gap { index: usize, expected: u64, got: u64 },
    StreamVersion { index: usize, expected: u64, got: u64 },
    Rejected { index: usize, kind: ApplyError }
}
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Projection {
    journal: Vec<Record>,
    accounts: BTreeMap<String, Account>,
}

impl Projection {
    pub fn new() -> Self { Self { journal: Vec::new(), accounts: BTreeMap::new() } }

    pub fn ingest(&mut self, records: &[Record]) -> Result<usize, ProjectionError> {
        let mut provisional = self.clone();
        let mut added = 0;
        for (index, record) in records.iter().enumerate() {
            validate_id(&record.stream).map_err(|byte| ProjectionError::InvalidId { index, byte })?;
            if record.position == 0 { return Err(ProjectionError::ZeroPosition { index }); }
            if record.position <= provisional.journal.len() as u64 {
                if provisional.journal[(record.position - 1) as usize] != *record {
                    return Err(ProjectionError::DuplicateMismatch { index, position: record.position });
                }
                continue;
            }
            let expected_position = provisional.journal.len() as u64 + 1;
            if record.position != expected_position {
                return Err(ProjectionError::Gap { index, expected: expected_position, got: record.position });
            }
            let prior = provisional.accounts.get(&record.stream);
            let expected_version = prior.map(|a| a.version).unwrap_or(0) + 1;
            if record.version != expected_version {
                return Err(ProjectionError::StreamVersion { index, expected: expected_version, got: record.version });
            }
            let mut account = prior.cloned().unwrap_or_else(initial);
            step(&mut account, &record.event)
                .map_err(|kind| ProjectionError::Rejected { index, kind })?;
            provisional.accounts.insert(record.stream.clone(), account);
            provisional.journal.push(record.clone());
            added += 1;
        }
        *self = provisional;
        Ok(added)
    }

    pub fn account(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.accounts.get(id).cloned())
    }

    pub fn position(&self) -> u64 { self.journal.len() as u64 }

    pub fn totals(&self) -> Totals {
        let mut t = Totals { streams: 0, open: 0, closed: 0, balance: 0 };
        for a in self.accounts.values() {
            t.streams += 1;
            match a.status { Status::Open => t.open += 1, Status::Closed => t.closed += 1, Status::New => {} }
            t.balance += a.balance as i128;
        }
        t
    }
}
