pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

pub struct Runner {
    jobs: Vec<Job>,
    dependencies: Vec<Vec<usize>>,
    states: Vec<State>,
    attempts: Vec<u8>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    cycle_started: bool,
    last_time: u64,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job.id.bytes().all(|b| {
                    b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-'
                })
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if job.max_attempts == 0 || job.max_attempts > 32 {
                return Err(Error::BadAttempts(i));
            }
            if jobs[..i].iter().any(|prior| prior.id == job.id) {
                return Err(Error::Duplicate(i));
            }
        }

        let find_id = |id: &str| jobs.iter().position(|job| job.id == id);
        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut deps = Vec::with_capacity(job.dependencies.len());
            for (j, name) in job.dependencies.iter().enumerate() {
                if job.dependencies[..j].iter().any(|prior| prior == name) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match find_id(name) {
                    Some(index) => deps.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(deps);
        }

        let n = jobs.len();
        let mut indegree = vec![0usize; n];
        let mut dependents = vec![Vec::new(); n];
        for (job, deps) in dependencies.iter().enumerate() {
            indegree[job] = deps.len();
            for &dep in deps {
                dependents[dep].push(job);
            }
        }
        let mut queue = std::collections::VecDeque::new();
        for (i, &degree) in indegree.iter().enumerate() {
            if degree == 0 {
                queue.push_back(i);
            }
        }
        let mut visited = 0;
        while let Some(node) = queue.pop_front() {
            visited += 1;
            for &child in &dependents[node] {
                indegree[child] -= 1;
                if indegree[child] == 0 {
                    queue.push_back(child);
                }
            }
        }
        if visited != n {
            return Err(Error::Cycle);
        }

        Ok(Self {
            states: vec![State::Waiting { ready_at: 0 }; n],
            attempts: vec![0; n],
            jobs,
            dependencies,
            aging_ms,
            retry_base,
            retry_cap,
            cycle_started: false,
            last_time: 0,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.cycle_started
            && self.states.iter().any(|state| {
                matches!(state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for state in &mut self.states {
            *state = State::Waiting { ready_at: now };
        }
        self.attempts.fill(0);
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut choice: Option<(usize, u8)> = None;
        for i in 0..self.jobs.len() {
            let ready_at = match &self.states[i] {
                State::Waiting { ready_at } if *ready_at <= now => *ready_at,
                _ => continue,
            };
            if self.dependencies[i]
                .iter()
                .any(|&d| !matches!(self.states[d], State::Succeeded))
            {
                continue;
            }
            let priority = if self.aging_ms == 0 {
                self.jobs[i].priority
            } else {
                (u64::from(self.jobs[i].priority)
                    + ((now - ready_at) / self.aging_ms))
                    .min(9) as u8
            };
            if choice.map_or(true, |(_, best)| priority > best) {
                choice = Some((i, priority));
            }
        }

        self.last_time = now;
        let Some((i, _)) = choice else {
            return Ok(None);
        };
        self.attempts[i] += 1;
        let attempt = self.attempts[i];
        self.states[i] = State::Running { attempt };
        Ok(Some(Ticket {
            id: self.jobs[i].id.clone(),
            attempt,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(i) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        if !matches!(self.states[i], State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.states[i] = State::Succeeded;
        } else if self.attempts[i] >= self.jobs[i].max_attempts {
            self.states[i] = State::Failed;
            self.block_dependents();
        } else {
            let mut delay = self.retry_base;
            for _ in 1..self.attempts[i] {
                if delay >= self.retry_cap || delay > self.retry_cap / 2 {
                    delay = self.retry_cap;
                    break;
                }
                delay *= 2;
            }
            self.states[i] = State::Waiting {
                ready_at: now.saturating_add(delay.min(self.retry_cap)),
            };
        }
        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(i) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        Ok(self.states[i].clone())
    }

    fn block_dependents(&mut self) {
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if matches!(self.states[i], State::Waiting { .. })
                    && self.dependencies[i].iter().any(|&d| {
                        matches!(self.states[d], State::Failed | State::Blocked)
                    })
                {
                    self.states[i] = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }
}

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct Entry {
    job: Job,
    dependency_indices: Vec<usize>,
    status: State,
    attempt_count: u8,
}

pub struct Runner {
    entries: Vec<Entry>,
    index_by_id: HashMap<String, usize>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    timed_call_seen: bool,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut index_by_id = HashMap::new();
        for (index, job) in jobs.iter().enumerate() {
            if !is_valid_id(&job.id) {
                return Err(Error::BadId(index));
            } else if job.priority > 9 {
                return Err(Error::BadPriority(index));
            } else if job.max_attempts == 0 || job.max_attempts > 32 {
                return Err(Error::BadAttempts(index));
            } else if index_by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(index));
            }
            index_by_id.insert(job.id.clone(), index);
        }

        let mut dependency_indices = vec![Vec::new(); jobs.len()];
        for (i, job) in jobs.iter().enumerate() {
            let mut checked = Vec::<&str>::new();
            for (j, dependency) in job.dependencies.iter().enumerate() {
                if checked.contains(&dependency.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                checked.push(dependency);
                let Some(&dependency_index) = index_by_id.get(dependency) else {
                    return Err(Error::MissingDependency(i, j));
                };
                dependency_indices[i].push(dependency_index);
            }
        }

        if graph_has_cycle(&dependency_indices) {
            return Err(Error::Cycle);
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Entry {
                job,
                dependency_indices: std::mem::take(&mut dependency_indices[i]),
                status: State::Waiting { ready_at: 0 },
                attempt_count: 0,
            })
            .collect();

        Ok(Self {
            entries,
            index_by_id,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            timed_call_seen: false,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.ensure_time(now)?;
        if self.cycle_started
            && self.entries.iter().any(|e| {
                matches!(e.status, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for entry in self.entries.iter_mut() {
            entry.status = State::Waiting { ready_at: now };
            entry.attempt_count = 0;
        }
        self.cycle_started = true;
        self.commit_time(now);
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.ensure_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut candidate: Option<(usize, u8)> = None;
        for (index, entry) in self.entries.iter().enumerate() {
            let ready_at = match entry.status {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            let deps_done = entry
                .dependency_indices
                .iter()
                .all(|&d| matches!(self.entries[d].status, State::Succeeded));
            if !deps_done {
                continue;
            }

            let priority = if self.aging_ms == 0 {
                entry.job.priority
            } else {
                let boost = ((now - ready_at) / self.aging_ms).min(9) as u8;
                entry.job.priority.saturating_add(boost).min(9)
            };
            match candidate {
                None => candidate = Some((index, priority)),
                Some((old_index, old_priority))
                    if priority > old_priority
                        || (priority == old_priority && index < old_index) =>
                {
                    candidate = Some((index, priority));
                }
                _ => {}
            }
        }

        let ticket = match candidate {
            Some((index, _)) => {
                let entry = &mut self.entries[index];
                entry.attempt_count += 1;
                entry.status = State::Running {
                    attempt: entry.attempt_count,
                };
                Some(Ticket {
                    id: entry.job.id.clone(),
                    attempt: entry.attempt_count,
                })
            }
            None => None,
        };

        self.commit_time(now);
        Ok(ticket)
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.ensure_time(now)?;
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = *self.index_by_id.get(id).ok_or(Error::Unknown)?;
        if !matches!(self.entries[index].status, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.entries[index].status = State::Succeeded;
        } else if self.entries[index].attempt_count == self.entries[index].job.max_attempts {
            self.entries[index].status = State::Failed;
        } else {
            let delay = compute_delay(
                self.retry_base,
                self.retry_cap,
                self.entries[index].attempt_count,
            );
            self.entries[index].status = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.mark_blocked_to_fixed_point();
        self.commit_time(now);
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let index = *self.index_by_id.get(id).ok_or(Error::Unknown)?;
        Ok(self.entries[index].status.clone())
    }

    fn ensure_time(&self, now: u64) -> Result<(), Error> {
        if self.timed_call_seen && now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn commit_time(&mut self, now: u64) {
        self.last_time = now;
        self.timed_call_seen = true;
    }

    fn mark_blocked_to_fixed_point(&mut self) {
        loop {
            let mut changed = false;
            for i in 0..self.entries.len() {
                if matches!(self.entries[i].status, State::Waiting { .. }) {
                    let dependency_failed = self.entries[i].dependency_indices.iter().any(|&d| {
                        matches!(
                            self.entries[d].status,
                            State::Failed | State::Blocked
                        )
                    });
                    if dependency_failed {
                        self.entries[i].status = State::Blocked;
                        changed = true;
                    }
                }
            }
            if !changed {
                return;
            }
        }
    }
}

fn is_valid_id(value: &str) -> bool {
    let bytes = value.as_bytes();
    !bytes.is_empty()
        && bytes.len() <= 16
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn compute_delay(base: u64, cap: u64, attempt: u8) -> u64 {
    let mut value = base;
    let mut exponent = attempt.saturating_sub(1);
    while exponent > 0 && value < cap {
        if value > cap / 2 {
            return cap;
        }
        value *= 2;
        exponent -= 1;
    }
    value.min(cap)
}

fn graph_has_cycle(graph: &[Vec<usize>]) -> bool {
    let mut color = vec![0u8; graph.len()];
    for start in 0..graph.len() {
        if color[start] != 0 {
            continue;
        }
        let mut stack = vec![(start, false)];
        while let Some((node, leaving)) = stack.pop() {
            if leaving {
                color[node] = 2;
                continue;
            }
            if color[node] == 1 {
                return true;
            }
            if color[node] == 2 {
                continue;
            }
            color[node] = 1;
            stack.push((node, true));
            for &dependency in &graph[node] {
                if color[dependency] == 1 {
                    return true;
                }
                if color[dependency] == 0 {
                    stack.push((dependency, false));
                }
            }
        }
    }
    false
}

    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),

    NoCycle,

    NotRunning,

pub struct Runner {
    jobs: Vec<Job>,
    dependency_indices: Vec<Vec<usize>>,
    id_to_index: HashMap<String, usize>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    states: Vec<State>,
    attempts: Vec<u8>,
    cycle_started: bool,
    last_time: u64,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut id_to_index = HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if id_to_index.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            id_to_index.insert(job.id.clone(), i);
        }

        let mut dependency_indices = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = HashSet::new();
            let mut deps = Vec::with_capacity(job.dependencies.len());
            for (j, dependency) in job.dependencies.iter().enumerate() {
                if !seen.insert(dependency.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match id_to_index.get(dependency) {
                    Some(&index) => deps.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependency_indices.push(deps);
        }

        let n = jobs.len();
        let mut color = vec![0u8; n];
        fn visit(node: usize, deps: &[Vec<usize>], color: &mut [u8]) -> bool {
            if color[node] == 1 {
                return false;
            }
            if color[node] == 2 {
                return true;
            }
            color[node] = 1;
            for &dep in &deps[node] {
                if !visit(dep, deps, color) {
                    return false;
                }
            }
            color[node] = 2;
            true
        }
        for i in 0..n {
            if color[i] == 0 && !visit(i, &dependency_indices, &mut color) {
                return Err(Error::Cycle);
            }
        }

        Ok(Self {
            states: vec![State::Waiting { ready_at: 0 }; n],
            attempts: vec![0; n],
            jobs,
            dependency_indices,
            id_to_index,
            aging_ms,
            retry_base,
            retry_cap,
            cycle_started: false,
            last_time: 0,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.cycle_started
            && self.states.iter().any(|s| matches!(s, State::Waiting { .. } | State::Running { .. }))
        {
            return Err(Error::Busy);
        }

        self.states.fill(State::Waiting { ready_at: now });
        self.attempts.fill(0);
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut selected: Option<(usize, u8)> = None;
        for i in 0..self.jobs.len() {
            let ready_at = match self.states[i] {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !self.dependency_indices[i]
                .iter()
                .all(|&dep| matches!(self.states[dep], State::Succeeded))
            {
                continue;
            }

            let base = self.jobs[i].priority;
            let effective = if self.aging_ms == 0 {
                base
            } else {
                let aged = (now - ready_at) / self.aging_ms;
                (u64::from(base) + aged).min(9) as u8
            };
            if selected.is_none_or(|(_, best)| effective > best) {
                selected = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((i, _)) = selected else {
            return Ok(None);
        };

        self.attempts[i] += 1;
        let attempt = self.attempts[i];
        self.states[i] = State::Running { attempt };
        Ok(Some(Ticket {
            id: self.jobs[i].id.clone(),
            attempt,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(&i) = self.id_to_index.get(id) else {
            return Err(Error::Unknown);
        };
        if !matches!(self.states[i], State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.states[i] = State::Succeeded;
        } else if self.attempts[i] == self.jobs[i].max_attempts {
            self.states[i] = State::Failed;
            self.propagate_blocked();
        } else {
            let delay = self.retry_delay(self.attempts[i]);
            self.states[i] = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }
        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(&i) = self.id_to_index.get(id) else {
            return Err(Error::Unknown);
        };
        Ok(self.states[i].clone())
    }

    fn retry_delay(&self, attempt: u8) -> u64 {
        let mut delay = self.retry_base;
        for _ in 1..attempt {
            if delay >= self.retry_cap || delay > self.retry_cap / 2 {
                return self.retry_cap;
            }
            delay *= 2;
        }
        delay.min(self.retry_cap)
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if !matches!(self.states[i], State::Waiting { .. }) {
                    continue;
                }
                if self.dependency_indices[i].iter().any(|&dep| {
                    matches!(self.states[dep], State::Failed | State::Blocked)
                }) {
                    self.states[i] = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }
}

    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut lookup = std::collections::HashMap::new();
        for (i, j) in jobs.iter().enumerate() {
            if !valid_id(&j.id) {
                return Err(Error::BadId(i));
            }
            if j.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&j.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if lookup.insert(j.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut deps = Vec::new();
        for (i, j) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut row = Vec::new();
            for (k, name) in j.dependencies.iter().enumerate() {
                if !seen.insert(name) {
                    return Err(Error::DuplicateDependency(i, k));
                }
                match lookup.get(name) {
                    Some(&d) => row.push(d),
                    None => return Err(Error::MissingDependency(i, k)),
                }
            }
            deps.push(row);
        }

        let mut indegree = vec![0usize; jobs.len()];
        let mut children = vec![Vec::new(); jobs.len()];
        for (i, row) in deps.iter().enumerate() {
            for &d in row {
                indegree[i] += 1;
                children[d].push(i);
            }
        }
        let mut queue: std::collections::VecDeque<usize> = indegree
            .iter()
            .enumerate()
            .filter_map(|(i, &n)| (n == 0).then_some(i))
            .collect();
        let mut visited = 0;
        while let Some(i) = queue.pop_front() {
            visited += 1;
            for &child in &children[i] {
                indegree[child] -= 1;
                if indegree[child] == 0 {
                    queue.push_back(child);
                }
            }
        }
        if visited != jobs.len() {
            return Err(Error::Cycle);
        }

        let entries = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Entry {
                job,
                deps: deps[i].clone(),
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();
        Ok(Self {
            entries,
            aging: aging_ms,
            base: retry_base,
            cap: retry_cap,
            last: 0,
            active: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check(now)?;
        if self.active
            && self.entries.iter().any(|e| matches!(e.state, State::Waiting { .. } | State::Running { .. }))
        {
            return Err(Error::Busy);
        }
        for e in &mut self.entries {
            e.state = State::Waiting { ready_at: now };
            e.attempts = 0;
        }
        self.active = true;
        self.last = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check(now)?;
        if !self.active {
            return Err(Error::NoCycle);
        }
        let mut selected = None;
        for i in 0..self.entries.len() {
            let e = &self.entries[i];
            let start = match e.state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if e.deps.iter().any(|&d| !matches!(self.entries[d].state, State::Succeeded)) {
                continue;
            }
            let age = if self.aging == 0 { 0 } else { (now - start) / self.aging };
            let priority = ((e.job.priority as u64).saturating_add(age)).min(9) as u8;
            if selected.map_or(true, |(best, p)| priority > p || (priority == p && i < best)) {
                selected = Some((i, priority));
            }
        }
        self.last = now;
        if let Some((i, _)) = selected {
            let e = &mut self.entries[i];
            e.attempts += 1;
            e.state = State::Running { attempt: e.attempts };
            Ok(Some(Ticket { id: e.job.id.clone(), attempt: e.attempts }))
        } else {
            Ok(None)
        }
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check(now)?;
        if !self.active {
            return Err(Error::NoCycle);
        }
        let i = match self.entries.iter().position(|e| e.job.id == id) {
            Some(i) => i,
            None => return Err(Error::Unknown),
        };
        if !matches!(self.entries[i].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        self.last = now;
        if success {
            self.entries[i].state = State::Succeeded;
        } else if self.entries[i].attempts >= self.entries[i].job.max_attempts {
            self.entries[i].state = State::Failed;
            self.block_descendants();
        } else {
            let delay = retry_delay(self.base, self.cap, self.entries[i].attempts);
            self.entries[i].state = State::Waiting { ready_at: now.saturating_add(delay) };
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {

            return Err(Error::NoCycle);
        }
        self.entries.iter().find(|e| e.job.id == id).map(|e| e.state.clone()).ok_or(Error::Unknown)
    }

    fn check(&self, now: u64) -> Result<(), Error> {
        if now < self.last { Err(Error::TimeReversed) } else { Ok(()) }
    }

                });

                    self.entries[i].state = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                return;
            }
        }
    }

        }

        let mut ids = std::collections::HashMap::<String, usize>::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_id(&job.id) {

            if job.max_attempts == 0 || job.max_attempts > 32 {

            }

        for (i, job) in jobs.iter().enumerate() {
            let mut row = Vec::new();
            for (j, name) in job.dependencies.iter().enumerate() {

                };
                row.push(index);
            }

        let mut stack = Vec::new();

            if color[root] == 0 && cycle_dfs(root, &deps, &mut color, &mut stack) {
                return Err(Error::Cycle);
            }

            .into_iter()
            .zip(deps)

                job,
                deps,

                attempts: 0,

            last_time: 0,

            retry_cap,

        if self.cycle_started
            && self.entries.iter().any(|e| {
                matches!(e.state, State::Waiting { .. } | State::Running { .. })
            })

            if ready_at > now || !entry.deps.iter().all(|&d| self.entries[d].state == State::Succeeded) {
                continue;

            let p = if self.aging_ms == 0 {
                entry.job.priority
            } else {
                (u64::from(entry.job.priority) + (now - ready_at) / self.aging_ms).min(9) as u8
            };
            if chosen.is_none_or(|(_, best)| p > best) {
                chosen = Some((i, p));
            }
        }

            entry.state = State::Running {

            };
            Some(Ticket {

            })
        } else {

        if !self.cycle_started {
            return Err(Error::NoCycle);

            .entries
            .iter()
            .position(|entry| entry.job.id == id)
            .ok_or(Error::Unknown)?;
        if !matches!(self.entries[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.entries[index].state = State::Succeeded;
        } else if self.entries[index].attempts == self.entries[index].job.max_attempts {
            self.entries[index].state = State::Failed;
            self.block_dependents();
        } else {
            let attempt = self.entries[index].attempts;
            let delay = retry_delay(self.retry_base, self.retry_cap, attempt);
            self.entries[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }
        self.last_time = now;
        Ok(())
    }

        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        self.entries
            .iter()
            .find(|entry| entry.job.id == id)
            .map(|entry| entry.state.clone())
            .ok_or(Error::Unknown)
    }

            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

                    continue;

                if self.entries[i]
                    .deps
                    .iter()
                    .any(|&d| matches!(self.entries[d].state, State::Failed | State::Blocked))
                {
                    self.entries[i].state = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }

            return cap;

        delay = delay.saturating_mul(2).min(cap);

fn cycle_dfs(node: usize, deps: &[Vec<usize>], color: &mut [u8], stack: &mut Vec<usize>) -> bool {

        return true;
    }
    if color[node] == 2 {

    for &dep in &deps[node] {

    }

    pub fn new(

        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,

                }

                    None => return Err(Error::MissingDependency(i, j)),

            dependencies.push(deps);

            return Err(Error::Cycle);
        }

            .zip(dependencies)
            .map(|(job, dependencies)| Entry {
                job,
                dependencies,
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            entries,
            by_id,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            has_timed_call: false,
            has_cycle: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        self.check_time(now)?;
        if self.has_cycle
            && self.entries.iter().any(|entry| {
                matches!(entry.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for entry in &mut self.entries {
            entry.state = State::Waiting { ready_at: now };
            entry.attempts = 0;
        }
        self.has_cycle = true;
        self.record_time(now);
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (i, entry) in self.entries.iter().enumerate() {
            let ready_at = match entry.state {
                State::Waiting { ready_at } if ready_at <= now => ready_at,
                _ => continue,
            };
            if !entry
                .dependencies
                .iter()
                .all(|&d| matches!(self.entries[d].state, State::Succeeded))
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                entry.job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                let boost = age.min(9) as u8;
                entry.job.priority.saturating_add(boost).min(9)
            };

            if best.map_or(true, |(best_i, best_priority)| {
                effective > best_priority || (effective == best_priority && i < best_i)
            }) {
                best = Some((i, effective));
            }
        }

            entry.attempts += 1;
            entry.state = State::Running {
                attempt: entry.attempts,
            };
            Some(Ticket {
                id: entry.job.id.clone(),
                attempt: entry.attempts,
            })
        } else {
            None
        };

        self.record_time(now);
        Ok(result)
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        self.check_time(now)?;
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let index = *self.by_id.get(id).ok_or(Error::Unknown)?;
        if !matches!(self.entries[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        if success {
            self.entries[index].state = State::Succeeded;
        } else if self.entries[index].attempts == self.entries[index].job.max_attempts {
            self.entries[index].state = State::Failed;
        } else {
            let delay = retry_delay(
                self.retry_base,
                self.retry_cap,
                self.entries[index].attempts,
            );
            self.entries[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }

        self.propagate_blocked();
        self.record_time(now);
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.has_cycle {
            return Err(Error::NoCycle);
        }
        let index = *self.by_id.get(id).ok_or(Error::Unknown)?;
        Ok(self.entries[index].state.clone())
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if self.has_timed_call && now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn record_time(&mut self, now: u64) {
        self.last_time = now;
        self.has_timed_call = true;
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut newly_blocked = Vec::new();
            for (i, entry) in self.entries.iter().enumerate() {
                if matches!(entry.state, State::Waiting { .. })
                    && entry.dependencies.iter().any(|&d| {
                        matches!(
                            self.entries[d].state,
                            State::Failed | State::Blocked
                        )
                    })

                    newly_blocked.push(i);
                }
            }
            if newly_blocked.is_empty() {
                break;
            }
            for i in newly_blocked {
                self.entries[i].state = State::Blocked;
            }
        }
    }
}

        } else if self.entries[index].attempts == self.entries[index].job.max_attempts {
            self.entries[index].state = State::Failed;
            self.propagate_blocked();
        } else {
            let attempt = self.entries[index].attempts;
            let delay = retry_delay(self.retry_base, self.retry_cap, attempt);
            self.entries[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        self.entries
            .iter()
            .find(|e| e.job.id == id)
            .map(|e| e.state.clone())
            .ok_or(Error::Unknown)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }

    fn propagate_blocked(&mut self) {
        loop {
            let to_block: Vec<usize> = self
                .entries
                .iter()
                .enumerate()
                .filter_map(|(i, entry)| {
                    if !matches!(entry.state, State::Waiting { .. }) {
                        return None;
                    }
                    entry.dependencies.iter().any(|&d| {
                        matches!(self.entries[d].state, State::Failed | State::Blocked)
                    }).then_some(i)
                })
                .collect();
            if to_block.is_empty() {
                break;
            }
            for i in to_block {
                self.entries[i].state = State::Blocked;
            }
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn retry_delay(base: u64, cap: u64, attempt: u8) -> u64 {
    let mut delay = base;
    for _ in 1..attempt {
        if delay >= cap || delay > cap / 2 {
            return cap;
        }
        delay *= 2;
    }
    delay.min(cap)
}
