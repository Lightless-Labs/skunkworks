pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut quote: Option<(char, usize)> = None;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        if let Some((delimiter, _)) = quote {
            if ch == delimiter {
                quote = None;
                continue;
            }

            if delimiter == '"' && ch == '\\' {
                let Some((_, next)) = chars.peek().copied() else {
                    return Err(SplitError::TrailingEscape { pos });
                };
                if matches!(next, '"' | '\\' | '$') {
                    chars.next();
                    current.push(next);
                } else if next == '\n' {
                    chars.next();
                } else {
                    current.push('\\');
                }
                continue;
            }

            current.push(ch);
            continue;
        }

        match ch {
            '\'' | '"' => {
                active = true;
                quote = Some((ch, pos));
            }
            '\\' => match chars.peek().copied() {
                None => return Err(SplitError::TrailingEscape { pos }),
                Some((_, '\n')) => {
                    chars.next();
                }
                Some((_, escaped)) => {
                    chars.next();
                    active = true;
                    current.push(escaped);
                }
            },
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    words.push(std::mem::take(&mut current));
                    active = false;
                }
            }
            _ => {
                active = true;
                current.push(ch);
            }
        }
    }

    if let Some((delimiter, pos)) = quote {
        return Err(if delimiter == '\'' {
            SplitError::UnterminatedSingle { pos }
        } else {
            SplitError::UnterminatedDouble { pos }
        });
    }

    if active {
        words.push(current);
    }
    Ok(words)
}

    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut current = String::new();
    let mut has_word = false;
    let mut state = 0u8; // 0: bare, 1: single-quoted, 2: double-quoted
    let mut quote_start = 0usize;
    let mut iter = input.char_indices();

    while let Some((pos, ch)) = iter.next() {
        if state == 1 {
            if ch == '\'' {
                state = 0;
            } else {
                current.push(ch);
            }
            continue;
        }

        if state == 2 {
            match ch {
                '"' => state = 0,
                '\\' => match iter.next() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, '\n')) => {}
                    Some((_, escaped @ ('"' | '\\' | '$'))) => current.push(escaped),
                    Some((next_pos, next)) => {
                        current.push('\\');
                        // The next scalar is processed normally in double quotes.
                        // Insert it directly, applying only quote closing behavior.
                        if next == '"' {
                            state = 0;
                        } else {
                            current.push(next);
                        }
                        let _ = next_pos;
                    }
                },
                _ => current.push(ch),
            }
            continue;
        }

        match ch {
            '\'' => {
                has_word = true;
                quote_start = pos;
                state = 1;
            }
            '"' => {
                has_word = true;
                quote_start = pos;
                state = 2;
            }
            '\\' => match iter.next() {
                None => return Err(SplitError::TrailingEscape { pos }),
                Some((_, '\n')) => {}
                Some((_, escaped)) => {
                    has_word = true;
                    current.push(escaped);
                }
            },
            ' ' | '\t' | '\r' | '\n' => {
                if has_word {
                    result.push(std::mem::take(&mut current));
                    has_word = false;
                }
            }
            _ => {
                has_word = true;
                current.push(ch);
            }
        }
    }

    if state == 1 {
        return Err(SplitError::UnterminatedSingle { pos: quote_start });
    }
    if state == 2 {
        return Err(SplitError::UnterminatedDouble { pos: quote_start });
    }
    if has_word {
        result.push(current);
    }

    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    #[derive(Clone, Copy)]
    enum State {
        Bare,
        Single(usize),
        Double(usize),
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut state = State::Bare;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match state {
            State::Bare => match ch {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    state = State::Single(pos);
                }
                '"' => {
                    active = true;
                    state = State::Double(pos);
                }
                '\\' => match chars.peek().copied() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, '\n')) => {
                        chars.next();
                    }
                    Some((_, next)) => {
                        chars.next();
                        word.push(next);
                        active = true;
                    }
                },
                _ => {
                    word.push(ch);
                    active = true;
                }
            },
            State::Single(open_pos) => {
                if ch == '\'' {
                    state = State::Bare;
                } else {
                    word.push(ch);
                    active = true;
                    let _ = open_pos;
                }
            }
            State::Double(open_pos) => match ch {
                '"' => state = State::Bare,
                '\\' => match chars.peek().copied() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, next @ ('"' | '\\' | '$'))) => {
                        chars.next();
                        word.push(next);
                        active = true;
                    }
                    Some((_, '\n')) => {
                        chars.next();
                    }
                    Some(_) => {
                        word.push('\\');
                        active = true;
                    }
                },
                _ => {
                    word.push(ch);
                    active = true;
                }
            },
        }
        if let State::Double(open_pos) = state {
            let _ = open_pos;
        }
    }

    match state {
        State::Bare => {}
        State::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        State::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

    TrailingEscape { pos: usize },

            }
            continue;
        }

                    result.push(std::mem::take(&mut current));
                    word_started = false;
                }
            }
            '\'' => {
                mode = 1;
                quote_start = byte_pos;
                word_started = true;
            }

                mode = 2;
                quote_start = byte_pos;
                word_started = true;
            }

                    return Err(SplitError::TrailingEscape { pos: byte_pos });

                if next != '\n' {
                    current.push(next);
                    word_started = true;
                }
            }
            _ => {
                current.push(c);
                word_started = true;
            }
        }

        2 => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
