- NUL is checked across the entire input before parsing, so it outranks any syntax error.
- Error positions are UTF-8 byte offsets; quote errors point to the opening delimiter.
- Only space, tab, CR, and LF separate words outside quotes.
- Empty quotes activate a word; a line continuation by itself does not.
- In double quotes, only `\"`, `\\`, `\$`, and backslash-LF consume the backslash.
- At EOF, a trailing double-quoted escape takes precedence over an unterminated double quote.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    #[derive(Clone, Copy)]
    enum Quote {
        None,
        Single(usize),
        Double(usize),
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut iter = input.char_indices().peekable();

    while let Some((pos, ch)) = iter.next() {
        match quote {
            Quote::Single(_) => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(ch);
                }
            }
            Quote::Double(_) => {
                if ch == '"' {
                    quote = Quote::None;
                } else if ch == '\\' {
                    match iter.peek().copied() {
                        None => return Err(SplitError::TrailingEscape { pos }),
                        Some((_, next)) if next == '"' || next == '\\' || next == '$' => {
                            iter.next();
                            word.push(next);
                        }
                        Some((_, '\n')) => {
                            iter.next();
                        }
                        Some(_) => word.push('\\'),
                    }
                } else {
                    word.push(ch);
                }
            }
            Quote::None => match ch {
                '\'' => {
                    active = true;
                    quote = Quote::Single(pos);
                }
                '"' => {
                    active = true;
                    quote = Quote::Double(pos);
                }
                '\\' => match iter.peek().copied() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, '\n')) => {
                        iter.next();
                    }
                    Some((_, next)) => {
                        iter.next();
                        active = true;
                        word.push(next);
                    }
                },
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                _ => {
                    active = true;
                    word.push(ch);
                }
            },
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut quote: Option<(char, usize)> = None;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        if let Some((delimiter, _)) = quote {
            if ch == delimiter {
                quote = None;
                continue;
            }

            if delimiter == '"' && ch == '\\' {
                let Some((_, next)) = chars.peek().copied() else {
                    return Err(SplitError::TrailingEscape { pos });
                };
                if matches!(next, '"' | '\\' | '$') {
                    chars.next();
                    current.push(next);
                } else if next == '\n' {
                    chars.next();
                } else {
                    current.push('\\');
                }
                continue;
            }

            current.push(ch);
            continue;
        }

        match ch {
            '\'' | '"' => {
                active = true;
                quote = Some((ch, pos));
            }
            '\\' => match chars.peek().copied() {
                None => return Err(SplitError::TrailingEscape { pos }),
                Some((_, '\n')) => {
                    chars.next();
                }
                Some((_, escaped)) => {
                    chars.next();
                    active = true;
                    current.push(escaped);
                }
            },
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    words.push(std::mem::take(&mut current));
                    active = false;
                }
            }
            _ => {
                active = true;
                current.push(ch);
            }
        }
    }

    if let Some((delimiter, pos)) = quote {
        return Err(if delimiter == '\'' {
            SplitError::UnterminatedSingle { pos }
        } else {
            SplitError::UnterminatedDouble { pos }
        });
    }

    if active {
        words.push(current);
    }
    Ok(words)
}
```
