- NUL is checked across the whole input before parsing, so it takes precedence over syntax errors.
- Error positions are UTF-8 byte offsets; quote and escape positions refer to their opening or backslash characters.
- Only ASCII space, tab, CR, and LF separate words outside quotes.
- Opening quotes activate a word even when empty; a line continuation alone does not.
- In double quotes, backslash is special only before `"`, `\`, `$`, or LF; otherwise both characters are retained.
- EOF after a double-quoted trailing backslash reports `TrailingEscape` before an unterminated quote.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote: Option<(char, usize)> = None;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match quote {
            Some(('\'', _)) => {
                if ch == '\'' {
                    quote = None;
                } else {
                    word.push(ch);
                }
            }
            Some(('"', _)) => {
                if ch == '"' {
                    quote = None;
                } else if ch == '\\' {
                    let Some((_, next)) = chars.next() else {
                        return Err(SplitError::TrailingEscape { pos });
                    };
                    match next {
                        '"' | '\\' | '$' => word.push(next),
                        '\n' => {}
                        _ => {
                            word.push('\\');
                            word.push(next);
                        }
                    }
                } else {
                    word.push(ch);
                }
            }
            None => match ch {
                '\'' | '"' => {
                    active = true;
                    quote = Some((ch, pos));
                }
                '\\' => {
                    if let Some((_, next)) = chars.next() {
                        if next != '\n' {
                            active = true;
                            word.push(next);
                        }
                    } else {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                }
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                _ => {
                    active = true;
                    word.push(ch);
                }
            },
        }
    }

    if let Some((quote_char, pos)) = quote {
        return if quote_char == '\'' {
            Err(SplitError::UnterminatedSingle { pos })
        } else {
            Err(SplitError::UnterminatedDouble { pos })
        };
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut started = false;
    let mut mode = 0u8; // 0: bare, 1: single quoted, 2: double quoted
    let mut quote_start = 0usize;
    let mut iter = input.char_indices();

    while let Some((pos, ch)) = iter.next() {
        match mode {
            1 => {
                if ch == '\'' {
                    mode = 0;
                } else {
                    current.push(ch);
                }
            }
            2 => {
                if ch == '"' {
                    mode = 0;
                } else if ch == '\\' {
                    match iter.next() {
                        None => return Err(SplitError::TrailingEscape { pos }),
                        Some((_, next)) if matches!(next, '"' | '\\' | '$') => current.push(next),
                        Some((_, '\n')) => {}
                        Some((_, next)) => {
                            current.push('\\');
                            current.push(next);
                        }
                    }
                } else {
                    current.push(ch);
                }
            }
            _ => match ch {
                '\'' => {
                    started = true;
                    mode = 1;
                    quote_start = pos;
                }
                '"' => {
                    started = true;
                    mode = 2;
                    quote_start = pos;
                }
                '\\' => match iter.next() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, '\n')) => {}
                    Some((_, next)) => {
                        started = true;
                        current.push(next);
                    }
                },
                ' ' | '\t' | '\r' | '\n' => {
                    if started {
                        words.push(std::mem::take(&mut current));
                        started = false;
                    }
                }
                _ => {
                    started = true;
                    current.push(ch);
                }
            },
        }
    }

    match mode {
        1 => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        2 => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        _ => {}
    }

    if started {
        words.push(current);
    }
    Ok(words)
}
```
