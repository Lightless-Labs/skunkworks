pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Validate every dependency before any other validation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    // Resource validation is task-major, then resource-major.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    // Kahn traversal with a min-heap represented by a sorted vector.
    let mut indegree = vec![0usize; n];
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while !available.is_empty() {
        let id = available.remove(0);
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let pos = available.binary_search(&child).unwrap_or_else(|p| p);
                available.insert(pos, child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        let removed = {
            let mut flags = vec![false; n];
            for &id in &order {
                flags[id] = true;
            }
            flags
        };
        for id in 0..n {
            if !removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let mut max_child = 0u64;
        for &child in &children[id] {
            max_child = max_child.max(tail[child]);
        }
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    let mut trace = format!("workers={};resources={}\n", cooldowns.len(), capacity.len());
    let mut used = vec![0u64; capacity.len()];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut worker_available = vec![0u64; cooldowns.len()];
    let mut running = Vec::<(u64, usize, usize)>::new(); // finish, task, worker
    let mut finished_count = 0usize;
    let mut now = 0u64;
    let mut makespan = 0u64;

    while finished_count < n {
        // Complete all jobs ending now in ascending task ID.
        let mut ending: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|(finish, _, _)| *finish == now)
            .collect();
        ending.sort_by_key(|&(_, task_id, _)| task_id);

        for (_, task_id, worker) in ending {
            running.retain(|&(_, id, _)| id != task_id);
            completed[task_id] = true;
            finished_count += 1;
            for r in 0..capacity.len() {
                used[r] -= tasks[task_id].needs[r] as u64;
            }
            trace.push_str(&format!(
                "{}|end|{}|{}|{}\n",
                now,
                task_id,
                worker,
                resource_string(&used)
            ));
        }

        // Admit one task at a time, rechecking eligibility and resources.
        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now {
                    continue;
                }
                if tasks[id].deps.iter().any(|&pred| !completed[pred]) {
                    continue;
                }
                let fits = (0..capacity.len()).all(|r| {
                    (tasks[id].needs[r] as u64)
                        <= (capacity[r] as u64 - used[r])
                });
                if !fits {
                    continue;
                }
                if let Some(prev) = best {
                    let better = tail[id] > tail[prev]
                        || (tail[id] == tail[prev]
                            && (tasks[id].priority > tasks[prev].priority
                                || (tasks[id].priority == tasks[prev].priority && id < prev)));
                    if better {
                        best = Some(id);
                    }
                } else {
                    best = Some(id);
                }
            }

            let Some(task_id) = best else {
                break;
            };

            let worker_count = cooldowns.len();
            let mut chosen_worker = None;
            for offset in 0..worker_count {
                let worker = (task_id % worker_count + offset) % worker_count;
                if worker_available[worker] <= now {
                    chosen_worker = Some(worker);
                    break;
                }
            }
            let Some(worker) = chosen_worker else {
                break;
            };

            let finish = now
                .checked_add(tasks[task_id].duration)
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Finish,
                })?;
            let available_at = finish
                .checked_add(cooldowns[worker])
                .ok_or(Error::Overflow {
                    task: task_id,
                    stage: Stage::Turnaround,
                })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            for r in 0..capacity.len() {
                used[r] += tasks[task_id].needs[r] as u64;
            }
            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                resource_string(&used)
            ));

            if tasks[task_id].duration == 0 {
                completed[task_id] = true;
                finished_count += 1;
                for r in 0..capacity.len() {
                    used[r] -= tasks[task_id].needs[r] as u64;
                }
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    resource_string(&used)
                ));
            } else {
                running.push((finish, task_id, worker));
            }
            makespan = makespan.max(finish);
        }

        if finished_count == n {
            break;
        }

        // Advance to the earliest relevant strictly-future event.
        let mut next: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next = Some(next.map_or(finish, |v| v.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |v| v.min(tasks[id].release)));
            }
        }
        for worker in 0..worker_available.len() {
            if worker_available[worker] > now {
                next = Some(next.map_or(worker_available[worker], |v| v.min(worker_available[worker])));
            }
        }

        if let Some(next_time) = next {
            now = next_time;
        } else {
            // Given validated acyclic dependencies, this can only be reached
            // if an impossible internal state occurs.
            break;
        }
    }

    for id in 0..n {
        if runs[id].is_none() {
            // This is unreachable for a valid acyclic graph and finite releases.
            return Err(Error::Cycle { blocked: vec![id] });
        }
    }

    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

pub fn schedule(tasks: &[Task], cooldowns: &[u64], capacity: &[u32]) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Dependency validation has strict precedence over all other work.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; n];
        for (position, &dep) in task.deps.iter().enumerate() {
            if dep >= n {
                return Err(Error::BadDependency { task: task_id, position, value: dep });
            }
            if dep == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if seen[dep] {
                return Err(Error::DuplicateDependency { task: task_id, position, value: dep });
            }
            seen[dep] = true;
        }
    }

    if cooldowns.is_empty() || cooldowns.len() > 64 {
        return Err(Error::Workers { count: cooldowns.len() });
    }

    // Validate resources task-by-task, preserving the required error ordering.
    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &cap)) in task.needs.iter().zip(capacity.iter()).enumerate() {
            if need > cap {
                return Err(Error::ResourceExceeded { task: task_id, resource, need, capacity: cap });
            }
        }
    }

    let mut indegree: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    // Sorted available IDs; insertion keeps the next removal deterministic.
    let mut available = std::collections::BTreeSet::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.insert(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(&id) = available.iter().next() {
        available.remove(&id);
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.insert(child);
            }
        }
    }

    if order.len() != n {
        let removed = {
            let mut flags = vec![false; n];
            for &id in &order {
                flags[id] = true;
            }
            flags
        };
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let mut max_child = 0u64;
        for &child in &children[id] {
            max_child = max_child.max(tail[child]);
        }
        tail[id] = tasks[id]
            .duration
            .checked_add(max_child)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let worker_count = cooldowns.len();
    let mut worker_available = vec![0u64; worker_count];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut used = vec![0u64; capacity.len()];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut running: Vec<(usize, usize, u64)> = Vec::new(); // task, worker, finish
    let mut trace = format!("workers={};resources={}\n", worker_count, capacity.len());
    let mut now = 0u64;
    let mut makespan = 0u64;
    let mut done_count = 0usize;

    while done_count < n {
        // Complete every task ending now, in ascending task ID.
        let mut ending: Vec<(usize, usize)> = running
            .iter()
            .filter(|entry| entry.2 == now)
            .map(|entry| (entry.0, entry.1))
            .collect();
        ending.sort_by_key(|entry| entry.0);

        for (task_id, worker) in ending {
            if let Some(pos) = running.iter().position(|entry| entry.0 == task_id) {
                running.remove(pos);
            }
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] -= need as u64;
            }
            completed[task_id] = true;
            done_count += 1;
            makespan = makespan.max(now);
            trace.push_str(&format!(
                "{}|end|{}|{}|{}\n",
                now,
                task_id,
                worker,
                usage_string(&used)
            ));
            for &child in &children[task_id] {
                remaining_deps[child] -= 1;
            }
        }

        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now || remaining_deps[id] != 0 {
                    continue;
                }
                let fits = tasks[id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(r, &need)| (need as u64) <= (capacity[r] as u64 - used[r]));
                if !fits {
                    continue;
                }
                let candidate_better = match best {
                    None => true,
                    Some(old) => {
                        tail[id] > tail[old]
                            || (tail[id] == tail[old]
                                && (tasks[id].priority > tasks[old].priority
                                    || (tasks[id].priority == tasks[old].priority && id < old)))
                    }
                };
                if candidate_better {
                    best = Some(id);
                }
            }

            let task_id = match best {
                Some(id) => id,
                None => break,
            };

            let mut worker = None;
            for offset in 0..worker_count {
                let candidate = (task_id % worker_count + offset) % worker_count;
                if worker_available[candidate] <= now {
                    worker = Some(candidate);
                    break;
                }
            }
            let worker = match worker {
                Some(w) => w,
                None => break,
            };

            let finish = now.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_available[worker] = available_at;
            runs[task_id] = Some(Run { start: now, finish, worker });
            for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                used[r] += need as u64;
            }
            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                usage_string(&used)
            ));

            if tasks[task_id].duration == 0 {
                for (r, &need) in tasks[task_id].needs.iter().enumerate() {
                    used[r] -= need as u64;
                }
                completed[task_id] = true;
                done_count += 1;
                makespan = makespan.max(now);
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    usage_string(&used)
                ));
                for &child in &children[task_id] {
                    remaining_deps[child] -= 1;
                }
            } else {
                running.push((task_id, worker, finish));
            }
        }

        if done_count == n {
            break;
        }

        let mut next: Option<u64> = None;
        for &(_, _, finish) in &running {
            if finish > now {
                next = Some(next.map_or(finish, |old| old.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |old| old.min(tasks[id].release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next = Some(next.map_or(available_at, |old| old.min(available_at)));
            }
        }

        match next {
            Some(t) => now = t,
            None => {
                // With a validated acyclic graph, progress must be possible at some event.
                // This guard prevents an accidental infinite loop if invariants are violated.
                return Err(Error::Cycle { blocked: (0..n).filter(|&id| !completed[id]).collect() });
            }
        }
    }

    let final_makespan = makespan;
    trace.push_str(&format!("makespan={}\n", final_makespan));
    let runs = runs.into_iter().map(Option::unwrap).collect();

    Ok(Schedule { runs, makespan: final_makespan, trace })
}

pub fn schedule(tasks: &[Task], cooldowns: &[u64], capacity: &[u32]) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Dependency validation is deliberately completed before any other check.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }
            if value == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
        }
    }

    let worker_count = cooldowns.len();
    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers { count: worker_count });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            children[pred].push(task_id);
        }
    }

    // Sorted-vector Kahn queue gives the smallest currently available ID.
    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }
    let mut topo = Vec::with_capacity(n);
    while !available.is_empty() {
        let id = available.remove(0);
        topo.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let at = available.binary_search(&child).unwrap_or_else(|p| p);
                available.insert(at, child);
            }
        }
    }
    if topo.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &id in &topo {
            removed[id] = true;
        }
        for id in 0..n {
            if !removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut tail = vec![0u64; n];
    for &id in topo.iter().rev() {
        let mut longest_child = 0u64;
        for &child in &children[id] {
            longest_child = longest_child.max(tail[child]);
        }
        tail[id] = tasks[id]
            .duration
            .checked_add(longest_child)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let mut remaining = vec![tasks.iter().map(|t| t.deps.len()).collect::<Vec<_>>()];
    let mut unmet = remaining.pop().unwrap_or_default();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut worker_free = vec![0u64; worker_count];
    let mut used = vec![0u64; capacity.len()];
    let mut running = Vec::<(u64, usize, usize)>::new(); // (finish, task, worker)
    let mut trace = format!("workers={};resources={}\n", worker_count, capacity.len());
    let mut now = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        // Complete all tasks ending now in ascending task ID.
        let mut ending: Vec<(u64, usize, usize)> = running
            .iter()
            .copied()
            .filter(|(finish, _, _)| *finish == now)
            .collect();
        ending.sort_by_key(|&(_, task, _)| task);
        if !ending.is_empty() {
            running.retain(|(finish, _, _)| *finish != now);
            for (finish, task_id, worker) in ending {
                for r in 0..used.len() {
                    used[r] -= u64::from(tasks[task_id].needs[r]);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    resource_text(&used)
                ));
                for &child in &children[task_id] {
                    unmet[child] -= 1;
                }
            }
        }

        // Repeatedly admit one task, then recompute eligibility and ranking.
        let mut admitted = false;
        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || unmet[id] != 0
                    || worker_free.iter().all(|&free| free > now)
                {
                    continue;
                }
                let fits = tasks[id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r]);
                if !fits {
                    continue;
                }
                let better = match best {
                    None => true,
                    Some(old) => {
                        tail[id] > tail[old]
                            || (tail[id] == tail[old]
                                && (tasks[id].priority > tasks[old].priority
                                    || (tasks[id].priority == tasks[old].priority && id < old)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }

            let Some(task_id) = best else { break };
            let start = now;
            let finish = start.checked_add(tasks[task_id].duration).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Finish,
            })?;

            let preferred = task_id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&w| worker_free[w] <= now)
                .expect("availability was checked");

            let free_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: task_id,
                stage: Stage::Turnaround,
            })?;

            started[task_id] = true;
            worker_free[worker] = free_at;
            for r in 0..used.len() {
                used[r] += u64::from(tasks[task_id].needs[r]);
            }
            runs[task_id] = Some(Run { start, finish, worker });
            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                now,
                task_id,
                worker,
                resource_text(&used)
            ));
            running.push((finish, task_id, worker));
            admitted = true;

            // A zero-duration task finishes immediately, before the next selection.
            if finish == now {
                running.retain(|&(_, id, _)| id != task_id);
                for r in 0..used.len() {
                    used[r] -= u64::from(tasks[task_id].needs[r]);
                }
                completed[task_id] = true;
                completed_count += 1;
                makespan = makespan.max(finish);
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    now,
                    task_id,
                    worker,
                    resource_text(&used)
                ));
                for &child in &children[task_id] {
                    unmet[child] -= 1;
                }
            }
        }

        if completed_count == n {
            break;
        }

        // If no more starts are possible now, advance to the next relevant event.
        let mut next: Option<u64> = None;
        for &(finish, _, _) in &running {
            if finish > now {
                next = Some(next.map_or(finish, |v| v.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                let has_worker_event = worker_free.iter().any(|&free| free > now);
                if !has_worker_event || worker_free.iter().any(|&free| free <= now) {
                    next = Some(next.map_or(tasks[id].release, |v| v.min(tasks[id].release)));
                }
            }
        }
        for &free in &worker_free {
            if free > now {
                next = Some(next.map_or(free, |v| v.min(free)));
            }
        }

        if let Some(t) = next {
            now = t;
        } else if !admitted && running.is_empty() {
            // With a valid acyclic graph, this can only be a future release.
            let future = (0..n)
                .filter(|&id| !started[id])
                .map(|id| tasks[id].release)
                .filter(|&release| release > now)
                .min();
            if let Some(t) = future {
                now = t;
            } else {
                break;
            }
        } else if running.is_empty() {
            break;
        }
    }

    let final_makespan = makespan;
    trace.push_str(&format!("makespan={}\n", final_makespan));
    let runs = runs
        .into_iter()
        .map(|r| r.expect("acyclic tasks are all scheduled"))
        .collect();
    Ok(Schedule { runs, makespan: final_makespan, trace })
}

pub fn schedule(tasks: &[Task], cooldowns: &[u64], capacity: &[u32]) -> Result<Schedule, Error> {
    let n = tasks.len();

    for (id, task) in tasks.iter().enumerate() {
        let mut seen = std::collections::BTreeSet::new();
        for (position, &dep) in task.deps.iter().enumerate() {
            if dep >= n {
                return Err(Error::BadDependency { task: id, position, value: dep });
            }
            if dep == id {
                return Err(Error::SelfDependency { task: id, position });
            }
            if !seen.insert(dep) {
                return Err(Error::DuplicateDependency { task: id, position, value: dep });
            }
        }
    }

    if !(1..=64).contains(&cooldowns.len()) {
        return Err(Error::Workers { count: cooldowns.len() });
    }

    for (id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (r, (&need, &cap)) in task.needs.iter().zip(capacity).enumerate() {
            if need > cap {
                return Err(Error::ResourceExceeded {
                    task: id,
                    resource: r,
                    need,
                    capacity: cap,
                });
            }
        }
    }

    let mut indegree: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut children = vec![Vec::new(); n];
    for (id, task) in tasks.iter().enumerate() {
        for &dep in &task.deps {
            children[dep].push(id);
        }
    }

    let mut ready = std::collections::BinaryHeap::new();
    for id in 0..n {
        if indegree[id] == 0 {
            ready.push(std::cmp::Reverse(id));
        }
    }
    let mut order = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(id)) = ready.pop() {
        order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                ready.push(std::cmp::Reverse(child));
            }
        }
    }
    if order.len() != n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tail = vec![0u64; n];
    for &id in order.iter().rev() {
        let child_tail = children[id].iter().map(|&child| tail[child]).max().unwrap_or(0);
        tail[id] = tasks[id]
            .duration
            .checked_add(child_tail)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let workers = cooldowns.len();
    let mut available_at = vec![0u64; workers];
    let mut started = vec![false; n];
    let mut remaining = tasks.iter().map(|task| task.deps.len()).collect::<Vec<_>>();
    let mut used = vec![0u64; capacity.len()];
    let mut running = Vec::<(u64, usize, usize)>::new(); // finish, task, worker
    let mut runs = vec![None; n];
    let mut trace = format!("workers={};resources={}\n", workers, capacity.len());
    let mut time = 0u64;
    let mut completed_count = 0usize;
    let mut makespan = 0u64;

    while completed_count < n {
        let mut finished_now: Vec<(usize, usize)> = running
            .iter()
            .filter(|&&(finish, _, _)| finish == time)
            .map(|&(_, task, worker)| (task, worker))
            .collect();
        finished_now.sort_unstable_by_key(|&(task, _)| task);

        for (task, worker) in finished_now {
            let index = running.iter().position(|&(_, id, _)| id == task).unwrap();
            running.swap_remove(index);
            for (r, &need) in tasks[task].needs.iter().enumerate() {
                used[r] -= u64::from(need);
            }
            completed_count += 1;
            makespan = makespan.max(time);
            trace.push_str(&format!(
                "{}|end|{}|{}|{}\n",
                time,
                task,
                worker,
                format_used(&used)
            ));
            for &child in &children[task] {
                remaining[child] -= 1;
            }
        }

        loop {
            let mut eligible = Vec::new();
            for id in 0..n {
                if started[id] || tasks[id].release > time || remaining[id] != 0 {
                    continue;
                }
                let fits = tasks[id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(r, &need)| u64::from(need) <= u64::from(capacity[r]) - used[r]);
                if fits {
                    eligible.push(id);
                }
            }
            eligible.sort_by(|&a, &b| {
                tail[b]
                    .cmp(&tail[a])
                    .then_with(|| tasks[b].priority.cmp(&tasks[a].priority))
                    .then_with(|| a.cmp(&b))
            });

            let Some(task) = eligible.first().copied() else {
                break;
            };

            let mut chosen_worker = None;
            for offset in 0..workers {
                let worker = (task % workers + offset) % workers;
                if available_at[worker] <= time {
                    chosen_worker = Some(worker);
                    break;
                }
            }
            let Some(worker) = chosen_worker else {
                break;
            };

            let finish = time.checked_add(tasks[task].duration).ok_or(Error::Overflow {
                task,
                stage: Stage::Finish,
            })?;
            let worker_ready = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task,
                stage: Stage::Turnaround,
            })?;

            started[task] = true;
            available_at[worker] = worker_ready;
            runs[task] = Some(Run { start: time, finish, worker });
            for (r, &need) in tasks[task].needs.iter().enumerate() {
                used[r] += u64::from(need);
            }
            trace.push_str(&format!(
                "{}|start|{}|{}|{}\n",
                time,
                task,
                worker,
                format_used(&used)
            ));

            if tasks[task].duration == 0 {
                for (r, &need) in tasks[task].needs.iter().enumerate() {
                    used[r] -= u64::from(need);
                }
                completed_count += 1;
                makespan = makespan.max(time);
                trace.push_str(&format!(
                    "{}|end|{}|{}|{}\n",
                    time,
                    task,
                    worker,
                    format_used(&used)
                ));
                for &child in &children[task] {
                    remaining[child] -= 1;
                }
            } else {
                running.push((finish, task, worker));
            }
        }

        if completed_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > time {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };
        for &(finish, _, _) in &running {
            consider(finish);
        }
        for id in 0..n {
            if !started[id] {
                consider(tasks[id].release);
            }
        }
        for &ready_time in &available_at {
            consider(ready_time);
        }

        match next_time {
            Some(next) => time = next,
            None => {
                return Err(Error::Cycle {
                    blocked: (0..n).filter(|&id| !started[id]).collect(),
                });
            }
        }
    }

    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

pub fn schedule(tasks: &[Task], cooldowns: &[u64], capacity: &[u32]) -> Result<Schedule, Error> {
    let n = tasks.len();
    for (i, task) in tasks.iter().enumerate() {
        for (p, &dep) in task.deps.iter().enumerate() {
            if dep >= n {
                return Err(Error::BadDependency { task: i, position: p, value: dep });
            }
            if dep == i {
                return Err(Error::SelfDependency { task: i, position: p });
            }
            if task.deps[..p].contains(&dep) {
                return Err(Error::DuplicateDependency { task: i, position: p, value: dep });
            }
        }
    }

    let workers = cooldowns.len();
    if !(1..=64).contains(&workers) {
        return Err(Error::Workers { count: workers });
    }

    for (i, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: i,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (r, &need) in task.needs.iter().enumerate() {
            if need > capacity[r] {
                return Err(Error::ResourceExceeded {
                    task: i,
                    resource: r,
                    need,
                    capacity: capacity[r],
                });
            }
        }
    }

    use std::collections::BinaryHeap;
    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (i, task) in tasks.iter().enumerate() {
        indegree[i] = task.deps.len();
        for &d in &task.deps {
            children[d].push(i);
        }
    }

    let mut ready = BinaryHeap::new();
    for i in 0..n {
        if indegree[i] == 0 {
            ready.push(std::cmp::Reverse(i));
        }
    }
    let mut topo = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(i)) = ready.pop() {
        topo.push(i);
        for &child in &children[i] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                ready.push(std::cmp::Reverse(child));
            }
        }
    }
    if topo.len() != n {
        let mut removed = vec![false; n];
        for &i in &topo {
            removed[i] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&i| !removed[i]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &i in topo.iter().rev() {
        let longest = children[i].iter().map(|&c| tails[c]).max().unwrap_or(0);
        tails[i] = tasks[i].duration.checked_add(longest).ok_or(Error::Overflow {
            task: i,
            stage: Stage::Tail,
        })?;
    }

    let mut unmet: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut started = vec![false; n];
    let mut finished = vec![false; n];
    let mut run_slots: Vec<Option<Run>> = vec![None; n];
    let mut worker_available = vec![0u64; workers];
    let mut used = vec![0u64; capacity.len()];
    let mut active: Vec<(u64, usize, usize)> = Vec::new();
    let mut now = 0u64;
    let mut done_count = 0usize;
    let mut makespan = 0u64;
    let mut trace = format!("workers={};resources={}\n", workers, capacity.len());

    while done_count < n {
        let mut completed_now: Vec<_> = active.iter().copied().filter(|x| x.0 == now).collect();
        completed_now.sort_by_key(|x| x.1);
        if !completed_now.is_empty() {
            active.retain(|x| x.0 != now);
            for (finish, id, worker) in completed_now {
                for r in 0..used.len() {
                    used[r] -= tasks[id].needs[r] as u64;
                }
                finished[id] = true;
                done_count += 1;
                makespan = makespan.max(finish);
                trace.push_str(&format!("{}|end|{}|{}|{}\n", now, id, worker, resource_text(&used)));
                for &child in &children[id] {
                    unmet[child] -= 1;
                }
            }
        }

        loop {
            let mut candidate: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > now || unmet[id] != 0 {
                    continue;
                }
                if !(0..workers).any(|w| worker_available[w] <= now) {
                    continue;
                }
                let fits = (0..capacity.len()).all(|r| {
                    (tasks[id].needs[r] as u64) <= (capacity[r] as u64 - used[r])
                });
                if !fits {
                    continue;
                }
                if candidate.map_or(true, |old| {
                    tails[id] > tails[old]
                        || (tails[id] == tails[old]
                            && (tasks[id].priority > tasks[old].priority
                                || (tasks[id].priority == tasks[old].priority && id < old)))
                }) {
                    candidate = Some(id);
                }
            }
            let Some(id) = candidate else { break };

            let finish = now.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let first = id % workers;
            let worker = (0..workers)
                .map(|k| (first + k) % workers)
                .find(|&w| worker_available[w] <= now)
                .unwrap();
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Turnaround,
            })?;

            started[id] = true;
            worker_available[worker] = available_at;
            for r in 0..used.len() {
                used[r] += tasks[id].needs[r] as u64;
            }
            run_slots[id] = Some(Run { start: now, finish, worker });
            trace.push_str(&format!("{}|start|{}|{}|{}\n", now, id, worker, resource_text(&used)));
            active.push((finish, id, worker));

            if finish == now {
                active.retain(|x| x.1 != id);
                for r in 0..used.len() {
                    used[r] -= tasks[id].needs[r] as u64;
                }
                finished[id] = true;
                done_count += 1;
                makespan = makespan.max(finish);
                trace.push_str(&format!("{}|end|{}|{}|{}\n", now, id, worker, resource_text(&used)));
                for &child in &children[id] {
                    unmet[child] -= 1;
                }
            }
        }

        if done_count == n {
            break;
        }

        let mut next = None;
        for &(finish, _, _) in &active {
            if finish > now {
                next = Some(next.map_or(finish, |v: u64| v.min(finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next = Some(next.map_or(tasks[id].release, |v: u64| v.min(tasks[id].release)));
            }
        }
        for &t in &worker_available {
            if t > now {
                next = Some(next.map_or(t, |v: u64| v.min(t)));
            }
        }
        if let Some(t) = next {
            now = t;
        } else {
            break;
        }
    }

    let runs = run_slots.into_iter().map(Option::unwrap).collect();
    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule { runs, makespan, trace })
}

    // Dependency validation is deliberately completed before every other check.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }

            }

                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
        }
    }

    if !(1..=64).contains(&worker_count) {
        return Err(Error::Workers { count: worker_count });
    }

        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });

            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }

    let mut children = vec![Vec::<usize>::new(); n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &dep in &task.deps {
            children[dep].push(task_id);
        }
    }

    let mut available = Vec::<usize>::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }

    let mut kahn = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(pos, _)| pos)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        kahn.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if kahn.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &id in &kahn {
            removed[id] = true;
        }
        for id in 0..n {
            if !removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut tails = vec![0u64; n];
    for &id in kahn.iter().rev() {
        let mut successor_tail = 0u64;
        for &child in &children[id] {
            successor_tail = successor_tail.max(tails[child]);
        }
        tails[id] = tasks[id]
            .duration
            .checked_add(successor_tail)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let mut remaining_deps: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut workers_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut used = vec![0u128; capacity.len()];
    let mut time = 0u64;
    let mut makespan = 0u64;

    let mut trace = String::new();
    trace.push_str("workers=");
    trace.push_str(&worker_count.to_string());
    trace.push_str(";resources=");
    trace.push_str(&capacity.len().to_string());
    trace.push('\n');

    loop {
        // Complete every task finishing now, in ascending task ID.
        let mut ending: Vec<usize> = running
            .iter()
            .enumerate()
            .filter_map(|(index, run)| if run.finish == time { Some(index) } else { None })
            .collect();
        ending.sort_by_key(|&index| running[index].task);
        for index in ending.into_iter().rev() {
            let run = running.remove(index);
            completed[run.task] = true;
            makespan = makespan.max(run.finish);
            for resource in 0..capacity.len() {
                used[resource] -= tasks[run.task].needs[resource] as u128;
            }
            append_event(&mut trace, time, "end", run.task, run.worker, &used);
        }

        let mut started_any = false;
        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > time {
                    continue;
                }
                if tasks[id].deps.iter().any(|&dep| !completed[dep]) {
                    continue;
                }
                let fits = (0..capacity.len()).all(|r| {
                    tasks[id].needs[r] as u128 <= capacity[r] as u128 - used[r]
                });
                if !fits {
                    continue;
                }
                let worker_exists = (0..worker_count).any(|worker| workers_available[worker] <= time);
                if !worker_exists {
                    continue;
                }
                let better = match best {
                    None => true,
                    Some(current) => {
                        tails[id] > tails[current]
                            || (tails[id] == tails[current]
                                && (tasks[id].priority > tasks[current].priority
                                    || (tasks[id].priority == tasks[current].priority && id < current)))
                    }
                };
                if better {
                    best = Some(id);
                }
            }

            let Some(id) = best else { break };
            let preferred = id % worker_count;
            let worker = (0..worker_count)
                .map(|offset| (preferred + offset) % worker_count)
                .find(|&worker| workers_available[worker] <= time)
                .unwrap();

            let finish = time.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let available = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Turnaround,
            })?;

            started[id] = true;
            for resource in 0..capacity.len() {
                used[resource] += tasks[id].needs[resource] as u128;
            }
            workers_available[worker] = available;
            runs[id] = Some(Run { start: time, finish, worker });
            running.push(Running { task: id, worker, finish, available });
            append_event(&mut trace, time, "start", id, worker, &used);
            started_any = true;

            if finish == time {
                let index = running.iter().position(|run| run.task == id).unwrap();
                let run = running.remove(index);
                completed[id] = true;
                makespan = makespan.max(time);
                for resource in 0..capacity.len() {
                    used[resource] -= tasks[id].needs[resource] as u128;
                }
                append_event(&mut trace, time, "end", id, worker, &used);
            }
        }

        if started.iter().all(|value| *value) && running.is_empty() {
            break;
        }

        let mut next = None::<u64>;
        for run in &running {
            if run.finish > time {
                next = Some(next.map_or(run.finish, |v| v.min(run.finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > time {
                next = Some(next.map_or(tasks[id].release, |v| v.min(tasks[id].release)));
            }
        }
        for &available in &workers_available {
            if available > time {
                next = Some(next.map_or(available, |v| v.min(available)));
            }
        }

        if let Some(next_time) = next {
            time = next_time;
        } else if !started_any {
            // A valid acyclic input always has a future event when unfinished work remains.
            break;
        }
    }

    let runs = runs.into_iter().map(Option::unwrap).collect::<Vec<_>>();
    trace.push_str("makespan=");
    trace.push_str(&makespan.to_string());
    trace.push('\n');

    Ok(Schedule { runs, makespan, trace })
}

pub struct Task {

    Turnaround,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency { task: usize, position: usize, value: usize },
    SelfDependency { task: usize, position: usize },
    DuplicateDependency { task: usize, position: usize, value: usize },
    Cycle { blocked: Vec<usize> },
    Workers { count: usize },
    Overflow { task: usize, stage: Stage },
    ResourceShape { task: usize, expected: usize, actual: usize },
    ResourceExceeded { task: usize, resource: usize, need: u32, capacity: u32 },
}

fn append_resources(trace: &mut String, used: &[u64]) {
    if used.is_empty() {
        trace.push('-');
    } else {
        for (i, value) in used.iter().enumerate() {
            if i > 0 {
                trace.push(',');
            }
            trace.push_str(&value.to_string());
        }
    }
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            } else if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            } else if task.deps[..position].iter().any(|&prior| prior == value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&cooldowns.len()) {
        return Err(Error::Workers { count: cooldowns.len() });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, &need) in task.needs.iter().enumerate() {
            if need > capacity[resource] {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: capacity[resource],
                });
            }
        }
    }

    let mut remaining = vec![0usize; n];
    let mut successors = vec![Vec::<usize>::new(); n];
    for (id, task) in tasks.iter().enumerate() {
        remaining[id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(id);
        }
    }

    // A binary heap as a max-heap of reverse IDs implements the min frontier.
    let mut heap = std::collections::BinaryHeap::new();
    for id in 0..n {
        if remaining[id] == 0 {
            heap.push(std::cmp::Reverse(id));
        }
    }
    let mut order = Vec::with_capacity(n);
    while let Some(std::cmp::Reverse(id)) = heap.pop() {
        order.push(id);
        for &child in &successors[id] {
            remaining[child] -= 1;
            if remaining[child] == 0 {
                heap.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != n {
        let mut done = vec![false; n];
        for &id in &order {
            done[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !done[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in order.iter().rev() {
        let max_successor = successors[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_successor)
            .ok_or(Error::Overflow { task: id, stage: Stage::Tail })?;
    }

    let workers = cooldowns.len();
    let mut worker_free = vec![0u64; workers];
    let mut started = vec![false; n];
    let mut done = vec![false; n];
    let mut active: Vec<(u64, usize, usize)> = Vec::new();
    let mut used = vec![0u64; capacity.len()];
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut trace = format!("workers={};resources={}\n", workers, capacity.len());
    let mut current = 0u64;
    let mut count_done = 0usize;
    let mut makespan = 0u64;

    while count_done < n {
        let mut due: Vec<(u64, usize, usize)> = active
            .iter()
            .copied()
            .filter(|(finish, _, _)| *finish == current)
            .collect();
        due.sort_by_key(|&(_, id, _)| id);

        for (_, id, worker) in due {
            active.retain(|&(_, active_id, _)| active_id != id);
            done[id] = true;
            count_done += 1;
            for r in 0..used.len() {
                used[r] -= tasks[id].needs[r] as u64;
            }
            trace.push_str(&format!("{}|end|{}|{}|", current, id, worker));
            append_resources(&mut trace, &used);
            trace.push('\n');
        }

        loop {
            let mut candidate: Option<usize> = None;
            for id in 0..n {
                if started[id] || tasks[id].release > current {
                    continue;
                }
                if tasks[id].deps.iter().any(|&p| !done[p]) {
                    continue;
                }
                let fits = (0..capacity.len())
                    .all(|r| tasks[id].needs[r] as u64 <= capacity[r] as u64 - used[r]);
                if !fits {
                    continue;
                }
                let better = candidate.is_none_or(|old| {
                    tails[id] > tails[old]
                        || (tails[id] == tails[old]
                            && (tasks[id].priority > tasks[old].priority
                                || (tasks[id].priority == tasks[old].priority && id < old)))
                });
                if better {
                    candidate = Some(id);
                }
            }

            let Some(id) = candidate else { break };
            let mut worker = None;
            for offset in 0..workers {
                let w = (id % workers + offset) % workers;
                if worker_free[w] <= current {
                    worker = Some(w);
                    break;
                }
            }
            let Some(worker) = worker else { break };

            let finish = current.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let free_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Turnaround,
            })?;

            started[id] = true;
            worker_free[worker] = free_at;
            runs[id] = Some(Run { start: current, finish, worker });
            for r in 0..used.len() {
                used[r] += tasks[id].needs[r] as u64;
            }
            trace.push_str(&format!("{}|start|{}|{}|", current, id, worker));
            append_resources(&mut trace, &used);
            trace.push('\n');

            if tasks[id].duration == 0 {
                done[id] = true;
                count_done += 1;
                for r in 0..used.len() {
                    used[r] -= tasks[id].needs[r] as u64;
                }
                trace.push_str(&format!("{}|end|{}|{}|", current, id, worker));
                append_resources(&mut trace, &used);
                trace.push('\n');
            } else {
                active.push((finish, id, worker));
            }
            makespan = makespan.max(finish);
        }

        if count_done == n {
            break;
        }

        let mut next: Option<u64> = None;
        let mut offer = |event: u64| {
            if event > current {
                next = Some(next.map_or(event, |old| old.min(event)));
            }
        };
        for &(finish, _, _) in &active {
            offer(finish);
        }
        for id in 0..n {
            if !started[id] {
                offer(tasks[id].release);
            }
        }
        for &free_at in &worker_free {
            offer(free_at);
        }
        if let Some(next_time) = next {
            current = next_time;
        } else {
            break;
        }
    }

    let runs = runs.into_iter().map(|r| r.unwrap()).collect();
    trace.push_str(&format!("makespan={}\n", makespan));
    Ok(Schedule { runs, makespan, trace })
}
