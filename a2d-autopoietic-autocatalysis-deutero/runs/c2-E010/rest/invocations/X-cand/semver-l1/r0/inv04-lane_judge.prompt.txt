pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_core_stop(b: u8) -> bool {
    matches!(b, b'.' | b'-' | b'+')
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return Err(if *pos == bytes.len() || is_core_stop(bytes[*pos]) {
            VersionError::MissingCore { at: *pos }
        } else {
            VersionError::InvalidChar { at: *pos }
        });
    }
    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut value = 0u64;
    for &d in digits {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((d - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    pos: &mut usize,
    stop_at_plus: bool,
    check_pre_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();

    loop {
        let start = *pos;
        while *pos < bytes.len()
            && bytes[*pos] != b'.'
            && !(stop_at_plus && bytes[*pos] == b'+')
        {
            let b = bytes[*pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: *pos });
            }
            *pos += 1;
        }

        if start == *pos {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        let id = &s[start..*pos];
        if check_pre_zero
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: start });
        }
        out.push(id.to_owned());

        if *pos == bytes.len() || (stop_at_plus && bytes[*pos] == b'+') {
            break;
        }
        // The only remaining delimiter is a dot.
        *pos += 1;
        if *pos == bytes.len() || (stop_at_plus && bytes[*pos] == b'+') {
            return Err(VersionError::EmptyIdentifier { at: *pos });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(if pos >= s.len() || is_core_stop(s.as_bytes()[pos]) {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;

    let minor = parse_core(s, &mut pos)?;
    if pos >= s.len() || s.as_bytes()[pos] != b'.' {
        return Err(if pos >= s.len() || is_core_stop(s.as_bytes()[pos]) {
            VersionError::MissingCore { at: pos }
        } else {
            VersionError::InvalidChar { at: pos }
        });
    }
    pos += 1;

    let patch = parse_core(s, &mut pos)?;
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < s.len() {
        match s.as_bytes()[pos] {
            b'-' => {
                pos += 1;
                pre = parse_identifiers(s, &mut pos, true, true)?;
                if pos < s.len() {
                    // A plus terminates pre-release and begins build metadata.
                    pos += 1;
                    build = parse_identifiers(s, &mut pos, false, false)?;
                }
            }
            b'+' => {
                pos += 1;
                build = parse_identifiers(s, &mut pos, false, false)?;
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    Ok(Version { major, minor, patch, pre, build })
}

fn all_digits(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_pre_id(a: &str, b: &str) -> Ordering {
    let an = all_digits(a);
    let bn = all_digits(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_pre_id(x, y);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = vec![Vec::new()];
    let mut pos = 0;

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_ws(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }
        let token_start = pos;
        while pos < bytes.len() && !is_range_ws(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;

        let mut operand_start = token_start;
        let op = match bytes[token_start] {
            b'=' => { operand_start += 1; Op::Eq }
            b'<' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2; Op::Le
            }
            b'<' => { operand_start += 1; Op::Lt }
            b'>' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                operand_start += 2; Op::Ge
            }
            b'>' => { operand_start += 1; Op::Gt }
            _ => Op::Eq,
        };

        if operand_start >= token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }
        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;
        arms[0].push(Comparator { op, version });
    }

    if arms[0].is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { arms })
}

        return false;

    r.arms.iter().any(|arm| {
        let admitted = v.pre.is_empty()
            || arm.iter().any(|c| {
                !c.version.pre.is_empty()
                    && c.version.major == v.major
                    && c.version.minor == v.minor
                    && c.version.patch == v.patch

            && arm.iter().all(|c| {
                let ord = compare_versions(v, &c.version);
                match c.op {
                    Op::Eq => ord == Ordering::Equal,
                    Op::Lt => ord == Ordering::Less,
                    Op::Le => ord != Ordering::Greater,
                    Op::Gt => ord == Ordering::Greater,
                    Op::Ge => ord != Ordering::Less,
                }
            })
    })
}

pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn missing_or_invalid(s: &str, at: usize) -> VersionError {
    match s.as_bytes().get(at) {
        None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at },
        _ => VersionError::InvalidChar { at },
    }
}

fn core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while bytes.get(*pos).is_some_and(u8::is_ascii_digit) {
        *pos += 1;
    }
    if start == *pos {
        return Err(missing_or_invalid(s, *pos));
    }
    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut value = 0u64;
    for digit in digits {
        value = value.checked_mul(10)
            .and_then(|n| n.checked_add((digit - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn identifiers(
    s: &str,
    pos: &mut usize,
    stop_plus: bool,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    loop {
        let start = *pos;
        while let Some(&b) = bytes.get(*pos) {
            if b == b'.' || (stop_plus && b == b'+') {
                break;
            }
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: *pos });
            }
            *pos += 1;
        }
        if start == *pos {
            return Err(VersionError::EmptyIdentifier { at: start });
        }
        let id = &s[start..*pos];
        if pre && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1 && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: start });
        }
        result.push(id.to_string());

        match bytes.get(*pos) {
            None => break,
            Some(b'+') if stop_plus => break,
            Some(b'.') => {
                *pos += 1;
                if *pos == bytes.len() || (stop_plus && bytes[*pos] == b'+') {
                    return Err(VersionError::EmptyIdentifier { at: *pos });
                }
            }
            _ => break,
        }
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = core(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return Err(missing_or_invalid(s, pos));
    }
    pos += 1;
    let minor = core(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return Err(missing_or_invalid(s, pos));
    }
    pos += 1;
    let patch = core(s, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    match s.as_bytes().get(pos) {
        None => {}
        Some(b'-') => {
            pos += 1;
            pre = identifiers(s, &mut pos, true, true)?;
            if s.as_bytes().get(pos) == Some(&b'+') {
                pos += 1;
                build = identifiers(s, &mut pos, false, false)?;
            }
        }
        Some(b'+') => {
            pos += 1;
            build = identifiers(s, &mut pos, false, false)?;
        }
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    Ok(Version { major, minor, patch, pre, build })
}

fn numeric(id: &str) -> bool {
    !id.is_empty() && id.bytes().all(|b| b.is_ascii_digit())
}

fn pre_cmp(a: &str, b: &str) -> Ordering {
    match (numeric(a), numeric(b)) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then(a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core_order = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core_order != Ordering::Equal {
        return core_order;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,

                if ord != Ordering::Equal {
                    return ord;
                }
            }

        }
    }
}

            VersionError::MissingCore { at: start }
        } else {
            VersionError::InvalidChar { at: start }
        });
    }
    if *p - start > 1 && b[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let value = s[start..*p]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })?;
    Ok(value)
}

fn dot(s: &str, p: &mut usize) -> Result<(), VersionError> {
    let b = s.as_bytes();
    let at = *p;
    if at < b.len() && b[at] == b'.' {
        *p += 1;
        Ok(())
    } else if at == b.len() || matches!(b[at], b'-' | b'+') {
        Err(VersionError::MissingCore { at })
    } else {
        Err(VersionError::InvalidChar { at })
    }
}

fn ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn ids(s: &str, p: &mut usize, stop_plus: bool, pre: bool) -> Result<Vec<String>, VersionError> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    loop {
        let start = *p;
        while *p < b.len() && b[*p] != b'.' && !(stop_plus && b[*p] == b'+') {
            if !ident_char(b[*p]) {
                return Err(VersionError::InvalidChar { at: *p });
            }
            *p += 1;
        }
        if start == *p {
            return Err(VersionError::EmptyIdentifier { at: start });
        }
        let value = &s[start..*p];
        if pre
            && value.len() > 1
            && value.as_bytes()[0] == b'0'
            && value.as_bytes().iter().all(|x| digit(*x))
        {
            return Err(VersionError::LeadingZero { at: start });
        }
        out.push(value.to_owned());
        if *p == b.len() || (stop_plus && b[*p] == b'+') {
            return Ok(out);
        }
        *p += 1;
    }
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut p = 0;
    let major = core(s, &mut p)?;
    dot(s, &mut p)?;
    let minor = core(s, &mut p)?;
    dot(s, &mut p)?;
    let patch = core(s, &mut p)?;

    let b = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if p < b.len() && b[p] == b'-' {
        p += 1;
        pre = ids(s, &mut p, true, true)?;
    }
    if p < b.len() && b[p] == b'+' {
        p += 1;
        build = ids(s, &mut p, false, false)?;
    } else if p < b.len() {
        return Err(VersionError::InvalidChar { at: p });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|&b| digit(b))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal {
        return core;
    }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(&b.pre) {
                let xn = numeric(x);
                let yn = numeric(y);
                let ord = match (xn, yn) {
                    (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
                    (true, false) => Ordering::Less,
                    (false, true) => Ordering::Greater,
                    (false, false) => x.as_bytes().cmp(y.as_bytes()),
                };
                if ord != Ordering::Equal {
                    return ord;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    clauses: Vec<(Op, Version)>,
    admit_cores: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut clauses = Vec::new();
    let mut admit_cores = Vec::new();
    let mut p = 0;
    while p < b.len() {
        while p < b.len() && matches!(b[p], b' ' | b'\t' | b'\r' | b'\n') {
            p += 1;
        }
        if p == b.len() {
            break;
        }
        let start = p;
        while p < b.len() && !matches!(b[p], b' ' | b'\t' | b'\r' | b'\n') {
            p += 1;
        }
        let end = p;
        let token = &b[start..end];
        let (op, n) = if token.starts_with(b">=") {
            (Op::Ge, 2)
        } else if token.starts_with(b"<=") {
            (Op::Le, 2)
        } else if token.starts_with(b">") {
            (Op::Gt, 1)
        } else if token.starts_with(b"<") {
            (Op::Lt, 1)
        } else if token.starts_with(b"=") {
            (Op::Eq, 1)
        } else {
            (Op::Eq, 0)
        };
        if n == token.len() || !digit(token[n]) {
            return Err(RangeError::Syntax { at: start });
        }
        let operand_at = start + n;
        let version = parse_version(&s[operand_at..end]).map_err(|error| RangeError::Version {
            at: operand_at,
            error,
        })?;
        if !version.pre.is_empty() {
            admit_cores.push((version.major, version.minor, version.patch));
        }
        clauses.push((op, version));
    }

    if clauses.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { clauses, admit_cores })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty() && !r.admit_cores.contains(&(v.major, v.minor, v.patch)) {
        return false;
    }
    r.clauses.iter().all(|(op, version)| {
        let ord = compare_versions(v, version);
        match op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    })
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn number(s: &str, start: usize, end: usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    if start == end {
        if matches!(bytes.get(start), None | Some(b'.' | b'-' | b'+')) {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    let mut value = 0u64;
    for i in start..end {
        let d = (bytes[i] - b'0') as u64;
        if i == start && end - start > 1 && d == 0 {
            return Err(VersionError::LeadingZero { at: start });
        }
        value = value.checked_mul(10)
            .and_then(|n| n.checked_add(d))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn identifiers(s: &str, start: usize, end: usize, pre: bool) -> Result<Vec<String>, VersionError> {
    let b = s.as_bytes();
    let mut result = Vec::new();
    let mut p = start;
    loop {
        let mut q = p;
        while q < end && b[q] != b'.' {
            if !(b[q].is_ascii_alphanumeric() || b[q] == b'-') {
                return Err(VersionError::InvalidChar { at: q });
            }
            q += 1;
        }
        if p == q {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
        let id = &s[p..q];
        if pre && id.len() > 1 && id.bytes().all(|x| x.is_ascii_digit()) && id.starts_with('0') {
            return Err(VersionError::LeadingZero { at: p });
        }
        result.push(id.to_owned());
        if q == end { break; }
        p = q + 1;
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() { return Err(VersionError::Empty); }
    let b = s.as_bytes();
    let mut p = 0;
    let mut fields = [0u64; 3];

    for field in 0..3 {
        let start = p;
        while p < b.len() && b[p].is_ascii_digit() { p += 1; }
        fields[field] = number(s, start, p)?;
        if field < 2 {
            if b.get(p) != Some(&b'.') {
                if matches!(b.get(p), None | Some(b'.' | b'-' | b'+')) {
                    return Err(VersionError::MissingCore { at: p });
                }
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
    }

    let (pre, build) = if p == b.len() {
        (Vec::new(), Vec::new())
    } else if b[p] == b'-' {
        let ps = p + 1;
        let mut pe = ps;
        while pe < b.len() && b[pe] != b'+' { pe += 1; }
        let pre = identifiers(s, ps, pe, true)?;
        let build = if pe < b.len() { identifiers(s, pe + 1, b.len(), false)? } else { Vec::new() };
        (pre, build)
    } else if b[p] == b'+' {
        (Vec::new(), identifiers(s, p + 1, b.len(), false)?)
    } else {
        return Err(VersionError::InvalidChar { at: p });
    };

    Ok(Version { major: fields[0], minor: fields[1], patch: fields[2], pre, build })
}

fn numeric(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|x| x.is_ascii_digit())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    let core = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core != Ordering::Equal { return core; }
    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.pre.iter().zip(b.pre.iter()) {
                let xn = numeric(x);
                let yn = numeric(y);
                let c = match (xn, yn) {
                    (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
                    (true, false) => Ordering::Less,
                    (false, true) => Ordering::Greater,
                    _ => x.as_bytes().cmp(y.as_bytes()),
                };
                if c != Ordering::Equal { return c; }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<(OrderingOp, Version)>,
}

#[derive(Clone, Copy, Debug)]
enum OrderingOp { Eq, Lt, Le, Gt, Ge }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut p = 0;
    let mut comps = Vec::new();

    while p < b.len() {
        while p < b.len() && matches!(b[p], b' ' | b'\t' | b'\r' | b'\n') { p += 1; }
        if p == b.len() { break; }
        let ts = p;
        while p < b.len() && !matches!(b[p], b' ' | b'\t' | b'\r' | b'\n') { p += 1; }
        let te = p;
        let t = &s[ts..te];

        let (op, n) = match t.as_bytes() {
            [b'>', b'=', ..] => (OrderingOp::Ge, 2),
            [b'<', b'=', ..] => (OrderingOp::Le, 2),
            [b'>', ..] => (OrderingOp::Gt, 1),
            [b'<', ..] => (OrderingOp::Lt, 1),
            [b'=', ..] => (OrderingOp::Eq, 1),
            _ => (OrderingOp::Eq, 0),
        };

        if ts + n == te || !b[ts + n].is_ascii_digit() {
            return Err(RangeError::Syntax { at: ts });
        }
        let operand = &s[ts + n..te];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: ts + n,
            error,
        })?;
        comps.push((op, version));
    }

    if comps.is_empty() { return Err(RangeError::Empty { at: 0 }); }
    Ok(Range { comparators: comps })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let admits_pre = r.comparators.iter().any(|(_, operand)| {
        !operand.pre.is_empty()
            && (operand.major, operand.minor, operand.patch) == (v.major, v.minor, v.patch)
    });
    if !v.pre.is_empty() && !admits_pre { return false; }

    r.comparators.iter().all(|(op, operand)| {
        let c = compare_versions(v, operand);
        match op {
            OrderingOp::Eq => c == Ordering::Equal,
            OrderingOp::Lt => c == Ordering::Less,
            OrderingOp::Le => c != Ordering::Greater,
            OrderingOp::Gt => c == Ordering::Greater,
            OrderingOp::Ge => c != Ordering::Less,
        }
    })
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }

    if *pos == start {
        if start == bytes.len()
            || matches!(bytes[start], b'.' | b'-' | b'+')
        {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &b in digits {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn require_dot(s: &str, pos: &mut usize) -> Result<(), VersionError> {
    let bytes = s.as_bytes();
    let at = *pos;
    if at < bytes.len() && bytes[at] == b'.' {
        *pos += 1;
        Ok(())
    } else if at == bytes.len() || matches!(bytes[at], b'-' | b'+') {
        Err(VersionError::MissingCore { at })
    } else {
        Err(VersionError::InvalidChar { at })
    }
}

fn valid_ident_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    pos: &mut usize,
    end_at_plus: bool,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();

    loop {
        let start = *pos;
        while *pos < bytes.len()
            && bytes[*pos] != b'.'
            && !(end_at_plus && bytes[*pos] == b'+')
        {
            if !valid_ident_char(bytes[*pos]) {
                return Err(VersionError::InvalidChar { at: *pos });
            }
            *pos += 1;
        }

        if *pos == start {
            return Err(VersionError::EmptyIdentifier { at: start });
        }

        let ident = &s[start..*pos];
        if prerelease
            && ident.len() > 1
            && ident.as_bytes().iter().all(|b| is_digit(*b))
            && ident.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: start });
        }
        result.push(ident.to_owned());

        if *pos == bytes.len() || (end_at_plus && bytes[*pos] == b'+') {
            break;
        }
        // The only remaining delimiter is a dot.
        *pos += 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    require_dot(s, &mut pos)?;
    let minor = parse_core(s, &mut pos)?;
    require_dot(s, &mut pos)?;
    let patch = parse_core(s, &mut pos)?;

    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < bytes.len() && bytes[pos] == b'-' {
        pos += 1;
        pre = parse_identifiers(s, &mut pos, true, true)?;
    }

    if pos < bytes.len() && bytes[pos] == b'+' {
        pos += 1;
        build = parse_identifiers(s, &mut pos, false, false)?;
    } else if pos < bytes.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|b| is_digit(*b))
}

fn compare_numeric_ident(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (ai, bi) in a.pre.iter().zip(&b.pre) {
                    let an = is_numeric(ai);
                    let bn = is_numeric(bi);
                    let ord = match (an, bn) {
                        (true, true) => compare_numeric_ident(ai, bi),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => ai.as_bytes().cmp(bi.as_bytes()),
                    };
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
    admits: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut i = 0;
    let mut comparators = Vec::new();
    let mut admits = Vec::new();
    let mut saw_token = false;

    while i < bytes.len() {
        while i < bytes.len() && matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        saw_token = true;
        let token_start = i;
        while i < bytes.len() && !matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n') {
            i += 1;
        }
        let token_end = i;
        let token = &s[token_start..token_end];
        let tb = token.as_bytes();

        let (op, operand_offset) = if tb.starts_with(b">=") {
            (Op::Ge, 2)
        } else if tb.starts_with(b"<=") {
            (Op::Le, 2)
        } else if tb.starts_with(b">") {
            (Op::Gt, 1)
        } else if tb.starts_with(b"<") {
            (Op::Lt, 1)
        } else if tb.starts_with(b"=") {
            (Op::Eq, 1)
        } else {
            (Op::Eq, 0)
        };

        if operand_offset >= tb.len() || !is_digit(tb[operand_offset]) {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand_start = token_start + operand_offset;
        let version = parse_version(&s[operand_start..token_end]).map_err(|error| {
            RangeError::Version {
                at: operand_start,
                error,
            }
        })?;

        if !version.pre.is_empty() {
            admits.push((version.major, version.minor, version.patch));
        }
        comparators.push(Comparator { op, version });
    }

    if !saw_token {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        comparators,
        admits,
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    if !v.pre.is_empty()
        && !r.admits.contains(&(v.major, v.minor, v.patch))
    {
        return false;
    }

    r.comparators.iter().all(|c| {
        let ord = compare_versions(v, &c.version);
        match c.op {
            Op::Eq => ord == Ordering::Equal,
            Op::Lt => ord == Ordering::Less,
            Op::Le => ord != Ordering::Greater,
            Op::Gt => ord == Ordering::Greater,
            Op::Ge => ord != Ordering::Less,
        }
    })
}
