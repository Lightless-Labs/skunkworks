pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Expression<'a> {
    start: usize,
    end: usize,
    operator: Option<u8>,
    names: Vec<&'a str>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_byte(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn validate_expression<'a>(
    template: &'a str,
    open: usize,
    close: usize,
) -> Result<Expression<'a>, Error> {
    let bytes = template.as_bytes();

    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut content_start = open + 1;
    let operator = if content_start < close
        && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    {
        let op = bytes[content_start];
        content_start += 1;
        if op != b'+' && op != b'#' {
            return Err(Error::UnsupportedOperator { pos: open + 1 });
        }
        Some(op)
    } else {
        None
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = content_start;

    loop {
        let spec_end = (spec_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let spec = &template[spec_start..spec_end];
        let spec_bytes = spec.as_bytes();
        let mut invalid = None;

        if !is_name_start(spec_bytes[0]) {
            invalid = Some(0);
        } else {
            for (i, &byte) in spec_bytes.iter().enumerate().skip(1) {
                if !is_name_byte(byte) {
                    invalid = Some(i);
                    break;
                }
            }
        }

        if let Some(offset) = invalid {
            return Err(Error::InvalidName {
                pos: spec_start + offset,
            });
        }

        if seen.iter().any(|&name| name == spec) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }

        seen.push(spec);
        names.push(spec);

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        names,
    })
}

fn encode(value: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut output = String::with_capacity(bytes.len());

    let is_reserved = |byte: u8| {
        matches!(
            byte,
            b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' |
            b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
        )
    };

    let mut i = 0;
    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            output.push('%');
            output.push((bytes[i + 1] as char).to_ascii_uppercase());
            output.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        if byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || (reserved && is_reserved(byte))
        {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }

    output
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|&&(binding, _)| binding == name)
        .map(|&(_, value)| value)
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let close = match (i + 1..bytes.len()).find(|&j| bytes[j] == b'}') {
                    Some(close) => close,
                    None => return Err(Error::Unclosed { pos: i }),
                };
                let expression = validate_expression(template, i, close)?;
                i = expression.end;
                expressions.push(expression);
            }
            _ => i += 1,
        }
    }

    let mut output = String::with_capacity(template.len());
    let mut literal_start = 0;

    for expression in expressions {
        output.push_str(&template[literal_start..expression.start]);

        let reserved = matches!(expression.operator, Some(b'+' | b'#'));
        let mut fields = Vec::new();

        for name in expression.names {
            if let Some(value) = lookup(vars, name) {
                fields.push(encode(value, reserved));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                output.push('#');
            }
            for (index, field) in fields.iter().enumerate() {
                if index != 0 {
                    output.push(',');
                }
                output.push_str(field);
            }
        }

        literal_start = expression.end;
    }

    output.push_str(&template[literal_start..]);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<u8>,
    names: Vec<(usize, usize)>,
}

fn valid_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn valid_continue(byte: u8) -> bool {
    valid_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(
    bytes: &[u8],
    open: usize,
    close: usize,
) -> Result<Expression, Error> {
    if let Some(pos) = (open + 1..close).find(|&pos| bytes[pos] == b'{') {
        return Err(Error::NestedOpen { pos });
    }

    let mut cursor = open + 1;
    let operator = if cursor < close
        && matches!(bytes[cursor], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    {
        let byte = bytes[cursor];
        cursor += 1;
        if !matches!(byte, b'+' | b'#') {
            return Err(Error::UnsupportedOperator { pos: cursor - 1 });
        }
        Some(byte)
    } else {
        None
    };

    if cursor == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<(usize, usize)> = Vec::new();
    let mut start = cursor;

    loop {
        let end = (start..close)
            .find(|&pos| bytes[pos] == b',')
            .unwrap_or(close);

        if end == start {
            return Err(Error::EmptyVariable { pos: start });
        }

        let mut pos = start;
        if !valid_start(bytes[pos]) {
            return Err(Error::InvalidName { pos });
        }
        pos += 1;
        while pos < end {
            if !valid_continue(bytes[pos]) {
                return Err(Error::InvalidName { pos });
            }
            pos += 1;
        }

        if seen
            .iter()
            .any(|&(old_start, old_end)| bytes[old_start..old_end] == bytes[start..end])
        {
            return Err(Error::DuplicateVariable { pos: start });
        }

        seen.push((start, end));
        names.push((start, end));

        if end == close {
            break;
        }
        start = end + 1;
    }

    Ok(Expression {
        start: open,
        end: close + 1,
        operator,
        names,
    })
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::with_capacity(bytes.len());

    for i in 0..bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            result.push('%');
            result.push((bytes[i + 1] as char).to_ascii_uppercase());
            result.push((bytes[i + 2] as char).to_ascii_uppercase());
            // The following two bytes are handled by skipping their loop iterations.
            continue;
        }

        if byte == b'%' && i > 0
            && bytes[i - 1] == b'%'
            && i + 1 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
        {
            // Already emitted as part of a triplet.
            continue;
        }
        if byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || (allow_reserved
                && matches!(
                    byte,
                    b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                        | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
                ))
        {
            result.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 15) as usize] as char);
        }
    }

    result
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        if bytes[pos] == b'}' {
            return Err(Error::UnexpectedClose { pos });
        }
        if bytes[pos] != b'{' {
            pos += 1;
            continue;
        }

        let open = pos;
        let close = match (open + 1..bytes.len()).find(|&i| bytes[i] == b'}') {
            Some(close) => close,
            None => return Err(Error::Unclosed { pos: open }),
        };

        expressions.push(parse_expression(bytes, open, close)?);
        pos = close + 1;
    }

    let mut result = String::with_capacity(template.len());
    let mut literal = 0;

    for expression in expressions {
        result.push_str(&template[literal..expression.start]);

        let mut has_field = false;
        let reserved = matches!(expression.operator, Some(b'+' | b'#'));
        if expression.operator == Some(b'#') {
            // Emit the prefix only after confirming a binding contributes a field.
            let any_present = expression.names.iter().any(|&(start, end)| {
                let name = &template[start..end];
                vars.iter().rev().any(|&(key, _)| key == name)
            });
            if any_present {
                result.push('#');
            }
        }

        for (index, &(start, end)) in expression.names.iter().enumerate() {
            let name = &template[start..end];
            if let Some((_, value)) = vars.iter().rev().find(|&&(key, _)| key == name) {
                if has_field || (index > 0 && expression.names[..index].iter().any(|&(a, b)| {
                    let earlier = &template[a..b];
                    vars.iter().rev().any(|&&(key, _)| key == earlier)
                })) {
                    result.push(',');
                }
                result.push_str(&encode_value(value, reserved));
                has_field = true;
            }
        }

        literal = expression.end;
    }

    result.push_str(&template[literal..]);
    Ok(result)
}
