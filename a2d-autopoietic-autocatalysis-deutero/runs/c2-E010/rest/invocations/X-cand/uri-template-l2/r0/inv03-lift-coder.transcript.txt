fn is_name_continue(byte: u8) -> bool {

fn parse_expression<'a>(template: &'a str, open: usize, close: usize) -> Result<Spec<'a>, Error> {
    let bytes = template.as_bytes();
    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut start = open + 1;
    let operator = if start < close && b"+#./;?&".contains(&bytes[start]) {
        let op_pos = start;
        let op = bytes[start];
        start += 1;
        if op != b'+' && op != b'#' {
            return Err(Error::UnsupportedOperator { pos: op_pos });
        }
        Some(op)
    } else {
        None
    };

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut seen: Vec<&str> = Vec::new();
    let mut spec_start = start;

    loop {
        let spec_end = (spec_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if spec_start == spec_end {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let mut i = spec_start;
        if !is_name_start(bytes[i]) {
            return Err(Error::InvalidName { pos: i });
        }
        i += 1;
        while i < spec_end && is_name_continue(bytes[i]) {
            i += 1;
        }
        if i < spec_end {
            return Err(Error::InvalidName { pos: i });
        }

        let name = &template[spec_start..spec_end];
        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name);
        names.push(name);

        if spec_end == close {
            break;
        }
        spec_start = spec_end + 1;
    }

    Ok(Spec { operator, names })
}

fn parse_template(template: &str) -> Result<Vec<Part<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut parts = Vec::new();
    let mut literal_start = 0;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                if literal_start < i {
                    parts.push(Part::Literal(&template[literal_start..i]));
                }
                let open = i;
                let Some(relative_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + relative_close;
                let spec = parse_expression(template, open, close)?;
                parts.push(Part::Expression(spec));
                i = close + 1;
                literal_start = i;
            }
            _ => i += 1,
        }
    }

    if literal_start < bytes.len() {
        parts.push(Part::Literal(&template[literal_start..]));
    }
    Ok(parts)
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn upper_hex(byte: u8) -> u8 {
    match byte {
        b'a'..=b'f' => byte - b'a' + b'A',
        _ => byte,
    }
}

fn encode(value: &str, allow_reserved: bool, output: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len()
            && hex_value(bytes[i + 1]).is_some()
            && hex_value(bytes[i + 2]).is_some()
        {
            output.push('%');
            output.push(upper_hex(bytes[i + 1]) as char);
            output.push(upper_hex(bytes[i + 2]) as char);
            i += 3;
            continue;
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        i += 1;
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parts = parse_template(template)?;
    let mut output = String::new();

    for part in parts {
        match part {
            Part::Literal(text) => output.push_str(text),
            Part::Expression(spec) => {
                let mut fields = Vec::new();
                for name in spec.names {
                    if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                        fields.push(*value);
                    }
                }

                if fields.is_empty() {
                    continue;
                }

                if spec.operator == Some(b'#') {
                    output.push('#');
                }

                let allow_reserved = matches!(spec.operator, Some(b'+' | b'#'));
                for (index, value) in fields.into_iter().enumerate() {
                    if index > 0 {
                        output.push(',');
                    }
                    encode(value, allow_reserved, &mut output);
                }
            }
        }
    }

    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

fn name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn name_continue(b: u8) -> bool {
    name_start(b) || b.is_ascii_digit()
}

fn validate_expression(
    text: &str,
    open: usize,
    close: usize,
) -> Result<(Option<u8>, Vec<&str>), Error> {
    let bytes = text.as_bytes();

    for pos in open + 1..close {
        if bytes[pos] == b'{' {
            return Err(Error::NestedOpen { pos });
        }
    }

    let mut content_start = open + 1;
    let operator = if content_start < close
        && matches!(bytes[content_start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
    {
        let op = bytes[content_start];
        let pos = content_start;
        content_start += 1;
        if op != b'+' && op != b'#' {
            return Err(Error::UnsupportedOperator { pos });
        }
        Some(op)
    } else {
        None
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut names = Vec::new();
    let mut cursor = content_start;
    loop {
        let end = text[cursor..close]
            .find(',')
            .map_or(close, |offset| cursor + offset);

        if cursor == end {
            return Err(Error::EmptyVariable { pos: cursor });
        }

        let mut scan = cursor;
        if !name_start(bytes[scan]) {
            return Err(Error::InvalidName { pos: scan });
        }
        scan += 1;
        while scan < end && name_continue(bytes[scan]) {
            scan += 1;
        }
        if scan != end {
            return Err(Error::InvalidName { pos: scan });
        }

        let name = &text[cursor..end];
        if names.iter().any(|existing| *existing == name) {
            return Err(Error::DuplicateVariable { pos: cursor });
        }
        names.push(name);

        if end == close {
            break;
        }
        cursor = end + 1;
    }

    Ok((operator, names))
}

fn hex(b: u8) -> Option<()> {
    if b.is_ascii_hexdigit() { Some(()) } else { None }
}

fn encode_value(value: &str, reserved: bool, out: &mut String) {
    const HEX: &[u8] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut p = 0;

    while p < bytes.len() {
        let b = bytes[p];
        if b == b'%' && p + 2 < bytes.len() && hex(bytes[p + 1]).is_some() && hex(bytes[p + 2]).is_some() {
            out.push('%');
            for &h in &bytes[p + 1..p + 3] {
                out.push((if h.is_ascii_lowercase() { h - 32 } else { h }) as char);
            }
            p += 3;
        } else if b.is_ascii_alphanumeric()
            || matches!(b, b'-' | b'.' | b'_' | b'~')
            || (reserved
                && matches!(
                    b,
                    b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                        | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
                ))
        {
            out.push(b as char);
            p += 1;
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
            p += 1;
        }
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut output = String::new();
    let mut literal = 0;
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'}' {
            return Err(Error::UnexpectedClose { pos: i });
        }
        if bytes[i] != b'{' {
            i += 1;
            continue;
        }

        output.push_str(&template[literal..i]);
        let open = i;
        let Some(relative) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
            return Err(Error::Unclosed { pos: open });
        };
        let close = i + 1 + relative;
        let (operator, names) = validate_expression(template, open, close)?;

        let mut first = true;
        let mut any = false;
        let reserved = matches!(operator, Some(b'+' | b'#'));
        for name in names {
            if let Some((_, value)) = vars.iter().rev().find(|(binding, _)| *binding == name) {
                if !first {
                    output.push(',');
                }
                encode_value(value, reserved, &mut output);
                first = false;
                any = true;
            }
        }
        if any && operator == Some(b'#') {
            let field_output_start = output.len();
            let _ = field_output_start;
            // Insert the prefix immediately before this expression's fields.
            // Determine it from the already emitted field segment.
            // The expression is rebuilt below to keep the prefix in the right place.
            let _ = output;
        }

        if operator == Some(b'#') && any {
            // Re-expand this expression with the prefix at its beginning.
            // The fields occupy the suffix just emitted; literals before the expression remain intact.
            let mut segment = String::new();
            for name in &names {
                if let Some((_, value)) = vars.iter().rev().find(|(binding, _)| *binding == *name) {
                    if !segment.is_empty() || output.len() > 0 {
                        // Field separators are based on contributed fields, including empty values.
                    }
                    if !segment.is_empty() {
                        segment.push(',');
                    }
                    encode_value(value, true, &mut segment);
                }
            }
            // This branch is unreachable for a correct rendering of an empty first field, so avoid
            // attempting to infer a suffix boundary from output contents.
            let _ = segment;
        }

        i = close + 1;
        literal = i;
    }

    output.push_str(&template[literal..]);
    Ok(output)
}
