fn number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    range: Option<(u32, u32)>,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }

    let mut n = 0u32;
    for &byte in bytes.get(start..end).unwrap_or_default() {
        n = n
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }

    if let Some((lo, hi)) = range {
        if !(lo..=hi).contains(&n) {
            return Err(ParseError::OutOfRange { field, offset: start });
        }
    }
    Ok(n)
}

fn digits(bytes: &[u8]) -> bool {
    bytes.len() > 0 && bytes.iter().all(|b| b.is_ascii_digit())

fn apply_item(
    bytes: &[u8],
    base_offset: usize,
    field: Field,
    bits: &mut u64,
) -> Result<(), ParseError> {
    if bytes.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: base_offset,
        });
    }

    let mut slash = None;
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'/' {
            if slash.is_some() {
                return Err(syntax(field, base_offset));
            }
            slash = Some(i);
        }
    }

    let base_end = slash.unwrap_or(bytes.len());
    let base = &bytes[..base_end];

    let mut dash = None;
    for (i, &b) in base.iter().enumerate() {
        if b == b'-' {
            if dash.is_some() {
                return Err(syntax(field, base_offset));
            }
            dash = Some(i);
        }
    }

    enum Kind {
        Star,
        Single(usize, usize),
        Range(usize, usize, usize, usize),
    }

    let kind = if base == b"*" {
        Kind::Star
    } else if let Some(d) = dash {
        if !digits(&base[..d]) || !digits(&base[d + 1..]) {
            return Err(syntax(field, base_offset));
        }
        Kind::Range(0, d, d + 1, base.len())
    } else if digits(base) {
        Kind::Single(0, base.len())
    } else {
        return Err(syntax(field, base_offset));
    };

    if let Some(s) = slash {
        if !digits(&bytes[s + 1..]) {
            return Err(syntax(field, base_offset));
        }
    }

    let (min, max) = field_info(field);
    let (start, end) = match kind {
        Kind::Star => (min, max),
        Kind::Single(a, b) => {
            let v = number(base, a, b, field, Some((min, max)))?;
            (v, if slash.is_some() { max } else { v })
        }
        Kind::Range(a, b, c, d) => {
            let lo = number(base, a, b, field, Some((min, max)))?;
            let hi = number(base, c, d, field, Some((min, max)))?;
            if lo > hi {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: base_offset + c,
                });
            }
            (lo, hi)
        }
    };

    let step = if let Some(s) = slash {
        let n = number(bytes, s + 1, bytes.len(), field, None)?;
        if n == 0 || n > max - min + 1 {
            return Err(ParseError::BadStep {
                field,
                offset: base_offset + s + 1,
            });
        }
        n
    } else {
        1
    };

    let mut v = start;
    while v <= end {
        *bits |= 1u64 << v;
        v += step;
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<(u64, bool), ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit() || b == b'*' || b == b'-' || b == b'/' || b == b',') {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let mut bits = 0u64;
    let mut start = 0usize;
    for end in 0..=bytes.len() {
        if end == bytes.len() || bytes[end] == b',' {
            apply_item(&bytes[start..end], start, field, &mut bits)?;
            start = end + 1;
        }
    }

    Ok((bits, text.as_bytes() == b"*"))
}

        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,

        let (bits, _) = parse_field(fields[i], field_kinds[i])?;
        selected[i] = bits;
    }

    Ok(Cron {

                return Err(ParseError::InvalidSyntax { field, offset: item_start });

                start == end || !item_bytes[start..end].iter().all(u8::is_ascii_digit)
            })
        {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let mut start_value = minimum as u32;
        let mut end_value = maximum as u32;

        if let Some((start, end)) = start_span {
            start_value = parse_number(bytes, item_start + start, item_start + end, field)?;
            if start_value < minimum as u32 || start_value > maximum as u32 {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + start,
                });
            }
        }

        if let Some((start, end)) = end_span {
            end_value = parse_number(bytes, item_start + start, item_start + end, field)?;
            if end_value < minimum as u32 || end_value > maximum as u32 {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + start,
                });
            }
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + start,
                });
            }
        } else if kind == 1 && step_span.is_none() {
            end_value = start_value;
        }

        let step = if let Some((start, end)) = step_span {
            let step = parse_number(bytes, item_start + start, item_start + end, field)?;
            if step == 0 || step > width as u32 {
                return Err(ParseError::BadStep {
                    field,
                    offset: item_start + start,
                });
            }
            step
        } else {
            1
        };

        let mut value = start_value;
        while value <= end_value {
            selected.insert(value as u8);
            match value.checked_add(step) {
                Some(next) if next > value => value = next,
                _ => break,
            }
        }

        item_start += item_bytes.len() + 1;
    }

    Ok(selected.into_iter().collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];

    for (text, field) in parts.iter().zip(fields) {
        selected.push(parse_field(text, field)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_stars: [parts[2] == "*", parts[4] == "*"],
    })
}

    max: u8,

const FIELDS: [Field; 5] = [

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn error(field: Field, offset: usize, kind: fn(Field, usize) -> ParseError) -> ParseError {
    kind(field, offset)
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }
    let mut value = 0u32;
    for &b in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })?;
    }
    Ok(value)
}

fn select_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();

    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let d = domain(field);
    let width = (d.max - d.min + 1) as u32;
    let mut selected = vec![false; d.max as usize + 1];
    let mut item_start = 0;

    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();

        if slash_positions.len() > 1 {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }

        let (base_end, step_bounds) = if let Some(&slash) = slash_positions.first() {
            if slash == 0 || slash + 1 == item.len() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            (slash, Some((slash + 1, item.len())))
        } else {
            (item.len(), None)
        };

        let base = &item[..base_end];
        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        let base_kind = if is_star {
            if !hyphens.is_empty() {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            0
        } else if hyphens.is_empty() {
            if base.is_empty() || !base.iter().all(u8::is_ascii_digit) {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            1
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            if h == 0
                || h + 1 == base.len()
                || !base[..h].iter().all(u8::is_ascii_digit)
                || !base[h + 1..].iter().all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
            2
        } else {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        };

        let (start, end) = match base_kind {
            0 => (d.min as u32, d.max as u32),
            1 => {
                let n = parse_number(bytes, item_start, item_start + base.len(), field)?;
                if n < d.min as u32 || n > d.max as u32 {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: item_start,
                    });
                }
                let end = if step_bounds.is_some() {
                    d.max as u32
                } else {
                    n
                };
                (n, end)
            }
            _ => {
                let h = hyphens[0];
                let left_start = item_start;
                let right_start = item_start + h + 1;
                let a = parse_number(bytes, left_start, item_start + h, field)?;
                if a < d.min as u32 || a > d.max as u32 {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: left_start,
                    });
                }
                let b = parse_number(bytes, right_start, item_end.min(right_start + (base.len() - h - 1)), field)?;
                if b < d.min as u32 || b > d.max as u32 {
                    return Err(ParseError::OutOfRange {
                        field,
                        offset: right_start,
                    });
                }
                if a > b {
                    return Err(ParseError::ReversedRange {
                        field,
                        offset: right_start,
                    });
                }
                (a, b)
            }
        };

        let step = if let Some((step_start, step_end)) = step_bounds {
            let n = parse_number(bytes, item_start + step_start, item_start + step_end, field)?;
            if n < 1 || n > width {
                return Err(ParseError::BadStep {
                    field,
                    offset: item_start + step_start,
                });
            }
            n
        } else {
            1
        };

        let mut n = start;
        while n <= end {
            selected[n as usize] = true;
            match n.checked_add(step) {
                Some(next) => n = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(n, &yes)| yes.then_some(n as u8))
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });

    for (i, (&field, part)) in FIELDS.iter().zip(parts.iter()).enumerate() {

    Ok(Cron {

        day_stars: [parts[2].as_bytes() == b"*", parts[4].as_bytes() == b"*"],
    })
}

fn read_num(
    b: &[u8],
    s: usize,
    e: usize,
    f: Field,
    bounds: Bounds,
    step: bool,
) -> Result<u32, ParseError> {
    if e - s > 1 && b[s] == b'0' {
        return Err(ParseError::LeadingZero { field: f, offset: s });
    }
    let mut n = 0u32;
    for &c in &b[s..e] {
        n = n.checked_mul(10)
            .and_then(|v| v.checked_add((c - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field: f, offset: s })?;
    }
    if step {
        if n == 0 || n > bounds.width {
            return Err(ParseError::BadStep { field: f, offset: s });
        }
    } else if n < bounds.lo || n > bounds.hi {
        return Err(ParseError::OutOfRange { field: f, offset: s });
    }
    Ok(n)
}

fn parse_one(
    b: &[u8],
    s: usize,
    e: usize,
    f: Field,
    bounds: Bounds,
    out: &mut [bool; 60],
) -> Result<(), ParseError> {
    if s == e {
        return Err(ParseError::EmptyItem { field: f, offset: s });
    }

    let mut slash = None;
    let mut hyphen = None;
    for i in s..e {
        match b[i] {
            b'/' => {
                if slash.is_some() {
                    return Err(ParseError::InvalidSyntax { field: f, offset: s });
                }
                slash = Some(i);
            }
            b'-' if slash.is_none() => {
                if hyphen.is_some() {
                    return Err(ParseError::InvalidSyntax { field: f, offset: s });
                }
                hyphen = Some(i);
            }
            _ => {}
        }
    }

    let base_end = slash.unwrap_or(e);
    let step = if let Some(k) = slash {
        if k == s || k + 1 == e {
            return Err(ParseError::InvalidSyntax { field: f, offset: s });
        }
        if !b[k + 1..e].iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax { field: f, offset: s });
        }
        Some(read_num(b, k + 1, e, f, bounds, true)?)
    } else {
        None
    };

    let (lo, hi, range_end) = if b[s..base_end] == *b"*" {
        if hyphen.is_some() {
            return Err(ParseError::InvalidSyntax { field: f, offset: s });
        }
        (bounds.lo, bounds.hi, false)
    } else if let Some(h) = hyphen {
        if h == s || h + 1 == base_end
            || !b[s..h].iter().all(u8::is_ascii_digit)
            || !b[h + 1..base_end].iter().all(u8::is_ascii_digit)
        {
            return Err(ParseError::InvalidSyntax { field: f, offset: s });
        }
        let left = read_num(b, s, h, f, bounds, false)?;
        let right = read_num(b, h + 1, base_end, f, bounds, false)?;
        if left > right {
            return Err(ParseError::ReversedRange { field: f, offset: h + 1 });
        }
        (left, right, true)
    } else {
        if !b[s..base_end].iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax { field: f, offset: s });
        }
        let n = read_num(b, s, base_end, f, bounds, false)?;
        (n, if step.is_some() { bounds.hi } else { n }, false)
    };

    let _ = range_end;
    let stride = step.unwrap_or(1);
    let mut n = lo;
    while n <= hi {
        out[n as usize] = true;
        match n.checked_add(stride) {
            Some(next) => n = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_field(
    text: &str,
    f: Field,
    bounds: Bounds,
) -> Result<Vec<u8>, ParseError> {
    let b = text.as_bytes();
    for (i, &c) in b.iter().enumerate() {
        if !c.is_ascii_digit() && !matches!(c, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field: f, offset: i });
        }
    }

    let mut chosen = [false; 60];
    let mut s = 0;
    loop {
        let e = b[s..].iter().position(|&c| c == b',').map(|p| s + p).unwrap_or(b.len());
        parse_one(b, s, e, f, bounds, &mut chosen)?;
        if e == b.len() {
            break;
        }
        s = e + 1;
    }

    Ok((bounds.lo..=bounds.hi)
        .filter(|&n| chosen[n as usize])
        .map(|n| n as u8)
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {

        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),

fn num(bytes: &[u8], a: usize, b: usize, field: Field) -> Result<u32, ParseError> {
    if b - a > 1 && bytes[a] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: a });
    }
    let mut n = 0u32;
    for &d in &bytes[a..b] {
        n = n
            .checked_mul(10)
            .and_then(|x| x.checked_add(u32::from(d - b'0')))
            .ok_or(ParseError::NumberOverflow { field, offset: a })?;
    }
    Ok(n)
}

fn parse_one(
    bytes: &[u8],
    base_offset: usize,
    field: Field,
    set: &mut [bool],
) -> Result<(), ParseError> {
    let item = &bytes[base_offset..];
    let end = item.iter().position(|&b| b == b',').unwrap_or(item.len());
    let item = &item[..end];
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: base_offset,
        });
    }

    let (min, max, width) = bounds(field);
    let slash_count = item.iter().filter(|&&b| b == b'/').count();
    let slash = item.iter().position(|&b| b == b'/');
    let base_end = slash.unwrap_or(item.len());
    let base = &item[..base_end];

    let shape = if base == b"*" {
        Some((None, None, true))
    } else {
        let dash_count = base.iter().filter(|&&b| b == b'-').count();
        if dash_count == 0 && !base.is_empty() && base.iter().all(u8::is_ascii_digit) {
            Some((Some((0, base.len())), None, false))
        } else if dash_count == 1 {
            let dash = base.iter().position(|&b| b == b'-').unwrap();
            if dash > 0
                && dash + 1 < base.len()
                && base[..dash].iter().all(u8::is_ascii_digit)
                && base[dash + 1..].iter().all(u8::is_ascii_digit)
            {
                Some((Some((0, dash)), Some((dash + 1, base.len())), false))
            } else {
                None
            }
        } else {
            None
        }
    };

    let (start_span, end_span, star) = shape.ok_or(ParseError::InvalidSyntax {
        field,
        offset: base_offset,
    })?;

    if slash_count > 1
        || slash.is_some_and(|s| {
            s + 1 == item.len() || !item[s + 1..].iter().all(u8::is_ascii_digit)
        })
    {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: base_offset,
        });
    }

    let mut start = min as u32;
    let mut finish = max as u32;

    if let Some((a, b)) = start_span {
        start = num(bytes, base_offset + a, base_offset + b, field)?;
        if start < min as u32 || start > max as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: base_offset + a,
            });
        }
    }

    if let Some((a, b)) = end_span {
        finish = num(bytes, base_offset + a, base_offset + b, field)?;
        if finish < min as u32 || finish > max as u32 {
            return Err(ParseError::OutOfRange {
                field,
                offset: base_offset + a,
            });
        }
        if start > finish {
            return Err(ParseError::ReversedRange {
                field,
                offset: base_offset + a,
            });
        }
    } else if !star && slash.is_none() {
        finish = start;
    }

    let step = if let Some(s) = slash {
        let n = num(bytes, base_offset + s + 1, base_offset + item.len(), field)?;
        if n == 0 || n > width as u32 {
            return Err(ParseError::BadStep {
                field,
                offset: base_offset + s + 1,
            });
        }
        n
    } else {
        1
    };

    let mut v = start;
    while v <= finish {
        set[v as usize] = true;
        v += step;
    }
    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = text.as_bytes();
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() && !matches!(b, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset: i });
        }
    }

    let (min, max, _) = bounds(field);
    let mut set = vec![false; max as usize + 1];
    let mut offset = 0;
    loop {
        parse_one(bytes, offset, field, &mut set)?;
        match bytes[offset..].iter().position(|&b| b == b',') {
            Some(i) => {
                offset += i + 1;
                if offset > bytes.len() {
                    break;
                }
            }
            None => break,
        }
    }

    Ok((min..=max)
        .filter(|&v| set[v as usize])
        .collect())
}

fn bounds(field: Field) -> (u8, u8, u32) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),

fn number(bytes: &[u8], start: usize, end: usize, field: Field) -> Result<u32, ParseError> {
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset: start });
    }
    bytes[start..end].iter().try_fold(0u32, |n, &b| {
        n.checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset: start })
    })
}

fn parse_one(field_text: &str, field: Field) -> Result<u64, ParseError> {
    let bytes = field_text.as_bytes();
    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit() || matches!(b, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max, width) = bounds(field);
    let mut mask = 0u64;
    let mut start = 0usize;

    loop {
        let end = bytes[start..]
            .iter()
            .position(|&b| b == b',')
            .map_or(bytes.len(), |n| start + n);
        if end == start {
            return Err(ParseError::EmptyItem { field, offset: start });
        }

        let slashes: Vec<usize> = (start..end).filter(|&i| bytes[i] == b'/').collect();
        if slashes.len() > 1 {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }
        let (base_end, step_start) = match slashes.first() {
            Some(&p) if p > start && p + 1 < end => (p, Some(p + 1)),
            Some(_) => return Err(ParseError::InvalidSyntax { field, offset: start }),
            None => (end, None),
        };

        let hyphens: Vec<usize> = (start..base_end).filter(|&i| bytes[i] == b'-').collect();
        let kind;
        let (a_start, a_end, b_start, b_end);
        if base_end - start == 1 && bytes[start] == b'*' {
            if !hyphens.is_empty() {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
            kind = 0;
            a_start = start;
            a_end = start;
            b_start = start;
            b_end = start;
        } else if hyphens.is_empty()
            && bytes[start..base_end].iter().all(u8::is_ascii_digit)
            && base_end > start
        {
            kind = 1;
            a_start = start;
            a_end = base_end;
            b_start = start;
            b_end = base_end;
        } else if hyphens.len() == 1 {
            let h = hyphens[0];
            if h == start
                || h + 1 == base_end
                || !bytes[start..h].iter().all(u8::is_ascii_digit)
                || !bytes[h + 1..base_end].iter().all(u8::is_ascii_digit)
            {
                return Err(ParseError::InvalidSyntax { field, offset: start });
            }
            kind = 2;
            a_start = start;
            a_end = h;
            b_start = h + 1;
            b_end = base_end;
        } else {
            return Err(ParseError::InvalidSyntax { field, offset: start });
        }

        let (lo, hi) = if kind == 0 {
            (min as u32, max as u32)
        } else {
            let a = number(bytes, a_start, a_end, field)?;
            if a < min as u32 || a > max as u32 {
                return Err(ParseError::OutOfRange { field, offset: a_start });
            }
            if kind == 1 {
                (a, if step_start.is_some() { max as u32 } else { a })
            } else {
                let b = number(bytes, b_start, b_end, field)?;
                if b < min as u32 || b > max as u32 {
                    return Err(ParseError::OutOfRange { field, offset: b_start });
                }
                if a > b {
                    return Err(ParseError::ReversedRange { field, offset: b_start });
                }
                (a, b)
            }
        };

        let step = if let Some(p) = step_start {
            let n = number(bytes, p, end, field)?;
            if n == 0 || n > width {
                return Err(ParseError::BadStep { field, offset: p });
            }
            n
        } else {
            1
        };

        let mut n = lo;
        while n <= hi {
            mask |= 1u64 << n;
            n += step;
        }

        if end == bytes.len() {
            break;
        }
        start = end + 1;
    }

    Ok(mask)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut sets = [0u64; 5];
    for i in 0..5 {
        sets[i] = parse_one(parts[i], fields[i])?;
    }

    Ok(Cron {
        sets,
        day_stars: [parts[2].as_bytes() == b"*", parts[4].as_bytes() == b"*"],
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let (min, max, _) = bounds(field);
        (min..=max)
            .filter(|&n| self.sets[field as usize] & (1u64 << n) != 0)
            .collect()
    }
}

    start: usize,
    end: usize,
    field: Field,
    bounds: Option<(u32, u32)>,

fn is_digits(bytes: &[u8]) -> bool {
    !bytes.is_empty() && bytes.iter().all(u8::is_ascii_digit)
}

fn parse_item(
    item: &[u8],
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item.is_empty() {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_offset,
        });
    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    let (base_end, step_range) = match slash_positions.as_slice() {
        [] => (item.len(), None),
        [slash] => (*slash, Some((slash + 1, item.len()))),
        _ => return Err(invalid_syntax(field, item_offset)),
    };

    let base = &item[..base_end];
    let is_star = base == b"*";
    let hyphens: Vec<usize> = base
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'-').then_some(i))
        .collect();

    let base_kind = if is_star && hyphens.is_empty() {
        0
    } else if hyphens.is_empty() && is_digits(base) {
        1
    } else if hyphens.len() == 1 {
        let h = hyphens[0];
        if is_digits(&base[..h]) && is_digits(&base[h + 1..]) {
            2
        } else {
            return Err(invalid_syntax(field, item_offset));
        }
    } else {
        return Err(invalid_syntax(field, item_offset));
    };

    if let Some((step_start, step_end)) = step_range {
        if !is_digits(&item[step_start..step_end]) {
            return Err(invalid_syntax(field, item_offset));
        }
    }

    let (min, max) = field_info(field);
    let (start, end) = match base_kind {
        0 => (min, max),
        1 => {
            let v = parse_number(base, 0, base.len(), field, Some((min, max)))?;
            (v, if step_range.is_some() { max } else { v })
        }
        2 => {
            let h = hyphens[0];
            let a = parse_number(base, 0, h, field, Some((min, max)))?;
            let b = parse_number(base, h + 1, base.len(), field, Some((min, max)))?;
            if a > b {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_offset + h + 1,
                });
            }
            (a, b)
        }
        _ => unreachable!(),
    };

    let step = if let Some((step_start, step_end)) = step_range {
        let step = parse_number(item, step_start, step_end, field, None)?;
        let width = max - min + 1;
        if step == 0 || step > width {
            return Err(ParseError::BadStep {
                field,
                offset: item_offset + step_start,
            });
        }
        step
    } else {
        1
    };

    let mut value = start;
    while value <= end {
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(text: &str, field: Field) -> Result<(Vec<u8>, bool), ParseError> {
    let bytes = text.as_bytes();

    for (offset, &b) in bytes.iter().enumerate() {
        if !(b.is_ascii_digit() || matches!(b, b'*' | b'-' | b'/' | b',')) {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = field_info(field);
    let mut selected = vec![false; (max - min + 1) as usize];

    let mut item_start = 0usize;
    for i in 0..=bytes.len() {
        if i == bytes.len() || bytes[i] == b',' {
            parse_item(&bytes[item_start..i], item_start, field, &mut selected)?;
            item_start = i + 1;
        }
    }

    let values = selected
        .iter()
        .enumerate()
        .filter_map(|(i, &yes)| yes.then_some(min as u8 + i as u8))
        .collect();
    Ok((values, text == "*"))
}
