#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Specifier {
    name: String,
    start: usize,
}

#[derive(Clone)]
struct Expression {
    operator: Option<u8>,
    operator_pos: Option<usize>,
    specifiers: Vec<Specifier>,
}

#[derive(Clone)]
enum Segment {
    Literal(String),
    Expression(Expression),
}

fn is_operator(byte: u8) -> bool {
    matches!(byte, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&')
}

fn is_supported_operator(byte: u8) -> bool {
    is_operator(byte)
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression, Error> {
    let bytes = template.as_bytes();

    if let Some(inner) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos: inner });
    }

    let mut content_start = open + 1;
    let mut operator = None;
    let mut operator_pos = None;

    if content_start < close && is_operator(bytes[content_start]) {
        let op = bytes[content_start];
        operator_pos = Some(content_start);
        if !is_supported_operator(op) {
            return Err(Error::UnsupportedOperator {
                pos: content_start,
            });
        }
        operator = Some(op);
        content_start += 1;
    }

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specifiers = Vec::new();
    let mut used = std::collections::HashSet::<String>::new();
    let mut start = content_start;

    loop {
        let end = (start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if start == end {
            return Err(Error::EmptyVariable { pos: start });
        }

        let mut invalid_at = None;
        for i in start..end {
            let byte = bytes[i];
            let valid = if i == start {
                is_name_start(byte)
            } else {
                is_name_continue(byte)
            };
            if !valid {
                invalid_at = Some(i);
                break;
            }
        }

        if let Some(pos) = invalid_at {
            return Err(Error::InvalidName { pos });
        }

        let name = template[start..end].to_owned();
        if !used.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        specifiers.push(Specifier { name, start });

        if end == close {
            break;
        }
        start = end + 1;
    }

    Ok(Expression {
        operator,
        operator_pos,
        specifiers,
    })
}

fn validate(template: &str) -> Result<Vec<Segment>, Error> {
    let bytes = template.as_bytes();
    let mut segments = Vec::new();
    let mut cursor = 0;
    let mut literal_start = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                if literal_start < cursor {
                    segments.push(Segment::Literal(
                        template[literal_start..cursor].to_owned(),
                    ));
                }

                let open = cursor;
                let close = match (open + 1..bytes.len()).find(|&i| bytes[i] == b'}') {
                    Some(close) => close,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                let expression = parse_expression(template, open, close)?;
                segments.push(Segment::Expression(expression));
                cursor = close + 1;
                literal_start = cursor;
            }
            _ => cursor += 1,
        }
    }

    if literal_start < bytes.len() {
        segments.push(Segment::Literal(template[literal_start..].to_owned()));
    }

    Ok(segments)
}

fn lookup<'a>(name: &str, vars: &'a [(&str, &str)]) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(candidate, _)| *candidate == name)
        .map(|(_, value)| *value)
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> u8 {
    match value {
        0..=9 => b'0' + value,
        _ => b'A' + (value - 10),
    }
}

fn encode(value: &str, reserved_allowed: bool) -> String {
    let bytes = value.as_bytes();
    let mut output = String::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.push('%');
                output.push(hex_digit(high) as char);
                output.push(hex_digit(low) as char);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved_allowed && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4) as char);
            output.push(hex_digit(byte & 0x0f) as char);
        }
        i += 1;
    }

    output
}

fn expand_expression(expression: &Expression, vars: &[(&str, &str)]) -> String {
    let operator = expression.operator;
    let reserved_allowed = matches!(operator, Some(b'+') | Some(b'#'));
    let mut fields = Vec::<String>::new();

    for specifier in &expression.specifiers {
        if let Some(value) = lookup(&specifier.name, vars) {
            let encoded = encode(value, reserved_allowed);
            let field = match operator {
                Some(b';') if encoded.is_empty() => specifier.name.clone(),
                Some(b';' | b'?' | b'&') => {
                    format!("{}={}", specifier.name, encoded)
                }
                _ => encoded,
            };
            fields.push(field);
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let (separator, prefix) = match operator {
        Some(b'.') => (".", "."),
        Some(b'/') => ("/", "/"),
        Some(b';') => (";", ";"),
        Some(b'?') => ("&", "?"),
        Some(b'&') => ("&", "&"),
        _ => (",", ""),
    };

    let mut output = String::new();
    output.push_str(prefix);
    output.push_str(&fields.join(separator));
    output
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let segments = validate(template)?;
    let mut output = String::new();

    for segment in segments {
        match segment {
            Segment::Literal(text) => output.push_str(&text),
            Segment::Expression(expression) => {
                output.push_str(&expand_expression(&expression, vars));
            }
        }
    }

    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
enum Part {
    Literal(std::ops::Range<usize>),
    Expr {
        operator: Option<u8>,
        fields: Vec<(std::ops::Range<usize>, String)>,
    },
}

fn valid_first(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn valid_rest(b: u8) -> bool {
    valid_first(b) || b.is_ascii_digit()
}

fn parse_expr(
    source: &str,
    open: usize,
    close: usize,
) -> Result<(Option<u8>, Vec<(std::ops::Range<usize>, String)>), Error> {
    let bytes = source.as_bytes();
    if let Some(pos) = (open + 1..close).find(|&i| bytes[i] == b'{') {
        return Err(Error::NestedOpen { pos });
    }

    let mut start = open + 1;
    let mut op = None;
    if start < close && matches!(bytes[start], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
        op = Some(bytes[start]);
        start += 1;
    }

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut result = Vec::new();
    let mut seen = std::collections::HashSet::<String>::new();
    let mut field_start = start;

    loop {
        let field_end = (field_start..close)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(close);

        if field_start == field_end {
            return Err(Error::EmptyVariable { pos: field_start });
        }

        for i in field_start..field_end {
            let okay = if i == field_start {
                valid_first(bytes[i])
            } else {
                valid_rest(bytes[i])
            };
            if !okay {
                return Err(Error::InvalidName { pos: i });
            }
        }

        let name = source[field_start..field_end].to_owned();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: field_start });
        }
        result.push((field_start..field_end, name));

        if field_end == close {
            break;
        }
        field_start = field_end + 1;
    }

    Ok((op, result))
}

fn scan(source: &str) -> Result<Vec<Part>, Error> {
    let bytes = source.as_bytes();
    let mut parts = Vec::new();
    let mut pos = 0;
    let mut lit = 0;

    while pos < bytes.len() {
        if bytes[pos] == b'}' {
            return Err(Error::UnexpectedClose { pos });
        }
        if bytes[pos] != b'{' {
            pos += 1;
            continue;
        }

        if lit < pos {
            parts.push(Part::Literal(lit..pos));
        }
        let open = pos;
        let close = match (open + 1..bytes.len()).find(|&i| bytes[i] == b'}') {
            Some(c) => c,
            None => return Err(Error::Unclosed { pos: open }),
        };

        let (operator, fields) = parse_expr(source, open, close)?;
        parts.push(Part::Expr { operator, fields });
        pos = close + 1;
        lit = pos;
    }

    if lit < bytes.len() {
        parts.push(Part::Literal(lit..bytes.len()));
    }
    Ok(parts)
}

fn hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn digit(n: u8) -> u8 {
    if n < 10 { b'0' + n } else { b'A' + n - 10 }
}

fn ordinary(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, allow_reserved: bool, out: &mut String) {
    let bytes = value.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len() {
            if let (Some(h), Some(l)) = (hex(bytes[i + 1]), hex(bytes[i + 2])) {
                out.push('%');
                out.push(digit(h));
                out.push(digit(l));
                i += 3;
                continue;
            }
        }
        if ordinary(b) || (allow_reserved && reserved(b)) {
            out.push(b as char);
        } else {
            out.push('%');
            out.push(digit(b >> 4));
            out.push(digit(b & 15));
        }
        i += 1;
    }
}

fn render_expr(source: &str, operator: Option<u8>, fields: &[(std::ops::Range<usize>, String)], vars: &[(&str, &str)], out: &mut String) {
    let allow_reserved = matches!(operator, Some(b'+' | b'#'));
    let mut rendered = Vec::new();

    for (_, name) in fields {
        let value = vars.iter().rev().find(|(n, _)| *n == name).map(|(_, v)| *v);
        if let Some(value) = value {
            let mut encoded = String::new();
            encode(value, allow_reserved, &mut encoded);
            let field = match operator {
                Some(b';') if encoded.is_empty() => name.clone(),
                Some(b';' | b'?' | b'&') => format!("{name}={encoded}"),
                _ => encoded,
            };
            rendered.push(field);
        }
    }

    if rendered.is_empty() {
        return;
    }

    match operator {
        Some(b'.') => out.push('.'),
        Some(b'/') => out.push('/'),
        Some(b';') => out.push(';'),
        Some(b'?') => out.push('?'),
        Some(b'&') => out.push('&'),
        Some(b'#') => out.push('#'),
        _ => {}
    }

    let separator = match operator {
        Some(b'.') => ".",
        Some(b'/') => "/",
        Some(b';' | b'?' | b'&') => "&".replace('&', if operator == Some(b';') { ';' } else { '&' }).as_str(),
        _ => ",",
    };

    // Use an explicit loop so each separator is independent of earlier output.
    for (index, field) in rendered.iter().enumerate() {
        if index > 0 {
            let sep = match operator {
                Some(b'.') => ".",
                Some(b'/') => "/",
                Some(b';') => ";",
                Some(b'?' | b'&') => "&",
                _ => ",",
            };
            out.push_str(sep);
        }
        out.push_str(field);
    }

    let _ = (source, separator);
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let parts = scan(template)?;
    let mut output = String::new();

    for part in parts {
        match part {
            Part::Literal(range) => output.push_str(&template[range]),
            Part::Expr { operator, fields } => {
                render_expr(template, operator, &fields, vars, &mut output);
            }
        }
    }

    Ok(output)
}
