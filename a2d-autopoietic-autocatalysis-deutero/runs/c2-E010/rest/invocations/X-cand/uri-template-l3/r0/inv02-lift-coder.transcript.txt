pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone, Copy)]
struct Spec<'a> {
    name: &'a str,
    start: usize,
}

#[derive(Clone, Copy)]
struct Expr<'a> {
    op: Option<u8>,
    specs: Vec<Spec<'a>>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_byte(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn validate<'a>(template: &'a str) -> Result<Vec<Expr<'a>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let close = match bytes[open + 1..].iter().position(|&b| b == b'}') {
                    Some(offset) => open + 1 + offset,
                    None => return Err(Error::Unclosed { pos: open }),
                };

                if let Some(offset) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + offset,
                    });
                }

                let mut content_start = open + 1;
                let mut op = None;
                if content_start < close
                    && matches!(
                        bytes[content_start],
                        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
                    )
                {
                    let operator = bytes[content_start];
                    if !matches!(operator, b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                    op = Some(operator);
                    content_start += 1;
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut specs = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut segment_start = content_start;

                loop {
                    let segment_end = bytes[segment_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| segment_start + offset)
                        .unwrap_or(close);

                    if segment_start == segment_end {
                        return Err(Error::EmptyVariable { pos: segment_start });
                    }

                    let name_bytes = &bytes[segment_start..segment_end];
                    if !is_name_start(name_bytes[0]) {
                        return Err(Error::InvalidName { pos: segment_start });
                    }
                    for (offset, &b) in name_bytes.iter().enumerate().skip(1) {
                        if !is_name_byte(b) {
                            return Err(Error::InvalidName {
                                pos: segment_start + offset,
                            });
                        }
                    }

                    let name = &template[segment_start..segment_end];
                    if seen.iter().any(|&old| old == name) {
                        return Err(Error::DuplicateVariable { pos: segment_start });
                    }
                    seen.push(name);
                    specs.push(Spec {
                        name,
                        start: segment_start,
                    });

                    if segment_end == close {
                        break;
                    }
                    segment_start = segment_end + 1;
                }

                expressions.push(Expr { op, specs });
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    Ok(expressions)
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn encode(value: &str, reserved: bool, out: &mut String) {
    let bytes = value.as_bytes();
    let mut i = 0;
    const HEX: &[u8; 16] = b"0123456789ABCDEF";

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && hex_value(bytes[i + 1]).is_some()
            && hex_value(bytes[i + 2]).is_some()
        {
            out.push('%');
            out.push(HEX[hex_value(bytes[i + 1]).unwrap() as usize] as char);
            out.push(HEX[hex_value(bytes[i + 2]).unwrap() as usize] as char);
            i += 3;
        } else if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
            i += 1;
        }
    }
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate(template)?;
    let bytes = template.as_bytes();
    let mut out = String::new();
    let mut cursor = 0;
    let mut expr_index = 0;

    while cursor < bytes.len() {
        if bytes[cursor] != b'{' {
            cursor += 1;
            continue;
        }

        out.push_str(&template[cursor..bytes[cursor..]
            .iter()
            .position(|&b| b == b'}')
            .map(|offset| cursor + offset)
            .unwrap() + 1]);

        // The preceding push is replaced by literal and expression handling below.
        let _ = &expressions[expr_index];
        expr_index += 1;
    }

    // Rebuild output in a single left-to-right pass over the validated expressions.
    out.clear();
    cursor = 0;
    for expr in &expressions {
        let open = bytes[cursor..]
            .iter()
            .position(|&b| b == b'{')
            .map(|offset| cursor + offset)
            .unwrap_or(bytes.len());
        out.push_str(&template[cursor..open]);

        let reserved = matches!(expr.op, Some(b'+') | Some(b'#'));
        let mut fields: Vec<String> = Vec::new();

        for spec in &expr.specs {
            let value = vars
                .iter()
                .rev()
                .find(|(name, _)| *name == spec.name)
                .map(|(_, value)| *value);
            if let Some(value) = value {
                let mut encoded = String::new();
                encode(value, reserved, &mut encoded);
                if matches!(expr.op, Some(b';') | Some(b'?') | Some(b'&')) {
                    if expr.op == Some(b';') && encoded.is_empty() {
                        fields.push(spec.name.to_owned());
                    } else {
                        fields.push(format!("{}={}", spec.name, encoded));
                    }
                } else {
                    fields.push(encoded);
                }
            }
        }

        if !fields.is_empty() {
            match expr.op {
                Some(b'#') => out.push('#'),
                Some(b'.') => out.push('.'),
                Some(b'/') => out.push('/'),
                Some(b';') => out.push(';'),
                Some(b'?') => out.push('?'),
                Some(b'&') => out.push('&'),
                _ => {}
            }

            let separator = match expr.op {
                Some(b'.') => '.',
                Some(b'/') => '/',
                Some(b';') => ';',
                Some(b'?') | Some(b'&') => '&',
                _ => ',',
            };
            for (index, field) in fields.iter().enumerate() {
                if index != 0 {
                    out.push(separator);
                }
                out.push_str(field);
            }
        }

        let open_pos = open;
        let close_pos = bytes[open_pos + 1..]
            .iter()
            .position(|&b| b == b'}')
            .map(|offset| open_pos + 1 + offset)
            .unwrap();
        cursor = close_pos + 1;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Expression {
    start: usize,
    end: usize,
    operator: Option<u8>,
    names: Vec<(usize, usize)>,
}

fn name_first(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn name_rest(byte: u8) -> bool {
    name_first(byte) || byte.is_ascii_digit()
}

fn parse(template: &str) -> Result<Vec<Expression>, Error> {
    let b = template.as_bytes();
    let mut result = Vec::new();
    let mut pos = 0;

    while pos < b.len() {
        if b[pos] == b'}' {
            return Err(Error::UnexpectedClose { pos });
        }
        if b[pos] != b'{' {
            pos += 1;
            continue;
        }

        let start = pos;
        let end = match b[start + 1..].iter().position(|&c| c == b'}') {
            Some(n) => start + 1 + n,
            None => return Err(Error::Unclosed { pos: start }),
        };
        if let Some(n) = b[start + 1..end].iter().position(|&c| c == b'{') {
            return Err(Error::NestedOpen { pos: start + 1 + n });
        }

        let mut content = start + 1;
        let mut operator = None;
        if content < end && matches!(b[content], b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') {
            operator = Some(b[content]);
            content += 1;
        }
        if content == end {
            return Err(Error::EmptyExpression { pos: end });
        }

        let mut names = Vec::new();
        let mut seen: Vec<&str> = Vec::new();
        let mut item = content;
        loop {
            let item_end = b[item..end]
                .iter()
                .position(|&c| c == b',')
                .map(|n| item + n)
                .unwrap_or(end);
            if item == item_end {
                return Err(Error::EmptyVariable { pos: item });
            }

            let part = &b[item..item_end];
            if !name_first(part[0]) {
                return Err(Error::InvalidName { pos: item });
            }
            if let Some(n) = part.iter().enumerate().skip(1).find_map(|(n, &c)| {
                if name_rest(c) { None } else { Some(n) }
            }) {
                return Err(Error::InvalidName { pos: item + n });
            }

            let name = &template[item..item_end];
            if seen.contains(&name) {
                return Err(Error::DuplicateVariable { pos: item });
            }
            seen.push(name);
            names.push((item, item_end));

            if item_end == end {
                break;
            }
            item = item_end + 1;
        }

        result.push(Expression {
            start,
            end,
            operator,
            names,
        });
        pos = end + 1;
    }

    Ok(result)
}

fn hex(c: u8) -> Option<u8> {
    match c {
        b'0'..=b'9' => Some(c - b'0'),
        b'a'..=b'f' => Some(c - b'a' + 10),
        b'A'..=b'F' => Some(c - b'A' + 10),
        _ => None,
    }
}

fn encode_value(value: &str, allow_reserved: bool) -> String {
    let b = value.as_bytes();
    let mut result = String::new();
    let mut i = 0;
    const H: &[u8; 16] = b"0123456789ABCDEF";

    while i < b.len() {
        if b[i] == b'%' && i + 2 < b.len() {
            if let (Some(x), Some(y)) = (hex(b[i + 1]), hex(b[i + 2])) {
                result.push('%');
                result.push(H[x as usize] as char);
                result.push(H[y as usize] as char);
                i += 3;
                continue;
            }
        }

        let c = b[i];
        let reserved = matches!(
            c,
            b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
        );
        if c.is_ascii_alphanumeric() || matches!(c, b'-' | b'.' | b'_' | b'~') || (allow_reserved && reserved) {
            result.push(c as char);
        } else {
            result.push('%');
            result.push(H[(c >> 4) as usize] as char);
            result.push(H[(c & 15) as usize] as char);
        }
        i += 1;
    }
    result
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let b = template.as_bytes();
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        let mut fields = Vec::new();

        for (start, end) in expression.names {
            let name = &template[start..end];
            if let Some((_, value)) = vars.iter().rev().find(|(key, _)| *key == name) {
                let encoded = encode_value(value, matches!(expression.operator, Some(b'+') | Some(b'#')));
                let field = match expression.operator {
                    Some(b';') if encoded.is_empty() => name.to_owned(),
                    Some(b';' | b'?' | b'&') => format!("{name}={encoded}"),
                    _ => encoded,
                };
                fields.push(field);
            }
        }

        if !fields.is_empty() {
            let (prefix, separator) = match expression.operator {
                Some(b'#') => ("#", ','),
                Some(b'.') => (".", '.'),
                Some(b'/') => ("/", '/'),
                Some(b';') => (";", ';'),
                Some(b'?') => ("?", '&'),
                Some(b'&') => ("&", '&'),
                _ => ("", ','),
            };
            output.push_str(prefix);
            for (i, field) in fields.iter().enumerate() {
                if i > 0 {
                    output.push(separator);
                }
                output.push_str(field);
            }
        }

        cursor = expression.end + 1;
        let _ = b;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}
