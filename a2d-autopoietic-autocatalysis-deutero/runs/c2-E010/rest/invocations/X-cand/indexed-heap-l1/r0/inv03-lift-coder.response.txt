- Key validation trims only ASCII spaces, then reports invalid bytes using offsets in the original input.
- Validation errors take precedence over duplicate and lookup outcomes.
- Keys are stored and compared in ASCII lowercase; values remain unchanged.
- Items rank by signed priority, value byte length, then retained insertion age.
- Removal followed by reinsertion gives the key a new age; age counters use checked renumbering to avoid overflow.
- Since no key-size or queue-size limit is specified, both implementations use standard-library collections without imposing one.

### Solution A

```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

#[derive(Debug, Clone)]
struct Entry {
    item: Item,
    age: u128,
}

#[derive(Debug, Clone)]
pub struct IndexedHeap {
    entries: HashMap<String, Entry>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    fn validate_key(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }
        if start == end {
            return Err(Error::EmptyKey);
        }

        for (offset, &byte) in bytes[start..end].iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic()
            } else {
                byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + offset });
            }
        }

        Ok(input[start..end].to_ascii_lowercase())
    }

    fn best_key(&self) -> Option<&str> {
        self.entries
            .iter()
            .min_by(|(_, a), (_, b)| {
                a.item
                    .priority
                    .cmp(&b.item.priority)
                    .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
                    .then_with(|| a.age.cmp(&b.age))
            })
            .map(|(key, _)| key.as_str())
    }

    fn renumber_ages(&mut self) {
        let mut ordered: Vec<(String, u128)> = self
            .entries
            .iter()
            .map(|(key, entry)| (key.clone(), entry.age))
            .collect();
        ordered.sort_by_key(|(_, age)| *age);
        for (age, (key, _)) in ordered.into_iter().enumerate() {
            if let Some(entry) = self.entries.get_mut(&key) {
                entry.age = age as u128;
            }
        }
        self.next_age = self.entries.len() as u128;
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let canonical = Self::validate_key(key)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Duplicate { key: canonical });
        }

        if self.next_age == u128::MAX {
            self.renumber_ages();
        }
        let age = self.next_age;
        self.next_age += 1;

        self.entries.insert(
            canonical.clone(),
            Entry {
                item: Item {
                    key: canonical,
                    priority,
                    value: value.to_owned(),
                },
                age,
            },
        );
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        let key = self.best_key()?;
        self.entries.get(key).map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.best_key()?.to_owned();
        self.entries.remove(&key).map(|entry| entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical = Self::validate_key(key)?;
        Ok(self.entries.get(&canonical).map(|entry| entry.item.clone()))
    }
}
```

### Solution B

```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

#[derive(Debug, Clone)]
struct Entry {
    item: Item,
    age: u128,
}

#[derive(Debug, Clone)]
pub struct IndexedHeap {
    entries: HashMap<String, Entry>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn validate_key(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let start = bytes.iter().position(|&b| b != b' ').unwrap_or(bytes.len());
        let end = bytes
            .iter()
            .rposition(|&b| b != b' ')
            .map(|i| i + 1)
            .unwrap_or(start);
        if start == end {
            return Err(Error::EmptyKey);
        }

        let mut canonical = Vec::with_capacity(end - start);
        for (relative, &byte) in bytes[start..end].iter().enumerate() {
            let valid = match relative {
                0 => byte.is_ascii_alphabetic(),
                _ => byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-',
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + relative });
            }
            canonical.push(byte.to_ascii_lowercase());
        }

        // All accepted bytes are ASCII, so this conversion cannot fail.
        Ok(String::from_utf8(canonical).expect("validated ASCII key"))
    }

    fn best_key(&self) -> Option<&String> {
        self.entries.iter().min_by(|(_, a), (_, b)| {
            (a.item.priority, a.item.value.len(), a.age)
                .cmp(&(b.item.priority, b.item.value.len(), b.age))
        }).map(|(key, _)| key)
    }

    fn reset_ages(&mut self) {
        let mut keys: Vec<String> = self.entries.keys().cloned().collect();
        keys.sort_by_key(|key| self.entries.get(key).map(|entry| entry.age));
        for (new_age, key) in keys.into_iter().enumerate() {
            if let Some(entry) = self.entries.get_mut(&key) {
                entry.age = new_age as u128;
            }
        }
        self.next_age = self.entries.len() as u128;
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let canonical = Self::validate_key(key)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Duplicate { key: canonical });
        }

        if self.next_age == u128::MAX {
            self.reset_ages();
        }
        let age = self.next_age;
        self.next_age += 1;
        self.entries.insert(
            canonical.clone(),
            Entry {
                item: Item {
                    key: canonical,
                    priority,
                    value: value.to_owned(),
                },
                age,
            },
        );
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_key()
            .and_then(|key| self.entries.get(key))
            .map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.best_key()?.clone();
        self.entries.remove(&key).map(|entry| entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical = Self::validate_key(key)?;
        Ok(self.entries.get(&canonical).map(|entry| entry.item.clone()))
    }
}
```
