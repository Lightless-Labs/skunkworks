#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

struct Entry {
    item: Item,
    age: u64,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    positions: HashMap<String, usize>,
    next_age: u64,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            positions: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonicalize(key)?;
        if self.positions.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let entry = Entry {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };
        self.next_age = self.next_age.wrapping_add(1);
        self.entries.push(entry);
        let index = self.entries.len() - 1;
        self.positions.insert(self.entries[index].item.key.clone(), index);
        self.sift_up(index);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.entries.is_empty() {
            return None;
        }
        let last = self.entries.len() - 1;
        self.swap_entries(0, last);
        let removed = self.entries.pop().expect("nonempty heap");
        self.positions.remove(&removed.item.key);
        if !self.entries.is_empty() {
            self.sift_down(0);
        }
        Some(removed.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        Ok(self
            .positions
            .get(&key)
            .map(|&index| self.entries[index].item.clone()))
    }

    fn less(&self, a: usize, b: usize) -> bool {
        let left = &self.entries[a];
        let right = &self.entries[b];
        (
            left.item.priority,
            left.item.value.len(),
            left.age,
        ) < (
            right.item.priority,
            right.item.value.len(),
            right.age,
        )
    }

    fn swap_entries(&mut self, a: usize, b: usize) {
        if a == b {
            return;
        }
        self.entries.swap(a, b);
        self.positions.insert(self.entries[a].item.key.clone(), a);
        self.positions.insert(self.entries[b].item.key.clone(), b);
    }

    fn sift_up(&mut self, mut index: usize) {
        while index > 0 {
            let parent = (index - 1) / 2;
            if !self.less(index, parent) {
                break;
            }
            self.swap_entries(index, parent);
            index = parent;
        }
    }

    fn sift_down(&mut self, mut index: usize) {
        loop {
            let left = index * 2 + 1;
            if left >= self.entries.len() {
                break;
            }
            let right = left + 1;
            let best = if right < self.entries.len() && self.less(right, left) {
                right
            } else {
                left
            };
            if !self.less(best, index) {
                break;
            }
            self.swap_entries(index, best);
            index = best;
        }
    }
}

        };
        if !valid {
            return Err(Error::InvalidKey { at: start + relative });
        }
    }

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    positions: HashMap<String, usize>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            positions: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonicalize(key)?;
        if self.positions.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let entry = Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };
        self.next_age = self.next_age.wrapping_add(1);
        let index = self.entries.len();
        self.entries.push(entry);
        self.positions.insert(key, index);
        self.up(index);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.entries.is_empty() {
            return None;
        }

        let last = self.entries.len() - 1;
        self.exchange(0, last);
        let entry = self.entries.pop()?;
        self.positions.remove(&entry.item.key);
        if !self.entries.is_empty() {
            self.down(0);
        }
        Some(entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        let result = match self.positions.get(&key) {
            Some(&index) => Some(self.entries[index].item.clone()),
            None => None,
        };
        Ok(result)
    }

    fn precedes(&self, a: usize, b: usize) -> bool {
        let a = &self.entries[a];
        let b = &self.entries[b];

        if a.item.priority != b.item.priority {
            return a.item.priority < b.item.priority;
        }
        if a.item.value.len() != b.item.value.len() {
            return a.item.value.len() < b.item.value.len();
        }
        a.age < b.age
    }

    fn exchange(&mut self, a: usize, b: usize) {
        if a == b {
            return;
        }
        self.entries.swap(a, b);
        let key_a = self.entries[a].item.key.clone();
        let key_b = self.entries[b].item.key.clone();
        self.positions.insert(key_a, a);
        self.positions.insert(key_b, b);
    }

    fn up(&mut self, mut index: usize) {
        while index > 0 {
            let parent = (index - 1) / 2;
            if !self.precedes(index, parent) {
                break;
            }
            self.exchange(index, parent);
            index = parent;
        }
    }

    fn down(&mut self, mut index: usize) {
        loop {
            let left = index * 2 + 1;
            if left >= self.entries.len() {
                return;
            }
            let right = left + 1;
            let child = if right < self.entries.len() && self.precedes(right, left) {
                right
            } else {
                left
            };
            if !self.precedes(child, index) {
                return;
            }
            self.exchange(index, child);
            index = child;
        }
    }
}

fn canonicalize(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for index in start..end {
        let byte = bytes[index];
        let accepted = if index == start {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !accepted {
            return Err(Error::InvalidKey { at: index });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

use std::cmp::Ordering;
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u64,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    next_age: u64,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = canonicalize(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        if self.next_age == u64::MAX {
            self.renumber_ages();
        }
        let age = self.next_age;
        self.next_age += 1;

        let index = self.entries.len();
        self.entries.push(Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age,
        });
        self.indices.insert(key, index);
        self.sift_up(index);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.entries.is_empty() {
            return None;
        }

        let last = self.entries.len() - 1;
        self.swap_entries(0, last);
        let removed = self.entries.pop().expect("heap was nonempty");
        self.indices.remove(&removed.item.key);

        if !self.entries.is_empty() {
            self.sift_down(0);
        }
        Some(removed.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        Ok(self
            .indices
            .get(&key)
            .map(|&index| self.entries[index].item.clone()))
    }

    fn compare(&self, a: usize, b: usize) -> Ordering {
        let a = &self.entries[a];
        let b = &self.entries[b];
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn swap_entries(&mut self, a: usize, b: usize) {
        self.entries.swap(a, b);
        let key_a = self.entries[a].item.key.clone();
        let key_b = self.entries[b].item.key.clone();
        self.indices.insert(key_a, a);
        self.indices.insert(key_b, b);
    }

    fn sift_up(&mut self, mut index: usize) {
        while index > 0 {
            let parent = (index - 1) / 2;
            if self.compare(index, parent) != Ordering::Less {
                break;
            }
            self.swap_entries(index, parent);
            index = parent;
        }
    }

    fn sift_down(&mut self, mut index: usize) {
        loop {
            let left = index * 2 + 1;
            if left >= self.entries.len() {
                break;
            }
            let right = left + 1;
            let best = if right < self.entries.len()
                && self.compare(right, left) == Ordering::Less
            {
                right
            } else {
                left
            };
            if self.compare(best, index) != Ordering::Less {
                break;
            }
            self.swap_entries(index, best);
            index = best;
        }
    }

    fn renumber_ages(&mut self) {
        let mut order: Vec<usize> = (0..self.entries.len()).collect();
        order.sort_by_key(|&index| self.entries[index].age);
        for (age, index) in order.into_iter().enumerate() {
            self.entries[index].age = age as u64;
        }
        self.next_age = self.entries.len() as u64;
    }
}

fn canonicalize(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    let trimmed = &bytes[start..end];
    for (offset, &byte) in trimmed.iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(trimmed
        .iter()
        .map(|byte| byte.to_ascii_lowercase() as char)
        .collect())
}

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
}

#[derive(Debug, Clone)]
struct Entry {
    item: Item,
    age: u128,
}

#[derive(Debug, Clone)]
pub struct IndexedHeap {
    entries: HashMap<String, Entry>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn validate_key(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let start = bytes.iter().position(|&b| b != b' ').unwrap_or(bytes.len());
        let end = bytes
            .iter()
            .rposition(|&b| b != b' ')
            .map(|i| i + 1)
            .unwrap_or(start);
        if start == end {
            return Err(Error::EmptyKey);
        }

        let mut canonical = Vec::with_capacity(end - start);
        for (relative, &byte) in bytes[start..end].iter().enumerate() {
            let valid = match relative {
                0 => byte.is_ascii_alphabetic(),
                _ => byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-',
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + relative });
            }
            canonical.push(byte.to_ascii_lowercase());
        }

        // All accepted bytes are ASCII, so this conversion cannot fail.
        Ok(String::from_utf8(canonical).expect("validated ASCII key"))
    }

    fn best_key(&self) -> Option<&String> {
        self.entries.iter().min_by(|(_, a), (_, b)| {
            (a.item.priority, a.item.value.len(), a.age)
                .cmp(&(b.item.priority, b.item.value.len(), b.age))
        }).map(|(key, _)| key)
    }

    fn reset_ages(&mut self) {
        let mut keys: Vec<String> = self.entries.keys().cloned().collect();
        keys.sort_by_key(|key| self.entries.get(key).map(|entry| entry.age));
        for (new_age, key) in keys.into_iter().enumerate() {
            if let Some(entry) = self.entries.get_mut(&key) {
                entry.age = new_age as u128;
            }
        }
        self.next_age = self.entries.len() as u128;
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let canonical = Self::validate_key(key)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Duplicate { key: canonical });
        }

        if self.next_age == u128::MAX {
            self.reset_ages();
        }
        let age = self.next_age;
        self.next_age += 1;
        self.entries.insert(
            canonical.clone(),
            Entry {
                item: Item {
                    key: canonical,
                    priority,
                    value: value.to_owned(),
                },
                age,
            },
        );
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_key()
            .and_then(|key| self.entries.get(key))
            .map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.best_key()?.clone();
        self.entries.remove(&key).map(|entry| entry.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let canonical = Self::validate_key(key)?;
        Ok(self.entries.get(&canonical).map(|entry| entry.item.clone()))
    }
}
