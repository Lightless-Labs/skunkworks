    pub fn cols(&self) -> usize { self.cols }

            }

                }

            }
        }

                let factor = out.data[row * out.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..out.cols {
                    let product = factor.mul(&out.data[p * out.cols + c]).map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                    out.data[row * out.cols + c] = out.data[row * out.cols + c].sub(&product).map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok((out, pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let (matrix, pivots, swaps) = self.eliminate(self.cols)?;
        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let augmented = Self { rows: n, cols: width, data };
        let (reduced, pivots, _) = augmented.eliminate(n)?;
        if pivots.len() < n { return Err(MatrixError::Singular { rank: pivots.len() }); }
        let mut right = Vec::with_capacity(n * n);
        for r in 0..n {
            right.extend_from_slice(&reduced.data[r * width + n..r * width + width]);
        }
        Ok(Self { rows: n, cols: n, data: right })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]

    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

pub enum ParseKind { Empty, InvalidByte, NonCanonical, OutOfRange, ZeroDenominator }

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational { num: i64, den: i64 }

fn gcd128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 { let r = a % b; a = b; b = r; }
    a
}

impl Rational {
    fn build(n: i128, d: i128) -> Result<Self, RationalError> {
        if d == 0 { return Err(RationalError::ZeroDenominator); }
        let (mut n, mut d) = if d < 0 { (-n, -d) } else { (n, d) };
        if n == 0 { return Ok(Self { num: 0, den: 1 }); }
        let g = gcd128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> { Self::build(num as i128, den as i128) }
    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }
    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::build(self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128, self.den as i128 * rhs.den as i128)
    }
    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::build(self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128, self.den as i128 * rhs.den as i128)
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::build(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }
    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        Self::build(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }
    pub fn neg(&self) -> Result<Self, RationalError> { Self::build(-(self.num as i128), self.den as i128) }

    pub fn parse(s: &str) -> Result<Self, RationalError> {
        let b = s.as_bytes();
        let error = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() { return Err(error(0, ParseKind::Empty)); }
        let neg = b[0] == b'-';
        let start = if neg { 1 } else { 0 };
        let mut i = start;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == start { return Err(error(i, ParseKind::InvalidByte)); }
        let num_end = i;
        let slash = if i < b.len() && b[i] == b'/' { Some(i) } else { None };
        let den_range = if let Some(slash) = slash {
            i = slash + 1;
            let ds = i;
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if ds == i { return Err(error(i, ParseKind::InvalidByte)); }
            if i < b.len() { return Err(error(i, ParseKind::InvalidByte)); }
            Some((ds, i, slash))
        } else {
            if i < b.len() { return Err(error(i, ParseKind::InvalidByte)); }
            None
        };
        let ns = start;
        let nl = num_end - ns;
        if nl > 1 && b[ns] == b'0' { return Err(error(ns + 1, ParseKind::NonCanonical)); }
        if neg && nl == 1 && b[ns] == b'0' { return Err(error(0, ParseKind::NonCanonical)); }
        if let Some((ds, de, _)) = den_range {
            if de - ds > 1 && b[ds] == b'0' { return Err(error(ds + 1, ParseKind::NonCanonical)); }
        }
        let nm = decimal(&b[ns..num_end]).unwrap_or(u128::MAX);
        let lim = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        if nm > lim { return Err(error(0, ParseKind::OutOfRange)); }
        let dm = if let Some((ds, de, _)) = den_range {
            let v = decimal(&b[ds..de]).unwrap_or(u128::MAX);
            if v > i64::MAX as u128 { return Err(error(ds, ParseKind::OutOfRange)); }
            Some(v)
        } else { None };
        if let Some(d) = dm {
            let (ds, _, sl) = den_range.unwrap();
            if d == 0 { return Err(error(ds, ParseKind::ZeroDenominator)); }
            if d == 1 { return Err(error(sl, ParseKind::NonCanonical)); }
            if nm == 0 { return Err(error(sl, ParseKind::NonCanonical)); }
            if gcd128(nm, d) != 1 { return Err(error(sl, ParseKind::NonCanonical)); }
        }
        let signed = if neg { -(nm as i128) } else { nm as i128 };
        Ok(Self { num: signed as i64, den: dm.unwrap_or(1) as i64 })
    }
}

    for &c in s { x = x.checked_mul(10)?.checked_add((c - b'0') as u128)?; }
    Some(x)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow, DataLength { expected: usize, actual: usize }, Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize }, Singular { rank: usize },
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix { rows: usize, cols: usize, data: Vec<Rational> }

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let n = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if n != data.len() { return Err(MatrixError::DataLength { expected: n, actual: data.len() }); }
        Ok(Self { rows, cols, data })
    }
    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }
    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols { return Err(MatrixError::Index { row, col }); }
        Ok(self.data[row * self.cols + col])
    }
    pub fn transpose(&self) -> Self {
        let mut d = Vec::with_capacity(self.data.len());
        for r in 0..self.cols { for c in 0..self.rows { d.push(self.data[c * self.cols + r]); } }
        Self { rows: self.cols, cols: self.rows, data: d }
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch { left: (self.rows,self.cols), right: (rhs.rows,rhs.cols) });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut d = Vec::with_capacity(len);
        let z = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut sum = z;
                for k in 0..self.cols {
                    let product = self.data[r*self.cols+k].mul(&rhs.data[k*rhs.cols+c])
                        .map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                    sum = sum.add(&product)
                        .map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                }
                d.push(sum);
            }
        }
        Ok(Self { rows:self.rows, cols:rhs.cols, data:d })
    }

    fn reduce_columns(&self, candidates: usize) -> Result<(Self, Vec<usize>, Vec<(usize,usize)>), MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0;
        for col in 0..candidates {
            if p == m.rows { break; }
            let mut chosen = None;
            for row in p..m.rows {
                let v = m.data[row*m.cols+col];
                if v.num == 0 { continue; }
                let key = (v.den, v.num.unsigned_abs(), row);
                if chosen.map_or(true, |(bd, bn, br)| key < (bd,bn,br)) {
                    chosen = Some((v.den,v.num.unsigned_abs(),row));
                }
            }
            let Some((_,_,q)) = chosen else { continue };
            if q != p {
                for c in 0..m.cols { m.data.swap(p*m.cols+c,q*m.cols+c); }
                swaps.push((p,q));
            }
            let pivot = m.data[p*m.cols+col];
            for c in 0..m.cols {
                let idx = p*m.cols+c;
                m.data[idx] = m.data[idx].div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row:p,col:c,step:p,source })?;
            }
            for row in 0..m.rows {
                if row == p { continue; }
                let factor = m.data[row*m.cols+col];
                if factor.num == 0 { continue; }
                for c in 0..m.cols {
                    let idx = row*m.cols+c;
                    let term = factor.mul(&m.data[p*m.cols+c])
                        .map_err(|source| MatrixError::Arithmetic { row,col:c,step:p,source })?;
                    m.data[idx] = m.data[idx].sub(&term)
                        .map_err(|source| MatrixError::Arithmetic { row,col:c,step:p,source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok((m,pivots,swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let (matrix,pivots,swaps) = self.reduce_columns(self.cols)?;
        Ok(Reduction { matrix,pivots,swaps })
    }
    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }
    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols { return Err(MatrixError::NotSquare { rows:self.rows,cols:self.cols }); }
        let n = self.rows;
        let w = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let size = n.checked_mul(w).ok_or(MatrixError::ShapeOverflow)?;
        let mut d = Vec::with_capacity(size);
        for r in 0..n {
            d.extend_from_slice(&self.data[r*n..(r+1)*n]);
            for c in 0..n { d.push(if r == c { Rational { num:1,den:1 } } else { Rational { num:0,den:1 } }); }
        }
        let aug = Self { rows:n,cols:w,data:d };
        let (red,pivots,_) = aug.reduce_columns(n)?;
        if pivots.len() < n { return Err(MatrixError::Singular { rank:pivots.len() }); }
        let mut right = Vec::with_capacity(n*n);
        for r in 0..n { right.extend_from_slice(&red.data[r*w+n..r*w+w]); }
        Ok(Self { rows:n,cols:n,data:right })
    }
}
