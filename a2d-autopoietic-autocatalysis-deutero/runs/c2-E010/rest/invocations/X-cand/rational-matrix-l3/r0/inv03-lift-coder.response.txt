- Syntax is validated across the whole input before canonicality or range checks; I use byte offsets and the specified check order.
- Rational operations reduce the mathematical result before checking bounds, using wider intermediates to handle `i64::MIN`.
- Matrix multiplication follows the prescribed product-then-add order, including temporary overflow; zero-sized outputs perform no scalar work.
- RREF chooses pivots by denominator, then unsigned numerator magnitude, then current row; row-operation errors use the pivot ordinal as `step`.
- Inverse checks squareness, augmented width, and storage size in that order, then runs the same elimination over the first `n` columns.
- For dimensions whose storage product is zero, I preserve the explicit shape and avoid iterating the unused large dimension.

use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        Self::from_i128(n, self.den as i128 * rhs.den as i128)
    }
    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        Self::from_i128(n, self.den as i128 * rhs.den as i128)
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }
    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        Self::from_i128(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }
    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let fail = |at, kind| Err(RationalError::Parse { at, kind });
        if b.is_empty() { return fail(0, ParseKind::Empty); }

        let mut i = 0;
        let neg = b[0] == b'-';
        if neg { i = 1; }
        let nstart = i;
        if i == b.len() { return fail(i, ParseKind::InvalidByte); }
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == nstart { return fail(i, ParseKind::InvalidByte); }
        let nend = i;
        let mut dstart = None;
        let mut dend = None;
        if i < b.len() {
            if b[i] != b'/' { return fail(i, ParseKind::InvalidByte); }
            i += 1;
            dstart = Some(i);
            if i == b.len() { return fail(i, ParseKind::InvalidByte); }
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == dstart.unwrap() { return fail(i, ParseKind::InvalidByte); }
            dend = Some(i);
            if i < b.len() { return fail(i, ParseKind::InvalidByte); }
        }

        let n_digits = nend - nstart;
        if n_digits >= 2 && b[nstart] == b'0' {
            return fail(nstart + 1, ParseKind::NonCanonical);
        }
        if neg && n_digits == 1 && b[nstart] == b'0' {
            return fail(0, ParseKind::NonCanonical);
        }
        if let (Some(ds), Some(de)) = (dstart, dend) {
            if de - ds >= 2 && b[ds] == b'0' {
                return fail(ds + 1, ParseKind::NonCanonical);
            }
        }

        let limit: u128 = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        let mut mag = 0u128;
        for &ch in &b[nstart..nend] {
            mag = match mag.checked_mul(10).and_then(|x| x.checked_add((ch - b'0') as u128)) {
                Some(v) if v <= limit => v,
                _ => return fail(0, ParseKind::OutOfRange),
            };
        }
        let num = if neg {
            if mag == (1u128 << 63) { i64::MIN } else { -(mag as i64) }
        } else { mag as i64 };

        if let (Some(ds), Some(de)) = (dstart, dend) {
            let mut denmag = 0u128;
            for &ch in &b[ds..de] {
                denmag = match denmag.checked_mul(10).and_then(|x| x.checked_add((ch - b'0') as u128)) {
                    Some(v) if v <= i64::MAX as u128 => v,
                    _ => return fail(ds, ParseKind::OutOfRange),
                };
            }
            if denmag == 0 { return fail(ds, ParseKind::ZeroDenominator); }
            if denmag == 1 { return fail(ds - 1, ParseKind::NonCanonical); }
            if num == 0 { return fail(ds - 1, ParseKind::NonCanonical); }
            if gcd_u128(mag, denmag) > 1 { return fail(ds - 1, ParseKind::NonCanonical); }
            return Self::new(num, denmag as i64).map_err(|_| RationalError::Overflow);
        }
        Ok(Self { num, den: 1 })
    }
}

        if self.den == 1 { write!(f, "{}", self.num) } else { write!(f, "{}/{}", self.num, self.den) }
    }

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }
    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }
    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        self.data.get(row * self.cols + col).copied().ok_or(MatrixError::Index { row, col })
    }
    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for c in 0..self.cols {
            for r in 0..self.rows {
                data.push(self.data[r * self.cols + c]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch { left: (self.rows, self.cols), right: (rhs.rows, rhs.cols) });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                out.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: out })
    }

    fn rref_limit(&self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let mut work = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        for c in 0..candidate_cols {
            if p == self.rows { break; }
            let mut best: Option<usize> = None;
            for r in p..self.rows {
                let x = work.data[r * work.cols + c];
                if x.num == 0 { continue; }
                let key = (x.den, x.num.unsigned_abs(), r);
                if best.map_or(true, |br| {
                    let y = work.data[br * work.cols + c];
                    key < (y.den, y.num.unsigned_abs(), br)
                }) {
                    best = Some(r);
                }
            }
            let Some(q) = best else { continue };
            if q != p {
                for j in 0..work.cols { work.data.swap(p * work.cols + j, q * work.cols + j); }
                swaps.push((p, q));
            }
            let pivot = work.data[p * work.cols + c];
            for j in 0..work.cols {
                let cur = work.data[p * work.cols + j];
                work.data[p * work.cols + j] = cur.div(&pivot).map_err(|source| MatrixError::Arithmetic { row: p, col: j, step: p, source })?;
            }
            for r in 0..work.rows {
                if r == p { continue; }
                let factor = work.data[r * work.cols + c];
                if factor.num == 0 { continue; }
                for j in 0..work.cols {
                    let product = factor.mul(&work.data[p * work.cols + j]).map_err(|source| MatrixError::Arithmetic { row: r, col: j, step: p, source })?;
                    work.data[r * work.cols + j] = work.data[r * work.cols + j].sub(&product).map_err(|source| MatrixError::Arithmetic { row: r, col: j, step: p, source })?;
                }
            }
            pivots.push(c);
            p += 1;
        }
        Ok(Reduction { matrix: work, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> { self.rref_limit(self.cols) }
    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..r * n + n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let augmented = Self { rows: n, cols: width, data };
        let red = augmented.rref_limit(n)?;
        if red.pivots.len() < n { return Err(MatrixError::Singular { rank: red.pivots.len() }); }
        let mut right = Vec::with_capacity(n * n);
        for r in 0..n {
            right.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Self { rows: n, cols: n, data: right })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}
impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational { num: i64, den: i64 }

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 { let t = a % b; a = b; b = t; }
    a
}
impl Rational {
    fn build(n: i128, d: i128) -> Result<Self, RationalError> {
        if d == 0 { return Err(RationalError::ZeroDenominator); }
        let (n, d) = if d < 0 { (-n, -d) } else { (n, d) };
        if n == 0 { return Ok(Self { num: 0, den: 1 }); }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        let n = n / g;
        let d = d / g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 { Err(RationalError::ZeroDenominator) } else { Self::build(num as i128, den as i128) }
    }
    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }
    pub fn add(&self, r: &Self) -> Result<Self, RationalError> {
        Self::build(self.num as i128 * r.den as i128 + r.num as i128 * self.den as i128, self.den as i128 * r.den as i128)
    }
    pub fn sub(&self, r: &Self) -> Result<Self, RationalError> {
        Self::build(self.num as i128 * r.den as i128 - r.num as i128 * self.den as i128, self.den as i128 * r.den as i128)
    }
    pub fn mul(&self, r: &Self) -> Result<Self, RationalError> {
        Self::build(self.num as i128 * r.num as i128, self.den as i128 * r.den as i128)
    }
    pub fn div(&self, r: &Self) -> Result<Self, RationalError> {
        if r.num == 0 { return Err(RationalError::DivideByZero); }
        Self::build(self.num as i128 * r.den as i128, self.den as i128 * r.num as i128)
    }
    pub fn neg(&self) -> Result<Self, RationalError> { Self::build(-(self.num as i128), self.den as i128) }

    pub fn parse(s: &str) -> Result<Self, RationalError> {
        let b = s.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() { return Err(err(0, ParseKind::Empty)); }
        let mut i = usize::from(b[0] == b'-');
        if i == b.len() { return Err(err(i, ParseKind::InvalidByte)); }
        let ns = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == ns { return Err(err(i, ParseKind::InvalidByte)); }
        let ne = i;
        let (ds, de) = if i < b.len() {
            if b[i] != b'/' { return Err(err(i, ParseKind::InvalidByte)); }
            i += 1;
            let ds = i;
            if i == b.len() { return Err(err(i, ParseKind::InvalidByte)); }
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == ds { return Err(err(i, ParseKind::InvalidByte)); }
            if i < b.len() { return Err(err(i, ParseKind::InvalidByte)); }
            (Some(ds), Some(i))
        } else { (None, None) };

        if ne - ns > 1 && b[ns] == b'0' { return Err(err(ns + 1, ParseKind::NonCanonical)); }
        if b[0] == b'-' && ne - ns == 1 && b[ns] == b'0' { return Err(err(0, ParseKind::NonCanonical)); }
        if let (Some(ds), Some(de)) = (ds, de) {
            if de - ds > 1 && b[ds] == b'0' { return Err(err(ds + 1, ParseKind::NonCanonical)); }
        }
        let parse_mag = |start: usize, end: usize, limit: u128| -> Option<u128> {
            let mut x = 0u128;
            for &ch in &b[start..end] {
                x = x.checked_mul(10)?.checked_add((ch - b'0') as u128)?;
                if x > limit { return None; }
            }
            Some(x)
        };
        let neg = b[0] == b'-';
        let nm = parse_mag(ns, ne, if neg { 1u128 << 63 } else { i64::MAX as u128 })
            .ok_or_else(|| err(0, ParseKind::OutOfRange))?;
        let n = if neg { if nm == 1u128 << 63 { i64::MIN } else { -(nm as i64) } } else { nm as i64 };
        if let (Some(ds), Some(de)) = (ds, de) {
            let dm = parse_mag(ds, de, i64::MAX as u128).ok_or_else(|| err(ds, ParseKind::OutOfRange))?;
            if dm == 0 { return Err(err(ds, ParseKind::ZeroDenominator)); }
            if dm == 1 { return Err(err(ds - 1, ParseKind::NonCanonical)); }
            if n == 0 { return Err(err(ds - 1, ParseKind::NonCanonical)); }
            if gcd_u128(nm, dm) != 1 { return Err(err(ds - 1, ParseKind::NonCanonical)); }
            Ok(Self { num: n, den: dm as i64 })
        } else { Ok(Self { num: n, den: 1 }) }
    }
}

pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix { rows: usize, cols: usize, data: Vec<Rational> }

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected { return Err(MatrixError::DataLength { expected, actual: data.len() }); }
        Ok(Self { rows, cols, data })
    }
    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }
    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols { return Err(MatrixError::Index { row, col }); }
        Ok(self.data[row * self.cols + col])
    }
    pub fn transpose(&self) -> Self {
        let mut d = Vec::with_capacity(self.data.len());
        for c in 0..self.cols { for r in 0..self.rows { d.push(self.data[r * self.cols + c]); } }
        Self { rows: self.cols, cols: self.rows, data: d }
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows { return Err(MatrixError::DimensionMismatch { left: (self.rows,self.cols), right: (rhs.rows,rhs.cols) }); }
        let count = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(count);
        for r in 0..self.rows { for c in 0..rhs.cols {
            let mut acc = Self::zero();
            for k in 0..self.cols {
                let x = self.data[r*self.cols+k].mul(&rhs.data[k*rhs.cols+c])
                    .map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                acc = acc.add(&x).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
            }
            out.push(acc);
        }}
        Ok(Self { rows:self.rows, cols:rhs.cols, data:out })
    }
    fn zero() -> Rational { Rational { num:0, den:1 } }

    fn eliminate(&self, candidates: usize) -> Result<Reduction, MatrixError> {
        let mut m = self.clone(); let mut pivots=Vec::new(); let mut swaps=Vec::new(); let mut p=0;
        for c in 0..candidates {
            if p == m.rows { break; }
            let q = (p..m.rows).filter(|&r| m.data[r*m.cols+c].num != 0).min_by_key(|&r| {
                let x=m.data[r*m.cols+c]; (x.den,x.num.unsigned_abs(),r)
            });
            let Some(q)=q else { continue };
            if p!=q { for j in 0..m.cols { m.data.swap(p*m.cols+j,q*m.cols+j); } swaps.push((p,q)); }
            let pivot=m.data[p*m.cols+c];
            for j in 0..m.cols {
                let value=m.data[p*m.cols+j].div(&pivot).map_err(|source| MatrixError::Arithmetic {row:p,col:j,step:p,source})?;
                m.data[p*m.cols+j]=value;
            }
            for r in 0..m.rows {
                if r==p { continue; }
                let factor=m.data[r*m.cols+c];
                if factor.num==0 { continue; }
                for j in 0..m.cols {
                    let prod=factor.mul(&m.data[p*m.cols+j]).map_err(|source| MatrixError::Arithmetic {row:r,col:j,step:p,source})?;
                    m.data[r*m.cols+j]=m.data[r*m.cols+j].sub(&prod).map_err(|source| MatrixError::Arithmetic {row:r,col:j,step:p,source})?;
                }
            }
            pivots.push(c); p+=1;
        }
        Ok(Reduction { matrix:m,pivots,swaps })
    }
    pub fn rref(&self) -> Result<Reduction, MatrixError> { self.eliminate(self.cols) }
    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }
    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows!=self.cols { return Err(MatrixError::NotSquare {rows:self.rows,cols:self.cols}); }
        let n=self.rows;
        let width=n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let count=n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut d=Vec::with_capacity(count);
        for r in 0..n {
            d.extend_from_slice(&self.data[r*n..r*n+n]);
            for c in 0..n { d.push(if r==c { Rational {num:1,den:1} } else {Self::zero()}); }
        }
        let red=Self {rows:n,cols:width,data:d}.eliminate(n)?;
        if red.pivots.len()<n { return Err(MatrixError::Singular {rank:red.pivots.len()}); }
        let mut right=Vec::with_capacity(n*n);
        for r in 0..n { right.extend_from_slice(&red.matrix.data[r*width+n..r*width+width]); }
        Ok(Self {rows:n,cols:n,data:right})
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction { pub matrix: Matrix, pub pivots: Vec<usize>, pub swaps: Vec<(usize,usize)> }
impl Reduction { pub fn rank(&self) -> usize { self.pivots.len() } }
