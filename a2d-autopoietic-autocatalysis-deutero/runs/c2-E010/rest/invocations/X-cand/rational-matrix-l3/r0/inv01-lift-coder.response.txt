- Syntax errors take precedence over every canonicality and magnitude error; parsing validates the whole byte string first.
- Canonicality checks run in the specified order, and parse never normalizes fractions.
- Rational operations use wide intermediates, reduce the exact result, then check representable bounds.
- Matrix multiplication and elimination report the first bounded scalar-operation error in the specified order.
- Empty dimensions preserve their explicit shapes; algorithms avoid work over huge dimensions when there are no output entries.
- Inverse checks augmented width and storage size before elimination, and reports arithmetic failures before singularity.

pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= g;
        den /= g;
        if num < i64::MIN as i128 || num > i64::MAX as i128 || den > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: num as i64, den: den as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        Self::from_i128(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() { return Err(err(0, ParseKind::Empty)); }

        let mut i = 0;
        if b[0] == b'-' { i = 1; }
        let ns = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == ns {
            return Err(err(i, if i == b.len() { ParseKind::InvalidByte } else { ParseKind::InvalidByte }));
        }
        let ne = i;
        let mut ds = None;
        let mut de = None;
        if i < b.len() {
            if b[i] != b'/' { return Err(err(i, ParseKind::InvalidByte)); }
            let slash = i;
            i += 1;
            let start = i;
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == start {
                return Err(err(i, ParseKind::InvalidByte));
            }
            ds = Some((start, slash));
            de = Some(i);
            if i < b.len() { return Err(err(i, ParseKind::InvalidByte)); }
        }

        let negative = b[0] == b'-';
        if ne - ns >= 2 && b[ns] == b'0' {
            return Err(err(ns + 1, ParseKind::NonCanonical));
        }
        if negative && ne - ns == 1 && b[ns] == b'0' {
            return Err(err(0, ParseKind::NonCanonical));
        }
        if let (Some((start, _)), Some(end)) = (ds, de) {
            if end - start >= 2 && b[start] == b'0' {
                return Err(err(start + 1, ParseKind::NonCanonical));
            }
        }

        let nmag = parse_u128(&b[ns..ne]).unwrap_or(u128::MAX);
        let nlimit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        if nmag > nlimit { return Err(err(0, ParseKind::OutOfRange)); }

        let dmag = if let (Some((start, _)), Some(end)) = (ds, de) {
            let v = parse_u128(&b[start..end]).unwrap_or(u128::MAX);
            if v > i64::MAX as u128 { return Err(err(start, ParseKind::OutOfRange)); }
            Some(v)
        } else { None };

        if let Some(d) = dmag {
            if d == 0 {
                return Err(err(ds.unwrap().0, ParseKind::ZeroDenominator));
            }
            let slash = ds.unwrap().1;
            if d == 1 { return Err(err(slash, ParseKind::NonCanonical)); }
            if nmag == 0 { return Err(err(slash, ParseKind::NonCanonical)); }
            if gcd_u128(nmag, d) > 1 { return Err(err(slash, ParseKind::NonCanonical)); }
        }

        let signed = if negative { -(nmag as i128) } else { nmag as i128 };
        Ok(Self { num: signed as i64, den: dmag.unwrap_or(1) as i64 })
    }

pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data })
    }

    fn eliminate(&self, candidate_cols: usize) -> Result<(Self, Vec<usize>, Vec<(usize, usize)>), MatrixError> {
        let mut out = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        for col in 0..candidate_cols {
            if p == out.rows { break; }
            let mut best: Option<(usize, i64, u64, usize)> = None;
            for row in p..out.rows {
                let v = out.data[row * out.cols + col];
                if v.num == 0 { continue; }
                let key = (v.den, v.num.unsigned_abs(), row);
                if best.map_or(true, |(_, d, n, r)| key < (d, n, r)) {
                    best = Some((row, v.den, v.num.unsigned_abs(), row));
                }
            }
            let Some((q, _, _, _)) = best else { continue };
            if q != p {
                for c in 0..out.cols { out.data.swap(p * out.cols + c, q * out.cols + c); }
                swaps.push((p, q));
            }
            let pivot = out.data[p * out.cols + col];
            for c in 0..out.cols {
                let v = out.data[p * out.cols + c];
                out.data[p * out.cols + c] = v.div(&pivot).map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
            }
            for row in 0..out.rows {
                if row == p { continue; }
                let factor = out.data[row * out.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..out.cols {
                    let product = factor.mul(&out.data[p * out.cols + c]).map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                    out.data[row * out.cols + c] = out.data[row * out.cols + c].sub(&product).map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok((out, pivots, swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let (matrix, pivots, swaps) = self.eliminate(self.cols)?;
        Ok(Reduction { matrix, pivots, swaps })
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let augmented = Self { rows: n, cols: width, data };
        let (reduced, pivots, _) = augmented.eliminate(n)?;
        if pivots.len() < n { return Err(MatrixError::Singular { rank: pivots.len() }); }
        let mut right = Vec::with_capacity(n * n);
        for r in 0..n {
            right.extend_from_slice(&reduced.data[r * width + n..r * width + width]);
        }
        Ok(Self { rows: n, cols: n, data: right })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

pub struct Rational { num: i64, den: i64 }

fn gcd128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 { let r = a % b; a = b; b = r; }
    a
}

impl Rational {
    fn build(n: i128, d: i128) -> Result<Self, RationalError> {
        if d == 0 { return Err(RationalError::ZeroDenominator); }
        let (mut n, mut d) = if d < 0 { (-n, -d) } else { (n, d) };
        if n == 0 { return Ok(Self { num: 0, den: 1 }); }
        let g = gcd128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> { Self::build(num as i128, den as i128) }
    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }
    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::build(self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128, self.den as i128 * rhs.den as i128)
    }
    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::build(self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128, self.den as i128 * rhs.den as i128)
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::build(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }
    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        Self::build(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }
    pub fn neg(&self) -> Result<Self, RationalError> { Self::build(-(self.num as i128), self.den as i128) }

    pub fn parse(s: &str) -> Result<Self, RationalError> {
        let b = s.as_bytes();
        let error = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() { return Err(error(0, ParseKind::Empty)); }
        let neg = b[0] == b'-';
        let start = if neg { 1 } else { 0 };
        let mut i = start;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == start { return Err(error(i, ParseKind::InvalidByte)); }
        let num_end = i;
        let slash = if i < b.len() && b[i] == b'/' { Some(i) } else { None };
        let den_range = if let Some(slash) = slash {
            i = slash + 1;
            let ds = i;
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if ds == i { return Err(error(i, ParseKind::InvalidByte)); }
            if i < b.len() { return Err(error(i, ParseKind::InvalidByte)); }
            Some((ds, i, slash))
        } else {
            if i < b.len() { return Err(error(i, ParseKind::InvalidByte)); }
            None
        };
        let ns = start;
        let nl = num_end - ns;
        if nl > 1 && b[ns] == b'0' { return Err(error(ns + 1, ParseKind::NonCanonical)); }
        if neg && nl == 1 && b[ns] == b'0' { return Err(error(0, ParseKind::NonCanonical)); }
        if let Some((ds, de, _)) = den_range {
            if de - ds > 1 && b[ds] == b'0' { return Err(error(ds + 1, ParseKind::NonCanonical)); }
        }
        let nm = decimal(&b[ns..num_end]).unwrap_or(u128::MAX);
        let lim = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        if nm > lim { return Err(error(0, ParseKind::OutOfRange)); }
        let dm = if let Some((ds, de, _)) = den_range {
            let v = decimal(&b[ds..de]).unwrap_or(u128::MAX);
            if v > i64::MAX as u128 { return Err(error(ds, ParseKind::OutOfRange)); }
            Some(v)
        } else { None };
        if let Some(d) = dm {
            let (ds, _, sl) = den_range.unwrap();
            if d == 0 { return Err(error(ds, ParseKind::ZeroDenominator)); }
            if d == 1 { return Err(error(sl, ParseKind::NonCanonical)); }
            if nm == 0 { return Err(error(sl, ParseKind::NonCanonical)); }
            if gcd128(nm, d) != 1 { return Err(error(sl, ParseKind::NonCanonical)); }
        }
        let signed = if neg { -(nm as i128) } else { nm as i128 };
        Ok(Self { num: signed as i64, den: dm.unwrap_or(1) as i64 })
    }
}

impl Matrix {

        let n = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if n != data.len() { return Err(MatrixError::DataLength { expected: n, actual: data.len() }); }
        Ok(Self { rows, cols, data })

    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }
    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols { return Err(MatrixError::Index { row, col }); }
        Ok(self.data[row * self.cols + col])
    }
    pub fn transpose(&self) -> Self {
        let mut d = Vec::with_capacity(self.data.len());
        for r in 0..self.cols { for c in 0..self.rows { d.push(self.data[c * self.cols + r]); } }
        Self { rows: self.cols, cols: self.rows, data: d }
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch { left: (self.rows,self.cols), right: (rhs.rows,rhs.cols) });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut d = Vec::with_capacity(len);
        let z = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut sum = z;
                for k in 0..self.cols {
                    let product = self.data[r*self.cols+k].mul(&rhs.data[k*rhs.cols+c])
                        .map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                    sum = sum.add(&product)
                        .map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                }
                d.push(sum);
            }
        }
        Ok(Self { rows:self.rows, cols:rhs.cols, data:d })
    }

    fn reduce_columns(&self, candidates: usize) -> Result<(Self, Vec<usize>, Vec<(usize,usize)>), MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0;
        for col in 0..candidates {
            if p == m.rows { break; }
            let mut chosen = None;
            for row in p..m.rows {
                let v = m.data[row*m.cols+col];
                if v.num == 0 { continue; }
                let key = (v.den, v.num.unsigned_abs(), row);
                if chosen.map_or(true, |(bd, bn, br)| key < (bd,bn,br)) {
                    chosen = Some((v.den,v.num.unsigned_abs(),row));
                }
            }
            let Some((_,_,q)) = chosen else { continue };
            if q != p {
                for c in 0..m.cols { m.data.swap(p*m.cols+c,q*m.cols+c); }
                swaps.push((p,q));
            }
            let pivot = m.data[p*m.cols+col];
            for c in 0..m.cols {
                let idx = p*m.cols+c;
                m.data[idx] = m.data[idx].div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row:p,col:c,step:p,source })?;
            }
            for row in 0..m.rows {
                if row == p { continue; }
                let factor = m.data[row*m.cols+col];
                if factor.num == 0 { continue; }
                for c in 0..m.cols {
                    let idx = row*m.cols+c;
                    let term = factor.mul(&m.data[p*m.cols+c])
                        .map_err(|source| MatrixError::Arithmetic { row,col:c,step:p,source })?;
                    m.data[idx] = m.data[idx].sub(&term)
                        .map_err(|source| MatrixError::Arithmetic { row,col:c,step:p,source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok((m,pivots,swaps))
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        let (matrix,pivots,swaps) = self.reduce_columns(self.cols)?;
        Ok(Reduction { matrix,pivots,swaps })
    }
    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }
    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols { return Err(MatrixError::NotSquare { rows:self.rows,cols:self.cols }); }
        let n = self.rows;
        let w = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let size = n.checked_mul(w).ok_or(MatrixError::ShapeOverflow)?;
        let mut d = Vec::with_capacity(size);
        for r in 0..n {
            d.extend_from_slice(&self.data[r*n..(r+1)*n]);
            for c in 0..n { d.push(if r == c { Rational { num:1,den:1 } } else { Rational { num:0,den:1 } }); }
        }
        let aug = Self { rows:n,cols:w,data:d };
        let (red,pivots,_) = aug.reduce_columns(n)?;
        if pivots.len() < n { return Err(MatrixError::Singular { rank:pivots.len() }); }
        let mut right = Vec::with_capacity(n*n);
        for r in 0..n { right.extend_from_slice(&red.data[r*w+n..r*w+w]); }
        Ok(Self { rows:n,cols:n,data:right })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction { pub matrix: Matrix, pub pivots: Vec<usize>, pub swaps: Vec<(usize,usize)> }
impl Reduction { pub fn rank(&self) -> usize { self.pivots.len() } }
