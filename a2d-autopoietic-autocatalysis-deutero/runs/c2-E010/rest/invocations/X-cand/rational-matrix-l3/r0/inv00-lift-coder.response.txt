- Parsing validates the entire byte grammar before any canonicality or range checks; byte offsets are used throughout.
- Fractions are reduced with widened integer arithmetic before bounds checks, including `i64::MIN`.
- Matrix multiplication reports dimension mismatch before shape overflow and follows the specified scalar operation order.
- RREF chooses pivots by denominator, absolute numerator, then current row; it updates every required column in order.
- Inverse checks augmented width and storage size before elimination, then reports arithmetic errors before singularity.
- Empty dimensions preserve their explicit shapes and avoid loops over unused large dimensions.

        if data.len() != expected {

    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }

            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                out.push(acc);
            }
        }
        Ok(Self { rows: self.rows, cols: rhs.cols, data: out })
    }

    fn eliminate(&self, candidate_cols: usize, update_cols: usize) -> Result<Reduction, MatrixError> {
        let mut work = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;

        for col in 0..candidate_cols {
            if p == work.rows { break; }
            let mut best: Option<(i64, u128, usize)> = None;
            for row in p..work.rows {
                let v = work.data[row * work.cols + col];
                if v.num == 0 { continue; }
                let key = (v.den, (v.num as i128).unsigned_abs(), row);
                if best.map_or(true, |old| key < old) {
                    best = Some(key);
                }
            }
            let Some((_, _, q)) = best else { continue; };
            if q != p {
                for c in 0..update_cols {
                    work.data.swap(p * work.cols + c, q * work.cols + c);
                }
                swaps.push((p, q));
            }

            let pivot = work.data[p * work.cols + col];
            let mut normalized = Vec::with_capacity(update_cols);
            for c in 0..update_cols {
                let value = work.data[p * work.cols + c].div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic { row: p, col: c, step: p, source })?;
                normalized.push(value);
            }
            for c in 0..update_cols {
                work.data[p * work.cols + c] = normalized[c];
            }

            for row in 0..work.rows {
                if row == p { continue; }
                let factor = work.data[row * work.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..update_cols {
                    let product = factor.mul(&work.data[p * work.cols + c])
                        .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                    let value = work.data[row * work.cols + c].sub(&product)
                        .map_err(|source| MatrixError::Arithmetic { row, col: c, step: p, source })?;
                    work.data[row * work.cols + c] = value;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix: work, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(total);
        for r in 0..n {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            for c in 0..n {
                data.push(if r == c { Rational { num: 1, den: 1 } } else { Rational { num: 0, den: 1 } });
            }
        }
        let aug = Matrix { rows: n, cols: width, data };
        let red = aug.eliminate(n, width)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut right = Vec::with_capacity(total / 2);
        for r in 0..n {
            right.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Matrix { rows: n, cols: n, data: right })
    }

}

impl Rational {
    fn normalized(mut n: i128, mut d: i128) -> Result<Self, RationalError> {
        if d == 0 { return Err(RationalError::ZeroDenominator); }
        if d < 0 { n = -n; d = -d; }
        if n == 0 { return Ok(Self { num: 0, den: 1 }); }
        let g = gcd(n.unsigned_abs(), d as u128) as i128;
        n /= g; d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {

        }
        Ok(Self { num: n as i64, den: d as i64 })

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    }
    pub fn sub(&self, x: &Self) -> Result<Self, RationalError> {
        Self::normalized(self.num as i128 * x.den as i128 - x.num as i128 * self.den as i128, self.den as i128 * x.den as i128)
    }
    pub fn mul(&self, x: &Self) -> Result<Self, RationalError> {
        Self::normalized(self.num as i128 * x.num as i128, self.den as i128 * x.den as i128)
    }
    pub fn div(&self, x: &Self) -> Result<Self, RationalError> {
        if x.num == 0 { return Err(RationalError::DivideByZero); }
        Self::normalized(self.num as i128 * x.den as i128, self.den as i128 * x.num as i128)
    }
    pub fn neg(&self) -> Result<Self, RationalError> { Self::normalized(-(self.num as i128), self.den as i128) }
    pub fn parse(s: &str) -> Result<Self, RationalError> {
        let b = s.as_bytes();
        let fail = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() { return Err(fail(0, ParseKind::Empty)); }
        let mut i = usize::from(b[0] == b'-');
        let ns = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == ns { return Err(fail(i, ParseKind::InvalidByte)); }
        let ne = i;
        let mut ds = None;
        let mut de = None;
        if i < b.len() {
            if b[i] != b'/' { return Err(fail(i, ParseKind::InvalidByte)); }
            i += 1; ds = Some(i);
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == ds.unwrap() { return Err(fail(i, ParseKind::InvalidByte)); }
            de = Some(i);
            if i < b.len() { return Err(fail(i, ParseKind::InvalidByte)); }
        }
        let neg = b[0] == b'-';
        if ne - ns >= 2 && b[ns] == b'0' { return Err(fail(ns + 1, ParseKind::NonCanonical)); }
        if neg && ne - ns == 1 && b[ns] == b'0' { return Err(fail(0, ParseKind::NonCanonical)); }
        if let Some(x) = ds {
            if de.unwrap() - x >= 2 && b[x] == b'0' { return Err(fail(x + 1, ParseKind::NonCanonical)); }
        }
        let magnitude = |a: usize, z: usize, limit: u128| -> Option<u128> {
            let mut v = 0u128;
            for &c in &b[a..z] {
                v = v.checked_mul(10)?.checked_add((c - b'0') as u128)?;
                if v > limit { return None; }
            }
            Some(v)
        };
        let nm = magnitude(ns, ne, if neg { 1u128 << 63 } else { i64::MAX as u128 })
            .ok_or_else(|| fail(0, ParseKind::OutOfRange))?;
        let n = if neg { if nm == 1u128 << 63 { i64::MIN } else { -(nm as i64) } } else { nm as i64 };
        let Some(x) = ds else { return Ok(Self { num: n, den: 1 }); };
        let dm = magnitude(x, de.unwrap(), i64::MAX as u128)
            .ok_or_else(|| fail(x, ParseKind::OutOfRange))?;
        if dm == 0 { return Err(fail(x, ParseKind::ZeroDenominator)); }
        if dm == 1 || n == 0 { return Err(fail(x - 1, ParseKind::NonCanonical)); }
        if gcd(nm, dm) > 1 { return Err(fail(x - 1, ParseKind::NonCanonical)); }
        Ok(Self { num: n, den: dm as i64 })
    }

impl Matrix {

        Self { rows: self.cols, cols: self.rows, data: d }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch { left: (self.rows, self.cols), right: (rhs.rows, rhs.cols) });
        }
        let count = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(count);
        for r in 0..self.rows { for c in 0..rhs.cols {
            let mut sum = Rational { num: 0, den: 1 };
            for k in 0..self.cols {
                let a = self.data[r * self.cols + k];
                let b = rhs.data[k * rhs.cols + c];
                let t = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row:r, col:c, step:k, source })?;
                sum = sum.add(&t).map_err(|source| MatrixError::Arithmetic { row:r, col:c, step:k, source })?;
            }
            out.push(sum);
        }}
        Ok(Self { rows:self.rows, cols:rhs.cols, data:out })
    }

    fn reduce(&self, candidates: usize, width: usize) -> Result<Reduction, MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0;
        for col in 0..candidates {
            if p == m.rows { break; }
            let mut chosen = None;
            for r in p..m.rows {
                let x = m.data[r * m.cols + col];
                if x.num != 0 {
                    let key = (x.den, (x.num as i128).unsigned_abs(), r);
                    if chosen.map_or(true, |(d, a, rr)| key < (d, a, rr)) { chosen = Some(key); }
                }
            }
            let Some((_, _, q)) = chosen else { continue; };
            if q != p {
                for c in 0..width { m.data.swap(p * m.cols + c, q * m.cols + c); }
                swaps.push((p, q));
            }
            let pivot = m.data[p * m.cols + col];
            let mut normalized = Vec::with_capacity(width);
            for c in 0..width {
                normalized.push(m.data[p * m.cols + c].div(&pivot).map_err(|source|
                    MatrixError::Arithmetic { row:p, col:c, step:p, source })?);
            }
            for c in 0..width { m.data[p * m.cols + c] = normalized[c]; }
            for r in 0..m.rows {
                if r == p { continue; }
                let factor = m.data[r * m.cols + col];
                if factor.num == 0 { continue; }
                for c in 0..width {
                    let product = factor.mul(&m.data[p * m.cols + c]).map_err(|source|
                        MatrixError::Arithmetic { row:r, col:c, step:p, source })?;
                    m.data[r * m.cols + c] = m.data[r * m.cols + c].sub(&product).map_err(|source|
                        MatrixError::Arithmetic { row:r, col:c, step:p, source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix:m, pivots, swaps })
    }
    pub fn rref(&self) -> Result<Reduction, MatrixError> { self.reduce(self.cols, self.cols) }
    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }
    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols { return Err(MatrixError::NotSquare { rows:self.rows, cols:self.cols }); }
        let n = self.rows;
        let w = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let total = n.checked_mul(w).ok_or(MatrixError::ShapeOverflow)?;
        let mut d = Vec::with_capacity(total);
        for r in 0..n {
            d.extend_from_slice(&self.data[r*n..(r+1)*n]);
            for c in 0..n { d.push(if r == c { Rational { num:1, den:1 } } else { Rational { num:0, den:1 } }); }
        }
        let aug = Self { rows:n, cols:w, data:d };
        let red = aug.reduce(n, w)?;
        let rank = red.pivots.len();
        if rank < n { return Err(MatrixError::Singular { rank }); }
        let mut result = Vec::with_capacity(n*n);
        for r in 0..n { result.extend_from_slice(&red.matrix.data[r*w+n..(r+1)*w]); }
        Ok(Self { rows:n, cols:n, data:result })
    }
}
