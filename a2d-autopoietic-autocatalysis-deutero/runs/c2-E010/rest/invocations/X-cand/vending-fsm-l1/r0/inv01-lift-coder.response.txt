- Catalog validation checks empty input first, then each product in order: ID syntax, duplicate ID, and price.
- IDs are 1–8 lowercase ASCII letters; runtime lookup is exact and case sensitive.
- Coin validation precedes the inclusive 100 credit limit; rejected calls leave state unchanged.
- Vending checks selection, underpayment, overpayment, then canonical tender; only exact canonical payment succeeds.
- Canonical tender uses greedy denominations 10, 4, 3, 1; stored escrow order remains unchanged.
- Refund returns escrow in reverse insertion order, including when the machine has no credit.

### Solution A

```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    ExactPayment { price: u16, credit: u16 },
    TenderMismatch {
        index: usize,
        expected: u16,
        found: u16,
    },
}

pub struct Machine {
    products: HashMap<String, u16>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=8).contains(&bytes.len()) && bytes.iter().all(|b| b.is_ascii_lowercase())
}

fn canonical_tender(mut amount: u16) -> Vec<u16> {
    let mut coins = Vec::new();
    for denomination in [10u16, 4, 3, 1] {
        while amount >= denomination {
            coins.push(denomination);
            amount -= denomination;
        }
    }
    coins
}

impl Machine {
    pub fn new(products: Vec<Product>) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut catalog = HashMap::with_capacity(products.len());
        for product in products {
            if !valid_id(&product.id) {
                return Err(Error::BadId { id: product.id });
            }
            if catalog.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id });
            }
            catalog.insert(product.id, product.price);
        }

        Ok(Self {
            products: catalog,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if ![1, 3, 4, 10].contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(_) = self.products.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let price = self.products[id];
        let credit = self.credit;

        if credit < price {
            return Err(Error::Underpaid { price, credit });
        }
        if credit > price {
            return Err(Error::ExactPayment { price, credit });
        }

        let expected = canonical_tender(price);
        let mut found = self.escrow.clone();
        found.sort_unstable_by(|a, b| b.cmp(a));

        for index in 0..expected.len() {
            if expected[index] != found[index] {
                return Err(Error::TenderMismatch {
                    index,
                    expected: expected[index],
                    found: found[index],
                });
            }
        }

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: Vec::new(),
        };
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}
```

```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    ExactPayment { price: u16, credit: u16 },
    TenderMismatch {
        index: usize,
        expected: u16,
        found: u16,
    },
}

pub struct Machine {
    products: Vec<Product>,
    selected: Option<String>,
    credit: u16,
    escrow: Vec<u16>,
}

fn valid_id(id: &str) -> bool {
    let length = id.len();
    (1..=8).contains(&length) && id.bytes().all(|byte| byte.is_ascii_lowercase())
}

fn canonical_tender(mut amount: u16) -> Vec<u16> {
    let mut result = Vec::new();
    while amount >= 10 {
        result.push(10);
        amount -= 10;
    }
    while amount >= 4 {
        result.push(4);
        amount -= 4;
    }
    while amount >= 3 {
        result.push(3);
        amount -= 3;
    }
    while amount > 0 {
        result.push(1);
        amount -= 1;
    }
    result
}

impl Machine {
    pub fn new(products: Vec<Product>) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut accepted = Vec::with_capacity(products.len());
        for product in products {
            if !valid_id(&product.id) {
                return Err(Error::BadId { id: product.id });
            }
            if accepted.iter().any(|saved: &Product| saved.id == product.id) {
                return Err(Error::DuplicateId { id: product.id });
            }
            if product.price == 0 || product.price > 100 {
                return Err(Error::BadPrice { id: product.id });
            }
            accepted.push(product);
        }

        Ok(Self {
            products: accepted,
            selected: None,
            credit: 0,
            escrow: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if coin != 1 && coin != 3 && coin != 4 && coin != 10 {
            return Err(Error::BadCoin { coin });
        }
        if self.credit > 100 - coin {
            return Err(Error::CreditLimit);
        }
        self.escrow.push(coin);
        self.credit += coin;
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let product_exists = self.products.iter().any(|product| product.id == id);
        if !product_exists {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let selected_id = match self.selected.as_deref() {
            Some(id) => id,
            None => return Err(Error::NoSelection),
        };
        let product = self
            .products
            .iter()
            .find(|product| product.id == selected_id)
            .expect("selected IDs always refer to catalog products");
        let price = product.price;
        let credit = self.credit;

        if credit < price {
            return Err(Error::Underpaid { price, credit });
        }
        if credit > price {
            return Err(Error::ExactPayment { price, credit });
        }

        let expected = canonical_tender(price);
        let mut found = self.escrow.clone();
        found.sort_unstable_by(|left, right| right.cmp(left));
        for index in 0..expected.len() {
            if expected[index] != found[index] {
                return Err(Error::TenderMismatch {
                    index,
                    expected: expected[index],
                    found: found[index],
                });
            }
        }

        let receipt = Receipt {
            id: selected_id.to_owned(),
            price,
            change: Vec::new(),
        };
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.selected = None;
        self.credit = 0;
        self.escrow.clear();
        Ok(coins)
    }
}
```
