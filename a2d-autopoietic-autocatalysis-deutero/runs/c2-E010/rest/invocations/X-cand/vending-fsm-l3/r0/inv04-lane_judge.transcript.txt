const DENOMS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

fn change_for(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut greedy_counts = [0u16; 4];
    let mut remaining = amount as u32;

    for i in (0..4).rev() {
        let coin = DENOMS[i] as u32;
        let take = (tubes[i] as u32).min(remaining / coin);
        greedy_counts[i] = take as u16;
        remaining -= take * coin;
    }

    let counts = if remaining == 0 {
        greedy_counts
    } else {
        let mut best: Option<[u16; 4]> = None;

        fn search(
            index: usize,
            remaining: u32,
            tubes: &[u16; 4],
            counts: &mut [u16; 4],
            best: &mut Option<[u16; 4]>,
        ) {
            if index == 4 {
                if remaining != 0 {
                    return;
                }
                let candidate = *counts;
                let candidate_total: u32 = candidate.iter().map(|&x| x as u32).sum();
                let replace = match best {
                    None => true,
                    Some(current) => {
                        let current_total: u32 = current.iter().map(|&x| x as u32).sum();
                        candidate_total < current_total
                            || (candidate_total == current_total
                                && (3..=0).rev().any(|_| false))
                    }
                };

                let replace = if let Some(current) = best.as_ref() {
                    let current_total: u32 = current.iter().map(|&x| x as u32).sum();
                    if candidate_total != current_total {
                        candidate_total < current_total
                    } else {
                        let mut greater = false;
                        let mut different = false;
                        for i in (0..4).rev() {
                            if candidate[i] != current[i] {
                                greater = candidate[i] > current[i];
                                different = true;
                                break;
                            }
                        }
                        different && greater
                    }
                } else {
                    true
                };

                if replace {
                    *best = Some(candidate);
                }
                return;
            }

            let coin = DENOMS[index] as u32;
            let max_take = (tubes[index] as u32).min(remaining / coin);
            for take in 0..=max_take {
                counts[index] = take as u16;
                search(
                    index + 1,
                    remaining - take * coin,
                    tubes,
                    counts,
                    best,
                );
            }
            counts[index] = 0;
        }

        let _ = {
            let _ = best;
            false
        };
        let mut chosen = None;
        let mut counts = [0u16; 4];
        search(0, amount as u32, tubes, &mut counts, &mut best);
        chosen = best;
        chosen?
    };

    let mut result = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            result.push(DENOMS[i]);
        }
    }
    Some(result)
}

    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {

            return Err(Error::EmptyCatalog);

                return Err(Error::BadId {

                });
            }

            }
        }

            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,

        let Some(index) = DENOMS.iter().position(|&d| d == coin) else {
            return Err(Error::BadCoin { coin });
        };
        let new_credit = self.credit as u32 + coin as u32;
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit = new_credit as u16;
        self.escrow.push(coin);
        self.log(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        let _ = index;
        Ok(())
    }

            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(id.to_owned());
        self.log(EventKind::Selected { id: id.to_owned() });
        Ok(())

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let index = self.product_index(&id).expect("selected ID is in catalog");

            return Err(Error::SoldOut { id });
        }
        if self.credit < product.price {
            return Err(Error::Underpaid {
                price: product.price,
                credit: self.credit,
            });

        let amount = self.credit - price;
        let change = change_for(amount, &self.tubes).ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for coin in &change {
            let i = DENOMS.iter().position(|&d| d == *coin).expect("valid change coin");
            proposed[i] -= 1;
        }
        for coin in &self.escrow {
            let i = DENOMS.iter().position(|&d| d == *coin).expect("valid escrow coin");
            proposed[i] = proposed[i] + 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        self.products[index].stock -= 1;
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        let receipt = Receipt {
            id: id.clone(),
            price,
            change: change.clone(),
        };
        self.log(EventKind::Vended {
            id,
            price,
            change,
        });
        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }

        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = self.products[index].stock as u32 + count as u32;
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }

        self.products[index].stock = new_stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: new_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

        self.products.iter().position(|p| p.id == id)
    }

    fn log(&mut self, kind: EventKind) {
        let seq = self.events.len();
        self.events.push(Event { seq, kind });
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let total = u32::from(self.credit) + u32::from(coin);
        if total > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit = total as u16;
        self.escrow.push(coin);
        self.log(EventKind::Inserted {
            coin,
            credit: self.credit,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.find(id).is_none() {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        self.log(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = self.selected.clone().ok_or(Error::NoSelection)?;
        let index = self.find(&id).expect("selected catalog product");
        if self.products[index].stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = self.products[index].price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        let amount = self.credit - price;
        let change = choose_change_dp(amount, &self.tubes)
            .ok_or(Error::NoChange { amount })?;

        let mut proposed = self.tubes;
        for &coin in &change {
            let i = DENOMS.iter().position(|&d| d == coin).expect("valid change coin");
            proposed[i] -= 1;
        }
        for &coin in &self.escrow {
            let i = DENOMS.iter().position(|&d| d == coin).expect("valid escrow coin");
            proposed[i] += 1;
        }
        for (i, &count) in proposed.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        self.products[index].stock -= 1;
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut result = self.products.clone();
        result.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        result
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let index = self.find(id).ok_or_else(|| Error::UnknownProduct {
            id: id.to_owned(),
        })?;
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if price == 0 || price > 100 {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[index].price;
        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let index = self.find(id).ok_or_else(|| Error::UnknownProduct {
            id: id.to_owned(),
        })?;
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let total = u32::from(self.products[index].stock) + u32::from(count);
        if total > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[index].stock = total as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: total as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }
}

// Greedy first, then a value-indexed dynamic program for minimum coin count.
// Equal-count solutions prefer more 10s, then 4s, then 3s, then 1s.
fn choose_change_dp(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / DENOMS[i]);
        greedy[i] = take;
        remaining -= take * DENOMS[i];
    }
    if remaining == 0 {
        return Some(to_descending(&greedy));
    }

    let target = usize::from(amount);
    let mut dp: Vec<Option<[u16; 4]>> = vec![None; target + 1];
    dp[0] = Some([0; 4]);

    for i in 0..4 {
        for _ in 0..tubes[i] {
            let value = usize::from(DENOMS[i]);
            for sum in (value..=target).rev() {
                if let Some(prev) = dp[sum - value] {
                    let mut candidate = prev;
                    candidate[i] += 1;
                    let should_replace = match dp[sum] {
                        None => true,
                        Some(current) => {
                            let nc: u32 = candidate.iter().map(|&n| u32::from(n)).sum();
                            let oc: u32 = current.iter().map(|&n| u32::from(n)).sum();
                            nc < oc
                                || (nc == oc
                                    && (0..4).rev().find_map(|j| {
                                        if candidate[j] != current[j] {
                                            Some(candidate[j] > current[j])
                                        } else {
                                            None
                                        }
                                    }) == Some(true))
                        }
                    };
                    if should_replace {
                        dp[sum] = Some(candidate);
                    }
                }
            }
        }
    }

    dp[target].map(|counts| to_descending(&counts))
}

fn to_descending(counts: &[u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            result.push(DENOMS[i]);
        }
    }
    result
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut rem = amount as u32;
    let mut greedy = [0u16; 4];
    for i in (0..4).rev() {
        let denom = DENOMS[i] as u32;
        let take = (tubes[i] as u32).min(rem / denom);
        greedy[i] = take as u16;
        rem -= take * denom;
    }

    let selected = if rem == 0 {
        greedy
    } else {
        let mut best: Option<[u16; 4]> = None;
        fn visit(
            idx: usize,
            rem: u32,
            tubes: &[u16; 4],
            cur: &mut [u16; 4],
            best: &mut Option<[u16; 4]>,
        ) {
            if idx == 4 {
                if rem != 0 {
                    return;
                }
                let n: u32 = cur.iter().map(|v| *v as u32).sum();
                let replace = match *best {
                    None => true,
                    Some(b) => {
                        let bn: u32 = b.iter().map(|v| *v as u32).sum();
                        if n != bn {
                            n < bn
                        } else {
                            (0..4).rev().find_map(|i| {
                                (cur[i] != b[i]).then_some(cur[i] > b[i])
                            }).unwrap_or(false)
                        }
                    }
                };
                if replace {
                    *best = Some(*cur);
                }
                return;
            }

            let denom = DENOMS[idx] as u32;
            let max = (tubes[idx] as u32).min(rem / denom);
            for n in 0..=max {
                cur[idx] = n as u16;
                visit(idx + 1, rem - n * denom, tubes, cur, best);
            }
            cur[idx] = 0;
        }

        let mut cur = [0u16; 4];
        visit(0, amount as u32, tubes, &mut cur, &mut best);
        best?
    };

    let mut result = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..selected[i] {
            result.push(DENOMS[i]);
        }
    }
    Some(result)
}

        self.products[index].price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(index) = self.product_index(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let new_stock = self.products[index].stock as u32 + count as u32;
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[index].stock = new_stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: new_stock as u16,
        });
        Ok(())
    }

    }

// Returns counts in ascending denomination order.
fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / DENOMS[i]);
        greedy[i] = take;
        remaining -= take * DENOMS[i];
    }
    if remaining == 0 {
        return Some(greedy);
    }

    let mut best: Option<[u16; 4]> = None;
    let mut best_count = u32::MAX;

    fn visit(
        i: i32,
        remaining: u16,
        tubes: &[u16; 4],
        counts: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
        best_count: &mut u32,
    ) {
        if i < 0 {
            if remaining != 0 {
                return;
            }
            let total: u32 = counts.iter().map(|&x| x as u32).sum();
            let better = match best {
                None => true,
                Some(old) => {
                    total < *best_count
                        || (total == *best_count
                            && (0..4).rev().find_map(|j| {
                                if counts[j] != old[j] {
                                    Some(counts[j] > old[j])
                                } else {
                                    None
                                }
                            }) == Some(true))
                }
            };
            if better {
                *best = Some(*counts);
                *best_count = total;
            }
            return;
        }

        let idx = i as usize;
        let denom = DENOMS[idx];
        let max_take = tubes[idx].min(remaining / denom);
        for take in 0..=max_take {
            counts[idx] = take;
            visit(
                i - 1,
                remaining - take * denom,
                tubes,
                counts,
                best,
                best_count,
            );
        }
        counts[idx] = 0;
    }

    visit(
        3,
        amount,
        tubes,
        &mut [0; 4],
        &mut best,
        &mut best_count,
    );
    best
}

const COINS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    (1..=8).contains(&id.len()) && id.bytes().all(|b| b.is_ascii_lowercase())
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }
        let mut catalog = BTreeMap::new();
        for product in products {
            if !valid_id(&product.id) {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if catalog.contains_key(&product.id) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit {
                    id: product.id.clone(),
                });
            }
            catalog.insert(product.id.clone(), product);
        }
        for (i, &count) in tubes.iter().enumerate() {
            if count > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }
        Ok(Self {
            catalog,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            events: Vec::new(),
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    fn log(&mut self, kind: EventKind) {
        self.events.push(Event {
            seq: self.events.len(),
            kind,
        });
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !COINS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let credit = self.credit as u32 + coin as u32;
        if credit > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit = credit as u16;
        self.escrow.push(coin);
        self.log(EventKind::Inserted { coin, credit: self.credit });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if !self.catalog.contains_key(id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        self.selected = Some(id.to_owned());
        self.log(EventKind::Selected { id: id.to_owned() });
        Ok(())
    }

    fn change(&self, amount: u16) -> Option<Vec<u16>> {
        if amount == 0 {
            return Some(Vec::new());
        }
        let mut rem = amount as u32;
        let mut chosen = [0u16; 4];
        for i in (0..4).rev() {
            let n = (rem / COINS[i] as u32).min(self.tubes[i] as u32);
            chosen[i] = n as u16;
            rem -= n * COINS[i] as u32;
        }
        if rem == 0 {
            return Some(to_vector(chosen));
        }

        let mut best: Option<([u16; 4], usize)> = None;
        let mut candidate = [0u16; 4];
        fn enumerate(
            i: usize,
            rest: u32,
            count: usize,
            tubes: &[u16; 4],
            v: &mut [u16; 4],
            best: &mut Option<([u16; 4], usize)>,
        ) {
            if i == 4 {
                if rest == 0 {
                    let replace = match best {
                        None => true,
                        Some((old, old_count)) => {
                            count < *old_count
                                || (count == *old_count
                                    && (0..4)
                                        .rev()
                                        .find_map(|j| {
                                            (v[j] != old[j]).then_some(v[j] > old[j])
                                        })
                                        .unwrap_or(false))
                        }
                    };
                    if replace {
                        *best = Some((*v, count));
                    }
                }
                return;
            }
            let denom = COINS[i] as u32;
            let limit = tubes[i].min((rest / denom) as u16);
            for n in 0..=limit {
                if let Some((_, best_count)) = best {
                    if count + n as usize > *best_count {
                        break;
                    }
                }
                v[i] = n;
                enumerate(
                    i + 1,
                    rest - n as u32 * denom,
                    count + n as usize,
                    tubes,
                    v,
                    best,
                );
            }
            v[i] = 0;
        }
        enumerate(0, amount as u32, 0, &self.tubes, &mut candidate, &mut best);
        best.map(|(counts, _)| to_vector(counts))
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let id = match self.selected.as_deref() {
            Some(id) => id.to_owned(),
            None => return Err(Error::NoSelection),
        };
        let (price, stock) = {
            let p = self.catalog.get(&id).expect("selected product exists");
            (p.price, p.stock)
        };
        if stock == 0 {
            return Err(Error::SoldOut { id });
        }
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }
        let amount = self.credit - price;
        let change = match self.change(amount) {
            Some(c) => c,
            None => return Err(Error::NoChange { amount }),
        };
        let mut proposed = self.tubes;
        for &coin in &change {
            let i = COINS.iter().position(|&c| c == coin).unwrap();
            proposed[i] -= 1;
        }
        for &coin in &self.escrow {
            let i = COINS.iter().position(|&c| c == coin).unwrap();
            proposed[i] += 1;
        }
        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: COINS[i] });
            }
        }
        self.catalog.get_mut(&id).expect("selected product exists").stock -= 1;
        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins = self.escrow.iter().rev().copied().collect::<Vec<_>>();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        self.log(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        self.catalog.values().cloned().collect()
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.catalog.contains_key(id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }

            return Err(Error::Busy);
        }

            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let p = self.catalog.get_mut(id).unwrap();
        let old = p.price;
        p.price = price;
        self.log(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.catalog.contains_key(id) {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        }
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let p = self.catalog.get_mut(id).unwrap();
        let new_stock = p.stock as u32 + count as u32;
        if new_stock > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        p.stock = new_stock as u16;
        self.log(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: new_stock as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }
}

    for i in (0..4).rev() {
        out.extend(std::iter::repeat(COINS[i]).take(counts[i] as usize));

                    break;

                    idx + 1,

                    next_used,
                    tubes,
                    candidate,
                    best,
                    best_total,
                    max_coins,
                );

            0,
            amount as u32,
            0,
            &self.tubes,
            &mut candidate,
            &mut best,
            &mut best_total,
            max_coins,
        );

pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

pub struct Machine {
    items: Vec<Product>,
    balance: u16,
    choice: Option<String>,
    inserted: Vec<u16>,
    tubes: [u16; 4],
    history: Vec<Event>,
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

fn valid_id(id: &str) -> bool {
    !id.is_empty() && id.len() <= 8 && id.bytes().all(|b| b.is_ascii_lowercase())
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }
        let mut prior = Vec::<String>::new();
        for p in &products {
            if !valid_id(&p.id) {
                return Err(Error::BadId { id: p.id.clone() });
            }
            if prior.iter().any(|x| x == &p.id) {
                return Err(Error::DuplicateId { id: p.id.clone() });
            }
            prior.push(p.id.clone());
            if p.price == 0 || p.price > 100 {
                return Err(Error::BadPrice { id: p.id.clone() });
            }
            if p.stock > 100 {
                return Err(Error::StockLimit { id: p.id.clone() });
            }
        }
        for (i, n) in tubes.iter().enumerate() {
            if *n > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        Ok(Self {
            items: products,
            balance: 0,
            choice: None,
            inserted: Vec::new(),
            tubes,
            history: Vec::new(),
        })
    }

    fn find(&self, id: &str) -> Option<usize> {
        self.items.iter().position(|p| p.id == id)
    }

    fn push_event(&mut self, kind: EventKind) {
        self.history.push(Event {
            seq: self.history.len(),
            kind,
        });
    }

    pub fn state(&self) -> State {
        State {
            credit: self.balance,
            selected: self.choice.clone(),
            escrow: self.inserted.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if !DENOMS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let sum = self.balance as u32 + coin as u32;
        if sum > 100 {
            return Err(Error::CreditLimit);
        }
        self.balance = sum as u16;
        self.inserted.push(coin);
        self.push_event(EventKind::Inserted {
            coin,
            credit: self.balance,
        });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(i) = self.find(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.choice.as_deref() == Some(self.items[i].id.as_str()) {
            return Err(Error::SameSelection);
        }
        self.choice = Some(self.items[i].id.clone());
        self.push_event(EventKind::Selected {
            id: self.items[i].id.clone(),
        });
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(id) = self.choice.clone() else {
            return Err(Error::NoSelection);
        };
        let Some(i) = self.find(&id) else {
            return Err(Error::NoSelection);
        };
        if self.items[i].stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = self.items[i].price;
        if self.balance < price {
            return Err(Error::Underpaid {
                price,
                credit: self.balance,
            });
        }
        let amount = self.balance - price;
        let counts = match change_counts(amount, &self.tubes) {
            Some(c) => c,
            None => return Err(Error::NoChange { amount }),
        };
        let mut next_tubes = self.tubes;
        for k in 0..4 {
            next_tubes[k] -= counts[k];
        }
        for &coin in &self.inserted {
            let k = DENOMS.iter().position(|&d| d == coin).unwrap_or(0);
            next_tubes[k] += 1;
        }
        for k in 0..4 {
            if next_tubes[k] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[k] });
            }
        }

        let mut change = Vec::new();
        for k in (0..4).rev() {
            for _ in 0..counts[k] {
                change.push(DENOMS[k]);
            }
        }

        self.next_commit_vend(i, id, price, change, next_tubes)
    }

    fn next_commit_vend(
        &mut self,
        index: usize,
        id: String,
        price: u16,
        change: Vec<u16>,
        tubes: [u16; 4],
    ) -> Result<Receipt, Error> {
        self.items[index].stock -= 1;
        self.tubes = tubes;
        self.balance = 0;
        self.choice = None;
        self.inserted.clear();
        self.push_event(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let coins: Vec<u16> = self.inserted.iter().rev().copied().collect();
        self.balance = 0;
        self.choice = None;
        self.inserted.clear();
        self.push_event(EventKind::Refunded {
            coins: coins.clone(),
        });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut v = self.items.clone();
        v.sort_by(|a, b| a.id.cmp(&b.id));
        v
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        let Some(i) = self.find(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.balance != 0 {
            return Err(Error::Busy);
        }
        if price == 0 || price > 100 {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.items[i].price;
        self.items[i].price = price;
        self.push_event(EventKind::PriceChanged {
            id: id.to_owned(),
            old,
            new: price,
        });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        let Some(i) = self.find(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let total = self.items[i].stock as u32 + count as u32;
        if total > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.items[i].stock = total as u16;
        self.push_event(EventKind::Restocked {
            id: id.to_owned(),
            added: count,
            stock: total as u16,
        });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.history
    }
}

// Dynamic-programming implementation. Tie breaking compares descending count
// vectors after minimizing total coin count.
fn change_counts(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    let mut greedy = [0u16; 4];
    let mut left = amount;
    for i in (0..4).rev() {
        let n = tubes[i].min(left / DENOMS[i]);
        greedy[i] = n;
        left -= n * DENOMS[i];
    }
    if left == 0 {
        return Some(greedy);
    }

    let target = amount as usize;
    let mut best: Vec<Option<[u16; 4]>> = vec![None; target + 1];
    best[0] = Some([0; 4]);

    for value in 1..=target {
        let mut candidate_best: Option<[u16; 4]> = None;
        for coin_index in 0..4 {
            let denomination = DENOMS[coin_index] as usize;
            if value < denomination {
                continue;
            }
            if let Some(mut candidate) = best[value - denomination] {
                if candidate[coin_index] >= tubes[coin_index] {
                    continue;
                }
                candidate[coin_index] += 1;
                let preferable = match candidate_best {
                    None => true,
                    Some(current) => {
                        let candidate_total: u32 =
                            candidate.iter().map(|&x| x as u32).sum();
                        let current_total: u32 =
                            current.iter().map(|&x| x as u32).sum();
                        candidate_total < current_total
                            || (candidate_total == current_total
                                && (0..4).rev().find_map(|k| {
                                    if candidate[k] != current[k] {
                                        Some(candidate[k] > current[k])
                                    } else {
                                        None
                                    }
                                }) == Some(true))
                    }
                };
                if preferable {
                    candidate_best = Some(candidate);
                }
            }
        }
        best[value] = candidate_best;
    }
    best[target]
}

fn choose_change(amount: u16, tubes: &[u16; 4]) -> Option<Vec<u16>> {
    if amount == 0 {
        return Some(Vec::new());
    }

    let mut greedy_counts = [0u16; 4];
    let mut remaining = amount;
    for i in (0..4).rev() {
        let take = tubes[i].min(remaining / DENOMS[i]);
        greedy_counts[i] = take;
        remaining -= take * DENOMS[i];
    }
    if remaining == 0 {
        return Some(counts_to_descending(&greedy_counts));
    }

    let mut best: Option<[u16; 4]> = None;
    search_change(0, amount, tubes, &mut [0; 4], &mut best);
    best.map(|counts| counts_to_descending(&counts))
}

fn search_change(
    index: usize,
    remaining: u16,
    tubes: &[u16; 4],
    counts: &mut [u16; 4],
    best: &mut Option<[u16; 4]>,
) {
    if index == 4 {
        if remaining != 0 {
            return;
        }
        let replace = match best {
            None => true,
            Some(old) => {
                let total_new: u32 = counts.iter().map(|&x| u32::from(x)).sum();
                let total_old: u32 = old.iter().map(|&x| u32::from(x)).sum();
                total_new < total_old
                    || (total_new == total_old
                        && (0..4).rev().find_map(|i| {
                            if counts[i] != old[i] {
                                Some(counts[i] > old[i])
                            } else {
                                None
                            }
                        }) == Some(true))
            }
        };
        if replace {
            *best = Some(*counts);
        }
        return;
    }

    let coin = DENOMS[index];
    let max_take = tubes[index].min(remaining / coin);
    for take in 0..=max_take {
        counts[index] = take;
        search_change(index + 1, remaining - take * coin, tubes, counts, best);
    }
    counts[index] = 0;
}

fn counts_to_descending(counts: &[u16; 4]) -> Vec<u16> {
    let mut result = Vec::new();
    for i in (0..4).rev() {
        for _ in 0..counts[i] {
            result.push(DENOMS[i]);
        }
    }
    result
}
