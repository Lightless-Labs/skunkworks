    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

    cwd: Vec<String>,

        entries.insert(Vec::new(), Entry::Dir);
        Self {
            entries,
            cwd: Vec::new(),
        }
    }

        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

            return Err(Error::InvalidPath(0));

            Vec::new()
        } else {
            self.cwd.clone()
        };

            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                name => parts.push(name.to_string()),
            }
        }
        Ok(parts)
    }

                Some(entry) if end == parts.len() => return Ok(entry),

                Some(Entry::File(_)) => unreachable!(),

        }
        unreachable!()

            return None;

            Some((parent, name)) => Some((parent.to_owned(), name)),
            None => None,
        }
    }

        let (parent, _) = Self::parent_and_name(path).expect("non-root path has parent");

            None => Err(Error::Missing(parent)),
            Some(Entry::File(_)) => Err(Error::NotDir(parent)),
            Some(Entry::Dir) => Ok(()),

                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Ok(self.entries.get(&prefix).unwrap()),

}

            nodes,
            working: Vec::new(),
        }
    }

        if parts.is_empty() {
            "/".to_string()
        } else {
            let mut path = String::new();
            for part in parts {
                path.push('/');
                path.push_str(part);
            }
            path
        }
    }

        if let Some(i) = path.as_bytes().iter().position(|b| *b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));

        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            self.working.clone()
        };

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                name => parts.push(name.to_string()),
            }
        }
        Ok(parts)
    }

        let mut prefix = Vec::new();
        for component in parts.iter().take(parts.len().saturating_sub(1)) {
            prefix.push(component.clone());
            let path = Self::canonical(&prefix);
            match self.nodes.get(&prefix) {
                None => return Err(Error::Missing(path)),
                Some(Node::Directory) => {}
                Some(Node::Bytes(_)) => return Err(Error::NotDir(path)),
            }
        }
        Ok(())
    }

    fn lookup(&self, parts: &[String]) -> Result<&Node, Error> {
        let mut prefix = Vec::new();
        for (i, component) in parts.iter().enumerate() {
            prefix.push(component.clone());
            let path = Self::canonical(&prefix);
            match self.nodes.get(&prefix) {
                None => return Err(Error::Missing(path)),
                Some(Node::Bytes(_)) if i + 1 < parts.len() => {

                }
                Some(node) if i + 1 == parts.len() => return Ok(node),
                Some(Node::Directory) => {}
                Some(Node::Bytes(_)) => unreachable!(),

        }
    }

    fn lookup_parent(&self, path: &str) -> Result<(), Error> {
        let parts: Vec<&str> = path.split('/').filter(|s| !s.is_empty()).collect();
        if parts.is_empty() {
            return Ok(());
        }

        for end in 0..parts.len() - 1 {
            let prefix = Self::prefix(&parts[..=end]);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn lookup_full(&self, path: &str) -> Result<&Entry, Error> {
        let parts: Vec<&str> = path.split('/').filter(|s| !s.is_empty()).collect();
        if parts.is_empty() {
            return self.entries.get("/").ok_or_else(|| Error::Missing("/".to_string()));
        }

        let mut prefix_parts = Vec::new();
        for (i, part) in parts.iter().enumerate() {
            prefix_parts.push(*part);
            let prefix = Self::prefix(&prefix_parts);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(entry) if i + 1 < parts.len() && matches!(entry, Entry::File(_)) => {
                    return Err(Error::NotDir(prefix));
                }
                Some(entry) if i + 1 < parts.len() && matches!(entry, Entry::Dir) => {}
                Some(entry) => return Ok(entry),
            }
        }
        unreachable!()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        self.lookup_parent(&canonical)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }
        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        self.lookup_parent(&canonical)?;

        match self.entries.get(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(_)) | None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_full(&canonical)? {
            Entry::Dir => Err(Error::IsDir(canonical)),
            Entry::File(data) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        let entry = self.lookup_full(&canonical)?;
        if !matches!(entry, Entry::Dir) {
            return Err(Error::NotDir(canonical));
        }

        let prefix = if canonical == "/" {
            "/".to_string()
        } else {
            format!("{}/", canonical)
        };
        let mut result = Vec::new();
        for (candidate, child) in &self.entries {
            if candidate == &canonical || !candidate.starts_with(&prefix) {
                continue;
            }
            let remainder = &candidate[prefix.len()..];
            if !remainder.is_empty() && !remainder.contains('/') {
                let kind = match child {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                result.push((remainder.to_string(), kind));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_full(&canonical)? {
            Entry::Dir => {
                self.cwd = canonical;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(canonical)),
        }
    }
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(index) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(index));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts: Vec<&str> = Vec::new();
        if !path.starts_with('/') {
            parts.extend(self.cwd.split('/').filter(|part| !part.is_empty()));
        }

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other),
            }
        }

        if parts.is_empty() {
            Ok("/".to_string())
        } else {
            let mut result = String::new();
            for part in parts {
                result.push('/');
                result.push_str(part);
            }
            Ok(result)
        }
    }

    fn parent_path(path: &str) -> &str {
        match path.rfind('/') {
            Some(0) | None => "/",
            Some(index) => &path[..index],
        }
    }

    fn lookup_parent(&self, path: &str) -> Result<(), Error> {
        let parent = Self::parent_path(path);
        let mut prefix = String::new();

        for component in parent.split('/').filter(|part| !part.is_empty()) {
            prefix.push('/');
            prefix.push_str(component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => return Err(Error::NotDir(prefix)),
            }
        }
        Ok(())
    }

    fn lookup_full(&self, path: &str) -> Result<&Entry, Error> {
        let mut prefix = String::new();
        let components: Vec<&str> = path.split('/').filter(|part| !part.is_empty()).collect();

        for (index, component) in components.iter().enumerate() {
            prefix.push('/');
            prefix.push_str(component);
            match self.entries.get(&prefix) {
                None => return Err(Error::Missing(prefix)),
                Some(entry @ Entry::File(_)) if index + 1 < components.len() => {
                    let _ = entry;
                    return Err(Error::NotDir(prefix));
                }
                Some(entry) if index + 1 == components.len() => return Ok(entry),
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => unreachable!(),
            }
        }

        self.entries.get("/").ok_or_else(|| Error::Missing("/".to_string()))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::Exists(canonical));
        }
        self.lookup_parent(&canonical)?;
        if self.entries.contains_key(&canonical) {
            return Err(Error::Exists(canonical));
        }
        self.entries.insert(canonical, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        if canonical == "/" {
            return Err(Error::IsDir(canonical));
        }
        self.lookup_parent(&canonical)?;
        match self.entries.get(&canonical) {
            Some(Entry::Dir) => Err(Error::IsDir(canonical)),
            Some(Entry::File(_)) => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
            None => {
                self.entries
                    .insert(canonical, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_full(&canonical)? {
            Entry::Dir => Err(Error::IsDir(canonical)),
            Entry::File(bytes) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_full(&canonical)? {
            Entry::File(_) => return Err(Error::NotDir(canonical)),
            Entry::Dir => {}
        }

        let mut children = Vec::new();
        let prefix = if canonical == "/" {
            "/".to_string()
        } else {
            format!("{}/", canonical)
        };

        for (entry_path, entry) in &self.entries {
            if !entry_path.starts_with(&prefix) || entry_path == &canonical {
                continue;
            }
            let remainder = &entry_path[prefix.len()..];
            if remainder.is_empty() || remainder.contains('/') {
                continue;
            }
            let kind = match entry {
                Entry::Dir => Kind::Dir,
                Entry::File(_) => Kind::File,
            };
            children.push((remainder.to_string(), kind));
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.normalize(path)?;
        match self.lookup_full(&canonical)? {
            Entry::Dir => {
                self.cwd = canonical;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(canonical)),
        }
    }

use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<Vec<String>, Entry>,
    cwd: Vec<String>,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert(Vec::new(), Entry::Dir);
        Self {
            entries,
            cwd: Vec::new(),
        }
    }

    fn canonical(parts: &[String]) -> String {
        if parts.is_empty() {
            "/".to_string()
        } else {
            format!("/{}", parts.join("/"))
        }
    }

    fn normalize(&self, path: &str) -> Result<Vec<String>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts = if path.starts_with('/') {
            Vec::new()
        } else {
            self.cwd.clone()
        };

        for component in path.split('/') {
            match component {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                name => parts.push(name.to_string()),
            }
        }
        Ok(parts)
    }

    fn parent(path: &[String]) -> (&[String], &str) {
        let name = path.last().map(String::as_str).unwrap_or("");
        (&path[..path.len() - 1], name)
    }

    fn check_traversal(&self, path: &[String]) -> Result<(), Error> {
        let mut prefix = Vec::new();
        for component in path.iter().take(path.len().saturating_sub(1)) {
            prefix.push(component.clone());
            match self.entries.get(&prefix) {
                Some(Entry::Dir) => {}
                Some(Entry::File(_)) => {
                    return Err(Error::NotDir(Self::canonical(&prefix)));
                }
                None => return Err(Error::Missing(Self::canonical(&prefix))),
            }
        }
        Ok(())
    }

    fn get_final(&self, path: &[String]) -> Result<Option<&Entry>, Error> {
        self.check_traversal(path)?;
        Ok(self.entries.get(path))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path.is_empty() {
            return Err(Error::Exists("/".to_string()));
        }
        self.check_traversal(&path)?;
        let (parent, _) = Self::parent(&path);
        match self.entries.get(parent) {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => {
                return Err(Error::NotDir(Self::canonical(parent)));
            }
            None => return Err(Error::Missing(Self::canonical(parent))),
        }
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(Self::canonical(&path)));
        }
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path.is_empty() {
            return Err(Error::IsDir("/".to_string()));
        }
        self.check_traversal(&path)?;
        let (parent, _) = Self::parent(&path);
        match self.entries.get(parent) {
            Some(Entry::Dir) => {}
            Some(Entry::File(_)) => {
                return Err(Error::NotDir(Self::canonical(parent)));
            }
            None => return Err(Error::Missing(Self::canonical(parent))),
        }
        match self.entries.get(&path) {
            Some(Entry::Dir) => Err(Error::IsDir(Self::canonical(&path))),
            Some(Entry::File(_)) | None => {
                self.entries.insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        match self.get_final(&path)? {
            None => Err(Error::Missing(Self::canonical(&path))),
            Some(Entry::Dir) => Err(Error::IsDir(Self::canonical(&path))),
            Some(Entry::File(bytes)) => Ok(bytes.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        match self.get_final(&path)? {
            None => return Err(Error::Missing(Self::canonical(&path))),
            Some(Entry::File(_)) => return Err(Error::NotDir(Self::canonical(&path))),
            Some(Entry::Dir) => {}
        }

        let mut result = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path.len() == path.len() + 1 && entry_path.starts_with(&path) {
                let name = entry_path.last().unwrap().clone();
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                result.push((name, kind));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        Self::canonical(&self.cwd)
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        match self.get_final(&path)? {
            None => Err(Error::Missing(Self::canonical(&path))),
            Some(Entry::File(_)) => Err(Error::NotDir(Self::canonical(&path))),
            Some(Entry::Dir) => {
                self.cwd = path;
                Ok(())
            }
        }
    }
