- Width is validated before input bytes; I return `WidthZero` before checking anything else.
- Width is measured in bytes, which equals columns for the permitted ASCII characters.
- Only space, tab, LF, and CR separate words; all other allowed printable characters are preserved.
- Long-word splitting follows the stated “available < 2” rule, even when one column could be used.
- Exact-width chunks emit without adding an empty trailing line; a short final chunk can share a line with the next word.
- The maximum width is 256, so line-length arithmetic is bounded; both implementations use checked or subtraction-safe fit logic.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    for (index, &byte) in input.as_bytes().iter().enumerate() {
        if !(matches!(byte, 32..=126 | b'\t' | b'\n' | b'\r')) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();

    for word in input.split(|c| matches!(c, ' ' | '\t' | '\n' | '\r')).filter(|w| !w.is_empty()) {
        let bytes = word.as_bytes();
        let mut pos = 0;

        while pos < bytes.len() {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = bytes.len() - pos;
            if remaining <= width {
                let needed = remaining + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(&word[pos..]);
                    pos = bytes.len();
                } else {
                    if !line.is_empty() {
                        lines.push(std::mem::take(&mut line));
                    }
                }
            } else if line.is_empty() {
                line.push_str(&word[pos..pos + width]);
                pos += width;
                lines.push(std::mem::take(&mut line));
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&word[pos..pos + available]);
                    pos += available;
                    lines.push(std::mem::take(&mut line));
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }

    Ok(lines)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let source = input.as_bytes();
    for (index, &byte) in source.iter().enumerate() {
        let allowed = (32..=126).contains(&byte) || byte == 9 || byte == 10 || byte == 13;
        if !allowed {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut output = Vec::new();
    let mut line = Vec::<u8>::new();
    let mut cursor = 0;

    while cursor < source.len() {
        while cursor < source.len()
            && matches!(source[cursor], b' ' | b'\t' | b'\n' | b'\r')
        {
            cursor += 1;
        }
        if cursor == source.len() {
            break;
        }

        let start = cursor;
        while cursor < source.len()
            && !matches!(source[cursor], b' ' | b'\t' | b'\n' | b'\r')
        {
            cursor += 1;
        }
        let word = &source[start..cursor];
        let mut pos = 0;

        while pos < word.len() {
            if line.len() == width {
                output.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
            }

            let remaining = word.len() - pos;
            if remaining <= width {
                let separator = usize::from(!line.is_empty());
                if line.len() + separator + remaining <= width {
                    if separator == 1 {
                        line.push(b' ');
                    }
                    line.extend_from_slice(&word[pos..]);
                    pos = word.len();
                } else {
                    output.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                }
            } else if line.is_empty() {
                line.extend_from_slice(&word[pos..pos + width]);
                pos += width;
                output.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(b' ');
                    line.extend_from_slice(&word[pos..pos + available]);
                    pos += available;
                    output.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                } else {
                    output.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                }
            }
        }
    }

    if !line.is_empty() {
        output.push(String::from_utf8(line).unwrap());
    }

    Ok(output)
}
```
