```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(matches!(byte, b' '..=b'~' | b'\t' | b'\n' | b'\r')) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && matches!(bytes[i], b' ' | b'\t' | b'\n' | b'\r') {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }

        let start = i;
        while i < bytes.len() && !matches!(bytes[i], b' ' | b'\t' | b'\n' | b'\r') {
            i += 1;
        }
        let mut word_pos = start;

        while word_pos < i {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = i - word_pos;
            if remaining <= width {
                let needed = remaining + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(&input[word_pos..i]);
                    word_pos = i;
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            } else if line.is_empty() {
                let end = word_pos + width;
                lines.push(input[word_pos..end].to_string());
                word_pos = end;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(&input[word_pos..word_pos + available]);
                    word_pos += available;
                    lines.push(std::mem::take(&mut line));
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }
    Ok(lines)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let b = input.as_bytes();
    for (index, &byte) in b.iter().enumerate() {
        if !(matches!(byte, 32..=126 | 9 | 10 | 13)) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut result = Vec::new();
    let mut line = Vec::<u8>::new();
    let mut pos = 0;

    while pos < b.len() {
        while pos < b.len() && matches!(b[pos], 32 | 9 | 10 | 13) {
            pos += 1;
        }
        if pos == b.len() {
            break;
        }

        let word_start = pos;
        while pos < b.len() && !matches!(b[pos], 32 | 9 | 10 | 13) {
            pos += 1;
        }
        let word_end = pos;
        let mut part = word_start;

        while part < word_end {
            if line.len() == width {
                result.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
            }

            let suffix_len = word_end - part;
            if suffix_len <= width {
                let separator_len = usize::from(!line.is_empty());
                if line.len() + separator_len + suffix_len <= width {
                    if separator_len != 0 {
                        line.push(b' ');
                    }
                    line.extend_from_slice(&b[part..word_end]);
                    part = word_end;
                } else {
                    result.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                }
            } else if line.is_empty() {
                result.push(String::from_utf8(b[part..part + width].to_vec()).unwrap());
                part += width;
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(b' ');
                    line.extend_from_slice(&b[part..part + available]);
                    part += available;
                    result.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                } else {
                    result.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                }
            }
        }
    }

    if !line.is_empty() {
        result.push(String::from_utf8(line).unwrap());
    }
    Ok(result)
}
```
