- Validate width before reading input; width zero takes precedence over oversized width checks.
- Count UTF-8 bytes, and report the first disallowed byte’s zero-based byte index.
- Treat only space, tab, LF, and CR as separators; collapse runs and omit empty lines.
- Preserve every other allowed printable byte literally, including hyphens and backslashes.
- Apply the split rules exactly, including leaving one available column unused and keeping short tails available for the next word.
- Width is capped at 256, so line-length and separator arithmetic stay bounded.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(matches!(byte, 32..=126 | 9 | 10 | 13)) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        while pos < bytes.len() && matches!(bytes[pos], b' ' | b'\t' | b'\n' | b'\r') {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let start = pos;
        while pos < bytes.len() && !matches!(bytes[pos], b' ' | b'\t' | b'\n' | b'\r') {
            pos += 1;
        }
        let mut suffix = &bytes[start..pos];

        while !suffix.is_empty() {
            if line.len() == width {
                lines.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
            }

            if suffix.len() <= width {
                let needed = suffix.len() + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(b' ');
                    }
                    line.extend_from_slice(suffix);
                    suffix = &[];
                } else {
                    lines.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                }
            } else if line.is_empty() {
                line.extend_from_slice(&suffix[..width]);
                lines.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                suffix = &suffix[width..];
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(b' ');
                    line.extend_from_slice(&suffix[..available]);
                    lines.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                    suffix = &suffix[available..];
                } else {
                    lines.push(String::from_utf8(std::mem::take(&mut line)).unwrap());
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(String::from_utf8(line).unwrap());
    }
    Ok(lines)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let input_bytes = input.as_bytes();
    for (index, &byte) in input_bytes.iter().enumerate() {
        let permitted = (32..=126).contains(&byte) || matches!(byte, 9 | 10 | 13);
        if !permitted {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let words: Vec<&[u8]> = input_bytes
        .split(|byte| matches!(*byte, b' ' | b'\t' | b'\n' | b'\r'))
        .filter(|word| !word.is_empty())
        .collect();

    let mut output = Vec::new();
    let mut current = Vec::<u8>::new();

    for word in words {
        let mut offset = 0;
        while offset < word.len() {
            if current.len() == width {
                output.push(String::from_utf8(std::mem::take(&mut current)).unwrap());
            }

            let remaining = &word[offset..];
            if remaining.len() <= width {
                let separator = if current.is_empty() { 0 } else { 1 };
                if current.len() + separator + remaining.len() <= width {
                    if separator == 1 {
                        current.push(b' ');
                    }
                    current.extend_from_slice(remaining);
                    offset = word.len();
                } else {
                    output.push(String::from_utf8(std::mem::take(&mut current)).unwrap());
                }
            } else if current.is_empty() {
                current.extend_from_slice(&remaining[..width]);
                output.push(String::from_utf8(std::mem::take(&mut current)).unwrap());
                offset += width;
            } else {
                let available = width - current.len() - 1;
                if available < 2 {
                    output.push(String::from_utf8(std::mem::take(&mut current)).unwrap());
                } else {
                    current.push(b' ');
                    current.extend_from_slice(&remaining[..available]);
                    output.push(String::from_utf8(std::mem::take(&mut current)).unwrap());
                    offset += available;
                }
            }
        }
    }

    if !current.is_empty() {
        output.push(String::from_utf8(current).unwrap());
    }
    Ok(output)
}
```
