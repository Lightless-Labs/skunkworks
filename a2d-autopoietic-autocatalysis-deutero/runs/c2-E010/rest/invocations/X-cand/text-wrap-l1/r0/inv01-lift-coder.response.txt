- Width is validated before input bytes; invalid width takes precedence.
- Width `1` is valid; a one-column gap is left unused when splitting beside a nonempty line.
- Separators collapse, and leading or trailing separators create no empty output.
- Long words split by bytes, which is safe because accepted word bytes are ASCII.
- A split word’s short remainder may share a line with the next word.
- Both solutions preserve punctuation and use checked-by-invariant width arithmetic; Solution B uses a separate tokenizing and packing approach.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let bytes = input.as_bytes();
    for (index, &byte) in bytes.iter().enumerate() {
        if !(matches!(byte, 32..=126 | b'\t' | b'\n' | b'\r')) {
            return Err(WrapError::InvalidByte { index, byte });
        }
    }

    let mut lines = Vec::new();
    let mut line = String::new();
    let mut pos = 0;

    while pos < bytes.len() {
        while pos < bytes.len() && matches!(bytes[pos], b' ' | b'\t' | b'\n' | b'\r') {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let start = pos;
        while pos < bytes.len() && !matches!(bytes[pos], b' ' | b'\t' | b'\n' | b'\r') {
            pos += 1;
        }
        let mut word_pos = start;

        while word_pos < pos {
            if line.len() == width {
                lines.push(std::mem::take(&mut line));
            }

            let remaining = pos - word_pos;
            if remaining <= width {
                let needed = remaining + usize::from(!line.is_empty());
                if line.len() + needed <= width {
                    if !line.is_empty() {
                        line.push(' ');
                    }
                    line.push_str(std::str::from_utf8(&bytes[word_pos..pos]).unwrap());
                    word_pos = pos;
                } else {
                    if !line.is_empty() {
                        lines.push(std::mem::take(&mut line));
                    }
                }
            } else if line.is_empty() {
                line.push_str(std::str::from_utf8(&bytes[word_pos..word_pos + width]).unwrap());
                word_pos += width;
                lines.push(std::mem::take(&mut line));
            } else {
                let available = width - line.len() - 1;
                if available >= 2 {
                    line.push(' ');
                    line.push_str(
                        std::str::from_utf8(&bytes[word_pos..word_pos + available]).unwrap(),
                    );
                    word_pos += available;
                    lines.push(std::mem::take(&mut line));
                } else {
                    lines.push(std::mem::take(&mut line));
                }
            }
        }
    }

    if !line.is_empty() {
        lines.push(line);
    }
    Ok(lines)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    InvalidByte { index: usize, byte: u8 },
}

pub fn wrap(input: &str, width: usize) -> Result<Vec<String>, WrapError> {
    if width == 0 {
        return Err(WrapError::WidthZero);
    }
    if width > 256 {
        return Err(WrapError::WidthTooLarge { width });
    }

    let data = input.as_bytes();
    for (i, &b) in data.iter().enumerate() {
        if !((b >= 32 && b <= 126) || b == 9 || b == 10 || b == 13) {
            return Err(WrapError::InvalidByte { index: i, byte: b });
        }
    }

    let is_separator = |b: u8| b == b' ' || b == b'\t' || b == b'\n' || b == b'\r';
    let mut words: Vec<&str> = Vec::new();
    let mut cursor = 0;

    while cursor < data.len() {
        while cursor < data.len() && is_separator(data[cursor]) {
            cursor += 1;
        }
        if cursor == data.len() {
            break;
        }
        let begin = cursor;
        while cursor < data.len() && !is_separator(data[cursor]) {
            cursor += 1;
        }
        words.push(std::str::from_utf8(&data[begin..cursor]).unwrap());
    }

    let mut result = Vec::new();
    let mut current = String::new();

    for word in words {
        let mut offset = 0;
        while offset < word.len() {
            if current.len() == width {
                result.push(std::mem::take(&mut current));
            }

            let remaining = word.len() - offset;
            if remaining <= width {
                let separator_len = if current.is_empty() { 0 } else { 1 };
                if current.len() + separator_len + remaining <= width {
                    if separator_len == 1 {
                        current.push(' ');
                    }
                    current.push_str(&word[offset..]);
                    offset = word.len();
                } else {
                    result.push(std::mem::take(&mut current));
                }
            } else if current.is_empty() {
                current.push_str(&word[offset..offset + width]);
                offset += width;
                result.push(std::mem::take(&mut current));
            } else {
                let available = width - current.len() - 1;
                if available < 2 {
                    result.push(std::mem::take(&mut current));
                } else {
                    current.push(' ');
                    current.push_str(&word[offset..offset + available]);
                    offset += available;
                    result.push(std::mem::take(&mut current));
                }
            }
        }
    }

    if !current.is_empty() {
        result.push(current);
    }
    Ok(result)
}
```
