#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&i| (jobs[i].arrival, i));

    let mut completion = vec![(0u32, 0u64); jobs.len()];
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut ready: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut event_pos = 0usize;
    let mut last_id: Option<u32> = None;
    let mut completed = 0usize;

    let admit_through = |time: u64,
                         event_pos: &mut usize,
                         ready: &mut std::collections::VecDeque<usize>,
                         completion: &mut Vec<(u32, u64)>,
                         completed: &mut usize| {
        while *event_pos < events.len() {
            let i = events[*event_pos];
            let arrival = u64::from(jobs[i].arrival);
            if arrival > time {
                break;
            }
            *event_pos += 1;
            if jobs[i].ticks == 0 {
                completion[i] = (jobs[i].id, arrival);
                *completed += 1;
            } else {
                ready.push_back(i);
            }
        }
    };

    while completed < jobs.len() {
        admit_through(
            now,
            &mut event_pos,
            &mut ready,
            &mut completion,
            &mut completed,
        );

        if let Some(i) = ready.pop_front() {
            if last_id.is_some_and(|id| id != jobs[i].id) {
                now += u64::from(switch_cost);
            }

            admit_through(
                now,
                &mut event_pos,
                &mut ready,
                &mut completion,
                &mut completed,
            );

            let consumed = remaining[i].min(quantum);
            let start = now;
            now += u64::from(consumed);
            remaining[i] -= consumed;
            let finished = remaining[i] == 0;

            runs.push(Run {
                id: jobs[i].id,
                start,
                end: now,
                finished,
            });

            admit_through(
                now,
                &mut event_pos,
                &mut ready,
                &mut completion,
                &mut completed,
            );

            if finished {
                completion[i] = (jobs[i].id, now);
                completed += 1;
            } else {
                ready.push_back(i);
            }
            last_id = Some(jobs[i].id);
        } else if event_pos < events.len() {
            let next_time = u64::from(jobs[events[event_pos]].arrival);
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_id = None;
            }
        }
    }

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: now,
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if quantum == 0 || quantum > 1000 {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for i in 0..jobs.len() {
        for j in 0..i {
            if jobs[j].id == jobs[i].id {
                return Err(Error::DuplicateId { index: i });
            }
        }
        if jobs[i].arrival > 1000 {
            return Err(Error::Arrival { index: i });
        }
        if jobs[i].ticks > 1000 {
            return Err(Error::Ticks { index: i });
        }
    }

    let mut pending: Vec<usize> = (0..jobs.len()).collect();
    pending.sort_by(|&a, &b| {
        jobs[a]
            .arrival
            .cmp(&jobs[b].arrival)
            .then_with(|| a.cmp(&b))
    });

    let mut ready = std::collections::VecDeque::new();
    let mut remaining: Vec<u32> = jobs.iter().map(|j| j.ticks).collect();
    let mut completion = vec![(0u32, 0u64); jobs.len()];
    let mut done = vec![false; jobs.len()];
    let mut done_count = 0usize;
    let mut next = 0usize;
    let mut time = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    loop {
        while next < pending.len()
            && u64::from(jobs[pending[next]].arrival) <= time
        {
            let i = pending[next];
            next += 1;
            if jobs[i].ticks == 0 {
                completion[i] = (jobs[i].id, u64::from(jobs[i].arrival));
                done[i] = true;
                done_count += 1;
            } else {
                ready.push_back(i);
            }
        }

        if done_count == jobs.len() {
            break;
        }

        if ready.is_empty() {
            let i = pending[next];
            let arrival = u64::from(jobs[i].arrival);
            if arrival > time {
                if let Some(interval) = idle.last_mut() {
                    if interval.1 == time {
                        interval.1 = arrival;
                    } else {
                        idle.push((time, arrival));
                    }
                } else {
                    idle.push((time, arrival));
                }
                time = arrival;
                last_dispatch = None;
            }
            continue;
        }

        let i = ready.pop_front().unwrap();
        if let Some(previous) = last_dispatch {
            if previous != jobs[i].id {
                time += u64::from(switch_cost);
            }
        }

        while next < pending.len()
            && u64::from(jobs[pending[next]].arrival) <= time
        {
            let arriving = pending[next];
            next += 1;
            if jobs[arriving].ticks == 0 {
                completion[arriving] =
                    (jobs[arriving].id, u64::from(jobs[arriving].arrival));
                done[arriving] = true;
                done_count += 1;
            } else {
                ready.push_back(arriving);
            }
        }

        let start = time;
        let used = remaining[i].min(quantum);
        time += u64::from(used);
        remaining[i] -= used;
        let finished = remaining[i] == 0;
        runs.push(Run {
            id: jobs[i].id,
            start,
            end: time,
            finished,
        });

        while next < pending.len()
            && u64::from(jobs[pending[next]].arrival) <= time
        {
            let arriving = pending[next];
            next += 1;
            if jobs[arriving].ticks == 0 {
                completion[arriving] =
                    (jobs[arriving].id, u64::from(jobs[arriving].arrival));
                done[arriving] = true;
                done_count += 1;
            } else {
                ready.push_back(arriving);
            }
        }

        if finished {
            completion[i] = (jobs[i].id, time);
            done[i] = true;
            done_count += 1;
        } else {
            ready.push_back(i);
        }
        last_dispatch = Some(jobs[i].id);
    }

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
