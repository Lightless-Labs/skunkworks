    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|previous| previous.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut ready: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut event_cursor = 0usize;
    let mut last_dispatched: Option<u32> = None;

    let admit_through = |limit: u64,
                         event_cursor: &mut usize,
                         ready: &mut std::collections::VecDeque<usize>,
                         completion: &mut Vec<Option<u64>>| {
        while *event_cursor < events.len() {
            let index = events[*event_cursor];
            let arrival = u64::from(jobs[index].arrival);
            if arrival > limit {
                break;
            }
            *event_cursor += 1;
            if jobs[index].ticks == 0 {
                completion[index] = Some(arrival);
            } else {
                ready.push_back(index);
            }
        }
    };

    while completion.iter().any(Option::is_none) {
        admit_through(now, &mut event_cursor, &mut ready, &mut completion);

        if ready.is_empty() {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = u64::from(jobs[events[event_cursor]].arrival);
            if next_time > now {
                if let Some((_, end)) = idle.last_mut() {
                    if *end == now {
                        *end = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatched = None;
            }
            continue;
        }

        let index = ready.pop_front().expect("ready queue is nonempty");
        let id = jobs[index].id;

        if last_dispatched.is_some_and(|previous| previous != id) {
            now += u64::from(switch_cost);
        }

        admit_through(now, &mut event_cursor, &mut ready, &mut completion);

        let consumed = remaining[index].min(quantum);
        let start = now;
        now += u64::from(consumed);
        remaining[index] -= consumed;
        let finished = remaining[index] == 0;
        runs.push(Run {
            id,
            start,
            end: now,
            finished,
        });

        admit_through(now, &mut event_cursor, &mut ready, &mut completion);

        if finished {
            completion[index] = Some(now);
        } else {
            ready.push_back(index);
        }
        last_dispatched = Some(id);
    }

    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].expect("every job completes")))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: now,
    })
}
