```rust
    Context(String),
    Remove(String),
    Add(String),
}

struct Hunk {
    old_start: i128,
    old_count: usize,
    new_start: i128,
    new_count: usize,
    header_line: usize,
    rows: Vec<Row>,
}

fn parse_decimal(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.as_bytes()[0] == b'0') {
        return None;
    }
    if !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let mut n = 0usize;
    for b in s.bytes() {
        n = n.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if n > 128 {
            return None;
        }
    }
    Some(n)
}

fn parse_hunk_header(line: &str) -> Option<(i128, usize, i128, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, suffix) = rest.split_once(" @@")?;
    if suffix != "" {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return None;
    }
    let a = parse_decimal(a)?;
    let b = parse_decimal(b)?;
    let c = parse_decimal(c)?;
    let d = parse_decimal(d)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    let old_start = if b > 0 { a - 1 } else { a } as i128;
    let new_start = if d > 0 { c - 1 } else { c } as i128;
    Some((old_start, b, new_start, d))
}

fn patch_lines(patch: &str) -> Vec<&str> {

    if patch.ends_with('\n') {
        v.pop();
```

```rust
    let rest = line.strip_prefix("@@ -")?;

    if count > 0 {
        if start == 0 {
            return None;
        }
        Some((start - 1) as i64)
    } else {
        Some(start as i64)
    }
}

                        new_seen += 1;
                    }
                    b'-' => old_seen += 1,
                    b'+' => new_seen += 1,
                    _ => unreachable!(),
                }
                rows.push((prefix, payload.to_owned()));
                i += 1;
            }

            if old_seen != b || new_seen != d {
                return Err(ApplyError::CountMismatch { line: header_line });
            }
            if !rows.iter().any(|(p, _)| *p == b'+' || *p == b'-') {
                return Err(ApplyError::BadBody { line: header_line });
            }

            if old_start < previous_old_end
                || new_start != old_start + cumulative_delta
            {
                return Err(ApplyError::Misordered { line: header_line });
            }
            previous_old_end = old_start + b as i64;
            cumulative_delta += d as i64 - b as i64;

            hunks.push(Hunk {
                line: header_line,
                old_start: old_start as usize,
                old_count: b,
                new_start: new_start as usize,
                new_count: d,
                rows,
            });

            if hunks.len() > 16 {
                return Err(ApplyError::TooLarge);
            }
        }
    }

    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut work: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1].split('\n').map(str::to_owned).collect()
    };
    if work.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let (mut net, mut drift) = (0i64, 0i64);
    for (idx, h) in hunks.iter().enumerate() {
        let lead = h.rows.iter().take_while(|(p, _)| *p == b' ').count();
        let trail = h.rows.iter().rev().take_while(|(p, _)| *p == b' ').count();
        let expected = h.old_start as i64 + net + drift;

        let mut best_rank: Option<(usize, i64)> = None;
        let mut best: Vec<(usize, usize, usize)> = Vec::new();

        for l in 0..=options.fuzz.min(lead) {
            for r in 0..=options.fuzz.min(trail) {
                if l + r > h.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                for (ri, (p, text)) in h.rows.iter().enumerate() {
                    if (ri < l) || (ri >= h.rows.len().saturating_sub(r) && r > 0) {
                        continue;
                    }
                    if *p == b' ' || *p == b'-' {
                        old_payloads.push(text.as_str());
                    }
                }
                let low = l;
                let high = h.old_count - r;
                if old_payloads.len() != high - low {
                    continue;
                }

                for s in 0..=work.len().saturating_sub(h.old_count) {
                    if s + h.old_count > work.len() {
                        continue;
                    }
                    let dist = (s as i64 - expected).abs();
                    if dist > options.max_offset as i64 {
                        continue;
                    }
                    if work[s + low..s + high]
                        .iter()
                        .map(String::as_str)
                        .ne(old_payloads.iter().copied())
                    {
                        continue;
                    }
                    let rank = (l + r, dist);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best.clear();
                            best.push((s, l, r));
                        }
                        Some(prev) if rank < prev => {
                            best_rank = Some(rank);
                            best.clear();
                            best.push((s, l, r));
                        }
                        Some(prev) if rank == prev => best.push((s, l, r)),
                        _ => {}
                    }
                }
            }
        }

        if best.is_empty() {
            return Err(ApplyError::NoMatch { hunk: idx + 1 });
        }

        let mut starts: Vec<usize> = best.iter().map(|x| x.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: idx + 1,
                first: starts[0],
                second: starts[1],
            });
        }
        let s = starts[0];
        let (_, l, r) = best
            .iter()
            .filter(|x| x.0 == s)
            .min_by_key(|x| x.1)
            .copied()
            .unwrap();

        let mut replacement = Vec::new();
        for (ri, (p, text)) in h.rows.iter().enumerate() {
            if ri < l || (r > 0 && ri >= h.rows.len() - r) {
                continue;
            }
            if *p == b' ' || *p == b'+' {
                replacement.push(text.clone());
            }
        }

        let begin = s + l;
        let end = s + h.old_count - r;
        work.splice(begin..end, replacement);
        if work.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        drift = s as i64 - (h.old_start as i64 + net);
        net += h.new_count as i64 - h.old_count as i64;
    }

    let mut out = String::new();
    for line in work {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]

    pub max_offset: usize,
    pub fuzz: usize,

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct Hunk {
    line: usize,
    old_start: i64,
    old_count: usize,
    new_start: i64,
    new_count: usize,
    body: Vec<(u8, String)>,
}

fn parse_number(s: &str) -> Option<usize> {
    if s.is_empty()
        || (s.len() > 1 && s.starts_with('0'))
        || !s.bytes().all(|b| b.is_ascii_digit())
    {
        return None;
    }
    let n = s.parse::<usize>().ok()?;
    (n <= 128).then_some(n)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, end) = rest.split_once(" @@")?;
    if !end.is_empty() {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    Some((parse_number(a)?, parse_number(b)?, parse_number(c)?, parse_number(d)?))
}

fn physical_lines(s: &str) -> Vec<&str> {
    let mut v: Vec<&str> = s.split_terminator('\n').collect();
    if s.is_empty() {
        v.clear();
    }
    v
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i).copied() != Some(*expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() <= 2 {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut i = 2;
    let mut previous_old_end: i64 = 0;
    let mut net: i64 = 0;

    while i < lines.len() {
        let header_line = i + 1;
        let Some((a, b, c, d)) = parse_hunk_header(lines[i]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        if (b > 0 && a == 0) || (d > 0 && c == 0) {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let old_start = if b > 0 { (a - 1) as i64 } else { a as i64 };
        let new_start = if d > 0 { (c - 1) as i64 } else { c as i64 };

        i += 1;
        let mut body = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while i < lines.len() && !lines[i].starts_with("@@") {
            let row = lines[i];
            let Some((&prefix, payload)) = row.as_bytes().split_first() else {
                return Err(ApplyError::BadBody { line: i + 1 });
            };
            if !matches!(prefix, b' ' | b'-' | b'+') || payload.contains(&b'\r') {
                return Err(ApplyError::BadBody { line: i + 1 });
            }
            let payload = std::str::from_utf8(payload).expect("patch is valid UTF-8");
            if prefix != b'+' {
                old_seen += 1;
            }
            if prefix != b'-' {
                new_seen += 1;
            }
            body.push((prefix, payload.to_owned()));
            i += 1;
        }
        if old_seen != b || new_seen != d {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !body.iter().any(|(p, _)| *p == b'+' || *p == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }
        if old_start < previous_old_end || new_start != old_start + net {
            return Err(ApplyError::Misordered { line: header_line });
        }
        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }
        previous_old_end = old_start + b as i64;
        net += d as i64 - b as i64;
        hunks.push(Hunk {
            line: header_line,
            old_start,
            old_count: b,
            new_start,
            new_count: d,
            body,
        });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let lines = physical_lines(source);
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines.into_iter().map(str::to_owned).collect())
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }
    let hunks = parse_patch(patch)?;
    let mut lines = source_lines(source)?;
    let mut net = 0i64;
    let mut drift = 0i64;

    for (hi, hunk) in hunks.iter().enumerate() {
        let lead = hunk.body.iter().take_while(|(p, _)| *p == b' ').count();
        let trail = hunk.body.iter().rev().take_while(|(p, _)| *p == b' ').count();
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let expected = hunk.old_start + net + drift;

        let mut best_rank: Option<(usize, i64)> = None;
        let mut best: Vec<(usize, usize, usize)> = Vec::new();
        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > lead + trail || l + r > hunk.old_count {
                    continue;
                }
                let start_body = l;
                let end_body = hunk.body.len().saturating_sub(r);
                let old_payloads: Vec<&str> = hunk.body[start_body..end_body]
                    .iter()
                    .filter(|(p, _)| *p != b'+')
                    .map(|(_, s)| s.as_str())
                    .collect();
                let span = hunk.old_count - l - r;
                if old_payloads.len() != span {
                    continue;
                }
                if span > lines.len() {
                    continue;
                }
                for s in 0..=lines.len() - span {
                    if s + hunk.old_count > lines.len() {
                        continue;
                    }
                    let distance = (s as i64 - expected).abs();
                    if distance > options.max_offset as i64 {
                        continue;
                    }
                    if old_payloads
                        .iter()
                        .enumerate()
                        .all(|(j, p)| lines[s + l + j] == *p)
                    {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(current) if rank < current => {
                                best_rank = Some(rank);
                                best.clear();
                                best.push((s, l, r));
                            }
                            Some(current) if rank == current => best.push((s, l, r)),
                            _ => {}
                        }
                    }
                }
            }
        }

        if best.is_empty() {
            return Err(ApplyError::NoMatch { hunk: hi + 1 });
        }
        let mut starts: Vec<usize> = best.iter().map(|x| x.0).collect();
        starts.sort_unstable();
        starts.dedup();
        if starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: hi + 1,
                first: starts[0],
                second: starts[1],
            });
        }
        let s = starts[0];
        let (_, l, r) = best
            .iter()
            .filter(|x| x.0 == s)
            .min_by_key(|x| x.1)
            .copied()
            .unwrap();

        let body_start = l;
        let body_end = hunk.body.len() - r;
        let mut replacement = Vec::new();
        for (prefix, payload) in &hunk.body[body_start..body_end] {
            if *prefix != b'-' {
                replacement.push(payload.clone());
            }
        }
        lines.splice(s + l..s + hunk.old_count - r, replacement);
        if lines.len() > 128 {
            return Err(ApplyError::TooLarge);
        }
        let previous_net = net;
        net += hunk.new_count as i64 - hunk.old_count as i64;
        drift = s as i64 - (hunk.old_start + previous_net);
    }

    let mut out = String::new();
    for line in lines {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
enum Part {
    C(String),
    D(String),
    I(String),
}

struct H {
    old: i128,
    old_n: usize,
    new: i128,
    new_n: usize,
    rows: Vec<Part>,
}

fn num(x: &str) -> Option<usize> {
    if x.is_empty() || (x.len() > 1 && x.starts_with('0')) {
        return None;
    }
    let mut v = 0usize;
    for b in x.bytes() {
        if !b.is_ascii_digit() {
            return None;
        }
        v = v.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if v > 128 {
            return None;
        }
    }
    Some(v)
}

fn header(x: &str) -> Option<(i128, usize, i128, usize)> {
    let x = x.strip_prefix("@@ -")?;
    let (old, x) = x.split_once(" +")?;
    let x = x.strip_suffix(" @@")?;
    let (new, _) = x.split_once(',')?;
    let (a, b) = old.split_once(',')?;
    let (_, d) = x.split_once(',')?;
    let a = num(a)?;
    let b = num(b)?;
    let c = num(new)?;
    let d = num(d)?;
    if (b != 0 && a == 0) || (d != 0 && c == 0) {
        return None;
    }
    Some((
        if b == 0 { a as i128 } else { (a - 1) as i128 },
        b,
        if d == 0 { c as i128 } else { (c - 1) as i128 },
        d,
    ))
}

fn split_patch(p: &str) -> Vec<&str> {
    let mut v: Vec<_> = p.split('\n').collect();
    if p.ends_with('\n') {
        v.pop();
    }
    v
}

fn err_header() -> ApplyError {
    ApplyError::BadHunkHeader { line: 3 }
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let mut parsed = Vec::<H>::new();
    if !patch.is_empty() {
        if !patch.ends_with('\n') {
            return Err(ApplyError::BadHeader {
                line: patch.bytes().filter(|b| *b == b'\n').count() + 1,
            });
        }
        let lines = split_patch(patch);
        if lines.get(0) != Some(&"--- before") {
            return Err(ApplyError::BadHeader { line: 1 });
        }
        if lines.get(1) != Some(&"+++ after") {
            return Err(ApplyError::BadHeader { line: 2 });
        }
        if lines.len() == 2 {
            return Err(err_header());
        }

        let mut i = 2;
        let mut end_old = 0i128;
        let mut delta = 0i128;
        while i < lines.len() {
            let hl = i + 1;
            let Some((old, old_n, new, new_n)) = header(lines[i]) else {
                return Err(ApplyError::BadHunkHeader { line: hl });
            };
            i += 1;
            let mut rows = Vec::new();
            let mut old_got = 0;
            let mut new_got = 0;
            let mut changed = false;
            while i < lines.len() && !lines[i].starts_with("@@") {
                let ln = lines[i];
                if ln.is_empty() || ln.as_bytes().contains(&b'\r') {
                    return Err(ApplyError::BadBody { line: i + 1 });
                }
                let (kind, payload) = (ln.as_bytes()[0], &ln[1..]);
                match kind {
                    b' ' => {
                        old_got += 1;
                        new_got += 1;
                        rows.push(Part::C(payload.to_owned()));
                    }
                    b'-' => {
                        old_got += 1;
                        changed = true;
                        rows.push(Part::D(payload.to_owned()));
                    }
                    b'+' => {
                        new_got += 1;
                        changed = true;
                        rows.push(Part::I(payload.to_owned()));
                    }
                    _ => return Err(ApplyError::BadBody { line: i + 1 }),
                }
                i += 1;
            }
            if old_got != old_n || new_got != new_n {
                return Err(ApplyError::CountMismatch { line: hl });
            }
            if !changed {
                return Err(ApplyError::BadBody { line: hl });
            }
            if old < end_old || new != old + delta || old + delta < 0 {
                return Err(ApplyError::Misordered { line: hl });
            }
            end_old = old + old_n as i128;
            delta += new_n as i128 - old_n as i128;
            parsed.push(H {
                old,
                old_n,
                new,
                new_n,
                rows,
            });
            if parsed.len() == 17 {
                return Err(ApplyError::TooLarge);
            }
        }
        if parsed.is_empty() {
            return Err(err_header());
        }
    }

    if let Some((byte, _)) = source.bytes().enumerate().find(|(_, b)| *b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut lines: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1]
            .split('\n')
            .map(ToOwned::to_owned)
            .collect()
    };
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net = 0i128;
    let mut drift = 0i128;
    for (hi, h) in parsed.iter().enumerate() {
        let mut lead = 0;
        while lead < h.rows.len() && matches!(h.rows[lead], Part::C(_)) {
            lead += 1;
        }
        let mut trail = 0;
        while trail < h.rows.len() && matches!(h.rows[h.rows.len() - 1 - trail], Part::C(_)) {
            trail += 1;
        }
        let expected = h.old + net + drift;
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let mut best_rank: Option<(usize, i128)> = None;
        let mut by_start = std::collections::BTreeMap::<usize, usize>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > h.old_n {
                    continue;
                }
                let mut old_payloads = Vec::<&str>::new();
                let mut seen_c = 0usize;
                for p in &h.rows {
                    match p {
                        Part::C(s) => {
                            let drop = seen_c < l || seen_c >= h.old_n - r;
                            seen_c += 1;
                            if !drop {
                                old_payloads.push(s);
                            }
                        }
                        Part::D(s) => old_payloads.push(s),
                        Part::I(_) => {}
                    }
                }
                if h.old_n > lines.len() {
                    continue;
                }
                for s in 0..=lines.len() - h.old_n {
                    let distance = (s as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    let lo = s + l;
                    let hi_line = s + h.old_n - r;
                    if hi_line > lines.len() || lo > hi_line {
                        continue;
                    }
                    if lines[lo..hi_line]
                        .iter()
                        .map(String::as_str)
                        .eq(old_payloads.iter().copied())
                    {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                by_start.clear();
                                by_start.insert(s, l);
                            }
                            Some(br) if rank < br => {
                                best_rank = Some(rank);
                                by_start.clear();
                                by_start.insert(s, l);
                            }
                            Some(br) if rank == br => {
                                by_start
                                    .entry(s)
                                    .and_modify(|v| *v = (*v).min(l))
                                    .or_insert(l);
                            }
                            _ => {}
                        }
                    }
                }
            }
        }

        let Some(_) = best_rank else {
            return Err(ApplyError::NoMatch { hunk: hi + 1 });
        };
        if by_start.len() > 1 {
            let mut it = by_start.keys();
            let first = *it.next().unwrap();
            let second = *it.next().unwrap();
            return Err(ApplyError::Ambiguous {
                hunk: hi + 1,
                first,
                second,
            });
        }
        let (&s, &l) = by_start.iter().next().unwrap();

        let mut eligible_r = Vec::new();
        for r in 0..=max_r {
            if l + r > h.old_n {
                continue;
            }
            let mut old_payloads = Vec::<&str>::new();
            let mut seen_c = 0usize;
            for p in &h.rows {
                match p {
                    Part::C(t) => {
                        let drop = seen_c < l || seen_c >= h.old_n - r;
                        seen_c += 1;
                        if !drop {
                            old_payloads.push(t);
                        }
                    }
                    Part::D(t) => old_payloads.push(t),
                    Part::I(_) => {}
                }
            }
            if lines[s + l..s + h.old_n - r]
                .iter()
                .map(String::as_str)
                .eq(old_payloads.iter().copied())
            {
                eligible_r.push(r);
            }
        }
        let r = *eligible_r.iter().min().unwrap_or(&0);
        let mut replacement = Vec::new();
        let mut seen_c = 0usize;
        for p in &h.rows {
            match p {
                Part::C(t) => {
                    let drop = seen_c < l || seen_c >= h.old_n - r;
                    seen_c += 1;
                    if !drop {
                        replacement.push(t.clone());
                    }
                }
                Part::I(t) => replacement.push(t.clone()),
                Part::D(_) => {}
            }
        }
        let begin = s + l;
        let end = s + h.old_n - r;
        lines.splice(begin..end, replacement);
        if lines.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let old_net = net;
        net += h.new_n as i128 - h.old_n as i128;
        drift = s as i128 - (h.old + old_net);
    }

    let mut result = String::new();
    for line in lines {
        result.push_str(&line);
        result.push('\n');
    }
    Ok(result)
}
```
