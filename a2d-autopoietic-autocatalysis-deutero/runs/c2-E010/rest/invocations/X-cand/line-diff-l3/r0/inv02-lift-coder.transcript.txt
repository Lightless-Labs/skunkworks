fn parse_decimal(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.as_bytes()[0] == b'0') {
        return None;
    }
    if !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let mut n = 0usize;
    for b in s.bytes() {
        n = n.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if n > 128 {
            return None;
        }
    }
    Some(n)
}

fn parse_hunk_header(line: &str) -> Option<(i128, usize, i128, usize)> {

        return None;

    let (c, d) = new.split_once(',')?;

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let mut hunks = Vec::new();
    if !patch.is_empty() {
        if !patch.ends_with('\n') {
            return Err(ApplyError::BadHeader {
                line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
            });
        }
        let lines = patch_lines(patch);
        if lines.first().copied() != Some("--- before") {
            return Err(ApplyError::BadHeader { line: 1 });
        }
        if lines.get(1).copied() != Some("+++ after") {
            return Err(ApplyError::BadHeader { line: 2 });
        }
        if lines.len() < 3 {
            return Err(ApplyError::BadHunkHeader { line: 3 });
        }
        let mut i = 2usize;
        let mut prev_old_end: i128 = 0;
        let mut cumulative_delta: i128 = 0;

        while i < lines.len() {
            let header_line = i + 1;
            let Some((old_start, old_count, new_start, new_count)) =
                parse_hunk_header(lines[i])
            else {
                return Err(ApplyError::BadHunkHeader { line: header_line });
            };
            i += 1;
            let mut rows = Vec::new();
            let mut old_seen = 0usize;
            let mut new_seen = 0usize;
            let mut changed = false;

            while i < lines.len() && !lines[i].starts_with("@@") {
                let line = lines[i];
                if line.as_bytes().contains(&b'\r') || line.is_empty() {
                    return Err(ApplyError::BadBody { line: i + 1 });
                }
                let payload = &line[1..];
                match line.as_bytes()[0] {
                    b' ' => {
                        old_seen += 1;
                        new_seen += 1;
                        rows.push(Row::Context(payload.to_owned()));
                    }
                    b'-' => {
                        old_seen += 1;
                        changed = true;
                        rows.push(Row::Remove(payload.to_owned()));
                    }
                    b'+' => {
                        new_seen += 1;
                        changed = true;
                        rows.push(Row::Add(payload.to_owned()));
                    }
                    _ => return Err(ApplyError::BadBody { line: i + 1 }),
                }
                i += 1;
            }
            if old_seen != old_count || new_seen != new_count {
                return Err(ApplyError::CountMismatch { line: header_line });
            }
            if !changed {
                return Err(ApplyError::BadBody { line: header_line });
            }
            if old_start < prev_old_end {
                return Err(ApplyError::Misordered { line: header_line });
            }
            let required_new = old_start + cumulative_delta;
            if required_new < 0 || new_start != required_new {
                return Err(ApplyError::Misordered { line: header_line });
            }
            prev_old_end = old_start + old_count as i128;
            cumulative_delta += new_count as i128 - old_count as i128;

            hunks.push(Hunk {
                old_start,
                old_count,
                new_start,
                new_count,
                header_line,
                rows,
            });
            if hunks.len() > 16 {
                return Err(ApplyError::TooLarge);
            }
        }
        if hunks.is_empty() {
            return Err(ApplyError::BadHunkHeader { line: 3 });
        }
    }

    if let Some((byte, _)) = source.bytes().enumerate().find(|(_, b)| *b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }

        Vec::new()
    } else {
        source[..source.len() - 1].split('\n').map(str::to_owned).collect()
    };

        return Err(ApplyError::TooLarge);
    }

    let mut net: i128 = 0;
    let mut drift: i128 = 0;
    for (idx, h) in hunks.iter().enumerate() {
        let mut lead = 0usize;
        while lead < h.rows.len() && matches!(h.rows[lead], Row::Context(_)) {
            lead += 1;
        }
        let mut trail = 0usize;
        while trail < h.rows.len() && matches!(h.rows[h.rows.len() - 1 - trail], Row::Context(_)) {
            trail += 1;
        }

        let mut best: Option<(usize, i128, usize)> = None;
        let mut best_starts = Vec::<usize>::new();
        let mut best_for_start = std::collections::BTreeMap::<usize, usize>::new();

        for l in 0..=options.fuzz.min(lead) {
            for r in 0..=options.fuzz.min(trail) {
                if l + r > h.old_count {
                    continue;
                }
                let mut old_trimmed = Vec::<&str>::new();
                let mut new_trimmed = Vec::<&str>::new();
                let mut seen_context = 0usize;
                for row in &h.rows {
                    match row {
                        Row::Context(s) => {
                            let drop = (seen_context < l) || (seen_context >= h.old_count - r);
                            seen_context += 1;
                            if !drop {
                                old_trimmed.push(s);
                                new_trimmed.push(s);
                            }
                        }
                        Row::Remove(s) => old_trimmed.push(s),
                        Row::Add(s) => new_trimmed.push(s),
                    }
                }

                let min_s = 0usize;
                let max_s = work.len().saturating_sub(h.old_count);
                if h.old_count > work.len() {
                    continue;
                }
                for s in min_s..=max_s {
                    let dist = (s as i128 - expected).abs();
                    if dist > options.max_offset as i128 {
                        continue;
                    }
                    let start = s + l;
                    let end = s + h.old_count - r;
                    if end > work.len() || start > end {
                        continue;
                    }
                    let mut expected_old = old_trimmed.iter();
                    let matches = work[start..end]
                        .iter()
                        .all(|line| expected_old.next().copied() == Some(line.as_str()))
                        && expected_old.next().is_none();
                    if !matches {
                        continue;
                    }
                    let rank = (l + r, dist);
                    match best {
                        None => {
                            best = Some((rank.0, rank.1, s));
                            best_starts.clear();
                            best_starts.push(s);
                            best_for_start.clear();
                            best_for_start.insert(s, l);
                        }
                        Some((bf, bd, _)) if rank < (bf, bd) => {
                            best = Some((rank.0, rank.1, s));
                            best_starts.clear();
                            best_starts.push(s);
                            best_for_start.clear();
                            best_for_start.insert(s, l);
                        }

                            if !best_for_start.contains_key(&s) {
                                best_starts.push(s);
                                best_for_start.insert(s, l);
                            } else if l < best_for_start[&s] {
                                best_for_start.insert(s, l);
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        if best.is_none() {
            return Err(ApplyError::NoMatch { hunk: idx + 1 });
        }
        best_starts.sort_unstable();
        best_starts.dedup();
        if best_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: idx + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let l = best_for_start[&s];

        let mut trail_choices = Vec::<usize>::new();
        for r in 0..=options.fuzz.min(trail) {
            let mut old = Vec::<&str>::new();
            let mut seen = 0usize;
            for row in &h.rows {
                match row {
                    Row::Context(t) => {
                        let drop = (seen < l) || (seen >= h.old_count - r);
                        seen += 1;
                        if !drop {
                            old.push(t);
                        }
                    }
                    Row::Remove(t) => old.push(t),
                    Row::Add(_) => {}
                }
            }
            if s + h.old_count - r <= work.len()
                && work[s + l..s + h.old_count - r]
                    .iter()
                    .map(String::as_str)
                    .eq(old.into_iter())
            {
                trail_choices.push(r);
            }
        }
        let r = *trail_choices.iter().min().unwrap_or(&0);

        let mut replacement = Vec::<String>::new();
        let mut seen_context = 0usize;
        for row in &h.rows {
            match row {
                Row::Context(t) => {
                    let drop = (seen_context < l) || (seen_context >= h.old_count - r);
                    seen_context += 1;
                    if !drop {
                        replacement.push(t.clone());
                    }
                }
                Row::Add(t) => replacement.push(t.clone()),
                Row::Remove(_) => {}
            }
        }
        let replace_start = s + l;
        let replace_end = s + h.old_count - r;
        work.splice(replace_start..replace_end, replacement);

        if work.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += h.new_count as i128 - h.old_count as i128;
        drift = s as i128 - (h.old_start + previous_net);
    }

    let mut out = String::new();
    for line in work {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}

fn num(x: &str) -> Option<usize> {
    if x.is_empty() || (x.len() > 1 && x.starts_with('0')) {
        return None;
    }
    let mut v = 0usize;
    for b in x.bytes() {
        if !b.is_ascii_digit() {
            return None;
        }
        v = v.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if v > 128 {
            return None;
        }
    }
    Some(v)
}

fn header(x: &str) -> Option<(i128, usize, i128, usize)> {
    let x = x.strip_prefix("@@ -")?;
    let (old, x) = x.split_once(" +")?;
    let x = x.strip_suffix(" @@")?;
    let (new, _) = x.split_once(',')?;
    let (a, b) = old.split_once(',')?;
    let (_, d) = x.split_once(',')?;
    let a = num(a)?;
    let b = num(b)?;
    let c = num(new)?;
    let d = num(d)?;
    if (b != 0 && a == 0) || (d != 0 && c == 0) {
        return None;
    }
    Some((
        if b == 0 { a as i128 } else { (a - 1) as i128 },
        b,
        if d == 0 { c as i128 } else { (c - 1) as i128 },
        d,
    ))
}

fn split_patch(p: &str) -> Vec<&str> {

    if p.ends_with('\n') {
        v.pop();
    }
    v
}

        result.push('\n');
    }
