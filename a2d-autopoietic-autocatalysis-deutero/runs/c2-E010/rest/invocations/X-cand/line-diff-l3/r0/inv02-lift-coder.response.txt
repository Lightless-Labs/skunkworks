- Validation order: options first; patch termination and all patch parsing before source validation or matching.
- Physical lines use LF separators, with no extra line after a final LF; CR is rejected in source and in body payloads.
- Fuzz drops only edge context rows; candidates and ambiguity are ranked by full hunk start.
- Offset distance and relocation bookkeeping use signed arithmetic; neither expected positions nor anchors are clamped.
- A structurally valid seventeenth hunk is rejected after its own ordering checks, without parsing later hunks.
- Both implementations preserve the same readings; Solution B uses a separate parsed representation and matching strategy.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
enum Row {
    Context(String),
    Remove(String),
    Add(String),
}

struct Hunk {
    old_start: i128,
    old_count: usize,
    new_start: i128,
    new_count: usize,
    header_line: usize,
    rows: Vec<Row>,
}

fn parse_decimal(s: &str) -> Option<usize> {
    if s.is_empty() || (s.len() > 1 && s.as_bytes()[0] == b'0') {
        return None;
    }
    if !s.bytes().all(|b| b.is_ascii_digit()) {
        return None;
    }
    let mut n = 0usize;
    for b in s.bytes() {
        n = n.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if n > 128 {
            return None;
        }
    }
    Some(n)
}

fn parse_hunk_header(line: &str) -> Option<(i128, usize, i128, usize)> {
    let rest = line.strip_prefix("@@ -")?;
    let (old, rest) = rest.split_once(" +")?;
    let (new, suffix) = rest.split_once(" @@")?;
    if suffix != "" {
        return None;
    }
    let (a, b) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if a.contains(',') || b.contains(',') || c.contains(',') || d.contains(',') {
        return None;
    }
    let a = parse_decimal(a)?;
    let b = parse_decimal(b)?;
    let c = parse_decimal(c)?;
    let d = parse_decimal(d)?;
    if (b > 0 && a == 0) || (d > 0 && c == 0) {
        return None;
    }
    let old_start = if b > 0 { a - 1 } else { a } as i128;
    let new_start = if d > 0 { c - 1 } else { c } as i128;
    Some((old_start, b, new_start, d))
}

fn patch_lines(patch: &str) -> Vec<&str> {
    let mut v: Vec<&str> = patch.split('\n').collect();
    if patch.ends_with('\n') {
        v.pop();
    }
    v
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let mut hunks = Vec::new();
    if !patch.is_empty() {
        if !patch.ends_with('\n') {
            return Err(ApplyError::BadHeader {
                line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
            });
        }
        let lines = patch_lines(patch);
        if lines.first().copied() != Some("--- before") {
            return Err(ApplyError::BadHeader { line: 1 });
        }
        if lines.get(1).copied() != Some("+++ after") {
            return Err(ApplyError::BadHeader { line: 2 });
        }
        if lines.len() < 3 {
            return Err(ApplyError::BadHunkHeader { line: 3 });
        }
        let mut i = 2usize;
        let mut prev_old_end: i128 = 0;
        let mut cumulative_delta: i128 = 0;

        while i < lines.len() {
            let header_line = i + 1;
            let Some((old_start, old_count, new_start, new_count)) =
                parse_hunk_header(lines[i])
            else {
                return Err(ApplyError::BadHunkHeader { line: header_line });
            };
            i += 1;
            let mut rows = Vec::new();
            let mut old_seen = 0usize;
            let mut new_seen = 0usize;
            let mut changed = false;

            while i < lines.len() && !lines[i].starts_with("@@") {
                let line = lines[i];
                if line.as_bytes().contains(&b'\r') || line.is_empty() {
                    return Err(ApplyError::BadBody { line: i + 1 });
                }
                let payload = &line[1..];
                match line.as_bytes()[0] {
                    b' ' => {
                        old_seen += 1;
                        new_seen += 1;
                        rows.push(Row::Context(payload.to_owned()));
                    }
                    b'-' => {
                        old_seen += 1;
                        changed = true;
                        rows.push(Row::Remove(payload.to_owned()));
                    }
                    b'+' => {
                        new_seen += 1;
                        changed = true;
                        rows.push(Row::Add(payload.to_owned()));
                    }
                    _ => return Err(ApplyError::BadBody { line: i + 1 }),
                }
                i += 1;
            }
            if old_seen != old_count || new_seen != new_count {
                return Err(ApplyError::CountMismatch { line: header_line });
            }
            if !changed {
                return Err(ApplyError::BadBody { line: header_line });
            }
            if old_start < prev_old_end {
                return Err(ApplyError::Misordered { line: header_line });
            }
            let required_new = old_start + cumulative_delta;
            if required_new < 0 || new_start != required_new {
                return Err(ApplyError::Misordered { line: header_line });
            }
            prev_old_end = old_start + old_count as i128;
            cumulative_delta += new_count as i128 - old_count as i128;

            hunks.push(Hunk {
                old_start,
                old_count,
                new_start,
                new_count,
                header_line,
                rows,
            });
            if hunks.len() > 16 {
                return Err(ApplyError::TooLarge);
            }
        }
        if hunks.is_empty() {
            return Err(ApplyError::BadHunkHeader { line: 3 });
        }
    }

    if let Some((byte, _)) = source.bytes().enumerate().find(|(_, b)| *b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut work: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1].split('\n').map(str::to_owned).collect()
    };
    if work.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net: i128 = 0;
    let mut drift: i128 = 0;
    for (idx, h) in hunks.iter().enumerate() {
        let mut lead = 0usize;
        while lead < h.rows.len() && matches!(h.rows[lead], Row::Context(_)) {
            lead += 1;
        }
        let mut trail = 0usize;
        while trail < h.rows.len() && matches!(h.rows[h.rows.len() - 1 - trail], Row::Context(_)) {
            trail += 1;
        }

        let expected = h.old_start + net + drift;
        let mut best: Option<(usize, i128, usize)> = None;
        let mut best_starts = Vec::<usize>::new();
        let mut best_for_start = std::collections::BTreeMap::<usize, usize>::new();

        for l in 0..=options.fuzz.min(lead) {
            for r in 0..=options.fuzz.min(trail) {
                if l + r > h.old_count {
                    continue;
                }
                let mut old_trimmed = Vec::<&str>::new();
                let mut new_trimmed = Vec::<&str>::new();
                let mut seen_context = 0usize;
                for row in &h.rows {
                    match row {
                        Row::Context(s) => {
                            let drop = (seen_context < l) || (seen_context >= h.old_count - r);
                            seen_context += 1;
                            if !drop {
                                old_trimmed.push(s);
                                new_trimmed.push(s);
                            }
                        }
                        Row::Remove(s) => old_trimmed.push(s),
                        Row::Add(s) => new_trimmed.push(s),
                    }
                }

                let min_s = 0usize;
                let max_s = work.len().saturating_sub(h.old_count);
                if h.old_count > work.len() {
                    continue;
                }
                for s in min_s..=max_s {
                    let dist = (s as i128 - expected).abs();
                    if dist > options.max_offset as i128 {
                        continue;
                    }
                    let start = s + l;
                    let end = s + h.old_count - r;
                    if end > work.len() || start > end {
                        continue;
                    }
                    let mut expected_old = old_trimmed.iter();
                    let matches = work[start..end]
                        .iter()
                        .all(|line| expected_old.next().copied() == Some(line.as_str()))
                        && expected_old.next().is_none();
                    if !matches {
                        continue;
                    }
                    let rank = (l + r, dist);
                    match best {
                        None => {
                            best = Some((rank.0, rank.1, s));
                            best_starts.clear();
                            best_starts.push(s);
                            best_for_start.clear();
                            best_for_start.insert(s, l);
                        }
                        Some((bf, bd, _)) if rank < (bf, bd) => {
                            best = Some((rank.0, rank.1, s));
                            best_starts.clear();
                            best_starts.push(s);
                            best_for_start.clear();
                            best_for_start.insert(s, l);
                        }
                        Some((bf, bd, _)) if rank == (bf, bd) => {
                            if !best_for_start.contains_key(&s) {
                                best_starts.push(s);
                                best_for_start.insert(s, l);
                            } else if l < best_for_start[&s] {
                                best_for_start.insert(s, l);
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        if best.is_none() {
            return Err(ApplyError::NoMatch { hunk: idx + 1 });
        }
        best_starts.sort_unstable();
        best_starts.dedup();
        if best_starts.len() > 1 {
            return Err(ApplyError::Ambiguous {
                hunk: idx + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }
        let s = best_starts[0];
        let l = best_for_start[&s];

        let mut trail_choices = Vec::<usize>::new();
        for r in 0..=options.fuzz.min(trail) {
            let mut old = Vec::<&str>::new();
            let mut seen = 0usize;
            for row in &h.rows {
                match row {
                    Row::Context(t) => {
                        let drop = (seen < l) || (seen >= h.old_count - r);
                        seen += 1;
                        if !drop {
                            old.push(t);
                        }
                    }
                    Row::Remove(t) => old.push(t),
                    Row::Add(_) => {}
                }
            }
            if s + h.old_count - r <= work.len()
                && work[s + l..s + h.old_count - r]
                    .iter()
                    .map(String::as_str)
                    .eq(old.into_iter())
            {
                trail_choices.push(r);
            }
        }
        let r = *trail_choices.iter().min().unwrap_or(&0);

        let mut replacement = Vec::<String>::new();
        let mut seen_context = 0usize;
        for row in &h.rows {
            match row {
                Row::Context(t) => {
                    let drop = (seen_context < l) || (seen_context >= h.old_count - r);
                    seen_context += 1;
                    if !drop {
                        replacement.push(t.clone());
                    }
                }
                Row::Add(t) => replacement.push(t.clone()),
                Row::Remove(_) => {}
            }
        }
        let replace_start = s + l;
        let replace_end = s + h.old_count - r;
        work.splice(replace_start..replace_end, replacement);

        if work.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += h.new_count as i128 - h.old_count as i128;
        drift = s as i128 - (h.old_start + previous_net);
    }

    let mut out = String::new();
    for line in work {
        out.push_str(&line);
        out.push('\n');
    }
    Ok(out)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
enum Part {
    C(String),
    D(String),
    I(String),
}

struct H {
    old: i128,
    old_n: usize,
    new: i128,
    new_n: usize,
    rows: Vec<Part>,
}

fn num(x: &str) -> Option<usize> {
    if x.is_empty() || (x.len() > 1 && x.starts_with('0')) {
        return None;
    }
    let mut v = 0usize;
    for b in x.bytes() {
        if !b.is_ascii_digit() {
            return None;
        }
        v = v.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if v > 128 {
            return None;
        }
    }
    Some(v)
}

fn header(x: &str) -> Option<(i128, usize, i128, usize)> {
    let x = x.strip_prefix("@@ -")?;
    let (old, x) = x.split_once(" +")?;
    let x = x.strip_suffix(" @@")?;
    let (new, _) = x.split_once(',')?;
    let (a, b) = old.split_once(',')?;
    let (_, d) = x.split_once(',')?;
    let a = num(a)?;
    let b = num(b)?;
    let c = num(new)?;
    let d = num(d)?;
    if (b != 0 && a == 0) || (d != 0 && c == 0) {
        return None;
    }
    Some((
        if b == 0 { a as i128 } else { (a - 1) as i128 },
        b,
        if d == 0 { c as i128 } else { (c - 1) as i128 },
        d,
    ))
}

fn split_patch(p: &str) -> Vec<&str> {
    let mut v: Vec<_> = p.split('\n').collect();
    if p.ends_with('\n') {
        v.pop();
    }
    v
}

fn err_header() -> ApplyError {
    ApplyError::BadHunkHeader { line: 3 }
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let mut parsed = Vec::<H>::new();
    if !patch.is_empty() {
        if !patch.ends_with('\n') {
            return Err(ApplyError::BadHeader {
                line: patch.bytes().filter(|b| *b == b'\n').count() + 1,
            });
        }
        let lines = split_patch(patch);
        if lines.get(0) != Some(&"--- before") {
            return Err(ApplyError::BadHeader { line: 1 });
        }
        if lines.get(1) != Some(&"+++ after") {
            return Err(ApplyError::BadHeader { line: 2 });
        }
        if lines.len() == 2 {
            return Err(err_header());
        }

        let mut i = 2;
        let mut end_old = 0i128;
        let mut delta = 0i128;
        while i < lines.len() {
            let hl = i + 1;
            let Some((old, old_n, new, new_n)) = header(lines[i]) else {
                return Err(ApplyError::BadHunkHeader { line: hl });
            };
            i += 1;
            let mut rows = Vec::new();
            let mut old_got = 0;
            let mut new_got = 0;
            let mut changed = false;
            while i < lines.len() && !lines[i].starts_with("@@") {
                let ln = lines[i];
                if ln.is_empty() || ln.as_bytes().contains(&b'\r') {
                    return Err(ApplyError::BadBody { line: i + 1 });
                }
                let (kind, payload) = (ln.as_bytes()[0], &ln[1..]);
                match kind {
                    b' ' => {
                        old_got += 1;
                        new_got += 1;
                        rows.push(Part::C(payload.to_owned()));
                    }
                    b'-' => {
                        old_got += 1;
                        changed = true;
                        rows.push(Part::D(payload.to_owned()));
                    }
                    b'+' => {
                        new_got += 1;
                        changed = true;
                        rows.push(Part::I(payload.to_owned()));
                    }
                    _ => return Err(ApplyError::BadBody { line: i + 1 }),
                }
                i += 1;
            }
            if old_got != old_n || new_got != new_n {
                return Err(ApplyError::CountMismatch { line: hl });
            }
            if !changed {
                return Err(ApplyError::BadBody { line: hl });
            }
            if old < end_old || new != old + delta || old + delta < 0 {
                return Err(ApplyError::Misordered { line: hl });
            }
            end_old = old + old_n as i128;
            delta += new_n as i128 - old_n as i128;
            parsed.push(H {
                old,
                old_n,
                new,
                new_n,
                rows,
            });
            if parsed.len() == 17 {
                return Err(ApplyError::TooLarge);
            }
        }
        if parsed.is_empty() {
            return Err(err_header());
        }
    }

    if let Some((byte, _)) = source.bytes().enumerate().find(|(_, b)| *b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let mut lines: Vec<String> = if source.is_empty() {
        Vec::new()
    } else {
        source[..source.len() - 1]
            .split('\n')
            .map(ToOwned::to_owned)
            .collect()
    };
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }

    let mut net = 0i128;
    let mut drift = 0i128;
    for (hi, h) in parsed.iter().enumerate() {
        let mut lead = 0;
        while lead < h.rows.len() && matches!(h.rows[lead], Part::C(_)) {
            lead += 1;
        }
        let mut trail = 0;
        while trail < h.rows.len() && matches!(h.rows[h.rows.len() - 1 - trail], Part::C(_)) {
            trail += 1;
        }
        let expected = h.old + net + drift;
        let max_l = options.fuzz.min(lead);
        let max_r = options.fuzz.min(trail);
        let mut best_rank: Option<(usize, i128)> = None;
        let mut by_start = std::collections::BTreeMap::<usize, usize>::new();

        for l in 0..=max_l {
            for r in 0..=max_r {
                if l + r > h.old_n {
                    continue;
                }
                let mut old_payloads = Vec::<&str>::new();
                let mut seen_c = 0usize;
                for p in &h.rows {
                    match p {
                        Part::C(s) => {
                            let drop = seen_c < l || seen_c >= h.old_n - r;
                            seen_c += 1;
                            if !drop {
                                old_payloads.push(s);
                            }
                        }
                        Part::D(s) => old_payloads.push(s),
                        Part::I(_) => {}
                    }
                }
                if h.old_n > lines.len() {
                    continue;
                }
                for s in 0..=lines.len() - h.old_n {
                    let distance = (s as i128 - expected).abs();
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    let lo = s + l;
                    let hi_line = s + h.old_n - r;
                    if hi_line > lines.len() || lo > hi_line {
                        continue;
                    }
                    if lines[lo..hi_line]
                        .iter()
                        .map(String::as_str)
                        .eq(old_payloads.iter().copied())
                    {
                        let rank = (l + r, distance);
                        match best_rank {
                            None => {
                                best_rank = Some(rank);
                                by_start.clear();
                                by_start.insert(s, l);
                            }
                            Some(br) if rank < br => {
                                best_rank = Some(rank);
                                by_start.clear();
                                by_start.insert(s, l);
                            }
                            Some(br) if rank == br => {
                                by_start
                                    .entry(s)
                                    .and_modify(|v| *v = (*v).min(l))
                                    .or_insert(l);
                            }
                            _ => {}
                        }
                    }
                }
            }
        }

        let Some(_) = best_rank else {
            return Err(ApplyError::NoMatch { hunk: hi + 1 });
        };
        if by_start.len() > 1 {
            let mut it = by_start.keys();
            let first = *it.next().unwrap();
            let second = *it.next().unwrap();
            return Err(ApplyError::Ambiguous {
                hunk: hi + 1,
                first,
                second,
            });
        }
        let (&s, &l) = by_start.iter().next().unwrap();

        let mut eligible_r = Vec::new();
        for r in 0..=max_r {
            if l + r > h.old_n {
                continue;
            }
            let mut old_payloads = Vec::<&str>::new();
            let mut seen_c = 0usize;
            for p in &h.rows {
                match p {
                    Part::C(t) => {
                        let drop = seen_c < l || seen_c >= h.old_n - r;
                        seen_c += 1;
                        if !drop {
                            old_payloads.push(t);
                        }
                    }
                    Part::D(t) => old_payloads.push(t),
                    Part::I(_) => {}
                }
            }
            if lines[s + l..s + h.old_n - r]
                .iter()
                .map(String::as_str)
                .eq(old_payloads.iter().copied())
            {
                eligible_r.push(r);
            }
        }
        let r = *eligible_r.iter().min().unwrap_or(&0);
        let mut replacement = Vec::new();
        let mut seen_c = 0usize;
        for p in &h.rows {
            match p {
                Part::C(t) => {
                    let drop = seen_c < l || seen_c >= h.old_n - r;
                    seen_c += 1;
                    if !drop {
                        replacement.push(t.clone());
                    }
                }
                Part::I(t) => replacement.push(t.clone()),
                Part::D(_) => {}
            }
        }
        let begin = s + l;
        let end = s + h.old_n - r;
        lines.splice(begin..end, replacement);
        if lines.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let old_net = net;
        net += h.new_n as i128 - h.old_n as i128;
        drift = s as i128 - (h.old + old_net);
    }

    let mut result = String::new();
    for line in lines {
        result.push_str(&line);
        result.push('\n');
    }
    Ok(result)
}
```
