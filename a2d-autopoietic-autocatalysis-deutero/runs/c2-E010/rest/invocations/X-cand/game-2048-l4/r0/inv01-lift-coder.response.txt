- Validation order is shape, then every tile, then movement; invalid tiles take precedence over merge overflow.
- Line processing stays row-major for horizontal moves and column-major for vertical moves, even when scanning from the far edge.
- Merges use original tile indices in scan order; newly merged outputs cannot merge again in the same move.
- Full boards consume no RNG; non-full boards consume exactly two draws using wrapping LCG arithmetic.
- Failed steps preserve both current state and undo history; unchanged steps preserve history too.
- The requirements specify one behavior for each edge case, so Solution B uses an independent move algorithm with the same specified behavior.

fn validate(board: &Board) -> Result<(), Error> {
    if !(1..=8).contains(&board.width)
        || !(1..=8).contains(&board.height)
        || board.cells.len() != board.width * board.height
    {
        return Err(Error::Shape);
    }
    for (index, &value) in board.cells.iter().enumerate() {
        if value != 0 && (value < 2 || value > 2_147_483_648 || !value.is_power_of_two()) {
            return Err(Error::InvalidTile { index, value });
        }
    }
    Ok(())
}

fn line_indices(board: &Board, direction: Direction, line: usize) -> Vec<usize> {
    match direction {
        Direction::Left => (0..board.width).map(|x| line * board.width + x).collect(),
        Direction::Right => (0..board.width)
            .rev()
            .map(|x| line * board.width + x)
            .collect(),
        Direction::Up => (0..board.height).map(|y| y * board.width + line).collect(),
        Direction::Down => (0..board.height)
            .rev()
            .map(|y| y * board.width + line)
            .collect(),
    }
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;
    let mut cells = vec![0; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let line_count = match direction {
        Direction::Left | Direction::Right => board.height,
        Direction::Up | Direction::Down => board.width,
    };

    for line in 0..line_count {
        let indices = line_indices(board, direction, line);
        let tiles: Vec<(usize, u32)> = indices
            .iter()
            .filter_map(|&index| {
                let value = board.cells[index];
                (value != 0).then_some((index, value))
            })
            .collect();

        let mut output_offset = 0usize;
        let mut i = 0usize;
        while i < tiles.len() {
            let (first_index, first_value) = tiles[i];
            if i + 1 < tiles.len() && tiles[i + 1].1 == first_value {
                let second_index = tiles[i + 1].0;
                if first_value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: first_index,
                        second: second_index,
                    });
                }
                let value = first_value * 2;
                let to = indices[output_offset];
                cells[to] = value;
                merges.push(Merge {
                    from: [first_index, second_index],
                    to,
                    value,
                });
                score = score.checked_add(value as u64).ok_or(Error::ScoreOverflow)?;
                output_offset += 1;
                i += 2;
            } else {
                let to = indices[output_offset];
                cells[to] = first_value;
                output_offset += 1;
                i += 1;
            }
        }
    }

    let changed = cells != board.cells;
    Ok(MoveResult {
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        changed,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut result = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        if let Ok(movement) = move_board(board, direction) {
            if movement.changed {
                result.push(direction);
            }
        }
    }
    Ok(result)
}

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
    validate(board)?;
    let empties: Vec<usize> = board
        .cells
        .iter()
        .enumerate()
        .filter_map(|(index, &value)| (value == 0).then_some(index))
        .collect();

    if empties.is_empty() {
        return Ok(SpawnResult {
            board: board.clone(),
            rng,
            spawn: None,
        });
    }

    let a = next_rng(rng);
    let index = empties[(a % empties.len() as u64) as usize];
    let b = next_rng(a);
    let value = if b % 10 == 0 { 4 } else { 2 };
    let mut spawned = board.clone();
    spawned.cells[index] = value;

    Ok(SpawnResult {
        board: spawned,
        rng: b,
        spawn: Some(Spawn { index, value }),
    })
}

pub fn play_turn(board: &Board, rng: u64, direction: Direction) -> Result<Turn, Error> {

            board: movement.board,

            score: 0,
            changed: false,

            spawn: None,

}

    validate(&initial.board)?;

        current: initial,
        previous: None,
    })
}

pub fn snapshot(game: &Game) -> State {
    game.current.clone()
}

pub fn step(game: &mut Game, direction: Direction) -> Result<Turn, Error> {
    let movement = move_board(&game.current.board, direction)?;
    if !movement.changed {
        return Ok(Turn {
            board: movement.board,
            rng: game.current.rng,
            score: 0,
            changed: false,
            merges: Vec::new(),
            spawn: None,
        });
    }

    let cumulative_score = game
        .current
        .score
        .checked_add(movement.score)
        .ok_or(Error::ScoreOverflow)?;
    let spawned = spawn_tile(&movement.board, game.current.rng)?;
    let turn = Turn {
        board: spawned.board.clone(),
        rng: spawned.rng,
        score: movement.score,
        changed: true,
        merges: movement.merges,
        spawn: spawned.spawn,
    };

    let old = game.current.clone();
    game.current = State {
        board: spawned.board,
        score: cumulative_score,
        rng: spawned.rng,
    };
    game.previous = Some(old);
    Ok(turn)
}

pub fn undo(game: &mut Game) -> bool {
    if let Some(previous) = game.previous.take() {
        game.current = previous;
        true
    } else {
        false
    }
}

pub fn can_undo(game: &Game) -> bool {
    game.previous.is_some()
}

pub fn game_over(game: &Game) -> bool {
    legal_moves(&game.current.board)
        .map(|moves| moves.is_empty())
        .unwrap_or(true)
}

pub fn replay(initial: State, commands: &[Command]) -> Result<Replay, ReplayError> {
    let mut game = match new_game(initial.clone()) {
        Ok(game) => game,
        Err(error) => {
            return Err(ReplayError {
                index: None,
                error,
                state: initial,
                can_undo: false,
                events: Vec::new(),
            });
        }
    };
    let mut events = Vec::new();

    for (index, command) in commands.iter().enumerate() {
        match command {
            Command::Move(direction) => match step(&mut game, *direction) {
                Ok(turn) => events.push(Event::Moved(turn)),
                Err(error) => {
                    return Err(ReplayError {
                        index: Some(index),
                        error,
                        state: snapshot(&game),
                        can_undo: can_undo(&game),
                        events,
                    });
                }
            },
            Command::Undo => events.push(Event::Undone(undo(&mut game))),
            Command::Check => events.push(Event::Checked(game_over(&game))),
        }
    }

    Ok(Replay {
        state: snapshot(&game),
        can_undo: can_undo(&game),
        events,
    })
}

pub fn move_board(board: &Board, direction: Direction) -> Result<MoveResult, Error> {
    validate(board)?;

    let mut cells = vec![0u32; board.cells.len()];
    let mut merges = Vec::new();
    let mut score = 0u64;

    let vertical = matches!(direction, Direction::Up | Direction::Down);
    let outer = if vertical { board.width } else { board.height };
    let inner = if vertical { board.height } else { board.width };
    let reverse = matches!(direction, Direction::Right | Direction::Down);

    for line in 0..outer {
        let mut sources = Vec::new();
        for offset in 0..inner {
            let position = if reverse { inner - 1 - offset } else { offset };
            let index = if vertical {
                position * board.width + line
            } else {
                line * board.width + position
            };
            let value = board.cells[index];
            if value != 0 {
                sources.push((index, value));
            }
        }

        let mut cursor = 0usize;
        let mut source = 0usize;
        while source < sources.len() {
            let (index, value) = sources[source];
            let destination = {
                let position = if reverse { inner - 1 - cursor } else { cursor };
                if vertical {
                    position * board.width + line
                } else {
                    line * board.width + position
                }
            };

            if source + 1 < sources.len() && sources[source + 1].1 == value {
                let other = sources[source + 1].0;
                if value == 2_147_483_648 {
                    return Err(Error::MergeOverflow {
                        first: index,
                        second: other,
                    });
                }
                let merged = value * 2;
                cells[destination] = merged;
                merges.push(Merge {
                    from: [index, other],
                    to: destination,
                    value: merged,
                });
                score = score.checked_add(merged as u64).ok_or(Error::ScoreOverflow)?;
                source += 2;
            } else {
                cells[destination] = value;
                source += 1;
            }
            cursor += 1;
        }
    }

    Ok(MoveResult {
        changed: cells != board.cells,
        board: Board {
            width: board.width,
            height: board.height,
            cells,
        },
        score,
        merges,
    })
}

pub fn legal_moves(board: &Board) -> Result<Vec<Direction>, Error> {
    validate(board)?;
    let mut legal = Vec::new();
    for direction in [
        Direction::Up,
        Direction::Left,
        Direction::Down,
        Direction::Right,
    ] {
        match move_board(board, direction) {
            Ok(result) if result.changed => legal.push(direction),
            _ => {}
        }

    Ok(legal)

pub fn spawn_tile(board: &Board, rng: u64) -> Result<SpawnResult, Error> {
