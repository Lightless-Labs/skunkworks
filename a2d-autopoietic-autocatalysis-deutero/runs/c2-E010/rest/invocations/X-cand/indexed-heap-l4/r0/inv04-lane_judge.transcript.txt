pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,

pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    capacity: Option<usize>,
    next_age: u128,
}

fn canonicalize(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();
    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

fn rank_cmp(a: &Entry, b: &Entry) -> Ordering {
    a.item
        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            capacity: None,
            next_age: 0,
        }
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self {
            entries: Vec::new(),
            capacity: Some(capacity),
            next_age: 0,
        }
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn find_index(&self, key: &str) -> Option<usize> {
        self.entries.iter().position(|e| e.item.key == key)
    }

    fn best_index(&self) -> Option<usize> {
        self.entries
            .iter()
            .enumerate()
            .min_by(|(_, a), (_, b)| rank_cmp(a, b))
            .map(|(i, _)| i)
    }

    fn worst_index(&self) -> Option<usize> {
        self.entries
            .iter()
            .enumerate()
            .max_by(|(_, a), (_, b)| rank_cmp(a, b))
            .map(|(i, _)| i)
    }

    fn sorted_entries(&self) -> Vec<&Entry> {
        let mut refs: Vec<&Entry> = self.entries.iter().collect();
        refs.sort_by(|a, b| rank_cmp(a, b));
        refs
    }

    fn trim_to_capacity(&mut self) -> Vec<Item> {
        let Some(limit) = self.capacity else {
            return Vec::new();
        };
        let mut evicted = Vec::new();
        while self.entries.len() > limit {
            let i = self.worst_index().expect("nonempty while over limit");
            evicted.push(self.entries.remove(i).item);
        }
        evicted
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut order: Vec<(usize, u128)> = self
                .entries
                .iter()
                .enumerate()
                .map(|(i, e)| (i, e.age))
                .collect();
            order.sort_by_key(|&(_, age)| age);
            for (new_age, (i, _)) in order.into_iter().enumerate() {
                self.entries[i].age = new_age as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = canonicalize(key)?;
        if self.find_index(&key).is_some() {
            return Err(Error::Duplicate { key });
        }

        let proposed = Entry {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };

        if let Some(limit) = self.capacity {
            if limit == 0 {
                return Ok(Admission::Discarded(proposed.item));
            }
            if self.entries.len() >= limit {
                let worst = self.worst_index().expect("full nonzero queue is nonempty");
                if rank_cmp(&proposed, &self.entries[worst]) != Ordering::Less {
                    return Ok(Admission::Discarded(proposed.item));
                }
                let evicted = self.entries.remove(worst).item;
                let age = self.fresh_age();
                self.entries.push(Entry { age, ..proposed });
                return Ok(Admission::Evicted(evicted));
            }
        }

        let age = self.fresh_age();
        self.entries.push(Entry { age, ..proposed });
        Ok(Admission::Inserted)
    }

        self.best_index().map(|i| self.entries[i].item.clone())

        self.best_index().map(|i| self.entries.remove(i).item)

                requested: priority,

        let i = self

            .ok_or_else(|| Error::Missing { key })?;

        self.capacity = capacity;
        let mut evicted = self.trim_to_capacity();
        evicted.reverse();
        evicted
    }

            .map(|e| e.item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,

            .sorted_entries()
            .into_iter()
            .cloned()
            .collect();

            for source in &source_order {
                if self.find_index(&source.item.key).is_some() {
                    return Err(Error::Collision {
                        key: source.item.key.clone(),
                    });
                }
            }
        }

        let mut moved = 0;
        let mut replaced = 0;
        let mut discarded = 0;

        for source in source_order {
            if let Some(i) = self.find_index(&source.item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[i].item.priority = source.item.priority;
                        self.entries[i].item.value = source.item.value;
                        replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                self.entries.push(Entry {
                    item: source.item,
                    age,
                });
                moved += 1;
            }
        }

        other.entries.clear();
        let evicted = self.trim_to_capacity();
        Ok(MergeReport {
            moved,
            replaced,
            discarded,
            evicted,
        })
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = canonicalize_key(key)?;
        if self.entries.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let proposed = Item {
            key: key.clone(),
            priority,
            value: value.to_owned(),
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed));
        }

        let age = self.next_age;
        if self.next_age < u128::MAX {
            self.next_age += 1;
        } else {
            self.renumber_ages();
        }

        if let Some(limit) = self.capacity {
            if self.entries.len() >= limit {
                let worst_key = self.worst_key().expect("nonempty full queue");
                let proposed_entry = Entry {
                    item: proposed.clone(),
                    age,
                };
                if compare_entries(&proposed_entry, self.entries.get(&worst_key).unwrap())
                    != std::cmp::Ordering::Less
                {
                    return Ok(Admission::Discarded(proposed));
                }
                let evicted = self.entries.remove(&worst_key).unwrap().item;
                self.entries.insert(
                    key,
                    Entry {
                        item: proposed,
                        age,
                    },
                );
                return Ok(Admission::Evicted(evicted));
            }
        }

        self.entries.insert(
            key,
            Entry {
                item: proposed,
                age,
            },
        );
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_key()
            .and_then(|key| self.entries.get(&key).map(|e| e.item.clone()))
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.best_key()?;
        self.entries.remove(&key).map(|e| e.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize_key(key)?;
        Ok(self.entries.get(&key).map(|e| e.item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize_key(key)?;
        let current = self
            .entries
            .get(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?
            .item
            .priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries.get_mut(&key).unwrap().item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize_key(key)?;
        let current = self
            .entries
            .get(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?
            .item
            .priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries.get_mut(&key).unwrap().item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize_key(key)?;
        let entry = self
            .entries
            .get_mut(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        Ok(std::mem::replace(&mut entry.item.value, value.to_owned()))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize_key(key)?;
        self.entries
            .remove(&key)
            .map(|entry| entry.item)
            .ok_or(Error::Missing { key })
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let excess = capacity
            .map(|limit| self.entries.len().saturating_sub(limit))
            .unwrap_or(0);
        let mut evicted = Vec::with_capacity(excess);
        for _ in 0..excess {
            let key = self.worst_key().unwrap();
            evicted.push(self.entries.remove(&key).unwrap().item);
        }
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        let mut ordered: Vec<&Entry> = self.entries.values().collect();
        ordered.sort_by(|a, b| compare_entries(a, b));
        ordered.into_iter().map(|e| e.item.clone()).collect::<Vec<_>>().into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source = other.ordered_keys();
        if policy == CollisionPolicy::Reject {
            for key in &source {
                if self.entries.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for key in source {
            let source_entry = other.entries.remove(&key).unwrap();
            if let Some(receiver_entry) = self.entries.get_mut(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        receiver_entry.item.priority = source_entry.item.priority;
                        receiver_entry.item.value = source_entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.allocate_age();
                self.entries.insert(
                    key,
                    Entry {
                        item: source_entry.item,
                        age,
                    },
                );
                report.moved += 1;
            }
        }

        if let Some(limit) = self.capacity {
            let excess = self.entries.len().saturating_sub(limit);
            for _ in 0..excess {
                let key = self.worst_key().unwrap();
                report.evicted.push(self.entries.remove(&key).unwrap().item);
            }
        }
        Ok(report)
    }

    fn ordered_keys(&self) -> Vec<String> {
        let mut entries: Vec<&Entry> = self.entries.values().collect();
        entries.sort_by(|a, b| compare_entries(a, b));
        entries.into_iter().map(|e| e.item.key.clone()).collect()
    }

    fn best_key(&self) -> Option<String> {
        self.entries
            .values()
            .min_by(|a, b| compare_entries(a, b))
            .map(|e| e.item.key.clone())
    }

    fn worst_key(&self) -> Option<String> {
        self.entries
            .values()
            .max_by(|a, b| compare_entries(a, b))
            .map(|e| e.item.key.clone())
    }

    fn allocate_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            self.renumber_ages();
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    fn renumber_ages(&mut self) {
        let mut entries: Vec<(String, u128)> = self
            .entries
            .values()
            .map(|e| (e.item.key.clone(), e.age))
            .collect();
        entries.sort_by_key(|(_, age)| *age);
        for (index, (key, _)) in entries.into_iter().enumerate() {
            if let Some(entry) = self.entries.get_mut(&key) {
                entry.age = index as u128;
            }
        }
        self.next_age = self.entries.len() as u128;
    }
}

fn compare_entries(a: &Entry, b: &Entry) -> std::cmp::Ordering {
    a.item
        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))
}

fn canonicalize_key(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();
    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))

impl IndexedHeap {
    pub fn new() -> Self {

    pub fn with_capacity(capacity: usize) -> Self {
        Self::with_capacity_inner(Some(capacity))

            capacity,
            next_age: 0,

    fn ensure_age_room(&mut self) {
        if self.next_age == usize::MAX {
            let mut order: Vec<usize> = (0..self.entries.len()).collect();
            order.sort_by_key(|&i| self.entries[i].age);
            for (age, i) in order.into_iter().enumerate() {
                self.entries[i].age = age;
            }
            self.next_age = self.entries.len();
        }

        let idx = self.entries.len();
        self.indices.insert(item.key.clone(), idx);
        self.entries.push(Entry { item, age });

    fn remove_index(&mut self, idx: usize) -> Entry {
        let removed = self.entries.swap_remove(idx);
        self.indices.remove(&removed.item.key);
        if idx < self.entries.len() {
            self.indices.insert(self.entries[idx].item.key.clone(), idx);
        }
        removed

        self.indices.get(key).copied()

        indices.sort_by(|&a, &b| cmp_entry(&self.entries[a], &self.entries[b]));

        self.entries.len()

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(item));
        }
        if let Some(cap) = self.capacity {
            if self.entries.len() >= cap {
                let worst = self
                    .ordered_indices()
                    .last()
                    .copied()
                    .expect("full nonzero queue has an entry");
                let proposed = Entry {
                    item: item.clone(),
                    age: self.next_age,
                };
                if cmp_entry(&proposed, &self.entries[worst]) != std::cmp::Ordering::Less {
                    return Ok(Admission::Discarded(item));

                let evicted = self.remove_index(worst).item;
                self.ensure_age_room();
                let age = self.next_age;
                self.next_age += 1;
                self.insert_raw(item, age);
                return Ok(Admission::Evicted(evicted));
            }
        }
        self.ensure_age_room();
        let age = self.next_age;
        self.next_age += 1;
        self.insert_raw(item, age);
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.ordered_indices()

            .map(|&i| self.entries[i].item.clone())

        let idx = self.ordered_indices().first().copied()?;
        Some(self.remove_index(idx).item)

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {

        let current = self.entries[idx].item.priority;
        if priority >= current {

        self.entries[idx].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let idx = self.find_index(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.entries[idx].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection { current, requested: priority });
        }
        self.entries[idx].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize(key)?;
        let idx = self.find_index(&key).ok_or_else(|| Error::Missing { key })?;
        Ok(std::mem::replace(&mut self.entries[idx].item.value, value.to_owned()))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize(key)?;
        let idx = self.find_index(&key).ok_or_else(|| Error::Missing { key })?;
        Ok(self.remove_index(idx).item)
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut evicted = Vec::new();
        if let Some(cap) = capacity {
            while self.entries.len() > cap {
                let idx = self.ordered_indices().last().copied().unwrap();
                evicted.push(self.remove_index(idx).item);
            }
        }
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.ordered_indices()
            .into_iter()
            .map(|i| self.entries[i].item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source_order = other.ordered_indices();
        if policy == CollisionPolicy::Reject {
            for &idx in &source_order {
                let key = &other.entries[idx].item.key;
                if self.indices.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let mut moved = 0;
        let mut replaced = 0;
        let mut discarded = 0;
        for idx in source_order {
            let source = other.entries[idx].item.clone();
            if let Some(&dest_idx) = self.indices.get(&source.key) {
                match policy {
                    CollisionPolicy::KeepSelf => discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[dest_idx].item.priority = source.priority;
                        self.entries[dest_idx].item.value = source.value;
                        replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                self.ensure_age_room();
                let age = self.next_age;
                self.next_age += 1;
                self.insert_raw(source, age);
                moved += 1;
            }
        }
        other.entries.clear();
        other.indices.clear();

        let mut evicted = Vec::new();
        if let Some(cap) = self.capacity {
            while self.entries.len() > cap {
                let idx = self.ordered_indices().last().copied().unwrap();
                evicted.push(self.remove_index(idx).item);
            }
        }
        Ok(MergeReport {
            moved,
            replaced,
            discarded,
            evicted,
        })
    }
}

    fn canonical(input: &str) -> Result<String, Error> {
        let data = input.as_bytes();
        let mut lo = 0usize;
        let mut hi = data.len();
        while lo < hi && data[lo] == 0x20 {
            lo += 1;
        }
        while hi > lo && data[hi - 1] == 0x20 {
            hi -= 1;
        }
        if lo == hi {
            return Err(Error::EmptyKey);
        }

        for i in lo..hi {
            let b = data[i];
            let valid = if i == lo {
                b.is_ascii_alphabetic()
            } else {
                b.is_ascii_alphabetic() || b.is_ascii_digit() || b == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: i });
            }
        }
        Ok(input[lo..hi].to_ascii_lowercase())

    fn cmp(a: &Node, b: &Node) -> Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then(a.item.value.len().cmp(&b.item.value.len()))
            .then(a.age.cmp(&b.age))
    }

    fn refresh_map(&mut self) {
        self.by_key.clear();
        for (i, node) in self.nodes.iter().enumerate() {
            self.by_key.insert(node.item.key.clone(), i);
        }
    }

    fn normalize_ages(&mut self) {
        if self.age_counter < u128::MAX {
            return;
        }
        let mut indices: Vec<usize> = (0..self.nodes.len()).collect();
        indices.sort_by_key(|&i| self.nodes[i].age);
        for (age, index) in indices.into_iter().enumerate() {
            self.nodes[index].age = age as u128;
        }
        self.age_counter = self.nodes.len() as u128;
    }

    fn next_age(&mut self) -> u128 {
        self.normalize_ages();
        let age = self.age_counter;
        self.age_counter += 1;
        age
    }

        self.nodes.sort_by(Self::cmp);
        self.refresh_map();
    }

        self.nodes

            .enumerate()

            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age: self.age_counter,
        };

            return Ok(Admission::Discarded(proposed.item));
        }

        if let Some(limit) = self.limit {
            if self.nodes.len() == limit {
                let worst_index = self.worst().expect("nonzero full queue has an item");
                if Self::cmp(&proposed, &self.nodes[worst_index]) != Ordering::Less {
                    return Ok(Admission::Discarded(proposed.item));

                self.nodes.push(Node { age, ..proposed });
                self.sort();
                return Ok(Admission::Evicted(old));
            }
        }

        let age = self.next_age();
        self.nodes.push(Node { age, ..proposed });
        self.sort();
        Ok(Admission::Inserted)
    }

                evicted.push(self.remove_at(i).item);

        self.nodes.iter().map(|n| n.item.clone()).collect::<Vec<_>>().into_iter()

            item: n.item.clone(),
            age: n.age,

            for node in &source {
                if self.by_key.contains_key(&node.item.key) {
                    return Err(Error::Collision { key: node.item.key.clone() });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),

            if let Some(&i) = self.by_key.get(&node.item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.nodes[i].item.priority = node.item.priority;
                        self.nodes[i].item.value = node.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),

            } else {
                let age = self.next_age();
                self.nodes.push(Node { item: node.item, age });
                report.moved += 1;
            }
            self.sort();
        }

        other.nodes.clear();
        other.by_key.clear();

        if let Some(limit) = self.limit {
            while self.nodes.len() > limit {
                let i = self.worst().expect("over limit means nonempty");
                report.evicted.push(self.remove_at(i).item);
            }
        }
        Ok(report)
    }
}

    let bytes = input.as_bytes();
    let start = bytes.iter().take_while(|&&b| b == b' ').count();
    let end = bytes
        .iter()
        .rposition(|&b| b != b' ')
        .map(|i| i + 1)

            .map(|(k, _)| k.clone())
    }

    fn age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut keys: Vec<String> = self.entries.keys().cloned().collect();
            keys.sort_by_key(|k| self.entries[k].age);
            for (n, key) in keys.into_iter().enumerate() {
                self.entries.get_mut(&key).unwrap().age = n as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let n = self.next_age;
        self.next_age += 1;
        n
    }

    fn trim(&mut self) -> Vec<Item> {
        let mut out = Vec::new();
        while self.capacity.is_some_and(|c| self.entries.len() > c) {
            let key = self.worst_key().unwrap();
            out.push(self.entries.remove(&key).unwrap().item);
        }
        out
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = canonicalize(key)?;
        if self.entries.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }
        let proposed = Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed.item));
        }

        if self.capacity.is_some_and(|c| self.entries.len() >= c) {
            let worst = self.worst_key().unwrap();
            if cmp_entry(&proposed, &self.entries[&worst]) != Ordering::Less {
                return Ok(Admission::Discarded(proposed.item));
            }
            let evicted = self.entries.remove(&worst).unwrap().item;
            let age = self.age();
            self.entries.insert(key, Entry { age, ..proposed });
            return Ok(Admission::Evicted(evicted));
        }

        let age = self.age();
        self.entries.insert(key, Entry { age, ..proposed });
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_key().and_then(|k| self.entries.get(&k).map(|e| e.item.clone()))
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.best_key()?;
        self.entries.remove(&key).map(|e| e.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        Ok(self.entries.get(&key).map(|e| e.item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let e = self.entries.get_mut(&key).ok_or_else(|| Error::Missing { key: key.clone() })?;
        if priority >= e.item.priority {
            return Err(Error::WrongDirection { current: e.item.priority, requested: priority });
        }
        let old = e.item.priority;
        e.item.priority = priority;
        Ok(old)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let e = self.entries.get_mut(&key).ok_or_else(|| Error::Missing { key: key.clone() })?;
        if priority <= e.item.priority {
            return Err(Error::WrongDirection { current: e.item.priority, requested: priority });
        }
        let old = e.item.priority;
        e.item.priority = priority;
        Ok(old)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize(key)?;
        let e = self.entries.get_mut(&key).ok_or_else(|| Error::Missing { key: key.clone() })?;
        Ok(std::mem::replace(&mut e.item.value, value.to_owned()))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize(key)?;
        self.entries
            .remove(&key)
            .map(|e| e.item)
            .ok_or(Error::Missing { key })
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut out = self.trim();
        out.reverse();
        out
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        let mut values: Vec<&Entry> = self.entries.values().collect();
        values.sort_by(|a, b| cmp_entry(a, b));
        values.into_iter().map(|e| e.item.clone()).collect::<Vec<_>>().into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let mut source: Vec<Entry> = other.entries.values().cloned().collect();
        source.sort_by(cmp_entry);

        if policy == CollisionPolicy::Reject {
            for e in &source {
                if self.entries.contains_key(&e.item.key) {
                    return Err(Error::Collision { key: e.item.key.clone() });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for mut incoming in source {
            let key = incoming.item.key.clone();
            if let Some(existing) = self.entries.get_mut(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        existing.item.priority = incoming.item.priority;
                        existing.item.value = incoming.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                incoming.age = self.age();
                self.entries.insert(key, incoming);
                report.moved += 1;
            }
        }

        other.entries.clear();
        report.evicted = self.trim();
        Ok(report)
    }
}

use std::cmp::Ordering;
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    capacity: Option<usize>,
    next_age: u128,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self::with_optional_capacity(None)
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self::with_optional_capacity(Some(capacity))
    }

    fn with_optional_capacity(capacity: Option<usize>) -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            capacity,
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    fn canonical_key(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();
        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }
        if start == end {
            return Err(Error::EmptyKey);
        }

        for (offset, &byte) in bytes[start..end].iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic()
            } else {
                byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + offset });
            }
        }
        Ok(input[start..end].to_ascii_lowercase())
    }

    // Less means a is better than b.
    fn compare(a: &Entry, b: &Entry) -> Ordering {
        a.item
            .priority
            .cmp(&b.item.priority)
            .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
            .then_with(|| a.age.cmp(&b.age))
    }

    fn rebuild_indices(&mut self) {
        self.indices.clear();
        for (index, entry) in self.entries.iter().enumerate() {
            self.indices.insert(entry.item.key.clone(), index);
        }
    }

    fn sort_best_first(&mut self) {
        self.entries.sort_by(Self::compare);
        self.rebuild_indices();
    }

    fn worst_index(&self) -> Option<usize> {
        self.entries
            .iter()
            .enumerate()
            .max_by(|(_, a), (_, b)| Self::compare(a, b))
            .map(|(index, _)| index)
    }

    fn remove_index(&mut self, index: usize) -> Entry {
        let removed = self.entries.remove(index);
        self.indices.remove(&removed.item.key);
        self.rebuild_indices();
        removed
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            // Renumber in current age order while preserving every relative age.
            let mut order: Vec<usize> = (0..self.entries.len()).collect();
            order.sort_by_key(|&i| self.entries[i].age);
            for (age, index) in order.into_iter().enumerate() {
                self.entries[index].age = age as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = Self::canonical_key(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let proposed = Entry {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed.item));
        }

        if let Some(limit) = self.capacity {
            if self.entries.len() >= limit {
                let worst_index = self.worst_index().expect("full nonzero queue is nonempty");
                if Self::compare(&proposed, &self.entries[worst_index]) != Ordering::Less {
                    return Ok(Admission::Discarded(proposed.item));
                }
                let evicted = self.remove_index(worst_index).item;
                let age = self.fresh_age();
                self.entries.push(Entry { age, ..proposed });
                self.sort_best_first();
                return Ok(Admission::Evicted(evicted));
            }
        }

        let age = self.fresh_age();
        self.entries.push(Entry { age, ..proposed });
        self.sort_best_first();
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.entries.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.entries.is_empty() {
            None
        } else {
            Some(self.remove_index(0).item)
        }
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = Self::canonical_key(key)?;
        Ok(self.indices.get(&key).map(|&i| self.entries[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.entries[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        self.sort_best_first();
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonical_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let current = self.entries[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        self.sort_best_first();
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = Self::canonical_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        let previous = std::mem::replace(
            &mut self.entries[index].item.value,
            value.to_owned(),
        );
        self.sort_best_first();
        Ok(previous)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = Self::canonical_key(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing { key })?;
        Ok(self.remove_index(index).item)
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut evicted = Vec::new();
        if let Some(limit) = capacity {
            while self.entries.len() > limit {
                let index = self.worst_index().expect("excess implies nonempty");
                evicted.push(self.remove_index(index).item);
            }
        }
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.entries
            .iter()
            .map(|entry| entry.item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source: Vec<Entry> = other.entries.clone();
        if policy == CollisionPolicy::Reject {
            for entry in &source {
                if self.indices.contains_key(&entry.item.key) {
                    return Err(Error::Collision {
                        key: entry.item.key.clone(),
                    });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for entry in source {
            match self.indices.get(&entry.item.key).copied() {
                None => {
                    let age = self.fresh_age();
                    self.entries.push(Entry {
                        item: entry.item,
                        age,
                    });
                    report.moved += 1;
                }
                Some(index) => match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[index].item.priority = entry.item.priority;
                        self.entries[index].item.value = entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                },
            }
        }

        self.sort_best_first();
        other.entries.clear();
        other.indices.clear();

        if let Some(limit) = self.capacity {
            while self.entries.len() > limit {
                let index = self.worst_index().expect("excess implies nonempty");
                report.evicted.push(self.remove_index(index).item);
            }
        }
        Ok(report)
    }
}
