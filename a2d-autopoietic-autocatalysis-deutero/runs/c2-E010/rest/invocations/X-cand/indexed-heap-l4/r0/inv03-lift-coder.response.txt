- Keys are trimmed only for ASCII spaces, then validated byte by byte; error offsets refer to the original input.
- Ranking uses priority, UTF-8 value byte length, then insertion age; ties keep the older item first.
- At capacity, a new item loses a complete rank tie; capacity zero discards every valid nonduplicate proposal.
- Merge checks Reject collisions in source rank order before mutation; successful merges rebase transferred ages, then trim the destination.
- Capacity trimming returns evictions worst-to-best; merge counts are computed before trimming.
- Both solutions use owned vectors and linear searches, avoiding index bookkeeping while preserving the specified behavior.

### Solution A

```rust
use std::cmp::Ordering;
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    capacity: Option<usize>,
    next_age: u128,
}

fn canonicalize(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();
    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for (offset, &byte) in bytes[start..end].iter().enumerate() {
        let valid = if offset == 0 {
            byte.is_ascii_alphabetic()
        } else {
            byte.is_ascii_alphanumeric() || byte == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

fn rank_cmp(a: &Entry, b: &Entry) -> Ordering {
    a.item
        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            capacity: None,
            next_age: 0,
        }
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self {
            entries: Vec::new(),
            capacity: Some(capacity),
            next_age: 0,
        }
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn find_index(&self, key: &str) -> Option<usize> {
        self.entries.iter().position(|e| e.item.key == key)
    }

    fn best_index(&self) -> Option<usize> {
        self.entries
            .iter()
            .enumerate()
            .min_by(|(_, a), (_, b)| rank_cmp(a, b))
            .map(|(i, _)| i)
    }

    fn worst_index(&self) -> Option<usize> {
        self.entries
            .iter()
            .enumerate()
            .max_by(|(_, a), (_, b)| rank_cmp(a, b))
            .map(|(i, _)| i)
    }

    fn sorted_entries(&self) -> Vec<&Entry> {
        let mut refs: Vec<&Entry> = self.entries.iter().collect();
        refs.sort_by(|a, b| rank_cmp(a, b));
        refs
    }

    fn trim_to_capacity(&mut self) -> Vec<Item> {
        let Some(limit) = self.capacity else {
            return Vec::new();
        };
        let mut evicted = Vec::new();
        while self.entries.len() > limit {
            let i = self.worst_index().expect("nonempty while over limit");
            evicted.push(self.entries.remove(i).item);
        }
        evicted
    }

    fn fresh_age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut order: Vec<(usize, u128)> = self
                .entries
                .iter()
                .enumerate()
                .map(|(i, e)| (i, e.age))
                .collect();
            order.sort_by_key(|&(_, age)| age);
            for (new_age, (i, _)) in order.into_iter().enumerate() {
                self.entries[i].age = new_age as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = canonicalize(key)?;
        if self.find_index(&key).is_some() {
            return Err(Error::Duplicate { key });
        }

        let proposed = Entry {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };

        if let Some(limit) = self.capacity {
            if limit == 0 {
                return Ok(Admission::Discarded(proposed.item));
            }
            if self.entries.len() >= limit {
                let worst = self.worst_index().expect("full nonzero queue is nonempty");
                if rank_cmp(&proposed, &self.entries[worst]) != Ordering::Less {
                    return Ok(Admission::Discarded(proposed.item));
                }
                let evicted = self.entries.remove(worst).item;
                let age = self.fresh_age();
                self.entries.push(Entry { age, ..proposed });
                return Ok(Admission::Evicted(evicted));
            }
        }

        let age = self.fresh_age();
        self.entries.push(Entry { age, ..proposed });
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_index().map(|i| self.entries[i].item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        self.best_index().map(|i| self.entries.remove(i).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        Ok(self.find_index(&key).map(|i| self.entries[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let i = self
            .find_index(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = self.entries[i].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[i].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let i = self
            .find_index(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = self.entries[i].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[i].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize(key)?;
        let i = self
            .find_index(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let old = std::mem::replace(&mut self.entries[i].item.value, value.to_owned());
        Ok(old)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize(key)?;
        let i = self
            .find_index(&key)
            .ok_or_else(|| Error::Missing { key })?;
        Ok(self.entries.remove(i).item)
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut evicted = self.trim_to_capacity();
        evicted.reverse();
        evicted
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.sorted_entries()
            .into_iter()
            .map(|e| e.item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source_order: Vec<Entry> = other
            .sorted_entries()
            .into_iter()
            .cloned()
            .collect();

        if policy == CollisionPolicy::Reject {
            for source in &source_order {
                if self.find_index(&source.item.key).is_some() {
                    return Err(Error::Collision {
                        key: source.item.key.clone(),
                    });
                }
            }
        }

        let mut moved = 0;
        let mut replaced = 0;
        let mut discarded = 0;

        for source in source_order {
            if let Some(i) = self.find_index(&source.item.key) {
                match policy {
                    CollisionPolicy::KeepSelf => discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[i].item.priority = source.item.priority;
                        self.entries[i].item.value = source.item.value;
                        replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let age = self.fresh_age();
                self.entries.push(Entry {
                    item: source.item,
                    age,
                });
                moved += 1;
            }
        }

        other.entries.clear();
        let evicted = self.trim_to_capacity();
        Ok(MergeReport {
            moved,
            replaced,
            discarded,
            evicted,
        })
    }
}
```

### Solution B

```rust
use std::cmp::Ordering;
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: BTreeMap<String, Entry>,
    capacity: Option<usize>,
    next_age: u128,
}

fn canonicalize(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let start = bytes.iter().take_while(|&&b| b == b' ').count();
    let end = bytes
        .iter()
        .rposition(|&b| b != b' ')
        .map(|i| i + 1)
        .unwrap_or(start);
    if start >= end {
        return Err(Error::EmptyKey);
    }

    for (offset, &b) in bytes[start..end].iter().enumerate() {
        let ok = if offset == 0 {
            b.is_ascii_alphabetic()
        } else {
            b.is_ascii_alphanumeric() || b == b'-'
        };
        if !ok {
            return Err(Error::InvalidKey { at: start + offset });
        }
    }
    Ok(input[start..end].to_ascii_lowercase())
}

fn cmp_entry(a: &Entry, b: &Entry) -> Ordering {
    (a.item.priority, a.item.value.len(), a.age).cmp(&(
        b.item.priority,
        b.item.value.len(),
        b.age,
    ))
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: BTreeMap::new(),
            capacity: None,
            next_age: 0,
        }
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self {
            entries: BTreeMap::new(),
            capacity: Some(capacity),
            next_age: 0,
        }
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.entries.is_empty()
    }

    fn best_key(&self) -> Option<String> {
        self.entries
            .iter()
            .min_by(|(_, a), (_, b)| cmp_entry(a, b))
            .map(|(k, _)| k.clone())
    }

    fn worst_key(&self) -> Option<String> {
        self.entries
            .iter()
            .max_by(|(_, a), (_, b)| cmp_entry(a, b))
            .map(|(k, _)| k.clone())
    }

    fn age(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut keys: Vec<String> = self.entries.keys().cloned().collect();
            keys.sort_by_key(|k| self.entries[k].age);
            for (n, key) in keys.into_iter().enumerate() {
                self.entries.get_mut(&key).unwrap().age = n as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let n = self.next_age;
        self.next_age += 1;
        n
    }

    fn trim(&mut self) -> Vec<Item> {
        let mut out = Vec::new();
        while self.capacity.is_some_and(|c| self.entries.len() > c) {
            let key = self.worst_key().unwrap();
            out.push(self.entries.remove(&key).unwrap().item);
        }
        out
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = canonicalize(key)?;
        if self.entries.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }
        let proposed = Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age: self.next_age,
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed.item));
        }

        if self.capacity.is_some_and(|c| self.entries.len() >= c) {
            let worst = self.worst_key().unwrap();
            if cmp_entry(&proposed, &self.entries[&worst]) != Ordering::Less {
                return Ok(Admission::Discarded(proposed.item));
            }
            let evicted = self.entries.remove(&worst).unwrap().item;
            let age = self.age();
            self.entries.insert(key, Entry { age, ..proposed });
            return Ok(Admission::Evicted(evicted));
        }

        let age = self.age();
        self.entries.insert(key, Entry { age, ..proposed });
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.best_key().and_then(|k| self.entries.get(&k).map(|e| e.item.clone()))
    }

    pub fn pop(&mut self) -> Option<Item> {
        let key = self.best_key()?;
        self.entries.remove(&key).map(|e| e.item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        Ok(self.entries.get(&key).map(|e| e.item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let e = self.entries.get_mut(&key).ok_or_else(|| Error::Missing { key: key.clone() })?;
        if priority >= e.item.priority {
            return Err(Error::WrongDirection { current: e.item.priority, requested: priority });
        }
        let old = e.item.priority;
        e.item.priority = priority;
        Ok(old)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let e = self.entries.get_mut(&key).ok_or_else(|| Error::Missing { key: key.clone() })?;
        if priority <= e.item.priority {
            return Err(Error::WrongDirection { current: e.item.priority, requested: priority });
        }
        let old = e.item.priority;
        e.item.priority = priority;
        Ok(old)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize(key)?;
        let e = self.entries.get_mut(&key).ok_or_else(|| Error::Missing { key: key.clone() })?;
        Ok(std::mem::replace(&mut e.item.value, value.to_owned()))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize(key)?;
        self.entries
            .remove(&key)
            .map(|e| e.item)
            .ok_or(Error::Missing { key })
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        let mut out = self.trim();
        out.reverse();
        out
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        let mut values: Vec<&Entry> = self.entries.values().collect();
        values.sort_by(|a, b| cmp_entry(a, b));
        values.into_iter().map(|e| e.item.clone()).collect::<Vec<_>>().into_iter()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let mut source: Vec<Entry> = other.entries.values().cloned().collect();
        source.sort_by(cmp_entry);

        if policy == CollisionPolicy::Reject {
            for e in &source {
                if self.entries.contains_key(&e.item.key) {
                    return Err(Error::Collision { key: e.item.key.clone() });
                }
            }
        }

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for mut incoming in source {
            let key = incoming.item.key.clone();
            if let Some(existing) = self.entries.get_mut(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        existing.item.priority = incoming.item.priority;
                        existing.item.value = incoming.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                incoming.age = self.age();
                self.entries.insert(key, incoming);
                report.moved += 1;
            }
        }

        other.entries.clear();
        report.evicted = self.trim();
        Ok(report)
    }
}
```
