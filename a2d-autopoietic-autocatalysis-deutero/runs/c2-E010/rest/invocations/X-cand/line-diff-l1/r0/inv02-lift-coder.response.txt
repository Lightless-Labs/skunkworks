- Check both length limits first; 128 elements is allowed.
- Validate all Old lines before New lines, scanning each side by index.
- Treat only LF and CR as embedded terminators; empty strings are valid lines.
- Follow the specified LCS lookahead and parity tie-break exactly.
- At exhausted suffixes, emit the remaining operations in order.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    TooLarge {
        old_len: usize,
        new_len: usize,
    },
    EmbeddedNewline {
        side: Side,
        index: usize,
    },
}

pub fn diff_lines(old: &[String], new: &[String]) -> Result<Vec<Edit>, DiffError> {
    if old.len() > 128 || new.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old.len(),
            new_len: new.len(),
        });
    }

    for (index, line) in old.iter().enumerate() {
        if line.contains('\n') || line.contains('\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::Old,
                index,
            });
        }
    }

    for (index, line) in new.iter().enumerate() {
        if line.contains('\n') || line.contains('\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::New,
                index,
            });
        }
    }

    let rows = old.len() + 1;
    let cols = new.len() + 1;
    let mut lcs = vec![vec![0usize; cols]; rows];

    for i in (0..old.len()).rev() {
        for j in (0..new.len()).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::with_capacity(old.len() + new.len());
    let (mut i, mut j) = (0, 0);

    while i < old.len() && j < new.len() {
        if old[i] == new[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].clone(),
            });
            i += 1;
            j += 1;
        } else {
            let delete_lcs = lcs[i + 1][j];
            let insert_lcs = lcs[i][j + 1];

            if delete_lcs > insert_lcs || (delete_lcs == insert_lcs && (i + j) % 2 == 0) {
                edits.push(Edit::Delete {
                    old_index: i,
                    line: old[i].clone(),
                });
                i += 1;
            } else {
                edits.push(Edit::Insert {
                    new_index: j,
                    line: new[j].clone(),
                });
                j += 1;
            }
        }
    }

    while i < old.len() {
        edits.push(Edit::Delete {
            old_index: i,
            line: old[i].clone(),
        });
        i += 1;
    }

    while j < new.len() {
        edits.push(Edit::Insert {
            new_index: j,
            line: new[j].clone(),
        });
        j += 1;
    }

    Ok(edits)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    TooLarge {
        old_len: usize,
        new_len: usize,
    },
    EmbeddedNewline {
        side: Side,
        index: usize,
    },
}

pub fn diff_lines(old: &[String], new: &[String]) -> Result<Vec<Edit>, DiffError> {
    if old.len() > 128 || new.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old.len(),
            new_len: new.len(),
        });
    }

    for index in 0..old.len() {
        if old[index].chars().any(|c| c == '\n' || c == '\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::Old,
                index,
            });
        }
    }
    for index in 0..new.len() {
        if new[index].chars().any(|c| c == '\n' || c == '\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::New,
                index,
            });
        }
    }

    let n = old.len();
    let m = new.len();
    let mut suffix_lcs = vec![0usize; (n + 1) * (m + 1)];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            let here = i * (m + 1) + j;
            suffix_lcs[here] = if old[i] == new[j] {
                1 + suffix_lcs[(i + 1) * (m + 1) + j + 1]
            } else {
                suffix_lcs[(i + 1) * (m + 1) + j]
                    .max(suffix_lcs[i * (m + 1) + j + 1])
            };
        }
    }

    let mut result = Vec::with_capacity(n + m);
    let (mut i, mut j) = (0usize, 0usize);

    while i < n && j < m {
        if old[i] == new[j] {
            result.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].clone(),
            });
            i += 1;
            j += 1;
            continue;
        }

        let after_delete = suffix_lcs[(i + 1) * (m + 1) + j];
        let after_insert = suffix_lcs[i * (m + 1) + j + 1];
        if after_delete > after_insert
            || (after_delete == after_insert && (i + j) % 2 == 0)
        {
            result.push(Edit::Delete {
                old_index: i,
                line: old[i].clone(),
            });
            i += 1;
        } else {
            result.push(Edit::Insert {
                new_index: j,
                line: new[j].clone(),
            });
            j += 1;
        }
    }

    for (index, line) in old.iter().enumerate().skip(i) {
        result.push(Edit::Delete {
            old_index: index,
            line: line.clone(),
        });
    }
    for (index, line) in new.iter().enumerate().skip(j) {
        result.push(Edit::Insert {
            new_index: index,
            line: line.clone(),
        });
    }

    Ok(result)
}
```
