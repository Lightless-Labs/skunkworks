- Check both length limits first; return both actual lengths if either exceeds 128.
- Validate all Old lines before any New lines; LF and CR are the only invalid characters.
- Empty strings are valid lines; empty slices contain no lines.
- At equal cursors, Keep immediately; otherwise follow the suffix-LCS comparison and parity tie-break exactly.
- The maximum LCS table is small enough for a fixed-size dynamic programming implementation without overflow concerns.
- Solution B uses a different LCS table layout while preserving the specified decisions.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    TooLarge { old_len: usize, new_len: usize },
    EmbeddedNewline { side: Side, index: usize },
}

pub fn diff_lines(old: &[String], new: &[String]) -> Result<Vec<Edit>, DiffError> {
    if old.len() > 128 || new.len() > 128 {
        return Err(DiffError::TooLarge {
            old_len: old.len(),
            new_len: new.len(),
        });
    }

    for (index, line) in old.iter().enumerate() {
        if line.contains('\n') || line.contains('\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::Old,
                index,
            });
        }
    }

    for (index, line) in new.iter().enumerate() {
        if line.contains('\n') || line.contains('\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::New,
                index,
            });
        }
    }

    let old_len = old.len();
    let new_len = new.len();
    let mut lcs = vec![vec![0usize; new_len + 1]; old_len + 1];

    for i in (0..old_len).rev() {
        for j in (0..new_len).rev() {
            lcs[i][j] = if old[i] == new[j] {
                lcs[i + 1][j + 1] + 1
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let mut edits = Vec::with_capacity(old_len + new_len);
    let (mut i, mut j) = (0usize, 0usize);

    while i < old_len && j < new_len {
        if old[i] == new[j] {
            edits.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].clone(),
            });
            i += 1;
            j += 1;
        } else if lcs[i + 1][j] > lcs[i][j + 1] {
            edits.push(Edit::Delete {
                old_index: i,
                line: old[i].clone(),
            });
            i += 1;
        } else if lcs[i + 1][j] < lcs[i][j + 1] {
            edits.push(Edit::Insert {
                new_index: j,
                line: new[j].clone(),
            });
            j += 1;
        } else if (i + j) % 2 == 0 {
            edits.push(Edit::Delete {
                old_index: i,
                line: old[i].clone(),
            });
            i += 1;
        } else {
            edits.push(Edit::Insert {
                new_index: j,
                line: new[j].clone(),
            });
            j += 1;
        }
    }

    while i < old_len {
        edits.push(Edit::Delete {
            old_index: i,
            line: old[i].clone(),
        });
        i += 1;
    }

    while j < new_len {
        edits.push(Edit::Insert {
            new_index: j,
            line: new[j].clone(),
        });
        j += 1;
    }

    Ok(edits)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Side {
    Old,
    New,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Edit {
    Keep {
        old_index: usize,
        new_index: usize,
        line: String,
    },
    Delete {
        old_index: usize,
        line: String,
    },
    Insert {
        new_index: usize,
        line: String,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DiffError {
    TooLarge { old_len: usize, new_len: usize },
    EmbeddedNewline { side: Side, index: usize },
}

pub fn diff_lines(old: &[String], new: &[String]) -> Result<Vec<Edit>, DiffError> {
    let m = old.len();
    let n = new.len();

    if m > 128 || n > 128 {
        return Err(DiffError::TooLarge {
            old_len: m,
            new_len: n,
        });
    }

    for (index, line) in old.iter().enumerate() {
        if line.chars().any(|ch| ch == '\n' || ch == '\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::Old,
                index,
            });
        }
    }

    for (index, line) in new.iter().enumerate() {
        if line.chars().any(|ch| ch == '\n' || ch == '\r') {
            return Err(DiffError::EmbeddedNewline {
                side: Side::New,
                index,
            });
        }
    }

    // Store the suffix LCS values in a flat row-major table.
    let width = n + 1;
    let mut lcs = vec![0usize; (m + 1) * width];

    for i in (0..m).rev() {
        for j in (0..n).rev() {
            let here = i * width + j;
            lcs[here] = if old[i] == new[j] {
                lcs[(i + 1) * width + j + 1] + 1
            } else {
                lcs[(i + 1) * width + j].max(lcs[i * width + j + 1])
            };
        }
    }

    let mut result = Vec::with_capacity(m + n);
    let (mut i, mut j) = (0usize, 0usize);

    while i < m && j < n {
        if old[i] == new[j] {
            result.push(Edit::Keep {
                old_index: i,
                new_index: j,
                line: old[i].clone(),
            });
            i += 1;
            j += 1;
            continue;
        }

        let delete_score = lcs[(i + 1) * width + j];
        let insert_score = lcs[i * width + j + 1];

        if delete_score > insert_score
            || (delete_score == insert_score && (i + j) % 2 == 0)
        {
            result.push(Edit::Delete {
                old_index: i,
                line: old[i].clone(),
            });
            i += 1;
        } else {
            result.push(Edit::Insert {
                new_index: j,
                line: new[j].clone(),
            });
            j += 1;
        }
    }

    for (index, line) in old.iter().enumerate().skip(i) {
        result.push(Edit::Delete {
            old_index: index,
            line: line.clone(),
        });
    }

    for (index, line) in new.iter().enumerate().skip(j) {
        result.push(Edit::Insert {
            new_index: index,
            line: line.clone(),
        });
    }

    Ok(result)
}
```
