- Configuration checks run first, in the stated order, even for empty input.
- Marker validation checks each marker’s immediate original neighbors; control errors take precedence.
- A maximal normalized LF run of length *n* creates *n−1* empty lines and splits paragraphs, including at the input edges.
- Layout costs use pre-justification widths; ties compare full line-break vectors and then pre-justification strings.
- Scalar counts, not graphemes or display cells, determine widths; Rust string ordering resolves the final tie.
- The two solutions use the same interpretation, with independent dynamic-programming approaches.

### Solution A

```rust
use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    text: String,
    consumed: usize,
    marker: bool,
    pieces: usize,
    gaps: usize,
    final_line: bool,
}

#[derive(Clone)]
struct Plan {
    cost: u64,
    marker_lines: usize,
    lines: Vec<Line>,
}

fn plan_cmp(a: &Plan, b: &Plan) -> Ordering {
    a.cost
        .cmp(&b.cost)
        .then_with(|| a.marker_lines.cmp(&b.marker_lines))
        .then_with(|| a.lines.len().cmp(&b.lines.len()))
        .then_with(|| {
            for (x, y) in a.lines.iter().zip(&b.lines) {
                match y.consumed.cmp(&x.consumed) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        })
        .then_with(|| {
            for (x, y) in a.lines.iter().zip(&b.lines) {
                match x.text.cmp(&y.text) {
                    Ordering::Equal => {}
                    other => return other,
                }
            }
            Ordering::Equal
        })
}

fn better(slot: &mut Option<Plan>, candidate: Plan) {
    if slot.as_ref().map_or(true, |old| plan_cmp(&candidate, old) == Ordering::Less) {
        *slot = Some(candidate);
    }
}

fn line_cost(width: usize, displayed: usize, final_line: bool) -> u64 {
    let gap = (width - displayed) as u64;
    let square = gap * gap;
    if final_line { square / 4 } else { square }
}

fn parse_words(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut spelling = Vec::new();
    let mut markers = Vec::new();
    let flush = |words: &mut Vec<Word>, spelling: &mut Vec<char>, markers: &mut Vec<usize>| {
        if !spelling.is_empty() {
            words.push(Word {
                chars: std::mem::take(spelling),
                markers: std::mem::take(markers),
            });
        }
    };
    let mut marker_offset = 0usize;
    for &c in chars {
        if c == '~' {
            markers.push(marker_offset);
        } else if c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00a0}' {
            flush(&mut words, &mut spelling, &mut markers);
            marker_offset = 0;
        } else {
            spelling.push(c);
            marker_offset += 1;
        }
    }
    flush(&mut words, &mut spelling, &mut markers);
    words
}

fn solve_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    if words.is_empty() {
        return Vec::new();
    }
    let mut states: Vec<(usize, usize, bool)> = Vec::new();
    let mut ids = std::collections::HashMap::new();
    for wi in 0..words.len() {
        for off in 0..words[wi].chars.len() {
            for first in [false, true] {
                let id = states.len();
                states.push((wi, off, first));
                ids.insert((wi, off, first), id);
            }
        }
    }
    let terminal = states.len();
    let mut dp: Vec<Option<Plan>> = vec![None; terminal + 1];
    dp[terminal] = Some(Plan { cost: 0, marker_lines: 0, lines: Vec::new() });

    for id in (0..terminal).rev() {
        let (wi, off, first) = states[id];
        let indent = if first { options.first_indent } else { options.rest_indent };
        let cap = options.width - indent;
        let mut candidates: Vec<(usize, usize, bool, usize, usize, String, usize)> = Vec::new();
        let mut text = String::new();
        let mut consumed = 0usize;
        let mut pieces = 0usize;
        let mut w = wi;
        let mut start = off;
        let mut occupied_width = 0usize;

        loop {
            let word = &words[w];
            let remaining = word.chars.len() - start;
            if pieces > 0 {
                if occupied_width + 1 >= cap {
                    break;
                }
                text.push(' ');
                occupied_width += 1;
            }
            let available = cap - occupied_width;
            let marker_fit = word.markers.iter().copied()
                .filter(|&m| m > start && m < word.chars.len() && m - start + 1 <= available)
                .collect::<Vec<_>>();

            if remaining <= available {
                text.extend(word.chars[start..].iter());
                occupied_width += remaining;
                consumed += remaining;
                pieces += 1;
                let next = if w + 1 == words.len() {
                    terminal
                } else {
                    *ids.get(&(w + 1, 0, false)).unwrap()
                };
                candidates.push((next, consumed, false, pieces, 0, text.clone(), occupied_width));
            } else {
                for m in marker_fit {
                    let mut split_text = text.clone();
                    split_text.extend(word.chars[start..m].iter());
                    split_text.push('-');
                    let next = *ids.get(&(w, m, false)).unwrap();
                    candidates.push((next, consumed + (m - start), true, pieces + 1, 0, split_text, occupied_width + (m - start) + 1));
                }

                if pieces == 0 && marker_fit.is_empty() {
                    let take = available;
                    if take > 0 && remaining > take {
                        let mut raw_text = text.clone();
                        raw_text.extend(word.chars[start..start + take].iter());
                        let next = *ids.get(&(w, start + take, false)).unwrap();
                        candidates.push((next, consumed + take, false, 1, 1, raw_text, occupied_width + take));
                    }
                }
                break;
            }

            w += 1;
            if w == words.len() {
                break;
            }
            start = 0;
        }

        let mut best = None;
        for (next, used, marker, piece_count, raw, body, displayed) in candidates {
            let suffix = dp[next].as_ref().unwrap();
            let final_line = suffix.lines.is_empty();
            let mut line_text = " ".repeat(indent);
            line_text.push_str(&body);
            let extra = line_cost(options.width, displayed + indent, final_line)
                + if marker { options.hyphen_cost as u64 } else { 0 }
                + if raw != 0 { (2 * options.width + 1) as u64 } else { 0 }
                + if !final_line && piece_count == 1 { 3 } else { 0 };
            let mut lines = Vec::with_capacity(suffix.lines.len() + 1);
            lines.push(Line {
                text: line_text,
                consumed: used,
                marker,
                pieces: piece_count,
                gaps: piece_count.saturating_sub(1),
                final_line,
            });
            lines.extend(suffix.lines.clone());
            better(&mut best, Plan {
                cost: suffix.cost + extra,
                marker_lines: suffix.marker_lines + usize::from(marker),
                lines,
            });
        }
        dp[id] = best;
    }
    dp[ids[&(0, 0, true)]].take().unwrap().lines
}

fn render_justified(lines: &mut [Line], width: usize, justify: bool) -> Vec<String> {
    let count = lines.len();
    lines.iter_mut().enumerate().map(|(i, line)| {
        if justify && !line.final_line && line.pieces >= 2 && line.gaps > 0 {
            let current = line.text.chars().count();
            let extra = width - current;
            let base = extra / line.gaps;
            let rem = extra % line.gaps;
            let start = i % line.gaps;
            let mut out = String::new();
            let mut gap_idx = 0usize;
            let chars: Vec<char> = line.text.chars().collect();
            for (j, &c) in chars.iter().enumerate() {
                if c == ' ' && j >= line.text.len().saturating_sub(line.text.len()) {
                    out.push(c);
                    let add = base + usize::from((gap_idx + line.gaps - start) % line.gaps < rem);
                    out.push_str(&" ".repeat(add));
                    gap_idx += 1;
                } else {
                    out.push(c);
                }
            }
            out
        } else {
            line.text.clone()
        }
    }).collect()
}

fn normalize(input: &str) -> Vec<char> {
    let mut out = Vec::new();
    let mut it = input.chars().peekable();
    while let Some(c) = it.next() {
        if c == '\r' {
            if it.peek() == Some(&'\n') {
                it.next();
            }
            out.push('\n');
        } else {
            out.push(c);
        }
    }
    out
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }
    for &(index, c) in &chars {
        if c.is_control() && c != '\t' && c != '\r' && c != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let ok = i > 0 && i + 1 < chars.len()
                && chars[i - 1].1 != '~' && chars[i + 1].1 != '~'
                && !matches!(chars[i - 1].1, ' ' | '\t' | '\r' | '\n' | '\u{00a0}')
                && !matches!(chars[i + 1].1, ' ' | '\t' | '\r' | '\n' | '\u{00a0}');
            if !ok {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let normalized = normalize(input);
    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            if j - i >= 2 {
                let words = parse_words(&segment);
                let mut lines = solve_paragraph(&words, options);
                result.extend(render_justified(&mut lines, options.width, options.justify));
                for _ in 1..(j - i) {
                    result.push(String::new());
                }
                segment.clear();
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }
    let words = parse_words(&segment);
    let mut lines = solve_paragraph(&words, options);
    result.extend(render_justified(&mut lines, options.width, options.justify));
    Ok(result)
}
```

### Solution B

```rust
use std::cmp::Ordering;
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    text: Vec<char>,
    breaks: Vec<usize>,
}

#[derive(Clone)]
struct Edge {
    next: (usize, usize, bool),
    body: String,
    consumed: usize,
    displayed: usize,
    marker: bool,
    raw: bool,
    pieces: usize,
}

#[derive(Clone)]
struct ResultPlan {
    cost: u64,
    markers: usize,
    lines: Vec<(String, usize, bool, usize)>,
}

fn compare(a: &ResultPlan, b: &ResultPlan) -> Ordering {
    a.cost.cmp(&b.cost)
        .then(a.markers.cmp(&b.markers))
        .then(a.lines.len().cmp(&b.lines.len()))
        .then_with(|| {
            for ((_, x, _, _), (_, y, _, _)) in a.lines.iter().zip(&b.lines) {
                if x != y {
                    return y.cmp(x);
                }
            }
            Ordering::Equal
        })
        .then_with(|| {
            for ((x, _, _, _), (y, _, _, _)) in a.lines.iter().zip(&b.lines) {
                if x != y {
                    return x.cmp(y);
                }
            }
            Ordering::Equal
        })
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse(chars: &[char]) -> Vec<Word> {
    let mut out = Vec::new();
    let mut text = Vec::new();
    let mut breaks = Vec::new();
    let mut pos = 0usize;
    for &c in chars {
        match c {
            '~' => breaks.push(pos),
            c if is_sep(c) => {
                if !text.is_empty() {
                    out.push(Word { text: std::mem::take(&mut text), breaks: std::mem::take(&mut breaks) });
                }
                pos = 0;
            }
            _ => {
                text.push(c);
                pos += 1;
            }
        }
    }
    if !text.is_empty() {
        out.push(Word { text, breaks });
    }
    out
}

fn options_for(
    words: &[Word],
    wi: usize,
    off: usize,
    first: bool,
    options: &WrapOptions,
) -> Vec<Edge> {
    let indent = if first { options.first_indent } else { options.rest_indent };
    let cap = options.width - indent;
    let mut edges = Vec::new();
    let mut body = String::new();
    let mut used = 0usize;
    let mut pieces = 0usize;
    let mut width = 0usize;
    let mut w = wi;
    let mut start = off;

    loop {
        if pieces > 0 {
            if width + 1 >= cap {
                break;
            }
            body.push(' ');
            width += 1;
        }
        let word = &words[w];
        let rem = word.text.len() - start;
        let room = cap - width;
        let fitting: Vec<usize> = word.breaks.iter().copied()
            .filter(|&b| b > start && b < word.text.len() && b - start + 1 <= room)
            .collect();

        if rem <= room {
            body.extend(word.text[start..].iter());
            width += rem;
            used += rem;
            pieces += 1;
            let next = if w + 1 == words.len() { (words.len(), 0, false) } else { (w + 1, 0, false) };
            edges.push(Edge {
                next, body: body.clone(), consumed: used, displayed: width + indent,
                marker: false, raw: false, pieces,
            });
        } else {
            for b in fitting {
                let mut split = body.clone();
                split.extend(word.text[start..b].iter());
                split.push('-');
                edges.push(Edge {
                    next: (w, b, false),
                    body: split,
                    consumed: used + b - start,
                    displayed: width + b - start + 1 + indent,
                    marker: true,
                    raw: false,
                    pieces: pieces + 1,
                });
            }
            if pieces == 0 && fitting.is_empty() && rem > room {
                let mut raw = body.clone();
                raw.extend(word.text[start..start + room].iter());
                edges.push(Edge {
                    next: (w, start + room, false),
                    body: raw,
                    consumed: room,
                    displayed: width + room + indent,
                    marker: false,
                    raw: true,
                    pieces: 1,
                });
            }
            break;
        }

        w += 1;
        if w == words.len() {
            break;
        }
        start = 0;
    }
    edges
}

fn render(words: &[Word], opts: &WrapOptions) -> Vec<String> {
    let mut memo: HashMap<(usize, usize, bool), ResultPlan> = HashMap::new();

    fn solve(
        state: (usize, usize, bool),
        words: &[Word],
        opts: &WrapOptions,
        memo: &mut HashMap<(usize, usize, bool), ResultPlan>,
    ) -> ResultPlan {
        if state.0 == words.len() {
            return ResultPlan { cost: 0, markers: 0, lines: Vec::new() };
        }
        if let Some(result) = memo.get(&state) {
            return result.clone();
        }
        let (wi, off, first) = state;
        let indent = if first { opts.first_indent } else { opts.rest_indent };
        let edges = options_for(words, wi, off, first, opts);
        let mut best: Option<ResultPlan> = None;

        for edge in edges {
            let suffix = solve(edge.next, words, opts, memo);
            let final_line = suffix.lines.is_empty();
            let gap = opts.width as u64 - edge.displayed as u64;
            let square = gap * gap;
            let mut cost = if final_line { square / 4 } else { square };
            if edge.marker { cost += opts.hyphen_cost as u64; }
            if edge.raw { cost += (2 * opts.width + 1) as u64; }
            if !final_line && edge.pieces == 1 { cost += 3; }
            cost += suffix.cost;

            let mut line = " ".repeat(indent);
            line.push_str(&edge.body);
            let mut lines = Vec::with_capacity(suffix.lines.len() + 1);
            lines.push((line, edge.consumed, edge.marker, edge.pieces));
            lines.extend(suffix.lines.clone());
            let candidate = ResultPlan { cost, markers: suffix.markers + usize::from(edge.marker), lines };
            if best.as_ref().map_or(true, |b| compare(&candidate, b) == Ordering::Less) {
                best = Some(candidate);
            }
        }
        let answer = best.unwrap();
        memo.insert(state, answer.clone());
        answer
    }

    let plan = solve((0, 0, true), words, opts, &mut memo);
    let n = plan.lines.len();
    plan.lines.into_iter().enumerate().map(|(i, (line, _, _, pieces))| {
        if !opts.justify || i + 1 == n || pieces < 2 {
            return line;
        }
        let gaps = pieces - 1;
        let current = line.chars().count();
        let extra = opts.width - current;
        let q = extra / gaps;
        let r = extra % gaps;
        let start = i % gaps;
        let mut out = String::new();
        let mut gap = 0usize;
        for c in line.chars() {
            out.push(c);
            if c == ' ' {
                let phase = (gap + gaps - start) % gaps;
                out.push_str(&" ".repeat(q + usize::from(phase < r)));
                gap += 1;
            }
        }
        out
    }).collect()
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 { return Err(WrapError::WidthZero); }
    if options.width > 256 { return Err(WrapError::WidthTooLarge { width: options.width }); }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    if original.len() > 512 {
        return Err(WrapError::InputTooLong { chars: original.len() });
    }
    for &(index, c) in &original {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..original.len() {
        if original[i].1 == '~' {
            let valid = i > 0 && i + 1 < original.len()
                && original[i - 1].1 != '~' && original[i + 1].1 != '~'
                && !is_sep(original[i - 1].1) && !is_sep(original[i + 1].1);
            if !valid {
                return Err(WrapError::InvalidMarker { index: original[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(c) = iter.next() {
        if c == '\r' {
            if iter.peek() == Some(&'\n') { iter.next(); }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i + 1;
            while j < normalized.len() && normalized[j] == '\n' { j += 1; }
            if j - i >= 2 {
                result.extend(render(&parse(&segment), options));
                result.extend((1..j - i).map(|_| String::new()));
                segment.clear();
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }
    result.extend(render(&parse(&segment), options));
    Ok(result)
}
```
