        return a.cost < b.cost;
    }
    if a.markers != b.markers {
        return a.markers < b.markers;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();

    for (x, y) in a.lines.iter().zip(&b.lines) {
        if x.consumed != y.consumed {

fn emit(
    base: &Plan,
    mut line: Line,
    next: (usize, usize, bool),
    best_state: &mut [Option<Plan>],
    states: &mut Vec<(usize, usize, bool)>,
    words: &[Word],
    opt: &WrapOptions,
) {
    let final_line = next.0 == words.len();
    let mut plan = base.clone();
    let square = (line.gap as u64) * (line.gap as u64);
    add_cost(&mut plan, if final_line { square / 4 } else { square });
    if line.marker_ended {
        plan.markers += 1;
        add_cost(&mut plan, opt.hyphen_cost as u64);
    }
    if line.raw {
        add_cost(&mut plan, (2 * opt.width + 1) as u64);
    }
    if !final_line && line.pieces.len() == 1 {
        add_cost(&mut plan, 3);
    }
    plan.lines.push(line.clone());
    let i = (next.0 * 520 + next.1) * 2 + usize::from(next.2);
    let replace = match &best_state[i] {
        None => true,
        Some(old) => better(&plan, old),
    };
    if replace {
        best_state[i] = Some(plan);
        states.push(next);
    }
    let _ = words;
}

fn is_sep(c: char) -> bool {
    c == ' ' || c == '\t' || c == '\n' || c == '\r' || c == '\u{00a0}'
}

fn parse_segment(s: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();
    for &c in s {
        if is_sep(c) {
            if !chars.is_empty() {
                words.push(Word { chars: std::mem::take(&mut chars), markers: std::mem::take(&mut markers) });
            }
        } else if c == '~' {
            markers.push(chars.len());
        } else {
            chars.push(c);
        }
    }
    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }
    words
}

fn render_lines(lines: &[Line], opt: &WrapOptions) -> Vec<String> {
    let mut out = Vec::new();
    for (line_index, line) in lines.iter().enumerate() {
        let mut parts = line.pieces.clone();
        let justify = opt.justify && line_index + 1 < lines.len() && parts.len() >= 2;
        let extra = if justify { line.gap } else { 0 };
        let gaps = parts.len().saturating_sub(1);
        let mut s = " ".repeat(line.indent);
        for i in 0..parts.len() {
            if i != 0 {
                let add = if justify {
                    let base = extra / gaps;
                    let rem = extra % gaps;
                    1 + base + usize::from((i - 1 + gaps - (line_index % gaps)) % gaps < rem)
                } else { 1 };
                s.push_str(&" ".repeat(add));
            }
            s.push_str(&parts[i]);
        }
        out.push(s);
    }
    out
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 { return Err(WrapError::WidthZero); }
    if options.width > 256 { return Err(WrapError::WidthTooLarge { width: options.width }); }
    if options.first_indent >= options.width { return Err(WrapError::FirstIndentTooWide { indent: options.first_indent }); }
    if options.rest_indent >= options.width { return Err(WrapError::RestIndentTooWide { indent: options.rest_indent }); }
    if options.hyphen_cost > 1_000_000 { return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost }); }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 { return Err(WrapError::InputTooLong { chars: chars.len() }); }
    for &(index, c) in &chars {
        if c.is_control() && c != '\t' && c != '\r' && c != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let prev_ok = i > 0 && chars[i - 1].1 != '~' && !is_sep(chars[i - 1].1);
            let next_ok = i + 1 < chars.len() && chars[i + 1].1 != '~' && !is_sep(chars[i + 1].1);
            if !prev_ok || !next_ok {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::<char>::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' { i += 1; }
        } else {
            normalized.push(c);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' { j += 1; }
            if j - i >= 2 {
                let words = parse_segment(&segment);
                if !words.is_empty() {
                    let lines = optimize(&words, options);
                    output.extend(render_lines(&lines, options));
                }
                segment.clear();
                output.extend(std::iter::repeat(String::new()).take(j - i - 1));
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }
    let words = parse_segment(&segment);
    if !words.is_empty() {
        let lines = optimize(&words, options);
        output.extend(render_lines(&lines, options));
    }
    Ok(output)
}

struct Candidate {
    lines: Vec<Line>,
    cost: u64,
    markers: usize,
    consumed: Vec<usize>,

fn rendered(line: &Line, justify: bool, width: usize, idx: usize) -> String {
    let gaps = line.pieces.len().saturating_sub(1);
    let (each, rem, start) = if justify && gaps > 0 {
        let extra = width - line.width;
        (extra / gaps, extra % gaps, idx % gaps)
    } else {
        (0, 0, 0)
    };
    let mut s = " ".repeat(line.indent);
    for (i, p) in line.pieces.iter().enumerate() {
        if i != 0 {
            s.push(' ');
            let g = i - 1;
            let dist = (g + gaps - start) % gaps;
            s.push_str(&" ".repeat(each + usize::from(gaps > 0 && dist < rem)));
        }
        s.push_str(p);
    }
    s
}

fn better(a: &Candidate, b: &Candidate) -> bool {
    (a.cost, a.markers, a.lines.len()) < (b.cost, b.markers, b.lines.len())
        || ((a.cost, a.markers, a.lines.len()) == (b.cost, b.markers, b.lines.len())
            && (a.consumed > b.consumed
                || (a.consumed == b.consumed && a.rendered < b.rendered)))
}

        words.iter().map(|w| vec![None; w.chars.len()]).collect();

        first: bool,
        opts: &WrapOptions,
        memo: &mut [Vec<Option<Candidate>>],

            return Some(Candidate {
                lines: vec![],
                cost: 0,
                markers: 0,
                consumed: vec![],
                rendered: vec![],
            });

            return Some(c.clone());
        }

                    tail.cost += cost;
                    tail.consumed.insert(0, rem);
                    tail.rendered.insert(0, rendered(&line, false, opts.width, 0));
                    update_best(&mut best, tail);
                }
            }

            for &m in &w.markers {
                if m > xo && m < w.chars.len() {
                    let n = m - xo;
                    if used + sep + n + 1 <= cap {
                        let mut ps = pieces.clone();
                        let mut p: String = w.chars[xo..m].iter().collect();
                        p.push('-');
                        ps.push(p);
                        let display = used + sep + n + 1;
                        let gap = (opts.width - indent - display) as u64;
                        let mut cost = gap * gap + opts.hyphen_cost as u64;
                        if ps.len() == 1 {
                            cost += 3;
                        }
                        let line = Line { indent, pieces: ps, width: indent + display, marker: true, penalty: cost };
                        if let Some(mut tail) = go(words, xw, m, false, opts, memo) {
                            tail.lines.insert(0, line.clone());
                            tail.cost += cost;
                            tail.markers += 1;
                            tail.consumed.insert(0, n);
                            tail.rendered.insert(0, rendered(&line, false, opts.width, 0));
                            update_best(&mut best, tail);
                        }
                    }
                }
            }

            if xw + 1 < words.len() && used + sep + rem + 1 <= cap {
                used += sep + rem;
                pieces.push(w.chars[xo..].iter().collect());
                xw += 1;
                xo = 0;
            } else {
                break;
            }
        }

        if pieces.is_empty() {
            let w = &words[wi];
            let rem = w.chars.len() - off;
            if rem > cap {
                let has_marker = w.markers.iter().any(|&m| m > off && m < w.chars.len() && m - off + 1 <= cap);
                if !has_marker {
                    let n = cap;
                    let piece: String = w.chars[off..off + n].iter().collect();
                    let gap = (opts.width - indent - n) as u64;
                    let cost = gap * gap + (2 * opts.width + 1) as u64;
                    let line = Line { indent, pieces: vec![piece], width: indent + n, marker: false, penalty: cost };
                    if let Some(mut tail) = go(words, wi, off + n, false, opts, memo) {
                        tail.lines.insert(0, line.clone());
                        tail.cost += cost;
                        tail.consumed.insert(0, n);
                        tail.rendered.insert(0, rendered(&line, false, opts.width, 0));
                        update_best(&mut best, tail);
                    }
                }
            }
        }

        memo[wi][off] = best.clone();
        best
    }

    fn update_best(best: &mut Option<Candidate>, candidate: Candidate) {
        if best.as_ref().map_or(true, |b| better(&candidate, b)) {
            *best = Some(candidate);
        }
    }

    let _ = states;
    let Some(plan) = go(words, 0, 0, true, opts, &mut memo) else {
        return Vec::new();
    };
    let mut out = Vec::new();
    for (i, line) in plan.lines.iter().enumerate() {
        let final_line = i + 1 == plan.lines.len();
        out.push(rendered(
            line,
            opts.justify && !final_line && line.pieces.len() >= 2 && !line.marker,
            opts.width,
            i,
        ));
    }
    out
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn parse(chars: &[char]) -> Vec<Word> {
    let mut words = vec![];
    let mut spelling = vec![];
    let mut marks = vec![];
    for &c in chars {
        match c {
            c if is_sep(c) => {
                if !spelling.is_empty() {
                    words.push(Word {
                        chars: std::mem::take(&mut spelling),
                        markers: std::mem::take(&mut marks),
                    });
                }
            }
            '~' => marks.push(spelling.len()),
            _ => spelling.push(c),
        }
    }
    if !spelling.is_empty() {
        words.push(Word { chars: spelling, markers: marks });
    }
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide { indent: options.first_indent });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide { indent: options.rest_indent });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost });
    }

    let indexed: Vec<(usize, char)> = input.char_indices().collect();
    if indexed.len() > 512 {
        return Err(WrapError::InputTooLong { chars: indexed.len() });
    }
    for &(index, c) in &indexed {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..indexed.len() {
        if indexed[i].1 == '~' {
            let valid = i > 0 && i + 1 < indexed.len()
                && !is_sep(indexed[i - 1].1) && indexed[i - 1].1 != '~'
                && !is_sep(indexed[i + 1].1) && indexed[i + 1].1 != '~';
            if !valid {
                return Err(WrapError::InvalidMarker { index: indexed[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < indexed.len() {
        if indexed[i].1 == '\r' {
            normalized.push('\n');
            if i + 1 < indexed.len() && indexed[i + 1].1 == '\n' {
                i += 2;
            } else {
                i += 1;
            }
        } else {
            normalized.push(indexed[i].1);
            i += 1;
        }
    }

    let mut output = vec![];
    let mut segment = vec![];
    let mut p = 0;
    while p < normalized.len() {
        if normalized[p] == '\n' {
            let mut q = p + 1;
            while q < normalized.len() && normalized[q] == '\n' {
                q += 1;
            }
            if q - p >= 2 {
                output.extend(optimize(&parse(&segment), options));
                segment.clear();
                for _ in 0..q - p - 1 {
                    output.push(String::new());
                }
                p = q;
                continue;
            }
        }
        segment.push(normalized[p]);
        p += 1;
    }
    output.extend(optimize(&parse(&segment), options));
    Ok(output)
}

struct Candidate {

    marker_count: usize,

fn line_text(l: &Line) -> String {
    let mut s = " ".repeat(l.indent);
    for (i, p) in l.pieces.iter().enumerate() {
        if i > 0 { s.push(' '); }
        s.push_str(p);
    }
    s
}

fn preferable(a: &Candidate, b: &Candidate) -> bool {
    if a.cost != b.cost { return a.cost < b.cost; }
    if a.marker_count != b.marker_count { return a.marker_count < b.marker_count; }
    if a.lines.len() != b.lines.len() { return a.lines.len() < b.lines.len(); }
    for (x, y) in a.lines.iter().zip(&b.lines) {
        if x.consumed != y.consumed { return x.consumed > y.consumed; }
    }
    for (x, y) in a.lines.iter().zip(&b.lines) {
        let sx = line_text(x);
        let sy = line_text(y);
        if sx != sy { return sx < sy; }
    }
    false
}

fn separators(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn parse_words(cs: &[char]) -> Vec<Word> {
    let mut out = Vec::new();
    let mut spelling = Vec::new();
    let mut marks = Vec::new();
    for &c in cs {
        match c {
            c if separators(c) => {
                if !spelling.is_empty() {
                    out.push(Word { chars: std::mem::take(&mut spelling), markers: std::mem::take(&mut marks) });
                }
            }
            '~' => marks.push(spelling.len()),
            _ => spelling.push(c),
        }
    }
    if !spelling.is_empty() { out.push(Word { chars: spelling, markers: marks }); }
    out
}

fn solve(words: &[Word], opt: &WrapOptions) -> Vec<Line> {
    use std::collections::HashMap;
    fn key(w: usize, o: usize, f: bool) -> (usize, usize, bool) { (w, o, f) }
    let mut memo: HashMap<(usize, usize, bool), Candidate> = HashMap::new();

    fn rec(
        w: usize, off: usize, first: bool, words: &[Word], opt: &WrapOptions,
        memo: &mut std::collections::HashMap<(usize, usize, bool), Candidate>,
    ) -> Candidate {
        if w == words.len() {
            return Candidate { cost: 0, marker_count: 0, lines: Vec::new() };
        }
        let k = key(w, off, first);
        if let Some(v) = memo.get(&k) { return v.clone(); }
        let indent = if first { opt.first_indent } else { opt.rest_indent };
        let cap = opt.width - indent;
        let word = &words[w];
        let rem = word.chars.len() - off;
        let mut choices: Vec<(Line, usize, usize)> = Vec::new();

        // marker endings in the current word
        for &m in &word.markers {
            if m > off && m < word.chars.len() && m - off + 1 <= cap {
                let mut p: String = word.chars[off..m].iter().collect();
                p.push('-');
                choices.push((Line {
                    indent, pieces: vec![p], consumed: m - off, marker: true, raw: false,
                    gap: cap - (m - off + 1),
                }, w, m));
            }
        }

        // Completed words, optionally several on a line.
        let mut display = 0usize;
        let mut pieces = Vec::new();
        let mut x = w;
        loop {
            if x >= words.len() { break; }
            let begin = if x == w { off } else { 0 };
            let len = words[x].chars.len() - begin;
            let needed = display + usize::from(!pieces.is_empty()) + len;
            if needed > cap { break; }
            display = needed;
            pieces.push(words[x].chars[begin..].iter().collect::<String>());
            choices.push((Line {
                indent, pieces: pieces.clone(), consumed: len, marker: false, raw: false,
                gap: cap - display,
            }, x + 1, 0));
            x += 1;
        }

        // Raw split only on an empty line and only when no marker can be used.
        if rem > cap && !word.markers.iter().any(|&m| m > off && m < word.chars.len() && m - off + 1 <= cap) {
            choices.push((Line {
                indent, pieces: vec![word.chars[off..off + cap].iter().collect()],
                consumed: cap, marker: false, raw: true, gap: 0,
            }, if off + cap == word.chars.len() { w + 1 } else { w }, if off + cap == word.chars.len() { 0 } else { off + cap }));
        }

        let mut best: Option<Candidate> = None;
        for (line, nw, no) in choices {
            let tail = rec(nw, no, false, words, opt, memo);
            let final_line = nw == words.len();
            let sq = (line.gap as u64) * (line.gap as u64);
            let mut cand = Candidate {
                cost: tail.cost + if final_line { sq / 4 } else { sq },
                marker_count: tail.marker_count + usize::from(line.marker),
                lines: std::iter::once(line.clone()).chain(tail.lines.into_iter()).collect(),
            };
            if line.marker { cand.cost += opt.hyphen_cost as u64; }
            if line.raw { cand.cost += (2 * opt.width + 1) as u64; }
            if !final_line && line.pieces.len() == 1 { cand.cost += 3; }
            if best.as_ref().map_or(true, |b| preferable(&cand, b)) { best = Some(cand); }
        }
        let answer = best.unwrap_or(Candidate { cost: 0, marker_count: 0, lines: Vec::new() });
        memo.insert(k, answer.clone());
        answer
    }

    rec(0, 0, true, words, opt, &mut memo).lines
}

fn render(lines: &[Line], opt: &WrapOptions) -> Vec<String> {
    let mut out = Vec::new();
    for (li, line) in lines.iter().enumerate() {
        let g = line.pieces.len().saturating_sub(1);
        let do_justify = opt.justify && li + 1 < lines.len() && g > 0;
        let mut s = " ".repeat(line.indent);
        for i in 0..line.pieces.len() {
            if i > 0 {
                let base = if do_justify { line.gap / g } else { 0 };
                let rem = if do_justify { line.gap % g } else { 0 };
                let start = if g == 0 { 0 } else { li % g };
                let rank = (i - 1 + g - start) % g;
                s.push_str(&" ".repeat(1 + base + usize::from(do_justify && rank < rem)));
            }
            s.push_str(&line.pieces[i]);
        }
        out.push(s);
    }
    out
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 { return Err(WrapError::WidthZero); }
    if options.width > 256 { return Err(WrapError::WidthTooLarge { width: options.width }); }
    if options.first_indent >= options.width { return Err(WrapError::FirstIndentTooWide { indent: options.first_indent }); }
    if options.rest_indent >= options.width { return Err(WrapError::RestIndentTooWide { indent: options.rest_indent }); }
    if options.hyphen_cost > 1_000_000 { return Err(WrapError::HyphenCostTooLarge { cost: options.hyphen_cost }); }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    if original.len() > 512 { return Err(WrapError::InputTooLong { chars: original.len() }); }
    for &(index, c) in &original {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }
    for i in 0..original.len() {
        if original[i].1 == '~' {
            let left = i > 0 && original[i - 1].1 != '~' && !separators(original[i - 1].1);
            let right = i + 1 < original.len() && original[i + 1].1 != '~' && !separators(original[i + 1].1);
            if !left || !right { return Err(WrapError::InvalidMarker { index: original[i].0 }); }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < original.len() {
        if original[i].1 == '\r' {
            normalized.push('\n');
            if i + 1 < original.len() && original[i + 1].1 == '\n' { i += 1; }
        } else {
            normalized.push(original[i].1);
        }
        i += 1;
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    i = 0;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut j = i;
            while j < normalized.len() && normalized[j] == '\n' { j += 1; }
            if j - i >= 2 {
                let words = parse_words(&segment);
                if !words.is_empty() { result.extend(render(&solve(&words, options), options)); }
                segment.clear();
                result.extend(std::iter::repeat(String::new()).take(j - i - 1));
                i = j;
                continue;
            }
        }
        segment.push(normalized[i]);
        i += 1;
    }
    let words = parse_words(&segment);
    if !words.is_empty() { result.extend(render(&solve(&words, options), options)); }
    Ok(result)
}

    WidthZero,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {

    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    used: usize,
    consumed: usize,
    marker: bool,
    raw: bool,
    cost: u64,
}

#[derive(Clone)]
struct Plan {
    lines: Vec<Line>,
    cost: u64,
    marker_count: usize,
    consumed: Vec<usize>,
    strings: Vec<String>,
}

fn plan_better(a: &Plan, b: &Plan) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_count != b.marker_count {
        return a.marker_count < b.marker_count;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed != b.consumed {
        return a.consumed > b.consumed;
    }
    a.strings < b.strings
}

fn push_piece(line: &mut Line, piece: String, width: usize) {
    if !line.pieces.is_empty() {
        line.used += 1;
    }
    line.used += piece.chars().count();
    line.pieces.push(piece);
    let _ = width;
}

fn line_text(line: &Line, justify: bool, width: usize, line_index: usize) -> String {
    let gaps = line.pieces.len().saturating_sub(1);
    let extra = if justify && !line.marker && !line.raw && gaps > 0 {
        let e = width.saturating_sub(line.used);
        Some((e / gaps, e % gaps, line_index % gaps))
    } else {
        None
    };
    let mut out = " ".repeat(line.indent);
    for (i, piece) in line.pieces.iter().enumerate() {
        if i > 0 {
            out.push(' ');
            if let Some((each, rem, start)) = extra {
                let gap = i - 1;
                let distance = (gap + gaps - start) % gaps;
                out.push_str(&" ".repeat(each + usize::from(distance < rem)));
            }
        }
        out.push_str(piece);
    }
    out
}

fn solve(
    words: &[Word],
    wi: usize,
    off: usize,
    first: bool,
    opts: &WrapOptions,
    memo: &mut Vec<Vec<Vec<Option<Plan>>>>,
) -> Option<Plan> {
    if wi == words.len() {
        return Some(Plan {
            lines: Vec::new(),
            cost: 0,
            marker_count: 0,
            consumed: Vec::new(),
            strings: Vec::new(),
        });
    }
    let fi = usize::from(first);
    if let Some(p) = &memo[wi][off][fi] {
        return Some(p.clone());
    }

    let indent = if first { opts.first_indent } else { opts.rest_indent };
    let cap = opts.width - indent;
    let mut best: Option<Plan> = None;
    let mut pieces: Vec<String> = Vec::new();
    let mut used = 0usize;
    let mut cur_wi = wi;
    let mut cur_off = off;
    let mut piece_count = 0usize;

    loop {
        let word = &words[cur_wi];
        let rem = word.chars.len() - cur_off;
        let separator = usize::from(!pieces.is_empty());

        // Complete this word if its remaining suffix fits.
        if used + separator + rem <= cap {
            let mut next_pieces = pieces.clone();
            next_pieces.push(word.chars[cur_off..].iter().collect());
            let displayed = used + separator + rem;
            let consumed = rem;
            let gap = opts.width - (indent + displayed);
            let final_line = cur_wi + 1 == words.len();
            let mut line_cost = (gap as u64) * (gap as u64);
            if final_line {
                line_cost /= 4;
            }
            if !final_line && next_pieces.len() == 1 {
                line_cost += 3;
            }
            let line = Line {
                indent,
                pieces: next_pieces,
                used: displayed,
                consumed,
                marker: false,
                raw: false,
                cost: line_cost,
            };
            if let Some(mut tail) = solve(words, cur_wi + 1, 0, false, opts, memo) {
                tail.lines.insert(0, line.clone());
                tail.cost += line_cost;
                tail.consumed.insert(0, consumed);
                tail.strings.insert(0, line_text(&line, false, opts.width, 0));
                let count = line.marker as usize;
                tail.marker_count += count;
                if best.as_ref().map_or(true, |b| plan_better(&tail, b)) {
                    best = Some(tail);
                }
            }
        }

        // Marker-ended split within the current word.
        for &m in &word.markers {
            if m > cur_off && m < word.chars.len() {
                let prefix = m - cur_off;
                if used + separator + prefix + 1 <= cap {
                    let mut next_pieces = pieces.clone();
                    let mut text: String = word.chars[cur_off..m].iter().collect();
                    text.push('-');
                    next_pieces.push(text);
                    let displayed = used + separator + prefix + 1;
                    let gap = opts.width - (indent + displayed);
                    let mut line_cost = (gap as u64) * (gap as u64) + opts.hyphen_cost as u64;
                    if next_pieces.len() == 1 {
                        line_cost += 3;
                    }
                    let line = Line {
                        indent,
                        pieces: next_pieces,
                        used: displayed,
                        consumed: prefix,
                        marker: true,
                        raw: false,
                        cost: line_cost,
                    };
                    if let Some(mut tail) = solve(words, cur_wi, m, false, opts, memo) {
                        tail.lines.insert(0, line.clone());
                        tail.cost += line_cost;
                        tail.marker_count += 1;
                        tail.consumed.insert(0, prefix);
                        tail.strings.insert(0, line_text(&line, false, opts.width, 0));
                        if best.as_ref().map_or(true, |b| plan_better(&tail, b)) {
                            best = Some(tail);
                        }
                    }
                }
            }
        }

        // Extend the line by completing this word and trying the next word.
        if cur_wi + 1 < words.len() && used + separator + rem + 1 <= cap {
            used += separator + rem;
            pieces.push(word.chars[cur_off..].iter().collect());
            piece_count += 1;
            cur_wi += 1;
            cur_off = 0;
        } else {
            break;
        }
    }

    // Raw split is only for an otherwise empty line, with no fitting marker.
    if pieces.is_empty() {
        let word = &words[wi];
        let rem = word.chars.len() - off;
        if rem > cap {
            let marker_fits = word.markers.iter().any(|&m| {
                m > off && m < word.chars.len() && (m - off) + 1 <= cap
            });
            if !marker_fits {
                let n = cap;
                let next_off = off + n;
                let piece: String = word.chars[off..next_off].iter().collect();
                let line = Line {
                    indent,
                    pieces: vec![piece],
                    used: n,
                    consumed: n,
                    marker: false,
                    raw: true,
                    cost: (opts.width - (indent + n)) as u64
                        * (opts.width - (indent + n)) as u64
                        + (2 * opts.width + 1) as u64,
                };
                if let Some(mut tail) = solve(words, wi, next_off, false, opts, memo) {
                    tail.lines.insert(0, line.clone());
                    tail.cost += line.cost;
                    tail.consumed.insert(0, n);
                    tail.strings.insert(0, line_text(&line, false, opts.width, 0));
                    if best.as_ref().map_or(true, |b| plan_better(&tail, b)) {
                        best = Some(tail);
                    }
                }
            }
        }
    }

    memo[wi][off][fi] = best.clone();
    best
}

fn layout_paragraph(words: &[Word], opts: &WrapOptions) -> Vec<String> {
    if words.is_empty() {
        return Vec::new();
    }
    let max_len = words.iter().map(|w| w.chars.len()).max().unwrap_or(0);
    let mut memo = words
        .iter()
        .map(|w| {
            (0..=w.chars.len())
                .map(|_| vec![None, None])
                .collect::<Vec<_>>()
        })
        .collect::<Vec<_>>();
    let _ = max_len;
    let Some(plan) = solve(words, 0, 0, true, opts, &mut memo) else {
        return Vec::new();
    };

    let mut out = Vec::with_capacity(plan.lines.len());
    for (i, line) in plan.lines.iter().enumerate() {
        let final_line = i + 1 == plan.lines.len();
        let justify = opts.justify && !final_line && line.pieces.len() >= 2;
        out.push(line_text(line, justify, opts.width, i));
    }
    out
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00A0}')
}

fn parse_segment(chars: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Vec::new();
    let mut markers = Vec::new();

    let flush = |words: &mut Vec<Word>, current: &mut Vec<char>, markers: &mut Vec<usize>| {
        if !current.is_empty() {
            words.push(Word {
                chars: std::mem::take(current),
                markers: std::mem::take(markers),
            });
        }
    };

    for &c in chars {
        if is_sep(c) {
            flush(&mut words, &mut current, &mut markers);
        } else if c == '~' {
            markers.push(current.len());
        } else {
            current.push(c);
        }
    }
    flush(&mut words, &mut current, &mut markers);
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(index, c) in &chars {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let prev_ok = i > 0 && !is_sep(chars[i - 1].1) && chars[i - 1].1 != '~';
            let next_ok = i + 1 < chars.len()
                && !is_sep(chars[i + 1].1)
                && chars[i + 1].1 != '~';
            if !prev_ok || !next_ok {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let c = chars[i].1;
        if c == '\r' {
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                normalized.push('\n');
                i += 2;
            } else {
                normalized.push('\n');
                i += 1;
            }
        } else {
            normalized.push(c);
            i += 1;
        }
    }

    if normalized.is_empty() {
        return Ok(Vec::new());
    }

    let mut result = Vec::new();
    let mut segment = Vec::new();
    let mut p = 0;
    while p < normalized.len() {
        if normalized[p] == '\n' {
            let mut q = p;
            while q < normalized.len() && normalized[q] == '\n' {
                q += 1;
            }
            if q - p >= 2 {
                let words = parse_segment(&segment);
                result.extend(layout_paragraph(&words, options));
                segment.clear();
                for _ in 1..(q - p) {
                    result.push(String::new());
                }
                p = q;
                continue;
            }
        }
        segment.push(normalized[p]);
        p += 1;
    }
    let words = parse_segment(&segment);
    result.extend(layout_paragraph(&words, options));
    Ok(result)
}
