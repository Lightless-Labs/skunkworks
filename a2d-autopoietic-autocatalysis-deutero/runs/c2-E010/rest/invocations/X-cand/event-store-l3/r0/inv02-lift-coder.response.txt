- Error precedence follows the numbered rules exactly; `VersionOverflow` is checked before any event-specific validation.
- ID length is checked after scanning earlier bytes, so an invalid byte before byte 24 takes precedence.
- Empty appends still validate ID and expectation, but never create a stream.
- Snapshots are taken at every positive interval boundary crossed by a successful batch; the latest boundary snapshot is retained.
- A supplied snapshot may be structurally valid without being historically reachable or interval-aligned.
- Both solutions use the same required semantics; Solution B uses a separate replay helper that starts from an account.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Status { New, Open, Closed }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Account {
    pub status: Status,
    pub balance: i64,
    pub limit: i64,
    pub version: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { limit: i64 },
    Deposit { amount: i64 },
    Withdraw { amount: i64, floor: i64 },
    Close { expected_version: u64 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    VersionOverflow, InvalidLimit, AlreadyOpened, NotOpened, Closed,
    InvalidAmount, LimitExceeded, InvalidFloor, InsufficientFunds,
    FloorViolation, CloseVersion { expected: u64, actual: u64 }, NonZeroBalance
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ReplayError { pub index: usize, pub kind: ApplyError }

fn initial_account() -> Account {
    Account { status: Status::New, balance: 0, limit: 0, version: 0 }
}

fn apply_event(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    if account.version == u64::MAX {
        return Err(ApplyError::VersionOverflow);
    }

    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.status = Status::Open;
            account.limit = *limit;
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.status = Status::Closed;
        }
    }

    account.version += 1;
    Ok(())
}

fn apply_events_from(mut account: Account, events: &[Event]) -> Result<Account, ReplayError> {
    for (index, event) in events.iter().enumerate() {
        apply_event(&mut account, event)
            .map_err(|kind| ReplayError { index, kind })?;
    }
    Ok(account)
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    apply_events_from(initial_account(), events)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Expected { Any, Missing, Exact(u64) }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StoreError {
    InvalidId { byte: usize },
    Conflict { expected: Expected, actual: Option<u64> },
    Rejected(ReplayError)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Snapshot { pub account: Account, pub tag: u64 }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SnapshotError { ZeroInterval, InvalidVersion, InvalidState, TagMismatch }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RestoreError { Snapshot(SnapshotError), Tail(ReplayError) }

#[derive(Debug, Clone, PartialEq, Eq)]
struct Stream {
    account: Account,
    events: Vec<Event>,
    snapshot: Option<Snapshot>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    snapshot_every: u64,
    streams: BTreeMap<String, Stream>,
}

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 {
            return Err(SnapshotError::ZeroInterval);
        }
        Ok(Self { snapshot_every, streams: BTreeMap::new() })
    }

    pub fn append(&mut self, id: &str, expected: Expected, events: &[Event]) -> Result<Account, StoreError> {
        validate_id(id)?;
        let current = self.streams.get(id);
        let actual = current.map(|s| s.account.version);
        let matches = match expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(v) => actual == Some(v),
        };
        if !matches {
            return Err(StoreError::Conflict { expected, actual });
        }

        if events.is_empty() {
            return Ok(current.map(|s| s.account.clone()).unwrap_or_else(initial_account));
        }

        let mut account = current.map(|s| s.account.clone()).unwrap_or_else(initial_account);
        let old_version = account.version;
        let mut new_snapshot = current.and_then(|s| s.snapshot.clone());
        let mut batch_snapshot = None;

        for (index, event) in events.iter().enumerate() {
            apply_event(&mut account, event)
                .map_err(|kind| StoreError::Rejected(ReplayError { index, kind }))?;
            if account.version > 0 && account.version % self.snapshot_every == 0 {
                batch_snapshot = Some(Snapshot {
                    tag: snapshot_tag(&account),
                    account: account.clone(),
                });
            }
        }

        if batch_snapshot.is_some() {
            new_snapshot = batch_snapshot;
        } else if old_version == 0 {
            // Keep the absent snapshot absent until the first boundary.
            new_snapshot = None;
        }

        let mut history = current.map(|s| s.events.clone()).unwrap_or_default();
        history.extend_from_slice(events);
        self.streams.insert(id.to_owned(), Stream {
            account: account.clone(),
            events: history,
            snapshot: new_snapshot,
        });
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|s| s.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|s| {
            s.events.iter().enumerate()
                .filter(|(i, _)| (*i as u64) + 1 > after)
                .map(|(_, e)| e.clone())
                .collect()
        }))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).and_then(|s| s.snapshot.clone()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|s| {
            let start = s.snapshot.as_ref().map(|x| x.account.version).unwrap_or(0) as usize;
            s.events.iter().skip(start).cloned().collect()
        }))
    }
}

fn validate_id(id: &str) -> Result<(), StoreError> {
    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Err(StoreError::InvalidId { byte: 0 });
    }
    for (i, &b) in bytes.iter().enumerate() {
        if i == 24 {
            return Err(StoreError::InvalidId { byte: 24 });
        }
        let valid = if i == 0 {
            b.is_ascii_lowercase()
        } else {
            b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-'
        };
        if !valid {
            return Err(StoreError::InvalidId { byte: i });
        }
    }
    if bytes.len() > 24 {
        return Err(StoreError::InvalidId { byte: 24 });
    }
    Ok(())
}

pub fn snapshot_tag(account: &Account) -> u64 {
    let status_code = match account.status {
        Status::New => 0,
        Status::Open => 0xA5,
        Status::Closed => 0x5A,
    };
    account.version.rotate_left(7)
        ^ (account.limit as u64).wrapping_mul(31)
        ^ (account.balance as u64).rotate_left(17)
        ^ status_code
}

fn validate_snapshot(snapshot: &Snapshot) -> Result<(), SnapshotError> {
    let a = &snapshot.account;
    if a.version == 0 {
        return Err(SnapshotError::InvalidVersion);
    }
    if a.status == Status::New
        || !(1..=1_000_000_000).contains(&a.limit)
        || a.balance < 0
        || a.balance > a.limit
        || (a.status == Status::Closed && (a.balance != 0 || a.version < 2))
    {
        return Err(SnapshotError::InvalidState);
    }
    if snapshot.tag != snapshot_tag(a) {
        return Err(SnapshotError::TagMismatch);
    }
    Ok(())
}

pub fn rebuild(snapshot: Option<&Snapshot>, tail: &[Event]) -> Result<Account, RestoreError> {
    match snapshot {
        None => replay(tail).map_err(RestoreError::Tail),
        Some(s) => {
            validate_snapshot(s).map_err(RestoreError::Snapshot)?;
            apply_events_from(s.account.clone(), tail).map_err(RestoreError::Tail)
        }
    }
}
```

```rust
    }

    match event {

                return Err(ApplyError::AlreadyOpened);

            if account.status == Status::Closed { return Err(ApplyError::Closed); }
            if *amount <= 0 { return Err(ApplyError::InvalidAmount); }
            if *amount > account.limit - account.balance { return Err(ApplyError::LimitExceeded); }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            if account.status == Status::New { return Err(ApplyError::NotOpened); }
            if account.status == Status::Closed { return Err(ApplyError::Closed); }
            if *amount <= 0 { return Err(ApplyError::InvalidAmount); }
            if *floor < 0 || *floor > account.limit { return Err(ApplyError::InvalidFloor); }
            if *amount > account.balance { return Err(ApplyError::InsufficientFunds); }
            if account.balance - *amount < *floor { return Err(ApplyError::FloorViolation); }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            if account.status == Status::New { return Err(ApplyError::NotOpened); }
            if account.status == Status::Closed { return Err(ApplyError::Closed); }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion { expected: *expected_version, actual: account.version });
            }
            if account.balance != 0 { return Err(ApplyError::NonZeroBalance); }
            account.status = Status::Closed;
        }
    }

    account.version += 1;
    Ok(())
}

fn apply_from(mut account: Account, events: &[Event]) -> Result<Account, ReplayError> {
    for (index, event) in events.iter().enumerate() {
        if let Err(kind) = apply_event(&mut account, event) {
            return Err(ReplayError { index, kind });
        }
    }
    Ok(account)
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    apply_from(initial_account(), events)
}

}

    snapshot_every: u64,

}

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 { return Err(SnapshotError::ZeroInterval); }
        Ok(Self { snapshot_every, streams: BTreeMap::new() })
    }

    pub fn append(&mut self, id: &str, expected: Expected, events: &[Event]) -> Result<Account, StoreError> {
        validate_id(id)?;
        let existing = self.streams.get(id);
        let actual = existing.map(|stream| stream.account.version);
        let matched = match expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(version) => actual == Some(version),
        };
        if !matched {
            return Err(StoreError::Conflict { expected, actual });
        }

        let mut account = existing.map(|stream| stream.account.clone()).unwrap_or_else(initial_account);
        if events.is_empty() {
            return Ok(account);
        }

        let mut snapshot = existing.and_then(|stream| stream.snapshot.clone());
        let mut history = existing.map(|stream| stream.events.clone()).unwrap_or_default();

        for (index, event) in events.iter().enumerate() {
            apply_event(&mut account, event)
                .map_err(|kind| StoreError::Rejected(ReplayError { index, kind }))?;
            history.push(event.clone());

            if account.version % self.snapshot_every == 0 {
                snapshot = Some(Snapshot {
                    account: account.clone(),
                    tag: snapshot_tag(&account),
                });
            }
        }

        self.streams.insert(id.to_owned(), Stream {
            account: account.clone(),
            events: history,
            snapshot,
        });
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {

        Ok(self.streams.get(id).map(|stream| stream.account.clone()))

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {

                .filter(|(index, _)| (*index as u64) + 1 > after)
                .map(|(_, event)| event.clone())
                .collect()

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|stream| {
            let start = stream.snapshot.as_ref()
                .map(|snapshot| snapshot.account.version as usize)
                .unwrap_or(0);
            stream.events.iter().skip(start).cloned().collect()
        }))
    }
}
```
