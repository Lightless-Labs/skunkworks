- Event validation follows the stated order; version overflow is checked before event-specific checks when rebuilding from a snapshot.
- Snapshot version zero is invalid before structural checks; a valid snapshot is accepted without requiring interval alignment or a provable event history.
- IDs are scanned byte by byte; an invalid byte before the 24-byte limit takes precedence over the length error.
- Empty appends still check ID and expected version, but never create a stream.
- Snapshots capture every reached positive interval boundary, including intermediate states within a batch.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Status { New, Open, Closed }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Account {
    pub status: Status,
    pub balance: i64,
    pub limit: i64,
    pub version: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { limit: i64 },
    Deposit { amount: i64 },
    Withdraw { amount: i64, floor: i64 },
    Close { expected_version: u64 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    VersionOverflow, InvalidLimit, AlreadyOpened, NotOpened, Closed,
    InvalidAmount, LimitExceeded, InvalidFloor, InsufficientFunds,
    FloorViolation, CloseVersion { expected: u64, actual: u64 }, NonZeroBalance
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ReplayError {
    pub index: usize,
    pub kind: ApplyError,
}

fn initial_account() -> Account {
    Account { status: Status::New, balance: 0, limit: 0, version: 0 }
}

fn apply_one(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    if account.version == u64::MAX {
        return Err(ApplyError::VersionOverflow);
    }

    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.status = Status::Open;
            account.limit = *limit;
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.status = Status::Closed;
        }
    }

    account.version += 1;
    Ok(())
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    let mut account = initial_account();
    for (index, event) in events.iter().enumerate() {
        if let Err(kind) = apply_one(&mut account, event) {
            return Err(ReplayError { index, kind });
        }
    }
    Ok(account)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Expected { Any, Missing, Exact(u64) }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StoreError {
    InvalidId { byte: usize },
    Conflict { expected: Expected, actual: Option<u64> },
    Rejected(ReplayError),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Snapshot {
    pub account: Account,
    pub tag: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SnapshotError { ZeroInterval, InvalidVersion, InvalidState, TagMismatch }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RestoreError { Snapshot(SnapshotError), Tail(ReplayError) }

#[derive(Debug, Clone, PartialEq, Eq)]
struct Stream {
    account: Account,
    events: Vec<Event>,
    snapshot: Option<Snapshot>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    snapshot_every: u64,
    streams: BTreeMap<String, Stream>,
}

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 {
            return Err(SnapshotError::ZeroInterval);
        }
        Ok(Self { snapshot_every, streams: BTreeMap::new() })
    }

    pub fn append(&mut self, id: &str, expected: Expected, events: &[Event]) -> Result<Account, StoreError> {
        validate_id(id)?;
        let actual = self.streams.get(id).map(|s| s.account.version);
        let matches = match expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(v) => actual == Some(v),
        };
        if !matches {
            return Err(StoreError::Conflict { expected, actual });
        }

        let Some(existing) = self.streams.get(id) else {
            if events.is_empty() {
                return Ok(initial_account());
            }
            let mut account = initial_account();
            let mut new_events = Vec::with_capacity(events.len());
            let mut snap = None;
            for (index, event) in events.iter().enumerate() {
                if let Err(kind) = apply_one(&mut account, event) {
                    return Err(StoreError::Rejected(ReplayError { index, kind }));
                }
                new_events.push(event.clone());
                if account.version % self.snapshot_every == 0 {
                    snap = Some(make_snapshot(&account));
                }
            }
            let result = account.clone();
            self.streams.insert(id.to_owned(), Stream { account, events: new_events, snapshot: snap });
            return Ok(result);
        };

        if events.is_empty() {
            return Ok(existing.account.clone());
        }

        let mut account = existing.account.clone();
        let mut new_events = Vec::with_capacity(events.len());
        let mut snap = existing.snapshot.clone();
        for (index, event) in events.iter().enumerate() {
            if let Err(kind) = apply_one(&mut account, event) {
                return Err(StoreError::Rejected(ReplayError { index, kind }));
            }
            new_events.push(event.clone());
            if account.version % self.snapshot_every == 0 {
                snap = Some(make_snapshot(&account));
            }
        }

        let result = account.clone();
        let stream = self.streams.get_mut(id).expect("stream was checked");
        stream.account = account;
        stream.events.extend(new_events);
        stream.snapshot = snap;
        Ok(result)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|s| s.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|s| {
            s.events.iter().enumerate()
                .filter(|(i, _)| (i as u128) + 1 > after as u128)
                .map(|(_, event)| event.clone())
                .collect()
        }))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).and_then(|s| s.snapshot.clone()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|s| {
            let start = s.snapshot.as_ref().map(|snap| snap.account.version as usize).unwrap_or(0);
            s.events.iter().skip(start).cloned().collect()
        }))
    }
}

fn validate_id(id: &str) -> Result<(), StoreError> {
    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Err(StoreError::InvalidId { byte: 0 });
    }
    for (i, &b) in bytes.iter().enumerate() {
        let valid = if i == 0 { b.is_ascii_lowercase() } else {
            b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-'
        };
        if !valid {
            return Err(StoreError::InvalidId { byte: i });
        }
    }
    if bytes.len() > 24 {
        return Err(StoreError::InvalidId { byte: 24 });
    }
    Ok(())
}

fn make_snapshot(account: &Account) -> Snapshot {
    Snapshot { account: account.clone(), tag: snapshot_tag(account) }
}

pub fn snapshot_tag(account: &Account) -> u64 {
    let code = match account.status {
        Status::New => 0,
        Status::Open => 0xA5,
        Status::Closed => 0x5A,
    };
    account.version.rotate_left(7)
        ^ (account.limit as u64).wrapping_mul(31)
        ^ (account.balance as u64).rotate_left(17)
        ^ code
}

fn validate_snapshot(snapshot: &Snapshot) -> Result<(), SnapshotError> {
    let a = &snapshot.account;
    if a.version == 0 {
        return Err(SnapshotError::InvalidVersion);
    }
    if a.status == Status::New
        || !(1..=1_000_000_000).contains(&a.limit)
        || a.balance < 0
        || a.balance > a.limit
        || (a.status == Status::Closed && (a.balance != 0 || a.version < 2))
    {
        return Err(SnapshotError::InvalidState);
    }
    if snapshot.tag != snapshot_tag(a) {
        return Err(SnapshotError::TagMismatch);
    }
    Ok(())
}

pub fn rebuild(snapshot: Option<&Snapshot>, tail: &[Event]) -> Result<Account, RestoreError> {
    let mut account = if let Some(snapshot) = snapshot {
        validate_snapshot(snapshot).map_err(RestoreError::Snapshot)?;
        snapshot.account.clone()
    } else {
        initial_account()
    };

    for (index, event) in tail.iter().enumerate() {
        if let Err(kind) = apply_one(&mut account, event) {
            return Err(RestoreError::Tail(ReplayError { index, kind }));
        }
    }
    Ok(account)
}
```

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Status { New, Open, Closed }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Account {
    pub status: Status,
    pub balance: i64,
    pub limit: i64,
    pub version: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { limit: i64 },
    Deposit { amount: i64 },
    Withdraw { amount: i64, floor: i64 },
    Close { expected_version: u64 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    VersionOverflow, InvalidLimit, AlreadyOpened, NotOpened, Closed,
    InvalidAmount, LimitExceeded, InvalidFloor, InsufficientFunds,
    FloorViolation, CloseVersion { expected: u64, actual: u64 }, NonZeroBalance
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ReplayError { pub index: usize, pub kind: ApplyError }

fn initial_account() -> Account {
    Account { status: Status::New, balance: 0, limit: 0, version: 0 }
}

fn apply_one(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    if account.version == u64::MAX {
        return Err(ApplyError::VersionOverflow);
    }
    match event {
        Event::Open { limit } => {
            if *limit < 1 || *limit > 1_000_000_000 { return Err(ApplyError::InvalidLimit); }
            if account.status != Status::New { return Err(ApplyError::AlreadyOpened); }
            account.status = Status::Open;
            account.limit = *limit;
        }
        Event::Deposit { amount } => {
            if account.status == Status::New { return Err(ApplyError::NotOpened); }
            if account.status == Status::Closed { return Err(ApplyError::Closed); }
            if *amount <= 0 { return Err(ApplyError::InvalidAmount); }
            if *amount > account.limit - account.balance { return Err(ApplyError::LimitExceeded); }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            if account.status == Status::New { return Err(ApplyError::NotOpened); }
            if account.status == Status::Closed { return Err(ApplyError::Closed); }
            if *amount <= 0 { return Err(ApplyError::InvalidAmount); }
            if *floor < 0 || *floor > account.limit { return Err(ApplyError::InvalidFloor); }
            if *amount > account.balance { return Err(ApplyError::InsufficientFunds); }
            if account.balance - *amount < *floor { return Err(ApplyError::FloorViolation); }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            if account.status == Status::New { return Err(ApplyError::NotOpened); }
            if account.status == Status::Closed { return Err(ApplyError::Closed); }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion { expected: *expected_version, actual: account.version });
            }
            if account.balance != 0 { return Err(ApplyError::NonZeroBalance); }
            account.status = Status::Closed;
        }
    }
    account.version = account.version.checked_add(1).ok_or(ApplyError::VersionOverflow)?;
    Ok(())
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    let mut account = initial_account();
    for (index, event) in events.iter().enumerate() {
        apply_one(&mut account, event).map_err(|kind| ReplayError { index, kind })?;
    }
    Ok(account)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Expected { Any, Missing, Exact(u64) }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StoreError {
    InvalidId { byte: usize },
    Conflict { expected: Expected, actual: Option<u64> },
    Rejected(ReplayError)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Snapshot { pub account: Account, pub tag: u64 }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SnapshotError { ZeroInterval, InvalidVersion, InvalidState, TagMismatch }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RestoreError { Snapshot(SnapshotError), Tail(ReplayError) }

#[derive(Debug, Clone, PartialEq, Eq)]
struct Stream {
    events: Vec<Event>,
    snapshot: Option<Snapshot>,
    account: Account,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    interval: u64,
    streams: BTreeMap<String, Stream>,
}

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 { return Err(SnapshotError::ZeroInterval); }
        Ok(Self { interval: snapshot_every, streams: BTreeMap::new() })
    }

    pub fn append(&mut self, id: &str, expected: Expected, events: &[Event]) -> Result<Account, StoreError> {
        validate_id(id)?;
        let actual = self.streams.get(id).map(|stream| stream.account.version);
        let ok = match expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(v) => actual == Some(v),
        };
        if !ok {
            return Err(StoreError::Conflict { expected, actual });
        }

        let base = self.streams.get(id).map(|stream| stream.account.clone()).unwrap_or_else(initial_account);
        if events.is_empty() { return Ok(base); }

        let mut account = base;
        let mut latest = self.streams.get(id).and_then(|stream| stream.snapshot.clone());
        for (index, event) in events.iter().enumerate() {
            apply_one(&mut account, event)
                .map_err(|kind| StoreError::Rejected(ReplayError { index, kind }))?;
            if account.version % self.interval == 0 {
                latest = Some(Snapshot { account: account.clone(), tag: snapshot_tag(&account) });
            }
        }

        let mut full_events = self.streams.get(id)
            .map(|stream| stream.events.clone())
            .unwrap_or_default();
        full_events.extend_from_slice(events);

        self.streams.insert(id.to_owned(), Stream {
            account: account.clone(),
            events: full_events,
            snapshot: latest,
        });
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|stream| stream.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|stream| {
            stream.events.iter().enumerate()
                .filter(|(index, _)| (*index as u128) + 1 > after as u128)
                .map(|(_, event)| event.clone())
                .collect()
        }))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).and_then(|stream| stream.snapshot.clone()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|stream| {
            let version = stream.snapshot.as_ref().map(|snap| snap.account.version);
            match version {
                None => stream.events.clone(),
                Some(v) => stream.events.iter().skip(usize::try_from(v).unwrap_or(usize::MAX)).cloned().collect(),
            }
        }))
    }
}

fn validate_id(id: &str) -> Result<(), StoreError> {
    let bytes = id.as_bytes();
    if bytes.is_empty() { return Err(StoreError::InvalidId { byte: 0 }); }
    for (i, b) in bytes.iter().copied().enumerate() {
        let valid = if i == 0 { b.is_ascii_lowercase() } else {
            b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-'
        };
        if !valid { return Err(StoreError::InvalidId { byte: i }); }
    }
    if bytes.len() > 24 { return Err(StoreError::InvalidId { byte: 24 }); }
    Ok(())
}

pub fn snapshot_tag(account: &Account) -> u64 {
    let status = match account.status {
        Status::New => 0,
        Status::Open => 0xA5,
        Status::Closed => 0x5A,
    };
    account.version.rotate_left(7)
        ^ (account.limit as u64).wrapping_mul(31)
        ^ (account.balance as u64).rotate_left(17)
        ^ status
}

fn validate_snapshot(snapshot: &Snapshot) -> Result<(), SnapshotError> {
    let a = &snapshot.account;
    if a.version == 0 { return Err(SnapshotError::InvalidVersion); }
    if a.status == Status::New
        || a.limit < 1 || a.limit > 1_000_000_000
        || a.balance < 0 || a.balance > a.limit
        || (a.status == Status::Closed && (a.balance != 0 || a.version < 2))

        return Err(SnapshotError::InvalidState);
    }
    if snapshot.tag != snapshot_tag(a) { return Err(SnapshotError::TagMismatch); }

pub fn rebuild(snapshot: Option<&Snapshot>, tail: &[Event]) -> Result<Account, RestoreError> {
    let mut account = match snapshot {
        Some(s) => {
            validate_snapshot(s).map_err(RestoreError::Snapshot)?;
            s.account.clone()
        }
        None => initial_account(),
    };
    for (index, event) in tail.iter().enumerate() {
        apply_one(&mut account, event)
            .map_err(|kind| RestoreError::Tail(ReplayError { index, kind }))?;
    }
    Ok(account)
}
```
