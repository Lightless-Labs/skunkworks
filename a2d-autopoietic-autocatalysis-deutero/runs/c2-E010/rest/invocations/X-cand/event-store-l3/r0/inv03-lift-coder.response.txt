- Lifecycle checks precede amount checks for deposits and withdrawals; Open checks its limit first.
- Invalid IDs report the first bad byte, with the 24-byte length violation reported at index 24 unless an earlier byte is invalid.
- Conflict checks happen before event validation, and empty successful appends never create streams.
- Snapshots are captured at every positive interval boundary crossed by a successful batch; the latest such boundary is retained.
- Snapshot validation checks version, structure, then tag; rebuilding validates a supplied snapshot before examining the tail.
- Version overflow is checked before any event-specific rule, and a version may advance from `u64::MAX - 1` to `u64::MAX` once.

### Solution A

```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Status { New, Open, Closed }

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Account {
    pub status: Status,
    pub balance: i64,
    pub limit: i64,
    pub version: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { limit: i64 },
    Deposit { amount: i64 },
    Withdraw { amount: i64, floor: i64 },
    Close { expected_version: u64 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    VersionOverflow, InvalidLimit, AlreadyOpened, NotOpened, Closed,
    InvalidAmount, LimitExceeded, InvalidFloor, InsufficientFunds,
    FloorViolation, CloseVersion { expected: u64, actual: u64 }, NonZeroBalance
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ReplayError {
    pub index: usize,
    pub kind: ApplyError,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Expected { Any, Missing, Exact(u64) }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StoreError {
    InvalidId { byte: usize },
    Conflict { expected: Expected, actual: Option<u64> },
    Rejected(ReplayError),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Snapshot {
    pub account: Account,
    pub tag: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SnapshotError { ZeroInterval, InvalidVersion, InvalidState, TagMismatch }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RestoreError { Snapshot(SnapshotError), Tail(ReplayError) }

#[derive(Debug, Clone, PartialEq, Eq)]
struct Stream {
    events: Vec<Event>,
    account: Account,
    snapshot: Option<Snapshot>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    snapshot_every: u64,
    streams: BTreeMap<String, Stream>,
}

fn initial_account() -> Account {
    Account { status: Status::New, balance: 0, limit: 0, version: 0 }
}

fn apply_one(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    if account.version == u64::MAX {
        return Err(ApplyError::VersionOverflow);
    }

    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.status = Status::Open;
            account.limit = *limit;
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.status = Status::Closed;
        }
    }

    account.version += 1;
    Ok(())
}

fn replay_from(account: &Account, events: &[Event]) -> Result<Account, ReplayError> {
    let mut result = account.clone();
    for (index, event) in events.iter().enumerate() {
        if let Err(kind) = apply_one(&mut result, event) {
            return Err(ReplayError { index, kind });
        }
    }
    Ok(result)
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    replay_from(&initial_account(), events)
}

fn validate_id(id: &str) -> Result<(), StoreError> {
    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Err(StoreError::InvalidId { byte: 0 });
    }
    for (i, &b) in bytes.iter().enumerate().take(24) {
        let valid = if i == 0 {
            b.is_ascii_lowercase()
        } else {
            b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-'
        };
        if !valid {
            return Err(StoreError::InvalidId { byte: i });
        }
    }
    if bytes.len() > 24 {
        return Err(StoreError::InvalidId { byte: 24 });
    }
    Ok(())
}

fn matches_expected(expected: &Expected, actual: Option<u64>) -> bool {
    match expected {
        Expected::Any => true,
        Expected::Missing => actual.is_none(),
        Expected::Exact(v) => actual == Some(*v),
    }
}

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 {
            return Err(SnapshotError::ZeroInterval);
        }
        Ok(Self { snapshot_every, streams: BTreeMap::new() })
    }

    pub fn append(&mut self, id: &str, expected: Expected, events: &[Event]) -> Result<Account, StoreError> {
        validate_id(id)?;
        let actual = self.streams.get(id).map(|s| s.account.version);
        if !matches_expected(&expected, actual) {
            return Err(StoreError::Conflict { expected, actual });
        }

        let base = self.streams.get(id).map(|s| s.account.clone()).unwrap_or_else(initial_account);
        if events.is_empty() {
            return Ok(base);
        }

        let mut account = base.clone();
        let mut newest_snapshot = self.streams.get(id).and_then(|s| s.snapshot.clone());
        let old_version = base.version;
        for (index, event) in events.iter().enumerate() {
            if let Err(kind) = apply_one(&mut account, event) {
                return Err(StoreError::Rejected(ReplayError { index, kind }));
            }
            if account.version % self.snapshot_every == 0 {
                newest_snapshot = Some(Snapshot {
                    tag: snapshot_tag(&account),
                    account: account.clone(),
                });
            }
        }

        let stream = self.streams.entry(id.to_owned()).or_insert_with(|| Stream {
            events: Vec::new(),
            account: initial_account(),
            snapshot: None,
        });
        stream.events.extend_from_slice(events);
        stream.account = account.clone();
        stream.snapshot = newest_snapshot;

        let _ = old_version;
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).map(|s| s.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        let Some(stream) = self.streams.get(id) else { return Ok(None) };
        let mut result = Vec::new();
        for (i, event) in stream.events.iter().enumerate() {
            let version = (i as u64) + 1;
            if version > after {
                result.push(event.clone());
            }
        }
        Ok(Some(result))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        validate_id(id)?;
        Ok(self.streams.get(id).and_then(|s| s.snapshot.clone()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id)?;
        let Some(stream) = self.streams.get(id) else { return Ok(None) };
        let start = stream.snapshot.as_ref().map(|s| s.account.version as usize).unwrap_or(0);
        Ok(Some(stream.events.iter().skip(start).cloned().collect()))
    }
}

pub fn snapshot_tag(account: &Account) -> u64 {
    let status_code = match account.status {
        Status::New => 0,
        Status::Open => 0xA5,
        Status::Closed => 0x5A,
    };
    account.version.rotate_left(7)
        ^ (account.limit as u64).wrapping_mul(31)
        ^ (account.balance as u64).rotate_left(17)
        ^ status_code
}

fn validate_snapshot(snapshot: &Snapshot) -> Result<(), SnapshotError> {
    let a = &snapshot.account;
    if a.version == 0 {
        return Err(SnapshotError::InvalidVersion);
    }
    if a.status == Status::New
        || !(1..=1_000_000_000).contains(&a.limit)
        || a.balance < 0
        || a.balance > a.limit
        || (a.status == Status::Closed && (a.balance != 0 || a.version < 2))
    {
        return Err(SnapshotError::InvalidState);
    }
    if snapshot.tag != snapshot_tag(a) {
        return Err(SnapshotError::TagMismatch);
    }
    Ok(())
}

pub fn rebuild(snapshot: Option<&Snapshot>, tail: &[Event]) -> Result<Account, RestoreError> {
    let base = match snapshot {
        None => initial_account(),
        Some(s) => {
            validate_snapshot(s).map_err(RestoreError::Snapshot)?;
            s.account.clone()
        }
    };
    replay_from(&base, tail).map_err(RestoreError::Tail)
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]

pub enum Expected { Any, Missing, Exact(u64) }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SnapshotError { ZeroInterval, InvalidVersion, InvalidState, TagMismatch }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum RestoreError { Snapshot(SnapshotError), Tail(ReplayError) }

#[derive(Debug, Clone, PartialEq, Eq)]
struct Stream {
    history: Vec<Event>,
    current: Account,
    checkpoints: Vec<Snapshot>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Store {
    interval: u64,
    streams: BTreeMap<String, Stream>,
}

fn new_account() -> Account {
    Account { status: Status::New, balance: 0, limit: 0, version: 0 }
}

fn apply(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    if account.version.checked_add(1).is_none() {
        return Err(ApplyError::VersionOverflow);
    }
    match event {
        Event::Open { limit } => {
            if *limit < 1 || *limit > 1_000_000_000 {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.status = Status::Open;
            account.limit = *limit;
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => (),
            }
            if *amount <= 0 { return Err(ApplyError::InvalidAmount); }
            let remaining = account.limit - account.balance;
            if *amount > remaining { return Err(ApplyError::LimitExceeded); }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => (),
            }
            if *amount <= 0 { return Err(ApplyError::InvalidAmount); }
            if *floor < 0 || *floor > account.limit { return Err(ApplyError::InvalidFloor); }
            if *amount > account.balance { return Err(ApplyError::InsufficientFunds); }
            if account.balance - *amount < *floor { return Err(ApplyError::FloorViolation); }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => (),
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion { expected: *expected_version, actual: account.version });
            }
            if account.balance != 0 { return Err(ApplyError::NonZeroBalance); }
            account.status = Status::Closed;
        }
    }
    account.version += 1;
    Ok(())
}

fn replay_from(start: &Account, events: &[Event]) -> Result<Account, ReplayError> {
    let mut account = start.clone();
    for (index, event) in events.iter().enumerate() {
        if let Err(kind) = apply(&mut account, event) {
            return Err(ReplayError { index, kind });
        }
    }
    Ok(account)
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    replay_from(&new_account(), events)
}

fn check_id(id: &str) -> Result<(), StoreError> {
    let bytes = id.as_bytes();
    if bytes.len() == 0 { return Err(StoreError::InvalidId { byte: 0 }); }
    for (index, b) in bytes.iter().enumerate() {
        if index == 24 { return Err(StoreError::InvalidId { byte: 24 }); }
        let okay = if index == 0 {
            b.is_ascii_lowercase()
        } else {
            b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-'
        };
        if !okay { return Err(StoreError::InvalidId { byte: index }); }
    }
    Ok(())
}

fn expected_matches(expected: &Expected, actual: Option<u64>) -> bool {
    match expected {
        Expected::Any => true,
        Expected::Missing => actual.is_none(),
        Expected::Exact(v) => actual == Some(*v),
    }
}

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 { return Err(SnapshotError::ZeroInterval); }
        Ok(Self { interval: snapshot_every, streams: BTreeMap::new() })
    }

    pub fn append(&mut self, id: &str, expected: Expected, events: &[Event]) -> Result<Account, StoreError> {
        check_id(id)?;
        let actual = self.streams.get(id).map(|s| s.current.version);
        if !expected_matches(&expected, actual) {
            return Err(StoreError::Conflict { expected, actual });
        }
        let base = self.streams.get(id).map(|s| s.current.clone()).unwrap_or_else(new_account);
        if events.len() == 0 { return Ok(base); }

        let mut current = base;
        let mut new_checkpoints = self.streams.get(id).map(|s| s.checkpoints.clone()).unwrap_or_default();
        for (index, event) in events.iter().enumerate() {
            apply(&mut current, event)
                .map_err(|kind| StoreError::Rejected(ReplayError { index, kind }))?;
            if current.version % self.interval == 0 {
                new_checkpoints.push(Snapshot { tag: snapshot_tag(&current), account: current.clone() });
            }
        }
        let stream = self.streams.entry(id.to_owned()).or_insert_with(|| Stream {
            history: Vec::new(),
            current: new_account(),
            checkpoints: Vec::new(),
        });
        stream.history.extend(events.iter().cloned());
        stream.current = current.clone();
        stream.checkpoints = new_checkpoints;
        Ok(current)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        check_id(id)?;
        Ok(self.streams.get(id).map(|s| s.current.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        check_id(id)?;
        let Some(s) = self.streams.get(id) else { return Ok(None); };
        Ok(Some(s.history.iter().enumerate()
            .filter(|(i, _)| (*i as u64) + 1 > after)
            .map(|(_, e)| e.clone())
            .collect()))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        check_id(id)?;
        Ok(self.streams.get(id).and_then(|s| s.checkpoints.last().cloned()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        check_id(id)?;
        let Some(s) = self.streams.get(id) else { return Ok(None); };
        let start = s.checkpoints.last().map(|x| x.account.version as usize).unwrap_or(0);
        Ok(Some(s.history.iter().skip(start).cloned().collect()))
    }
}

pub fn snapshot_tag(account: &Account) -> u64 {
    let code = match account.status { Status::New => 0, Status::Open => 0xA5, Status::Closed => 0x5A };
    account.version.rotate_left(7)
        ^ (account.limit as u64).wrapping_mul(31)
        ^ (account.balance as u64).rotate_left(17)
        ^ code
}

fn validate(snapshot: &Snapshot) -> Result<(), SnapshotError> {
    let a = &snapshot.account;
    if a.version == 0 { return Err(SnapshotError::InvalidVersion); }
    if matches!(a.status, Status::New)
        || a.limit < 1 || a.limit > 1_000_000_000
        || a.balance < 0 || a.balance > a.limit
        || (a.status == Status::Closed && (a.balance != 0 || a.version < 2))
    {
        return Err(SnapshotError::InvalidState);
    }
    if snapshot.tag != snapshot_tag(a) { return Err(SnapshotError::TagMismatch); }
    Ok(())
}

pub fn rebuild(snapshot: Option<&Snapshot>, tail: &[Event]) -> Result<Account, RestoreError> {
    match snapshot {
        None => replay(tail).map_err(RestoreError::Tail),
        Some(s) => {
            validate(s).map_err(RestoreError::Snapshot)?;
            replay_from(&s.account, tail).map_err(RestoreError::Tail)
        }
    }
}
```
