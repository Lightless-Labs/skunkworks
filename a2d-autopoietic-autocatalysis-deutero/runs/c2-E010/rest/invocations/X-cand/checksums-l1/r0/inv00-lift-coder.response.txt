### Solution A

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = 0xffff_ffffu32;

    for &byte in data {
        crc ^= u32::from(byte);
        for _ in 0..8 {
            let old_low_bit = crc & 1;
            crc >>= 1;
            if old_low_bit != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }

    crc ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    const MOD: u32 = 65_521;
    let mut a = 1u32;
    let mut b = 0u32;

    for &byte in data {
        a = (a + u32::from(byte)) % MOD;
        b = (b + a) % MOD;
    }

    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    const ADLER_MOD: u32 = 65_521;
    let mut crc = 0xffff_ffffu32;
    let mut a = 1u32;
    let mut b = 0u32;
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }

        for &byte in &data[span.start..span.end] {
            crc ^= u32::from(byte);
            for _ in 0..8 {
                let old_low_bit = crc & 1;
                crc >>= 1;
                if old_low_bit != 0 {
                    crc ^= 0xedb8_8320;
                }
            }

            a = (a + u32::from(byte)) % ADLER_MOD;
            b = (b + a) % ADLER_MOD;
        }

        previous_end = span.end;
    }

    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}
```

### Solution B

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

struct Accumulator {
    crc: u32,
    a: u32,
    b: u32,
}

impl Accumulator {
    fn new() -> Self {
        Self {
            crc: 0xffff_ffff,
            a: 1,
            b: 0,
        }
    }

    fn update(&mut self, byte: u8) {
        self.crc ^= u32::from(byte);
        for _ in 0..8 {
            self.crc = (self.crc >> 1)
                ^ if self.crc & 1 != 0 {
                    0xedb8_8320
                } else {
                    0
                };
        }

        const MOD: u32 = 65_521;
        self.a = (self.a + u32::from(byte)) % MOD;
        self.b = (self.b + self.a) % MOD;
    }

    fn finish(self) -> Hashes {
        Hashes {
            crc32: self.crc ^ 0xffff_ffff,
            adler32: (self.b << 16) | self.a,
        }
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut acc = Accumulator::new();
    for &byte in data {
        acc.update(byte);
    }
    acc.finish().crc32
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut acc = Accumulator::new();
    for &byte in data {
        acc.update(byte);
    }
    acc.finish().adler32
}

pub fn checksums(data: &[u8]) -> Hashes {
    let mut acc = Accumulator::new();
    for &byte in data {
        acc.update(byte);
    }
    acc.finish()
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    let mut acc = Accumulator::new();
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }

        for &byte in &data[span.start..span.end] {
            acc.update(byte);
        }

        previous_end = span.end;
    }

    Ok(acc.finish())
}
```
