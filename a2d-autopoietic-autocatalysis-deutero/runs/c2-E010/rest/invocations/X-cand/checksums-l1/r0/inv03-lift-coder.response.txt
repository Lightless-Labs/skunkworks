- Check each span in supplied order; within it, reversed takes precedence over bounds, and bounds precede overlap.
- Treat `end == data.len()` as valid; spans are half-open.
- Empty spans contribute no bytes but still update the ordering boundary.
- Hash selected bytes as one concatenated stream, retaining checksum state across spans.
- Use bounded integer arithmetic for Adler-32 so long inputs cannot overflow.
- Since no alternate interpretation fits the contract, Solution B uses an independent checksum implementation.

### Solution A

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

const ADLER_MOD: u32 = 65_521;

fn update_crc32(mut crc: u32, data: &[u8]) -> u32 {
    for &byte in data {
        crc ^= u32::from(byte);
        for _ in 0..8 {
            let old_low_bit = crc & 1;
            crc >>= 1;
            if old_low_bit != 0 {
                crc ^= 0xedb8_8320;
            }
        }
    }
    crc
}

fn update_adler32(mut a: u32, mut b: u32, data: &[u8]) -> (u32, u32) {
    for &byte in data {
        a = (a + u32::from(byte)) % ADLER_MOD;
        b = (b + a) % ADLER_MOD;
    }
    (a, b)
}

pub fn crc32(data: &[u8]) -> u32 {
    update_crc32(0xffff_ffff, data) ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let (a, b) = update_adler32(1, 0, data);
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    let mut crc = 0xffff_ffff;
    let mut a = 1;
    let mut b = 0;
    let mut previous_end = 0;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }

        let selected = &data[span.start..span.end];
        crc = update_crc32(crc, selected);
        (a, b) = update_adler32(a, b, selected);
        previous_end = span.end;
    }

    Ok(Hashes {
        crc32: crc ^ 0xffff_ffff,
        adler32: (b << 16) | a,
    })
}
```

### Solution B

```rust
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Span {
    pub start: usize,
    pub end: usize,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Hashes {
    pub crc32: u32,
    pub adler32: u32,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SpanError {
    Reversed { span: usize },
    OutOfBounds { span: usize, offset: usize },
    Overlap { span: usize },
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut crc = !0u32;
    for &byte in data {
        crc = crc ^ u32::from(byte);
        let mut bit = 0;
        while bit < 8 {
            crc = if crc & 1 == 1 {
                (crc >> 1) ^ 0xedb8_8320
            } else {
                crc >> 1
            };
            bit += 1;
        }
    }
    !crc
}

pub fn adler32(data: &[u8]) -> u32 {
    const MOD: u64 = 65_521;
    let mut a = 1u64;
    let mut b = 0u64;
    for &byte in data {
        a = (a + u64::from(byte)) % MOD;
        b = (b + a) % MOD;
    }
    ((b as u32) << 16) | a as u32
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    let mut crc = !0u32;
    let mut a = 1u32;
    let mut b = 0u32;
    let mut previous_end = 0usize;

    for (index, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: index });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.start,
            });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds {
                span: index,
                offset: span.end,
            });
        }
        if index != 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: index });
        }

        for &byte in &data[span.start..span.end] {
            crc ^= u32::from(byte);
            for _ in 0..8 {
                crc = (crc >> 1) ^ (0xedb8_8320 & (0u32.wrapping_sub(crc & 1)));
            }

            a += u32::from(byte);
            if a >= 65_521 {
                a -= 65_521;
            }
            b = (b + a) % 65_521;
        }

        previous_end = span.end;
    }

    Ok(Hashes {
        crc32: !crc,
        adler32: (b << 16) | a,
    })
}
```
