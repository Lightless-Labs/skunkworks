- NUL is preflighted across the full input, so it wins even over an earlier syntax error.
- Braced expansion first searches for an unescaped `}`; missing closure wins over invalid contents.
- A trailing backslash in double quotes wins at EOF over the still-open quote.
- Only ASCII space, tab, CR, and LF split words; other Unicode whitespace stays literal.
- Wildcard offsets are UTF-8 byte offsets in the completed word; expansion wildcards inherit the quote context.
- Solution B uses the same interpretations with a separate scanner implementation.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn first_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter().find_map(|(k, v)| if *k == name { Some(*v) } else { None })
}

fn append_literal(word: &mut Word, c: char, active: bool) {
    let at = word.text.len();
    word.text.push(c);
    if active && matches!(c, '*' | '?' | '[') {
        word.glob.push(at);
    }
}

fn append_data(word: &mut Word, text: &str, active: bool) {
    for c in text.chars() {
        append_literal(word, c, active);
    }
}

fn decode_default(s: &str) -> String {
    let mut out = String::new();
    let mut it = s.char_indices().peekable();
    while let Some((_, c)) = it.next() {
        if c == '\\' {
            if let Some((_, next)) = it.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn expand_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let bytes = input.as_bytes();
    let mut i = dollar + 2;
    let mut close = None;
    while i < bytes.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < bytes.len() {
                let n = input[i..].chars().next().unwrap();
                i += n.len_utf8();
            }
        } else if c == '}' {
            close = Some(i);
            break;
        } else {
            i += c.len_utf8();
        }
    }
    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let content = &input[dollar + 2..close];

    let mut chars = content.char_indices();
    let first = chars.next().map(|(_, c)| c).ok_or(SplitError::InvalidExpansion { pos: dollar })?;
    if !is_name_start(first) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    let mut name_end = first.len_utf8();
    for (idx, c) in chars {
        if is_name_continue(c) {
            name_end = idx + c.len_utf8();
        } else {
            break;
        }
    }
    let name = &content[..name_end];
    let rest = &content[name_end..];
    let value = if rest.is_empty() {
        first_env(env, name).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match first_env(env, name) {
            Some(v) if !v.is_empty() => v.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    Ok((close + 1, value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut word = Word { text: String::new(), glob: Vec::new() };
    let mut active_word = false;
    let mut quote = Quote::None;
    let mut quote_start = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                    i = next_i;
                } else {
                    append_literal(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    let slash = i;
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let d = input[next_i..].chars().next().unwrap();
                    let after = next_i + d.len_utf8();
                    if matches!(d, '"' | '\\' | '$' | '\n') {
                        if d != '\n' {
                            append_literal(&mut word, d, false);
                        }
                        i = after;
                    } else {
                        append_literal(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if next_i < input.len() {
                        let d = input[next_i..].chars().next().unwrap();
                        if d == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                    }
                    if next_i < input.len() && input.as_bytes()[next_i] == b'{' {
                        active_word = true;
                        let (end, value) = expand_braced(input, i, env)?;
                        append_data(&mut word, &value, false);
                        i = end;
                    } else if next_i < input.len() {
                        let d = input[next_i..].chars().next().unwrap();
                        if is_name_start(d) {
                            let mut end = next_i + d.len_utf8();
                            while end < input.len() {
                                let x = input[end..].chars().next().unwrap();
                                if is_name_continue(x) {
                                    end += x.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            active_word = true;
                            let name = &input[next_i..end];
                            let value = first_env(env, name).unwrap_or("");
                            append_data(&mut word, value, false);
                            i = end;
                        } else {
                            append_literal(&mut word, '$', false);
                            i = next_i;
                        }
                    } else {
                        append_literal(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    append_literal(&mut word, c, false);
                    i = next_i;
                }
            }
            Quote::None => {
                if is_sep(c) {
                    if active_word {
                        result.push(word);
                        word = Word { text: String::new(), glob: Vec::new() };
                        active_word = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active_word = true;
                    quote = Quote::Single;
                    quote_start = i;
                    i = next_i;
                } else if c == '"' {
                    active_word = true;
                    quote = Quote::Double;
                    quote_start = i;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let d = input[next_i..].chars().next().unwrap();
                    let after = next_i + d.len_utf8();
                    if d == '\n' {
                        i = after;
                    } else {
                        active_word = true;
                        append_literal(&mut word, d, false);
                        i = after;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if next_i < input.len() {
                        let d = input[next_i..].chars().next().unwrap();
                        if d == '(' {
                            return Err(SplitError::CommandSubstitution { pos: i });
                        }
                    }
                    if next_i < input.len() && input.as_bytes()[next_i] == b'{' {
                        active_word = true;
                        let (end, value) = expand_braced(input, i, env)?;
                        append_data(&mut word, &value, true);
                        i = end;
                    } else if next_i < input.len() {
                        let d = input[next_i..].chars().next().unwrap();
                        if is_name_start(d) {
                            let mut end = next_i + d.len_utf8();
                            while end < input.len() {
                                let x = input[end..].chars().next().unwrap();
                                if is_name_continue(x) {
                                    end += x.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            active_word = true;
                            let name = &input[next_i..end];
                            let value = first_env(env, name).unwrap_or("");
                            append_data(&mut word, value, true);
                            i = end;
                        } else {
                            active_word = true;
                            append_literal(&mut word, '$', true);
                            i = next_i;
                        }
                    } else {
                        active_word = true;
                        append_literal(&mut word, '$', true);
                        i = next_i;
                    }
                } else {
                    active_word = true;
                    append_literal(&mut word, c, true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),
        Quote::None => {}
    }
    if active_word {
        result.push(word);
    }
    Ok(result)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum State {
    Bare,
    Single(usize),
    Double(usize),
}

fn separator(c: char) -> bool {
    c == ' ' || c == '\t' || c == '\r' || c == '\n'
}

fn start_name(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn name_char(c: char) -> bool {
    start_name(c) || c.is_ascii_digit()
}

fn append(word: &mut Word, c: char, glob: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if glob && (c == '*' || c == '?' || c == '[') {
        word.glob.push(offset);
    }
}

fn append_string(word: &mut Word, s: &str, glob: bool) {
    for c in s.chars() {
        append(word, c, glob);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> &'a str {
    env.iter().find(|entry| entry.0 == name).map_or("", |entry| entry.1)
}

fn braced(input: &str, start: usize, env: &[(&str, &str)]) -> Result<(usize, String), SplitError> {
    let mut p = start + 2;
    let mut closing = None;
    while p < input.len() {
        let ch = input[p..].chars().next().unwrap();
        if ch == '\\' {
            p += 1;
            if p < input.len() {
                let escaped = input[p..].chars().next().unwrap();
                p += escaped.len_utf8();
            }
        } else if ch == '}' {
            closing = Some(p);
            break;
        } else {
            p += ch.len_utf8();
        }
    }
    let closing = closing.ok_or(SplitError::UnterminatedExpansion { pos: start })?;
    let body = &input[start + 2..closing];
    let mut iter = body.char_indices();
    let (first_at, first) = iter.next().ok_or(SplitError::InvalidExpansion { pos: start })?;
    if first_at != 0 || !start_name(first) {
        return Err(SplitError::InvalidExpansion { pos: start });
    }
    let mut end_name = first.len_utf8();
    for (at, ch) in iter {
        if name_char(ch) {
            end_name = at + ch.len_utf8();
        } else {
            break;
        }
    }
    let name = &body[..end_name];
    let suffix = &body[end_name..];
    if suffix.is_empty() {
        Ok((closing + 1, lookup(env, name).to_owned()))
    } else if let Some(default) = suffix.strip_prefix(":-") {
        let existing = env.iter().find(|entry| entry.0 == name).map(|entry| entry.1);
        let text = match existing {
            Some(value) if !value.is_empty() => value.to_owned(),
            _ => {
                let mut decoded = String::new();
                let mut chars = default.chars();
                while let Some(ch) = chars.next() {
                    if ch == '\\' {
                        if let Some(next) = chars.next() {
                            decoded.push(next);
                        }
                    } else {
                        decoded.push(ch);
                    }
                }
                decoded
            }
        };
        Ok((closing + 1, text))
    } else {
        Err(SplitError::InvalidExpansion { pos: start })
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(nul) = input.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos: nul });
    }

    let mut output = Vec::new();
    let mut current = Word { text: String::new(), glob: Vec::new() };
    let mut in_word = false;
    let mut state = State::Bare;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let after = cursor + ch.len_utf8();

        match state {
            State::Single(open) => {
                if ch == '\'' {
                    state = State::Bare;
                } else {
                    append(&mut current, ch, false);
                }
                let _ = open;
                cursor = after;
            }
            State::Double(open) => {
                if ch == '"' {
                    state = State::Bare;
                    cursor = after;
                } else if ch == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[after..].chars().next().unwrap();
                    if escaped == '"' || escaped == '\\' || escaped == '$' || escaped == '\n' {
                        if escaped != '\n' {
                            append(&mut current, escaped, false);
                        }
                        cursor = after + escaped.len_utf8();
                    } else {
                        append(&mut current, '\\', false);
                        cursor = after;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if after < input.len() && input[after..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if after < input.len() && input.as_bytes()[after] == b'{' {
                        let (end, value) = braced(input, cursor, env)?;
                        append_string(&mut current, &value, false);
                        cursor = end;
                    } else if after < input.len() {
                        let next = input[after..].chars().next().unwrap();
                        if start_name(next) {
                            let mut end = after + next.len_utf8();
                            while end < input.len() {
                                let c = input[end..].chars().next().unwrap();
                                if name_char(c) {
                                    end += c.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            append_string(&mut current, lookup(env, &input[after..end]), false);
                            cursor = end;
                        } else {
                            append(&mut current, '$', false);
                            cursor = after;
                        }
                    } else {
                        append(&mut current, '$', false);
                        cursor = after;
                    }
                    in_word = true;
                } else {
                    append(&mut current, ch, false);
                    cursor = after;
                }
                let _ = open;
            }
            State::Bare => {
                if separator(ch) {
                    if in_word {
                        output.push(current);
                        current = Word { text: String::new(), glob: Vec::new() };
                        in_word = false;
                    }
                    cursor = after;
                } else if ch == '\'' {
                    state = State::Single(cursor);
                    in_word = true;
                    cursor = after;
                } else if ch == '"' {
                    state = State::Double(cursor);
                    in_word = true;
                    cursor = after;
                } else if ch == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[after..].chars().next().unwrap();
                    cursor = after + escaped.len_utf8();
                    if escaped != '\n' {
                        append(&mut current, escaped, false);
                        in_word = true;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if after < input.len() && input[after..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if after < input.len() && input.as_bytes()[after] == b'{' {
                        let (end, value) = braced(input, cursor, env)?;
                        append_string(&mut current, &value, true);
                        cursor = end;
                    } else if after < input.len() {
                        let next = input[after..].chars().next().unwrap();
                        if start_name(next) {
                            let mut end = after + next.len_utf8();
                            while end < input.len() {
                                let c = input[end..].chars().next().unwrap();
                                if name_char(c) {
                                    end += c.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            append_string(&mut current, lookup(env, &input[after..end]), true);
                            cursor = end;
                        } else {
                            append(&mut current, '$', true);
                            cursor = after;
                        }
                    } else {
                        append(&mut current, '$', true);
                        cursor = after;
                    }
                    in_word = true;
                } else {
                    append(&mut current, ch, true);
                    in_word = true;
                    cursor = after;
                }
            }
        }
    }

    match state {
        State::Single(open) => return Err(SplitError::UnterminatedSingle { pos: open }),
        State::Double(open) => return Err(SplitError::UnterminatedDouble { pos: open }),
        State::Bare => {}
    }
    if in_word {
        output.push(current);
    }
    Ok(output)
}
```
