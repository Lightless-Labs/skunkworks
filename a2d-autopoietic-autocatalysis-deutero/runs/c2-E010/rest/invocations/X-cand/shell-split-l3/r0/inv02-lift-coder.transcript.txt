    name_start(c) || c.is_ascii_digit()

    word.text.push(c);
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn separated(c: char) -> bool {
    c == ' ' || c == '\t' || c == '\r' || c == '\n'
}
fn starts_name(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}
fn continues_name(c: char) -> bool {
    starts_name(c) || c.is_ascii_digit()
}
fn append(w: &mut Word, s: &str, glob_active: bool) {
    for ch in s.chars() {
        if glob_active && matches!(ch, '*' | '?' | '[') {
            w.glob.push(w.text.len());
        }
        w.text.push(ch);
    }
}
fn env_value<'a>(env: &'a [(&str, &str)], key: &str) -> Option<&'a str> {
    for (k, v) in env {
        if *k == key {
            return Some(*v);
        }
    }
    None
}

fn expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
    glob_active: bool,
) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    if !input[after_dollar..].starts_with('{') {
        let Some(ch) = input[after_dollar..].chars().next() else {
            return Ok((after_dollar, "$".into()));
        };
        if !starts_name(ch) {
            return Ok((after_dollar, "$".into()));
        }
        let mut end = after_dollar + ch.len_utf8();
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if continues_name(c) { end += c.len_utf8(); } else { break; }
        }
        let name = &input[after_dollar..end];
        return Ok((end, env_value(env, name).unwrap_or("").to_string()));
    }

    let start = after_dollar + 1;
    let mut scan = start;
    let mut close = None;
    while scan < input.len() {
        let c = input[scan..].chars().next().unwrap();
        let n = scan + c.len_utf8();
        if c == '\\' {
            if n < input.len() {
                scan = n + input[n..].chars().next().unwrap().len_utf8();
            } else {
                scan = n;
            }
        } else if c == '}' {
            close = Some(scan);
            break;
        } else {
            scan = n;
        }
    }
    let end = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let body = &input[start..end];
    let Some(first) = body.chars().next() else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    if !starts_name(first) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    let mut n = first.len_utf8();
    while n < body.len() {
        let c = body[n..].chars().next().unwrap();
        if continues_name(c) { n += c.len_utf8(); } else { break; }
    }
    let name = &body[..n];
    let value = env_value(env, name).unwrap_or("");
    let chosen;
    if n == body.len() {
        chosen = value;
    } else if body[n..].starts_with(":-") {
        let default = &body[n + 2..];
        if value.is_empty() {
            let mut decoded = String::new();
            let mut p = 0;
            while p < default.len() {
                let c = default[p..].chars().next().unwrap();
                let next = p + c.len_utf8();
                if c == '\\' && next < default.len() {
                    let escaped = default[next..].chars().next().unwrap();
                    decoded.push(escaped);
                    p = next + escaped.len_utf8();
                } else {
                    decoded.push(c);
                    p = next;
                }
            }
            chosen = decoded;
            return Ok((end + 1, chosen));
        } else {
            chosen = value;
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    let _ = glob_active;
    Ok((end + 1, chosen.to_string()))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|b| *b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut current = Word { text: String::new(), glob: Vec::new() };
    let mut have_word = false;
    let mut quote = Quote::None;
    let mut opener = 0;
    let mut pos = 0;

    while pos < input.len() {
        let ch = input[pos..].chars().next().unwrap();
        let npos = pos + ch.len_utf8();
        match quote {
            Quote::Single => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    current.text.push(ch);
                }
                pos = npos;
            }
            Quote::None => {
                if separated(ch) {
                    if have_word {
                        result.push(current);
                        current = Word { text: String::new(), glob: Vec::new() };
                        have_word = false;
                    }
                    pos = npos;
                } else if ch == '\'' || ch == '"' {
                    have_word = true;
                    opener = pos;
                    quote = if ch == '\'' { Quote::Single } else { Quote::Double };
                    pos = npos;
                } else if ch == '\\' {
                    if npos == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let escaped = input[npos..].chars().next().unwrap();
                    let after = npos + escaped.len_utf8();
                    if escaped != '\n' {
                        have_word = true;
                        append(&mut current, &escaped.to_string(), false);
                    }
                    pos = after;
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if ch == '$' && input[npos..].starts_with('(') {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if ch == '$' {
                    have_word = true;
                    let (end, value) = expansion(input, pos, env, true)?;
                    append(&mut current, &value, true);
                    pos = end;
                } else {
                    have_word = true;
                    append(&mut current, &ch.to_string(), true);
                    pos = npos;
                }
            }
            Quote::Double => {
                if ch == '"' {
                    quote = Quote::None;
                    pos = npos;
                } else if ch == '\\' {
                    if npos == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let escaped = input[npos..].chars().next().unwrap();
                    if matches!(escaped, '"' | '\\' | '$' | '\n') {
                        if escaped != '\n' {
                            current.text.push(escaped);
                        }
                        pos = npos + escaped.len_utf8();
                    } else {
                        current.text.push('\\');
                        pos = npos;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if ch == '$' && input[npos..].starts_with('(') {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if ch == '$' {
                    have_word = true;
                    let (end, value) = expansion(input, pos, env, false)?;
                    append(&mut current, &value, false);
                    pos = end;
                } else {
                    have_word = true;
                    current.text.push(ch);
                    pos = npos;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: opener }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: opener }),
        Quote::None => {}
    }
    if have_word {
        result.push(current);
    }
    Ok(result)
}
