pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

    pub glob: Vec<usize>,
}

    matches!(c, ' ' | '\t' | '\r' | '\n')
}

    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()

fn first_env<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter().find_map(|(k, v)| if *k == name { Some(*v) } else { None })

fn append_literal(word: &mut Word, c: char, active: bool) {
    let at = word.text.len();
    word.text.push(c);
    if active && matches!(c, '*' | '?' | '[') {
        word.glob.push(at);
    }
}

fn append_data(word: &mut Word, text: &str, active: bool) {
    for c in text.chars() {
        append_literal(word, c, active);
    }
}

fn decode_default(s: &str) -> String {
    let mut out = String::new();
    let mut it = s.char_indices().peekable();
    while let Some((_, c)) = it.next() {
        if c == '\\' {
            if let Some((_, next)) = it.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn expand_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let bytes = input.as_bytes();
    let mut i = dollar + 2;
    let mut close = None;
    while i < bytes.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < bytes.len() {
                let n = input[i..].chars().next().unwrap();
                i += n.len_utf8();
            }
        } else if c == '}' {
            close = Some(i);
            break;
        } else {
            i += c.len_utf8();
        }
    }
    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let content = &input[dollar + 2..close];

    let mut chars = content.char_indices();
    let first = chars.next().map(|(_, c)| c).ok_or(SplitError::InvalidExpansion { pos: dollar })?;
    if !is_name_start(first) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    let mut name_end = first.len_utf8();
    for (idx, c) in chars {
        if is_name_continue(c) {
            name_end = idx + c.len_utf8();
        } else {
            break;
        }
    }
    let name = &content[..name_end];
    let rest = &content[name_end..];
    let value = if rest.is_empty() {
        first_env(env, name).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match first_env(env, name) {
            Some(v) if !v.is_empty() => v.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    Ok((close + 1, value))
}

pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Mode {
    Bare,
    Single,
    Double,
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn name_continue(c: char) -> bool {
    name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn push_scalar(word: &mut Word, c: char, active: bool) {
    if active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn push_text(word: &mut Word, text: &str, active: bool) {
    for c in text.chars() {
        push_scalar(word, c, active);
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word { text: String::new(), glob: Vec::new() };
    let mut active_word = false;
    let mut mode = Mode::Bare;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match mode {
            Mode::Single => {
                if c == '\'' {
                    mode = Mode::Bare;
                } else {
                    word.text.push(c);
                }
                i = next_i;
            }
            Mode::Bare => {
                if is_sep(c) {
                    if active_word {
                        words.push(word);
                        word = Word { text: String::new(), glob: Vec::new() };
                        active_word = false;
                    }
                    i = next_i;
                } else if c == '\'' || c == '"' {
                    active_word = true;
                    quote_pos = i;
                    mode = if c == '\'' { Mode::Single } else { Mode::Double };
                    i = next_i;
                } else if c == '\\' {
                    let pos = i;
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if escaped == '\n' {
                        i = after;
                    } else {
                        active_word = true;
                        push_scalar(&mut word, escaped, false);
                        i = after;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        active_word = true;
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let x = input[j..].chars().next().unwrap();
                            let x_next = j + x.len_utf8();
                            if x == '\\' {
                                if x_next < input.len() {
                                    let skipped = input[x_next..].chars().next().unwrap();
                                    j = x_next + skipped.len_utf8();
                                } else {
                                    j = x_next;
                                }
                            } else if x == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = x_next;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];

                        let mut n = 0usize;
                        let chars: Vec<(usize, char)> = contents.char_indices().collect();
                        if chars.is_empty() || !name_start(chars[0].1) {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        n = chars[0].1.len_utf8();
                        while n < contents.len() {
                            let ch = contents[n..].chars().next().unwrap();
                            if name_continue(ch) {
                                n += ch.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &contents[..n];
                        let mut selected = lookup(name, env).unwrap_or("");
                        if n < contents.len() {
                            if !contents[n..].starts_with(":-") {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            }
                            let default = &contents[n + 2..];
                            if selected.is_empty() {
                                let mut decoded = String::new();
                                let mut k = 0usize;
                                while k < default.len() {
                                    let ch = default[k..].chars().next().unwrap();
                                    let after = k + ch.len_utf8();
                                    if ch == '\\' && after < default.len() {

                                        decoded.push(escaped);
                                        k = after + escaped.len_utf8();
                                    } else {
                                        decoded.push(ch);
                                        k = after;
                                    }
                                }

                            } else {

                            }
                        } else {

                                    if name_continue(x) {
                                        end += x.len_utf8();
                                    } else {

                                    }
                                }
                                active_word = true;
                                let name = &input[after_dollar..end];
                                if let Some(value) = lookup(name, env) {
                                    push_text(&mut word, value, true);
                                }
                                i = end;
                            } else {
                                active_word = true;
                                word.text.push('$');
                                i = next_i;
                            }
                        } else {
                            active_word = true;
                            word.text.push('$');
                            i = next_i;
                        }
                    }
                } else {
                    active_word = true;
                    push_scalar(&mut word, c, true);
                    i = next_i;
                }
            }
            Mode::Double => {
                if c == '"' {
                    mode = Mode::Bare;
                    i = next_i;
                } else if c == '\\' {
                    let pos = i;
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$') {
                        word.text.push(escaped);
                        i = after;
                    } else if escaped == '\n' {
                        i = after;
                    } else {
                        word.text.push('\\');
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        active_word = true;
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let x = input[j..].chars().next().unwrap();
                            let x_next = j + x.len_utf8();
                            if x == '\\' {
                                if x_next < input.len() {
                                    let skipped = input[x_next..].chars().next().unwrap();
                                    j = x_next + skipped.len_utf8();
                                } else {
                                    j = x_next;
                                }
                            } else if x == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = x_next;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let mut n = 0usize;
                        let first = contents.chars().next().ok_or(SplitError::InvalidExpansion { pos: dollar })?;
                        if !name_start(first) {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        n = first.len_utf8();
                        while n < contents.len() {
                            let ch = contents[n..].chars().next().unwrap();
                            if name_continue(ch) { n += ch.len_utf8(); } else { break; }
                        }
                        let name = &contents[..n];
                        let selected = lookup(name, env).unwrap_or("");
                        if n < contents.len() {
                            if !contents[n..].starts_with(":-") {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            }
                            if selected.is_empty() {
                                let default = &contents[n + 2..];
                                let mut decoded = String::new();
                                let mut k = 0usize;
                                while k < default.len() {
                                    let ch = default[k..].chars().next().unwrap();
                                    let after = k + ch.len_utf8();
                                    if ch == '\\' && after < default.len() {
                                        let escaped = default[after..].chars().next().unwrap();
                                        decoded.push(escaped);
                                        k = after + escaped.len_utf8();
                                    } else {
                                        decoded.push(ch);
                                        k = after;
                                    }
                                }
                                push_text(&mut word, &decoded, false);
                            } else {
                                push_text(&mut word, selected, false);
                            }
                        } else {
                            push_text(&mut word, selected, false);
                        }
                        i = close + 1;
                    } else if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if name_start(n) {
                            let mut end = next_i + n.len_utf8();
                            while end < input.len() {
                                let x = input[end..].chars().next().unwrap();
                                if name_continue(x) { end += x.len_utf8(); } else { break; }
                            }
                            active_word = true;
                            if let Some(value) = lookup(&input[next_i..end], env) {
                                push_text(&mut word, value, false);
                            }
                            i = end;
                        } else {
                            active_word = true;
                            word.text.push('$');
                            i = next_i;
                        }
                    } else {
                        active_word = true;
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    active_word = true;
                    word.text.push(c);
                    i = next_i;
                }
            }
        }
    }

    match mode {
        Mode::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Mode::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Mode::Bare => {}
    }
    if active_word {
        words.push(word);
    }
    Ok(words)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy)]
enum State {
    Bare,
    Single(usize),
    Double(usize),
}

struct Parser<'a> {
    src: &'a str,
    env: &'a [(&'a str, &'a str)],
    at: usize,
    state: State,
    words: Vec<Word>,
    current: Word,
    has_word: bool,
}

impl<'a> Parser<'a> {
    fn char_at(&self, at: usize) -> char {
        self.src[at..].chars().next().unwrap()
    }

    fn step(&self, at: usize) -> usize {
        at + self.char_at(at).len_utf8()
    }

    fn add(&mut self, c: char, glob: bool) {
        if glob && matches!(c, '*' | '?' | '[') {
            self.current.glob.push(self.current.text.len());
        }
        self.current.text.push(c);
    }

    fn add_string(&mut self, s: &str, glob: bool) {
        for c in s.chars() {
            self.add(c, glob);
        }
    }

    fn value(&self, name: &str) -> &str {
        self.env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v).unwrap_or("")
    }

    fn expansion(&self, start: usize) -> Result<Option<(usize, String)>, SplitError> {
        let d = start + 1;
        if d == self.src.len() {
            return Ok(None);
        }
        let c = self.char_at(d);
        if c == '{' {
            let content_start = d + 1;
            let mut p = content_start;
            let close = loop {
                if p >= self.src.len() {
                    return Err(SplitError::UnterminatedExpansion { pos: start });
                }
                let ch = self.char_at(p);
                let n = self.step(p);
                if ch == '\\' {
                    p = if n < self.src.len() { self.step(n) } else { n };
                } else if ch == '}' {
                    break p;
                } else {
                    p = n;
                }
            };
            let body = &self.src[content_start..close];
            let mut iter = body.char_indices();
            let (first_idx, first) = iter.next().ok_or(SplitError::InvalidExpansion { pos: start })?;
            if first_idx != 0 || !(first.is_ascii_alphabetic() || first == '_') {
                return Err(SplitError::InvalidExpansion { pos: start });
            }
            let mut name_end = first.len_utf8();
            for (idx, ch) in iter {
                if ch.is_ascii_alphanumeric() || ch == '_' {
                    name_end = idx + ch.len_utf8();
                } else {
                    break;
                }
            }
            let name = &body[..name_end];
            let rem = &body[name_end..];
            let result = if rem.is_empty() {
                self.value(name).to_owned()
            } else if let Some(default) = rem.strip_prefix(":-") {
                let val = self.env.iter().find(|(k, _)| *k == name).map(|(_, v)| *v);
                match val {
                    Some(v) if !v.is_empty() => v.to_owned(),
                    _ => {
                        let mut decoded = String::new();
                        let mut chars = default.chars();
                        while let Some(ch) = chars.next() {
                            if ch == '\\' {
                                if let Some(next) = chars.next() {
                                    decoded.push(next);
                                }
                            } else {
                                decoded.push(ch);
                            }
                        }
                        decoded
                    }
                }
            } else {
                return Err(SplitError::InvalidExpansion { pos: start });
            };
            Ok(Some((close + 1, result)))
        } else if c.is_ascii_alphabetic() || c == '_' {
            let mut end = self.step(d);
            while end < self.src.len() {
                let ch = self.char_at(end);
                if ch.is_ascii_alphanumeric() || ch == '_' {
                    end = self.step(end);
                } else {
                    break;
                }
            }
            Ok(Some((end, self.value(&self.src[d..end]).to_owned())))
        } else {
            Ok(None)
        }
    }

    fn parse(mut self) -> Result<Vec<Word>, SplitError> {
        while self.at < self.src.len() {
            let c = self.char_at(self.at);
            let next = self.step(self.at);
            match self.state {
                State::Single(_) => {
                    if c == '\'' {
                        self.state = State::Bare;
                    } else {
                        self.add(c, false);
                    }
                    self.at = next;
                }
                State::Double(_) => {
                    if c == '"' {
                        self.state = State::Bare;
                        self.at = next;
                    } else if c == '\\' {
                        if next == self.src.len() {
                            return Err(SplitError::TrailingEscape { pos: self.at });
                        }
                        let n = self.char_at(next);
                        if matches!(n, '"' | '\\' | '$') {
                            self.add(n, false);
                            self.at = self.step(next);
                        } else if n == '\n' {
                            self.at = self.step(next);
                        } else {
                            self.add('\\', false);
                            self.at = next;
                        }
                    } else if c == '`' {
                        return Err(SplitError::CommandSubstitution { pos: self.at });
                    } else if c == '$' {
                        if self.src[next..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: self.at });
                        }
                        if let Some((end, value)) = self.expansion(self.at)? {
                            self.has_word = true;
                            self.add_string(&value, false);
                            self.at = end;
                        } else {
                            self.add('$', false);
                            self.at = next;
                        }
                    } else {
                        self.add(c, false);
                        self.at = next;
                    }
                }
                State::Bare => {
                    if matches!(c, ' ' | '\t' | '\r' | '\n') {
                        if self.has_word {
                            self.words.push(std::mem::replace(
                                &mut self.current,
                                Word { text: String::new(), glob: Vec::new() },
                            ));
                            self.has_word = false;
                        }
                        self.at = next;
                    } else if c == '\'' {
                        self.has_word = true;
                        self.state = State::Single(self.at);
                        self.at = next;
                    } else if c == '"' {
                        self.has_word = true;
                        self.state = State::Double(self.at);
                        self.at = next;
                    } else if c == '\\' {
                        if next == self.src.len() {
                            return Err(SplitError::TrailingEscape { pos: self.at });
                        }
                        let n = self.char_at(next);
                        if n == '\n' {
                            self.at = self.step(next);
                        } else {
                            self.has_word = true;
                            self.add(n, false);
                            self.at = self.step(next);
                        }
                    } else if c == '`' {
                        return Err(SplitError::CommandSubstitution { pos: self.at });
                    } else if c == '$' {
                        if self.src[next..].starts_with('(') {
                            return Err(SplitError::CommandSubstitution { pos: self.at });
                        }
                        if let Some((end, value)) = self.expansion(self.at)? {
                            self.has_word = true;
                            self.add_string(&value, true);
                            self.at = end;
                        } else {
                            self.has_word = true;
                            self.add('$', true);
                            self.at = next;
                        }
                    } else {
                        self.has_word = true;
                        self.add(c, true);
                        self.at = next;
                    }
                }
            }
        }
        match self.state {
            State::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
            State::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
            State::Bare => {}
        }
        if self.has_word {
            self.words.push(self.current);
        }
        Ok(self.words)
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }
    Parser {
        src: input,
        env,
        at: 0,
        state: State::Bare,
        words: Vec::new(),
        current: Word { text: String::new(), glob: Vec::new() },
        has_word: false,
    }
    .parse()
}

pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

fn start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn cont(c: char) -> bool {
    c == '_' || c.is_ascii_alphanumeric()
}

fn scalar_at(s: &str, i: usize) -> Option<char> {
    s.get(i..)?.chars().next()
}

fn add(w: &mut Word, c: char, glob: bool) {
    let p = w.text.len();
    w.text.push(c);
    if glob && matches!(c, '*' | '?' | '[') {
        w.glob.push(p);
    }
}

fn add_str(w: &mut Word, s: &str, glob: bool) {
    for c in s.chars() {
        add(w, c, glob);
    }

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> &'a str {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
        .unwrap_or("")
}

fn parse_expansion(
    s: &str,
    at: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let mut p = at + 2;
    let end;
    loop {
        if p == s.len() {
            return Err(SplitError::UnterminatedExpansion { pos: at });
        }
        let c = scalar_at(s, p).unwrap();
        if c == '\\' {
            p += 1;
            if p < s.len() {
                p += scalar_at(s, p).unwrap().len_utf8();
            }
        } else if c == '}' {
            end = p;
            break;
        } else {
            p += c.len_utf8();
        }
    }

    let body = &s[at + 2..end];
    let mut name_end = 0;
    for c in body.chars() {
        if cont(c) {
            name_end += c.len_utf8();
        } else {
            break;
        }
    }
    let name = &body[..name_end];
    if !name.chars().next().map(start).unwrap_or(false) {
        return Err(SplitError::InvalidExpansion { pos: at });
    }
    let tail = &body[name_end..];
    let val = env_value(env, name);
    if tail.is_empty() {
        return Ok((end + 1, val.to_string()));
    }
    if let Some(default) = tail.strip_prefix(":-") {
        if val.is_empty() {
            let mut decoded = String::new();
            let mut it = default.chars();
            while let Some(c) = it.next() {
                if c == '\\' {
                    if let Some(next) = it.next() {
                        decoded.push(next);
                    } else {
                        decoded.push('\\');
                    }
                } else {
                    decoded.push(c);
                }
            }
            Ok((end + 1, decoded))
        } else {
            Ok((end + 1, val.to_string()))
        }
    } else {
        Err(SplitError::InvalidExpansion { pos: at })
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }
    let mut result = Vec::new();
    let mut word = Word { text: String::new(), glob: Vec::new() };
    let mut active = false;
    let mut q = 0u8;
    let mut qpos = 0;
    let mut p = 0usize;

    while p < input.len() {
        let c = scalar_at(input, p).unwrap();
        let size = c.len_utf8();
        if q == 1 {
            if c == '\'' { q = 0; } else { add(&mut word, c, false); }
            p += size;
            continue;
        }
        if q == 2 {
            match c {
                '"' => { q = 0; p += size; }
                '\\' => {
                    let escape_pos = p;
                    p += 1;
                    if p == input.len() { return Err(SplitError::TrailingEscape { pos: escape_pos }); }
                    let next = scalar_at(input, p).unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' { add(&mut word, next, false); }
                        p += next.len_utf8();
                    } else { add(&mut word, '\\', false); }
                }
                '`' => return Err(SplitError::CommandSubstitution { pos: p }),
                '$' => {
                    let n = p + 1;
                    if input.get(n..).map(|x| x.starts_with('(')).unwrap_or(false) {
                        return Err(SplitError::CommandSubstitution { pos: p });
                    }
                    if input.get(n..).map(|x| x.starts_with('{')).unwrap_or(false) {
                        active = true;
                        let (end, value) = parse_expansion(input, p, env)?;
                        add_str(&mut word, &value, false);
                        p = end;
                    } else if let Some(ch) = scalar_at(input, n) {
                        if start(ch) {
                            let mut end = n;
                            while let Some(ch) = scalar_at(input, end) {
                                if !cont(ch) { break; }
                                end += ch.len_utf8();
                            }
                            active = true;
                            add_str(&mut word, env_value(env, &input[n..end]), false);
                            p = end;
                        } else { add(&mut word, '$', false); p += 1; }
                    } else { add(&mut word, '$', false); p += 1; }
                }
                _ => { add(&mut word, c, false); p += size; }
            }
            continue;
        }

        match c {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    result.push(word);
                    word = Word { text: String::new(), glob: Vec::new() };
                    active = false;
                }
                p += size;
            }
            '\'' => { q = 1; qpos = p; active = true; p += size; }
            '"' => { q = 2; qpos = p; active = true; p += size; }
            '\\' => {
                let escape_pos = p;
                p += 1;
                if p == input.len() { return Err(SplitError::TrailingEscape { pos: escape_pos }); }
                let next = scalar_at(input, p).unwrap();
                if next != '\n' { active = true; add(&mut word, next, false); }
                p += next.len_utf8();
            }
            '`' => return Err(SplitError::CommandSubstitution { pos: p }),
            '$' => {
                let n = p + 1;
                if input.get(n..).map(|x| x.starts_with('(')).unwrap_or(false) {
                    return Err(SplitError::CommandSubstitution { pos: p });
                }
                if input.get(n..).map(|x| x.starts_with('{')).unwrap_or(false) {
                    active = true;
                    let (end, value) = parse_expansion(input, p, env)?;
                    add_str(&mut word, &value, true);
                    p = end;
                } else if let Some(ch) = scalar_at(input, n) {
                    if start(ch) {
                        let mut end = n;
                        while let Some(ch) = scalar_at(input, end) {
                            if !cont(ch) { break; }
                            end += ch.len_utf8();
                        }
                        active = true;
                        add_str(&mut word, env_value(env, &input[n..end]), true);
                        p = end;
                    } else {
                        active = true;
                        add(&mut word, '$', true);
                        p += 1;
                    }
                } else {
                    active = true;
                    add(&mut word, '$', true);
                    p += 1;
                }
            }
            _ => {
                active = true;
                add(&mut word, c, matches!(c, '*' | '?' | '['));
                p += size;
            }
        }
    }

    if q == 1 { return Err(SplitError::UnterminatedSingle { pos: qpos }); }
    if q == 2 { return Err(SplitError::UnterminatedDouble { pos: qpos }); }
    if active { result.push(word); }
    Ok(result)
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word { text: String::new(), glob: Vec::new() };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();

        match quote {
            Quote::Single(open) => {
                if c == '\'' {
                    quote = Quote::None;

                    push_char(&mut word, c, false);
                }
                i = next_i;
                let _ = open;
            }
            Quote::Double(open) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after = next_i + n.len_utf8();
                    if matches!(n, '"' | '\\' | '$') {
                        push_char(&mut word, n, false);
                        i = after;
                    } else if n == '\n' {
                        i = after;
                    } else {
                        push_char(&mut word, '\\', false);
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    let (end, value, was_expansion) = parse_expansion(input, i, env)?;
                    if was_expansion {
                        active = true;
                        push_text(&mut word, &value, false);
                        i = end;
                    } else {
                        push_char(&mut word, '$', false);
                        i = next_i;
                    }
                } else {
                    push_char(&mut word, c, false);
                    i = next_i;
                }
                let _ = open;
            }
            Quote::None => {
                if is_sep(c) {
                    if active {
                        words.push(word);
                        word = Word { text: String::new(), glob: Vec::new() };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let n = input[next_i..].chars().next().unwrap();
                    let after = next_i + n.len_utf8();
                    if n == '\n' {
                        i = after;
                    } else {
                        active = true;
                        push_char(&mut word, n, false);
                        i = after;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    let (end, value, was_expansion) = parse_expansion(input, i, env)?;
                    if was_expansion {
                        active = true;
                        push_text(&mut word, &value, true);
                        i = end;
                    } else {
                        active = true;
                        push_char(&mut word, '$', true);
                        i = next_i;
                    }
                } else {
                    active = true;
                    push_char(&mut word, c, true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(open) => return Err(SplitError::UnterminatedSingle { pos: open }),
        Quote::Double(open) => {
            let _ = open;
            return Err(SplitError::UnterminatedDouble { pos: open });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String, bool), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((after_dollar, String::new(), false));
    }
    let c = input[after_dollar..].chars().next().unwrap();

    if c == '{' {
        let content_start = after_dollar + 1;
        let mut pos = content_start;
        let mut close = None;
        while pos < input.len() {
            let ch = input[pos..].chars().next().unwrap();
            let end = pos + ch.len_utf8();
            if ch == '\\' {
                if end < input.len() {
                    let skipped = input[end..].chars().next().unwrap();
                    pos = end + skipped.len_utf8();
                } else {
                    pos = end;
                }
            } else if ch == '}' {
                close = Some(pos);
                break;
            } else {
                pos = end;
            }
        }
        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let content = &input[content_start..close];

        let mut name_end = 0;
        let mut chars = content.char_indices();
        match chars.next() {
            Some((_, first)) if name_start(first) => {
                name_end = first.len_utf8();
                for (idx, ch) in chars {
                    if name_char(ch) {
                        name_end = idx + ch.len_utf8();
                    } else {
                        break;
                    }
                }
            }
            _ => return Err(SplitError::InvalidExpansion { pos: dollar }),
        }

        let name = &content[..name_end];
        let rest = &content[name_end..];
        let selected = if rest.is_empty() {
            env_value(env, name).unwrap_or("").to_string()
        } else if let Some(default) = rest.strip_prefix(":-") {
            match env_value(env, name) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };
        Ok((close + 1, selected, true))
    } else if name_start(c) {
        let mut end = after_dollar + c.len_utf8();
        while end < input.len() {
            let ch = input[end..].chars().next().unwrap();
            if name_char(ch) {
                end += ch.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        Ok((end, env_value(env, name).unwrap_or("").to_string(), true))
    } else {
        Ok((after_dollar, String::new(), false))
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn append_char(word: &mut Word, c: char, wildcard_active: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if wildcard_active && matches!(c, '*' | '?' | '[') {
        word.glob.push(offset);
    }
}

fn append_string(word: &mut Word, value: &str, wildcard_active: bool) {
    for c in value.chars() {
        append_char(word, c, wildcard_active);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                out.push(next);
            } else {
                out.push('\\');
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn parse_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let mut close = None;
    let mut iter = input[dollar + 2..].char_indices().peekable();
    while let Some((rel, c)) = iter.next() {
        if c == '\\' {
            let _ = iter.next();
        } else if c == '}' {
            close = Some(dollar + 2 + rel);
            break;
        }
    }
    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[dollar + 2..close];

    let name_len = contents
        .char_indices()
        .take_while(|(_, c)| is_name_continue(*c))
        .last()
        .map(|(i, c)| i + c.len_utf8())
        .unwrap_or(0);

    if name_len == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_len];
    let mut name_chars = name.chars();
    if !name_chars.next().map(is_name_start).unwrap_or(false)
        || !name_chars.all(is_name_continue)
    {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let value = lookup(env, name).unwrap_or("");
    let rest = &contents[name_len..];
    if rest.is_empty() {
        return Ok((close + 1, value.to_string()));
    }
    if let Some(default) = rest.strip_prefix(":-") {
        if !value.is_empty() {
            Ok((close + 1, value.to_string()))
        } else {
            Ok((close + 1, decode_default(default)))
        }
    } else {
        Err(SplitError::InvalidExpansion { pos: dollar })
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut quote_start = 0;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let n = c.len_utf8();

        match quote {
            Quote::Single => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    append_char(&mut current, c, false);
                }
                i += n;
            }
            Quote::Double => {
                if c == '"' {
                    quote = Quote::None;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += n;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            append_char(&mut current, next, false);
                        }
                        i += next.len_utf8();
                    } else {
                        append_char(&mut current, '\\', false);
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_pos = i + n;

                        return Err(SplitError::CommandSubstitution { pos: i });

                        } else {
                            append_char(&mut current, '$', false);

                        }
                    } else {
                        append_char(&mut current, '$', false);
                        i += n;
                    }
                } else {
                    append_char(&mut current, c, false);
                    i += n;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(current);
                        current = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i += n;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single;
                    quote_start = i;
                    i += n;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double;
                    quote_start = i;
                    i += n;
                } else if c == '\\' {
                    let pos = i;
                    i += n;
                    if i >= input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next != '\n' {
                        active = true;
                        append_char(&mut current, next, false);
                    }
                    i += next.len_utf8();
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    let next_pos = i + n;
                    if next_pos < input.len() && input[next_pos..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if next_pos < input.len() && input[next_pos..].starts_with('{') {
                        active = true;
                        let (end, value) = parse_braced(input, i, env)?;
                        append_string(&mut current, &value, true);
                        i = end;
                    } else if next_pos < input.len() {
                        let next = input[next_pos..].chars().next().unwrap();

                                }

                            i = end;
                        } else {
                            active = true;
                            append_char(&mut current, '$', true);
                            i += n;
                        }

                        active = true;
                        append_char(&mut current, '$', true);
                        i += n;
                    }
                } else {
                    active = true;
                    append_char(&mut current, c, matches!(c, '*' | '?' | '['));
                    i += n;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: quote_start }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: quote_start }),

    }

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum State {
    Bare,
    Single(usize),
    Double(usize),
}

fn separator(c: char) -> bool {
    c == ' ' || c == '\t' || c == '\r' || c == '\n'
}

fn start_name(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn name_char(c: char) -> bool {
    start_name(c) || c.is_ascii_digit()
}

fn append(word: &mut Word, c: char, glob: bool) {
    let offset = word.text.len();
    word.text.push(c);
    if glob && (c == '*' || c == '?' || c == '[') {
        word.glob.push(offset);
    }
}

fn append_string(word: &mut Word, s: &str, glob: bool) {
    for c in s.chars() {
        append(word, c, glob);
    }
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> &'a str {
    env.iter().find(|entry| entry.0 == name).map_or("", |entry| entry.1)
}

fn braced(input: &str, start: usize, env: &[(&str, &str)]) -> Result<(usize, String), SplitError> {
    let mut p = start + 2;
    let mut closing = None;
    while p < input.len() {
        let ch = input[p..].chars().next().unwrap();
        if ch == '\\' {
            p += 1;
            if p < input.len() {
                let escaped = input[p..].chars().next().unwrap();
                p += escaped.len_utf8();
            }
        } else if ch == '}' {
            closing = Some(p);
            break;
        } else {
            p += ch.len_utf8();
        }
    }
    let closing = closing.ok_or(SplitError::UnterminatedExpansion { pos: start })?;
    let body = &input[start + 2..closing];
    let mut iter = body.char_indices();
    let (first_at, first) = iter.next().ok_or(SplitError::InvalidExpansion { pos: start })?;
    if first_at != 0 || !start_name(first) {
        return Err(SplitError::InvalidExpansion { pos: start });
    }
    let mut end_name = first.len_utf8();
    for (at, ch) in iter {
        if name_char(ch) {
            end_name = at + ch.len_utf8();
        } else {
            break;
        }
    }
    let name = &body[..end_name];
    let suffix = &body[end_name..];
    if suffix.is_empty() {
        Ok((closing + 1, lookup(env, name).to_owned()))
    } else if let Some(default) = suffix.strip_prefix(":-") {
        let existing = env.iter().find(|entry| entry.0 == name).map(|entry| entry.1);
        let text = match existing {
            Some(value) if !value.is_empty() => value.to_owned(),
            _ => {
                let mut decoded = String::new();
                let mut chars = default.chars();
                while let Some(ch) = chars.next() {
                    if ch == '\\' {
                        if let Some(next) = chars.next() {
                            decoded.push(next);
                        }
                    } else {
                        decoded.push(ch);
                    }
                }
                decoded
            }
        };
        Ok((closing + 1, text))
    } else {
        Err(SplitError::InvalidExpansion { pos: start })
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(nul) = input.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos: nul });
    }

    let mut output = Vec::new();
    let mut current = Word { text: String::new(), glob: Vec::new() };
    let mut in_word = false;
    let mut state = State::Bare;
    let mut cursor = 0;

    while cursor < input.len() {
        let ch = input[cursor..].chars().next().unwrap();
        let after = cursor + ch.len_utf8();

        match state {
            State::Single(open) => {
                if ch == '\'' {
                    state = State::Bare;
                } else {
                    append(&mut current, ch, false);
                }
                let _ = open;
                cursor = after;
            }
            State::Double(open) => {
                if ch == '"' {
                    state = State::Bare;
                    cursor = after;
                } else if ch == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[after..].chars().next().unwrap();
                    if escaped == '"' || escaped == '\\' || escaped == '$' || escaped == '\n' {
                        if escaped != '\n' {
                            append(&mut current, escaped, false);
                        }
                        cursor = after + escaped.len_utf8();
                    } else {
                        append(&mut current, '\\', false);
                        cursor = after;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if after < input.len() && input[after..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if after < input.len() && input.as_bytes()[after] == b'{' {
                        let (end, value) = braced(input, cursor, env)?;
                        append_string(&mut current, &value, false);
                        cursor = end;
                    } else if after < input.len() {
                        let next = input[after..].chars().next().unwrap();
                        if start_name(next) {
                            let mut end = after + next.len_utf8();
                            while end < input.len() {
                                let c = input[end..].chars().next().unwrap();
                                if name_char(c) {
                                    end += c.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            append_string(&mut current, lookup(env, &input[after..end]), false);
                            cursor = end;
                        } else {
                            append(&mut current, '$', false);
                            cursor = after;
                        }
                    } else {
                        append(&mut current, '$', false);
                        cursor = after;
                    }
                    in_word = true;
                } else {
                    append(&mut current, ch, false);
                    cursor = after;
                }
                let _ = open;
            }
            State::Bare => {
                if separator(ch) {
                    if in_word {
                        output.push(current);
                        current = Word { text: String::new(), glob: Vec::new() };
                        in_word = false;
                    }
                    cursor = after;
                } else if ch == '\'' {
                    state = State::Single(cursor);
                    in_word = true;
                    cursor = after;
                } else if ch == '"' {
                    state = State::Double(cursor);
                    in_word = true;
                    cursor = after;
                } else if ch == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[after..].chars().next().unwrap();
                    cursor = after + escaped.len_utf8();
                    if escaped != '\n' {
                        append(&mut current, escaped, false);
                        in_word = true;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos: cursor });
                } else if ch == '$' {
                    if after < input.len() && input[after..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: cursor });
                    }
                    if after < input.len() && input.as_bytes()[after] == b'{' {
                        let (end, value) = braced(input, cursor, env)?;
                        append_string(&mut current, &value, true);
                        cursor = end;
                    } else if after < input.len() {
                        let next = input[after..].chars().next().unwrap();
                        if start_name(next) {
                            let mut end = after + next.len_utf8();
                            while end < input.len() {
                                let c = input[end..].chars().next().unwrap();
                                if name_char(c) {
                                    end += c.len_utf8();
                                } else {
                                    break;
                                }
                            }
                            append_string(&mut current, lookup(env, &input[after..end]), true);
                            cursor = end;
                        } else {
                            append(&mut current, '$', true);
                            cursor = after;
                        }
                    } else {
                        append(&mut current, '$', true);
                        cursor = after;
                    }
                    in_word = true;
                } else {
                    append(&mut current, ch, true);
                    in_word = true;
                    cursor = after;
                }
            }
        }
    }

    match state {
        State::Single(open) => return Err(SplitError::UnterminatedSingle { pos: open }),
        State::Double(open) => return Err(SplitError::UnterminatedDouble { pos: open }),
        State::Bare => {}
    }
    if in_word {
        output.push(current);
    }
    Ok(output)
}
