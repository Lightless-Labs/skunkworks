- NUL is checked across the full input before parsing, so it outranks earlier syntax errors.
- Braced expansion searches for the first unescaped `}` before validating its contents; backslash skips one Unicode scalar during the search.
- A trailing backslash at EOF inside double quotes wins over an unclosed-double-quote error.
- Wildcard offsets count UTF-8 bytes in the assembled word; expansion values are appended atomically and inherit the quote context’s wildcard activity.
- Command markers are rejected as soon as encountered outside single quotes; backticks remain errors even after a retained double-quote backslash.
- Solution B uses the same specified semantics with a separate parser organization; there are no deliberately different readings.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Mode {
    Bare,
    Single,
    Double,
}

fn is_sep(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn name_continue(c: char) -> bool {
    name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find(|(key, _)| *key == name).map(|(_, value)| *value)
}

fn push_scalar(word: &mut Word, c: char, active: bool) {
    if active && matches!(c, '*' | '?' | '[') {
        word.glob.push(word.text.len());
    }
    word.text.push(c);
}

fn push_text(word: &mut Word, text: &str, active: bool) {
    for c in text.chars() {
        push_scalar(word, c, active);
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word { text: String::new(), glob: Vec::new() };
    let mut active_word = false;
    let mut mode = Mode::Bare;
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match mode {
            Mode::Single => {
                if c == '\'' {
                    mode = Mode::Bare;
                } else {
                    word.text.push(c);
                }
                i = next_i;
            }
            Mode::Bare => {
                if is_sep(c) {
                    if active_word {
                        words.push(word);
                        word = Word { text: String::new(), glob: Vec::new() };
                        active_word = false;
                    }
                    i = next_i;
                } else if c == '\'' || c == '"' {
                    active_word = true;
                    quote_pos = i;
                    mode = if c == '\'' { Mode::Single } else { Mode::Double };
                    i = next_i;
                } else if c == '\\' {
                    let pos = i;
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if escaped == '\n' {
                        i = after;
                    } else {
                        active_word = true;
                        push_scalar(&mut word, escaped, false);
                        i = after;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        active_word = true;
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let x = input[j..].chars().next().unwrap();
                            let x_next = j + x.len_utf8();
                            if x == '\\' {
                                if x_next < input.len() {
                                    let skipped = input[x_next..].chars().next().unwrap();
                                    j = x_next + skipped.len_utf8();
                                } else {
                                    j = x_next;
                                }
                            } else if x == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = x_next;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];

                        let mut n = 0usize;
                        let chars: Vec<(usize, char)> = contents.char_indices().collect();
                        if chars.is_empty() || !name_start(chars[0].1) {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        n = chars[0].1.len_utf8();
                        while n < contents.len() {
                            let ch = contents[n..].chars().next().unwrap();
                            if name_continue(ch) {
                                n += ch.len_utf8();
                            } else {
                                break;
                            }
                        }
                        let name = &contents[..n];
                        let mut selected = lookup(name, env).unwrap_or("");
                        if n < contents.len() {
                            if !contents[n..].starts_with(":-") {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            }
                            let default = &contents[n + 2..];
                            if selected.is_empty() {
                                let mut decoded = String::new();
                                let mut k = 0usize;
                                while k < default.len() {
                                    let ch = default[k..].chars().next().unwrap();
                                    let after = k + ch.len_utf8();
                                    if ch == '\\' && after < default.len() {
                                        let escaped = default[after..].chars().next().unwrap();
                                        decoded.push(escaped);
                                        k = after + escaped.len_utf8();
                                    } else {
                                        decoded.push(ch);
                                        k = after;
                                    }
                                }
                                push_text(&mut word, &decoded, true);
                            } else {
                                push_text(&mut word, selected, true);
                            }
                        } else {
                            push_text(&mut word, selected, true);
                        }
                        i = close + 1;
                    } else {
                        let after_dollar = next_i;
                        if after_dollar < input.len() {
                            let n = input[after_dollar..].chars().next().unwrap();
                            if name_start(n) {
                                let mut end = after_dollar + n.len_utf8();
                                while end < input.len() {
                                    let x = input[end..].chars().next().unwrap();
                                    if name_continue(x) {
                                        end += x.len_utf8();
                                    } else {
                                        break;
                                    }
                                }
                                active_word = true;
                                let name = &input[after_dollar..end];
                                if let Some(value) = lookup(name, env) {
                                    push_text(&mut word, value, true);
                                }
                                i = end;
                            } else {
                                active_word = true;
                                word.text.push('$');
                                i = next_i;
                            }
                        } else {
                            active_word = true;
                            word.text.push('$');
                            i = next_i;
                        }
                    }
                } else {
                    active_word = true;
                    push_scalar(&mut word, c, true);
                    i = next_i;
                }
            }
            Mode::Double => {
                if c == '"' {
                    mode = Mode::Bare;
                    i = next_i;
                } else if c == '\\' {
                    let pos = i;
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let escaped = input[next_i..].chars().next().unwrap();
                    let after = next_i + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$') {
                        word.text.push(escaped);
                        i = after;
                    } else if escaped == '\n' {
                        i = after;
                    } else {
                        word.text.push('\\');
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        active_word = true;
                        let dollar = i;
                        let content_start = next_i + 1;
                        let mut j = content_start;
                        let mut close = None;
                        while j < input.len() {
                            let x = input[j..].chars().next().unwrap();
                            let x_next = j + x.len_utf8();
                            if x == '\\' {
                                if x_next < input.len() {
                                    let skipped = input[x_next..].chars().next().unwrap();
                                    j = x_next + skipped.len_utf8();
                                } else {
                                    j = x_next;
                                }
                            } else if x == '}' {
                                close = Some(j);
                                break;
                            } else {
                                j = x_next;
                            }
                        }
                        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
                        let contents = &input[content_start..close];
                        let mut n = 0usize;
                        let first = contents.chars().next().ok_or(SplitError::InvalidExpansion { pos: dollar })?;
                        if !name_start(first) {
                            return Err(SplitError::InvalidExpansion { pos: dollar });
                        }
                        n = first.len_utf8();
                        while n < contents.len() {
                            let ch = contents[n..].chars().next().unwrap();
                            if name_continue(ch) { n += ch.len_utf8(); } else { break; }
                        }
                        let name = &contents[..n];
                        let selected = lookup(name, env).unwrap_or("");
                        if n < contents.len() {
                            if !contents[n..].starts_with(":-") {
                                return Err(SplitError::InvalidExpansion { pos: dollar });
                            }
                            if selected.is_empty() {
                                let default = &contents[n + 2..];
                                let mut decoded = String::new();
                                let mut k = 0usize;
                                while k < default.len() {
                                    let ch = default[k..].chars().next().unwrap();
                                    let after = k + ch.len_utf8();
                                    if ch == '\\' && after < default.len() {
                                        let escaped = default[after..].chars().next().unwrap();
                                        decoded.push(escaped);
                                        k = after + escaped.len_utf8();
                                    } else {
                                        decoded.push(ch);
                                        k = after;
                                    }
                                }
                                push_text(&mut word, &decoded, false);
                            } else {
                                push_text(&mut word, selected, false);
                            }
                        } else {
                            push_text(&mut word, selected, false);
                        }
                        i = close + 1;
                    } else if next_i < input.len() {
                        let n = input[next_i..].chars().next().unwrap();
                        if name_start(n) {
                            let mut end = next_i + n.len_utf8();
                            while end < input.len() {
                                let x = input[end..].chars().next().unwrap();
                                if name_continue(x) { end += x.len_utf8(); } else { break; }
                            }
                            active_word = true;
                            if let Some(value) = lookup(&input[next_i..end], env) {
                                push_text(&mut word, value, false);
                            }
                            i = end;
                        } else {
                            active_word = true;
                            word.text.push('$');
                            i = next_i;
                        }
                    } else {
                        active_word = true;
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    active_word = true;
                    word.text.push(c);
                    i = next_i;
                }
            }
        }
    }

    match mode {
        Mode::Single => return Err(SplitError::UnterminatedSingle { pos: quote_pos }),
        Mode::Double => return Err(SplitError::UnterminatedDouble { pos: quote_pos }),
        Mode::Bare => {}
    }
    if active_word {
        words.push(word);
    }
    Ok(words)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single,
    Double,
}

fn separated(c: char) -> bool {
    c == ' ' || c == '\t' || c == '\r' || c == '\n'
}
fn starts_name(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}
fn continues_name(c: char) -> bool {
    starts_name(c) || c.is_ascii_digit()
}
fn append(w: &mut Word, s: &str, glob_active: bool) {
    for ch in s.chars() {
        if glob_active && matches!(ch, '*' | '?' | '[') {
            w.glob.push(w.text.len());
        }
        w.text.push(ch);
    }
}
fn env_value<'a>(env: &'a [(&str, &str)], key: &str) -> Option<&'a str> {
    for (k, v) in env {
        if *k == key {
            return Some(*v);
        }
    }
    None
}

fn expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
    glob_active: bool,
) -> Result<(usize, String), SplitError> {
    let after_dollar = dollar + 1;
    if !input[after_dollar..].starts_with('{') {
        let Some(ch) = input[after_dollar..].chars().next() else {
            return Ok((after_dollar, "$".into()));
        };
        if !starts_name(ch) {
            return Ok((after_dollar, "$".into()));
        }
        let mut end = after_dollar + ch.len_utf8();
        while end < input.len() {
            let c = input[end..].chars().next().unwrap();
            if continues_name(c) { end += c.len_utf8(); } else { break; }
        }
        let name = &input[after_dollar..end];
        return Ok((end, env_value(env, name).unwrap_or("").to_string()));
    }

    let start = after_dollar + 1;
    let mut scan = start;
    let mut close = None;
    while scan < input.len() {
        let c = input[scan..].chars().next().unwrap();
        let n = scan + c.len_utf8();
        if c == '\\' {
            if n < input.len() {
                scan = n + input[n..].chars().next().unwrap().len_utf8();
            } else {
                scan = n;
            }
        } else if c == '}' {
            close = Some(scan);
            break;
        } else {
            scan = n;
        }
    }
    let end = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let body = &input[start..end];
    let Some(first) = body.chars().next() else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    if !starts_name(first) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    let mut n = first.len_utf8();
    while n < body.len() {
        let c = body[n..].chars().next().unwrap();
        if continues_name(c) { n += c.len_utf8(); } else { break; }
    }
    let name = &body[..n];
    let value = env_value(env, name).unwrap_or("");
    let chosen;
    if n == body.len() {
        chosen = value;
    } else if body[n..].starts_with(":-") {
        let default = &body[n + 2..];
        if value.is_empty() {
            let mut decoded = String::new();
            let mut p = 0;
            while p < default.len() {
                let c = default[p..].chars().next().unwrap();
                let next = p + c.len_utf8();
                if c == '\\' && next < default.len() {
                    let escaped = default[next..].chars().next().unwrap();
                    decoded.push(escaped);
                    p = next + escaped.len_utf8();
                } else {
                    decoded.push(c);
                    p = next;
                }
            }
            chosen = decoded;
            return Ok((end + 1, chosen));
        } else {
            chosen = value;
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    let _ = glob_active;
    Ok((end + 1, chosen.to_string()))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|b| *b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut current = Word { text: String::new(), glob: Vec::new() };
    let mut have_word = false;
    let mut quote = Quote::None;
    let mut opener = 0;
    let mut pos = 0;

    while pos < input.len() {
        let ch = input[pos..].chars().next().unwrap();
        let npos = pos + ch.len_utf8();
        match quote {
            Quote::Single => {
                if ch == '\'' {
                    quote = Quote::None;
                } else {
                    current.text.push(ch);
                }
                pos = npos;
            }
            Quote::None => {
                if separated(ch) {
                    if have_word {
                        result.push(current);
                        current = Word { text: String::new(), glob: Vec::new() };
                        have_word = false;
                    }
                    pos = npos;
                } else if ch == '\'' || ch == '"' {
                    have_word = true;
                    opener = pos;
                    quote = if ch == '\'' { Quote::Single } else { Quote::Double };
                    pos = npos;
                } else if ch == '\\' {
                    if npos == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let escaped = input[npos..].chars().next().unwrap();
                    let after = npos + escaped.len_utf8();
                    if escaped != '\n' {
                        have_word = true;
                        append(&mut current, &escaped.to_string(), false);
                    }
                    pos = after;
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if ch == '$' && input[npos..].starts_with('(') {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if ch == '$' {
                    have_word = true;
                    let (end, value) = expansion(input, pos, env, true)?;
                    append(&mut current, &value, true);
                    pos = end;
                } else {
                    have_word = true;
                    append(&mut current, &ch.to_string(), true);
                    pos = npos;
                }
            }
            Quote::Double => {
                if ch == '"' {
                    quote = Quote::None;
                    pos = npos;
                } else if ch == '\\' {
                    if npos == input.len() {
                        return Err(SplitError::TrailingEscape { pos });
                    }
                    let escaped = input[npos..].chars().next().unwrap();
                    if matches!(escaped, '"' | '\\' | '$' | '\n') {
                        if escaped != '\n' {
                            current.text.push(escaped);
                        }
                        pos = npos + escaped.len_utf8();
                    } else {
                        current.text.push('\\');
                        pos = npos;
                    }
                } else if ch == '`' {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if ch == '$' && input[npos..].starts_with('(') {
                    return Err(SplitError::CommandSubstitution { pos });
                } else if ch == '$' {
                    have_word = true;
                    let (end, value) = expansion(input, pos, env, false)?;
                    append(&mut current, &value, false);
                    pos = end;
                } else {
                    have_word = true;
                    current.text.push(ch);
                    pos = npos;
                }
            }
        }
    }

    match quote {
        Quote::Single => return Err(SplitError::UnterminatedSingle { pos: opener }),
        Quote::Double => return Err(SplitError::UnterminatedDouble { pos: opener }),
        Quote::None => {}
    }
    if have_word {
        result.push(current);
    }
    Ok(result)
}
```
