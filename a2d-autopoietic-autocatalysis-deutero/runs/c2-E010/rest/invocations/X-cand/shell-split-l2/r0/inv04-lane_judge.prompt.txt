pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

fn name_first(byte: u8) -> bool {
    byte == b'_' || byte.is_ascii_alphabetic()
}

fn name_byte(byte: u8) -> bool {
    name_first(byte) || byte.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> &'a str {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
        .unwrap_or("")
}

fn append_decoded_default(out: &mut String, source: &str) {
    let mut iter = source.chars();
    while let Some(c) = iter.next() {
        if c == '\\' {
            if let Some(escaped) = iter.next() {
                out.push(escaped);
            }
        } else {
            out.push(c);
        }
    }
}

fn expand(input: &str, pos: usize, env: &[(&str, &str)]) -> Result<(usize, Option<String>, bool), SplitError> {
    let bytes = input.as_bytes();
    let start = pos + 1;

    if start == bytes.len() {
        return Ok((start, None, false));
    }
    if bytes[start] == b'{' {
        let body = start + 1;
        let mut cursor = body;
        while cursor < bytes.len() {
            if bytes[cursor] == b'\\' {
                cursor += 1;
                if cursor < bytes.len() {
                    cursor += input[cursor..].chars().next().unwrap().len_utf8();
                }
            } else if bytes[cursor] == b'}' {
                break;
            } else {
                cursor += input[cursor..].chars().next().unwrap().len_utf8();
            }
        }
        if cursor == bytes.len() {
            return Err(SplitError::UnterminatedExpansion { pos });
        }

        let content = &input[body..cursor];
        let cb = content.as_bytes();
        if cb.is_empty() || !name_first(cb[0]) {
            return Err(SplitError::InvalidExpansion { pos });
        }
        let mut k = 1;
        while k < cb.len() && name_byte(cb[k]) {
            k += 1;
        }
        let name = &content[..k];
        let value = if k == cb.len() {
            lookup(env, name).to_owned()
        } else if cb[k..].starts_with(b":-") {
            match env.iter().find(|(key, _)| *key == name).map(|(_, v)| *v) {
                Some(v) if !v.is_empty() => v.to_owned(),
                _ => {
                    let mut out = String::new();
                    append_decoded_default(&mut out, &content[k + 2..]);
                    out
                }
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos });
        };
        return Ok((cursor + 1, Some(value), true));
    }

    if !name_first(bytes[start]) {
        return Ok((start, None, false));
    }
    let mut end = start + 1;
    while end < bytes.len() && name_byte(bytes[end]) {
        end += 1;
    }
    Ok((end, Some(lookup(env, &input[start..end]).to_owned()), true))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|b| *b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let bytes = input.as_bytes();
    let mut out = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut i = 0;

    while i < bytes.len() {
        let c = input[i..].chars().next().unwrap();
        match c {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    out.push(std::mem::take(&mut word));
                    active = false;
                }
                i += 1;
            }
            '\'' => {
                active = true;
                let opening = i;
                i += 1;
                loop {
                    if i >= bytes.len() {
                        return Err(SplitError::UnterminatedSingle { pos: opening });
                    }
                    let q = input[i..].chars().next().unwrap();
                    i += q.len_utf8();
                    if q == '\'' {
                        break;
                    }
                    word.push(q);
                }
            }
            '"' => {
                active = true;
                let opening = i;
                i += 1;
                let mut closed = false;
                while i < bytes.len() {
                    let q = input[i..].chars().next().unwrap();
                    if q == '"' {
                        i += 1;
                        closed = true;
                        break;
                    } else if q == '\\' {
                        let slash = i;
                        i += 1;
                        if i >= bytes.len() {
                            return Err(SplitError::TrailingEscape { pos: slash });
                        }
                        let next = input[i..].chars().next().unwrap();
                        if matches!(next, '"' | '\\' | '$') {
                            word.push(next);
                            i += next.len_utf8();
                        } else if next == '\n' {
                            i += 1;
                        } else {
                            word.push('\\');
                            word.push(next);
                            i += next.len_utf8();
                        }
                    } else if q == '$' {
                        let (end, replacement, makes_word) = expand(input, i, env)?;
                        if let Some(replacement) = replacement {
                            word.push_str(&replacement);
                        } else {
                            word.push('$');
                        }
                        active |= makes_word;
                        i = end;
                    } else {
                        word.push(q);
                        i += q.len_utf8();
                    }
                }
                if !closed {
                    return Err(SplitError::UnterminatedDouble { pos: opening });
                }
            }
            '\\' => {
                let slash = i;
                i += 1;
                if i == bytes.len() {
                    return Err(SplitError::TrailingEscape { pos: slash });
                }
                let next = input[i..].chars().next().unwrap();
                if next != '\n' {
                    word.push(next);
                    active = true;
                }
                i += next.len_utf8();
            }
            '$' => {
                let (end, replacement, makes_word) = expand(input, i, env)?;
                if let Some(replacement) = replacement {
                    word.push_str(&replacement);
                } else {
                    word.push('$');
                }
                active |= makes_word;
                i = end;
            }
            _ => {
                word.push(c);
                active = true;
                i += c.len_utf8();
            }
        }
    }

    if active {
        out.push(word);
    }
    Ok(out)
}

pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Mode {
    Bare,
    Single,
    Double,
}

fn sep(c: char) -> bool {
    c == ' ' || c == '\t' || c == '\r' || c == '\n'
}

fn start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn cont(c: char) -> bool {
    start(c) || c.is_ascii_digit()
}

fn value<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    for (key, value) in env {
        if *key == name {
            return Some(value);
        }
    }
    None
}

fn default_value(text: &str) -> String {
    let mut out = String::new();
    let mut it = text.chars();
    while let Some(c) = it.next() {
        if c == '\\' {
            match it.next() {
                Some(next) => out.push(next),
                None => out.push('\\'),
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn expansion(
    input: &str,
    pos: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let begin = pos + 1;
    if input[begin..].starts_with('{') {
        let body = begin + 1;
        let mut i = body;
        let mut end = None;

        while i < input.len() {
            let c = input[i..].chars().next().unwrap();
            if c == '\\' {
                i += 1;
                if i < input.len() {
                    i += input[i..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                end = Some(i);
                break;
            } else {
                i += c.len_utf8();
            }
        }

        let end = end.ok_or(SplitError::UnterminatedExpansion { pos })?;
        let body_text = &input[body..end];
        let mut n = 0;
        for c in body_text.chars() {
            if cont(c) {
                n += c.len_utf8();
            } else {
                break;
            }
        }

        let name = &body_text[..n];
        if name.is_empty() || !name.chars().next().map(start).unwrap_or(false) {
            return Err(SplitError::InvalidExpansion { pos });
        }

        let rest = &body_text[n..];
        if rest.is_empty() {
            Ok((value(name, env).unwrap_or("").to_string(), end + 1))
        } else if let Some(text) = rest.strip_prefix(":-") {
            if let Some(found) = value(name, env).filter(|v| !v.is_empty()) {
                Ok((found.to_string(), end + 1))
            } else {
                Ok((default_value(text), end + 1))
            }
        } else {
            Err(SplitError::InvalidExpansion { pos })
        }
    } else {
        let mut iter = input[begin..].char_indices();
        let Some((_, first)) = iter.next() else {
            return Ok(("$".to_string(), begin));
        };
        if !start(first) {
            return Ok(("$".to_string(), begin));
        }
        let mut finish = begin + first.len_utf8();
        for (offset, c) in iter {
            if cont(c) {
                finish = begin + offset + c.len_utf8();
            } else {
                break;
            }
        }
        Ok((value(&input[begin..finish], env).unwrap_or("").to_string(), finish))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut output = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut mode = Mode::Bare;
    let mut opening = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let after = i + c.len_utf8();

        match mode {
            Mode::Single => {
                if c == '\'' {
                    mode = Mode::Bare;
                } else {
                    current.push(c);
                }
                i = after;
            }
            Mode::Double => {
                if c == '"' {
                    mode = Mode::Bare;
                    i = after;
                } else if c == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let d = input[after..].chars().next().unwrap();
                    if d == '"' || d == '\\' || d == '$' {
                        current.push(d);
                        i = after + d.len_utf8();
                    } else if d == '\n' {
                        i = after + 1;
                    } else {
                        current.push('\\');
                        i = after;
                    }
                } else if c == '$' {
                    let (expanded, next) = expansion(input, i, env)?;
                    current.push_str(&expanded);
                    active = true;
                    i = next;
                } else {
                    current.push(c);
                    i = after;
                }
            }
            Mode::Bare => {
                if sep(c) {
                    if active {
                        output.push(std::mem::take(&mut current));
                        active = false;
                    }
                    i = after;
                } else if c == '\'' || c == '"' {
                    opening = i;
                    mode = if c == '\'' { Mode::Single } else { Mode::Double };
                    active = true;
                    i = after;
                } else if c == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let d = input[after..].chars().next().unwrap();
                    if d != '\n' {
                        current.push(d);
                        active = true;
                    }
                    i = after + d.len_utf8();
                } else if c == '$' {
                    let (expanded, next) = expansion(input, i, env)?;
                    current.push_str(&expanded);
                    active = true;
                    i = next;
                } else {
                    current.push(c);
                    active = true;
                    i = after;
                }
            }
        }
    }

    if mode == Mode::Single {
        return Err(SplitError::UnterminatedSingle { pos: opening });
    }
    if mode == Mode::Double {
        return Err(SplitError::UnterminatedDouble { pos: opening });
    }
    if active {
        output.push(current);
    }
    Ok(output)
}

    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    let bytes = input.as_bytes();
    if let Some(pos) = bytes.iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b' ' | b'\t' | b'\r' | b'\n' => {
                if active {
                    words.push(std::mem::take(&mut word));
                    active = false;
                }
                i += 1;
            }
            b'\'' => {
                active = true;
                let opening = i;
                i += 1;
                while i < bytes.len() && bytes[i] != b'\'' {
                    let ch = input[i..].chars().next().unwrap();
                    word.push(ch);
                    i += ch.len_utf8();
                }
                if i == bytes.len() {
                    return Err(SplitError::UnterminatedSingle { pos: opening });
                }
                i += 1;
            }
            b'"' => {
                active = true;
                let opening = i;
                i += 1;
                loop {
                    if i == bytes.len() {
                        return Err(SplitError::UnterminatedDouble { pos: opening });
                    }
                    match bytes[i] {
                        b'"' => {
                            i += 1;
                            break;
                        }
                        b'\\' => {
                            let slash = i;
                            i += 1;
                            if i == bytes.len() {
                                return Err(SplitError::TrailingEscape { pos: slash });
                            }
                            match bytes[i] {
                                b'"' | b'\\' | b'$' => {
                                    word.push(bytes[i] as char);
                                    i += 1;
                                }

                                _ => {
                                    word.push('\\');
                                    let ch = input[i..].chars().next().unwrap();
                                    word.push(ch);
                                    i += ch.len_utf8();
                                }
                            }
                        }
                        b'$' => {
                            if let Some((value, end)) = parse_expansion(input, i, env)? {
                                word.push_str(value);
                                active = true;
                                i = end;
                            } else {
                                word.push('$');
                                i += 1;
                            }
                        }
                        _ => {
                            let ch = input[i..].chars().next().unwrap();
                            word.push(ch);
                            i += ch.len_utf8();
                        }
                    }
                }
            }
            b'\\' => {
                let slash = i;
                i += 1;
                if i == bytes.len() {
                    return Err(SplitError::TrailingEscape { pos: slash });
                }
                if bytes[i] == b'\n' {
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    word.push(ch);
                    active = true;
                    i += ch.len_utf8();
                }
            }
            b'$' => {
                if let Some((value, end)) = parse_expansion(input, i, env)? {
                    word.push_str(value);
                    active = true;
                    i = end;
                } else {
                    word.push('$');
                    active = true;
                    i += 1;
                }
            }
            _ => {
                let ch = input[i..].chars().next().unwrap();
                word.push(ch);
                active = true;
                i += ch.len_utf8();
            }
        }
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn is_name_start(b: u8) -> bool {
    b == b'_' || b.is_ascii_alphabetic()
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> &'a str {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
        .unwrap_or("")
}

fn parse_expansion<'a>(
    input: &str,
    start: usize,
    env: &'a [(&str, &str)],
) -> Result<Option<(&'a str, usize)>, SplitError> {
    let bytes = input.as_bytes();
    if start + 1 >= bytes.len() {
        return Ok(None);
    }

    if bytes[start + 1] == b'{' {
        let mut close = start + 2;
        while close < bytes.len() {
            if bytes[close] == b'\\' {
                close += 1;
                if close < bytes.len() {
                    let ch = input[close..].chars().next().unwrap();
                    close += ch.len_utf8();
                }
            } else if bytes[close] == b'}' {
                break;
            } else {
                let ch = input[close..].chars().next().unwrap();
                close += ch.len_utf8();
            }
        }
        if close == bytes.len() {
            return Err(SplitError::UnterminatedExpansion { pos: start });
        }

        let body = &input[start + 2..close];
        let name_end = body
            .as_bytes()
            .iter()
            .position(|b| !is_name_continue(*b))
            .unwrap_or(body.len());
        let name = &body[..name_end];
        if name.is_empty() || !is_name_start(body.as_bytes()[0]) {
            return Err(SplitError::InvalidExpansion { pos: start });
        }

        let rest = &body[name_end..];
        let value = if rest.is_empty() {
            lookup(name, env)
        } else if let Some(default) = rest.strip_prefix(":-") {
            let found = env.iter().find_map(|(key, value)| {
                if *key == name && !value.is_empty() { Some(*value) } else { None }
            });
            if let Some(value) = found {
                value
            } else {
                return Ok(Some((Box::leak(decode_default(default).into_boxed_str()), close + 1)));
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: start });
        };
        Ok(Some((value, close + 1)))
    } else if is_name_start(bytes[start + 1]) {
        let mut end = start + 2;
        while end < bytes.len() && is_name_continue(bytes[end]) {
            end += 1;
        }
        let name = &input[start + 1..end];
        Ok(Some((lookup(name, env), end)))
    } else {
        Ok(None)
    }
}

fn decode_default(text: &str) -> String {
    let bytes = text.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'\\' {
            i += 1;
            if i < bytes.len() {
                let ch = text[i..].chars().next().unwrap();
                out.push(ch);
                i += ch.len_utf8();
            }
        } else {
            let ch = text[i..].chars().next().unwrap();
            out.push(ch);
            i += ch.len_utf8();
        }
    }
    out
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

fn start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn cont(c: char) -> bool {
    c == '_' || c.is_ascii_alphanumeric()
}

fn value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    for (key, val) in env {
        if *key == name {
            return Some(*val);
        }
    }
    None
}

fn decode(s: &str) -> String {
    let mut result = String::new();
    let mut it = s.chars();
    while let Some(c) = it.next() {
        if c == '\\' {
            match it.next() {
                Some(next) => result.push(next),
                None => result.push('\\'),
            }
        } else {
            result.push(c);
        }
    }
    result
}

fn parse_dollar(
    s: &str,
    at: usize,
    env: &[(&str, &str)],
) -> Result<(usize, String), SplitError> {
    let p = at + 1;
    if p < s.len() && s.as_bytes()[p] == b'{' {
        let begin = p + 1;
        let mut k = begin;
        let mut end = None;
        while k < s.len() {
            let c = s[k..].chars().next().unwrap();
            match c {
                '\\' => {
                    k += 1;
                    if k < s.len() {
                        k += s[k..].chars().next().unwrap().len_utf8();
                    }
                }
                '}' => {
                    end = Some(k);
                    break;
                }
                _ => k += c.len_utf8(),
            }
        }
        let end = end.ok_or(SplitError::UnterminatedExpansion { pos: at })?;
        let body = &s[begin..end];

        let mut j = 0;
        let mut chars = body.char_indices();
        let first = chars.next().map(|(_, c)| c)
            .filter(|c| start(*c))
            .ok_or(SplitError::InvalidExpansion { pos: at })?;
        j += first.len_utf8();
        for (offset, c) in chars {
            if cont(c) {
                j = offset + c.len_utf8();
            } else {
                break;
            }
        }

        let name = &body[..j];
        let tail = &body[j..];
        let out = if tail.is_empty() {
            value(env, name).unwrap_or("").to_string()
        } else if let Some(default) = tail.strip_prefix(":-") {
            match value(env, name) {
                Some(v) if !v.is_empty() => v.to_string(),
                _ => decode(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: at });
        };
        Ok((end + 1, out))
    } else {
        let mut j = p;
        if j < s.len() {
            let c = s[j..].chars().next().unwrap();
            if start(c) {
                j += c.len_utf8();
                while j < s.len() {
                    let next = s[j..].chars().next().unwrap();
                    if !cont(next) {
                        break;
                    }
                    j += next.len_utf8();
                }
                return Ok((j, value(env, &s[p..j]).unwrap_or("").to_string()));
            }
        }
        Ok((p, "$".to_string()))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|b| *b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut result = Vec::new();
    let mut current = String::new();
    let mut started = false;
    let mut state = 0u8; // 0 bare, 1 single, 2 double
    let mut quote_pos = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let width = c.len_utf8();

        match state {
            1 => {
                if c == '\'' {
                    state = 0;
                } else {
                    current.push(c);
                }
                i += width;
            }
            2 => match c {
                '"' => {
                    state = 0;
                    i += width;
                }
                '\\' => {
                    let escape_pos = i;
                    i += 1;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next == '"' || next == '\\' || next == '$' {
                        current.push(next);
                        i += next.len_utf8();
                    } else if next == '\n' {
                        i += 1;
                    } else {
                        current.push('\\');
                    }
                }
                '$' => {
                    let (next, expanded) = parse_dollar(input, i, env)?;
                    current.push_str(&expanded);
                    started = true;
                    i = next;
                }
                _ => {
                    current.push(c);
                    i += width;
                }
            },
            _ => match c {
                ' ' | '\t' | '\r' | '\n' => {
                    if started {
                        result.push(std::mem::take(&mut current));
                        started = false;
                    }
                    i += width;
                }
                '\'' | '"' => {
                    state = if c == '\'' { 1 } else { 2 };
                    quote_pos = i;
                    started = true;
                    i += width;
                }
                '\\' => {
                    let escape_pos = i;
                    i += 1;
                    if i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: escape_pos });
                    }
                    let next = input[i..].chars().next().unwrap();
                    if next != '\n' {
                        current.push(next);
                        started = true;
                    }
                    i += next.len_utf8();
                }
                '$' => {
                    let (next, expanded) = parse_dollar(input, i, env)?;
                    current.push_str(&expanded);
                    started = true;
                    i = next;
                }
                _ => {
                    current.push(c);
                    started = true;
                    i += width;
                }
            },
        }
    }

    if state == 1 {
        return Err(SplitError::UnterminatedSingle { pos: quote_pos });
    }
    if state == 2 {
        return Err(SplitError::UnterminatedDouble { pos: quote_pos });
    }
    if started {
        result.push(current);
    }
    Ok(result)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_dollar = dollar + 1;
    let rest = &input[after_dollar..];

    if rest.starts_with('{') {
        let content_start = after_dollar + 1;
        let mut close = None;
        let mut cursor = content_start;
        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    cursor += input[cursor..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let content = &input[content_start..close];

        let name_end = content
            .char_indices()
            .find_map(|(i, c)| if !is_name_continue(c) { Some(i) } else { None })
            .unwrap_or(content.len());

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &content[..name_end];
        let mut name_chars = name.chars();
        if !name_chars.next().map(is_name_start).unwrap_or(false)
            || !name_chars.all(is_name_continue)
        {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let tail = &content[name_end..];
        if tail.is_empty() {
            let value = lookup(name, env).unwrap_or("");
            Ok((value.to_string(), close + 1))
        } else if let Some(default) = tail.strip_prefix(":-") {
            if let Some(value) = lookup(name, env).filter(|v| !v.is_empty()) {
                Ok((value.to_string(), close + 1))
            } else {
                Ok((decode_default(default), close + 1))
            }
        } else {
            Err(SplitError::InvalidExpansion { pos: dollar })
        }
    } else {
        let mut chars = rest.char_indices();
        let Some((_, first)) = chars.next() else {
            return Ok(("$".to_string(), after_dollar));
        };
        if !is_name_start(first) {
            return Ok(("$".to_string(), after_dollar));
        }

        let mut end = after_dollar + first.len_utf8();
        for (offset, c) in chars {
            if is_name_continue(c) {
                end = after_dollar + offset + c.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        Ok((lookup(name, env).unwrap_or("").to_string(), end))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut cursor = 0;

    while cursor < input.len() {
        let c = input[cursor..].chars().next().unwrap();
        let next = cursor + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                cursor = next;
            }
            Quote::Double(_) => {
                if c == '"' {
                    quote = Quote::None;
                    cursor = next;
                } else if c == '\\' {
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next..].chars().next().unwrap();
                    let after = next + escaped.len_utf8();
                    if matches!(escaped, '"' | '\\' | '$') {
                        word.push(escaped);
                        cursor = after;
                    } else if escaped == '\n' {
                        cursor = after;
                    } else {
                        word.push('\\');
                        cursor = next;
                    }
                } else if c == '$' {
                    let (value, after) = parse_expansion(input, cursor, env)?;
                    word.push_str(&value);
                    active = true;
                    cursor = after;
                } else {
                    word.push(c);
                    cursor = next;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next;
                } else if c == '\'' {
                    quote = Quote::Single(cursor);
                    active = true;
                    cursor = next;
                } else if c == '"' {
                    quote = Quote::Double(cursor);
                    active = true;
                    cursor = next;
                } else if c == '\\' {
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next..].chars().next().unwrap();
                    let after = next + escaped.len_utf8();
                    if escaped == '\n' {
                        cursor = after;
                    } else {
                        word.push(escaped);
                        active = true;
                        cursor = after;
                    }
                } else if c == '$' {
                    let (value, after) = parse_expansion(input, cursor, env)?;
                    word.push_str(&value);
                    active = true;
                    cursor = after;
                } else {
                    word.push(c);
                    active = true;
                    cursor = next;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
