pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {

        return Err(SplitError::Nul { pos });

                    words.push(std::mem::take(&mut word));
                    active = false;

        return Ok(None);

            end += 1;

        Ok(Some((lookup(name, env), end)))

        Ok(None)
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    let bytes = input.as_bytes();
    if let Some(pos) = bytes.iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut word_active = false;
    let mut i = 0;
    let mut quote = 0u8; // 0 = unquoted, 1 = single, 2 = double
    let mut quote_start = 0;

    while i < bytes.len() {
        if quote == 1 {
            if bytes[i] == b'\'' {
                quote = 0;
                i += 1;
            } else {
                let ch = input[i..].chars().next().unwrap();
                current.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if quote == 2 {
            match bytes[i] {
                b'"' => {
                    quote = 0;
                    i += 1;
                }
                b'\\' => {
                    let slash = i;
                    i += 1;
                    if i == bytes.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    match bytes[i] {
                        b'"' | b'\\' | b'$' => {
                            current.push(bytes[i] as char);
                            i += 1;
                        }
                        b'\n' => i += 1,
                        _ => {
                            current.push('\\');
                            let ch = input[i..].chars().next().unwrap();
                            current.push(ch);
                            i += ch.len_utf8();
                        }
                    }
                }
                b'$' => {
                    if let Some((value, end)) = expansion(input, i, env)? {
                        current.push_str(value);
                        word_active = true;
                        i = end;
                    } else {
                        current.push('$');
                        i += 1;
                    }
                }
                _ => {
                    let ch = input[i..].chars().next().unwrap();
                    current.push(ch);
                    i += ch.len_utf8();
                }
            }
            continue;
        }

        match bytes[i] {
            b' ' | b'\t' | b'\r' | b'\n' => {
                if word_active {
                    words.push(std::mem::take(&mut current));
                    word_active = false;
                }
                i += 1;
            }
            b'\'' | b'"' => {
                quote = if bytes[i] == b'\'' { 1 } else { 2 };
                quote_start = i;
                word_active = true;
                i += 1;
            }
            b'\\' => {
                let slash = i;
                i += 1;
                if i == bytes.len() {
                    return Err(SplitError::TrailingEscape { pos: slash });
                }
                if bytes[i] == b'\n' {
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    current.push(ch);
                    word_active = true;
                    i += ch.len_utf8();
                }
            }
            b'$' => {
                if let Some((value, end)) = expansion(input, i, env)? {
                    current.push_str(value);
                    word_active = true;
                    i = end;
                } else {
                    current.push('$');
                    word_active = true;
                    i += 1;
                }
            }
            _ => {
                let ch = input[i..].chars().next().unwrap();
                current.push(ch);
                word_active = true;
                i += ch.len_utf8();
            }
        }
    }

    if quote == 1 {
        return Err(SplitError::UnterminatedSingle { pos: quote_start });
    }
    if quote == 2 {
        return Err(SplitError::UnterminatedDouble { pos: quote_start });
    }
    if word_active {
        words.push(current);
    }
    Ok(words)
}

fn name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn name_continue(b: u8) -> bool {
    name_start(b) || b.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find_map(|(key, value)| {
        if *key == name {
            Some(*value)
        } else {
            None
        }
    })
}

fn expansion<'a>(
    input: &str,
    dollar: usize,
    env: &'a [(&str, &str)],
) -> Result<Option<(String, usize)>, SplitError> {
    let bytes = input.as_bytes();
    if dollar + 1 >= bytes.len() {
        return Ok(None);
    }

    if bytes[dollar + 1] != b'{' {
        if !name_start(bytes[dollar + 1]) {
            return Ok(None);
        }
        let mut end = dollar + 2;
        while end < bytes.len() && name_continue(bytes[end]) {
            end += 1;
        }
        let name = &input[dollar + 1..end];
        return Ok(Some((lookup(name, env).unwrap_or("").to_owned(), end)));
    }

    let mut close = dollar + 2;
    while close < bytes.len() {
        if bytes[close] == b'\\' {
            close += 1;
            if close < bytes.len() {
                let ch = input[close..].chars().next().unwrap();
                close += ch.len_utf8();
            }
        } else if bytes[close] == b'}' {
            break;
        } else {
            let ch = input[close..].chars().next().unwrap();
            close += ch.len_utf8();
        }
    }
    if close == bytes.len() {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    }

    let body = &input[dollar + 2..close];
    let body_bytes = body.as_bytes();
    let mut name_end = 0;
    if body_bytes.first().copied().map(name_start) != Some(true) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    name_end = 1;
    while name_end < body_bytes.len() && name_continue(body_bytes[name_end]) {
        name_end += 1;
    }

    let name = &body[..name_end];
    let rest = &body[name_end..];
    let value = if rest.is_empty() {
        lookup(name, env).unwrap_or("").to_owned()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match lookup(name, env) {
            Some(v) if !v.is_empty() => v.to_owned(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    Ok(Some((value, close + 1)))
}

fn decode_default(text: &str) -> String {
    let mut chars = text.chars();
    let mut out = String::new();
    while let Some(ch) = chars.next() {
        if ch == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(ch);
        }
    }
    out
}
