// A None value denotes a literal dollar.
fn parse_expansion<'a>(
    input: &str,
    dollar: usize,
    env: &'a [(&str, &str)],
) -> Result<(usize, Option<String>, bool), SplitError> {
    let bytes = input.as_bytes();
    let after_dollar = dollar + 1;
    if after_dollar >= bytes.len() {
        return Ok((after_dollar, None, false));
    }

    if bytes[after_dollar] == b'{' {
        let content_start = after_dollar + 1;
        let mut j = content_start;
        let mut close = None;
        while j < bytes.len() {
            let c = input[j..].chars().next().unwrap();
            if c == '\\' {
                j += 1;
                if j < bytes.len() {
                    let next = input[j..].chars().next().unwrap();
                    j += next.len_utf8();
                }
            } else if c == '}' {
                close = Some(j);
                break;
            } else {
                j += c.len_utf8();
            }
        }
        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let name_end = contents
            .char_indices()
            .find_map(|(idx, c)| (!is_name_continue(c)).then_some(idx))
            .unwrap_or(contents.len());
        let name = &contents[..name_end];
        if name.is_empty()
            || !name.chars().next().map(is_name_start).unwrap_or(false)
        {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let rest = &contents[name_end..];
        let value = if rest.is_empty() {
            env_value(env, name).unwrap_or("").to_owned()
        } else if let Some(default) = rest.strip_prefix(":-") {
            match env_value(env, name) {
                Some(v) if !v.is_empty() => v.to_owned(),
                _ => decode_default(default),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };
        return Ok((close + 1, Some(value), true));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if !is_name_start(first) {
        return Ok((after_dollar, None, false));
    }

    let mut end = after_dollar + first.len_utf8();
    while end < bytes.len() {
        let c = input[end..].chars().next().unwrap();
        if !is_name_continue(c) {
            break;
        }
        end += c.len_utf8();
    }
    let name = &input[after_dollar..end];
    Ok((
        end,
        Some(env_value(env, name).unwrap_or("").to_owned()),
        true,
    ))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

fn name_first(byte: u8) -> bool {
    byte == b'_' || byte.is_ascii_alphabetic()
}

fn name_byte(byte: u8) -> bool {
    name_first(byte) || byte.is_ascii_digit()
}

fn lookup<'a>(env: &'a [(&str, &str)], name: &str) -> &'a str {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
        .unwrap_or("")
}

fn append_decoded_default(out: &mut String, source: &str) {
    let mut iter = source.chars();
    while let Some(c) = iter.next() {
        if c == '\\' {
            if let Some(escaped) = iter.next() {
                out.push(escaped);
            }
        } else {
            out.push(c);
        }
    }
}

fn expand(input: &str, pos: usize, env: &[(&str, &str)]) -> Result<(usize, Option<String>, bool), SplitError> {
    let bytes = input.as_bytes();
    let start = pos + 1;

    if start == bytes.len() {
        return Ok((start, None, false));
    }
    if bytes[start] == b'{' {
        let body = start + 1;
        let mut cursor = body;
        while cursor < bytes.len() {
            if bytes[cursor] == b'\\' {
                cursor += 1;
                if cursor < bytes.len() {
                    cursor += input[cursor..].chars().next().unwrap().len_utf8();
                }
            } else if bytes[cursor] == b'}' {
                break;
            } else {
                cursor += input[cursor..].chars().next().unwrap().len_utf8();
            }
        }
        if cursor == bytes.len() {
            return Err(SplitError::UnterminatedExpansion { pos });
        }

        let content = &input[body..cursor];
        let cb = content.as_bytes();
        if cb.is_empty() || !name_first(cb[0]) {
            return Err(SplitError::InvalidExpansion { pos });
        }
        let mut k = 1;
        while k < cb.len() && name_byte(cb[k]) {
            k += 1;
        }
        let name = &content[..k];
        let value = if k == cb.len() {
            lookup(env, name).to_owned()
        } else if cb[k..].starts_with(b":-") {
            match env.iter().find(|(key, _)| *key == name).map(|(_, v)| *v) {
                Some(v) if !v.is_empty() => v.to_owned(),
                _ => {
                    let mut out = String::new();
                    append_decoded_default(&mut out, &content[k + 2..]);
                    out
                }
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos });
        };
        return Ok((cursor + 1, Some(value), true));
    }

    if !name_first(bytes[start]) {
        return Ok((start, None, false));
    }
    let mut end = start + 1;
    while end < bytes.len() && name_byte(bytes[end]) {
        end += 1;
    }
    Ok((end, Some(lookup(env, &input[start..end]).to_owned()), true))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|b| *b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let bytes = input.as_bytes();
    let mut out = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut i = 0;

    while i < bytes.len() {
        let c = input[i..].chars().next().unwrap();
        match c {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    out.push(std::mem::take(&mut word));
                    active = false;
                }
                i += 1;
            }
            '\'' => {
                active = true;
                let opening = i;
                i += 1;
                loop {
                    if i >= bytes.len() {
                        return Err(SplitError::UnterminatedSingle { pos: opening });
                    }
                    let q = input[i..].chars().next().unwrap();
                    i += q.len_utf8();
                    if q == '\'' {
                        break;
                    }
                    word.push(q);
                }
            }
            '"' => {
                active = true;
                let opening = i;
                i += 1;
                let mut closed = false;
                while i < bytes.len() {
                    let q = input[i..].chars().next().unwrap();
                    if q == '"' {
                        i += 1;
                        closed = true;
                        break;
                    } else if q == '\\' {
                        let slash = i;
                        i += 1;
                        if i >= bytes.len() {
                            return Err(SplitError::TrailingEscape { pos: slash });
                        }
                        let next = input[i..].chars().next().unwrap();
                        if matches!(next, '"' | '\\' | '$') {
                            word.push(next);
                            i += next.len_utf8();
                        } else if next == '\n' {
                            i += 1;
                        } else {
                            word.push('\\');
                            word.push(next);
                            i += next.len_utf8();
                        }
                    } else if q == '$' {
                        let (end, replacement, makes_word) = expand(input, i, env)?;
                        if let Some(replacement) = replacement {
                            word.push_str(&replacement);
                        } else {
                            word.push('$');
                        }
                        active |= makes_word;
                        i = end;
                    } else {
                        word.push(q);
                        i += q.len_utf8();
                    }
                }
                if !closed {
                    return Err(SplitError::UnterminatedDouble { pos: opening });
                }
            }
            '\\' => {
                let slash = i;
                i += 1;
                if i == bytes.len() {
                    return Err(SplitError::TrailingEscape { pos: slash });
                }
                let next = input[i..].chars().next().unwrap();
                if next != '\n' {
                    word.push(next);
                    active = true;
                }
                i += next.len_utf8();
            }

                let (end, replacement, makes_word) = expand(input, i, env)?;
                if let Some(replacement) = replacement {
                    word.push_str(&replacement);
                } else {
                    word.push('$');
                }
                active |= makes_word;
                i = end;
            }
            _ => {
                word.push(c);
                active = true;
                i += c.len_utf8();
            }
        }
    }

    if active {
        out.push(word);
    }
    Ok(out)
}
