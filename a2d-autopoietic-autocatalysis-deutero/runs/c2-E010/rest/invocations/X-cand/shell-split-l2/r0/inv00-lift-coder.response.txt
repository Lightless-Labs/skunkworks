- NUL preflight scans the entire input first, so it outranks every syntax error.
- Braced expansion searches for its first unescaped `}` before validating the body; a missing brace is `UnterminatedExpansion`.
- A trailing backslash at EOF inside double quotes takes precedence over an unterminated-double-quote error.
- A backslash in default text decodes the following Unicode scalar; a final backslash before the closing brace is invalid because it escapes that brace.
- Empty quotes and expansions activate a word; a line continuation alone does not.
- Both solutions use byte offsets into the original input and preserve substituted values atomically.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    let bytes = input.as_bytes();
    if let Some(pos) = bytes.iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b' ' | b'\t' | b'\r' | b'\n' => {
                if active {
                    words.push(std::mem::take(&mut word));
                    active = false;
                }
                i += 1;
            }
            b'\'' => {
                active = true;
                let opening = i;
                i += 1;
                while i < bytes.len() && bytes[i] != b'\'' {
                    let ch = input[i..].chars().next().unwrap();
                    word.push(ch);
                    i += ch.len_utf8();
                }
                if i == bytes.len() {
                    return Err(SplitError::UnterminatedSingle { pos: opening });
                }
                i += 1;
            }
            b'"' => {
                active = true;
                let opening = i;
                i += 1;
                loop {
                    if i == bytes.len() {
                        return Err(SplitError::UnterminatedDouble { pos: opening });
                    }
                    match bytes[i] {
                        b'"' => {
                            i += 1;
                            break;
                        }
                        b'\\' => {
                            let slash = i;
                            i += 1;
                            if i == bytes.len() {
                                return Err(SplitError::TrailingEscape { pos: slash });
                            }
                            match bytes[i] {
                                b'"' | b'\\' | b'$' => {
                                    word.push(bytes[i] as char);
                                    i += 1;
                                }
                                b'\n' => i += 1,
                                _ => {
                                    word.push('\\');
                                    let ch = input[i..].chars().next().unwrap();
                                    word.push(ch);
                                    i += ch.len_utf8();
                                }
                            }
                        }
                        b'$' => {
                            if let Some((value, end)) = parse_expansion(input, i, env)? {
                                word.push_str(value);
                                active = true;
                                i = end;
                            } else {
                                word.push('$');
                                i += 1;
                            }
                        }
                        _ => {
                            let ch = input[i..].chars().next().unwrap();
                            word.push(ch);
                            i += ch.len_utf8();
                        }
                    }
                }
            }
            b'\\' => {
                let slash = i;
                i += 1;
                if i == bytes.len() {
                    return Err(SplitError::TrailingEscape { pos: slash });
                }
                if bytes[i] == b'\n' {
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    word.push(ch);
                    active = true;
                    i += ch.len_utf8();
                }
            }
            b'$' => {
                if let Some((value, end)) = parse_expansion(input, i, env)? {
                    word.push_str(value);
                    active = true;
                    i = end;
                } else {
                    word.push('$');
                    active = true;
                    i += 1;
                }
            }
            _ => {
                let ch = input[i..].chars().next().unwrap();
                word.push(ch);
                active = true;
                i += ch.len_utf8();
            }
        }
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn is_name_start(b: u8) -> bool {
    b == b'_' || b.is_ascii_alphabetic()
}

fn is_name_continue(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> &'a str {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
        .unwrap_or("")
}

fn parse_expansion<'a>(
    input: &str,
    start: usize,
    env: &'a [(&str, &str)],
) -> Result<Option<(&'a str, usize)>, SplitError> {
    let bytes = input.as_bytes();
    if start + 1 >= bytes.len() {
        return Ok(None);
    }

    if bytes[start + 1] == b'{' {
        let mut close = start + 2;
        while close < bytes.len() {
            if bytes[close] == b'\\' {
                close += 1;
                if close < bytes.len() {
                    let ch = input[close..].chars().next().unwrap();
                    close += ch.len_utf8();
                }
            } else if bytes[close] == b'}' {
                break;
            } else {
                let ch = input[close..].chars().next().unwrap();
                close += ch.len_utf8();
            }
        }
        if close == bytes.len() {
            return Err(SplitError::UnterminatedExpansion { pos: start });
        }

        let body = &input[start + 2..close];
        let name_end = body
            .as_bytes()
            .iter()
            .position(|b| !is_name_continue(*b))
            .unwrap_or(body.len());
        let name = &body[..name_end];
        if name.is_empty() || !is_name_start(body.as_bytes()[0]) {
            return Err(SplitError::InvalidExpansion { pos: start });
        }

        let rest = &body[name_end..];
        let value = if rest.is_empty() {
            lookup(name, env)
        } else if let Some(default) = rest.strip_prefix(":-") {
            let found = env.iter().find_map(|(key, value)| {
                if *key == name && !value.is_empty() { Some(*value) } else { None }
            });
            if let Some(value) = found {
                value
            } else {
                return Ok(Some((Box::leak(decode_default(default).into_boxed_str()), close + 1)));
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: start });
        };
        Ok(Some((value, close + 1)))
    } else if is_name_start(bytes[start + 1]) {
        let mut end = start + 2;
        while end < bytes.len() && is_name_continue(bytes[end]) {
            end += 1;
        }
        let name = &input[start + 1..end];
        Ok(Some((lookup(name, env), end)))
    } else {
        Ok(None)
    }
}

fn decode_default(text: &str) -> String {
    let bytes = text.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        if bytes[i] == b'\\' {
            i += 1;
            if i < bytes.len() {
                let ch = text[i..].chars().next().unwrap();
                out.push(ch);
                i += ch.len_utf8();
            }
        } else {
            let ch = text[i..].chars().next().unwrap();
            out.push(ch);
            i += ch.len_utf8();
        }
    }
    out
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    let bytes = input.as_bytes();
    if let Some(pos) = bytes.iter().position(|&b| b == 0) {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current = String::new();
    let mut word_active = false;
    let mut i = 0;
    let mut quote = 0u8; // 0 = unquoted, 1 = single, 2 = double
    let mut quote_start = 0;

    while i < bytes.len() {
        if quote == 1 {
            if bytes[i] == b'\'' {
                quote = 0;
                i += 1;
            } else {
                let ch = input[i..].chars().next().unwrap();
                current.push(ch);
                i += ch.len_utf8();
            }
            continue;
        }

        if quote == 2 {
            match bytes[i] {
                b'"' => {
                    quote = 0;
                    i += 1;
                }
                b'\\' => {
                    let slash = i;
                    i += 1;
                    if i == bytes.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    match bytes[i] {
                        b'"' | b'\\' | b'$' => {
                            current.push(bytes[i] as char);
                            i += 1;
                        }
                        b'\n' => i += 1,
                        _ => {
                            current.push('\\');
                            let ch = input[i..].chars().next().unwrap();
                            current.push(ch);
                            i += ch.len_utf8();
                        }
                    }
                }
                b'$' => {
                    if let Some((value, end)) = expansion(input, i, env)? {
                        current.push_str(value);
                        word_active = true;
                        i = end;
                    } else {
                        current.push('$');
                        i += 1;
                    }
                }
                _ => {
                    let ch = input[i..].chars().next().unwrap();
                    current.push(ch);
                    i += ch.len_utf8();
                }
            }
            continue;
        }

        match bytes[i] {
            b' ' | b'\t' | b'\r' | b'\n' => {
                if word_active {
                    words.push(std::mem::take(&mut current));
                    word_active = false;
                }
                i += 1;
            }
            b'\'' | b'"' => {
                quote = if bytes[i] == b'\'' { 1 } else { 2 };
                quote_start = i;
                word_active = true;
                i += 1;
            }
            b'\\' => {
                let slash = i;
                i += 1;
                if i == bytes.len() {
                    return Err(SplitError::TrailingEscape { pos: slash });
                }
                if bytes[i] == b'\n' {
                    i += 1;
                } else {
                    let ch = input[i..].chars().next().unwrap();
                    current.push(ch);
                    word_active = true;
                    i += ch.len_utf8();
                }
            }
            b'$' => {
                if let Some((value, end)) = expansion(input, i, env)? {
                    current.push_str(value);
                    word_active = true;
                    i = end;
                } else {
                    current.push('$');
                    word_active = true;
                    i += 1;
                }
            }
            _ => {
                let ch = input[i..].chars().next().unwrap();
                current.push(ch);
                word_active = true;
                i += ch.len_utf8();
            }
        }
    }

    if quote == 1 {
        return Err(SplitError::UnterminatedSingle { pos: quote_start });
    }
    if quote == 2 {
        return Err(SplitError::UnterminatedDouble { pos: quote_start });
    }
    if word_active {
        words.push(current);
    }
    Ok(words)
}

fn name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn name_continue(b: u8) -> bool {
    name_start(b) || b.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter().find_map(|(key, value)| {
        if *key == name {
            Some(*value)
        } else {
            None
        }
    })
}

fn expansion<'a>(
    input: &str,
    dollar: usize,
    env: &'a [(&str, &str)],
) -> Result<Option<(String, usize)>, SplitError> {
    let bytes = input.as_bytes();
    if dollar + 1 >= bytes.len() {
        return Ok(None);
    }

    if bytes[dollar + 1] != b'{' {
        if !name_start(bytes[dollar + 1]) {
            return Ok(None);
        }
        let mut end = dollar + 2;
        while end < bytes.len() && name_continue(bytes[end]) {
            end += 1;
        }
        let name = &input[dollar + 1..end];
        return Ok(Some((lookup(name, env).unwrap_or("").to_owned(), end)));
    }

    let mut close = dollar + 2;
    while close < bytes.len() {
        if bytes[close] == b'\\' {
            close += 1;
            if close < bytes.len() {
                let ch = input[close..].chars().next().unwrap();
                close += ch.len_utf8();
            }
        } else if bytes[close] == b'}' {
            break;
        } else {
            let ch = input[close..].chars().next().unwrap();
            close += ch.len_utf8();
        }
    }
    if close == bytes.len() {
        return Err(SplitError::UnterminatedExpansion { pos: dollar });
    }

    let body = &input[dollar + 2..close];
    let body_bytes = body.as_bytes();
    let mut name_end = 0;
    if body_bytes.first().copied().map(name_start) != Some(true) {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }
    name_end = 1;
    while name_end < body_bytes.len() && name_continue(body_bytes[name_end]) {
        name_end += 1;
    }

    let name = &body[..name_end];
    let rest = &body[name_end..];
    let value = if rest.is_empty() {
        lookup(name, env).unwrap_or("").to_owned()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match lookup(name, env) {
            Some(v) if !v.is_empty() => v.to_owned(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };
    Ok(Some((value, close + 1)))
}

fn decode_default(text: &str) -> String {
    let mut chars = text.chars();
    let mut out = String::new();
    while let Some(ch) = chars.next() {
        if ch == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(ch);
        }
    }
    out
}
```
