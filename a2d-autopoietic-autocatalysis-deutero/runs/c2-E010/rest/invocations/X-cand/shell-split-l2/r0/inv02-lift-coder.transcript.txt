pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| if *key == name { Some(*value) } else { None })
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();
    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            } else {
                result.push('\\');
            }
        } else {
            result.push(c);
        }
    }
    result
}

fn parse_expansion(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let after_dollar = dollar + 1;
    let rest = &input[after_dollar..];

    if rest.starts_with('{') {
        let content_start = after_dollar + 1;
        let mut close = None;
        let mut cursor = content_start;
        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            if c == '\\' {
                cursor += 1;
                if cursor < input.len() {
                    cursor += input[cursor..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                close = Some(cursor);
                break;
            } else {
                cursor += c.len_utf8();
            }
        }

        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let content = &input[content_start..close];

        let name_end = content
            .char_indices()
            .find_map(|(i, c)| if !is_name_continue(c) { Some(i) } else { None })
            .unwrap_or(content.len());

        if name_end == 0 {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let name = &content[..name_end];
        let mut name_chars = name.chars();
        if !name_chars.next().map(is_name_start).unwrap_or(false)
            || !name_chars.all(is_name_continue)
        {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let tail = &content[name_end..];
        if tail.is_empty() {
            let value = lookup(name, env).unwrap_or("");
            Ok((value.to_string(), close + 1))
        } else if let Some(default) = tail.strip_prefix(":-") {
            if let Some(value) = lookup(name, env).filter(|v| !v.is_empty()) {
                Ok((value.to_string(), close + 1))
            } else {
                Ok((decode_default(default), close + 1))
            }
        } else {
            Err(SplitError::InvalidExpansion { pos: dollar })
        }
    } else {
        let mut chars = rest.char_indices();
        let Some((_, first)) = chars.next() else {
            return Ok(("$".to_string(), after_dollar));
        };
        if !is_name_start(first) {
            return Ok(("$".to_string(), after_dollar));
        }

        let mut end = after_dollar + first.len_utf8();
        for (offset, c) in chars {
            if is_name_continue(c) {
                end = after_dollar + offset + c.len_utf8();
            } else {
                break;
            }
        }
        let name = &input[after_dollar..end];
        Ok((lookup(name, env).unwrap_or("").to_string(), end))
    }
}

        return Err(SplitError::Nul { pos });

                if c == '\'' {

                    word.push(c);
                }
                cursor = next;

                    quote = Quote::None;
                    cursor = next;

                    } else if escaped == '\n' {

                        word.push('\\');
                        cursor = next;

                    cursor = after;

                    word.push(c);
                    cursor = next;

            Quote::None => {

                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    cursor = next;
                } else if c == '\'' {
                    quote = Quote::Single(cursor);
                    active = true;
                    cursor = next;
                } else if c == '"' {
                    quote = Quote::Double(cursor);
                    active = true;
                    cursor = next;
                } else if c == '\\' {
                    if next == input.len() {
                        return Err(SplitError::TrailingEscape { pos: cursor });
                    }
                    let escaped = input[next..].chars().next().unwrap();
                    let after = next + escaped.len_utf8();
                    if escaped == '\n' {

                    } else {
                        word.push(escaped);
                        active = true;
                        cursor = after;

                    word.push_str(&value);
                    active = true;
                    cursor = after;
                } else {
                    word.push(c);
                    active = true;
                    cursor = next;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Mode {
    Bare,
    Single,
    Double,
}

fn sep(c: char) -> bool {
    c == ' ' || c == '\t' || c == '\r' || c == '\n'
}

fn start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn cont(c: char) -> bool {
    start(c) || c.is_ascii_digit()
}

fn value<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    for (key, value) in env {
        if *key == name {
            return Some(value);
        }
    }
    None
}

fn default_value(text: &str) -> String {
    let mut out = String::new();
    let mut it = text.chars();
    while let Some(c) = it.next() {
        if c == '\\' {
            match it.next() {
                Some(next) => out.push(next),
                None => out.push('\\'),
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn expansion(
    input: &str,
    pos: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let begin = pos + 1;
    if input[begin..].starts_with('{') {
        let body = begin + 1;
        let mut i = body;
        let mut end = None;

        while i < input.len() {
            let c = input[i..].chars().next().unwrap();
            if c == '\\' {
                i += 1;
                if i < input.len() {
                    i += input[i..].chars().next().unwrap().len_utf8();
                }
            } else if c == '}' {
                end = Some(i);
                break;
            } else {
                i += c.len_utf8();
            }
        }

        let end = end.ok_or(SplitError::UnterminatedExpansion { pos })?;
        let body_text = &input[body..end];
        let mut n = 0;
        for c in body_text.chars() {
            if cont(c) {
                n += c.len_utf8();
            } else {
                break;
            }
        }

        let name = &body_text[..n];
        if name.is_empty() || !name.chars().next().map(start).unwrap_or(false) {
            return Err(SplitError::InvalidExpansion { pos });
        }

        let rest = &body_text[n..];
        if rest.is_empty() {
            Ok((value(name, env).unwrap_or("").to_string(), end + 1))
        } else if let Some(text) = rest.strip_prefix(":-") {
            if let Some(found) = value(name, env).filter(|v| !v.is_empty()) {
                Ok((found.to_string(), end + 1))
            } else {
                Ok((default_value(text), end + 1))
            }
        } else {
            Err(SplitError::InvalidExpansion { pos })
        }
    } else {
        let mut iter = input[begin..].char_indices();
        let Some((_, first)) = iter.next() else {
            return Ok(("$".to_string(), begin));
        };
        if !start(first) {
            return Ok(("$".to_string(), begin));
        }
        let mut finish = begin + first.len_utf8();
        for (offset, c) in iter {
            if cont(c) {
                finish = begin + offset + c.len_utf8();
            } else {
                break;
            }
        }
        Ok((value(&input[begin..finish], env).unwrap_or("").to_string(), finish))
    }
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut output = Vec::new();
    let mut current = String::new();
    let mut active = false;
    let mut mode = Mode::Bare;
    let mut opening = 0usize;
    let mut i = 0usize;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let after = i + c.len_utf8();

        match mode {
            Mode::Single => {
                if c == '\'' {
                    mode = Mode::Bare;
                } else {
                    current.push(c);
                }
                i = after;
            }
            Mode::Double => {
                if c == '"' {
                    mode = Mode::Bare;
                    i = after;
                } else if c == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let d = input[after..].chars().next().unwrap();
                    if d == '"' || d == '\\' || d == '$' {
                        current.push(d);
                        i = after + d.len_utf8();
                    } else if d == '\n' {
                        i = after + 1;
                    } else {
                        current.push('\\');
                        i = after;
                    }
                } else if c == '$' {
                    let (expanded, next) = expansion(input, i, env)?;
                    current.push_str(&expanded);
                    active = true;
                    i = next;
                } else {
                    current.push(c);
                    i = after;
                }
            }
            Mode::Bare => {
                if sep(c) {
                    if active {
                        output.push(std::mem::take(&mut current));
                        active = false;
                    }
                    i = after;
                } else if c == '\'' || c == '"' {
                    opening = i;
                    mode = if c == '\'' { Mode::Single } else { Mode::Double };
                    active = true;
                    i = after;
                } else if c == '\\' {
                    if after == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let d = input[after..].chars().next().unwrap();
                    if d != '\n' {
                        current.push(d);
                        active = true;
                    }
                    i = after + d.len_utf8();
                } else if c == '$' {
                    let (expanded, next) = expansion(input, i, env)?;
                    current.push_str(&expanded);
                    active = true;
                    i = next;
                } else {
                    current.push(c);
                    active = true;
                    i = after;
                }
            }
        }
    }

    if mode == Mode::Single {
        return Err(SplitError::UnterminatedSingle { pos: opening });
    }
    if mode == Mode::Double {
        return Err(SplitError::UnterminatedDouble { pos: opening });
    }
    if active {
        output.push(current);
    }
    Ok(output)
}
