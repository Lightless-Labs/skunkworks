// Interpretation notes:
// - Validation follows the specified side order, then final-LF checks, then line limits.
// - A trailing LF terminates the last line; it does not add an empty line.
// - Insertions interact with positive-length changes only strictly inside their intervals.
// - Marker lines count as output lines; empty projections add no payload lines.
// - At most 128 lines per input keeps the LCS tables and change-component work bounded.
// - Solution A uses union-find for transitive change components.

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change<'a> {
    start: usize,
    end: usize,
    replacement: Vec<&'a str>,
    side: usize,
}

struct Component {
    start: usize,
    end: usize,
    changes: Vec<usize>,
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1].split('\n').collect()
}

fn first_cr(text: &str) -> Option<usize> {
    text.as_bytes().iter().position(|&b| b == b'\r')
}

fn validate(
    base: &str,
    left: &str,
    right: &str,
) -> Result<(Vec<&str>, Vec<&str>, Vec<&str>), MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = first_cr(text) {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let b = split_lines(base);
    let l = split_lines(left);
    let r = split_lines(right);
    if b.len() > 128 || l.len() > 128 || r.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: b.len(),
            left_len: l.len(),
            right_len: r.len(),
        });
    }
    Ok((b, l, r))
}

fn changes<'a>(base: &[&'a str], new: &[&'a str], side: usize) -> Vec<Change<'a>> {
    let n = base.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];
    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut run_start = None;
    let mut deletes = 0usize;
    let mut inserts = Vec::new();
    let mut out = Vec::new();

    let finish_run = |out: &mut Vec<Change<'a>>,
                      run_start: &mut Option<usize>,
                      deletes: &mut usize,
                      inserts: &mut Vec<&'a str>,
                      consumed: usize| {
        if let Some(start) = run_start.take() {
            out.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserts),
                side,
            });
            *deletes = 0;
        } else {
            let _ = consumed;
        }
    };

    while i < n || j < m {
        if i < n && j < m && base[i] == new[j] {
            finish_run(&mut out, &mut run_start, &mut deletes, &mut inserts, i);
            i += 1;
            j += 1;
        } else if i == n {
            run_start.get_or_insert(i);
            inserts.push(new[j]);
            j += 1;
        } else if j == m {
            run_start.get_or_insert(i);
            deletes += 1;
            i += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            let take_delete = if del > ins {
                true
            } else if ins > del {
                false
            } else {
                (i + j) % 2 == 0
            };
            if take_delete {
                run_start.get_or_insert(i);
                deletes += 1;
                i += 1;
            } else {
                run_start.get_or_insert(i);
                inserts.push(new[j]);
                j += 1;
            }
        }
    }
    finish_run(&mut out, &mut run_start, &mut deletes, &mut inserts, i);
    out
}

fn interacts(a: &Change<'_>, b: &Change<'_>) -> bool {
    match (a.start < a.end, b.start < b.end) {
        (true, true) => a.start.max(b.start) < a.end.min(b.end),
        (false, false) => a.start == b.start,
        (false, true) => b.start < a.start && a.start < b.end,
        (true, false) => a.start < b.start && b.start < a.end,
    }
}

fn find(parent: &mut [usize], x: usize) -> usize {
    if parent[x] != x {
        parent[x] = find(parent, parent[x]);
    }
    parent[x]
}

fn union(parent: &mut [usize], a: usize, b: usize) {
    let ra = find(parent, a);
    let rb = find(parent, b);
    if ra != rb {
        parent[rb] = ra;
    }
}

fn project<'a>(
    base: &[&'a str],
    component: &Component,
    all: &[Change<'a>],
    side: usize,
) -> Vec<&'a str> {
    let mut selected: Vec<usize> = component
        .changes
        .iter()
        .copied()
        .filter(|&idx| all[idx].side == side)
        .collect();
    selected.sort_by_key(|&idx| (all[idx].start, all[idx].end));

    let mut result = Vec::new();
    let mut cursor = component.start;
    for idx in selected {
        let change = &all[idx];
        result.extend_from_slice(&base[cursor..change.start]);
        result.extend(change.replacement.iter().copied());
        cursor = change.end;
    }
    result.extend_from_slice(&base[cursor..component.end]);
    result
}

fn push_line(output: &mut String, line: &str) {
    output.push_str(line);
    output.push('\n');
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let (base_lines, left_lines, right_lines) = validate(base, left, right)?;
    let left_changes = changes(&base_lines, &left_lines, 0);
    let right_changes = changes(&base_lines, &right_lines, 1);
    let mut all = left_changes;
    all.extend(right_changes);

    if all.is_empty() {
        let mut text = String::new();
        for line in base_lines {
            push_line(&mut text, line);
        }
        return Ok(MergeResult {
            text,
            conflicts: 0,
            regions: Vec::new(),
        });
    }

    let mut parent: Vec<usize> = (0..all.len()).collect();
    for i in 0..all.len() {
        for j in (i + 1)..all.len() {
            if interacts(&all[i], &all[j]) {
                union(&mut parent, i, j);
            }
        }
    }

    let mut grouped: Vec<Component> = Vec::new();
    let mut roots = Vec::new();
    for idx in 0..all.len() {
        let root = find(&mut parent, idx);
        let group_index = match roots.iter().position(|&r| r == root) {
            Some(pos) => pos,
            None => {
                roots.push(root);
                grouped.push(Component {
                    start: all[idx].start,
                    end: all[idx].end,
                    changes: Vec::new(),
                });
                grouped.len() - 1
            }
        };
        let component = &mut grouped[group_index];
        component.start = component.start.min(all[idx].start);
        component.end = component.end.max(all[idx].end);
        component.changes.push(idx);
    }

    grouped.sort_by_key(|c| (c.start, usize::from(c.start < c.end), c.end));

    let mut output = String::new();
    let mut output_lines = 0usize;
    let mut base_cursor = 0usize;
    let mut conflicts = Vec::new();

    for component in grouped {
        for line in &base_lines[base_cursor..component.start] {
            push_line(&mut output, line);
            output_lines += 1;
        }

        let base_projection = &base_lines[component.start..component.end];
        let left_projection = project(&base_lines, &component, &all, 0);
        let right_projection = project(&base_lines, &component, &all, 1);

        let is_conflict = left_projection != right_projection
            && left_projection != base_projection
            && right_projection != base_projection;

        if !is_conflict {
            let selected = if left_projection == right_projection {
                &left_projection
            } else if left_projection == base_projection {
                &right_projection
            } else {
                &left_projection
            };
            for line in selected {
                push_line(&mut output, line);
                output_lines += 1;
            }
        } else {
            match resolution {
                Resolution::PreferLeft => {
                    for line in &left_projection {
                        push_line(&mut output, line);
                        output_lines += 1;
                    }
                }
                Resolution::PreferRight => {
                    for line in &right_projection {
                        push_line(&mut output, line);
                        output_lines += 1;
                    }
                }
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output_lines;
                    push_line(&mut output, &format!("<<<<<<< LEFT {ordinal}"));
                    for line in &left_projection {
                        push_line(&mut output, line);
                    }
                    let c = component.end - component.start;
                    let s = if c > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    push_line(&mut output, &format!("||||||| BASE {s},{c}"));
                    for line in base_projection {
                        push_line(&mut output, line);
                    }
                    push_line(&mut output, "=======");
                    for line in &right_projection {
                        push_line(&mut output, line);
                    }
                    push_line(&mut output, &format!(">>>>>>> RIGHT {ordinal}"));
                    let output_len = 4
                        + left_projection.len()
                        + base_projection.len()
                        + right_projection.len();
                    output_lines += output_len;
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: c,
                        output_start,
                        output_len,
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    for line in &base_lines[base_cursor..] {
        push_line(&mut output, line);
    }

    Ok(MergeResult {
        text: output,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn lines(s: &str) -> Vec<&str> {
    if s.is_empty() {
        Vec::new()
    } else {
        s[..s.len() - 1].split('\n').collect()
    }
}

fn validate<'a>(
    base: &'a str,
    left: &'a str,
    right: &'a str,
) -> Result<(Vec<&'a str>, Vec<&'a str>, Vec<&'a str>), MergeError> {
    for (s, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(pos) = s.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte: pos });
        }
    }
    for (s, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !s.is_empty() && !s.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }
    let b = lines(base);
    let l = lines(left);
    let r = lines(right);
    if b.len() > 128 || l.len() > 128 || r.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: b.len(),
            left_len: l.len(),
            right_len: r.len(),
        });
    }
    Ok((b, l, r))
}

fn make_changes<'a>(old: &[&'a str], new: &[&'a str], side: usize) -> Vec<Change<'a>> {
    let (n, m) = (old.len(), new.len());
    let mut dp = vec![vec![0usize; m + 1]; n + 1];
    for i in (0..n).rev() {
        for j in (0..m).rev() {
            dp[i][j] = if old[i] == new[j] {
                dp[i + 1][j + 1] + 1
            } else {
                dp[i + 1][j].max(dp[i][j + 1])
            };
        }
    }
    let (mut i, mut j) = (0, 0);
    let mut changes = Vec::new();
    let mut start = None;
    let mut deleted = 0;
    let mut inserted = Vec::new();

    let flush = |changes: &mut Vec<Change<'a>>,
                 start: &mut Option<usize>,
                 deleted: &mut usize,
                 inserted: &mut Vec<&'a str>| {
        if let Some(s) = start.take() {
            changes.push(Change {
                start: s,
                end: s + *deleted,
                inserted: std::mem::take(inserted),
                side,
            });
            *deleted = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(&mut changes, &mut start, &mut deleted, &mut inserted);
            i += 1;
            j += 1;
        } else if i == n {
            start.get_or_insert(i);
            inserted.push(new[j]);
            j += 1;
        } else if j == m {
            start.get_or_insert(i);
            deleted += 1;
            i += 1;
        } else {
            let d = dp[i + 1][j];
            let ins = dp[i][j + 1];
            let delete = if d != ins {
                d > ins
            } else {
                (i + j) % 2 == 0
            };
            start.get_or_insert(i);
            if delete {
                deleted += 1;
                i += 1;
            } else {
                inserted.push(new[j]);
                j += 1;
            }
        }
    }
    flush(&mut changes, &mut start, &mut deleted, &mut inserted);
    changes
}

fn overlaps(a: &Change<'_>, b: &Change<'_>) -> bool {
    match (a.start == a.end, b.start == b.end) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn components(changes: &[Change<'_>]) -> Vec<Component> {
    let n = changes.len();
    let mut reach = vec![vec![false; n]; n];
    for i in 0..n {
        reach[i][i] = true;
        for j in 0..n {
            if overlaps(&changes[i], &changes[j]) {
                reach[i][j] = true;
            }
        }
    }
    for k in 0..n {
        for i in 0..n {
            if reach[i][k] {
                for j in 0..n {
                    reach[i][j] |= reach[k][j];
                }
            }
        }
    }
    let mut used = vec![false; n];
    let mut result = Vec::new();
    for i in 0..n {
        if used[i] {
            continue;
        }
        let ids: Vec<usize> = (0..n).filter(|&j| reach[i][j] && reach[j][i]).collect();
        for &j in &ids {
            used[j] = true;
        }
        let start = ids.iter().map(|&j| changes[j].start).min().unwrap();
        let end = ids.iter().map(|&j| changes[j].end).max().unwrap();
        result.push(Component { start, end, ids });
    }
    result.sort_by_key(|c| (c.start, usize::from(c.start < c.end), c.end));
    result
}

fn projection<'a>(
    base: &[&'a str],
    component: &Component,
    changes: &[Change<'a>],
    side: usize,
) -> Vec<&'a str> {
    let mut ids: Vec<usize> = component
        .ids
        .iter()
        .copied()
        .filter(|&id| changes[id].side == side)
        .collect();
    ids.sort_by_key(|&id| (changes[id].start, changes[id].end));
    let mut result = Vec::new();
    let mut cursor = component.start;
    for id in ids {
        let c = &changes[id];
        result.extend_from_slice(&base[cursor..c.start]);
        result.extend(c.inserted.iter().copied());
        cursor = c.end;
    }
    result.extend_from_slice(&base[cursor..component.end]);
    result
}

fn append(out: &mut String, line: &str) {
    out.push_str(line);
    out.push('\n');
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let (b, l, r) = validate(base, left, right)?;
    let mut edits = make_changes(&b, &l, 0);
    edits.extend(make_changes(&b, &r, 1));
    let groups = components(&edits);

    let mut text = String::new();
    let mut regions = Vec::new();
    let mut out_index = 0usize;
    let mut cursor = 0usize;

    for group in groups {
        for line in &b[cursor..group.start] {
            append(&mut text, line);
            out_index += 1;
        }
        let bp = &b[group.start..group.end];
        let lp = projection(&b, &group, &edits, 0);
        let rp = projection(&b, &group, &edits, 1);
        if lp == rp || lp == bp || rp == bp {
            let chosen = if lp == rp {
                &lp
            } else if lp == bp {
                &rp
            } else {
                &lp
            };
            for line in chosen {
                append(&mut text, line);
                out_index += 1;
            }
        } else {
            match resolution {
                Resolution::PreferLeft => {
                    for line in &lp {
                        append(&mut text, line);
                        out_index += 1;
                    }
                }
                Resolution::PreferRight => {
                    for line in &rp {
                        append(&mut text, line);
                        out_index += 1;
                    }
                }
                Resolution::Mark => {
                    let ordinal = regions.len() + 1;
                    let output_start = out_index;
                    append(&mut text, &format!("<<<<<<< LEFT {ordinal}"));
                    for line in &lp {
                        append(&mut text, line);
                    }
                    let len = group.end - group.start;
                    let base_number = if len == 0 { group.start } else { group.start + 1 };
                    append(
                        &mut text,
                        &format!("||||||| BASE {base_number},{len}"),
                    );
                    for line in bp {
                        append(&mut text, line);
                    }
                    append(&mut text, "=======");
                    for line in &rp {
                        append(&mut text, line);
                    }
                    append(&mut text, &format!(">>>>>>> RIGHT {ordinal}"));
                    let output_len = 4 + lp.len() + bp.len() + rp.len();
                    out_index += output_len;
                    regions.push(Conflict {
                        ordinal,
                        base_start: group.start,
                        base_len: len,
                        output_start,
                        output_len,
                    });
                }
            }
        }
        cursor = group.end;
    }

    for line in &b[cursor..] {
        append(&mut text, line);
    }
    Ok(MergeResult {
        text,
        conflicts: regions.len(),
        regions,
    })
}
