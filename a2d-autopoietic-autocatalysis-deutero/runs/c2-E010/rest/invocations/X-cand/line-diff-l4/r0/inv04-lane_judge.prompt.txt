fn split_lines(text: &str) -> Vec<String> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1]
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edit_script(old: &[String], new: &[String]) -> Vec<Op> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0, 0);
    let mut ops = Vec::new();
    while i < n && j < m {
        if old[i] == new[j] {
            ops.push(Op::Keep);
            i += 1;
            j += 1;
        } else {
            let delete_score = lcs[i + 1][j];
            let insert_score = lcs[i][j + 1];
            if delete_score > insert_score
                || (delete_score == insert_score && (i + j) % 2 == 0)
            {
                ops.push(Op::Delete);
                i += 1;
            } else {
                ops.push(Op::Insert);
                j += 1;
            }
        }
    }
    while i < n {
        ops.push(Op::Delete);
        i += 1;
    }
    while j < m {
        ops.push(Op::Insert);
        j += 1;
    }
    ops
}

fn changes_for(old: &[String], new: &[String], side: usize) -> Vec<Change> {
    let ops = edit_script(old, new);
    let (mut i, mut j) = (0usize, 0usize);
    let mut result = Vec::new();

    let mut active = false;
    let mut start = 0usize;
    let mut deletes = 0usize;
    let mut replacement = Vec::new();

    for op in ops {
        match op {
            Op::Keep => {
                if active {
                    result.push(Change {
                        start,
                        end: start + deletes,
                        replacement: std::mem::take(&mut replacement),
                        side,
                    });
                    active = false;
                    deletes = 0;
                }
                i += 1;
                j += 1;
            }
            Op::Delete => {
                if !active {
                    active = true;
                    start = i;
                }
                deletes += 1;
                i += 1;
            }
            Op::Insert => {
                if !active {
                    active = true;
                    start = i;
                }
                replacement.push(new[j].clone());
                j += 1;
            }
        }
    }
    if active {
        result.push(Change {
            start,
            end: start + deletes,
            replacement,
            side,
        });
    }
    result
}

fn interacts(a: &Change, b: &Change) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(base: &[String], start: usize, end: usize, changes: &[&Change]) -> Vec<String> {
    let mut ordered = changes.to_vec();
    ordered.sort_by_key(|c| (c.start, c.end));
    let mut result = Vec::new();
    let mut cursor = start;

    for change in ordered {
        if change.start > cursor {
            result.extend(base[cursor..change.start].iter().cloned());
        }
        result.extend(change.replacement.iter().cloned());
        cursor = change.end;
    }
    if cursor < end {
        result.extend(base[cursor..end].iter().cloned());
    }
    result
}

fn append_lines(out: &mut Vec<String>, lines: &[String]) {
    out.extend(lines.iter().cloned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base_lines = split_lines(base);
    let left_lines = split_lines(left);
    let right_lines = split_lines(right);

    if base_lines.len() > 128 || left_lines.len() > 128 || right_lines.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base_lines.len(),
            left_len: left_lines.len(),
            right_len: right_lines.len(),
        });
    }

    let mut changes = changes_for(&base_lines, &left_lines, 0);
    changes.extend(changes_for(&base_lines, &right_lines, 1));

    let count = changes.len();
    let mut parent: Vec<usize> = (0..count).collect();

    fn root(parent: &mut [usize], mut x: usize) -> usize {
        while parent[x] != x {
            parent[x] = parent[parent[x]];
            x = parent[x];
        }
        x
    }

    for i in 0..count {
        for j in (i + 1)..count {
            if interacts(&changes[i], &changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut grouped: Vec<Component> = Vec::new();
    let mut roots = Vec::new();
    for i in 0..count {
        let r = root(&mut parent, i);
        let position = roots.iter().position(|&x| x == r);
        let idx = if let Some(idx) = position {
            idx
        } else {
            roots.push(r);
            grouped.push(Component {
                start: changes[i].start,
                end: changes[i].end,
                changes: Vec::new(),
            });
            grouped.len() - 1
        };
        grouped[idx].start = grouped[idx].start.min(changes[i].start);
        grouped[idx].end = grouped[idx].end.max(changes[i].end);
        grouped[idx].changes.push(i);
    }

    grouped.sort_by_key(|c| (c.start, c.end != c.start, c.end));

    let mut output = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in grouped {
        if base_cursor < component.start {
            append_lines(&mut output, &base_lines[base_cursor..component.start]);
        }

        let base_projection = base_lines[component.start..component.end].to_vec();
        let left_changes: Vec<&Change> = component
            .changes
            .iter()
            .map(|&i| &changes[i])
            .filter(|c| c.side == 0)
            .collect();
        let right_changes: Vec<&Change> = component
            .changes
            .iter()
            .map(|&i| &changes[i])
            .filter(|c| c.side == 1)
            .collect();

        let left_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &left_changes,
        );
        let right_projection = projection(
            &base_lines,
            component.start,
            component.end,
            &right_changes,
        );

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    let region_len = component.end - component.start;
                    let base_marker_start = if region_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };

                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    output.push(format!(
                        "||||||| BASE {base_marker_start},{region_len}"
                    ));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_owned());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));

                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: region_len,
                        output_start,
                        output_len: 4
                            + left_projection.len()
                            + base_projection.len()
                            + right_projection.len(),
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    if base_cursor < base_lines.len() {
        append_lines(&mut output, &base_lines[base_cursor..]);
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

fn script(old: &[String], new: &[String], side: usize) -> Vec<Edit> {
    let (n, m) = (old.len(), new.len());
    let mut dp = vec![vec![0usize; m + 1]; n + 1];
    for i in (0..n).rev() {
        for j in (0..m).rev() {
            dp[i][j] = if old[i] == new[j] {
                dp[i + 1][j + 1] + 1
            } else {
                dp[i + 1][j].max(dp[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut consumed) = (0, 0, 0);
    let mut pending: Option<(usize, usize, Vec<String>)> = None;
    let mut edits = Vec::new();

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            if let Some((start, deleted, replacement)) = pending.take() {
                edits.push(Edit {
                    start,
                    end: start + deleted,
                    replacement,
                    side,
                });
            }
            i += 1;
            j += 1;
            consumed += 1;
        } else {
            let delete = if i == n {
                false
            } else if j == m {
                true
            } else {
                let d = dp[i + 1][j];
                let ins = dp[i][j + 1];
                d > ins || (d == ins && (i + j) % 2 == 0)
            };
            let p = pending.get_or_insert_with(|| (consumed, 0, Vec::new()));
            if delete {
                p.1 += 1;
                i += 1;
                consumed += 1;
            } else {
                p.2.push(new[j].clone());
                j += 1;
            }
        }
    }
    if let Some((start, deleted, replacement)) = pending {
        edits.push(Edit {
            start,
            end: start + deleted,
            replacement,
            side,
        });
    }
    edits
}

fn connected(a: &Edit, b: &Edit) -> bool {
    match (a.start == a.end, b.start == b.end) {
        (true, true) => a.start == b.start,
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|b| *b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }
    for (text, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let b = split_lines(base);
    let l = split_lines(left);
    let r = split_lines(right);
    if b.len() > 128 || l.len() > 128 || r.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: b.len(),
            left_len: l.len(),
            right_len: r.len(),
        });

    edits.extend(script(&b, &r, 1));

    let mut groups: Vec<(usize, usize, Vec<usize>)> = Vec::new();
    for i in 0..edits.len() {
        if visited[i] {
            continue;
        }

        let start = ids.iter().map(|&k| edits[k].start).min().unwrap();
        let end = ids.iter().map(|&k| edits[k].end).max().unwrap();
        ids.sort_by_key(|&k| (edits[k].start, edits[k].end));
        groups.push((start, end, ids));
    }
    groups.sort_by_key(|g| (g.0, usize::from(g.0 != g.1), g.1));

    let mut output = Vec::<String>::new();
    let mut records = Vec::<Conflict>::new();
    let mut cursor = 0;
    for (start, end, ids) in groups {
        output.extend(b[cursor..start].iter().cloned());
        let bp = b[start..end].to_vec();
        let left_edits: Vec<Edit> = ids
            .iter()
            .map(|&i| &edits[i])
            .filter(|e| e.side == 0)
            .cloned()
            .collect();
        let right_edits: Vec<Edit> = ids
            .iter()
            .map(|&i| &edits[i])
            .filter(|e| e.side == 1)
            .cloned()
            .collect();
        let lp = project(&b, start, end, &left_edits);
        let rp = project(&b, start, end, &right_edits);

        if lp == rp {
            output.extend(lp);
        } else if lp == bp {
            output.extend(rp);
        } else if rp == bp {
            output.extend(lp);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(lp),
                Resolution::PreferRight => output.extend(rp),
                Resolution::Mark => {
                    let ordinal = records.len() + 1;
                    let output_start = output.len();
                    append(&mut output, format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(lp.iter().cloned());
                    let count = end - start;
                    let base_label = if count == 0 { start } else { start + 1 };
                    append(
                        &mut output,
                        format!("||||||| BASE {base_label},{count}"),
                    );
                    output.extend(bp.iter().cloned());
                    append(&mut output, "=======".to_string());
                    output.extend(rp.iter().cloned());
                    append(&mut output, format!(">>>>>>> RIGHT {ordinal}"));
                    records.push(Conflict {
                        ordinal,
                        base_start: start,
                        base_len: count,
                        output_start,
                        output_len: 4 + lp.len() + bp.len() + rp.len(),
                    });
                }
            }
        }
        cursor = end;
    }
    output.extend(b[cursor..].iter().cloned());

    let mut text = String::new();
    for line in output {
        text.push_str(&line);
        text.push('\n');
    }
    Ok(MergeResult {
        text,
        conflicts: records.len(),
        regions: records,
    })
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1].split('\n').collect()
    }

fn make_changes<'a>(base: &[&'a str], new: &[&'a str], side: MergeSide) -> Vec<Change<'a>> {
    let n = base.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut changes = Vec::new();
    let mut in_change = false;
    let mut start = 0usize;
    let mut end = 0usize;
    let mut replacement = Vec::new();

    let finish = |changes: &mut Vec<Change<'a>>,
                  in_change: &mut bool,
                  start: &mut usize,
                  end: &mut usize,
                  replacement: &mut Vec<&'a str>| {
        if *in_change {
            changes.push(Change {
                start: *start,
                end: *end,
                replacement: std::mem::take(replacement),
                side: side.clone(),
            });
            *in_change = false;
        }
    };

    while i < n || j < m {
        if i < n && j < m && base[i] == new[j] {
            finish(
                &mut changes,
                &mut in_change,
                &mut start,
                &mut end,
                &mut replacement,
            );
            i += 1;
            j += 1;
        } else {
            if !in_change {
                in_change = true;
                start = i;
                end = i;
            }
            let delete = if i == n {
                false
            } else if j == m {
                true
            } else {
                let del = lcs[i + 1][j];
                let ins = lcs[i][j + 1];
                if del > ins {
                    true
                } else if ins > del {
                    false
                } else {
                    (i + j) % 2 == 0
                }
            };

            if delete {
                i += 1;
                end += 1;
            } else {
                replacement.push(new[j]);
                j += 1;
            }
        }
    }

    finish(
        &mut changes,
        &mut in_change,
        &mut start,
        &mut end,
        &mut replacement,
    );
    changes
}

fn interacts(a: &Change<'_>, b: &Change<'_>) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn project<'a>(
    base: &[&'a str],
    component: &Component,
    changes: &[Change<'a>],
    side: MergeSide,
) -> Vec<&'a str> {
    let mut selected: Vec<&Change<'a>> = component
        .changes
        .iter()
        .map(|&idx| &changes[idx])
        .filter(|change| change.side == side)
        .collect();
    selected.sort_by_key(|change| (change.start, change.end));

    let mut out = Vec::new();
    let mut cursor = component.start;
    for change in selected {
        for line in &base[cursor..change.start] {
            out.push(*line);
        }
        out.extend(change.replacement.iter().copied());
        cursor = change.end;
    }
    for line in &base[cursor..component.end] {
        out.push(*line);
    }
    out
}

fn append_lines(out: &mut Vec<String>, lines: &[&str]) {
    out.extend(lines.iter().map(|line| (*line).to_string()));
}

pub fn merge_three_way(
    base_text: &str,
    left_text: &str,
    right_text: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base_text, MergeSide::Base),
        (left_text, MergeSide::Left),
        (right_text, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base_text, MergeSide::Base),
        (left_text, MergeSide::Left),
        (right_text, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base = split_lines(base_text);
    let left = split_lines(left_text);
    let right = split_lines(right_text);

    if base.len() > 128 || left.len() > 128 || right.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base.len(),
            left_len: left.len(),
            right_len: right.len(),
        });
    }

    let mut changes = make_changes(&base, &left, MergeSide::Left);
    changes.extend(make_changes(&base, &right, MergeSide::Right));

    let mut parent: Vec<usize> = (0..changes.len()).collect();
    fn root(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = root(parent, parent[x]);
        }
        parent[x]
    }
    for i in 0..changes.len() {
        for j in i + 1..changes.len() {
            if interacts(&changes[i], &changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut components = Vec::<Component>::new();
    for idx in 0..changes.len() {
        let r = root(&mut parent, idx);
        if let Some(component) = components.iter_mut().find(|c| {
            c.changes
                .first()
                .map(|&existing| root(&mut parent, existing) == r)
                .unwrap_or(false)
        }) {
            component.changes.push(idx);
        } else {
            components.push(Component {
                changes: vec![idx],
                start: changes[idx].start,
                end: changes[idx].end,
            });
        }
    }

    for component in &mut components {
        component.start = component
            .changes
            .iter()
            .map(|&idx| changes[idx].start)
            .min()
            .unwrap_or(0);
        component.end = component
            .changes
            .iter()
            .map(|&idx| changes[idx].end)
            .max()
            .unwrap_or(component.start);
    }
    components.sort_by_key(|c| (c.start, if c.start == c.end { 0 } else { 1 }, c.end));

    let mut output = Vec::<String>::new();
    let mut records = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        for line in &base[base_cursor..component.start] {
            output.push((*line).to_string());
        }

        let base_projection: Vec<&str> = base[component.start..component.end].to_vec();
        let left_projection = project(&base, &component, &changes, MergeSide::Left);
        let right_projection = project(&base, &component, &changes, MergeSide::Right);

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = records.len() + 1;
                    let output_start = output.len();
                    let base_len = component.end - component.start;
                    let base_line_number = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    output.push(format!("||||||| BASE {base_line_number},{base_len}"));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_string());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    records.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: output.len() - output_start,
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    for line in &base[base_cursor..] {
        output.push((*line).to_string());
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: records.len(),
        regions: records,
    })
}

fn validate<'a>(

    left: &'a str,
    right: &'a str,

fn make_changes<'a>(old: &[&'a str], new: &[&'a str], side: usize) -> Vec<Change<'a>> {
    let (n, m) = (old.len(), new.len());
    let mut dp = vec![vec![0usize; m + 1]; n + 1];
    for i in (0..n).rev() {
        for j in (0..m).rev() {
            dp[i][j] = if old[i] == new[j] {
                dp[i + 1][j + 1] + 1
            } else {
                dp[i + 1][j].max(dp[i][j + 1])
            };
        }
    }
    let (mut i, mut j) = (0, 0);
    let mut changes = Vec::new();
    let mut start = None;
    let mut deleted = 0;
    let mut inserted = Vec::new();

    let flush = |changes: &mut Vec<Change<'a>>,
                 start: &mut Option<usize>,
                 deleted: &mut usize,
                 inserted: &mut Vec<&'a str>| {
        if let Some(s) = start.take() {
            changes.push(Change {
                start: s,
                end: s + *deleted,
                inserted: std::mem::take(inserted),
                side,
            });
            *deleted = 0;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(&mut changes, &mut start, &mut deleted, &mut inserted);
            i += 1;
            j += 1;
        } else if i == n {
            start.get_or_insert(i);
            inserted.push(new[j]);
            j += 1;
        } else if j == m {
            start.get_or_insert(i);
            deleted += 1;
            i += 1;
        } else {
            let d = dp[i + 1][j];
            let ins = dp[i][j + 1];
            let delete = if d != ins {
                d > ins
            } else {
                (i + j) % 2 == 0
            };
            start.get_or_insert(i);
            if delete {
                deleted += 1;
                i += 1;
            } else {
                inserted.push(new[j]);
                j += 1;
            }
        }
    }
    flush(&mut changes, &mut start, &mut deleted, &mut inserted);
    changes
}

fn overlaps(a: &Change<'_>, b: &Change<'_>) -> bool {
    match (a.start == a.end, b.start == b.end) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn components(changes: &[Change<'_>]) -> Vec<Component> {
    let n = changes.len();
    let mut reach = vec![vec![false; n]; n];
    for i in 0..n {
        reach[i][i] = true;
        for j in 0..n {
            if overlaps(&changes[i], &changes[j]) {
                reach[i][j] = true;
            }
        }
    }
    for k in 0..n {
        for i in 0..n {
            if reach[i][k] {
                for j in 0..n {
                    reach[i][j] |= reach[k][j];
                }
            }
        }
    }
    let mut used = vec![false; n];
    let mut result = Vec::new();
    for i in 0..n {
        if used[i] {
            continue;
        }
        let ids: Vec<usize> = (0..n).filter(|&j| reach[i][j] && reach[j][i]).collect();
        for &j in &ids {
            used[j] = true;
        }
        let start = ids.iter().map(|&j| changes[j].start).min().unwrap();
        let end = ids.iter().map(|&j| changes[j].end).max().unwrap();
        result.push(Component { start, end, ids });
    }
    result.sort_by_key(|c| (c.start, usize::from(c.start < c.end), c.end));
    result
}

fn projection<'a>(
    base: &[&'a str],
    component: &Component,
    changes: &[Change<'a>],
    side: usize,
) -> Vec<&'a str> {
    let mut ids: Vec<usize> = component
        .ids
        .iter()
        .copied()
        .filter(|&id| changes[id].side == side)
        .collect();
    ids.sort_by_key(|&id| (changes[id].start, changes[id].end));
    let mut result = Vec::new();
    let mut cursor = component.start;
    for id in ids {
        let c = &changes[id];
        result.extend_from_slice(&base[cursor..c.start]);
        result.extend(c.inserted.iter().copied());
        cursor = c.end;
    }
    result.extend_from_slice(&base[cursor..component.end]);
    result
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let (b, l, r) = validate(base, left, right)?;
    let mut edits = make_changes(&b, &l, 0);
    edits.extend(make_changes(&b, &r, 1));
    let groups = components(&edits);

    let mut text = String::new();
    let mut regions = Vec::new();
    let mut out_index = 0usize;
    let mut cursor = 0usize;

    for group in groups {
        for line in &b[cursor..group.start] {
            append(&mut text, line);
            out_index += 1;
        }
        let bp = &b[group.start..group.end];
        let lp = projection(&b, &group, &edits, 0);
        let rp = projection(&b, &group, &edits, 1);
        if lp == rp || lp == bp || rp == bp {
            let chosen = if lp == rp {
                &lp
            } else if lp == bp {
                &rp
            } else {
                &lp
            };
            for line in chosen {
                append(&mut text, line);
                out_index += 1;
            }
        } else {
            match resolution {
                Resolution::PreferLeft => {
                    for line in &lp {
                        append(&mut text, line);
                        out_index += 1;
                    }
                }
                Resolution::PreferRight => {
                    for line in &rp {
                        append(&mut text, line);
                        out_index += 1;
                    }
                }
                Resolution::Mark => {
                    let ordinal = regions.len() + 1;
                    let output_start = out_index;
                    append(&mut text, &format!("<<<<<<< LEFT {ordinal}"));
                    for line in &lp {
                        append(&mut text, line);
                    }
                    let len = group.end - group.start;
                    let base_number = if len == 0 { group.start } else { group.start + 1 };
                    append(
                        &mut text,
                        &format!("||||||| BASE {base_number},{len}"),
                    );
                    for line in bp {
                        append(&mut text, line);
                    }
                    append(&mut text, "=======");
                    for line in &rp {
                        append(&mut text, line);
                    }
                    append(&mut text, &format!(">>>>>>> RIGHT {ordinal}"));
                    let output_len = 4 + lp.len() + bp.len() + rp.len();
                    out_index += output_len;
                    regions.push(Conflict {
                        ordinal,
                        base_start: group.start,
                        base_len: len,
                        output_start,
                        output_len,
                    });
                }
            }
        }
        cursor = group.end;
    }

    for line in &b[cursor..] {
        append(&mut text, line);
    }
    Ok(MergeResult {
        text,
        conflicts: regions.len(),
        regions,
    })
}

    end: usize,
}

fn lines(s: &str) -> Vec<&str> {
    if s.is_empty() {
        vec![]
    } else {
        s.split_terminator('\n').collect()
    }
}

fn diff<'a>(old: &[&'a str], new: &[&'a str], side: usize) -> Vec<Change<'a>> {
    let (n, m) = (old.len(), new.len());
    let width = m + 1;
    let mut dp = vec![0usize; (n + 1) * (m + 1)];
    let at = |i: usize, j: usize| i * width + j;

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            let value = if old[i] == new[j] {
                1 + dp[at(i + 1, j + 1)]
            } else {
                dp[at(i + 1, j)].max(dp[at(i, j + 1)])
            };
            dp[at(i, j)] = value;
        }
    }

    let mut i = 0;
    let mut j = 0;
    let mut result = Vec::new();
    let mut active = false;
    let mut start = 0;
    let mut end = 0;
    let mut replacement = Vec::new();

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            if active {
                result.push(Change {
                    start,
                    end,
                    replacement: std::mem::take(&mut replacement),
                    side,
                });
                active = false;
            }
            i += 1;
            j += 1;
            continue;
        }

        if !active {
            active = true;
            start = i;
            end = i;
        }

        let delete = if i == n {
            false
        } else if j == m {
            true
        } else {
            let d = dp[at(i + 1, j)];
            let ins = dp[at(i, j + 1)];
            if d == ins {
                (i + j) % 2 == 0
            } else {
                d > ins
            }
        };

        if delete {
            i += 1;
            end += 1;
        } else {
            replacement.push(new[j]);
            j += 1;
        }
    }

    if active {
        result.push(Change {
            start,
            end,
            replacement,
            side,
        });
    }
    result
}

fn overlap(a: &Change<'_>, b: &Change<'_>) -> bool {
    match (a.start == a.end, b.start == b.end) {
        (true, true) => a.start == b.start,
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn side_projection<'a>(
    base: &[&'a str],
    region: &Region,
    all: &[Change<'a>],
    side: usize,
) -> Vec<&'a str> {
    let mut selected: Vec<_> = region
        .ids
        .iter()
        .map(|&id| &all[id])
        .filter(|c| c.side == side)
        .collect();
    selected.sort_by_key(|c| (c.start, c.end));

    let mut result = Vec::new();
    let mut cursor = region.start;
    for c in selected {
        result.extend(base[cursor..c.start].iter().copied());
        result.extend(c.replacement.iter().copied());
        cursor = c.end;
    }
    result.extend(base[cursor..region.end].iter().copied());
    result
}

fn append(out: &mut Vec<String>, items: &[&str]) {
    for item in items {
        out.push((*item).to_owned());
    }
}

pub fn merge_three_way(
    base_text: &str,
    left_text: &str,
    right_text: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let texts = [base_text, left_text, right_text];
    let sides = [MergeSide::Base, MergeSide::Left, MergeSide::Right];

    for i in 0..3 {
        if let Some(byte) = texts[i].bytes().position(|b| b == b'\r') {
            return Err(MergeError::CarriageReturn {
                side: sides[i].clone(),
                byte,
            });
        }
    }
    for i in 0..3 {
        if !texts[i].is_empty() && !texts[i].ends_with('\n') {
            return Err(MergeError::MissingFinalLf {
                side: sides[i].clone(),
            });
        }
    }

    let base = lines(base_text);
    let left = lines(left_text);
    let right = lines(right_text);
    if base.len() > 128 || left.len() > 128 || right.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base.len(),
            left_len: left.len(),
            right_len: right.len(),
        });
    }

    let mut changes = diff(&base, &left, 0);
    changes.extend(diff(&base, &right, 1));

    let mut parent: Vec<usize> = (0..changes.len()).collect();
    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }
    for i in 0..changes.len() {
        for j in i + 1..changes.len() {
            if overlap(&changes[i], &changes[j]) {
                let a = find(&mut parent, i);
                let b = find(&mut parent, j);
                if a != b {
                    parent[b] = a;
                }
            }
        }
    }

    let mut regions: Vec<Region> = Vec::new();
    for id in 0..changes.len() {
        let representative = find(&mut parent, id);
        if let Some(region) = regions.iter_mut().find(|r| {
            r.ids
                .first()
                .map(|&first| find(&mut parent, first) == representative)
                .unwrap_or(false)
        }) {
            region.ids.push(id);
        } else {
            regions.push(Region {
                ids: vec![id],
                start: changes[id].start,
                end: changes[id].end,
            });
        }
    }

    for region in &mut regions {
        region.start = region.ids.iter().map(|&id| changes[id].start).min().unwrap();
        region.end = region.ids.iter().map(|&id| changes[id].end).max().unwrap();
    }
    regions.sort_by_key(|r| (r.start, usize::from(r.start != r.end), r.end));

    let mut out = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut cursor = 0;

    for region in regions {
        out.extend(base[cursor..region.start].iter().map(|s| (*s).to_owned()));

        let base_part = base[region.start..region.end].to_vec();
        let left_part = side_projection(&base, &region, &changes, 0);
        let right_part = side_projection(&base, &region, &changes, 1);

        if left_part == right_part {
            append(&mut out, &left_part);
        } else if left_part == base_part {
            append(&mut out, &right_part);
        } else if right_part == base_part {
            append(&mut out, &left_part);
        } else {
            match resolution {
                Resolution::PreferLeft => append(&mut out, &left_part),
                Resolution::PreferRight => append(&mut out, &right_part),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = out.len();
                    let base_len = region.end - region.start;
                    let marker_start = if base_len == 0 {
                        region.start
                    } else {
                        region.start + 1
                    };
                    out.push(format!("<<<<<<< LEFT {ordinal}"));
                    append(&mut out, &left_part);
                    out.push(format!("||||||| BASE {marker_start},{base_len}"));
                    append(&mut out, &base_part);
                    out.push("=======".to_owned());
                    append(&mut out, &right_part);
                    out.push(format!(">>>>>>> RIGHT {ordinal}"));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: region.start,
                        base_len,
                        output_start,
                        output_len: out.len() - output_start,
                    });
                }
            }
        }

        cursor = region.end;
    }

    out.extend(base[cursor..].iter().map(|s| (*s).to_owned()));
    let mut text = String::new();
    for line in out {
        text.push_str(&line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change {
    start: usize,
    end: usize,
    replacement: Vec<String>,
    side: usize,
}

#[derive(Clone)]
struct Component {
    start: usize,
    end: usize,
    changes: Vec<Change>,
}

fn lines(s: &str) -> Vec<String> {
    if s.is_empty() {
        return Vec::new();
    }
    s.strip_suffix('\n')
        .unwrap()
        .split('\n')
        .map(str::to_owned)
        .collect()
}

fn edits(old: &[String], new: &[String]) -> Vec<Change> {
    let n = old.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];
    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if old[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j, mut base_pos) = (0, 0, 0);
    let mut active_start: Option<usize> = None;
    let mut deletes = 0usize;
    let mut inserted = Vec::new();
    let mut changes = Vec::new();

    let flush = |changes: &mut Vec<Change>,
                 active_start: &mut Option<usize>,
                 deletes: &mut usize,
                 inserted: &mut Vec<String>,
                 base_pos: usize| {
        if let Some(start) = active_start.take() {
            changes.push(Change {
                start,
                end: start + *deletes,
                replacement: std::mem::take(inserted),
                side: 0,
            });
            *deletes = 0;
            let _ = base_pos;
        }
    };

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            flush(
                &mut changes,
                &mut active_start,
                &mut deletes,
                &mut inserted,
                base_pos,
            );
            i += 1;
            j += 1;
            base_pos += 1;
        } else if i == n {
            active_start.get_or_insert(base_pos);
            inserted.push(new[j].clone());
            j += 1;
        } else if j == m {
            active_start.get_or_insert(base_pos);
            deletes += 1;
            i += 1;
            base_pos += 1;
        } else {
            let del = lcs[i + 1][j];
            let ins = lcs[i][j + 1];
            let take_delete = if del > ins {
                true
            } else if ins > del {
                false
            } else {
                (i + j) % 2 == 0
            };
            if take_delete {
                active_start.get_or_insert(base_pos);
                deletes += 1;
                i += 1;
                base_pos += 1;
            } else {
                active_start.get_or_insert(base_pos);
                inserted.push(new[j].clone());
                j += 1;
            }
        }
    }
    flush(
        &mut changes,
        &mut active_start,
        &mut deletes,
        &mut inserted,
        base_pos,
    );
    changes
}

fn interacts(a: &Change, b: &Change) -> bool {
    let az = a.start == a.end;
    let bz = b.start == b.end;
    match (az, bz) {
        (true, true) => a.start == b.start,
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn projection(base: &[String], start: usize, end: usize, changes: &[Change]) -> Vec<String> {
    let mut out = Vec::new();
    let mut cursor = start;
    for c in changes {
        if c.start < cursor || c.start > end || c.end > end {
            continue;
        }
        out.extend(base[cursor..c.start].iter().cloned());
        out.extend(c.replacement.iter().cloned());
        cursor = c.end;
    }
    out.extend(base[cursor..end].iter().cloned());
    out
}

fn emit(out: &mut Vec<String>, s: &str) {
    out.push(s.to_owned());
}

pub fn merge_three_way(
    base: &str,
    left: &str,
    right: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (s, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if let Some(byte) = s.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }
    for (s, side) in [
        (base, MergeSide::Base),
        (left, MergeSide::Left),
        (right, MergeSide::Right),
    ] {
        if !s.is_empty() && !s.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let b = lines(base);
    let l = lines(left);
    let r = lines(right);
    if b.len() > 128 || l.len() > 128 || r.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: b.len(),
            left_len: l.len(),
            right_len: r.len(),
        });
    }

    let mut left_changes = edits(&b, &l);
    let mut right_changes = edits(&b, &r);
    for c in &mut left_changes {
        c.side = 0;
    }
    for c in &mut right_changes {
        c.side = 1;
    }
    let all: Vec<Change> = left_changes
        .iter()
        .chain(right_changes.iter())
        .cloned()
        .collect();

    let mut seen = vec![false; all.len()];
    let mut components = Vec::new();
    for seed in 0..all.len() {
        if seen[seed] {
            continue;
        }
        seen[seed] = true;
        let mut stack = vec![seed];
        let mut members = Vec::new();
        while let Some(i) = stack.pop() {
            members.push(all[i].clone());
            for j in 0..all.len() {
                if !seen[j] && interacts(&all[i], &all[j]) {
                    seen[j] = true;
                    stack.push(j);
                }
            }
        }
        let start = members.iter().map(|c| c.start).min().unwrap();
        let end = members.iter().map(|c| c.end).max().unwrap();
        members.sort_by_key(|c| (c.start, c.end, c.side));
        components.push(Component {
            start,
            end,
            changes: members,
        });
    }
    components.sort_by_key(|c| (c.start, usize::from(c.start != c.end), c.end));

    let mut output = Vec::new();
    let mut conflicts = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        output.extend(b[base_cursor..component.start].iter().cloned());
        let base_projection = b[component.start..component.end].to_vec();
        let lc: Vec<Change> = component
            .changes
            .iter()
            .filter(|c| c.side == 0)
            .cloned()
            .collect();
        let rc: Vec<Change> = component
            .changes
            .iter()
            .filter(|c| c.side == 1)
            .cloned()
            .collect();
        let lp = projection(&b, component.start, component.end, &lc);
        let rp = projection(&b, component.start, component.end, &rc);

        if lp == rp {
            output.extend(lp);
        } else if lp == base_projection {
            output.extend(rp);
        } else if rp == base_projection {
            output.extend(lp);
        } else {
            match resolution {
                Resolution::PreferLeft => output.extend(lp),
                Resolution::PreferRight => output.extend(rp),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = output.len();
                    emit(&mut output, &format!("<<<<<<< LEFT {ordinal}"));
                    output.extend(lp.iter().cloned());
                    let c = component.end - component.start;
                    let s = if c > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    emit(&mut output, &format!("||||||| BASE {s},{c}"));
                    output.extend(base_projection.iter().cloned());
                    emit(&mut output, "=======");
                    output.extend(rp.iter().cloned());
                    emit(&mut output, &format!(">>>>>>> RIGHT {ordinal}"));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len: c,
                        output_start,
                        output_len: 4 + lp.len() + base_projection.len() + rp.len(),
                    });
                }
            }
        }
        base_cursor = component.end;
    }
    output.extend(b[base_cursor..].iter().cloned());

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }
    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
