- Validation follows R6 exactly: CR checks by Base/Left/Right, then final LF checks in that order, then line-count limits.
- Empty text has zero lines; a nonempty input ending in LF has no extra trailing line.
- Insertions interact with other insertions only at the same anchor, and with replacements only strictly inside their interval.
- Components are ordered by start, with zero-length regions before positive-length regions at the same start.
- Markers count as output lines; `output_start` and `output_len` use zero-based and line-count semantics from R40–R41.
- Both implementations use the specified LCS tie rule; Solution B stores the table in a flat vector.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,
    },
    TooLarge {
        base_len: usize,
        left_len: usize,
        right_len: usize,
    },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {
    pub ordinal: usize,
    pub base_start: usize,
    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,
}

#[derive(Clone)]
struct Change<'a> {
    start: usize,
    end: usize,
    replacement: Vec<&'a str>,
    side: MergeSide,
}

#[derive(Clone)]
struct Component {
    changes: Vec<usize>,
    start: usize,
    end: usize,
}

fn split_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        Vec::new()
    } else {
        text[..text.len() - 1].split('\n').collect()
    }
}

fn make_changes<'a>(base: &[&'a str], new: &[&'a str], side: MergeSide) -> Vec<Change<'a>> {
    let n = base.len();
    let m = new.len();
    let mut lcs = vec![vec![0usize; m + 1]; n + 1];

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            lcs[i][j] = if base[i] == new[j] {
                1 + lcs[i + 1][j + 1]
            } else {
                lcs[i + 1][j].max(lcs[i][j + 1])
            };
        }
    }

    let (mut i, mut j) = (0usize, 0usize);
    let mut changes = Vec::new();
    let mut in_change = false;
    let mut start = 0usize;
    let mut end = 0usize;
    let mut replacement = Vec::new();

    let finish = |changes: &mut Vec<Change<'a>>,
                  in_change: &mut bool,
                  start: &mut usize,
                  end: &mut usize,
                  replacement: &mut Vec<&'a str>| {
        if *in_change {
            changes.push(Change {
                start: *start,
                end: *end,
                replacement: std::mem::take(replacement),
                side: side.clone(),
            });
            *in_change = false;
        }
    };

    while i < n || j < m {
        if i < n && j < m && base[i] == new[j] {
            finish(
                &mut changes,
                &mut in_change,
                &mut start,
                &mut end,
                &mut replacement,
            );
            i += 1;
            j += 1;
        } else {
            if !in_change {
                in_change = true;
                start = i;
                end = i;
            }
            let delete = if i == n {
                false
            } else if j == m {
                true
            } else {
                let del = lcs[i + 1][j];
                let ins = lcs[i][j + 1];
                if del > ins {
                    true
                } else if ins > del {
                    false
                } else {
                    (i + j) % 2 == 0
                }
            };

            if delete {
                i += 1;
                end += 1;
            } else {
                replacement.push(new[j]);
                j += 1;
            }
        }
    }

    finish(
        &mut changes,
        &mut in_change,
        &mut start,
        &mut end,
        &mut replacement,
    );
    changes
}

fn interacts(a: &Change<'_>, b: &Change<'_>) -> bool {
    let a_zero = a.start == a.end;
    let b_zero = b.start == b.end;
    match (a_zero, b_zero) {
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, true) => a.start == b.start,
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn project<'a>(
    base: &[&'a str],
    component: &Component,
    changes: &[Change<'a>],
    side: MergeSide,
) -> Vec<&'a str> {
    let mut selected: Vec<&Change<'a>> = component
        .changes
        .iter()
        .map(|&idx| &changes[idx])
        .filter(|change| change.side == side)
        .collect();
    selected.sort_by_key(|change| (change.start, change.end));

    let mut out = Vec::new();
    let mut cursor = component.start;
    for change in selected {
        for line in &base[cursor..change.start] {
            out.push(*line);
        }
        out.extend(change.replacement.iter().copied());
        cursor = change.end;
    }
    for line in &base[cursor..component.end] {
        out.push(*line);
    }
    out
}

fn append_lines(out: &mut Vec<String>, lines: &[&str]) {
    out.extend(lines.iter().map(|line| (*line).to_string()));
}

pub fn merge_three_way(
    base_text: &str,
    left_text: &str,
    right_text: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    for (text, side) in [
        (base_text, MergeSide::Base),
        (left_text, MergeSide::Left),
        (right_text, MergeSide::Right),
    ] {
        if let Some(byte) = text.as_bytes().iter().position(|&b| b == b'\r') {
            return Err(MergeError::CarriageReturn { side, byte });
        }
    }

    for (text, side) in [
        (base_text, MergeSide::Base),
        (left_text, MergeSide::Left),
        (right_text, MergeSide::Right),
    ] {
        if !text.is_empty() && !text.ends_with('\n') {
            return Err(MergeError::MissingFinalLf { side });
        }
    }

    let base = split_lines(base_text);
    let left = split_lines(left_text);
    let right = split_lines(right_text);

    if base.len() > 128 || left.len() > 128 || right.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base.len(),
            left_len: left.len(),
            right_len: right.len(),
        });
    }

    let mut changes = make_changes(&base, &left, MergeSide::Left);
    changes.extend(make_changes(&base, &right, MergeSide::Right));

    let mut parent: Vec<usize> = (0..changes.len()).collect();
    fn root(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = root(parent, parent[x]);
        }
        parent[x]
    }
    for i in 0..changes.len() {
        for j in i + 1..changes.len() {
            if interacts(&changes[i], &changes[j]) {
                let ri = root(&mut parent, i);
                let rj = root(&mut parent, j);
                if ri != rj {
                    parent[rj] = ri;
                }
            }
        }
    }

    let mut components = Vec::<Component>::new();
    for idx in 0..changes.len() {
        let r = root(&mut parent, idx);
        if let Some(component) = components.iter_mut().find(|c| {
            c.changes
                .first()
                .map(|&existing| root(&mut parent, existing) == r)
                .unwrap_or(false)
        }) {
            component.changes.push(idx);
        } else {
            components.push(Component {
                changes: vec![idx],
                start: changes[idx].start,
                end: changes[idx].end,
            });
        }
    }

    for component in &mut components {
        component.start = component
            .changes
            .iter()
            .map(|&idx| changes[idx].start)
            .min()
            .unwrap_or(0);
        component.end = component
            .changes
            .iter()
            .map(|&idx| changes[idx].end)
            .max()
            .unwrap_or(component.start);
    }
    components.sort_by_key(|c| (c.start, if c.start == c.end { 0 } else { 1 }, c.end));

    let mut output = Vec::<String>::new();
    let mut records = Vec::new();
    let mut base_cursor = 0usize;

    for component in components {
        for line in &base[base_cursor..component.start] {
            output.push((*line).to_string());
        }

        let base_projection: Vec<&str> = base[component.start..component.end].to_vec();
        let left_projection = project(&base, &component, &changes, MergeSide::Left);
        let right_projection = project(&base, &component, &changes, MergeSide::Right);

        if left_projection == right_projection {
            append_lines(&mut output, &left_projection);
        } else if left_projection == base_projection {
            append_lines(&mut output, &right_projection);
        } else if right_projection == base_projection {
            append_lines(&mut output, &left_projection);
        } else {
            match resolution {
                Resolution::PreferLeft => append_lines(&mut output, &left_projection),
                Resolution::PreferRight => append_lines(&mut output, &right_projection),
                Resolution::Mark => {
                    let ordinal = records.len() + 1;
                    let output_start = output.len();
                    let base_len = component.end - component.start;
                    let base_line_number = if base_len > 0 {
                        component.start + 1
                    } else {
                        component.start
                    };
                    output.push(format!("<<<<<<< LEFT {ordinal}"));
                    append_lines(&mut output, &left_projection);
                    output.push(format!("||||||| BASE {base_line_number},{base_len}"));
                    append_lines(&mut output, &base_projection);
                    output.push("=======".to_string());
                    append_lines(&mut output, &right_projection);
                    output.push(format!(">>>>>>> RIGHT {ordinal}"));
                    records.push(Conflict {
                        ordinal,
                        base_start: component.start,
                        base_len,
                        output_start,
                        output_len: output.len() - output_start,
                    });
                }
            }
        }
        base_cursor = component.end;
    }

    for line in &base[base_cursor..] {
        output.push((*line).to_string());
    }

    let mut text = String::new();
    for line in &output {
        text.push_str(line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: records.len(),
        regions: records,
    })
}
```

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeSide {
    Base,
    Left,
    Right,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Resolution {
    Mark,
    PreferLeft,
    PreferRight,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum MergeError {
    CarriageReturn {
        side: MergeSide,
        byte: usize,
    },
    MissingFinalLf {
        side: MergeSide,

        base_len: usize,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Conflict {

    pub base_len: usize,
    pub output_start: usize,
    pub output_len: usize,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeResult {
    pub text: String,
    pub conflicts: usize,
    pub regions: Vec<Conflict>,

struct Change<'a> {
    start: usize,
    end: usize,
    replacement: Vec<&'a str>,
    side: usize,

#[derive(Clone)]
struct Region {
    ids: Vec<usize>,
    start: usize,
    end: usize,
}

fn lines(s: &str) -> Vec<&str> {
    if s.is_empty() {
        vec![]
    } else {
        s.split_terminator('\n').collect()
    }
}

fn diff<'a>(old: &[&'a str], new: &[&'a str], side: usize) -> Vec<Change<'a>> {
    let (n, m) = (old.len(), new.len());
    let width = m + 1;
    let mut dp = vec![0usize; (n + 1) * (m + 1)];
    let at = |i: usize, j: usize| i * width + j;

    for i in (0..n).rev() {
        for j in (0..m).rev() {
            let value = if old[i] == new[j] {
                1 + dp[at(i + 1, j + 1)]
            } else {
                dp[at(i + 1, j)].max(dp[at(i, j + 1)])
            };
            dp[at(i, j)] = value;
        }
    }

    let mut i = 0;
    let mut j = 0;
    let mut result = Vec::new();
    let mut active = false;
    let mut start = 0;
    let mut end = 0;
    let mut replacement = Vec::new();

    while i < n || j < m {
        if i < n && j < m && old[i] == new[j] {
            if active {
                result.push(Change {
                    start,
                    end,
                    replacement: std::mem::take(&mut replacement),
                    side,
                });
                active = false;
            }
            i += 1;
            j += 1;
            continue;
        }

        if !active {
            active = true;
            start = i;
            end = i;
        }

        let delete = if i == n {
            false
        } else if j == m {
            true
        } else {
            let d = dp[at(i + 1, j)];
            let ins = dp[at(i, j + 1)];
            if d == ins {
                (i + j) % 2 == 0
            } else {
                d > ins
            }
        };

        if delete {
            i += 1;
            end += 1;
        } else {
            replacement.push(new[j]);
            j += 1;
        }
    }

    if active {
        result.push(Change {
            start,
            end,
            replacement,
            side,
        });
    }
    result
}

fn overlap(a: &Change<'_>, b: &Change<'_>) -> bool {
    match (a.start == a.end, b.start == b.end) {
        (true, true) => a.start == b.start,
        (false, false) => a.start.max(b.start) < a.end.min(b.end),
        (true, false) => b.start < a.start && a.start < b.end,
        (false, true) => a.start < b.start && b.start < a.end,
    }
}

fn side_projection<'a>(
    base: &[&'a str],
    region: &Region,
    all: &[Change<'a>],
    side: usize,
) -> Vec<&'a str> {
    let mut selected: Vec<_> = region
        .ids
        .iter()
        .map(|&id| &all[id])
        .filter(|c| c.side == side)
        .collect();
    selected.sort_by_key(|c| (c.start, c.end));

    let mut result = Vec::new();
    let mut cursor = region.start;
    for c in selected {
        result.extend(base[cursor..c.start].iter().copied());
        result.extend(c.replacement.iter().copied());
        cursor = c.end;
    }
    result.extend(base[cursor..region.end].iter().copied());
    result
}

fn append(out: &mut Vec<String>, items: &[&str]) {
    for item in items {
        out.push((*item).to_owned());
    }
}

pub fn merge_three_way(
    base_text: &str,
    left_text: &str,
    right_text: &str,
    resolution: Resolution,
) -> Result<MergeResult, MergeError> {
    let texts = [base_text, left_text, right_text];
    let sides = [MergeSide::Base, MergeSide::Left, MergeSide::Right];

    for i in 0..3 {
        if let Some(byte) = texts[i].bytes().position(|b| b == b'\r') {
            return Err(MergeError::CarriageReturn {
                side: sides[i].clone(),
                byte,
            });
        }
    }
    for i in 0..3 {
        if !texts[i].is_empty() && !texts[i].ends_with('\n') {
            return Err(MergeError::MissingFinalLf {
                side: sides[i].clone(),
            });
        }
    }

    let base = lines(base_text);
    let left = lines(left_text);
    let right = lines(right_text);
    if base.len() > 128 || left.len() > 128 || right.len() > 128 {
        return Err(MergeError::TooLarge {
            base_len: base.len(),
            left_len: left.len(),
            right_len: right.len(),
        });
    }

    let mut changes = diff(&base, &left, 0);
    changes.extend(diff(&base, &right, 1));

    let mut parent: Vec<usize> = (0..changes.len()).collect();
    fn find(parent: &mut [usize], x: usize) -> usize {
        if parent[x] != x {
            parent[x] = find(parent, parent[x]);
        }
        parent[x]
    }
    for i in 0..changes.len() {
        for j in i + 1..changes.len() {
            if overlap(&changes[i], &changes[j]) {
                let a = find(&mut parent, i);
                let b = find(&mut parent, j);
                if a != b {
                    parent[b] = a;
                }
            }
        }
    }

    let mut regions: Vec<Region> = Vec::new();
    for id in 0..changes.len() {
        let representative = find(&mut parent, id);
        if let Some(region) = regions.iter_mut().find(|r| {
            r.ids
                .first()
                .map(|&first| find(&mut parent, first) == representative)
                .unwrap_or(false)
        }) {
            region.ids.push(id);
        } else {
            regions.push(Region {
                ids: vec![id],
                start: changes[id].start,
                end: changes[id].end,
            });
        }
    }

    for region in &mut regions {
        region.start = region.ids.iter().map(|&id| changes[id].start).min().unwrap();
        region.end = region.ids.iter().map(|&id| changes[id].end).max().unwrap();
    }
    regions.sort_by_key(|r| (r.start, usize::from(r.start != r.end), r.end));

    let mut out = Vec::<String>::new();
    let mut conflicts = Vec::new();
    let mut cursor = 0;

    for region in regions {
        out.extend(base[cursor..region.start].iter().map(|s| (*s).to_owned()));

        let base_part = base[region.start..region.end].to_vec();
        let left_part = side_projection(&base, &region, &changes, 0);
        let right_part = side_projection(&base, &region, &changes, 1);

        if left_part == right_part {
            append(&mut out, &left_part);
        } else if left_part == base_part {
            append(&mut out, &right_part);
        } else if right_part == base_part {
            append(&mut out, &left_part);
        } else {
            match resolution {
                Resolution::PreferLeft => append(&mut out, &left_part),
                Resolution::PreferRight => append(&mut out, &right_part),
                Resolution::Mark => {
                    let ordinal = conflicts.len() + 1;
                    let output_start = out.len();
                    let base_len = region.end - region.start;
                    let marker_start = if base_len == 0 {
                        region.start
                    } else {
                        region.start + 1
                    };
                    out.push(format!("<<<<<<< LEFT {ordinal}"));
                    append(&mut out, &left_part);
                    out.push(format!("||||||| BASE {marker_start},{base_len}"));
                    append(&mut out, &base_part);
                    out.push("=======".to_owned());
                    append(&mut out, &right_part);
                    out.push(format!(">>>>>>> RIGHT {ordinal}"));
                    conflicts.push(Conflict {
                        ordinal,
                        base_start: region.start,
                        base_len,
                        output_start,
                        output_len: out.len() - output_start,
                    });
                }
            }
        }

        cursor = region.end;
    }

    out.extend(base[cursor..].iter().map(|s| (*s).to_owned()));
    let mut text = String::new();
    for line in out {
        text.push_str(&line);
        text.push('\n');
    }

    Ok(MergeResult {
        text,
        conflicts: conflicts.len(),
        regions: conflicts,
    })
}
```
