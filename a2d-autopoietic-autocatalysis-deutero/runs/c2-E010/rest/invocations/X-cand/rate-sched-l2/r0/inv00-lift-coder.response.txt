- Layout validation scans the entire input first; I use the specified byte-offset precedence.
- A second slash is detected before base parsing; base bounds and range order are checked before step syntax.
- Poll intervals include zero on the first poll and exclude boundaries at or before the previous poll time thereafter.
- `Once` selects the latest match in the interval; `Skip` emits only when `now` itself is a matching minute boundary.
- The time limit keeps enumeration small; boundaries and arithmetic are handled without overflowing `u64`.
- For Solution B, I use a different representation for Cron’s selected values; parsing and polling semantics remain the same.

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();

        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let mut minutes = [false; 60];
        let mut hours = [false; 24];
        parse_field(text, 0, split, 60, &mut minutes)?;
        parse_field(text, split + 1, bytes.len(), 24, &mut hours)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut item_start = start;

    loop {
        let item_end = (item_start..end)
            .find(|&i| bytes[i] == b',')
            .unwrap_or(end);

        if item_start == item_end {
            return Err(CronError::Syntax(item_start));
        }
        parse_item(text, item_start, item_end, limit, selected)?;

        if item_end == end {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(())
}

fn parse_item(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let slash = (start..end).find(|&i| bytes[i] == b'/');
    if let Some(s) = slash {
        if let Some(second) = (s + 1..end).find(|&i| bytes[i] == b'/') {
            return Err(CronError::Syntax(second));
        }
        let base_has_range = (start..s).any(|i| bytes[i] == b'-');
        let base_is_star = s == start + 1 && bytes[start] == b'*';
        if !base_is_star && !base_has_range {
            return Err(CronError::Syntax(s));
        }
    }

    let base_end = slash.unwrap_or(end);
    let (range_start, range_end, dash) = if base_end == start + 1 && bytes[start] == b'*' {
        (0usize, limit - 1, None)
    } else if let Some(d) = (start..base_end).find(|&i| bytes[i] == b'-') {
        let left = parse_number(text, start, d, limit, false)?;
        let right = parse_number(text, d + 1, base_end, limit, false)?;
        if left > right {
            return Err(CronError::Descending(d));
        }
        (left, right, Some(d))
    } else {
        let n = parse_number(text, start, base_end, limit, false)?;
        (n, n, None)
    };

    let step = if let Some(s) = slash {
        parse_number(text, s + 1, end, if limit == 60 { 61 } else { 25 }, true)?
    } else {
        1
    };

    let _ = dash;
    let mut value = range_start;
    while value <= range_end {
        selected[value] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }
    Ok(())
}

fn parse_number(
    text: &str,
    start: usize,
    end: usize,
    max: usize,
    is_step: bool,
) -> Result<usize, CronError> {
    let bytes = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }

    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value: usize = 0;
    for i in start..end {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((bytes[i] - b'0') as usize))
        {
            Some(v) => v,
            None => {
                return Err(if is_step {
                    CronError::Step(start)
                } else {
                    CronError::Bounds(start)
                })
            }
        };
    }

    if is_step {
        if value == 0 || value > max {
            return Err(CronError::Step(start));
        }
    } else if value >= max {
        return Err(CronError::Bounds(start));
    }
    Ok(value)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    added_after: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        self.ids.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order: self.jobs.len(),
            added_after: self.last_poll,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let first_minute = match self.last_poll {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let last_minute = now / 60_000;
        let mut tagged: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let effective_first = match job.added_after {
                Some(added_at) => first_minute.max(added_at / 60_000 + 1),
                None => first_minute,
            };
            if effective_first > last_minute {
                continue;
            }

            match job.policy {
                Policy::All => {
                    for minute in effective_first..=last_minute {
                        if job.cron.matches(minute) {
                            tagged.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: minute * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Once => {
                    let mut latest = None;
                    for minute in effective_first..=last_minute {
                        if job.cron.matches(minute) {
                            latest = Some(minute);
                        }
                    }
                    if let Some(minute) = latest {
                        tagged.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: minute * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let minute = now / 60_000;
                        if minute >= effective_first && job.cron.matches(minute) {
                            tagged.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        tagged.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        Ok(tagged.into_iter().map(|x| x.0).collect())
    }
}
```

### Solution B
```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    minute_bits: u64,
    hour_bits: u32,
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();
        for (i, &byte) in bytes.iter().enumerate() {
            if byte.is_ascii_whitespace() && byte != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let mut separator = None;
        for (i, &byte) in bytes.iter().enumerate() {
            if byte == b' ' {
                if separator.is_some() {
                    return Err(CronError::Layout(i));
                }
                separator = Some(i);
            }
        }
        let sep = match separator {
            Some(i) => i,
            None => return Err(CronError::Layout(text.len())),
        };
        if sep == 0 || sep + 1 == bytes.len() {
            return Err(CronError::Layout(sep));
        }

        let mut minute_bits = 0u64;
        let mut hour_bits = 0u32;
        parse_field(text, 0, sep, 60, &mut minute_bits)?;
        parse_field(text, sep + 1, bytes.len(), 24, &mut hour_bits)?;
        Ok(Self {
            minute_bits,
            hour_bits,
        })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minute_bits & (1u64 << (minute % 60)) != 0
            && self.hour_bits & (1u32 << ((minute / 60) % 24)) != 0
    }
}

fn parse_field<T: BitField>(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut T,
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let mut from = start;
    loop {
        let to = (from..end).find(|&i| bytes[i] == b',').unwrap_or(end);
        if from == to {
            return Err(CronError::Syntax(from));
        }
        parse_item(text, from, to, limit, selected)?;
        if to == end {
            return Ok(());
        }
        from = to + 1;
    }
}

trait BitField {
    fn set(&mut self, index: usize);
}

impl BitField for u64 {
    fn set(&mut self, index: usize) {
        *self |= 1u64 << index;
    }
}
impl BitField for u32 {
    fn set(&mut self, index: usize) {
        *self |= 1u32 << index;
    }
}

fn parse_item<T: BitField>(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    selected: &mut T,
) -> Result<(), CronError> {
    let bytes = text.as_bytes();
    let slash = (start..end).find(|&i| bytes[i] == b'/');
    if let Some(s) = slash {
        if let Some(second) = (s + 1..end).find(|&i| bytes[i] == b'/') {
            return Err(CronError::Syntax(second));
        }
        let star = s == start + 1 && bytes[start] == b'*';
        let range = (start..s).any(|i| bytes[i] == b'-');
        if !star && !range {
            return Err(CronError::Syntax(s));
        }
    }

    let base_end = slash.unwrap_or(end);
    let (first, last) = if base_end == start + 1 && bytes[start] == b'*' {
        (0, limit - 1)
    } else if let Some(dash) = (start..base_end).find(|&i| bytes[i] == b'-') {
        let a = number(text, start, dash, limit, false)?;
        let b = number(text, dash + 1, base_end, limit, false)?;
        if a > b {
            return Err(CronError::Descending(dash));
        }
        (a, b)
    } else {
        let n = number(text, start, base_end, limit, false)?;
        (n, n)
    };

    let step = match slash {
        Some(s) => number(text, s + 1, end, if limit == 60 { 60 } else { 24 }, true)?,
        None => 1,
    };
    let mut n = first;
    while n <= last {
        selected.set(n);
        n += step;
    }
    Ok(())
}

fn number(
    text: &str,
    start: usize,
    end: usize,
    limit: usize,
    step: bool,
) -> Result<usize, CronError> {
    let bytes = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0usize;
    for i in start..end {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((bytes[i] - b'0') as usize))
        {
            Some(v) => v,
            None => {
                return Err(if step {
                    CronError::Step(start)
                } else {
                    CronError::Bounds(start)
                })
            }
        };
    }

    if step {
        if value == 0 || value > limit {
            return Err(CronError::Step(start));
        }
    } else if value >= limit {
        return Err(CronError::Bounds(start));
    }
    Ok(value)
}

pub enum Policy {
    Skip,
    Once,

}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
    added_after: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    ids: Vec<String>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            ids: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|value| value == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.ids.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
            added_after: self.last_poll,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let low = match self.last_poll {
            None => 0,
            Some(last) => last / 60_000 + 1,
        };
        let high = now / 60_000;
        let mut output: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let low = match job.added_after {
                Some(t) => low.max(t / 60_000 + 1),
                None => low,
            };
            if low > high {
                continue;
            }

            match job.policy {
                Policy::All => {
                    for m in low..=high {
                        if job.cron.matches(m) {
                            output.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: m * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Once => {
                    if let Some(m) = (low..=high).rev().find(|&m| job.cron.matches(m)) {
                        output.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: m * 60_000,
                            },
                            job.priority,
                            job.order,
                        ));
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let m = now / 60_000;
                        if m >= low && job.cron.matches(m) {
                            output.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        output.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(output.into_iter().map(|entry| entry.0).collect())
    }
}
```
