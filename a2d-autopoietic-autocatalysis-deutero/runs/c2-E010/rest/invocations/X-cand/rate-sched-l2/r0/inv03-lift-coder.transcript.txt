fn parse_number(
    text: &str,
    start: usize,
    end: usize,
    max: usize,
) -> Result<usize, CronError> {
    let bytes = text.as_bytes();
    if start == end {
        return Err(CronError::Syntax(start));
    }
    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }
    let mut value: usize = 0;
    for i in start..end {
        value = match value
            .checked_mul(10)
            .and_then(|v| v.checked_add((bytes[i] - b'0') as usize))
        {
            Some(v) => v,
            None => return Err(CronError::Bounds(start)),
        };
    }
    if value > max {
        return Err(CronError::Bounds(start));
    }
    Ok(value)
}

fn parse_field<const N: usize>(
    field: &str,
    field_offset: usize,
    max: usize,
) -> Result<[bool; N], CronError> {
    let mut selected = [false; N];
    let bytes = field.as_bytes();
    let mut item_start = 0;

    loop {
        let mut item_end = item_start;
        while item_end < bytes.len() && bytes[item_end] != b',' {
            item_end += 1;
        }
        if item_start == item_end {
            return Err(CronError::Syntax(field_offset + item_start));
        }

        let mut slash = None;
        for i in item_start..item_end {
            if bytes[i] == b'/' {
                if slash.is_some() {
                    return Err(CronError::Syntax(field_offset + i));
                }
                slash = Some(i);
            }
        }

        let base_end = slash.unwrap_or(item_end);
        let step = if let Some(slash_pos) = slash {
            let base_is_star = base_end == item_start + 1 && bytes[item_start] == b'*';
            let base_is_range = bytes[item_start..base_end].contains(&b'-');
            if !base_is_star && !base_is_range {
                return Err(CronError::Syntax(field_offset + slash_pos));
            }
            None
        } else {
            None
        };

        let is_star = base_end == item_start + 1 && bytes[item_start] == b'*';
        let dash = (item_start..base_end).find(|&i| bytes[i] == b'-');

        let (range_start, range_end, dash_pos) = if is_star {
            (0, max, None)
        } else if let Some(d) = dash {
            let left = parse_number(field, item_start, d, max)?;
            let right = parse_number(field, d + 1, base_end, max)?;
            if left > right {
                return Err(CronError::Descending(field_offset + d));
            }
            (left, right, Some(d))
        } else {
            let n = parse_number(field, item_start, base_end, max)?;
            (n, n, None)
        };

        let step_value = if let Some(slash_pos) = slash {
            let step_start = slash_pos + 1;
            let step_n = parse_number(field, step_start, item_end, if max == 59 { 60 } else { 24 })?;
            if step_n == 0 {
                return Err(CronError::Step(field_offset + step_start));
            }
            step_n
        } else {
            1
        };

        let _ = step;
        let _ = dash_pos;
        let mut value = range_start;
        while value <= range_end {
            selected[value] = true;
            match value.checked_add(step_value) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }
    Ok(selected)
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();
        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }
        let spaces: Vec<usize> = bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| if b == b' ' { Some(i) } else { None })
            .collect();
        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }
        let split = spaces[0];
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let minutes = parse_field::<60>(&text[..split], 0, 59)?;
        let hours = parse_field::<24>(&text[split + 1..], split + 1, 23)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize] && self.hours[((minute / 60) % 24) as usize]
    }
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    order: usize,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if !(1..=16).contains(&id.len())
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.jobs.iter().any(|job| job.id == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;
        let order = self.jobs.len();
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            order,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        let lower = self.last_poll.map(|t| t / 60_000);
        let upper = now / 60_000;
        let mut emitted: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let first = match lower {
                None => 0,
                Some(last) => last.saturating_add(1),
            };
            let matching: Vec<u64> = if first <= upper {
                (first..=upper)
                    .filter(|&m| job.cron.matches(m))
                    .collect()
            } else {
                Vec::new()
            };

            let chosen: Vec<u64> = match job.policy {
                Policy::All => matching,
                Policy::Once => matching.last().copied().into_iter().collect(),
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let m = now / 60_000;
                        if job.cron.matches(m) && first <= m && m <= upper {
                            vec![m]
                        } else {
                            Vec::new()
                        }
                    } else {
                        Vec::new()
                    }
                }
            };

            for m in chosen {
                emitted.push((
                    Occurrence {
                        id: job.id.clone(),
                        at: m * 60_000,
                    },
                    job.priority,
                    job.order,
                ));
            }
        }

        emitted.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.last_poll = Some(now);
        Ok(emitted.into_iter().map(|x| x.0).collect())
    }
}

    Skip,

    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

fn checked_digits(text: &str, start: usize, end: usize) -> Result<u64, CronError> {
    if start == end {
        return Err(CronError::Syntax(start));
    }
    let bytes = text.as_bytes();
    for i in start..end {
        if !bytes[i].is_ascii_digit() {
            return Err(CronError::Syntax(i));
        }
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(CronError::Number(start));
    }

    let mut value = 0u64;
    for i in start..end {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((bytes[i] - b'0') as u64))
            .ok_or(CronError::Bounds(start))?;
    }
    Ok(value)
}

fn parse_number_b(
    text: &str,
    start: usize,
    end: usize,
    max: u64,
    is_step: bool,
) -> Result<u64, CronError> {
    let n = checked_digits(text, start, end)?;
    if n > max {
        return Err(if is_step {
            CronError::Step(start)
        } else {
            CronError::Bounds(start)
        });
    }
    Ok(n)
}

fn field_b<const N: usize>(
    text: &str,
    offset: usize,
    max: u64,
) -> Result<[bool; N], CronError> {
    let bytes = text.as_bytes();
    let mut out = [false; N];
    let mut pos = 0;

    while pos <= bytes.len() {
        let end = bytes[pos..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| pos + n)
            .unwrap_or(bytes.len());
        if end == pos {
            return Err(CronError::Syntax(offset + pos));
        }

        let mut slash_at = None;
        for i in pos..end {
            if bytes[i] == b'/' {
                if slash_at.is_some() {
                    return Err(CronError::Syntax(offset + i));
                }
                slash_at = Some(i);
            }
        }

        let base_end = slash_at.unwrap_or(end);
        let range_dash = (pos..base_end).find(|&i| bytes[i] == b'-');
        if let Some(slash) = slash_at {
            let star = base_end == pos + 1 && bytes[pos] == b'*';
            if !star && range_dash.is_none() {
                return Err(CronError::Syntax(offset + slash));
            }
        }

        let (first, last, dash_for_error) = if base_end == pos + 1 && bytes[pos] == b'*' {
            (0, max, None)
        } else if let Some(dash) = range_dash {
            let lo = parse_number_b(text, pos, dash, max, false)?;
            let hi = parse_number_b(text, dash + 1, base_end, max, false)?;
            if lo > hi {
                return Err(CronError::Descending(offset + dash));
            }
            (lo, hi, Some(dash))
        } else {
            let n = parse_number_b(text, pos, base_end, max, false)?;
            (n, n, None)
        };
        let _ = dash_for_error;

        let step = if let Some(slash) = slash_at {
            let start = slash + 1;
            let max_step = if max == 59 { 60 } else { 24 };
            let n = parse_number_b(text, start, end, max_step, true)?;
            if n == 0 {
                return Err(CronError::Step(offset + start));
            }
            n
        } else {
            1
        };

        let mut n = first;
        while n <= last {
            out[n as usize] = true;
            match n.checked_add(step) {
                Some(next) => n = next,
                None => break,
            }
        }

        if end == bytes.len() {
            break;
        }
        pos = end + 1;
    }
    Ok(out)
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        let bytes = text.as_bytes();
        for (i, &b) in bytes.iter().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let mut space = None;
        for (i, &b) in bytes.iter().enumerate() {
            if b == b' ' {
                if space.is_some() {
                    return Err(CronError::Layout(i));
                }
                space = Some(i);
            }
        }
        let Some(split) = space else {
            return Err(CronError::Layout(text.len()));
        };
        if split == 0 || split + 1 == bytes.len() {
            return Err(CronError::Layout(split));
        }

        let minutes = field_b::<60>(&text[..split], 0, 59)?;
        let hours = field_b::<24>(&text[split + 1..], split + 1, 23)?;
        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        let minute_of_hour = (minute % 60) as usize;
        let hour_of_day = ((minute / 60) % 24) as usize;
        self.minutes[minute_of_hour] && self.hours[hour_of_day]
    }
}

            return Err(Error::TimeRange);
        }

        let first_minute = self.watermark.map_or(0, |t| t / 60_000 + 1);
        let last_minute = now / 60_000;
        let mut results: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            match job.policy {
                Policy::All => {
                    if first_minute <= last_minute {
                        for m in first_minute..=last_minute {
                            if job.cron.matches(m) {
                                results.push((
                                    Occurrence {
                                        id: job.id.clone(),
                                        at: m * 60_000,
                                    },
                                    job.priority,
                                    job.order,
                                ));
                            }
                        }
                    }
                }
                Policy::Once => {
                    if first_minute <= last_minute {
                        let mut latest = None;
                        for m in first_minute..=last_minute {
                            if job.cron.matches(m) {
                                latest = Some(m);
                            }
                        }
                        if let Some(m) = latest {
                            results.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: m * 60_000,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
                Policy::Skip => {
                    if now % 60_000 == 0 {
                        let m = now / 60_000;
                        if m >= first_minute && m <= last_minute && job.cron.matches(m) {
                            results.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.order,
                            ));
                        }
                    }
                }
            }
        }

        results.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });
        self.watermark = Some(now);
        Ok(results.into_iter().map(|x| x.0).collect())
    }
}
