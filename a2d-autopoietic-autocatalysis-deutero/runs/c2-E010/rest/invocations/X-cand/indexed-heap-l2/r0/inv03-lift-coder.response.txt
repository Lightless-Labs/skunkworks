- Key trimming removes only ASCII spaces; validation error offsets refer to the original UTF-8 byte positions.
- The first non-space byte must be a letter; later bytes may also be digits or hyphens.
- Key errors take precedence over lookup errors, and missing keys precede direction errors.
- Priority comparisons use signed `i64` ordering directly, including `i64::MIN` and `i64::MAX`.
- Ties use value byte length, then insertion age; updates preserve age.
- I use a binary heap with an index map in both solutions; Solution B uses a separate heap implementation strategy.

### Solution A

```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
}

struct Entry {
    item: Item,
    age: u64,
}

pub struct IndexedHeap {
    heap: Vec<Entry>,
    indices: HashMap<String, usize>,
    next_age: u64,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            heap: Vec::new(),
            indices: HashMap::new(),
            next_age: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.heap.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    fn canonicalize(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut start = 0;
        let mut end = bytes.len();

        while start < end && bytes[start] == b' ' {
            start += 1;
        }
        while end > start && bytes[end - 1] == b' ' {
            end -= 1;
        }

        if start == end {
            return Err(Error::EmptyKey);
        }

        for (offset, &byte) in bytes[start..end].iter().enumerate() {
            let valid = if offset == 0 {
                byte.is_ascii_alphabetic()
            } else {
                byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
            };
            if !valid {
                return Err(Error::InvalidKey { at: start + offset });
            }
        }

        let trimmed = &input[start..end];
        Ok(trimmed.to_ascii_lowercase())
    }

    fn better(a: &Entry, b: &Entry) -> bool {
        a.item.priority < b.item.priority
            || (a.item.priority == b.item.priority
                && (a.item.value.len() < b.item.value.len()
                    || (a.item.value.len() == b.item.value.len() && a.age < b.age)))
    }

    fn swap_slots(&mut self, a: usize, b: usize) {
        self.heap.swap(a, b);
        self.indices.insert(self.heap[a].item.key.clone(), a);
        self.indices.insert(self.heap[b].item.key.clone(), b);
    }

    fn sift_up(&mut self, mut index: usize) {
        while index > 0 {
            let parent = (index - 1) / 2;
            if !Self::better(&self.heap[index], &self.heap[parent]) {
                break;
            }
            self.swap_slots(index, parent);
            index = parent;
        }
    }

    fn sift_down(&mut self, mut index: usize) {
        let len = self.heap.len();
        loop {
            let left = index * 2 + 1;
            if left >= len {
                break;
            }
            let right = left + 1;
            let mut best = left;
            if right < len && Self::better(&self.heap[right], &self.heap[left]) {
                best = right;
            }
            if !Self::better(&self.heap[best], &self.heap[index]) {
                break;
            }
            self.swap_slots(index, best);
            index = best;
        }
    }

    fn repair(&mut self, index: usize) {
        if index > 0 && Self::better(&self.heap[index], &self.heap[(index - 1) / 2]) {
            self.sift_up(index);
        } else {
            self.sift_down(index);
        }
    }

    fn remove_at(&mut self, index: usize) -> Entry {
        let last = self.heap.len() - 1;
        self.heap.swap(index, last);
        let removed = self.heap.pop().expect("index is valid");
        self.indices.remove(&removed.item.key);

        if index < self.heap.len() {
            self.indices.insert(self.heap[index].item.key.clone(), index);
            self.repair(index);
        }
        removed
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = Self::canonicalize(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let age = self.next_age;
        self.next_age = self.next_age.wrapping_add(1);
        let index = self.heap.len();
        self.heap.push(Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age,
        });
        self.indices.insert(key, index);
        self.sift_up(index);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.heap.first().map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.heap.is_empty() {
            None
        } else {
            Some(self.remove_at(0).item)
        }
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = Self::canonicalize(key)?;
        Ok(self.indices.get(&key).map(|&i| self.heap[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonicalize(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing {
            key: key.clone(),
        })?;
        let current = self.heap[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.heap[index].item.priority = priority;
        self.repair(index);
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonicalize(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing {
            key: key.clone(),
        })?;
        let current = self.heap[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.heap[index].item.priority = priority;
        self.repair(index);
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = Self::canonicalize(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing {
            key: key.clone(),
        })?;
        let previous = std::mem::replace(&mut self.heap[index].item.value, value.to_owned());
        self.repair(index);
        Ok(previous)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = Self::canonicalize(key)?;
        let index = *self.indices.get(&key).ok_or_else(|| Error::Missing {
            key: key.clone(),
        })?;
        Ok(self.remove_at(index).item)
    }
}
```

### Solution B

```rust
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
}

struct Entry {
    item: Item,
    age: u64,
}

pub struct IndexedHeap {
    slots: Vec<Entry>,
    positions: HashMap<String, usize>,
    age_counter: u64,
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            slots: Vec::new(),
            positions: HashMap::new(),
            age_counter: 0,
        }
    }

    pub fn len(&self) -> usize {
        self.slots.len()
    }

    pub fn is_empty(&self) -> bool {
        self.slots.is_empty()
    }

    fn canonicalize(input: &str) -> Result<String, Error> {
        let bytes = input.as_bytes();
        let mut first = 0usize;
        let mut last = bytes.len();

        while first < last && bytes[first] == 0x20 {
            first += 1;
        }
        while first < last && bytes[last - 1] == 0x20 {
            last -= 1;
        }

        if first == last {
            return Err(Error::EmptyKey);
        }

        for pos in first..last {
            let byte = bytes[pos];
            let allowed = if pos == first {
                byte.is_ascii_alphabetic()
            } else {
                byte.is_ascii_alphabetic() || byte.is_ascii_digit() || byte == b'-'
            };
            if !allowed {
                return Err(Error::InvalidKey { at: pos });
            }
        }

        Ok(input[first..last].to_ascii_lowercase())
    }

    fn comes_before(left: &Entry, right: &Entry) -> bool {
        match left.item.priority.cmp(&right.item.priority) {
            std::cmp::Ordering::Less => true,
            std::cmp::Ordering::Greater => false,
            std::cmp::Ordering::Equal => match left.item.value.len().cmp(&right.item.value.len()) {
                std::cmp::Ordering::Less => true,
                std::cmp::Ordering::Greater => false,
                std::cmp::Ordering::Equal => left.age < right.age,
            },
        }
    }

    fn swap_entries(&mut self, i: usize, j: usize) {
        self.slots.swap(i, j);
        let ki = self.slots[i].item.key.clone();
        let kj = self.slots[j].item.key.clone();
        self.positions.insert(ki, i);
        self.positions.insert(kj, j);
    }

    fn bubble_up(&mut self, mut i: usize) {
        while i != 0 {
            let p = (i - 1) / 2;
            if !Self::comes_before(&self.slots[i], &self.slots[p]) {
                return;
            }
            self.swap_entries(i, p);
            i = p;
        }
    }

    fn bubble_down(&mut self, mut i: usize) {
        loop {
            let left = i.saturating_mul(2).saturating_add(1);
            if left >= self.slots.len() {
                return;
            }
            let right = left + 1;
            let child = if right < self.slots.len()
                && Self::comes_before(&self.slots[right], &self.slots[left])
            {
                right
            } else {
                left
            };
            if !Self::comes_before(&self.slots[child], &self.slots[i]) {
                return;
            }
            self.swap_entries(i, child);
            i = child;
        }
    }

    fn rebalance(&mut self, index: usize) {
        if index > 0 && Self::comes_before(&self.slots[index], &self.slots[(index - 1) / 2]) {
            self.bubble_up(index);
        } else {
            self.bubble_down(index);
        }
    }

    fn detach(&mut self, index: usize) -> Entry {
        let last = self.slots.len() - 1;
        self.slots.swap(index, last);
        let removed = self.slots.pop().unwrap();
        self.positions.remove(&removed.item.key);
        if index < self.slots.len() {
            self.positions
                .insert(self.slots[index].item.key.clone(), index);
            self.rebalance(index);
        }
        removed
    }

    pub fn push(&mut self, key: &str, priority: i64, value: &str) -> Result<(), Error> {
        let key = Self::canonicalize(key)?;
        if self.positions.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }
        let entry = Entry {
            item: Item {
                key: key.clone(),
                priority,
                value: value.to_owned(),
            },
            age: self.age_counter,
        };
        self.age_counter = self.age_counter.wrapping_add(1);
        let index = self.slots.len();
        self.slots.push(entry);
        self.positions.insert(key, index);
        self.bubble_up(index);
        Ok(())
    }

    pub fn peek(&self) -> Option<Item> {
        self.slots.get(0).map(|entry| entry.item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        if self.slots.is_empty() {
            None
        } else {
            Some(self.detach(0).item)
        }
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = Self::canonicalize(key)?;
        Ok(self.positions.get(&key).map(|&i| self.slots[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonicalize(key)?;
        let index = match self.positions.get(&key) {
            Some(&index) => index,
            None => return Err(Error::Missing { key }),
        };
        let current = self.slots[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.slots[index].item.priority = priority;
        self.rebalance(index);
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = Self::canonicalize(key)?;
        let index = match self.positions.get(&key) {
            Some(&index) => index,
            None => return Err(Error::Missing { key }),
        };
        let current = self.slots[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.slots[index].item.priority = priority;
        self.rebalance(index);
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = Self::canonicalize(key)?;
        let index = match self.positions.get(&key) {
            Some(&index) => index,
            None => return Err(Error::Missing { key }),
        };
        let old = std::mem::replace(&mut self.slots[index].item.value, value.to_owned());
        self.rebalance(index);
        Ok(old)
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = Self::canonicalize(key)?;
        let index = match self.positions.get(&key) {
            Some(&index) => index,
            None => return Err(Error::Missing { key }),
        };
        Ok(self.detach(index).item)
    }
}
```
