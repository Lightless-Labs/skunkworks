- Decode scans the payload before checking the suffix; an invalid payload byte takes precedence over malformed checksum syntax.
- A colon splits at the first occurrence. Any suffix other than exactly one ASCII digit `0`–`6` is `InvalidChecksum` at that colon.
- Empty payload wins immediately; checksum syntax is checked before leading zero, range, and checksum value.
- The range is inclusive at both `-(2^120-1)` and `2^120-1`; prefix overflow is checked after every digit.
- Checksum positions count payload characters from one on the left, with `T=2`; the suffix is excluded.
- Both solutions use checked arithmetic for decoding and bounded arithmetic for encoding.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1i128 << 120) - 1;

fn digit_value(byte: u8, system: &SignedSystem) -> Option<i128> {
    match system {
        SignedSystem::BalancedTernary => match byte {
            b'T' => Some(-1),
            b'0' => Some(0),
            b'1' => Some(1),
            _ => None,
        },
        SignedSystem::Negabinary => match byte {
            b'0' => Some(0),
            b'1' => Some(1),
            _ => None,
        },
    }
}

fn checksum(payload: &[u8]) -> u8 {
    payload
        .iter()
        .enumerate()
        .fold(0u64, |sum, (i, &b)| {
            let code = match b {
                b'T' => 2,
                b'1' => 1,
                _ => 0,
            };
            (sum + ((i as u64 + 1) * code)) % 7
        }) as u8
}

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    if !(-LIMIT..=LIMIT).contains(&value) {
        return Err(SignedError::OutOfRange);
    }
    let mut n = value;
    let mut digits = Vec::new();
    if n == 0 {
        digits.push(b'0');
    } else {
        while n != 0 {
            match system {
                SignedSystem::BalancedTernary => {
                    let rem = n.rem_euclid(3);
                    let d = if rem == 2 { -1 } else { rem };
                    digits.push(match d {
                        -1 => b'T',
                        0 => b'0',
                        _ => b'1',
                    });
                    n = (n - d) / 3;
                }
                SignedSystem::Negabinary => {
                    let d = n.rem_euclid(2);
                    digits.push(if d == 0 { b'0' } else { b'1' });
                    n = (n - d) / -2;
                }
            }
        }
    }
    digits.reverse();
    let mut out = String::from_utf8(digits).expect("ASCII digits");
    if with_checksum {
        let sum = checksum(out.as_bytes());
        out.push(':');
        out.push(char::from(b'0' + sum));
    }
    Ok(out)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(SignedError::Empty);
    }
    let colon = bytes.iter().position(|&b| b == b':');
    let payload_end = colon.unwrap_or(bytes.len());
    if payload_end == 0 {
        return Err(SignedError::Empty);
    }
    let payload = &bytes[..payload_end];

    for (i, &b) in payload.iter().enumerate() {
        if digit_value(b, &system).is_none() {
            return Err(SignedError::InvalidByte { index: i });
        }
    }

    if let Some(c) = colon {
        if bytes.len() != c + 2 || !(b'0'..=b'6').contains(&bytes[c + 1]) {
            return Err(SignedError::InvalidChecksum { index: c });
        }
    }

    if payload.len() > 1 && payload[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let base: i128 = match system {
        SignedSystem::BalancedTernary => 3,
        SignedSystem::Negabinary => -2,
    };
    let mut value: i128 = 0;
    for (i, &b) in payload.iter().enumerate() {
        let digit = digit_value(b, &system).unwrap_or(0);
        value = value
            .checked_mul(base)
            .and_then(|v| v.checked_add(digit))
            .ok_or(SignedError::Range { index: i })?;
        if !(-LIMIT..=LIMIT).contains(&value) {
            return Err(SignedError::Range { index: i });
        }
    }

    if let Some(c) = colon {
        let expected = checksum(payload);
        let found = bytes[c + 1] - b'0';
        if expected != found {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }
    Ok(value)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    let value = signed_decode(input, from)?;
    signed_encode(value, to, with_checksum)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1i128 << 120) - 1;

fn checksum(payload: &[u8]) -> u8 {
    let mut sum = 0usize;
    for (index, byte) in payload.iter().enumerate() {
        let code = match byte {
            b'T' => 2usize,
            b'1' => 1usize,
            _ => 0usize,
        };
        sum = (sum + ((index + 1) % 7) * code) % 7;
    }
    sum as u8
}

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    if value < -LIMIT || value > LIMIT {
        return Err(SignedError::OutOfRange);
    }

    let mut n = value;
    let mut reversed = Vec::new();
    if n == 0 {
        reversed.push(b'0');
    } else {
        while n != 0 {
            if system == SignedSystem::BalancedTernary {
                let r = n.rem_euclid(3);
                let d = if r == 2 { -1 } else { r };
                reversed.push(if d == -1 {
                    b'T'
                } else if d == 0 {
                    b'0'
                } else {
                    b'1'
                });
                n = (n - d) / 3;
            } else {
                let d = n.rem_euclid(2);
                reversed.push(if d == 0 { b'0' } else { b'1' });
                n = (n - d) / -2;
            }
        }
    }
    reversed.reverse();
    let mut answer = String::new();
    for b in reversed {
        answer.push(char::from(b));
    }
    if with_checksum {
        let digit = checksum(answer.as_bytes());
        answer.push(':');
        answer.push(char::from(b'0' + digit));
    }
    Ok(answer)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    let bytes = input.as_bytes();
    let colon = bytes.iter().position(|&b| b == b':');
    let end = colon.unwrap_or(bytes.len());

    if end == 0 {
        return Err(SignedError::Empty);
    }

    for (index, &byte) in bytes[..end].iter().enumerate() {
        let valid = match system {
            SignedSystem::BalancedTernary => matches!(byte, b'T' | b'0' | b'1'),
            SignedSystem::Negabinary => matches!(byte, b'0' | b'1'),
        };
        if !valid {
            return Err(SignedError::InvalidByte { index });
        }
    }

    if let Some(colon_index) = colon {
        if bytes.len() != colon_index + 2
            || !matches!(bytes[colon_index + 1], b'0'..=b'6')
        {
            return Err(SignedError::InvalidChecksum { index: colon_index });
        }
    }

    let payload = &bytes[..end];
    if payload.len() > 1 && payload[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let base = match system {
        SignedSystem::BalancedTernary => 3i128,
        SignedSystem::Negabinary => -2i128,
    };

    let mut acc = 0i128;
    for (index, &byte) in payload.iter().enumerate() {
        let digit = match byte {
            b'T' => -1,
            b'1' => 1,
            _ => 0,
        };
        acc = acc
            .checked_mul(base)
            .and_then(|v| v.checked_add(digit))
            .ok_or(SignedError::Range { index })?;
        if acc < -LIMIT || acc > LIMIT {
            return Err(SignedError::Range { index });
        }
    }

    if let Some(colon_index) = colon {
        let expected = checksum(payload);
        let found = bytes[colon_index + 1] - b'0';
        if expected != found {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }
    Ok(acc)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    let decoded = signed_decode(input, from)?;
    signed_encode(decoded, to, with_checksum)
}
```
