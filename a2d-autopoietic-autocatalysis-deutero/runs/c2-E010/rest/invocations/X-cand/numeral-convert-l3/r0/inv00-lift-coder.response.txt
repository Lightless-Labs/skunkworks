- Decode checks the entire payload alphabet before checking checksum syntax; the first invalid payload byte wins.
- The first colon separates payload and checksum; any suffix other than exactly one ASCII digit `0`–`6` is invalid at the colon.
- Leading zero and the first out-of-range prefix take precedence over a well-formed but incorrect checksum.
- Range checks use `L = 2^120 - 1` after every digit; checked `i128` arithmetic prevents panics.
- Checksums use one-based positions from the left and character codes `T=2`, `0=0`, `1=1`.
- Solution B uses a distinct encoding approach; both versions follow the same stated validation rules.

### Solution A

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1i128 << 120) - 1;

fn digit_value(byte: u8, system: &SignedSystem) -> Option<i128> {
    match system {
        SignedSystem::BalancedTernary => match byte {
            b'T' => Some(-1),
            b'0' => Some(0),
            b'1' => Some(1),
            _ => None,
        },
        SignedSystem::Negabinary => match byte {
            b'0' => Some(0),
            b'1' => Some(1),
            _ => None,
        },
    }
}

fn checksum(payload: &[u8]) -> u8 {
    let mut sum = 0u64;
    for (index, &byte) in payload.iter().enumerate() {
        let code = match byte {
            b'T' => 2u64,
            b'1' => 1,
            _ => 0,
        };
        sum = (sum + ((index as u64 + 1) % 7) * code) % 7;
    }
    sum as u8
}

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    if !(-LIMIT..=LIMIT).contains(&value) {
        return Err(SignedError::OutOfRange);
    }

    let mut n = value;
    let mut reversed = Vec::new();
    if n == 0 {
        reversed.push(b'0');
    } else {
        while n != 0 {
            match system {
                SignedSystem::BalancedTernary => {
                    let mut remainder = n.rem_euclid(3);
                    if remainder == 2 {
                        remainder = -1;
                    }
                    let digit = remainder;
                    reversed.push(match digit {
                        -1 => b'T',
                        0 => b'0',
                        _ => b'1',
                    });
                    n = (n - digit) / 3;
                }
                SignedSystem::Negabinary => {
                    let digit = n.rem_euclid(2);
                    reversed.push(if digit == 0 { b'0' } else { b'1' });
                    n = (n - digit) / -2;
                }
            }
        }
    }

    reversed.reverse();
    let mut output = String::from_utf8(reversed).expect("ASCII digits");
    if with_checksum {
        let check = checksum(output.as_bytes());
        output.push(':');
        output.push(char::from(b'0' + check));
    }
    Ok(output)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(SignedError::Empty);
    }

    let colon = bytes.iter().position(|&byte| byte == b':');
    let payload_end = colon.unwrap_or(bytes.len());
    if payload_end == 0 {
        return Err(SignedError::Empty);
    }
    let payload = &bytes[..payload_end];

    for (index, &byte) in payload.iter().enumerate() {
        if digit_value(byte, &system).is_none() {
            return Err(SignedError::InvalidByte { index });
        }
    }

    if let Some(colon_index) = colon {
        let suffix = &bytes[colon_index + 1..];
        if suffix.len() != 1 || !(b'0'..=b'6').contains(&suffix[0]) {
            return Err(SignedError::InvalidChecksum { index: colon_index });
        }
    }

    if payload.len() > 1 && payload[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let mut accumulator = 0i128;
    let base: i128 = match system {
        SignedSystem::BalancedTernary => 3,
        SignedSystem::Negabinary => -2,
    };

    for (index, &byte) in payload.iter().enumerate() {
        let digit = digit_value(byte, &system).expect("validated digit");
        accumulator = accumulator
            .checked_mul(base)
            .and_then(|n| n.checked_add(digit))
            .ok_or(SignedError::Range { index })?;
        if !(-LIMIT..=LIMIT).contains(&accumulator) {
            return Err(SignedError::Range { index });
        }
    }

    if let Some(colon_index) = colon {
        let expected = checksum(payload);
        let found = bytes[colon_index + 1] - b'0';
        if expected != found {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }

    Ok(accumulator)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    let value = signed_decode(input, from)?;
    signed_encode(value, to, with_checksum)
}
```

### Solution B

```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedSystem {
    BalancedTernary,
    Negabinary,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SignedError {
    Empty,
    InvalidByte { index: usize },
    InvalidChecksum { index: usize },
    LeadingZero,
    Range { index: usize },
    ChecksumMismatch { expected: u8, found: u8 },
    OutOfRange,
}

const LIMIT: i128 = (1i128 << 120) - 1;

fn digit(byte: u8, system: &SignedSystem) -> Option<i128> {
    match (system, byte) {
        (SignedSystem::BalancedTernary, b'T') => Some(-1),
        (_, b'0') => Some(0),
        (_, b'1') => Some(1),
        _ => None,
    }
}

fn checksum_bytes(bytes: &[u8]) -> u8 {
    bytes.iter().enumerate().fold(0u8, |sum, (i, &b)| {
        let code = if b == b'T' { 2u16 } else if b == b'1' { 1 } else { 0 };
        ((u16::from(sum) + (((i as u128 + 1) % 7) as u16) * code) % 7) as u8
    })
}

pub fn signed_encode(
    value: i128,
    system: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    if value < -LIMIT || value > LIMIT {
        return Err(SignedError::OutOfRange);
    }

    // Use little-endian output digits, then reverse them for display.
    let mut n = value;
    let mut digits = Vec::new();
    while n != 0 {
        let (d, next) = match system {
            SignedSystem::BalancedTernary => {
                let r = n.rem_euclid(3);
                let d = if r == 2 { -1 } else { r };
                (d, (n - d) / 3)
            }
            SignedSystem::Negabinary => {
                let d = n.rem_euclid(2);
                (d, (n - d) / -2)
            }
        };
        digits.push(match d {
            -1 => b'T',
            0 => b'0',
            _ => b'1',
        });
        n = next;
    }

    if digits.is_empty() {
        digits.push(b'0');
    }
    let mut lo = 0usize;
    let mut hi = digits.len();
    while lo < hi {
        hi -= 1;
        digits.swap(lo, hi);
        lo += 1;
    }

    let mut result = String::with_capacity(digits.len() + if with_checksum { 2 } else { 0 });
    for &b in &digits {
        result.push(char::from(b));
    }
    if with_checksum {
        let c = checksum_bytes(&digits);
        result.push(':');
        result.push(char::from(b'0' + c));
    }
    Ok(result)
}

pub fn signed_decode(input: &str, system: SignedSystem) -> Result<i128, SignedError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(SignedError::Empty);
    }

    let colon = bytes.iter().position(|&b| b == b':');
    let end = colon.unwrap_or(bytes.len());
    if end == 0 {
        return Err(SignedError::Empty);
    }

    for (i, &b) in bytes[..end].iter().enumerate() {
        if digit(b, &system).is_none() {
            return Err(SignedError::InvalidByte { index: i });
        }
    }

    if let Some(c) = colon {
        if bytes.len() != c + 2 || !(b'0'..=b'6').contains(&bytes[c + 1]) {
            return Err(SignedError::InvalidChecksum { index: c });
        }
    }

    let payload = &bytes[..end];
    if payload.len() > 1 && payload[0] == b'0' {
        return Err(SignedError::LeadingZero);
    }

    let base = match system {
        SignedSystem::BalancedTernary => 3i128,
        SignedSystem::Negabinary => -2i128,
    };
    let mut value = 0i128;
    for (i, &b) in payload.iter().enumerate() {
        let d = digit(b, &system).expect("digit validated above");
        value = match value.checked_mul(base).and_then(|x| x.checked_add(d)) {
            Some(x) if (-LIMIT..=LIMIT).contains(&x) => x,
            _ => return Err(SignedError::Range { index: i }),
        };
    }

    if let Some(c) = colon {
        let expected = checksum_bytes(payload);
        let found = bytes[c + 1] - b'0';
        if expected != found {
            return Err(SignedError::ChecksumMismatch { expected, found });
        }
    }

    Ok(value)
}

pub fn signed_recode(
    input: &str,
    from: SignedSystem,
    to: SignedSystem,
    with_checksum: bool,
) -> Result<String, SignedError> {
    match signed_decode(input, from) {
        Ok(value) => signed_encode(value, to, with_checksum),
        Err(error) => Err(error),
    }
}
```
