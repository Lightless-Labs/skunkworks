pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    let n = tasks.len();

    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = Vec::with_capacity(task.deps.len());
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }
            if value == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if seen.contains(&value) {
                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
            seen.push(value);
        }
    }

    let mut successors = vec![Vec::new(); n];
    let mut remaining = vec![0usize; n];
    for (id, task) in tasks.iter().enumerate() {
        remaining[id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(id);
        }
    }

    let mut ready = BinaryHeap::new();
    for (id, &degree) in remaining.iter().enumerate() {
        if degree == 0 {
            ready.push(Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(Reverse(id)) = ready.pop() {
        order.push(id);
        for &child in &successors[id] {
            remaining[child] -= 1;
            if remaining[child] == 0 {
                ready.push(Reverse(child));
            }
        }
    }

    if order.len() < n {
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        let blocked = (0..n).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut earliest = vec![Timing { start: 0, finish: 0 }; n];
    let mut makespan = 0u64;
    for &id in &order {
        let mut start = 0u64;
        for &pred in &tasks[id].deps {
            start = start.max(earliest[pred].finish);
        }
        let finish = match start.checked_add(tasks[id].duration) {
            Some(value) => value,
            None => return Err(Error::Overflow { task: id }),
        };
        earliest[id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut latest = vec![Timing { start: 0, finish: 0 }; n];
    for index in (0..order.len()).rev() {
        let id = order[index];
        let finish = match successors[id]
            .iter()
            .map(|&child| latest[child].start)
            .min()
        {
            Some(value) => value,
            None => target,
        };
        latest[id] = Timing {
            start: finish - tasks[id].duration,
            finish,
        };
    }

    let mut slack = vec![0u64; n];
    let baseline = target - makespan;
    let mut critical = Vec::new();
    for id in 0..n {
        slack[id] = latest[id].start - earliest[id].start;
        if slack[id] == baseline {
            critical.push(id);
        }
    }

    let mut path = Vec::new();
    if n != 0 {
        let sink = (0..n)

            .max()
            .unwrap();

            path.push(current);

                .deps
                .iter()
                .copied()
                .max_by_key(|&pred| (earliest[pred].finish, pred))
                .unwrap();
        }
        path.reverse();
    }

    Ok(Analysis { order, earliest, latest, slack, critical, path, makespan })
}

pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    validate(tasks)?;

    let n = tasks.len();
    let mut remaining: Vec<usize> = tasks.iter().map(|task| task.deps.len()).collect();
    let mut children = vec![Vec::<usize>::new(); n];
    for (child, task) in tasks.iter().enumerate() {
        for &parent in &task.deps {
            children[parent].push(child);
        }
    }

        if remaining[id] == 0 {
            heap.push(Reverse(id));
        }

            Some(finish) => finish,
            None => return Err(Error::Overflow { task: id }),

        earliest_finish[id] = finish;
        order.push(id);

        for &child in &children[id] {
            earliest_start[child] = earliest_start[child].max(finish);
            remaining[child] -= 1;
            if remaining[child] == 0 {
                heap.push(Reverse(child));
            }
        }

        let blocked = (0..n).filter(|&id| remaining[id] > 0).collect();

        .map(|id| Timing {
            start: earliest_start[id],
            finish: earliest_finish[id],
        })
        .collect::<Vec<_>>();

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut latest_start = vec![0u64; n];
    let mut latest_finish = vec![0u64; n];
    for index in (0..order.len()).rev() {
        let id = order[index];
        let finish_limit = children[id]
            .iter()
            .map(|&child| latest_start[child])
            .min()
            .unwrap_or(target);
        latest_finish[id] = finish_limit;
        latest_start[id] = finish_limit - tasks[id].duration;
    }

    let slack = (0..n)
        .map(|id| latest_start[id] - earliest_start[id])
        .collect::<Vec<_>>();
    let baseline = target - makespan;

        .filter(|&id| slack[id] == baseline)
        .collect::<Vec<_>>();

    for id in 0..n {
        if children[id].is_empty() && earliest_finish[id] == makespan {
            sink = Some(id);
        }

                None => break,
            }
        }
        path.reverse();

            start: latest_start[id],
            finish: latest_finish[id],
        })
        .collect::<Vec<_>>();

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}

    for (id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: id,
                    position,
                    value,
                });
            }
            if value == id {
                return Err(Error::SelfDependency {
                    task: id,
                    position,
                });
            }
            for earlier in 0..position {
                if task.deps[earlier] == value {
                    return Err(Error::DuplicateDependency {
                        task: id,
                        position,
                        value,
                    });
                }
            }
        }
    }

    let mut indegree = vec![0usize; n];
    for (id, task) in tasks.iter().enumerate() {
        indegree[id] = task.deps.len();
        for &pred in &task.deps {
            outgoing[pred].push(id);
        }
    }

        if indegree[id] == 0 {
            ready.push(Reverse(id));
        }

    let mut remaining = indegree;
    while let Some(Reverse(id)) = ready.pop() {
        order.push(id);

            remaining[child] -= 1;
            if remaining[child] == 0 {
                ready.push(Reverse(child));
            }
        }

        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),

    let mut makespan = 0;

            .iter()
            .fold(0u64, |best, &pred| best.max(early[pred].finish));
        let finish = match start.checked_add(tasks[id].duration) {
            Some(value) => value,
            None => return Err(Error::Overflow { task: id }),
        };
        early[id] = Timing { start, finish };
        if finish > makespan {
            makespan = finish;
        }
    }

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut late = vec![Timing { start: 0, finish: 0 }; n];
    for &id in order.iter().rev() {

            None => target,

            start: finish - tasks[id].duration,
            finish,
        };

        if slack[id] == baseline {
            critical.push(id);
        }

            && sink.map_or(true, |best: usize| id > best)

            sink = Some(id);

            path.push(id);
            if tasks[id].deps.is_empty() {

            let mut best = tasks[id].deps[0];

                if (early[pred].finish, pred) > (early[best].finish, best) {
                    best = pred;
                }
            }
            id = best;
        }
        path.reverse();
    }

    Ok(Analysis {
        order,
        earliest: early,
        latest: late,
        slack,
        critical,
        path,
        makespan,
    })
}

pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    let n = tasks.len();

    // Validate every dependency before doing any graph work.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency { task: task_id, position, value });
            }
            if value == task_id {
                return Err(Error::SelfDependency { task: task_id, position });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency { task: task_id, position, value });
            }
        }
    }

    let mut successors = vec![Vec::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(task_id);
        }
    }

    // Sorted vector keeps the smallest currently available ID at its end.
    let mut available = Vec::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }
    }

    let mut order = Vec::with_capacity(n);
    while let Some(id) = available.pop() {
        order.push(id);
        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                let at = available.partition_point(|&x| x > child);
                available.insert(at, child);
            }
        }
    }

    if order.len() != n {
        let mut blocked = Vec::new();
        let mut removed = vec![false; n];
        for &id in &order {
            removed[id] = true;
        }
        for id in 0..n {
            if !removed[id] {
                blocked.push(id);
            }
        }
        return Err(Error::Cycle { blocked });
    }

    let mut earliest = vec![Timing { start: 0, finish: 0 }; n];
    let mut makespan = 0;
    for &id in &order {
        let start = tasks[id]
            .deps
            .iter()
            .map(|&pred| earliest[pred].finish)
            .max()
            .unwrap_or(0);
        let finish = start
            .checked_add(tasks[id].duration)
            .ok_or(Error::Overflow { task: id })?;
        earliest[id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut latest = vec![Timing { start: 0, finish: 0 }; n];
    for &id in order.iter().rev() {
        let finish = if successors[id].is_empty() {
            target
        } else {
            successors[id]
                .iter()
                .map(|&child| latest[child].start)
                .min()
                .unwrap()
        };
        let start = finish - tasks[id].duration;
        latest[id] = Timing { start, finish };
    }

    let slack: Vec<u64> = (0..n)
        .map(|id| latest[id].start - earliest[id].start)
        .collect();
    let baseline = target - makespan;
    let critical = (0..n).filter(|&id| slack[id] == baseline).collect();

    let mut path = Vec::new();
    if n > 0 {
        let mut sink = None;
        for id in 0..n {
            if successors[id].is_empty()
                && earliest[id].finish == makespan
                && sink.map_or(true, |best| id > best)
            {
                sink = Some(id);
            }
        }
        let mut current = sink.unwrap();
        path.push(current);
        while !tasks[current].deps.is_empty() {
            let predecessor = tasks[current]
                .deps
                .iter()
                .copied()
                .max_by_key(|&pred| (earliest[pred].finish, pred))
                .unwrap();
            current = predecessor;
            path.push(current);
        }
        path.reverse();
    }

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}

    validate(tasks)?;

    let n = tasks.len();
    let mut indegree: Vec<usize> = tasks.iter().map(|t| t.deps.len()).collect();
    let mut successors = vec![Vec::<usize>::new(); n];
    for (child, task) in tasks.iter().enumerate() {
        for &parent in &task.deps {
            successors[parent].push(child);
        }
    }

    let mut available = BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(n);
    let mut start = vec![0u64; n];
    let mut finish = vec![0u64; n];

            .checked_add(tasks[id].duration)
            .ok_or(Error::Overflow { task: id })?;

            if start[child] < end {
                start[child] = end;

            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(Reverse(child));
            }
        }

        for (id, &degree) in indegree.iter().enumerate() {
            if degree != 0 {
                blocked.push(id);
            }

        return Err(Error::Cycle { blocked });

    let earliest: Vec<Timing> = (0..n)
        .map(|i| Timing { start: start[i], finish: finish[i] })
        .collect();

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut latest_start = vec![0u64; n];
    let mut latest_finish = vec![0u64; n];
    for &id in order.iter().rev() {
        let lf = if successors[id].is_empty() {
            target
        } else {
            successors[id]
                .iter()
                .map(|&child| latest_start[child])
                .min()
                .unwrap()
        };
        latest_finish[id] = lf;
        latest_start[id] = lf - tasks[id].duration;
    }

    let slack: Vec<u64> = (0..n).map(|i| latest_start[i] - start[i]).collect();
    let baseline = target - makespan;
    let critical: Vec<usize> = (0..n).filter(|&i| slack[i] == baseline).collect();

    let mut path = Vec::new();
    if n != 0 {
        let mut sink: Option<usize> = None;
        for id in 0..n {
            if successors[id].is_empty() && finish[id] == makespan {
                sink = Some(id);

            path.push(id);
            while !tasks[id].deps.is_empty() {
                let mut best = tasks[id].deps[0];
                for &parent in &tasks[id].deps[1..] {
                    if finish[parent] > finish[best]
                        || (finish[parent] == finish[best] && parent > best)

                        best = parent;
                    }

                id = best;
                path.push(id);
            }
            path.reverse();
        }
    }

    let latest: Vec<Timing> = (0..n)
        .map(|i| Timing { start: latest_start[i], finish: latest_finish[i] })
        .collect();

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}
