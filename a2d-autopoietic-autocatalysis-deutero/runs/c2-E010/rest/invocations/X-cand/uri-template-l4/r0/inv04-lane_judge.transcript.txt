fn parse<'a>(body: &'a str, body_pos: usize, close: usize) -> Result<(Op, Vec<Var<'a>>), Error> {
    let mut idx = 0;
    let op = match body.as_bytes().first().copied() {
        Some(b'+') => { idx = 1; Op::Plus }
        Some(b'#') => { idx = 1; Op::Hash }
        Some(b'.') => { idx = 1; Op::Dot }
        Some(b'/') => { idx = 1; Op::Slash }
        Some(b';') => { idx = 1; Op::Semi }
        Some(b'?') => { idx = 1; Op::Query }
        Some(b'&') => { idx = 1; Op::Amp }
        _ => Op::Simple,
    };
    if idx == body.len() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let rest = &body[idx..];
    let mut vars = Vec::new();
    let mut begin = 0;
    loop {
        let end = rest[begin..].find(',').map(|n| begin + n).unwrap_or(rest.len());
        let token = &rest[begin..end];
        let pos = body_pos + idx + begin;
        if token.is_empty() {
            return Err(Error::EmptyVariable { pos });
        }
        let m = token.bytes().position(|b| b == b'*' || b == b':');
        let name_end = m.unwrap_or(token.len());
        let name = &token[..name_end];
        if name.is_empty() {
            return Err(Error::InvalidName { pos });
        }
        let nb = name.as_bytes();
        if !(nb[0].is_ascii_alphabetic() || nb[0] == b'_') {
            return Err(Error::InvalidName { pos });
        }
        for (i, &b) in nb.iter().enumerate().skip(1) {
            if !(b.is_ascii_alphanumeric() || b == b'_') {
                return Err(Error::InvalidName { pos: pos + i });
            }
        }
        let modifier = if let Some(m) = m {
            let suffix = &token[m..];
            if suffix == "*" {
                Mod::Explode
            } else if suffix.starts_with(':') {
                let digits = &suffix[1..];
                if !(1..=4).contains(&digits.len())
                    || digits.as_bytes()[0] == b'0'
                    || !digits.bytes().all(|b| b.is_ascii_digit())
                {
                    return Err(Error::InvalidModifier { pos: pos + m });
                }
                let n = digits.parse::<usize>().unwrap_or(0);
                if n == 0 || n > 9999 {
                    return Err(Error::InvalidModifier { pos: pos + m });
                }
                Mod::Prefix(n)
            } else {
                return Err(Error::InvalidModifier { pos: pos + m });
            }
        } else {
            Mod::Plain
        };
        if vars.iter().any(|v: &Var<'_>| v.name == name) {
            return Err(Error::DuplicateVariable { pos });
        }
        vars.push(Var { name, modifier, pos });
        if end == rest.len() {
            break;
        }
        begin = end + 1;
    }
    Ok((op, vars))
}

fn encoded(s: &str, reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let b = s.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < b.len() {
        let c = b[i];
        if c == b'%' && i + 2 < b.len() && b[i + 1].is_ascii_hexdigit() && b[i + 2].is_ascii_hexdigit() {
            out.push('%');
            out.push((b[i + 1] as char).to_ascii_uppercase());
            out.push((b[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if c.is_ascii_alphanumeric() || matches!(c, b'-' | b'.' | b'_' | b'~')
            || (reserved && matches!(c, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='))
        {
            out.push(c as char);
            i += 1;
        } else {
            out.push('%');
            out.push(HEX[(c >> 4) as usize] as char);
            out.push(HEX[(c & 15) as usize] as char);
            i += 1;
        }
    }
    out
}

fn parameter(op: Op, name: &str, value: String) -> String {
    if op == Op::Semi && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn render(op: Op, vars_in_expr: &[Var<'_>], bindings: &[(&str, Value)]) -> Result<String, Error> {
    let reserved = matches!(op, Op::Plus | Op::Hash);
    let mut fields = Vec::new();
    for var in vars_in_expr {
        let selected = bindings.iter().rev().find(|(n, _)| *n == var.name).map(|(_, v)| v);
        let Some(value) = selected else { continue };
        if matches!(var.modifier, Mod::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: var.pos });
        }
        match value {
            Value::Scalar(s) => {
                let s = match var.modifier {
                    Mod::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                let value = encoded(&s, reserved);
                fields.push(if matches!(op, Op::Semi | Op::Query | Op::Amp) {
                    parameter(op, var.name, value)
                } else { value });
            }
            Value::List(list) => {
                if list.is_empty() { continue; }
                if matches!(var.modifier, Mod::Explode) {
                    for item in list {
                        let item = encoded(item, reserved);
                        fields.push(if matches!(op, Op::Semi | Op::Query | Op::Amp) {
                            parameter(op, var.name, item)
                        } else { item });
                    }
                } else {
                    let value = list.iter().map(|s| encoded(s, reserved)).collect::<Vec<_>>().join(",");
                    fields.push(if matches!(op, Op::Semi | Op::Query | Op::Amp) {
                        parameter(op, var.name, value)
                    } else { value });
                }
            }
            Value::Map(map) => {
                if map.is_empty() { continue; }
                for i in 0..map.len() {
                    if map[..i].iter().any(|(key, _)| *key == map[i].0) {
                        return Err(Error::DuplicateKey { pos: var.pos });
                    }
                }
                if matches!(var.modifier, Mod::Explode) {
                    for (key, value) in map {
                        let key = encoded(key, reserved);
                        let value = encoded(value, reserved);
                        fields.push(if op == Op::Semi && value.is_empty() {
                            key
                        } else {
                            format!("{key}={value}")
                        });
                    }
                } else {
                    let mut flat = Vec::with_capacity(map.len() * 2);
                    for (key, value) in map {
                        flat.push(encoded(key, reserved));
                        flat.push(encoded(value, reserved));
                    }
                    let value = flat.join(",");
                    fields.push(if matches!(op, Op::Semi | Op::Query | Op::Amp) {
                        parameter(op, var.name, value)
                    } else { value });
                }
            }
        }
    }
    if fields.is_empty() { return Ok(String::new()); }
    let (prefix, sep) = match op {
        Op::Simple | Op::Plus => ("", ","),
        Op::Hash => ("#", ","),
        Op::Dot => (".", "."),
        Op::Slash => ("/", "/"),
        Op::Semi => (";", ";"),
        Op::Query => ("?", "&"),
        Op::Amp => ("&", "&"),
    };
    Ok(format!("{prefix}{}", fields.join(sep)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut parsed: Vec<(usize, usize, Op, Vec<Var<'_>>)> = Vec::new();
    let b = template.as_bytes();
    let mut i = 0;
    while i < b.len() {
        match b[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(rel) = template[open + 1..].find('}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + rel;
                if let Some(nested) = template[open + 1..close].find('{') {
                    return Err(Error::NestedOpen { pos: open + 1 + nested });
                }
                let (op, specs) = parse(&template[open + 1..close], open + 1, close)?;
                parsed.push((open, close + 1, op, specs));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for (start, end, op, specs) in parsed {
        out.push_str(&template[cursor..start]);
        out.push_str(&render(op, &specs, vars)?);
        cursor = end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone)]
enum Mod {
    Plain,
    Explode,
    Prefix(usize),
}

struct Var {
    name: String,
    pos: usize,
    modifier: Mod,
}

struct Expression {
    operator: u8,
    vars: Vec<Var>,
}

enum Segment {
    Text(usize, usize),
    Expr(Expression),
}

fn parse_name(name: &[u8], pos: usize) -> Result<(), Error> {
    if name.is_empty() {
        return Err(Error::InvalidName { pos });
    }
    if !(name[0].is_ascii_alphabetic() || name[0] == b'_') {
        return Err(Error::InvalidName { pos });
    }
    for (i, b) in name.iter().enumerate().skip(1) {
        if !(b.is_ascii_alphanumeric() || *b == b'_') {
            return Err(Error::InvalidName { pos: pos + i });
        }
    }
    Ok(())
}

fn parse_modifier(suffix: &[u8], pos: usize) -> Result<Mod, Error> {
    if suffix == b"*" {
        return Ok(Mod::Explode);
    }
    if suffix.len() >= 2 && suffix[0] == b':' && suffix.len() <= 5
        && suffix[1].is_ascii_digit() && suffix[1] != b'0'
        && suffix[1..].iter().all(u8::is_ascii_digit)
    {
        let mut n = 0usize;
        for d in &suffix[1..] {
            n = n * 10 + usize::from(*d - b'0');
        }
        if (1..=9999).contains(&n) {
            return Ok(Mod::Prefix(n));
        }
    }
    Err(Error::InvalidModifier { pos })
}

fn parse_expression(s: &str, opening: usize, closing: usize) -> Result<Expression, Error> {
    let bytes = s.as_bytes();
    let mut p = opening + 1;
    let operator = if p < closing && b"+#./;?&".contains(&bytes[p]) {
        let op = bytes[p];
        p += 1;
        op
    } else if p < closing && b"!@$|=".contains(&bytes[p]) {
        return Err(Error::UnsupportedOperator { pos: p });
    } else {
        0
    };

    if p == closing {
        return Err(Error::EmptyExpression { pos: closing });
    }

    let mut result = Vec::new();
    let mut names = std::collections::HashSet::new();
    let mut begin = p;
    loop {
        let end = bytes[begin..closing].iter().position(|b| *b == b',')
            .map(|n| begin + n).unwrap_or(closing);
        if begin == end {
            return Err(Error::EmptyVariable { pos: begin });
        }

        let spec = &bytes[begin..end];
        let cut = spec.iter().position(|b| *b == b'*' || *b == b':').unwrap_or(spec.len());
        parse_name(&spec[..cut], begin)?;
        let modifier = if cut == spec.len() {
            Mod::Plain
        } else {
            parse_modifier(&spec[cut..], begin + cut)?
        };
        let name = s[begin..begin + cut].to_owned();
        if !names.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: begin });
        }
        result.push(Var { name, pos: begin, modifier });

        if end == closing {
            break;
        }
        begin = end + 1;
    }
    Ok(Expression { operator, vars: result })
}

fn parse_all(s: &str) -> Result<Vec<Segment>, Error> {
    let b = s.as_bytes();
    let mut res = Vec::new();
    let mut i = 0;
    let mut text_start = 0;
    while i < b.len() {
        if b[i] == b'}' {
            return Err(Error::UnexpectedClose { pos: i });
        }
        if b[i] != b'{' {
            i += 1;
            continue;
        }
        if i > text_start {
            res.push(Segment::Text(text_start, i));
        }
        let open = i;
        let close = match b[i + 1..].iter().position(|c| *c == b'}') {
            Some(n) => i + 1 + n,
            None => return Err(Error::Unclosed { pos: open }),
        };
        if let Some(n) = b[i + 1..close].iter().position(|c| *c == b'{') {
            return Err(Error::NestedOpen { pos: i + 1 + n });
        }
        res.push(Segment::Expr(parse_expression(s, open, close)?));
        i = close + 1;
        text_start = i;
    }
    if text_start < b.len() {
        res.push(Segment::Text(text_start, b.len()));
    }
    Ok(res)
}

fn percent_encode(s: &str, reserved: bool) -> String {
    let b = s.as_bytes();
    let mut out = String::new();
    let mut p = 0;
    while p < b.len() {
        if b[p] == b'%' && p + 2 < b.len() && b[p + 1].is_ascii_hexdigit() && b[p + 2].is_ascii_hexdigit() {
            out.push('%');
            out.push((b[p + 1] as char).to_ascii_uppercase());
            out.push((b[p + 2] as char).to_ascii_uppercase());
            p += 3;
            continue;
        }
        let x = b[p];
        if x.is_ascii_alphanumeric() || matches!(x, b'-' | b'.' | b'_' | b'~')
            || (reserved && b":/?#[]@!$&'()*+,;=".contains(&x))
        {
            out.push(x as char);
        } else {
            const H: &[u8] = b"0123456789ABCDEF";
            out.push('%');
            out.push(H[(x >> 4) as usize] as char);
            out.push(H[(x & 15) as usize] as char);
        }
        p += 1;
    }
    out
}

fn select_value<'a>(name: &str, bindings: &'a [(&str, Value)]) -> Option<&'a Value> {
    let mut found = None;
    for (key, value) in bindings {
        if *key == name {
            found = Some(value);
        }
    }
    found
}

fn wrap(name: &str, value: String, semicolon: bool) -> String {
    if semicolon && value.is_empty() {
        name.to_owned()
    } else {
        let mut out = String::with_capacity(name.len() + value.len() + 1);
        out.push_str(name);
        out.push('=');
        out.push_str(&value);
        out
    }
}

fn expand_variable(var: &Var, value: &Value, operator: u8) -> Result<Vec<String>, Error> {
    let reserved = operator == b'+' || operator == b'#';
    let named = matches!(operator, b';' | b'?' | b'&');
    match value {
        Value::Scalar(s) => {
            let sliced = match var.modifier {
                Mod::Prefix(n) => s.chars().take(n).collect::<String>(),
                _ => s.clone(),
            };
            let v = percent_encode(&sliced, reserved);
            Ok(vec![if named { wrap(&var.name, v, operator == b';') } else { v }])
        }
        Value::List(items) => {
            if matches!(var.modifier, Mod::Prefix(_)) {
                return Err(Error::TypeMismatch { pos: var.pos });
            }
            if items.is_empty() {
                return Ok(Vec::new());
            }
            let explode = matches!(var.modifier, Mod::Explode);
            if explode {
                Ok(items.iter().map(|item| {
                    let encoded = percent_encode(item, reserved);
                    if named { wrap(&var.name, encoded, operator == b';') } else { encoded }
                }).collect())
            } else {
                let joined = items.iter().map(|item| percent_encode(item, reserved)).collect::<Vec<_>>().join(",");
                Ok(vec![if named { wrap(&var.name, joined, operator == b';') } else { joined }])
            }
        }
        Value::Map(entries) => {
            if matches!(var.modifier, Mod::Prefix(_)) {
                return Err(Error::TypeMismatch { pos: var.pos });
            }
            if entries.is_empty() {
                return Ok(Vec::new());
            }
            let mut seen = std::collections::HashSet::new();
            for (key, _) in entries {
                if !seen.insert(key) {
                    return Err(Error::DuplicateKey { pos: var.pos });
                }
            }
            if matches!(var.modifier, Mod::Explode) {
                Ok(entries.iter().map(|(key, value)| {
                    let k = percent_encode(key, reserved);
                    let v = percent_encode(value, reserved);
                    if operator == b';' && v.is_empty() {
                        k
                    } else {
                        format!("{k}={v}")
                    }
                }).collect())
            } else {
                let mut flattened = Vec::with_capacity(entries.len() * 2);
                for (key, value) in entries {
                    flattened.push(percent_encode(key, reserved));
                    flattened.push(percent_encode(value, reserved));
                }
                let joined = flattened.join(",");
                Ok(vec![if named { wrap(&var.name, joined, operator == b';') } else { joined }])
            }
        }
    }
}

fn render_expression(expr: &Expression, bindings: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields = Vec::new();
    for var in &expr.vars {
        if let Some(value) = select_value(&var.name, bindings) {
            fields.extend(expand_variable(var, value, expr.operator)?);
        }
    }
    if fields.is_empty() {
        return Ok(String::new());
    }
    let (prefix, separator) = match expr.operator {
        0 | b'+' => ("", ","),
        b'#' => ("#", ","),
        b'.' => (".", "."),
        b'/' => ("/", "/"),
        b';' => (";", ";"),
        b'?' => ("?", "&"),
        b'&' => ("&", "&"),
        _ => ("", ","),
    };
    let mut out = String::from(prefix);
    out.push_str(&fields.join(separator));
    Ok(out)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let segments = parse_all(template)?;
    let mut output = String::new();
    for segment in segments {
        match segment {
            Segment::Text(start, end) => output.push_str(&template[start..end]),
            Segment::Expr(expr) => output.push_str(&render_expression(&expr, vars)?),
        }
    }
    Ok(output)
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continue,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Spec<'a> {
    name: &'a str,
    modifier: Modifier,
    pos: usize,
}

struct Expression<'a> {
    op: Operator,
    specs: Vec<Spec<'a>>,

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_byte(b: u8) -> bool {
    is_name_start(b) || b.is_ascii_digit()
}

fn parse_expression<'a>(content: &'a str, start: usize, close: usize) -> Result<Expression<'a>, Error> {
    let bytes = content.as_bytes();
    let mut offset = 0;
    let op = if let Some(&b) = bytes.first() {
        let recognized = match b {
            b'+' => Some(Operator::Reserved),
            b'#' => Some(Operator::Fragment),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semicolon),
            b'?' => Some(Operator::Query),
            b'&' => Some(Operator::Continue),
            _ => None,
        };
        if let Some(op) = recognized {
            offset = 1;
            op
        } else {
            Operator::Simple
        }
    } else {
        Operator::Simple
    };

    if offset == bytes.len() {
        return Err(Error::EmptyExpression { pos: close });
    }

    let remaining = &content[offset..];
    let mut specs = Vec::new();
    let mut local_start = 0;
    loop {
        let local_end = remaining[local_start..]
            .find(',')
            .map(|n| local_start + n)
            .unwrap_or(remaining.len());
        let part = &remaining[local_start..local_end];
        let pos = start + offset + local_start;
        if part.is_empty() {
            return Err(Error::EmptyVariable { pos });
        }

        let part_bytes = part.as_bytes();
        let modifier_at = part_bytes.iter().position(|&b| b == b'*' || b == b':');
        let name_end = modifier_at.unwrap_or(part.len());
        let name = &part[..name_end];
        if name.is_empty() {
            return Err(Error::InvalidName { pos });
        }

        let name_bytes = name.as_bytes();
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos });
        }
        if let Some(i) = name_bytes.iter().position(|&b| !is_name_byte(b)) {
            return Err(Error::InvalidName { pos: pos + i });
        }

        let modifier = if let Some(i) = modifier_at {
            let suffix = &part[i..];
            if suffix == "*" {
                Modifier::Explode
            } else if suffix.as_bytes().first() == Some(&b':') {
                let digits = &suffix[1..];
                let valid = (1..=4).contains(&digits.len())
                    && digits.as_bytes()[0].is_ascii_digit()
                    && digits.as_bytes()[0] != b'0'
                    && digits.bytes().all(|b| b.is_ascii_digit());
                if !valid {
                    return Err(Error::InvalidModifier { pos: pos + i });
                }
                let value = digits.parse::<usize>().unwrap_or(0);
                if !(1..=9999).contains(&value) {
                    return Err(Error::InvalidModifier { pos: pos + i });
                }
                Modifier::Prefix(value)
            } else {
                return Err(Error::InvalidModifier { pos: pos + i });
            }
        } else {
            Modifier::None
        };

        if specs.iter().any(|s: &Spec<'_>| s.name == name) {
            return Err(Error::DuplicateVariable { pos });
        }
        specs.push(Spec { name, modifier, pos });

        if local_end == remaining.len() {
            break;
        }
        local_start = local_end + 1;
    }

    Ok(Expression { op, specs })
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }
        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let allowed_reserved = reserved && b.is_ascii() && matches!(
            b, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
                | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
        );
        if unreserved || allowed_reserved {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
        }
        i += 1;
    }
    out
}

fn named_field(op: Operator, name: &str, value: String) -> String {
    match op {
        Operator::Semicolon if value.is_empty() => name.to_owned(),
        _ => format!("{name}={value}"),
    }
}

fn expand_expression(expr: &Expression<'_>, vars: &[(&str, Value)]) -> Result<String, Error> {
    let reserved = matches!(expr.op, Operator::Reserved | Operator::Fragment);
    let mut fields = Vec::new();

    for spec in &expr.specs {
        let value = vars.iter().rev().find(|(name, _)| *name == spec.name).map(|(_, v)| v);
        let Some(value) = value else { continue };

        if matches!(spec.modifier, Modifier::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(s) => {
                let value = match spec.modifier {
                    Modifier::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                let encoded = encode(&value, reserved);
                if matches!(expr.op, Operator::Semicolon | Operator::Query | Operator::Continue) {
                    fields.push(named_field(expr.op, spec.name, encoded));
                } else {
                    fields.push(encoded);
                }
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        if matches!(expr.op, Operator::Semicolon | Operator::Query | Operator::Continue) {
                            fields.push(named_field(expr.op, spec.name, encoded));
                        } else {
                            fields.push(encoded);
                        }
                    }
                } else {
                    let joined = items.iter().map(|s| encode(s, reserved)).collect::<Vec<_>>().join(",");
                    if matches!(expr.op, Operator::Semicolon | Operator::Query | Operator::Continue) {
                        fields.push(named_field(expr.op, spec.name, joined));
                    } else {
                        fields.push(joined);
                    }
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }
                for i in 0..entries.len() {
                    if entries[..i].iter().any(|(key, _)| key == &entries[i].0) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }
                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let key = encode(key, reserved);
                        let value = encode(value, reserved);
                        let field = if expr.op == Operator::Semicolon && value.is_empty() {
                            key
                        } else {
                            format!("{key}={value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    if matches!(expr.op, Operator::Semicolon | Operator::Query | Operator::Continue) {
                        fields.push(named_field(expr.op, spec.name, joined));
                    } else {
                        fields.push(joined);
                    }
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    let (prefix, separator) = match expr.op {
        Operator::Simple | Operator::Reserved => ("", ","),
        Operator::Fragment => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Query => ("?", "&"),
        Operator::Continue => ("&", "&"),
    };
    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut out = String::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = template[open + 1..].find('}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;
                if let Some(relative_nested) = template[open + 1..close].find('{') {
                    return Err(Error::NestedOpen { pos: open + 1 + relative_nested });
                }
                let expression = parse_expression(&template[open + 1..close], open + 1, close)?;
                cursor = close + 1;
                // Syntax is checked in a separate pass below before any bindings are expanded.
                out.push('\0');
                let _ = expression;
            }
            _ => cursor += 1,
        }
    }

    let mut result = String::new();
    let mut cursor = 0;
    let mut marker = 0;
    while cursor < bytes.len() {
        if bytes[cursor] == b'{' {
            let close = cursor + 1 + template[cursor + 1..].find('}').expect("validated expression");
            result.push_str(&template[marker..cursor]);
            let expression = parse_expression(&template[cursor + 1..close], cursor + 1, close)?;
            result.push_str(&expand_expression(&expression, vars)?);
            cursor = close + 1;
            marker = cursor;
        } else {
            cursor += 1;
        }
    }
    result.push_str(&template[marker..]);
    Ok(result)
}

    Unclosed { pos: usize },
    NestedOpen { pos: usize },

}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone)]
struct Spec {
    name: String,
    pos: usize,
    modifier: Modifier,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Clone, Copy)]
enum Operator {
    Simple,
    Plus,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continue,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Self::Simple | Self::Plus | Self::Fragment => ',',
            Self::Dot => '.',
            Self::Slash => '/',
            Self::Semicolon | Self::Continue => ';',
            Self::Query => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Self::Simple | Self::Plus => None,
            Self::Fragment => Some('#'),
            Self::Dot => Some('.'),
            Self::Slash => Some('/'),
            Self::Semicolon => Some(';'),
            Self::Query => Some('?'),
            Self::Continue => Some('&'),
        }
    }

    fn named(self) -> bool {
        matches!(self, Self::Semicolon | Self::Query | Self::Continue)
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Plus | Self::Fragment)
    }
}

struct Expression {
    start: usize,
    end: usize,
    operator: Operator,
    specs: Vec<Spec>,
}

fn is_name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}

fn is_name_continue(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'_'
}

fn parse_expression(
    template: &str,
    open: usize,
    close: usize,
) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;

    let operator = if content_start < close {
        match bytes[content_start] {
            b'+' => {
                content_start += 1;
                Operator::Plus
            }
            b'#' => {
                content_start += 1;
                Operator::Fragment
            }
            b'.' => {
                content_start += 1;
                Operator::Dot
            }
            b'/' => {
                content_start += 1;
                Operator::Slash
            }
            b';' => {
                content_start += 1;
                Operator::Semicolon
            }
            b'?' => {
                content_start += 1;
                Operator::Query
            }
            b'&' => {
                content_start += 1;
                Operator::Continue
            }
            b'!' | b'=' | b',' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: content_start });
            }
            _ => Operator::Simple,
        }
    } else {
        Operator::Simple
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut cursor = content_start;

    loop {
        let entry_start = cursor;
        while cursor < close && bytes[cursor] != b',' {
            cursor += 1;
        }
        let entry_end = cursor;

        if entry_start == entry_end {
            return Err(Error::EmptyVariable { pos: entry_start });
        }

        let mut modifier_at = entry_end;
        for i in entry_start..entry_end {
            if bytes[i] == b'*' || bytes[i] == b':' {
                modifier_at = i;
                break;
            }
        }

        if modifier_at == entry_start {
            return Err(Error::InvalidName { pos: entry_start });
        }

        if !is_name_start(bytes[entry_start]) {
            return Err(Error::InvalidName { pos: entry_start });
        }
        let mut i = entry_start + 1;
        while i < modifier_at {
            if !is_name_continue(bytes[i]) {
                return Err(Error::InvalidName { pos: i });
            }
            i += 1;
        }

        let modifier = if modifier_at == entry_end {
            Modifier::None
        } else if bytes[modifier_at] == b'*' {
            if modifier_at + 1 != entry_end {
                return Err(Error::InvalidModifier { pos: modifier_at });
            }
            Modifier::Explode
        } else {
            let suffix = &bytes[modifier_at + 1..entry_end];
            if suffix.is_empty()
                || suffix.len() > 4
                || !suffix[0].is_ascii_digit()
                || suffix[0] == b'0'
                || suffix.iter().any(|b| !b.is_ascii_digit())
            {
                return Err(Error::InvalidModifier { pos: modifier_at });
            }
            let mut n = 0usize;
            for &b in suffix {
                n = n * 10 + (b - b'0') as usize;
            }
            if !(1..=9999).contains(&n) {
                return Err(Error::InvalidModifier { pos: modifier_at });
            }
            Modifier::Prefix(n)
        };

        let name = template[entry_start..modifier_at].to_owned();
        if specs.iter().any(|s: &Spec| s.name == name) {
            return Err(Error::DuplicateVariable { pos: entry_start });
        }
        specs.push(Spec {
            name,
            pos: entry_start,
            modifier,
        });

        if cursor == close {
            break;
        }
        cursor += 1;
        if cursor == close {
            return Err(Error::EmptyVariable { pos: close });
        }
    }

    Ok(Expression {
        start: open,
        end: close,
        operator,
        specs,
    })
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(candidate, _)| *candidate == name)
        .map(|(_, value)| value)
}

fn prefix_chars(s: &str, n: usize) -> String {
    s.chars().take(n).collect()
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }

        let unreserved = b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~');
        let reserved_byte = reserved
            && matches!(
                b,
                b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$'
                    | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            );

        if unreserved || reserved_byte {
            out.push(b as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 0x0f) as usize] as char);
        }
        i += 1;
    }
    out
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    if matches!(operator, Operator::Semicolon) && value.is_empty() {
        name.to_owned()
    } else {

    }

fn expand_expression(expr: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expr.operator;
    let mut fields = Vec::new();

    for spec in &expr.specs {
        let Some(value) = lookup(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.pos });
        }

        match value {
            Value::Scalar(s) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(n) => prefix_chars(s, n),
                    _ => s.clone(),
                };
                let encoded = encode(&selected, op.reserved());
                fields.push(if op.named() {
                    named_field(&spec.name, encoded, op)
                } else {
                    encoded
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for item in items {
                        let encoded = encode(item, op.reserved());
                        fields.push(if op.named() {
                            named_field(&spec.name, encoded, op)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, op.reserved()))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if op.named() {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                for i in 0..entries.len() {
                    if entries[..i].iter().any(|(k, _)| k == &entries[i].0) {
                        return Err(Error::DuplicateKey { pos: spec.pos });
                    }
                }
                if entries.is_empty() {
                    continue;
                }
                let exploded = matches!(spec.modifier, Modifier::Explode);
                if exploded {
                    for (key, value) in entries {
                        let key = encode(key, op.reserved());
                        let value = encode(value, op.reserved());
                        if matches!(op, Operator::Semicolon) && value.is_empty() {
                            fields.push(key);
                        } else {
                            fields.push(format!("{key}={value}"));
                        }
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, op.reserved()));
                        components.push(encode(value, op.reserved()));
                    }
                    let joined = components.join(",");
                    fields.push(if op.named() {
                        named_field(&spec.name, joined, op)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut out = String::new();
    if let Some(prefix) = op.prefix() {
        out.push(prefix);
    }
    out.push_str(&fields.join(&op.separator().to_string()));
    Ok(out)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(relative_close) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + relative_close;
                if let Some(relative_inner) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_inner,
                    });
                }
                expressions.push(parse_expression(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut out = String::new();
    let mut cursor = 0;
    for expr in &expressions {
        out.push_str(&template[cursor..expr.start]);
        out.push_str(&expand_expression(expr, vars)?);
        cursor = expr.end + 1;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone, Copy)]
enum Op {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semi,
    Query,
    Amp,
}

impl Op {
    fn separator(self) -> &'static str {
        match self {
            Self::Simple | Self::Plus | Self::Hash => ",",
            Self::Dot => ".",
            Self::Slash => "/",
            Self::Semi | Self::Amp => ";",
            Self::Query => "&",
        }
    }

    fn prefix(self) -> &'static str {
        match self {
            Self::Simple | Self::Plus => "",
            Self::Hash => "#",
            Self::Dot => ".",
            Self::Slash => "/",
            Self::Semi => ";",
            Self::Query => "?",
            Self::Amp => "&",
        }
    }

    fn reserved(self) -> bool {
        matches!(self, Self::Plus | Self::Hash)
    }

    fn named(self) -> bool {
        matches!(self, Self::Semi | Self::Query | Self::Amp)
    }
}

#[derive(Clone, Copy)]
enum Mod {
    Plain,
    Explode,
    Prefix(usize),
}

struct Var {
    name: String,
    pos: usize,
    modifier: Mod,
}

struct Expr {
    open: usize,
    close: usize,
    op: Op,
    vars: Vec<Var>,
}

fn parse(template: &str, open: usize, close: usize) -> Result<Expr, Error> {
    let b = template.as_bytes();
    let mut p = open + 1;
    let op = if p < close {
        match b[p] {
            b'+' => { p += 1; Op::Plus }
            b'#' => { p += 1; Op::Hash }
            b'.' => { p += 1; Op::Dot }
            b'/' => { p += 1; Op::Slash }
            b';' => { p += 1; Op::Semi }
            b'?' => { p += 1; Op::Query }
            b'&' => { p += 1; Op::Amp }
            b'!' | b'=' | b',' | b'@' | b'|' => {
                return Err(Error::UnsupportedOperator { pos: p });
            }
            _ => Op::Simple,
        }
    } else {
        Op::Simple
    };

    if p == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut vars = Vec::new();
    while p <= close {
        let start = p;
        while p < close && b[p] != b',' {
            p += 1;
        }
        let end = p;
        if start == end {
            return Err(Error::EmptyVariable { pos: start });
        }

        let mod_at = (start..end)
            .find(|&i| b[i] == b'*' || b[i] == b':')
            .unwrap_or(end);
        if mod_at == start {
            return Err(Error::InvalidName { pos: start });
        }
        if !(b[start].is_ascii_alphabetic() || b[start] == b'_') {
            return Err(Error::InvalidName { pos: start });
        }
        for i in start + 1..mod_at {
            if !(b[i].is_ascii_alphanumeric() || b[i] == b'_') {
                return Err(Error::InvalidName { pos: i });
            }
        }

        let modifier = if mod_at == end {
            Mod::Plain
        } else if b[mod_at] == b'*' {
            if mod_at + 1 != end {
                return Err(Error::InvalidModifier { pos: mod_at });
            }
            Mod::Explode
        } else {
            let digits = &b[mod_at + 1..end];
            if digits.is_empty()
                || digits.len() > 4
                || digits[0] == b'0'
                || digits.iter().any(|d| !d.is_ascii_digit())
            {
                return Err(Error::InvalidModifier { pos: mod_at });
            }
            let n = digits.iter().fold(0usize, |n, d| n * 10 + (d - b'0') as usize);
            if n == 0 || n > 9999 {
                return Err(Error::InvalidModifier { pos: mod_at });
            }
            Mod::Prefix(n)
        };

        let name = template[start..mod_at].to_owned();
        if vars.iter().any(|v: &Var| v.name == name) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        vars.push(Var { name, pos: start, modifier });

        if p == close {
            break;
        }
        p += 1;
        if p == close {
            return Err(Error::EmptyVariable { pos: close });
        }
    }

    Ok(Expr { open, close, op, vars })
}

fn encode(s: &str, reserved: bool) -> String {
    let bytes = s.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let c = bytes[i];
        if c == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
            continue;
        }
        let safe = c.is_ascii_alphanumeric()
            || matches!(c, b'-' | b'.' | b'_' | b'~')
            || (reserved && matches!(
                c, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$'
                    | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
            ));
        if safe {
            out.push(c as char);
        } else {
            const H: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(H[(c >> 4) as usize] as char);
            out.push(H[(c & 15) as usize] as char);
        }
        i += 1;
    }
    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    for (candidate, value) in vars.iter().rev() {
        if *candidate == name {
            return Some(value);
        }
    }
    None
}

fn named(name: &str, value: String, op: Op) -> String {
    if matches!(op, Op::Semi) && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn evaluate(expr: &Expr, bindings: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields: Vec<String> = Vec::new();
    for var in &expr.vars {
        let Some(value) = lookup(&var.name, bindings) else {
            continue;
        };
        if matches!(var.modifier, Mod::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: var.pos });
        }
        match value {
            Value::Scalar(s) => {
                let text = match var.modifier {
                    Mod::Prefix(n) => s.chars().take(n).collect::<String>(),
                    _ => s.clone(),
                };
                let text = encode(&text, expr.op.reserved());
                fields.push(if expr.op.named() { named(&var.name, text, expr.op) } else { text });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                if matches!(var.modifier, Mod::Explode) {
                    for item in items {
                        let text = encode(item, expr.op.reserved());
                        fields.push(if expr.op.named() { named(&var.name, text, expr.op) } else { text });
                    }
                } else {
                    let text = items.iter()
                        .map(|s| encode(s, expr.op.reserved()))
                        .collect::<Vec<_>>().join(",");
                    fields.push(if expr.op.named() { named(&var.name, text, expr.op) } else { text });
                }
            }
            Value::Map(entries) => {
                for i in 0..entries.len() {
                    if entries[..i].iter().any(|(key, _)| key == &entries[i].0) {
                        return Err(Error::DuplicateKey { pos: var.pos });
                    }
                }
                if entries.is_empty() {
                    continue;
                }
                if matches!(var.modifier, Mod::Explode) {
                    for (key, val) in entries {
                        let key = encode(key, expr.op.reserved());
                        let val = encode(val, expr.op.reserved());
                        fields.push(if matches!(expr.op, Op::Semi) && val.is_empty() {
                            key
                        } else {
                            format!("{key}={val}")
                        });
                    }
                } else {
                    let mut parts = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        parts.push(encode(key, expr.op.reserved()));
                        parts.push(encode(val, expr.op.reserved()));
                    }
                    let text = parts.join(",");
                    fields.push(if expr.op.named() { named(&var.name, text, expr.op) } else { text });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    let mut result = String::from(expr.op.prefix());
    result.push_str(&fields.join(expr.op.separator()));
    Ok(result)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut parsed = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(distance) = bytes[open + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = open + 1 + distance;
                if let Some(distance) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: open + 1 + distance });
                }
                parsed.push(parse(template, open, close)?);
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut previous = 0;
    for expr in &parsed {
        output.push_str(&template[previous..expr.open]);
        output.push_str(&evaluate(expr, vars)?);
        previous = expr.close + 1;
    }
    output.push_str(&template[previous..]);
    Ok(output)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone)]
struct Spec {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Clone, Copy)]
enum Op {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semi,
    Query,
    Amp,
}

struct Expr {
    op: Op,
    specs: Vec<Spec>,
}

enum Part {
    Literal(String),
    Expression(Expr),
}

fn parse(template: &str) -> Result<Vec<Part>, Error> {
    let bytes = template.as_bytes();
    let mut parts = Vec::new();
    let mut literal_start = 0;
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                if literal_start < i {
                    parts.push(Part::Literal(template[literal_start..i].to_owned()));
                }
                let open = i;
                let close = match bytes[i + 1..].iter().position(|&b| b == b'}') {
                    Some(n) => i + 1 + n,
                    None => return Err(Error::Unclosed { pos: open }),
                };
                if let Some(n) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + n });
                }
                parts.push(Part::Expression(parse_expr(template, open, close)?));
                i = close + 1;
                literal_start = i;
            }
            _ => i += 1,
        }
    }
    if literal_start < bytes.len() {
        parts.push(Part::Literal(template[literal_start..].to_owned()));
    }
    Ok(parts)
}

fn parse_expr(template: &str, open: usize, close: usize) -> Result<Expr, Error> {
    let bytes = template.as_bytes();
    let mut start = open + 1;
    let op = if start < close {
        match bytes[start] {
            b'+' => { start += 1; Op::Reserved }
            b'#' => { start += 1; Op::Fragment }
            b'.' => { start += 1; Op::Dot }
            b'/' => { start += 1; Op::Slash }
            b';' => { start += 1; Op::Semi }
            b'?' => { start += 1; Op::Query }
            b'&' => { start += 1; Op::Amp }
            b'!' | b'@' | b'|' | b'=' | b',' => {
                return Err(Error::UnsupportedOperator { pos: start });
            }
            _ => Op::Simple,
        }
    } else {
        Op::Simple
    };

    if start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = std::collections::HashSet::new();
    let mut item_start = start;
    loop {
        let item_end = bytes[item_start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(close);
        if item_start == item_end {
            return Err(Error::EmptyVariable { pos: item_start });
        }

        let spec_bytes = &bytes[item_start..item_end];
        let cut = spec_bytes.iter().position(|&b| b == b'*' || b == b':').unwrap_or(spec_bytes.len());
        let name_bytes = &spec_bytes[..cut];
        if name_bytes.is_empty() {
            return Err(Error::InvalidName { pos: item_start });
        }

        let first = name_bytes[0];
        if !first.is_ascii_alphabetic() && first != b'_' {
            return Err(Error::InvalidName { pos: item_start });
        }
        for (offset, &b) in name_bytes.iter().enumerate().skip(1) {
            if !b.is_ascii_alphanumeric() && b != b'_' {
                return Err(Error::InvalidName { pos: item_start + offset });
            }
        }

        let modifier = if cut == spec_bytes.len() {
            Modifier::None
        } else {
            let mpos = item_start + cut;
            let suffix = &spec_bytes[cut..];
            if suffix == b"*" {
                Modifier::Explode
            } else if suffix.len() >= 2
                && suffix[0] == b':'
                && suffix.len() <= 5
                && suffix[1].is_ascii_digit()
                && suffix[1] != b'0'
                && suffix[1..].iter().all(u8::is_ascii_digit)
            {
                let mut n = 0usize;
                for &d in &suffix[1..] {
                    n = n * 10 + usize::from(d - b'0');
                }
                if (1..=9999).contains(&n) {
                    Modifier::Prefix(n)
                } else {
                    return Err(Error::InvalidModifier { pos: mpos });
                }
            } else {
                return Err(Error::InvalidModifier { pos: mpos });
            }
        };

        let name = template[item_start..item_start + cut].to_owned();
        if !seen.insert(name.clone()) {
            return Err(Error::DuplicateVariable { pos: item_start });
        }
        specs.push(Spec { name, start: item_start, modifier });

        if item_end == close {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(Expr { op, specs })
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(n, _)| *n == name).map(|(_, v)| v)
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && bytes[i + 1].is_ascii_hexdigit()
            && bytes[i + 2].is_ascii_hexdigit()
        {
            out.push('%');
            out.push((bytes[i + 1] as char).to_ascii_uppercase());
            out.push((bytes[i + 2] as char).to_ascii_uppercase());
            i += 3;
        } else if b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
            || (reserved && matches!(b, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='))
        {
            out.push(b as char);
            i += 1;
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            out.push('%');
            out.push(HEX[(b >> 4) as usize] as char);
            out.push(HEX[(b & 15) as usize] as char);
            i += 1;
        }
    }
    out
}

fn named_field(name: &str, value: String, op: Op) -> String {
    if matches!(op, Op::Semi) && value.is_empty() {
        name.to_owned()
    } else {
        format!("{name}={value}")
    }
}

fn fields_for(spec: &Spec, value: &Value, op: Op) -> Result<Vec<String>, Error> {
    let reserved = matches!(op, Op::Reserved | Op::Fragment);
    match value {
        Value::Scalar(s) => {
            let value = match spec.modifier {
                Modifier::Prefix(n) => s.chars().take(n).collect::<String>(),
                _ => s.clone(),
            };
            let encoded = encode(&value, reserved);
            Ok(vec![if matches!(op, Op::Semi | Op::Query | Op::Amp) {
                named_field(&spec.name, encoded, op)
            } else {
                encoded
            }])
        }
        Value::List(items) => {
            if matches!(spec.modifier, Modifier::Prefix(_)) {
                return Err(Error::TypeMismatch { pos: spec.start });
            }
            if items.is_empty() {
                return Ok(Vec::new());
            }
            let exploded = matches!(spec.modifier, Modifier::Explode);
            if exploded {
                Ok(items.iter().map(|item| {
                    let encoded = encode(item, reserved);
                    if matches!(op, Op::Semi | Op::Query | Op::Amp) {
                        named_field(&spec.name, encoded, op)
                    } else {
                        encoded
                    }
                }).collect())
            } else {
                let joined = items.iter().map(|item| encode(item, reserved)).collect::<Vec<_>>().join(",");
                Ok(vec![if matches!(op, Op::Semi | Op::Query | Op::Amp) {
                    named_field(&spec.name, joined, op)
                } else {
                    joined
                }])
            }
        }
        Value::Map(entries) => {
            if matches!(spec.modifier, Modifier::Prefix(_)) {
                return Err(Error::TypeMismatch { pos: spec.start });
            }
            if entries.is_empty() {
                return Ok(Vec::new());
            }
            let mut keys = std::collections::HashSet::new();
            for (key, _) in entries {
                if !keys.insert(key) {
                    return Err(Error::DuplicateKey { pos: spec.start });
                }
            }
            let exploded = matches!(spec.modifier, Modifier::Explode);
            if exploded {
                Ok(entries.iter().map(|(key, value)| {
                    let key = encode(key, reserved);
                    let value = encode(value, reserved);
                    if matches!(op, Op::Semi) && value.is_empty() {
                        key
                    } else {
                        format!("{key}={value}")
                    }
                }).collect())
            } else {
                let mut components = Vec::with_capacity(entries.len() * 2);
                for (key, value) in entries {
                    components.push(encode(key, reserved));
                    components.push(encode(value, reserved));
                }
                let joined = components.join(",");
                Ok(vec![if matches!(op, Op::Semi | Op::Query | Op::Amp) {
                    named_field(&spec.name, joined, op)
                } else {
                    joined
                }])
            }
        }
    }
}

fn expand_expr(expr: &Expr, vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields = Vec::new();
    for spec in &expr.specs {
        if let Some(value) = lookup(&spec.name, vars) {
            fields.extend(fields_for(spec, value, expr.op)?);
        }
    }
    if fields.is_empty() {
        return Ok(String::new());
    }
    let (prefix, separator) = match expr.op {
        Op::Simple | Op::Reserved => ("", ","),
        Op::Fragment => ("#", ","),
        Op::Dot => (".", "."),
        Op::Slash => ("/", "/"),
        Op::Semi => (";", ";"),
        Op::Query => ("?", "&"),
        Op::Amp => ("&", "&"),
    };
    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let parts = parse(template)?;
    let mut out = String::new();
    for part in parts {
        match part {
            Part::Literal(s) => out.push_str(&s),
            Part::Expression(expr) => out.push_str(&expand_expr(&expr, vars)?),
        }
    }
    Ok(out)
}

pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

    Plus,

    Slash,

    Amp,

enum Mod {

}

struct Spec {

    pos: usize,
    modifier: Mod,

fn name_start(b: u8) -> bool {
    b.is_ascii_alphabetic() || b == b'_'
}
fn name_char(b: u8) -> bool {
    name_start(b) || b.is_ascii_digit()
}

fn parse(template: &str) -> Result<Vec<(usize, usize, Op, Vec<Spec>)>, Error> {
    let b = template.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;
    while i < b.len() {
        if b[i] == b'}' {
            return Err(Error::UnexpectedClose { pos: i });
        }
        if b[i] != b'{' {
            i += 1;
            continue;
        }
        let start = i;
        let Some(rel_end) = b[start + 1..].iter().position(|x| *x == b'}') else {
            return Err(Error::Unclosed { pos: start });
        };
        let end = start + 1 + rel_end;
        if let Some(rel_nested) = b[start + 1..end].iter().position(|x| *x == b'{') {
            return Err(Error::NestedOpen { pos: start + 1 + rel_nested });
        }

        let mut p = start + 1;
        let op = match b[p] {
            b'+' => { p += 1; Op::Plus }
            b'#' => { p += 1; Op::Hash }
            b'.' => { p += 1; Op::Dot }
            b'/' => { p += 1; Op::Slash }
            b';' => { p += 1; Op::Semi }
            b'?' => { p += 1; Op::Query }
            b'&' => { p += 1; Op::Amp }
            b'!' | b'=' | b'@' | b'|' => return Err(Error::UnsupportedOperator { pos: p }),
            _ => Op::Simple,
        };
        if p == end {
            return Err(Error::EmptyExpression { pos: end });
        }

        let mut specs = Vec::new();
        let mut names = Vec::<String>::new();
        let mut begin = p;
        loop {
            let finish = b[begin..end].iter().position(|x| *x == b',')
                .map(|r| begin + r).unwrap_or(end);
            if begin == finish {
                return Err(Error::EmptyVariable { pos: begin });
            }
            let mod_idx = b[begin..finish].iter().position(|x| *x == b'*' || *x == b':');
            let name_end = mod_idx.map(|x| begin + x).unwrap_or(finish);
            if name_end == begin || !name_start(b[begin]) {
                return Err(Error::InvalidName { pos: begin });
            }
            for n in begin + 1..name_end {
                if !name_char(b[n]) {
                    return Err(Error::InvalidName { pos: n });
                }
            }
            let modifier = if let Some(mi) = mod_idx {
                let mp = begin + mi;
                let s = &b[mp..finish];
                if s == b"*" {
                    Mod::Explode
                } else if s.first() == Some(&b':')
                    && (2..=5).contains(&s.len())
                    && s[1] >= b'1' && s[1] <= b'9'
                    && s[1..].iter().all(u8::is_ascii_digit)
                {
                    let mut n = 0usize;
                    for x in &s[1..] { n = n * 10 + (x - b'0') as usize; }
                    if n <= 9999 { Mod::Prefix(n) }
                    else { return Err(Error::InvalidModifier { pos: mp }); }
                } else {
                    return Err(Error::InvalidModifier { pos: mp });
                }
            } else { Mod::None };

            let name = template[begin..name_end].to_owned();
            if names.iter().any(|x| x == &name) {
                return Err(Error::DuplicateVariable { pos: begin });
            }
            names.push(name.clone());
            specs.push(Spec { name, pos: begin, modifier });
            if finish == end { break; }
            begin = finish + 1;
        }
        result.push((start, end, op, specs));
        i = end + 1;
    }
    Ok(result)
}

fn hex(v: u8) -> Option<u8> {
    match v {
        b'0'..=b'9' => Some(v - b'0'),
        b'a'..=b'f' => Some(v - b'a' + 10),
        b'A'..=b'F' => Some(v - b'A' + 10),
        _ => None,
    }
}
fn safe(v: u8, reserved: bool) -> bool {
    v.is_ascii_alphanumeric() || matches!(v, b'-' | b'.' | b'_' | b'~')
        || (reserved && matches!(v, b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='))
}
fn enc(s: &str, reserved: bool) -> String {
    let b = s.as_bytes();
    let mut out = String::new();
    let mut i = 0;
    const H: &[u8] = b"0123456789ABCDEF";
    while i < b.len() {
        if b[i] == b'%' && i + 2 < b.len() && hex(b[i + 1]).is_some() && hex(b[i + 2]).is_some() {
            out.push('%');
            out.push(b[i + 1].to_ascii_uppercase() as char);
            out.push(b[i + 2].to_ascii_uppercase() as char);
            i += 3;
        } else if safe(b[i], reserved) {
            out.push(b[i] as char);
            i += 1;
        } else {
            out.push('%');
            out.push(H[(b[i] >> 4) as usize] as char);
            out.push(H[(b[i] & 15) as usize] as char);
            i += 1;
        }
    }
    out
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(n, _)| *n == name).map(|(_, v)| v)
}
fn named(op: Op, name: &str, value: String) -> String {
    match op {
        Op::Semi if value.is_empty() => name.to_owned(),
        Op::Semi | Op::Query | Op::Amp => format!("{name}={value}"),
        _ => value,
    }
}
fn op_reserved(op: Op) -> bool {
    matches!(op, Op::Plus | Op::Hash)
}
fn values(spec: &Spec, op: Op, vars: &[(&str, Value)]) -> Result<Vec<String>, Error> {
    let Some(v) = lookup(&spec.name, vars) else { return Ok(Vec::new()) };
    let reserved = op_reserved(op);
    match v {
        Value::Scalar(s) => {
            let raw = match spec.modifier {
                Mod::Prefix(n) => s.chars().take(n).collect::<String>(),
                _ => s.clone(),
            };
            Ok(vec![named(op, &spec.name, enc(&raw, reserved))])
        }
        Value::List(xs) => {
            if matches!(spec.modifier, Mod::Prefix(_)) {
                return Err(Error::TypeMismatch { pos: spec.pos });
            }
            if xs.is_empty() { return Ok(Vec::new()); }
            if matches!(spec.modifier, Mod::Explode) {
                Ok(xs.iter().map(|x| named(op, &spec.name, enc(x, reserved))).collect())
            } else {
                let joined = xs.iter().map(|x| enc(x, reserved)).collect::<Vec<_>>().join(",");
                Ok(vec![named(op, &spec.name, joined)])
            }
        }
        Value::Map(m) => {
            if matches!(spec.modifier, Mod::Prefix(_)) {
                return Err(Error::TypeMismatch { pos: spec.pos });
            }
            for i in 0..m.len() {
                if m[..i].iter().any(|(k, _)| k == &m[i].0) {
                    return Err(Error::DuplicateKey { pos: spec.pos });
                }
            }
            if m.is_empty() { return Ok(Vec::new()); }
            if matches!(spec.modifier, Mod::Explode) {
                Ok(m.iter().map(|(k, v)| {
                    let key = enc(k, reserved);
                    let val = enc(v, reserved);
                    if matches!(op, Op::Semi) && val.is_empty() { key }
                    else { format!("{key}={val}") }
                }).collect())
            } else {
                let mut all = Vec::with_capacity(m.len() * 2);
                for (k, v) in m {
                    all.push(enc(k, reserved));
                    all.push(enc(v, reserved));
                }
                Ok(vec![named(op, &spec.name, all.join(","))])
            }
        }
    }
}
fn render(op: Op, specs: &[Spec], vars: &[(&str, Value)]) -> Result<String, Error> {
    let mut fields = Vec::new();
    for s in specs { fields.extend(values(s, op, vars)?); }
    if fields.is_empty() { return Ok(String::new()); }
    let (prefix, sep) = match op {
        Op::Simple | Op::Plus => ("", ","),
        Op::Hash => ("#", ","),
        Op::Dot => (".", "."),
        Op::Slash => ("/", "/"),
        Op::Semi => (";", ";"),
        Op::Query => ("?", "&"),
        Op::Amp => ("&", "&"),
    };
    Ok(format!("{prefix}{}", fields.join(sep)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let parsed = parse(template)?;
    let mut out = String::new();
    let mut cursor = 0;
    for (start, end, op, specs) in parsed {
        out.push_str(&template[cursor..start]);
        out.push_str(&render(op, &specs, vars)?);
        cursor = end + 1;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
