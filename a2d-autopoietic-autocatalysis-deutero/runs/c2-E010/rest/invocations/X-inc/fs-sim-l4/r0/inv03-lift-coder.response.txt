```rust
use std::collections::{BTreeMap, BTreeSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

#[derive(Debug, Clone)]
pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

fn validate(path: &str) -> Result<(), Error> {
    if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(Error::InvalidPath(i));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty() && *part != ".")
        .map(str::to_owned)
        .collect()
}

fn normalize_from(base: &str, path: &str) -> String {
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(base)
    };
    for part in path.split('/') {
        match part {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            _ => parts.push(part.to_owned()),
        }
    }
    make_path(&parts)
}

fn make_path(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn parent(path: &str) -> String {
    if path == "/" {
        return "/".to_owned();
    }
    match path.rfind('/') {
        Some(0) | None => "/".to_owned(),
        Some(i) => path[..i].to_owned(),
    }
}

fn basename(path: &str) -> String {
    path.rsplit('/').next().unwrap_or("").to_owned()
}

fn is_within(path: &str, root: &str) -> bool {
    if root == "/" {
        return path.starts_with('/');
    }
    path == root
        || path
            .strip_prefix(root)
            .is_some_and(|suffix| suffix.starts_with('/'))
}

#[derive(Default)]
struct Walk {
    visited: BTreeSet<String>,
    expansions: usize,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_owned(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_owned(),
        }
    }

    fn canonical(&self, path: &str) -> Result<String, Error> {
        validate(path)?;
        Ok(normalize_from(&self.cwd, path))
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        create_final: bool,
    ) -> Result<String, Error> {
        self.resolve_with(path, follow_final, create_final, &mut Walk::default())
    }

    fn resolve_with(
        &self,
        path: &str,
        follow_final: bool,
        create_final: bool,
        walk: &mut Walk,
    ) -> Result<String, Error> {
        let mut pending = components(path);
        let mut physical = Vec::<String>::new();

        loop {
            if pending.is_empty() {
                return Ok(make_path(&physical));
            }

            let part = pending.remove(0);
            physical.push(part);
            let current = make_path(&physical);
            let is_final = pending.is_empty();

            let entry = match self.entries.get(&current) {
                Some(entry) => entry,
                None if is_final && create_final => {
                    return Ok(current);
                }
                None => return Err(Error::Missing(current)),
            };

            if !is_final && entry.kind() != Kind::Dir {
                return Err(Error::NotDir(current));
            }

            let should_follow = entry.kind() == Kind::Link && (!is_final || follow_final);
            if should_follow {
                if walk.visited.contains(&current) {
                    return Err(Error::Loop(current));
                }
                if walk.expansions >= 8 {
                    return Err(Error::Depth(current));
                }
                walk.visited.insert(current.clone());
                walk.expansions += 1;

                let target = match entry {
                    Entry::Link(target) => target,
                    _ => unreachable!(),
                };
                let target_path = if target.starts_with('/') {
                    normalize_from("/", target)
                } else {
                    normalize_from(&parent(&current), target)
                };

                let mut expanded = components(&target_path);
                expanded.extend(pending);
                let expanded_path = make_path(&expanded);
                pending = components(&expanded_path);
                physical.clear();
            }
        }
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let p = parent(path);
        let name = basename(path);
        let physical_parent = self.resolve(&p, true, false)?;
        match self.entries.get(&physical_parent) {
            Some(Entry::Dir) => Ok((physical_parent, name)),
            Some(_) => Err(Error::NotDir(physical_parent)),
            None => Err(Error::Missing(physical_parent)),
        }
    }

    fn child_path(dir: &str, name: &str) -> String {
        if dir == "/" {
            format!("/{name}")
        } else {
            format!("{dir}/{name}")
        }
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries.keys().any(|p| p != path && parent(p) == path)
    }

    fn subtree_paths(&self, path: &str) -> Vec<String> {
        self.entries
            .keys()
            .filter(|candidate| is_within(candidate, path))
            .cloned()
            .collect()
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.canonical(path)?;
        let (p, name) = self.resolve_parent(&canonical)?;
        let dest = Self::child_path(&p, &name);
        if self.entries.contains_key(&dest) {
            return Err(Error::Exists(dest));
        }
        self.entries.insert(dest, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = self.canonical(path)?;
        let resolved = self.resolve(&canonical, true, true)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => return Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) | None => {}
            Some(Entry::Link(_)) => unreachable!(),
        }
        let p = parent(&resolved);
        match self.entries.get(&p) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(p)),
            None => return Err(Error::Missing(p)),
        }
        self.entries
            .insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = self.canonical(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = self.canonical(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {}
            Some(_) => return Err(Error::NotDir(resolved)),
            None => return Err(Error::Missing(resolved)),
        }
        let mut result = Vec::new();
        for (candidate, entry) in &self.entries {
            if candidate != &resolved && parent(candidate) == resolved {
                result.push((basename(candidate), entry.kind()));
            }
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = self.canonical(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Dir) => {
                self.cwd = resolved;
                Ok(())
            }
            Some(_) => Err(Error::NotDir(resolved)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn rename(&mut self, from: &str, to: &str, overwrite: bool) -> Result<(), Error> {
        let source = self.canonical(from)?;
        let destination = self.canonical(to)?;
        if source == "/" || destination == "/" {
            return Err(Error::Root);
        }

        let (src_parent, src_name) = self.resolve_parent(&source)?;
        let src = Self::child_path(&src_parent, &src_name);
        let src_entry = self.entries.get(&src).cloned().ok_or_else(|| Error::Missing(src.clone()))?;

        if src_entry.kind() == Kind::Dir && is_within(&self.cwd, &src) {
            return Err(Error::Busy(src));
        }

        let (dst_parent, dst_name) = self.resolve_parent(&destination)?;
        let dst = Self::child_path(&dst_parent, &dst_name);

        if let Some(Entry::Dir(_)) = self.entries.get(&dst) {
            if is_within(&self.cwd, &dst) {
                return Err(Error::Busy(dst));
            }
        }

        if src == dst {
            return Ok(());
        }

        if src_entry.kind() == Kind::Dir && is_within(&dst, &src) {
            return Err(Error::Cycle(dst));
        }

        let dst_entry = self.entries.get(&dst).cloned();
        if dst_entry.is_some() && !overwrite {
            return Err(Error::Exists(dst));
        }

        if let Some(existing) = &dst_entry {
            match (src_entry.kind(), existing.kind()) {
                (Kind::Dir, Kind::Dir) => {
                    if self.has_children(&dst) {
                        return Err(Error::NotEmpty(dst));
                    }
                }
                (Kind::Dir, _) => return Err(Error::NotDir(dst)),
                (_, Kind::Dir) => return Err(Error::IsDir(dst)),
                _ => {}
            }
        }

        let source_paths = self.subtree_paths(&src);
        let mut moved = Vec::new();
        for old_path in source_paths {
            let suffix = &old_path[src.len()..];
            moved.push((old_path, format!("{dst}{suffix}")));
        }

        if dst_entry.is_some() {
            for old_path in self.subtree_paths(&dst) {
                self.entries.remove(&old_path);
            }
        }
        let mut saved = Vec::new();
        for (old_path, new_path) in moved {
            if let Some(entry) = self.entries.remove(&old_path) {
                saved.push((new_path, entry));
            }
        }
        for (new_path, entry) in saved {
            self.entries.insert(new_path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = self.canonical(path)?;
        if canonical == "/" {
            return Err(Error::Root);
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let resolved = Self::child_path(&p, &name);
        let entry = self
            .entries
            .get(&resolved)
            .cloned()
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        if entry.kind() == Kind::Dir && is_within(&self.cwd, &resolved) {
            return Err(Error::Busy(resolved));
        }
        if entry.kind() == Kind::Dir && self.has_children(&resolved) && !recursive {
            return Err(Error::NotEmpty(resolved));
        }
        for candidate in self.subtree_paths(&resolved) {
            self.entries.remove(&candidate);
        }
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let canonical = self.canonical(link)?;
        if target.is_empty() {
            return Err(Error::InvalidTarget(0));
        }
        if let Some(i) = target.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidTarget(i));
        }
        let (p, name) = self.resolve_parent(&canonical)?;
        let resolved = Self::child_path(&p, &name);
        if self.entries.contains_key(&resolved) {
            return Err(Error::Exists(resolved));
        }
        self.entries
            .insert(resolved, Entry::Link(target.to_owned()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = self.canonical(path)?;
        let resolved = self.resolve(&canonical, false, false)?;
        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = self.canonical(path)?;
        let resolved = self.resolve(&canonical, true, false)?;
        let root_entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;
        if root_entry.kind() == Kind::Link {
            unreachable!();
        }

        let paths = self.subtree_paths(&resolved);
        let mut sizes = BTreeMap::<String, u64>::new();
        for p in &paths {
            let size = match self.entries.get(p).expect("subtree path is stored") {
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
                Entry::Dir => 0,
            };
            sizes.insert(p.clone(), size);
        }

        for p in paths.iter().rev() {
            let total = match self.entries.get(p).expect("subtree path is stored") {
                Entry::Dir => self
                    .entries
                    .keys()
                    .filter(|child| child != &p && parent(child) == *p)
                    .map(|child| sizes.get(child).copied().unwrap_or(0))
                    .sum(),
                _ => sizes.get(p).copied().unwrap_or(0),
            };
            sizes.insert(p.clone(), total);
        }

        Ok(sizes.into_iter().collect())
    }
}
```
