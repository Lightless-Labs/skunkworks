```rust
use std::collections::{BTreeMap, HashSet};

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
    Link,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
    InvalidTarget(usize),
    Loop(String),
    Depth(String),
}

#[derive(Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
    Link(String),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        let (parent, name) = split_parent(&canonical);
        let physical_parent = self.resolve(&parent, true, false)?;
        self.require_dir(&physical_parent)?;
        let destination = join(&physical_parent, &name);

        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }

        self.entries.insert(destination, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical, true, true)?;

        match self.entries.get(&resolved) {
            Some(Entry::Dir) => return Err(Error::IsDir(resolved)),
            Some(Entry::File(_)) => {}
            Some(Entry::Link(_)) => unreachable!(),
            None => {
                let (parent, _) = split_parent(&resolved);
                self.require_dir(&parent)?;
            }
        }

        self.entries
            .insert(resolved, Entry::File(data.to_vec()));
        Ok(())
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical, true, false)?;

        match self.entries.get(&resolved) {
            Some(Entry::File(data)) => Ok(data.clone()),
            Some(Entry::Dir) => Err(Error::IsDir(resolved)),
            Some(Entry::Link(_)) => unreachable!(),
            None => unreachable!(),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical, true, false)?;
        self.require_dir(&resolved)?;

        let mut result = Vec::new();
        for (entry_path, entry) in &self.entries {
            if entry_path == &resolved || parent_of(entry_path) != resolved {
                continue;
            }
            result.push((basename(entry_path), entry.kind()));
        }
        result.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(result)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical, true, false)?;
        self.require_dir(&resolved)?;
        self.cwd = resolved;
        Ok(())
    }

    pub fn rename(
        &mut self,
        from: &str,
        to: &str,
        overwrite: bool,
    ) -> Result<(), Error> {
        let from_canonical = normalize(from, &self.cwd)?;
        let to_canonical = normalize(to, &self.cwd)?;

        if from_canonical == "/" || to_canonical == "/" {
            return Err(Error::Root);
        }

        let (from_parent, from_name) = split_parent(&from_canonical);
        let source_parent = self.resolve(&from_parent, true, false)?;
        self.require_dir(&source_parent)?;
        let source = join(&source_parent, &from_name);
        let source_entry = self
            .entries
            .get(&source)
            .cloned()
            .ok_or_else(|| Error::Missing(source.clone()))?;

        if matches!(source_entry, Entry::Dir) && contains_path(&source, &self.cwd) {
            return Err(Error::Busy(source));
        }

        let (to_parent, to_name) = split_parent(&to_canonical);
        let destination_parent = self.resolve(&to_parent, true, false)?;
        self.require_dir(&destination_parent)?;
        let destination = join(&destination_parent, &to_name);

        if matches!(self.entries.get(&destination), Some(Entry::Dir))
            && contains_path(&destination, &self.cwd)
        {
            return Err(Error::Busy(destination));
        }

        if source == destination {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && contains_path(&source, &destination) {
            return Err(Error::Cycle(destination));
        }

        let destination_entry = self.entries.get(&destination).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(destination));
        }

        if let Some(existing) = destination_entry {
            match (&source_entry, &existing) {
                (Entry::Dir, Entry::Dir) => {
                    if self.has_children(&destination) {
                        return Err(Error::NotEmpty(destination));
                    }
                }
                (Entry::Dir, _) => return Err(Error::NotDir(destination)),
                (_, Entry::Dir) => return Err(Error::IsDir(destination)),
                _ => {}
            }
        }

        let mut moved = Vec::new();
        for (path, entry) in &self.entries {
            if contains_path(&source, path) {
                let suffix = &path[source.len()..];
                moved.push((format!("{destination}{suffix}"), entry.clone()));
            }
        }

        if destination_entry.is_some() {
            self.remove_subtree(&destination);
        }
        self.remove_subtree(&source);
        for (path, entry) in moved {
            self.entries.insert(path, entry);
        }
        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let canonical = normalize(path, &self.cwd)?;
        if canonical == "/" {
            return Err(Error::Root);
        }

        let (parent, name) = split_parent(&canonical);
        let physical_parent = self.resolve(&parent, true, false)?;
        self.require_dir(&physical_parent)?;
        let target = join(&physical_parent, &name);
        let entry = self
            .entries
            .get(&target)
            .cloned()
            .ok_or_else(|| Error::Missing(target.clone()))?;

        if matches!(entry, Entry::Dir) && contains_path(&target, &self.cwd) {
            return Err(Error::Busy(target));
        }

        if matches!(entry, Entry::Dir) && self.has_children(&target) && !recursive {
            return Err(Error::NotEmpty(target));
        }

        self.remove_subtree(&target);
        Ok(())
    }

    pub fn symlink(&mut self, target: &str, link: &str) -> Result<(), Error> {
        let link_canonical = normalize(link, &self.cwd)?;
        validate_target(target)?;

        let (parent, name) = split_parent(&link_canonical);
        let physical_parent = self.resolve(&parent, true, false)?;
        self.require_dir(&physical_parent)?;
        let destination = join(&physical_parent, &name);

        if self.entries.contains_key(&destination) {
            return Err(Error::Exists(destination));
        }

        self.entries
            .insert(destination, Entry::Link(target.to_string()));
        Ok(())
    }

    pub fn read_link(&self, path: &str) -> Result<String, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical, false, false)?;

        match self.entries.get(&resolved) {
            Some(Entry::Link(target)) => Ok(target.clone()),
            Some(_) => Err(Error::InvalidTarget(0)),
            None => Err(Error::Missing(resolved)),
        }
    }

    pub fn du(&self, path: &str) -> Result<Vec<(String, u64)>, Error> {
        let canonical = normalize(path, &self.cwd)?;
        let resolved = self.resolve(&canonical, true, false)?;
        let entry = self
            .entries
            .get(&resolved)
            .ok_or_else(|| Error::Missing(resolved.clone()))?;

        let mut paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| contains_path(&resolved, candidate))
            .cloned()
            .collect();
        paths.sort();

        let mut sizes = BTreeMap::new();
        for entry_path in paths.iter().rev() {
            let size = match self.entries.get(entry_path).unwrap() {
                Entry::Dir => {
                    let mut total = 0u64;
                    for child in &paths {
                        if parent_of(child) == *entry_path {
                            total = total.saturating_add(*sizes.get(child).unwrap_or(&0));
                        }
                    }
                    total
                }
                Entry::File(data) => data.len() as u64,
                Entry::Link(target) => target.len() as u64,
            };
            sizes.insert(entry_path.clone(), size);
        }

        let _ = entry;
        Ok(paths
            .into_iter()
            .map(|entry_path| {
                let size = *sizes.get(&entry_path).unwrap_or(&0);
                (entry_path, size)
            })
            .collect())
    }

    fn require_dir(&self, path: &str) -> Result<(), Error> {
        match self.entries.get(path) {
            Some(Entry::Dir) => Ok(()),
            Some(_) => Err(Error::NotDir(path.to_string())),
            None => Err(Error::Missing(path.to_string())),
        }
    }

    fn resolve(
        &self,
        path: &str,
        follow_final: bool,
        allow_missing_final: bool,
    ) -> Result<String, Error> {
        let mut pending = components(path);
        let mut resolved = Vec::<String>::new();
        let mut visited = HashSet::<String>::new();
        let mut expansions = 0usize;

        while let Some(component) = pending.first().cloned() {
            pending.remove(0);
            resolved.push(component);
            let current = from_components(&resolved);

            if !pending.is_empty() {
                match self.entries.get(&current) {
                    None => return Err(Error::Missing(current)),
                    Some(Entry::Dir) => {}
                    Some(_) => return Err(Error::NotDir(current)),
                }
            } else if !follow_final && allow_missing_final {
                return Ok(current);
            }

            let should_follow = !pending.is_empty() || follow_final;
            let Some(Entry::Link(target)) = self.entries.get(&current) else {
                if self.entries.contains_key(&current) {
                    continue;
                }
                if allow_missing_final && pending.is_empty() {
                    return Ok(current);
                }
                return Err(Error::Missing(current));
            };

            if !should_follow {
                continue;
            }

            if !visited.insert(current.clone()) {
                return Err(Error::Loop(current));
            }
            if expansions >= 8 {
                return Err(Error::Depth(current));
            }
            expansions += 1;

            let mut target_components = if target.starts_with('/') {
                Vec::new()
            } else {
                resolved[..resolved.len() - 1].to_vec()
            };
            append_normalized(&mut target_components, target);
            target_components.extend(pending);
            let expanded = from_components(&target_components);
            pending = components(&expanded);
            resolved.clear();
        }

        let result = from_components(&resolved);
        if !self.entries.contains_key(&result) {
            if allow_missing_final {
                return Ok(result);
            }
            return Err(Error::Missing(result));
        }
        Ok(result)
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && parent_of(candidate) == path)
    }

    fn remove_subtree(&mut self, path: &str) {
        let keys: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| contains_path(path, candidate))
            .cloned()
            .collect();
        for key in keys {
            self.entries.remove(&key);
        }
    }
}

impl Entry {
    fn kind(&self) -> Kind {
        match self {
            Entry::Dir => Kind::Dir,
            Entry::File(_) => Kind::File,
            Entry::Link(_) => Kind::Link,
        }
    }
}

fn validate_path(path: &str) -> Result<(), Error> {
    if let Some(index) = path.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(Error::InvalidPath(index));
    }
    if path.is_empty() {
        return Err(Error::InvalidPath(0));
    }
    Ok(())
}

fn validate_target(target: &str) -> Result<(), Error> {
    if target.is_empty() {
        return Err(Error::InvalidTarget(0));
    }
    if let Some(index) = target.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(Error::InvalidTarget(index));
    }
    Ok(())
}

fn normalize(path: &str, cwd: &str) -> Result<String, Error> {
    validate_path(path)?;
    let mut parts = if path.starts_with('/') {
        Vec::new()
    } else {
        components(cwd)
    };
    append_normalized(&mut parts, path);
    Ok(from_components(&parts))
}

fn append_normalized(parts: &mut Vec<String>, path: &str) {
    for component in path.split('/') {
        match component {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            name => parts.push(name.to_string()),
        }
    }
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_string)
        .collect()
}

fn from_components(parts: &[String]) -> String {
    if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    }
}

fn split_parent(path: &str) -> (String, String) {
    if path == "/" {
        return ("/".to_string(), String::new());
    }
    match path.rfind('/') {
        Some(0) => ("/".to_string(), path[1..].to_string()),
        Some(index) => (path[..index].to_string(), path[index + 1..].to_string()),
        None => ("/".to_string(), path.to_string()),
    }
}

fn join(parent: &str, name: &str) -> String {
    if parent == "/" {
        format!("/{name}")
    } else {
        format!("{parent}/{name}")
    }
}

fn parent_of(path: &str) -> String {
    split_parent(path).0
}

fn basename(path: &str) -> String {
    split_parent(path).1
}

fn contains_path(parent: &str, child: &str) -> bool {
    if parent == "/" {
        return child.starts_with('/');
    }
    child == parent
        || child
            .strip_prefix(parent)
            .is_some_and(|suffix| suffix.starts_with('/'))
}
```
