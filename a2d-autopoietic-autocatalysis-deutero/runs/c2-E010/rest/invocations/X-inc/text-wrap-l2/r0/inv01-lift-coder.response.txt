#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    InvalidControl { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }

    for (index, ch) in input.char_indices() {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let normalized: Vec<char> = input
        .chars()
        .map(|ch| if ch == '\r' { '\n' } else { ch })
        .collect();

    let mut output = Vec::new();
    let mut segment_start = 0;
    let mut i = 0;

    while i < normalized.len() {
        if normalized[i] == '\n' {
            let run_start = i;
            while i < normalized.len() && normalized[i] == '\n' {
                i += 1;
            }
            if i - run_start >= 2 {
                layout_segment(&normalized[segment_start..run_start], options, &mut output);
                for _ in 1..(i - run_start) {
                    output.push(String::new());
                }
                segment_start = i;
            }
        } else {
            i += 1;
        }
    }

    layout_segment(&normalized[segment_start..], options, &mut output);
    Ok(output)
}

fn is_separator(ch: char) -> bool {
    ch == ' ' || ch == '\t' || ch == '\n' || ch == '\r' || ch == '\u{00A0}'
}

fn layout_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let mut words: Vec<Vec<char>> = Vec::new();
    let mut word = Vec::new();

    for &ch in segment {
        if is_separator(ch) {
            if !word.is_empty() {
                words.push(std::mem::take(&mut word));
            }
        } else {
            word.push(ch);
        }
    }
    if !word.is_empty() {
        words.push(word);
    }

    if words.is_empty() {
        return;
    }

    let mut first_line = true;
    let mut content = String::new();
    let mut content_len = 0usize;

    for word in words {
        let mut suffix: Vec<char> = word;

        while !suffix.is_empty() {
            let indent = if first_line {
                options.first_indent
            } else {
                options.rest_indent
            };
            let capacity = options.width - indent;

            if suffix.len() <= capacity {
                let needed = suffix.len() + usize::from(content_len > 0);
                if content_len + needed <= capacity {
                    if content_len > 0 {
                        content.push(' ');
                        content_len += 1;
                    }
                    for ch in suffix.drain(..) {
                        content.push(ch);
                        content_len += 1;
                    }
                } else {
                    emit_line(
                        &mut content,
                        content_len,
                        indent,
                        output,
                    );
                    content_len = 0;
                    first_line = false;
                }
                continue;
            }

            if content_len == 0 {
                for ch in suffix.drain(..capacity) {
                    content.push(ch);
                    content_len += 1;
                }
                emit_line(&mut content, content_len, indent, output);
                content_len = 0;
                first_line = false;
            } else {
                let available = capacity - content_len - 1;
                if available >= 2 {
                    content.push(' ');
                    content_len += 1;
                    for ch in suffix.drain(..available) {
                        content.push(ch);
                        content_len += 1;
                    }
                    emit_line(&mut content, content_len, indent, output);
                    content_len = 0;
                    first_line = false;
                } else {
                    emit_line(&mut content, content_len, indent, output);
                    content_len = 0;
                    first_line = false;
                }
            }
        }
    }

    if content_len > 0 {
        let indent = if first_line {
            options.first_indent
        } else {
            options.rest_indent
        };
        emit_line(&mut content, content_len, indent, output);
    }
}

fn emit_line(content: &mut String, _content_len: usize, indent: usize, output: &mut Vec<String>) {
    let mut line = " ".repeat(indent);
    line.push_str(content);
    output.push(line);
    content.clear();
}
