#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find(|(key, _)| *key == name)
        .map(|(_, value)| *value)
}

fn decode_default(text: &str) -> String {
    let mut result = String::new();
    let mut chars = text.char_indices().peekable();

    while let Some((_, c)) = chars.next() {
        if c == '\\' {
            if let Some((_, next)) = chars.next() {
                result.push(next);
            }
        } else {
            result.push(c);
        }
    }

    result
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if matches!(next, '"' | '\\' | '$') {
                        word.push(next);
                        active = true;
                        i = after_next;
                    } else if next == '\n' {
                        i = after_next;
                    } else {
                        word.push('\\');
                        active = true;
                        i = next_i;
                    }
                } else if c == '$' {
                    active = true;
                    let (new_i, expansion) = parse_expansion(input, i, env)?;
                    if let Some(value) = expansion {
                        word.push_str(value);
                        i = new_i;
                    } else {
                        word.push('$');
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }

                if matches!(quote, Quote::Double(_)) && i == input.len() {
                    let _ = open_pos;
                }
            }
            Quote::None => {
                if is_separator(c) {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    let after_next = next_i + next.len_utf8();
                    if next == '\n' {
                        i = after_next;
                    } else {
                        word.push(next);
                        active = true;
                        i = after_next;
                    }
                } else if c == '$' {
                    let (new_i, expansion) = parse_expansion(input, i, env)?;
                    if let Some(value) = expansion {
                        active = true;
                        word.push_str(value);
                        i = new_i;
                    } else {
                        word.push('$');
                        active = true;
                        i = next_i;
                    }
                } else {
                    word.push(c);
                    active = true;
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

fn parse_expansion<'a>(
    input: &'a str,
    dollar: usize,
    env: &'a [(&str, &str)],
) -> Result<(usize, Option<String>), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar >= input.len() {
        return Ok((after_dollar, None));
    }

    let first = input[after_dollar..].chars().next().unwrap();
    if first == '{' {
        let content_start = after_dollar + 1;
        let mut cursor = content_start;
        let mut closing = None;

        while cursor < input.len() {
            let c = input[cursor..].chars().next().unwrap();
            let next = cursor + c.len_utf8();
            if c == '\\' {
                if next < input.len() {
                    let escaped = input[next..].chars().next().unwrap();
                    cursor = next + escaped.len_utf8();
                } else {
                    cursor = next;
                }
            } else if c == '}' {
                closing = Some(cursor);
                break;
            } else {
                cursor = next;
            }
        }

        let close_pos = closing.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let content = &input[content_start..close_pos];

        let name_end = content
            .char_indices()
            .find_map(|(offset, c)| {
                if (offset == 0 && !is_name_start(c))
                    || (offset > 0 && !is_name_continue(c))
                {
                    Some(offset)
                } else {
                    None
                }
            })
            .unwrap_or(content.len());

        let name = &content[..name_end];
        if name.is_empty() {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let remainder = &content[name_end..];
        let value = if remainder.is_empty() {
            lookup(name, env).unwrap_or("").to_string()
        } else if let Some(default_text) = remainder.strip_prefix(":-") {
            match lookup(name, env) {
                Some(value) if !value.is_empty() => value.to_string(),
                _ => decode_default(default_text),
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok((close_pos + 1, Some(value)));
    }

    if !is_name_start(first) {
        return Ok((after_dollar, None));
    }

    let mut end = after_dollar;
    while end < input.len() {
        let c = input[end..].chars().next().unwrap();
        if !is_name_continue(c) {
            break;
        }
        end += c.len_utf8();
    }

    let name = &input[after_dollar..end];
    Ok((end, Some(lookup(name, env).unwrap_or("").to_string())))
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn public_basic_words() {
        assert_eq!(
            split("alpha 'two words' x\"y\" ''", &[]),
            Ok(vec![
                "alpha".to_string(),
                "two words".to_string(),
                "xy".to_string(),
                "".to_string()
            ])
        );
    }

    #[test]
    fn public_escapes() {
        assert_eq!(
            split("a\\ b \"c\\\"d\"", &[]),
            Ok(vec!["a b".to_string(), "c\"d".to_string()])
        );
    }

    #[test]
    fn public_environment_and_default() {
        assert_eq!(
            split(
                "$USER ${M:-guest} \"$HOME\"",
                &[("USER", "sam"), ("HOME", "/a b")]
            ),
            Ok(vec![
                "sam".to_string(),
                "guest".to_string(),
                "/a b".to_string()
            ])
        );
    }

    #[test]
    fn public_expansion_error() {
        assert_eq!(
            split("${!}", &[]),
            Err(SplitError::InvalidExpansion { pos: 0 })
        );
    }
}
