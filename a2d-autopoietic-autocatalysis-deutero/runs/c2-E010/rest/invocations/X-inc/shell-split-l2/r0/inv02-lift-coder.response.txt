#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
enum Quote {
    None,
    Single { opening: usize },
    Double { opening: usize },
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single { .. } => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.push(c);
                }
                i = next_i;
            }
            Quote::Double { .. } => {
                match c {
                    '"' => {
                        quote = Quote::None;
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let after = next_i + escaped.len_utf8();
                        if matches!(escaped, '"' | '\\' | '$') {
                            word.push(escaped);
                            i = after;
                        } else if escaped == '\n' {
                            i = after;
                        } else {
                            word.push('\\');
                            i = next_i;
                        }
                    }
                    '$' => {
                        let (end, value) = parse_expansion(input, i, env)?;
                        if let Some(value) = value {
                            active = true;
                            word.push_str(value);
                            i = end;
                        } else {
                            word.push('$');
                            i = next_i;
                        }
                    }
                    _ => {
                        word.push(c);
                        i = next_i;
                    }
                }
            }
            Quote::None => {
                match c {
                    '\'' => {
                        active = true;
                        quote = Quote::Single { opening: i };
                        i = next_i;
                    }
                    '"' => {
                        active = true;
                        quote = Quote::Double { opening: i };
                        i = next_i;
                    }
                    '\\' => {
                        if next_i == input.len() {
                            return Err(SplitError::TrailingEscape { pos: i });
                        }
                        let escaped = input[next_i..].chars().next().unwrap();
                        let after = next_i + escaped.len_utf8();
                        if escaped == '\n' {
                            i = after;
                        } else {
                            active = true;
                            word.push(escaped);
                            i = after;
                        }
                    }
                    '$' => {
                        let (end, value) = parse_expansion(input, i, env)?;
                        if let Some(value) = value {
                            active = true;
                            word.push_str(value);
                            i = end;
                        } else {
                            active = true;
                            word.push('$');
                            i = next_i;
                        }
                    }
                    c if is_separator(c) => {
                        if active {
                            words.push(std::mem::take(&mut word));
                            active = false;
                        }
                        i = next_i;
                    }
                    _ => {
                        active = true;
                        word.push(c);
                        i = next_i;
                    }
                }
            }
        }
    }

    match quote {
        Quote::Single { opening } => {
            return Err(SplitError::UnterminatedSingle { pos: opening });
        }
        Quote::Double { opening } => {
            return Err(SplitError::UnterminatedDouble { pos: opening });
        }
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}

// Returns (end byte offset, Some(value)) for an expansion, or (dollar+1, None)
// when the dollar is literal.
fn parse_expansion<'a>(
    input: &str,
    dollar: usize,
    env: &'a [(&str, &str)],
) -> Result<(usize, Option<String>), SplitError> {
    let after_dollar = dollar + 1;
    if after_dollar == input.len() {
        return Ok((after_dollar, None));
    }
    let c = input[after_dollar..].chars().next().unwrap();
    if c == '{' {
        let content_start = after_dollar + 1;
        let mut j = content_start;
        let mut close = None;
        while j < input.len() {
            let ch = input[j..].chars().next().unwrap();
            let end = j + ch.len_utf8();
            if ch == '\\' {
                if end < input.len() {
                    let skipped = input[end..].chars().next().unwrap();
                    j = end + skipped.len_utf8();
                } else {
                    j = end;
                }
            } else if ch == '}' {
                close = Some(j);
                break;
            } else {
                j = end;
            }
        }
        let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
        let contents = &input[content_start..close];

        let name_end = contents
            .char_indices()
            .find_map(|(offset, ch)| {
                if !is_name_continue(ch) {
                    Some(offset)
                } else {
                    None
                }
            })
            .unwrap_or(contents.len());

        let name = &contents[..name_end];
        if name.is_empty() || !name.chars().next().map(is_name_start).unwrap_or(false) {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        }

        let rest = &contents[name_end..];
        let value = if rest.is_empty() {
            lookup(name, env).unwrap_or("").to_string()
        } else if let Some(default) = rest.strip_prefix(":-") {
            let found = lookup(name, env).unwrap_or("");
            if !found.is_empty() {
                found.to_string()
            } else {
                decode_default(default)
            }
        } else {
            return Err(SplitError::InvalidExpansion { pos: dollar });
        };

        return Ok((close + 1, Some(value)));
    }

    if !is_name_start(c) {
        return Ok((after_dollar, None));
    }

    let mut end = after_dollar + c.len_utf8();
    while end < input.len() {
        let ch = input[end..].chars().next().unwrap();
        if !is_name_continue(ch) {
            break;
        }
        end += ch.len_utf8();
    }
    let name = &input[after_dollar..end];
    Ok((end, Some(lookup(name, env).unwrap_or("").to_string())))
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut i = 0;
    while i < text.len() {
        let c = text[i..].chars().next().unwrap();
        let next = i + c.len_utf8();
        if c == '\\' && next < text.len() {
            let escaped = text[next..].chars().next().unwrap();
            out.push(escaped);
            i = next + escaped.len_utf8();
        } else {
            out.push(c);
            i = next;
        }
    }
    out
}
