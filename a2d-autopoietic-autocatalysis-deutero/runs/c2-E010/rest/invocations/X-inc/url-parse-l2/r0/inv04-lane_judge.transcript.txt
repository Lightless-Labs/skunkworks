    InvalidScheme { at: usize },

    InvalidUserInfo { at: usize },

    InvalidUtf8 { at: usize },

    UserInfo,
    Path,
    Query,
    Fragment,

#[derive(Debug, Clone, PartialEq, Eq)]

    BadEscape { at: usize },

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let source = input.as_bytes();
    let mut decoded = Vec::with_capacity(source.len());
    let mut origins = Vec::with_capacity(source.len());
    let mut i = 0;

    while i < source.len() {
        if source[i] == b'%' {
            if i + 2 >= source.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let high = hex_value(source[i + 1]);
            let low = hex_value(source[i + 2]);
            match (high, low) {
                (Some(high), Some(low)) => {
                    decoded.push((high << 4) | low);
                    origins.push(i);
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
            i += 3;
        } else {
            decoded.push(source[i]);
            origins.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_offset = error.valid_up_to();
            let original_offset = origins.get(decoded_offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_offset })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut encoded = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = matches!(
            byte,
            b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-' | b'.' | b'_' | b'~'
        ) || match component {
            Component::UserInfo => byte == b':',
            Component::Path => byte == b'/',
            Component::Query => matches!(byte, b'/' | b'?'),
            Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
        };

        if safe {
            encoded.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            encoded.push('%');
            encoded.push(HEX[(byte >> 4) as usize] as char);
            encoded.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    encoded
}

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    for (offset, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + offset,
            });
        }
    }

    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|current| current.checked_add((byte - b'0') as u32))
        {
            Some(value) => value,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }

    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

fn map_codec_error(error: CodecError, base: usize) -> ParseError {
    match error {
        CodecError::BadEscape { at } => ParseError::BadEscape { at: base + at },
        CodecError::InvalidUtf8 { at } => ParseError::InvalidUtf8 { at: base + at },
    }
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(colon) => colon,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let mut at_sign = None;
    for (offset, byte) in authority.bytes().enumerate() {
        if byte == b'@' {
            if at_sign.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: authority_start + offset,
                });
            }
            at_sign = Some(offset);
        }
    }

    let (userinfo, host_start, host_port) = match at_sign {
        Some(offset) => {
            if offset == 0 {
                return Err(ParseError::InvalidUserInfo {
                    at: authority_start + offset,
                });
            }
            (
                Some(authority[..offset].to_owned()),
                authority_start + offset + 1,
                &authority[offset + 1..],
            )
        }
        None => (None, authority_start, authority),
    };

    let mut ipv6 = false;
    let (host, port) = if host_port.starts_with('[') {
        let opening = host_start;
        let close = match host_port[1..].find(']') {
            Some(offset) => offset + 1,
            None => return Err(ParseError::InvalidIpv6 { at: opening }),
        };
        let interior = &host_port[1..close];
        if Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: opening });
        }

        let suffix = &host_port[close + 1..];
        let parsed_port = if suffix.is_empty() {
            None
        } else if let Some(port_text) = suffix.strip_prefix(':') {
            Some(parse_port(port_text, host_start + close + 2)?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_start + close + 1,
            });
        };

        ipv6 = true;
        (interior.to_ascii_lowercase(), parsed_port)
    } else {
        let host_port_bytes = host_port.as_bytes();
        let split = host_port_bytes.iter().position(|&byte| byte == b':');
        let (host_text, parsed_port) = match split {
            Some(offset) => (
                &host_port[..offset],
                Some(parse_port(
                    &host_port[offset + 1..],
                    host_start + offset + 1,
                )?),
            ),
            None => (host_port, None),
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), parsed_port)
    };

    let fragment_marker = bytes[authority_end..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|offset| authority_end + offset);
    let component_end = fragment_marker.unwrap_or(bytes.len());

    let query_marker = bytes[authority_end..component_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|offset| authority_end + offset);

    let path_end = query_marker.unwrap_or(component_end);
    let path = if path_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..path_end].to_owned()
    };

    let query = query_marker.map(|marker| input[marker + 1..component_end].to_owned());
    let fragment = fragment_marker.map(|marker| input[marker + 1..].to_owned());

    if let Some(userinfo_text) = userinfo.as_deref() {
        let userinfo_offset = authority_start;
        decode_component(Component::UserInfo, userinfo_text)
            .map_err(|error| map_codec_error(error, userinfo_offset))?;
    }
    if path_end != authority_end {
        decode_component(Component::Path, &path)
            .map_err(|error| map_codec_error(error, authority_end))?;
    }
    if let (Some(query_text), Some(marker)) = (query.as_deref(), query_marker) {
        decode_component(Component::Query, query_text)
            .map_err(|error| map_codec_error(error, marker + 1))?;
    }
    if let (Some(fragment_text), Some(marker)) = (fragment.as_deref(), fragment_marker) {
        decode_component(Component::Fragment, fragment_text)
            .map_err(|error| map_codec_error(error, marker + 1))?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    let bytes = input.as_bytes();
    for (at, byte) in bytes.iter().copied().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|byte| *byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for at in 1..colon {
        let byte = bytes[at];
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if bytes.get(colon + 1) != Some(&b'/') || bytes.get(colon + 2) != Some(&b'/') {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|byte| matches!(*byte, b'/' | b'?' | b'#'))
        .map(|offset| authority_start + offset)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let mut at_sign = None;
    for (offset, byte) in authority.bytes().enumerate() {
        if byte == b'@' {
            if at_sign.is_some() {
                return Err(ParseError::InvalidUserInfo {
                    at: authority_start + offset,
                });
            }
            at_sign = Some(offset);
        }
    }

    let (userinfo, host_port, host_port_start) = if let Some(offset) = at_sign {
        if offset == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start,
            });
        }
        (
            Some(authority[..offset].to_owned()),
            &authority[offset + 1..],
            authority_start + offset + 1,
        )
    } else {
        (None, authority, authority_start)
    };

    let mut ipv6 = false;
    let (host, port) = if host_port.starts_with('[') {
        let open_at = host_port_start;
        let close = match host_port.as_bytes()[1..].iter().position(|b| *b == b']') {
            Some(offset) => offset + 1,
            None => return Err(ParseError::InvalidIpv6 { at: open_at }),
        };
        let interior = &host_port[1..close];
        if interior.contains('%') || Ipv6Addr::from_str(interior).is_err() {
            return Err(ParseError::InvalidIpv6 { at: open_at });
        }

        let suffix = &host_port[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                host_port_start + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: host_port_start + close + 1,
            });
        };
        ipv6 = true;
        (interior.to_ascii_lowercase(), port)
    } else {
        let split = host_port.find(':');
        let (host_text, port) = if let Some(offset) = split {
            (
                &host_port[..offset],
                Some(parse_port(
                    &host_port[offset + 1..],
                    host_port_start + offset + 1,
                )?),
            )
        } else {
            (host_port, None)
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost { at: host_port_start });
        }
        for (offset, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_port_start + offset,
                });
            }
        }
        (host_text.to_ascii_lowercase(), port)
    };

    let after_authority = &input[authority_end..];
    let fragment_relative = after_authority.find('#');
    let before_fragment = fragment_relative
        .map(|offset| &after_authority[..offset])
        .unwrap_or(after_authority);
    let fragment = fragment_relative.map(|offset| after_authority[offset + 1..].to_owned());

    let query_relative = before_fragment.find('?');
    let path_text = query_relative
        .map(|offset| &before_fragment[..offset])
        .unwrap_or(before_fragment);
    let query = query_relative.map(|offset| before_fragment[offset + 1..].to_owned());

    let path = if path_text.is_empty() {
        "/".to_owned()
    } else {
        path_text.to_owned()
    };

    let mut components: Vec<(Component, &str, usize)> = Vec::new();
    if let Some(userinfo) = userinfo.as_deref() {
        components.push((Component::UserInfo, userinfo, authority_start));
    }
    if !path_text.is_empty() {
        let path_start = authority_end;
        components.push((Component::Path, path_text, path_start));
    }
    if let Some(offset) = query_relative {
        let query_start = authority_end + offset + 1;
        components.push((
            Component::Query,
            query.as_deref().unwrap_or(""),
            query_start,
        ));
    }
    if let Some(offset) = fragment_relative {
        let fragment_start = authority_end + offset + 1;
        components.push((
            Component::Fragment,
            fragment.as_deref().unwrap_or(""),
            fragment_start,
        ));
    }

    for (component, text, component_start) in components {
        match decode_component(component, text) {
            Ok(_) => {}
            Err(CodecError::BadEscape { at }) => {
                return Err(ParseError::BadEscape {
                    at: component_start + at,
                });
            }
            Err(CodecError::InvalidUtf8 { at }) => {
                return Err(ParseError::InvalidUtf8 {
                    at: component_start + at,
                });
            }
        }
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host,
        port,
        path,
        query,
        fragment,
        ipv6,
    })
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

pub fn decode_component(_component: Component, input: &str) -> Result<String, CodecError> {
    let bytes = input.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut offsets = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        if bytes[i] == b'%' {
            if i + 2 >= bytes.len() {
                return Err(CodecError::BadEscape { at: i });
            }
            let hi = hex_value(bytes[i + 1]);
            let lo = hex_value(bytes[i + 2]);
            match (hi, lo) {
                (Some(hi), Some(lo)) => {
                    decoded.push((hi << 4) | lo);
                    offsets.push(i);
                    i += 3;
                }
                _ => return Err(CodecError::BadEscape { at: i }),
            }
        } else {
            decoded.push(bytes[i]);
            offsets.push(i);
            i += 1;
        }
    }

    match std::str::from_utf8(&decoded) {
        Ok(text) => Ok(text.to_owned()),
        Err(error) => {
            let decoded_offset = error.valid_up_to();
            let original_offset = offsets.get(decoded_offset).copied().unwrap_or(input.len());
            Err(CodecError::InvalidUtf8 { at: original_offset })
        }
    }
}

pub fn encode_component(component: Component, text: &str) -> String {
    let mut output = String::with_capacity(text.len());
    for &byte in text.as_bytes() {
        let safe = byte.is_ascii_alphanumeric()
            || matches!(byte, b'-' | b'.' | b'_' | b'~')
            || match component {
                Component::UserInfo => byte == b':',
                Component::Path => byte == b'/',
                Component::Query => matches!(byte, b'/' | b'?'),
                Component::Fragment => matches!(byte, b'/' | b'?' | b'#'),
            };

        if safe {
            output.push(byte as char);
        } else {
            const HEX: &[u8; 16] = b"0123456789ABCDEF";
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
    }
    output
}

        Err(CodecError::InvalidUtf8 { at }) => Err(ParseError::InvalidUtf8 {
            at: base_offset + at,

fn parse_port(port: &str, port_start: usize) -> Result<u16, ParseError> {
    if port.is_empty() {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    for (i, byte) in port.bytes().enumerate() {
        if !byte.is_ascii_digit() {
            return Err(ParseError::InvalidPort {
                at: port_start + i,
            });
        }
    }

    if port.len() >= 2 && port.as_bytes()[0] == b'0' {
        return Err(ParseError::InvalidPort { at: port_start });
    }

    let mut value: u32 = 0;
    for byte in port.bytes() {
        value = match value
            .checked_mul(10)
            .and_then(|n| n.checked_add((byte - b'0') as u32))
        {
            Some(n) => n,
            None => return Err(ParseError::InvalidPort { at: port_start }),
        };
    }

    if !(1..=65535).contains(&value) {
        return Err(ParseError::InvalidPort { at: port_start });
    }
    Ok(value as u16)
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    if input.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, byte) in input.bytes().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let bytes = input.as_bytes();
    let colon = match bytes.iter().position(|&b| b == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: input.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }
    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    if colon + 2 >= bytes.len()
        || bytes[colon + 1] != b'/'
        || bytes[colon + 2] != b'/'
    {
        return Err(ParseError::MissingAuthority { at: colon + 1 });
    }

    let authority_start = colon + 3;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&b| matches!(b, b'/' | b'?' | b'#'))
        .map(|relative| authority_start + relative)
        .unwrap_or(bytes.len());
    let authority = &input[authority_start..authority_end];

    let at_positions: Vec<usize> = authority
        .bytes()
        .enumerate()
        .filter_map(|(i, b)| (b == b'@').then_some(i))
        .collect();

    if at_positions.len() >= 2 {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at_positions[1],
        });
    }

    let (userinfo, hostport_start) = if let Some(&at) = at_positions.first() {
        if at == 0 {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        (Some(&authority[..at]), at + 1)
    } else {
        (None, 0)
    };

    let hostport = &authority[hostport_start..];
    let hostport_absolute = authority_start + hostport_start;

    let (host, ipv6, port) = if hostport.starts_with('[') {
        let close = match hostport.as_bytes()[1..]
            .iter()
            .position(|&b| b == b']')
        {
            Some(relative) => relative + 1,
            None => {
                return Err(ParseError::InvalidIpv6 {
                    at: hostport_absolute,
                });
            }
        };

        let interior = &hostport[1..close];
        if interior.parse::<std::net::Ipv6Addr>().is_err() {
            return Err(ParseError::InvalidIpv6 {
                at: hostport_absolute,
            });
        }

        let suffix = &hostport[close + 1..];
        let port = if suffix.is_empty() {
            None
        } else if suffix.starts_with(':') {
            Some(parse_port(
                &suffix[1..],
                hostport_absolute + close + 2,
            )?)
        } else {
            return Err(ParseError::InvalidHost {
                at: hostport_absolute + close + 1,
            });
        };

        (interior.to_ascii_lowercase(), true, port)
    } else {
        let split = hostport.find(':');
        let (host_text, port) = match split {
            Some(relative_colon) => {
                let port_start = hostport_absolute + relative_colon + 1;
                (
                    &hostport[..relative_colon],
                    Some(parse_port(&hostport[relative_colon + 1..], port_start)?),
                )
            }
            None => (hostport, None),
        };

        if host_text.is_empty() {
            return Err(ParseError::InvalidHost {
                at: hostport_absolute,
            });
        }
        for (relative, byte) in host_text.bytes().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: hostport_absolute + relative,
                });
            }
        }

        (host_text.to_ascii_lowercase(), false, port)
    };

    let tail = &input[authority_end..];
    let fragment_position = tail.find('#');
    let before_fragment = fragment_position.map(|p| &tail[..p]).unwrap_or(tail);
    let fragment = fragment_position.map(|p| &tail[p + 1..]);

    let query_position = before_fragment.find('?');
    let path_text = query_position
        .map(|p| &before_fragment[..p])
        .unwrap_or(before_fragment);
    let query = query_position.map(|p| &before_fragment[p + 1..]);

    let path = if path_text.is_empty() {
        "/".to_owned()
    } else {
        path_text.to_owned()
    };

    if let Some(userinfo_text) = userinfo {
        validate_component(
            Component::UserInfo,
            userinfo_text,
            authority_start,
        )?;
    }
    if !path_text.is_empty() {
        validate_component(
            Component::Path,
            path_text,
            authority_end,
        )?;
    }
    if let Some(query_text) = query {
        let query_offset = authority_end + query_position.unwrap() + 1;
        validate_component(Component::Query, query_text, query_offset)?;
    }
    if let Some(fragment_text) = fragment {
        let fragment_offset = authority_end + fragment_position.unwrap() + 1;
        validate_component(Component::Fragment, fragment_text, fragment_offset)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo: userinfo.map(str::to_owned),
        host,
        port,
        path,
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    })
}

            return Err(ParseError::InvalidIpv6 { at: host_start });
        }

        let suffix_start = close_rel + 1;
        let (port, suffix_valid) = if suffix_start == host_port.len() {
            (None, true)
        } else if host_port_bytes[suffix_start] == b':' {
            let port_start = host_start + suffix_start + 1;
            (Some(parse_port(&host_port[suffix_start + 1..], port_start)?), true)
        } else {
            (None, false)
        };

        if !suffix_valid {
            return Err(ParseError::InvalidHost {
                at: host_start + suffix_start,
            });
        }

        (interior, port, true)
    } else {
        let host_colon = host_port_bytes.iter().position(|&byte| byte == b':');
        let (host, port) = if let Some(colon_rel) = host_colon {
            let port_start = host_start + colon_rel + 1;
            (
                &host_port[..colon_rel],
                Some(parse_port(&host_port[colon_rel + 1..], port_start)?),
            )
        } else {
            (host_port, None)
        };

        if host.is_empty() {
            return Err(ParseError::InvalidHost { at: host_start });
        }
        for (offset, &byte) in host.as_bytes().iter().enumerate() {
            if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
                return Err(ParseError::InvalidHost {
                    at: host_start + offset,
                });
            }
        }
        (host, port, false)
    };

    let (fragment, before_fragment_end) = if let Some(hash_rel) = input[authority_end..]
        .bytes()
        .position(|byte| byte == b'#')
    {
        let hash = authority_end + hash_rel;
        (Some(&input[hash + 1..]), hash)
    } else {
        (None, input.len())
    };

    let (query, path_end) = if let Some(query_rel) = input[authority_end..before_fragment_end]
        .bytes()
        .position(|byte| byte == b'?')
    {
        let question = authority_end + query_rel;
        (Some(&input[question + 1..before_fragment_end]), question)
    } else {
        (None, before_fragment_end)
    };

    let path = &input[authority_end..path_end];
    let path = if path.is_empty() { "/" } else { path };

    if let Some(userinfo) = userinfo {
        validate_component(
            Component::UserInfo,
            userinfo,
            authority_start,
        )?;
    }
    if !path.is_empty() {
        validate_component(Component::Path, path, authority_end)?;
    }
    if let Some(query) = query {
        let query_start = path_end + 1;
        validate_component(Component::Query, query, query_start)?;
    }
    if let Some(fragment) = fragment {
        let fragment_start = before_fragment_end + 1;
        validate_component(Component::Fragment, fragment, fragment_start)?;
    }

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo: userinfo.map(str::to_owned),
        host: host.to_ascii_lowercase(),
        port,
        path: path.to_owned(),
        query: query.map(str::to_owned),
        fragment: fragment.map(str::to_owned),
        ipv6,
    })
}
