#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EndKind {
    Complete,
    Marker,
    Raw,
}

#[derive(Clone, Debug)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone, Debug)]
struct Candidate {
    cost: u64,
    marker_lines: usize,
    line_count: usize,
    consumed: Vec<usize>,
    rendered: Vec<String>,
    lines: Vec<Line>,
}

#[derive(Clone, Debug)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    consumed: usize,
    end_kind: EndKind,
}

impl Candidate {
    fn better_than(&self, other: &Self) -> bool {
        self.cost < other.cost
            || (self.cost == other.cost
                && (self.marker_lines < other.marker_lines
                    || (self.marker_lines == other.marker_lines
                        && (self.line_count < other.line_count
                            || (self.line_count == other.line_count
                                && (self.consumed > other.consumed
                                    || (self.consumed == other.consumed
                                        && self.rendered < other.rendered)))))))
    }
}

fn add_cost(a: u64, b: u64) -> u64 {
    a + b
}

fn state_index(word: usize, offset: usize, max_chars: usize) -> usize {
    word * (max_chars + 1) + offset
}

fn wrap_paragraph(words: &[Word], options: &WrapOptions) -> Vec<Line> {
    if words.is_empty() {
        return Vec::new();
    }

    let max_chars: usize = words.iter().map(|w| w.chars.len()).sum();
    let state_count = (words.len() + 1) * (max_chars + 1);
    let mut memo: Vec<Option<Candidate>> = vec![None; state_count * 2];

    fn solve(
        word_idx: usize,
        offset: usize,
        first: bool,
        words: &[Word],
        options: &WrapOptions,
        max_chars: usize,
        memo: &mut [Option<Candidate>],
    ) -> Candidate {
        if word_idx >= words.len() {
            return Candidate {
                cost: 0,
                marker_lines: 0,
                line_count: 0,
                consumed: Vec::new(),
                rendered: Vec::new(),
                lines: Vec::new(),
            };
        }

        let key = state_index(word_idx, offset, max_chars) * 2 + usize::from(!first);
        if let Some(c) = &memo[key] {
            return c.clone();
        }

        let indent = if first { options.first_indent } else { options.rest_indent };
        let capacity = options.width - indent;
        let word = &words[word_idx];
        let remaining = word.chars.len() - offset;

        let mut best: Option<Candidate> = None;

        let mut consider = |piece_text: String,
                            next_word: usize,
                            next_offset: usize,
                            kind: EndKind,
                            consumed_now: usize,
                            piece_count: usize| {
            let mut pieces = vec![piece_text];
            let mut consumed = consumed_now;
            let mut end_word = next_word;
            let mut end_offset = next_offset;

            let mut used = pieces[0].chars().count();
            while end_word < words.len() {
                let w = &words[end_word];
                let start = if end_offset == 0 { 0 } else { end_offset };
                let rem = w.chars.len() - start;
                let add_space = usize::from(!pieces.is_empty());
                if used + add_space + rem > capacity {
                    break;
                }
                pieces.push(w.chars[start..].iter().collect());
                used += add_space + rem;
                consumed += rem;
                end_word += 1;
                end_offset = 0;
            }

            let line_width = used + indent;
            let gap = options.width - line_width;
            let is_final = end_word >= words.len();
            let mut line_cost = (gap * gap) as u64;
            if is_final {
                line_cost /= 4;
            }
            if kind == EndKind::Marker {
                line_cost = add_cost(line_cost, options.hyphen_cost as u64);
            } else if kind == EndKind::Raw {
                line_cost = add_cost(line_cost, (2 * options.width + 1) as u64);
            }
            if !is_final && piece_count == 1 {
                line_cost = add_cost(line_cost, 3);
            }

            let tail = if end_word >= words.len() {
                Candidate {
                    cost: 0,
                    marker_lines: 0,
                    line_count: 0,
                    consumed: Vec::new(),
                    rendered: Vec::new(),
                    lines: Vec::new(),
                }
            } else {
                solve(
                    end_word,
                    end_offset,
                    false,
                    words,
                    options,
                    max_chars,
                    memo,
                )
            };

            let mut line = Line {
                indent,
                pieces: pieces.clone(),
                consumed,
                end_kind: kind,
            };
            let mut pre = " ".repeat(indent);
            for (i, p) in pieces.iter().enumerate() {
                if i > 0 {
                    pre.push(' ');
                }
                pre.push_str(p);
            }

            let mut candidate = Candidate {
                cost: line_cost + tail.cost,
                marker_lines: usize::from(kind == EndKind::Marker) + tail.marker_lines,
                line_count: 1 + tail.line_count,
                consumed: vec![consumed],
                rendered: vec![pre],
                lines: vec![line],
            };
            candidate.consumed.extend(tail.consumed);
            candidate.rendered.extend(tail.rendered);
            candidate.lines.extend(tail.lines);

            if best.as_ref().map_or(true, |b| candidate.better_than(b)) {
                best = Some(candidate);
            }
        };

        // Enumerate line endings after a completed word or at a marker in the
        // starting word. A line may also contain subsequent complete words.
        for end in offset + 1..=word.chars.len() {
            let prefix_len = end - offset;
            if prefix_len <= capacity {
                let complete = end == word.chars.len();
                if complete {
                    consider(
                        word.chars[offset..end].iter().collect(),
                        word_idx + 1,
                        0,
                        EndKind::Complete,
                        prefix_len,
                        1,
                    );
                }
                for &marker in &word.markers {
                    if marker == end && marker < word.chars.len() && prefix_len + 1 <= capacity {
                        let mut s: String = word.chars[offset..end].iter().collect();
                        s.push('-');
                        consider(
                            s,
                            word_idx,
                            end,
                            EndKind::Marker,
                            prefix_len,
                            1,
                        );
                    }
                }
            }
        }

        // Add complete following words to lines ending after a complete word.
        // This is represented by considering each possible complete-word run.
        if offset == 0 {
            for end_word in word_idx + 1..words.len() {
                let mut text = String::new();
                let mut used = 0usize;
                let mut consumed = 0usize;
                let mut possible = true;
                for wi in word_idx..=end_word {
                    if wi > word_idx {
                        used += 1;
                        text.push(' ');
                    }
                    let w = &words[wi];
                    used += w.chars.len();
                    consumed += w.chars.len();
                    text.extend(w.chars.iter());
                    if used > capacity {
                        possible = false;
                        break;
                    }
                }
                if possible {
                    let next = end_word + 1;
                    let tail = solve(next, 0, false, words, options, max_chars, memo);
                    let gap = options.width - indent - used;
                    let final_line = next == words.len();
                    let mut cost = (gap * gap) as u64;
                    if final_line {
                        cost /= 4;
                    }
                    let piece_count = end_word - word_idx + 1;
                    if !final_line && piece_count == 1 {
                        cost += 3;
                    }
                    let mut pre = " ".repeat(indent);
                    pre.push_str(&text);
                    let mut c = Candidate {
                        cost: cost + tail.cost,
                        marker_lines: tail.marker_lines,
                        line_count: 1 + tail.line_count,
                        consumed: vec![consumed],
                        rendered: vec![pre],
                        lines: vec![Line {
                            indent,
                            pieces: text.split(' ').map(str::to_string).collect(),
                            consumed,
                            end_kind: EndKind::Complete,
                        }],
                    };
                    c.consumed.extend(tail.consumed);
                    c.rendered.extend(tail.rendered);
                    c.lines.extend(tail.lines);
                    if best.as_ref().map_or(true, |b| c.better_than(b)) {
                        best = Some(c);
                    }
                }
            }
        }

        // Raw splits are allowed only on an empty line and only when no
        // marker prefix can fit with its hyphen.
        let marker_fits = word.markers.iter().any(|&m| {
            m > offset && m < word.chars.len() && m - offset + 1 <= capacity
        });
        if remaining > capacity && !marker_fits {
            let text: String = word.chars[offset..offset + capacity].iter().collect();
            let tail = solve(
                word_idx,
                offset + capacity,
                false,
                words,
                options,
                max_chars,
                memo,
            );
            let gap = 0usize;
            let mut c = Candidate {
                cost: (gap * gap) as u64 + (2 * options.width + 1) as u64 + tail.cost,
                marker_lines: tail.marker_lines,
                line_count: 1 + tail.line_count,
                consumed: vec![capacity],
                rendered: vec![format!("{}{}", " ".repeat(indent), text)],
                lines: vec![Line {
                    indent,
                    pieces: vec![text],
                    consumed: capacity,
                    end_kind: EndKind::Raw,
                }],
            };
            c.consumed.extend(tail.consumed);
            c.rendered.extend(tail.rendered);
            c.lines.extend(tail.lines);
            if best.as_ref().map_or(true, |b| c.better_than(b)) {
                best = Some(c);
            }
        }

        let result = best.expect("raw split guarantees a legal layout");
        memo[key] = Some(result.clone());
        result
    }

    solve(0, 0, true, words, options, max_chars, &mut memo).lines
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge { width: options.width });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars: Vec<(usize, char)> = input.char_indices().collect();
    if chars.len() > 512 {
        return Err(WrapError::InputTooLong { chars: chars.len() });
    }

    for &(idx, ch) in &chars {
        if ch.is_control() && ch != '\t' && ch != '\r' && ch != '\n' {
            return Err(WrapError::InvalidControl { index: idx });
        }
    }

    let is_sep = |c: char| matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00A0}');
    for i in 0..chars.len() {
        if chars[i].1 == '~' {
            let left_ok = i > 0 && chars[i - 1].1 != '~' && !is_sep(chars[i - 1].1);
            let right_ok = i + 1 < chars.len()
                && chars[i + 1].1 != '~'
                && !is_sep(chars[i + 1].1);
            if !left_ok || !right_ok {
                return Err(WrapError::InvalidMarker { index: chars[i].0 });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut i = 0;
    while i < chars.len() {
        let ch = chars[i].1;
        if ch == '\r' {
            normalized.push('\n');
            if i + 1 < chars.len() && chars[i + 1].1 == '\n' {
                i += 1;
            }
        } else {
            normalized.push(ch);
        }
        i += 1;
    }

    let mut output = Vec::new();
    let mut segment = Vec::new();
    let mut j = 0;
    while j < normalized.len() {
        if normalized[j] == '\n' {
            let start = j;
            while j < normalized.len() && normalized[j] == '\n' {
                j += 1;
            }
            let run = j - start;
            if run >= 2 {
                append_segment(&segment, options, &mut output);
                segment.clear();
                output.extend(std::iter::repeat(String::new()).take(run - 1));
            } else {
                segment.push('\n');
            }
        } else {
            segment.push(normalized[j]);
            j += 1;
        }
    }
    append_segment(&segment, options, &mut output);
    Ok(output)
}

fn append_segment(segment: &[char], options: &WrapOptions, output: &mut Vec<String>) {
    let is_sep = |c: char| matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00A0}');
    let mut words = Vec::new();
    let mut chars = Vec::new();
    let mut markers = Vec::new();

    for &ch in segment {
        if is_sep(ch) {
            if !chars.is_empty() {
                words.push(Word {
                    chars: std::mem::take(&mut chars),
                    markers: std::mem::take(&mut markers),
                });
            }
        } else if ch == '~' {
            markers.push(chars.len());
        } else {
            chars.push(ch);
        }
    }
    if !chars.is_empty() {
        words.push(Word { chars, markers });
    }

    let mut lines = wrap_paragraph(&words, options);
    let last = lines.len().saturating_sub(1);
    for (line_index, line) in lines.iter_mut().enumerate() {
        let pieces = &line.pieces;
        let piece_count = pieces.len();
        let mut rendered = " ".repeat(line.indent);
        for (i, piece) in pieces.iter().enumerate() {
            if i > 0 {
                let extra = if options.justify && line_index != last && piece_count >= 2 {
                    let displayed: usize = pieces.iter().map(|p| p.chars().count()).sum()
                        + piece_count - 1
                        + line.indent;
                    let e = options.width - displayed;
                    let gaps = piece_count - 1;
                    let base = e / gaps;
                    let rem = e % gaps;
                    1 + base + usize::from((i - 1 + gaps - (line_index % gaps)) % gaps < rem)
                } else {
                    1
                };
                rendered.push_str(&" ".repeat(extra));
            }
            rendered.push_str(piece);
        }
        output.push(rendered);
    }
}
