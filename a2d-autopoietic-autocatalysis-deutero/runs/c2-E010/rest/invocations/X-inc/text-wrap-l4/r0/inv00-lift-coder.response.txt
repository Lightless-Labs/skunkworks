#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WrapError {
    WidthZero,
    WidthTooLarge { width: usize },
    FirstIndentTooWide { indent: usize },
    RestIndentTooWide { indent: usize },
    HyphenCostTooLarge { cost: usize },
    InputTooLong { chars: usize },
    InvalidControl { index: usize },
    InvalidMarker { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct WrapOptions {
    pub width: usize,
    pub first_indent: usize,
    pub rest_indent: usize,
    pub hyphen_cost: usize,
    pub justify: bool,
}

#[derive(Clone)]
struct Word {
    chars: Vec<char>,
    markers: Vec<usize>,
}

#[derive(Clone)]
struct Line {
    indent: usize,
    pieces: Vec<String>,
    consumed: usize,
    marker_ended: bool,
    raw_split: bool,
    piece_count: usize,
    prejustify: String,
    gap: usize,
}

#[derive(Clone)]
struct Solution {
    cost: u64,
    marker_count: usize,
    lines: Vec<Line>,
    consumed_vector: Vec<usize>,
    string_vector: Vec<String>,
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\n' | '\r' | '\u{00a0}')
}

fn better(a: &Solution, b: &Solution) -> bool {
    if a.cost != b.cost {
        return a.cost < b.cost;
    }
    if a.marker_count != b.marker_count {
        return a.marker_count < b.marker_count;
    }
    if a.lines.len() != b.lines.len() {
        return a.lines.len() < b.lines.len();
    }
    if a.consumed_vector != b.consumed_vector {
        return a.consumed_vector > b.consumed_vector;
    }
    a.string_vector < b.string_vector
}

fn render_prejustify(indent: usize, pieces: &[String]) -> String {
    let mut s = " ".repeat(indent);
    for (i, piece) in pieces.iter().enumerate() {
        if i > 0 {
            s.push(' ');
        }
        s.push_str(piece);
    }
    s
}

fn solve_paragraph(words: &[Word], opts: &WrapOptions) -> Vec<Line> {
    use std::collections::HashMap;

    type State = (usize, usize, bool);
    let mut memo: HashMap<State, Option<Solution>> = HashMap::new();

    fn rec(
        state: State,
        words: &[Word],
        opts: &WrapOptions,
        memo: &mut HashMap<State, Option<Solution>>,
    ) -> Option<Solution> {
        if state.0 == words.len() {
            return Some(Solution {
                cost: 0,
                marker_count: 0,
                lines: Vec::new(),
                consumed_vector: Vec::new(),
                string_vector: Vec::new(),
            });
        }
        if let Some(cached) = memo.get(&state) {
            return cached.clone();
        }

        let (word_index, offset, first_line) = state;
        let indent = if first_line {
            opts.first_indent
        } else {
            opts.rest_indent
        };
        let capacity = opts.width - indent;
        let mut best: Option<Solution> = None;

        let mut consider = |end_word: usize,
                            end_offset: usize,
                            pieces: Vec<String>,
                            consumed: usize,
                            marker_ended: bool,
                            raw_split: bool,
                            next_state: State| {
            let Some(tail) = rec(next_state, words, opts, memo) else {
                return;
            };
            let piece_count = pieces.len();
            let prejustify = render_prejustify(indent, &pieces);
            let display_width = prejustify.chars().count();
            let gap = opts.width - display_width;
            let nonfinal = !tail.lines.is_empty();
            let mut line_cost = (gap * gap) as u64;
            if !nonfinal {
                line_cost /= 4;
            }
            if marker_ended {
                line_cost += opts.hyphen_cost as u64;
            }
            if raw_split {
                line_cost += (2 * opts.width + 1) as u64;
            }
            if nonfinal && piece_count == 1 {
                line_cost += 3;
            }

            let line = Line {
                indent,
                pieces,
                consumed,
                marker_ended,
                raw_split,
                piece_count,
                prejustify: prejustify.clone(),
                gap,
            };
            let mut candidate = Solution {
                cost: line_cost + tail.cost,
                marker_count: usize::from(marker_ended) + tail.marker_count,
                lines: Vec::with_capacity(1 + tail.lines.len()),
                consumed_vector: Vec::with_capacity(1 + tail.consumed_vector.len()),
                string_vector: Vec::with_capacity(1 + tail.string_vector.len()),
            };
            candidate.lines.push(line);
            candidate.lines.extend(tail.lines);
            candidate.consumed_vector.push(consumed);
            candidate.consumed_vector.extend(tail.consumed_vector);
            candidate.string_vector.push(prejustify);
            candidate.string_vector.extend(tail.string_vector);

            if best.as_ref().is_none_or(|current| better(&candidate, current)) {
                best = Some(candidate);
            }

            let _ = (end_word, end_offset);
        };

        let mut pieces = Vec::new();
        let mut used = 0usize;
        let mut wi = word_index;
        let mut start = offset;
        let mut consumed_total = 0usize;

        loop {
            if wi >= words.len() {
                break;
            }
            let word = &words[wi];
            let remaining = word.chars.len() - start;
            let separator_width = usize::from(!pieces.is_empty());
            if used + separator_width >= capacity {
                break;
            }
            let room = capacity - used - separator_width;

            if remaining <= room {
                if !pieces.is_empty() {
                    used += 1;
                }
                let text: String = word.chars[start..].iter().collect();
                used += remaining;
                consumed_total += remaining;
                pieces.push(text);

                let next = (wi + 1, 0, false);
                consider(
                    wi + 1,
                    0,
                    pieces.clone(),
                    consumed_total,
                    false,
                    false,
                    next,
                );
                wi += 1;
                start = 0;
                if used >= capacity {
                    break;
                }
                continue;
            }

            // Marker-terminated prefix choices.
            for &marker in word.markers.iter().filter(|&&m| m > start && m < start + remaining) {
                let prefix_len = marker - start;
                if prefix_len + 1 <= room {
                    let mut candidate_pieces = pieces.clone();
                    let mut prefix: String = word.chars[start..marker].iter().collect();
                    prefix.push('-');
                    candidate_pieces.push(prefix);
                    let consumed = consumed_total + prefix_len;
                    consider(
                        wi,
                        marker,
                        candidate_pieces,
                        consumed,
                        true,
                        false,
                        (wi, marker, false),
                    );
                }
            }

            // Raw split is available only on an otherwise empty line and
            // only when no remaining marker prefix plus hyphen fits.
            if pieces.is_empty() && remaining > capacity {
                let marker_fits = word.markers.iter().any(|&m| {
                    m > start && m < word.chars.len() && (m - start) + 1 <= capacity
                });
                if !marker_fits {
                    let take = capacity;
                    let text: String = word.chars[start..start + take].iter().collect();
                    consider(
                        wi,
                        start + take,
                        vec![text],
                        take,
                        false,
                        true,
                        (wi, start + take, false),
                    );
                }
            }

            break;
        }

        let result = best;
        memo.insert(state, result.clone());
        result
    }

    rec((0, 0, true), words, opts, &mut memo)
        .map(|solution| solution.lines)
        .unwrap_or_default()
}

fn parse_segment(segment: &[char]) -> Vec<Word> {
    let mut words = Vec::new();
    let mut current = Vec::new();
    let mut markers = Vec::new();

    let flush = |words: &mut Vec<Word>, current: &mut Vec<char>, markers: &mut Vec<usize>| {
        if !current.is_empty() {
            words.push(Word {
                chars: std::mem::take(current),
                markers: std::mem::take(markers),
            });
        }
    };

    for &c in segment {
        if is_separator(c) {
            flush(&mut words, &mut current, &mut markers);
        } else if c == '~' {
            markers.push(current.len());
        } else {
            current.push(c);
        }
    }
    flush(&mut words, &mut current, &mut markers);
    words
}

pub fn wrap(input: &str, options: &WrapOptions) -> Result<Vec<String>, WrapError> {
    if options.width == 0 {
        return Err(WrapError::WidthZero);
    }
    if options.width > 256 {
        return Err(WrapError::WidthTooLarge {
            width: options.width,
        });
    }
    if options.first_indent >= options.width {
        return Err(WrapError::FirstIndentTooWide {
            indent: options.first_indent,
        });
    }
    if options.rest_indent >= options.width {
        return Err(WrapError::RestIndentTooWide {
            indent: options.rest_indent,
        });
    }
    if options.hyphen_cost > 1_000_000 {
        return Err(WrapError::HyphenCostTooLarge {
            cost: options.hyphen_cost,
        });
    }

    let chars = input.chars().count();
    if chars > 512 {
        return Err(WrapError::InputTooLong { chars });
    }

    for (index, c) in input.char_indices() {
        if c.is_control() && !matches!(c, '\t' | '\r' | '\n') {
            return Err(WrapError::InvalidControl { index });
        }
    }

    let original: Vec<(usize, char)> = input.char_indices().collect();
    for i in 0..original.len() {
        if original[i].1 == '~' {
            let valid_neighbor = |j: isize| -> bool {
                if j < 0 || j as usize >= original.len() {
                    return false;
                }
                let c = original[j as usize].1;
                c != '~' && !is_separator(c)
            };
            if !valid_neighbor(i as isize - 1) || !valid_neighbor(i as isize + 1) {
                return Err(WrapError::InvalidMarker {
                    index: original[i].0,
                });
            }
        }
    }

    let mut normalized = Vec::new();
    let mut iter = input.chars().peekable();
    while let Some(c) = iter.next() {
        if c == '\r' {
            if iter.peek() == Some(&'\n') {
                iter.next();
            }
            normalized.push('\n');
        } else {
            normalized.push(c);
        }
    }

    let mut output = Vec::new();
    let mut segment_start = 0usize;
    let mut i = 0usize;
    while i < normalized.len() {
        if normalized[i] == '\n' {
            let mut end = i + 1;
            while end < normalized.len() && normalized[end] == '\n' {
                end += 1;
            }
            if end - i >= 2 {
                let words = parse_segment(&normalized[segment_start..i]);
                let lines = solve_paragraph(&words, options);
                append_lines(&mut output, lines, options);
                for _ in 1..(end - i) {
                    output.push(String::new());
                }
                segment_start = end;
                i = end;
                continue;
            }
        }
        i += 1;
    }

    let words = parse_segment(&normalized[segment_start..]);
    let lines = solve_paragraph(&words, options);
    append_lines(&mut output, lines, options);
    Ok(output)
}

fn append_lines(output: &mut Vec<String>, lines: Vec<Line>, opts: &WrapOptions) {
    let total = lines.len();
    for (line_index, line) in lines.into_iter().enumerate() {
        let mut rendered = " ".repeat(line.indent);
        if opts.justify && line_index + 1 < total && line.piece_count >= 2 {
            let gaps = line.piece_count - 1;
            let extra = opts.width - line.prejustify.chars().count();
            let base = extra / gaps;
            let remainder = extra % gaps;
            for (i, piece) in line.pieces.iter().enumerate() {
                if i > 0 {
                    rendered.push(' ');
                    rendered.push_str(&" ".repeat(base + usize::from((i - 1 + line_index % gaps) % gaps < remainder)));
                }
                rendered.push_str(piece);
            }
        } else {
            for (i, piece) in line.pieces.iter().enumerate() {
                if i > 0 {
                    rendered.push(' ');
                }
                rendered.push_str(piece);
            }
        }
        output.push(rendered);
    }
}
