use std::collections::{HashMap, HashSet};

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub ready_at: u64,
    pub priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Dispatch {
    pub id: String,
    pub effective_priority: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    BadId,
    BadPriority,
    Duplicate,
    Unknown,
    Cancelled,
    Completed,
}

#[derive(Clone, Copy, PartialEq, Eq)]
enum Status {
    Queued,
    Cancelled,
    Completed,
}

struct Entry {
    job: Job,
    sequence: u64,
    status: Status,
}

pub struct Scheduler {
    aging_ms: u64,
    last_time: u64,
    next_sequence: u64,
    entries: HashMap<String, Entry>,
    reserved: HashSet<String>,
}

impl Scheduler {
    pub fn new(aging_ms: u64) -> Self {
        Self {
            aging_ms,
            last_time: 0,
            next_sequence: 0,
            entries: HashMap::new(),
            reserved: HashSet::new(),
        }
    }

    pub fn insert(&mut self, now: u64, job: Job) -> Result<(), Error> {
        self.check_time(now)?;

        if !valid_id(&job.id) {
            return Err(Error::BadId);
        }
        if job.priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.reserved.contains(&job.id) {
            return Err(Error::Duplicate);
        }

        let sequence = self.next_sequence;
        self.next_sequence = self.next_sequence.wrapping_add(1);
        self.reserved.insert(job.id.clone());
        self.entries.insert(
            job.id.clone(),
            Entry {
                job,
                sequence,
                status: Status::Queued,
            },
        );
        self.last_time = now;
        Ok(())
    }

    pub fn cancel(&mut self, now: u64, id: &str) -> Result<(), Error> {
        self.check_time(now)?;

        let entry = self.entries.get_mut(id).ok_or(Error::Unknown)?;
        match entry.status {
            Status::Queued => entry.status = Status::Cancelled,
            Status::Cancelled => return Err(Error::Cancelled),
            Status::Completed => return Err(Error::Completed),
        }
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Dispatch>, Error> {
        self.check_time(now)?;

        let mut selected: Option<(&str, u8, u64)> = None;
        for (id, entry) in &self.entries {
            if entry.status != Status::Queued || entry.job.ready_at > now {
                continue;
            }

            let effective = effective_priority(&entry.job, now, self.aging_ms);
            let replace = match selected {
                None => true,
                Some((_, best_priority, best_sequence)) => {
                    effective > best_priority
                        || (effective == best_priority && entry.sequence < best_sequence)
                }
            };
            if replace {
                selected = Some((id.as_str(), effective, entry.sequence));
            }
        }

        let result = if let Some((id, effective_priority, _)) = selected {
            let entry = self.entries.get_mut(id).expect("selected entry exists");
            entry.status = Status::Completed;
            Some(Dispatch {
                id: entry.job.id.clone(),
                effective_priority,
            })
        } else {
            None
        };
        self.last_time = now;
        Ok(result)
    }

    fn check_time(&self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            Err(Error::TimeReversed)
        } else {
            Ok(())
        }
    }
}

fn valid_id(id: &str) -> bool {
    let bytes = id.as_bytes();
    (1..=16).contains(&bytes.len())
        && bytes
            .iter()
            .all(|&b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
}

fn effective_priority(job: &Job, now: u64, aging_ms: u64) -> u8 {
    if aging_ms == 0 {
        return job.priority;
    }

    let increments = (now - job.ready_at) / aging_ms;
    let room = (9 - job.priority) as u64;
    if increments >= room {
        9
    } else {
        job.priority + increments as u8
    }
}
