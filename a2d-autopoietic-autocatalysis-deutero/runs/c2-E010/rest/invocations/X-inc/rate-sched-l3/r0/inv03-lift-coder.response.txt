```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

#[derive(Clone)]
struct JobSpec {
    id: String,
    priority: u8,
    dependencies: Vec<usize>,
    max_attempts: u8,
}

struct Runtime {
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<JobSpec>,
    runtime: Vec<Runtime>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut ids = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            let valid_id = !job.id.is_empty()
                && job.id.len() <= 16
                && job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-');
            if !valid_id {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if ids.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut specs = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut dependencies = Vec::with_capacity(job.dependencies.len());
            for (di, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep) {
                    return Err(Error::DuplicateDependency(i, di));
                }
                let Some(&dep_index) = ids.get(dep) else {
                    return Err(Error::MissingDependency(i, di));
                };
                dependencies.push(dep_index);
            }
            specs.push(JobSpec {
                id: job.id.clone(),
                priority: job.priority,
                dependencies,
                max_attempts: job.max_attempts,
            });
        }

        fn visit(index: usize, jobs: &[JobSpec], marks: &mut [u8]) -> bool {
            if marks[index] == 1 {
                return false;
            }
            if marks[index] == 2 {
                return true;
            }
            marks[index] = 1;
            for &dep in &jobs[index].dependencies {
                if !visit(dep, jobs, marks) {
                    return false;
                }
            }
            marks[index] = 2;
            true
        }

        let mut marks = vec![0; specs.len()];
        for i in 0..specs.len() {
            if !visit(i, &specs, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let runtime = specs
            .iter()
            .map(|_| Runtime {
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs: specs,
            runtime,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.cycle_started
            && self.runtime.iter().any(|r| {
                matches!(r.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for runtime in &mut self.runtime {
            runtime.state = State::Waiting { ready_at: now };
            runtime.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (i, (job, runtime)) in self.jobs.iter().zip(&self.runtime).enumerate() {
            let State::Waiting { ready_at } = runtime.state else {
                continue;
            };
            if ready_at > now
                || !job
                    .dependencies
                    .iter()
                    .all(|&dep| matches!(self.runtime[dep].state, State::Succeeded))
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (job.priority as u64 + age).min(9) as u8
            };
            if best.is_none_or(|(best_i, best_priority)| {
                effective > best_priority || (effective == best_priority && i < best_i)
            }) {
                best = Some((i, effective));
            }
        }

        self.last_time = now;
        let Some((index, _)) = best else {
            return Ok(None);
        };

        let runtime = &mut self.runtime[index];
        runtime.attempts += 1;
        let attempt = runtime.attempts;
        runtime.state = State::Running { attempt };
        Ok(Some(Ticket {
            id: self.jobs[index].id.clone(),
            attempt,
        }))
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(index) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        if !matches!(self.runtime[index].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        self.last_time = now;
        if success {
            self.runtime[index].state = State::Succeeded;
        } else if self.runtime[index].attempts == self.jobs[index].max_attempts {
            self.runtime[index].state = State::Failed;
            self.propagate_blocked();
        } else {
            let attempt = self.runtime[index].attempts;
            let mut delay = self.retry_base;
            for _ in 1..attempt {
                delay = delay.saturating_mul(2).min(self.retry_cap);
            }
            delay = delay.min(self.retry_cap);
            self.runtime[index].state = State::Waiting {
                ready_at: now.saturating_add(delay),
            };
        }
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(index) = self.jobs.iter().position(|job| job.id == id) else {
            return Err(Error::Unknown);
        };
        Ok(self.runtime[index].state.clone())
    }

    fn propagate_blocked(&mut self) {
        loop {
            let mut changed = false;
            for i in 0..self.jobs.len() {
                if !matches!(self.runtime[i].state, State::Waiting { .. }) {
                    continue;
                }
                if self.jobs[i].dependencies.iter().any(|&dep| {
                    matches!(self.runtime[dep].state, State::Failed | State::Blocked)
                }) {
                    self.runtime[i].state = State::Blocked;
                    changed = true;
                }
            }
            if !changed {
                break;
            }
        }
    }
}
```
