```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64 },
    Running { attempt: u8 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Ticket {
    pub id: String,
    pub attempt: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    Busy,
    NoCycle,
    Unknown,
    NotRunning,
}

struct JobData {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    attempts: u8,
}

pub struct Runner {
    jobs: Vec<JobData>,
    by_id: std::collections::HashMap<String, usize>,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    last_time: u64,
    cycle_started: bool,
}

impl Runner {
    pub fn new(
        jobs: Vec<Job>,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if retry_base == 0 || retry_cap < retry_base {
            return Err(Error::Config);
        }

        let mut by_id = std::collections::HashMap::new();

        for (i, job) in jobs.iter().enumerate() {
            if job.id.is_empty()
                || job.id.len() > 16
                || !job
                    .id
                    .bytes()
                    .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
            {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if by_id.contains_key(&job.id) {
                return Err(Error::Duplicate(i));
            }
            by_id.insert(job.id.clone(), i);
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, dep) in job.dependencies.iter().enumerate() {
                if !seen.insert(dep.as_str()) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                let Some(&dep_index) = by_id.get(dep) else {
                    return Err(Error::MissingDependency(i, j));
                };
                resolved.push(dep_index);
            }
            dependencies.push(resolved);
        }

        fn has_cycle(
            node: usize,
            dependencies: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return true;
            }
            if marks[node] == 2 {
                return false;
            }
            marks[node] = 1;
            for &dep in &dependencies[node] {
                if has_cycle(dep, dependencies, marks) {
                    return true;
                }
            }
            marks[node] = 2;
            false
        }

        let mut marks = vec![0; jobs.len()];
        for i in 0..jobs.len() {
            if has_cycle(i, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let jobs = jobs
            .into_iter()
            .zip(dependencies)
            .map(|(job, dependencies)| JobData {
                job,
                dependencies,
                state: State::Waiting { ready_at: 0 },
                attempts: 0,
            })
            .collect();

        Ok(Self {
            jobs,
            by_id,
            aging_ms,
            retry_base,
            retry_cap,
            last_time: 0,
            cycle_started: false,
        })
    }

    pub fn begin_cycle(&mut self, now: u64) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if self.cycle_started
            && self.jobs.iter().any(|job| {
                matches!(job.state, State::Waiting { .. } | State::Running { .. })
            })
        {
            return Err(Error::Busy);
        }

        for job in &mut self.jobs {
            job.state = State::Waiting { ready_at: now };
            job.attempts = 0;
        }
        self.cycle_started = true;
        self.last_time = now;
        Ok(())
    }

    pub fn next_ready(&mut self, now: u64) -> Result<Option<Ticket>, Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }

        let mut best: Option<(usize, u8)> = None;
        for (i, job) in self.jobs.iter().enumerate() {
            let State::Waiting { ready_at } = job.state else {
                continue;
            };
            if ready_at > now
                || job.dependencies.iter().any(|&dep| {
                    !matches!(self.jobs[dep].state, State::Succeeded)
                })
            {
                continue;
            }

            let effective = if self.aging_ms == 0 {
                job.job.priority
            } else {
                let age = (now - ready_at) / self.aging_ms;
                (job.job.priority as u64 + age).min(9) as u8
            };

            if best.map_or(true, |(_, priority)| effective > priority) {
                best = Some((i, effective));
            }
        }

        self.last_time = now;
        if let Some((i, _)) = best {
            let job = &mut self.jobs[i];
            job.attempts += 1;
            job.state = State::Running {
                attempt: job.attempts,
            };
            Ok(Some(Ticket {
                id: job.job.id.clone(),
                attempt: job.attempts,
            }))
        } else {
            Ok(None)
        }
    }

    pub fn complete(&mut self, now: u64, id: &str, success: bool) -> Result<(), Error> {
        if now < self.last_time {
            return Err(Error::TimeReversed);
        }
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(&i) = self.by_id.get(id) else {
            return Err(Error::Unknown);
        };
        if !matches!(self.jobs[i].state, State::Running { .. }) {
            return Err(Error::NotRunning);
        }

        let terminal_failure;
        {
            let job = &mut self.jobs[i];
            if success {
                job.state = State::Succeeded;
                terminal_failure = false;
            } else if job.attempts == job.job.max_attempts {
                job.state = State::Failed;
                terminal_failure = true;
            } else {
                let shift = u32::from(job.attempts.saturating_sub(1));
                let delay = self
                    .retry_base
                    .checked_mul(1u64.checked_shl(shift).unwrap_or(u64::MAX))
                    .unwrap_or(u64::MAX)
                    .min(self.retry_cap);
                job.state = State::Waiting {
                    ready_at: now.saturating_add(delay),
                };
                terminal_failure = false;
            }
        }

        if terminal_failure {
            loop {
                let mut changed = false;
                for job in &mut self.jobs {
                    if matches!(job.state, State::Waiting { .. })
                        && job.dependencies.iter().any(|&dep| {
                            matches!(
                                self.jobs[dep].state,
                                State::Failed | State::Blocked
                            )
                        })
                    {
                        job.state = State::Blocked;
                        changed = true;
                    }
                }
                if !changed {
                    break;
                }
            }
        }

        self.last_time = now;
        Ok(())
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        if !self.cycle_started {
            return Err(Error::NoCycle);
        }
        let Some(&i) = self.by_id.get(id) else {
            return Err(Error::Unknown);
        };
        Ok(self.jobs[i].state.clone())
    }
}
```
