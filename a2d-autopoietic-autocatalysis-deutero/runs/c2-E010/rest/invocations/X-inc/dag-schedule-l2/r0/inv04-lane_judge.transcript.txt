pub fn analyze(tasks: &[Task], target: u64) -> Result<Analysis, Error> {
    let count = tasks.len();

    // Validate every dependency before doing any graph or timing work.
    for (task_id, task) in tasks.iter().enumerate() {
        let mut seen = vec![false; count];
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen[value] {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen[value] = true;
        }
    }

    // Dependencies point from predecessors to this task. Build successor
    // lists and indegrees while preserving IDs as the deterministic priority.
    let mut successors = vec![Vec::new(); count];
    let mut indegree = vec![0usize; count];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &pred in &task.deps {
            successors[pred].push(task_id);
        }
    }

    let mut available = std::collections::BinaryHeap::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(id));
        }
    }

    let mut order = Vec::with_capacity(count);
    while let Some(std::cmp::Reverse(id)) = available.pop() {
        order.push(id);
        for &child in &successors[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != count {
        let removed = {
            let mut flags = vec![false; count];
            for &id in &order {
                flags[id] = true;
            }
            flags
        };
        let blocked = (0..count).filter(|&id| !removed[id]).collect();
        return Err(Error::Cycle { blocked });
    }

    let mut earliest = vec![Timing { start: 0, finish: 0 }; count];
    for &id in &order {
        let start = tasks[id]
            .deps
            .iter()
            .map(|&pred| earliest[pred].finish)
            .max()
            .unwrap_or(0);
        let finish = start
            .checked_add(tasks[id].duration)
            .ok_or(Error::Overflow { task: id })?;
        earliest[id] = Timing { start, finish };
    }

    let makespan = earliest.iter().map(|timing| timing.finish).max().unwrap_or(0);
    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    let mut latest = vec![Timing { start: 0, finish: 0 }; count];
    for &id in order.iter().rev() {
        let finish = if successors[id].is_empty() {
            target
        } else {
            successors[id]
                .iter()
                .map(|&child| latest[child].start)
                .min()
                .unwrap()
        };
        let start = finish - tasks[id].duration;
        latest[id] = Timing { start, finish };
    }

    let slack: Vec<u64> = (0..count)
        .map(|id| latest[id].start - earliest[id].start)
        .collect();
    let baseline = target - makespan;
    let critical = (0..count)
        .filter(|&id| slack[id] == baseline)
        .collect();

    let mut path = Vec::new();
    if count > 0 {
        let mut current = (0..count)
            .filter(|&id| successors[id].is_empty() && earliest[id].finish == makespan)
            .max()
            .unwrap();

        loop {
            path.push(current);
            if tasks[current].deps.is_empty() {
                break;
            }
            current = *tasks[current]
                .deps
                .iter()
                .max_by_key(|&&pred| (earliest[pred].finish, pred))
                .unwrap();
        }
        path.reverse();
    }

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}

    for (task_id, task) in tasks.iter().enumerate() {

        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {

                    task: task_id,
                    position,
                    value,
                });

            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if seen.contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            seen.push(value);
        }
    }

    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }

    // Kahn traversal with the smallest currently available ID first.
    let mut available = Vec::new();
    for id in 0..n {
        if indegree[id] == 0 {
            available.push(id);
        }

    let mut order = Vec::with_capacity(n);
    let mut earliest = vec![Timing { start: 0, finish: 0 }; n];

    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|&(_, id)| *id)
            .map(|(position, _)| position)
            .unwrap();

        let start = earliest[id].start;
        let finish = match start.checked_add(tasks[id].duration) {
            Some(finish) => finish,
            None => return Err(Error::Overflow { task: id }),

        earliest[id] = Timing { start, finish };
        order.push(id);

        for &child in &successors[id] {
            if finish > earliest[child].start {
                earliest[child].start = finish;
            }
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if order.len() != n {
        let mut blocked: Vec<usize> = (0..n).filter(|&id| indegree[id] != 0).collect();
        blocked.sort_unstable();
        return Err(Error::Cycle { blocked });

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });

                .map(|&child| latest[child].start)
                .min()
                .unwrap()
        };

        latest[id] = Timing { start, finish };
    }

        .map(|id| latest[id].start - earliest[id].start)

    // Validate every dependency before performing any other computation.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= count {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    // Build successor edges and indegrees.
    let mut successors = vec![Vec::new(); count];
    let mut indegree = vec![0usize; count];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            successors[predecessor].push(task_id);
        }
    }

    // Kahn traversal using a min-heap, so the smallest currently available
    // task is always removed next.
    let mut available = std::collections::BinaryHeap::new();
    for (task_id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(std::cmp::Reverse(task_id));
        }
    }

    let mut order = Vec::with_capacity(count);
    let mut remaining_indegree = indegree;
    while let Some(std::cmp::Reverse(task_id)) = available.pop() {
        order.push(task_id);
        for &child in &successors[task_id] {
            remaining_indegree[child] -= 1;
            if remaining_indegree[child] == 0 {
                available.push(std::cmp::Reverse(child));
            }
        }
    }

    if order.len() != count {
        let removed = {
            let mut removed = vec![false; count];
            for &task_id in &order {
                removed[task_id] = true;
            }
            removed
        };
        let blocked = (0..count).filter(|&task_id| !removed[task_id]).collect();
        return Err(Error::Cycle { blocked });
    }

    // Compute earliest timings in Kahn order.
    let mut earliest = vec![Timing { start: 0, finish: 0 }; count];
    let mut makespan = 0;
    for &task_id in &order {
        let start = tasks[task_id]
            .deps
            .iter()
            .map(|&predecessor| earliest[predecessor].finish)
            .max()
            .unwrap_or(0);
        let finish = start
            .checked_add(tasks[task_id].duration)
            .ok_or(Error::Overflow { task: task_id })?;
        earliest[task_id] = Timing { start, finish };
        makespan = makespan.max(finish);
    }

    if target < makespan {
        return Err(Error::TargetTooSmall { minimum: makespan });
    }

    // Compute latest timings in reverse Kahn order.
    let mut latest = vec![Timing { start: 0, finish: 0 }; count];
    for &task_id in order.iter().rev() {
        let finish = if successors[task_id].is_empty() {
            target
        } else {
            successors[task_id]
                .iter()
                .map(|&child| latest[child].start)
                .min()
                .expect("nonempty successor list")
        };
        let start = finish - tasks[task_id].duration;
        latest[task_id] = Timing { start, finish };
    }

    let slack: Vec<u64> = (0..count)
        .map(|task_id| latest[task_id].start - earliest[task_id].start)
        .collect();
    let baseline = target - makespan;
    let critical = (0..count)
        .filter(|&task_id| slack[task_id] == baseline)
        .collect();

    // Select the greatest-ID sink finishing at the makespan, then walk back
    // using the specified local predecessor tie-break.
    let mut path = Vec::new();
    if count > 0 {
        let mut current = (0..count)
            .filter(|&task_id| {
                successors[task_id].is_empty() && earliest[task_id].finish == makespan
            })
            .max()
            .expect("a nonempty acyclic graph has a sink");

        loop {
            path.push(current);
            if tasks[current].deps.is_empty() {
                break;
            }
            current = tasks[current]
                .deps
                .iter()
                .copied()
                .max_by_key(|&predecessor| (earliest[predecessor].finish, predecessor))
                .expect("non-root task has a predecessor");
        }
        path.reverse();
    }

    Ok(Analysis {
        order,
        earliest,
        latest,
        slack,
        critical,
        path,
        makespan,
    })
}
