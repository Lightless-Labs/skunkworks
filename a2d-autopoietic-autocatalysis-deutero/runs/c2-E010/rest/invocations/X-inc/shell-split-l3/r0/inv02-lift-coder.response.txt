#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Quote {
    None,
    Single(usize),
    Double(usize),
}

fn is_name_start(c: char) -> bool {
    c.is_ascii_alphabetic() || c == '_'
}

fn is_name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn lookup<'a>(name: &str, env: &'a [(&str, &str)]) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append(word: &mut Word, text: &str, wildcards_active: bool) {
    for c in text.chars() {
        if wildcards_active && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn decode_default(text: &str) -> String {
    let mut out = String::new();
    let mut chars = text.chars().peekable();
    while let Some(c) = chars.next() {
        if c == '\\' {
            if let Some(next) = chars.next() {
                out.push(next);
            }
        } else {
            out.push(c);
        }
    }
    out
}

fn parse_braced(input: &str, dollar: usize) -> Result<(usize, String, Option<String>), SplitError> {
    let after_open = dollar + 2;
    let mut i = after_open;
    let mut close = None;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        if c == '\\' {
            i += 1;
            if i < input.len() {
                i += input[i..].chars().next().unwrap().len_utf8();
            }
        } else if c == '}' {
            close = Some(i);
            break;
        } else {
            i += c.len_utf8();
        }
    }

    let close = close.ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let contents = &input[after_open..close];

    let name_end = contents
        .char_indices()
        .find_map(|(idx, c)| (!is_name_char(c)).then_some(idx))
        .unwrap_or(contents.len());

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &contents[..name_end];
    let tail = &contents[name_end..];

    let default = if tail.is_empty() {
        None
    } else if let Some(text) = tail.strip_prefix(":-") {
        Some(text.to_string())
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((close + 1, name.to_string(), default))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = Word {
        text: String::new(),
        glob: Vec::new(),
    };
    let mut active = false;
    let mut quote = Quote::None;
    let mut i = 0;

    while i < input.len() {
        let c = input[i..].chars().next().unwrap();
        let next_i = i + c.len_utf8();

        match quote {
            Quote::Single(_) => {
                if c == '\'' {
                    quote = Quote::None;
                } else {
                    word.text.push(c);
                }
                i = next_i;
            }
            Quote::Double(open_pos) => {
                if c == '"' {
                    quote = Quote::None;
                    i = next_i;
                } else if c == '\\' {
                    let slash = i;
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: slash });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    if matches!(next, '"' | '\\' | '$' | '\n') {
                        if next != '\n' {
                            word.text.push(next);
                        }
                        i = next_i + next.len_utf8();
                    } else {
                        word.text.push('\\');
                        i = next_i;
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, name, default) = parse_braced(input, i)?;
                        let value = match default {
                            Some(text) => match lookup(&name, env) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(&text),
                            },
                            None => lookup(&name, env).unwrap_or("").to_string(),
                        };
                        append(&mut word, &value, false);
                        active = true;
                        i = end;
                    } else if next_i < input.len()
                        && is_name_start(input[next_i..].chars().next().unwrap())
                    {
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_char(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[next_i..end];
                        append(&mut word, lookup(name, env).unwrap_or(""), false);
                        active = true;
                        i = end;
                    } else {
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    word.text.push(c);
                    i = next_i;
                }
                let _ = open_pos;
            }
            Quote::None => {
                if matches!(c, ' ' | '\t' | '\r' | '\n') {
                    if active {
                        words.push(word);
                        word = Word {
                            text: String::new(),
                            glob: Vec::new(),
                        };
                        active = false;
                    }
                    i = next_i;
                } else if c == '\'' {
                    active = true;
                    quote = Quote::Single(i);
                    i = next_i;
                } else if c == '"' {
                    active = true;
                    quote = Quote::Double(i);
                    i = next_i;
                } else if c == '\\' {
                    if next_i == input.len() {
                        return Err(SplitError::TrailingEscape { pos: i });
                    }
                    let next = input[next_i..].chars().next().unwrap();
                    if next == '\n' {
                        i = next_i + next.len_utf8();
                    } else {
                        active = true;
                        word.text.push(next);
                        i = next_i + next.len_utf8();
                    }
                } else if c == '`' {
                    return Err(SplitError::CommandSubstitution { pos: i });
                } else if c == '$' {
                    if input[next_i..].starts_with('(') {
                        return Err(SplitError::CommandSubstitution { pos: i });
                    }
                    if input[next_i..].starts_with('{') {
                        let (end, name, default) = parse_braced(input, i)?;
                        let value = match default {
                            Some(text) => match lookup(&name, env) {
                                Some(value) if !value.is_empty() => value.to_string(),
                                _ => decode_default(&text),
                            },
                            None => lookup(&name, env).unwrap_or("").to_string(),
                        };
                        append(&mut word, &value, true);
                        active = true;
                        i = end;
                    } else if next_i < input.len()
                        && is_name_start(input[next_i..].chars().next().unwrap())
                    {
                        let mut end = next_i;
                        while end < input.len() {
                            let ch = input[end..].chars().next().unwrap();
                            if !is_name_char(ch) {
                                break;
                            }
                            end += ch.len_utf8();
                        }
                        let name = &input[next_i..end];
                        append(&mut word, lookup(name, env).unwrap_or(""), true);
                        active = true;
                        i = end;
                    } else {
                        active = true;
                        word.text.push('$');
                        i = next_i;
                    }
                } else {
                    active = true;
                    append(&mut word, &c.to_string(), true);
                    i = next_i;
                }
            }
        }
    }

    match quote {
        Quote::Single(pos) => return Err(SplitError::UnterminatedSingle { pos }),
        Quote::Double(pos) => return Err(SplitError::UnterminatedDouble { pos }),
        Quote::None => {}
    }

    if active {
        words.push(word);
    }
    Ok(words)
}
