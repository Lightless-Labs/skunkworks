use std::collections::BTreeSet;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
    UnknownSpecial,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [BTreeSet<u8>; 5],
    unrestricted_dom: bool,
    unrestricted_dow: bool,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WindowError {
    InvalidStart { component: DatePart },
    InvalidEnd { component: DatePart },
    ReversedWindow,
    ZeroCap,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct FireWindow {
    pub fires: Vec<CivilDateTime>,
    pub truncated: bool,
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn domain(field: Field) -> (u8, u8) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn names(field: Field) -> Option<&'static [(&'static str, u8)]> {
    match field {
        Field::Month => Some(&[
            ("JAN", 1), ("FEB", 2), ("MAR", 3), ("APR", 4), ("MAY", 5), ("JUN", 6),
            ("JUL", 7), ("AUG", 8), ("SEP", 9), ("OCT", 10), ("NOV", 11), ("DEC", 12),
        ]),
        Field::DayOfWeek => Some(&[
            ("SUN", 0), ("MON", 1), ("TUE", 2), ("WED", 3), ("THU", 4), ("FRI", 5),
            ("SAT", 6),
        ]),
        _ => None,
    }
}

fn syntax(field: Field, s: &str) -> Result<Option<u8>, ()> {
    if !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit()) {
        return Ok(None);
    }
    if let Some(table) = names(field) {
        for &(name, value) in table {
            if s.eq_ignore_ascii_case(name) {
                return Ok(Some(value));
            }
        }
    }
    Err(())
}

fn number(field: Field, s: &str, offset: usize) -> Result<u32, ParseError> {
    let bytes = s.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }
    let mut value = 0u32;
    for &b in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn endpoint(field: Field, s: &str, offset: usize) -> Result<u8, ParseError> {
    if let Ok(Some(value)) = syntax(field, s) {
        return Ok(value);
    }
    let v = number(field, s, offset)?;
    let (lo, hi) = domain(field);
    if v < lo as u32 || v > hi as u32 {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(v as u8)
}

fn parse_item(field: Field, item: &str, offset: usize, out: &mut BTreeSet<u8>) -> Result<(), ParseError> {
    let (lo, hi) = domain(field);
    let slash_count = item.bytes().filter(|&b| b == b'/').count();
    let (base, step_text) = if slash_count == 0 {
        (item, None)
    } else if slash_count == 1 {
        let p = item.find('/').unwrap();
        (&item[..p], Some((&item[p + 1..], offset + p + 1)))
    } else {
        return Err(ParseError::InvalidSyntax { field, offset });
    };

    if base.is_empty() || step_text.map_or(false, |(s, _)| s.is_empty()) {
        return Err(ParseError::InvalidSyntax { field, offset });
    }

    let hyphens = base.bytes().filter(|&b| b == b'-').count();
    let (left, right) = if hyphens == 0 {
        (base, None)
    } else if hyphens == 1 {
        let p = base.find('-').unwrap();
        (&base[..p], Some((&base[p + 1..], offset + p + 1)))
    } else {
        return Err(ParseError::InvalidSyntax { field, offset });
    };

    if left.is_empty() || right.map_or(false, |(s, _)| s.is_empty()) || left == "*" {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if right.is_some() && (left == "*" || right.unwrap().0 == "*") {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if left != "*" && syntax(field, left).is_err() {
        return Err(ParseError::InvalidSyntax { field, offset });
    }
    if let Some((r, _)) = right {
        if syntax(field, r).is_err() {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
    }
    if let Some((s, _)) = step_text {
        if !s.bytes().all(|b| b.is_ascii_digit()) {
            return Err(ParseError::InvalidSyntax { field, offset });
        }
    }

    let (start, end) = if left == "*" {
        (lo, hi)
    } else {
        let start = endpoint(field, left, offset)?;
        let end = if let Some((r, ro)) = right {
            endpoint(field, r, ro)?
        } else if step_text.is_some() {
            hi
        } else {
            start
        };
        if right.is_some() && start > end {
            return Err(ParseError::ReversedRange {
                field,
                offset: right.unwrap().1,
            });
        }
        (start, end)
    };

    let step = if let Some((s, so)) = step_text {
        let n = number(field, s, so)?;
        let width = (hi - lo + 1) as u32;
        if n == 0 || n > width {
            return Err(ParseError::BadStep { field, offset: so });
        }
        n as u8
    } else {
        1
    };

    let mut v = start;
    while v <= end {
        out.insert(v);
        match v.checked_add(step) {
            Some(next) => v = next,
            None => break,
        }
    }
    Ok(())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let expanded = if input.starts_with('@') {
        match input {
            "@hourly" => "0 * * * *",
            "@daily" => "0 0 * * *",
            "@weekly" => "0 0 * * 0",
            "@monthly" => "0 0 1 * *",
            "@yearly" => "0 0 1 1 *",
            _ => return Err(ParseError::UnknownSpecial),
        }
    } else {
        input
    };

    let parts: Vec<&str> = expanded.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }
    let mut values: [BTreeSet<u8>; 5] = std::array::from_fn(|_| BTreeSet::new());
    for i in 0..5 {
        let field = FIELDS[i];
        for (offset, b) in parts[i].bytes().enumerate() {
            if !(b.is_ascii_digit()
                || b.is_ascii_alphabetic()
                || matches!(b, b'*' | b'-' | b'/' | b','))
            {
                return Err(ParseError::InvalidCharacter { field, offset });
            }
        }
        let mut start = 0;
        for item in parts[i].split(',') {
            if item.is_empty() {
                return Err(ParseError::EmptyItem { field, offset: start });
            }
            parse_item(field, item, start, &mut values[i])?;
            start += item.len() + 1;
        }
    }
    Ok(Cron {
        values,
        unrestricted_dom: parts[2] == "*",
        unrestricted_dow: parts[4] == "*",
    })
}

fn leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn month_len(year: u16, month: u8) -> u8 {
    match month {
        2 => if leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn validate(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime { component: DatePart::Year });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Month });
    }
    if at.day == 0 || at.day > month_len(at.year, at.month) {
        return Err(DateError::InvalidDateTime { component: DatePart::Day });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime { component: DatePart::Hour });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime { component: DatePart::Minute });
    }
    Ok(())
}

fn weekday(at: CivilDateTime) -> u8 {
    let mut days = 0u32;
    for y in 2000..at.year {
        days += if leap(y) { 366 } else { 365 };
    }
    for m in 1..at.month {
        days += month_len(at.year, m) as u32;
    }
    days += at.day as u32 - 1;
    ((days + 6) % 7) as u8
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let i = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        self.values[i].iter().copied().collect()
    }

    fn day_matches(&self, at: CivilDateTime) -> bool {
        let dom = self.values[2].contains(&at.day);
        let dow = self.values[4].contains(&weekday(at));
        match (self.unrestricted_dom, self.unrestricted_dow) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                if at.day == month_len(at.year, at.month) { dom && dow } else { dom || dow }
            }
        }
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate(at)?;
        Ok(self.values[0].contains(&at.minute)
            && self.values[1].contains(&at.hour)
            && self.values[3].contains(&at.month)
            && self.day_matches(at))
    }

    fn search(&self, at: CivilDateTime, forward: bool, inclusive: bool) -> Option<CivilDateTime> {
        let mut date = CivilDateTime { hour: 0, minute: 0, ..at };
        loop {
            if self.values[3].contains(&date.month) && self.day_matches(date) {
                if forward {
                    for &h in &self.values[1] {
                        for &m in &self.values[0] {
                            let candidate = CivilDateTime { hour: h, minute: m, ..date };
                            if candidate > at || (inclusive && candidate == at) {
                                return Some(candidate);
                            }
                        }
                    }
                } else {
                    for &h in self.values[1].iter().rev() {
                        for &m in self.values[0].iter().rev() {
                            let candidate = CivilDateTime { hour: h, minute: m, ..date };
                            if candidate < at || (inclusive && candidate == at) {
                                return Some(candidate);
                            }
                        }
                    }
                }
            }
            if forward {
                if date.year == 2399 && date.month == 12 && date.day == 31 { return None; }
                if date.day < month_len(date.year, date.month) {
                    date.day += 1;
                } else if date.month < 12 {
                    date.month += 1;
                    date.day = 1;
                } else {
                    date.year += 1;
                    date.month = 1;
                    date.day = 1;
                }
            } else {
                if date.year == 2000 && date.month == 1 && date.day == 1 { return None; }
                if date.day > 1 {
                    date.day -= 1;
                } else if date.month > 1 {
                    date.month -= 1;
                    date.day = month_len(date.year, date.month);
                } else {
                    date.year -= 1;
                    date.month = 12;
                    date.day = 31;
                }
            }
            date.hour = if forward { 0 } else { 23 };
            date.minute = if forward { 0 } else { 59 };
            at = date;
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(after)?;
        Ok(self.search(after, true, false))
    }

    pub fn previous_fire(&self, before: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate(before)?;
        Ok(self.search(before, false, false))
    }

    pub fn fires_between(
        &self,
        start: CivilDateTime,
        end: CivilDateTime,
        cap: usize,
    ) -> Result<FireWindow, WindowError> {
        validate(start).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidStart { component },
        })?;
        validate(end).map_err(|e| match e {
            DateError::InvalidDateTime { component } => WindowError::InvalidEnd { component },
        })?;
        if start > end { return Err(WindowError::ReversedWindow); }
        if cap == 0 { return Err(WindowError::ZeroCap); }
        let mut fires = Vec::new();
        if start == end {
            return Ok(FireWindow { fires, truncated: false });
        }
        let mut cursor = start;
        loop {
            if let Some(fire) = self.search(cursor, true, true) {
                if fire >= end {
                    break;
                }
                if fires.len() == cap {
                    return Ok(FireWindow { fires, truncated: true });
                }
                fires.push(fire);
                cursor = fire;
                if cursor == CivilDateTime { year: 2399, month: 12, day: 31, hour: 23, minute: 59 } {
                    break;
                }
                cursor = increment_minute(cursor);
            } else {
                break;
            }
        }
        Ok(FireWindow { fires, truncated: false })
    }

    pub fn canonical(&self) -> String {
        let mut fields = Vec::new();
        for i in 0..5 {
            let field = FIELDS[i];
            let (lo, hi) = domain(field);
            let full = self.values[i].len() == (hi - lo + 1) as usize;
            let text = if i == 2 && self.unrestricted_dom {
                "*".to_string()
            } else if i == 4 && self.unrestricted_dow {
                "*".to_string()
            } else if full && matches!(field, Field::Minute | Field::Hour | Field::Month) {
                "*".to_string()
            } else if full && matches!(field, Field::DayOfMonth | Field::DayOfWeek) {
                "*/1".to_string()
            } else {
                self.values[i].iter().map(u8::to_string).collect::<Vec<_>>().join(",")
            };
            fields.push(text);
        }
        fields.join(" ")
    }
}

fn increment_minute(mut at: CivilDateTime) -> CivilDateTime {
    if at.minute < 59 {
        at.minute += 1;
    } else {
        at.minute = 0;
        if at.hour < 23 {
            at.hour += 1;
        } else {
            at.hour = 0;
            if at.day < month_len(at.year, at.month) {
                at.day += 1;
            } else if at.month < 12 {
                at.month += 1;
                at.day = 1;
            } else if at.year < 2399 {
                at.year += 1;
                at.month = 1;
                at.day = 1;
            }
        }
    }
    at
}
