#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Operator {
    Simple,
    Plus,
    Hash,
    Dot,
    Slash,
    Semicolon,
    Question,
    Ampersand,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug)]
struct Specifier<'a> {
    name: &'a str,
    start: usize,
    modifier: Modifier,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression<'a>(
    content: &'a str,
    content_start: usize,
    close_pos: usize,
) -> Result<(Operator, Vec<Specifier<'a>>), Error> {
    let bytes = content.as_bytes();
    let (operator, offset) = if let Some(&first) = bytes.first() {
        let operator = match first {
            b'+' => Some(Operator::Plus),
            b'#' => Some(Operator::Hash),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semicolon),
            b'?' => Some(Operator::Question),
            b'&' => Some(Operator::Ampersand),
            _ => None,
        };
        match operator {
            Some(operator) => (operator, 1),
            None if first.is_ascii_punctuation() && !is_name_start(first) => {
                return Err(Error::UnsupportedOperator {
                    pos: content_start,
                });
            }
            None => (Operator::Simple, 0),
        }
    } else {
        return Err(Error::EmptyExpression { pos: close_pos });
    };

    let remaining = &content[offset..];
    let remaining_start = content_start + offset;
    if remaining.is_empty() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<&str>::new();
    let mut segment_offset = 0;

    for segment in remaining.split(',') {
        let start = remaining_start + segment_offset;
        if segment.is_empty() {
            return Err(Error::EmptyVariable { pos: start });
        }

        let segment_bytes = segment.as_bytes();
        let modifier_offset = segment_bytes
            .iter()
            .position(|&byte| byte == b'*' || byte == b':')
            .unwrap_or(segment_bytes.len());

        let name = &segment[..modifier_offset];
        if name.is_empty() {
            return Err(Error::InvalidName { pos: start });
        }

        let name_bytes = name.as_bytes();
        if !is_name_start(name_bytes[0]) {
            return Err(Error::InvalidName { pos: start });
        }
        if let Some(index) = name_bytes.iter().position(|&byte| !is_name_continue(byte)) {
            return Err(Error::InvalidName { pos: start + index });
        }

        let suffix = &segment[modifier_offset..];
        let modifier = if suffix.is_empty() {
            Modifier::None
        } else if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            let valid = !digits.is_empty()
                && digits.len() <= 4
                && digits.as_bytes()[0].is_ascii_digit()
                && digits.as_bytes()[0] != b'0'
                && digits.bytes().all(|byte| byte.is_ascii_digit());
            let value = if valid {
                digits.parse::<usize>().ok()
            } else {
                None
            };
            match value {
                Some(value) if (1..=9999).contains(&value) => Modifier::Prefix(value),
                _ => {
                    return Err(Error::InvalidModifier {
                        pos: start + modifier_offset,
                    });
                }
            }
        } else {
            return Err(Error::InvalidModifier {
                pos: start + modifier_offset,
            });
        };

        if seen.contains(&name) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        seen.push(name);
        specs.push(Specifier {
            name,
            start,
            modifier,
        });

        segment_offset += segment.len() + 1;
    }

    Ok((operator, specs))
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> u8 {
    match value {
        0..=9 => b'0' + value,
        _ => b'A' + (value - 10),
    }
}

fn encode(input: &str, allow_reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut output = String::with_capacity(bytes.len());
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];
        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) =
                (hex_value(bytes[index + 1]), hex_value(bytes[index + 2]))
            {
                output.push('%');
                output.push(hex_digit(high) as char);
                output.push(hex_digit(low) as char);
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(hex_digit(byte >> 4) as char);
            output.push(hex_digit(byte & 0x0f) as char);
        }
        index += 1;
    }

    output
}

fn lookup<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn encode_named(name: &str, value: String, operator: Operator) -> String {
    match operator {
        Operator::Semicolon if value.is_empty() => name.to_owned(),
        _ => format!("{name}={value}"),
    }
}

fn expand_expression(
    operator: Operator,
    specs: &[Specifier<'_>],
    vars: &[(&str, Value)],
) -> Result<String, Error> {
    let allow_reserved = matches!(operator, Operator::Plus | Operator::Hash);
    let mut fields = Vec::<String>::new();

    for spec in specs {
        let Some(value) = lookup(spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_))
            && !matches!(value, Value::Scalar(_))
        {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        let explode = matches!(spec.modifier, Modifier::Explode);
        match value {
            Value::Scalar(scalar) => {
                let value = match spec.modifier {
                    Modifier::Prefix(count) => scalar.chars().take(count).collect::<String>(),
                    _ => scalar.clone(),
                };
                let encoded = encode(&value, allow_reserved);
                let field = if matches!(
                    operator,
                    Operator::Semicolon | Operator::Question | Operator::Ampersand
                ) {
                    encode_named(spec.name, encoded, operator)
                } else {
                    encoded
                };
                fields.push(field);
            }
            Value::List(items) => {
                if explode {
                    for item in items {
                        let encoded = encode(item, allow_reserved);
                        fields.push(if matches!(
                            operator,
                            Operator::Semicolon | Operator::Question | Operator::Ampersand
                        ) {
                            encode_named(spec.name, encoded, operator)
                        } else {
                            encoded
                        });
                    }
                } else if !items.is_empty() {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, allow_reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if matches!(
                        operator,
                        Operator::Semicolon | Operator::Question | Operator::Ampersand
                    ) {
                        encode_named(spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.contains(&key.as_str()) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                    keys.push(key);
                }

                if explode {
                    for (key, value) in entries {
                        let encoded_key = encode(key, allow_reserved);
                        let encoded_value = encode(value, allow_reserved);
                        fields.push(if operator == Operator::Semicolon && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        });
                    }
                } else if !entries.is_empty() {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, allow_reserved));
                        components.push(encode(value, allow_reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if matches!(
                        operator,
                        Operator::Semicolon | Operator::Question | Operator::Ampersand
                    ) {
                        encode_named(spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let (prefix, separator) = match operator {
        Operator::Simple | Operator::Plus => ("", ","),
        Operator::Hash => ("#", ","),
        Operator::Dot => (".", "."),
        Operator::Slash => ("/", "/"),
        Operator::Semicolon => (";", ";"),
        Operator::Question => ("?", "&"),
        Operator::Ampersand => ("&", "&"),
    };

    Ok(format!("{prefix}{}", fields.join(separator)))
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<(usize, usize, Operator, Vec<Specifier<'_>>)>::new();
    let mut index = 0;

    while index < bytes.len() {
        match bytes[index] {
            b'}' => return Err(Error::UnexpectedClose { pos: index }),
            b'{' => {
                let open_pos = index;
                let Some(relative_close) = bytes[index + 1..].iter().position(|&byte| byte == b'}')
                else {
                    return Err(Error::Unclosed { pos: open_pos });
                };
                let close_pos = index + 1 + relative_close;
                if let Some(relative_open) =
                    bytes[index + 1..close_pos].iter().position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: index + 1 + relative_open,
                    });
                }

                let content_start = open_pos + 1;
                let content = &template[content_start..close_pos];
                let (operator, specs) =
                    parse_expression(content, content_start, close_pos)?;
                expressions.push((open_pos, close_pos + 1, operator, specs));
                index = close_pos + 1;
            }
            _ => index += 1,
        }
    }

    let mut output = String::with_capacity(template.len());
    let mut copied_until = 0;
    for (open, end, operator, specs) in expressions {
        output.push_str(&template[copied_until..open]);
        output.push_str(&expand_expression(operator, &specs, vars)?);
        copied_until = end;
    }
    output.push_str(&template[copied_until..]);
    Ok(output)
}
