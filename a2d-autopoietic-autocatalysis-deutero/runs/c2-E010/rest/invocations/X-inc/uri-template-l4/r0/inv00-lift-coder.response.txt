#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Clone)]
struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

struct Expression {
    operator: Option<char>,
    specs: Vec<Specifier>,
}

fn name_char(c: char) -> bool {
    c.is_ascii_alphanumeric() || c == '_'
}

fn parse_expression(content: &str, content_pos: usize, close_pos: usize) -> Result<Expression, Error> {
    let mut operator = None;
    let mut body_start = 0;
    if let Some(c) = content.chars().next() {
        if "+#./;?&".contains(c) {
            operator = Some(c);
            body_start = c.len_utf8();
        } else if c.is_ascii_punctuation() && !name_char(c) {
            return Err(Error::UnsupportedOperator { pos: content_pos });
        }
    }

    if body_start == content.len() {
        return Err(Error::EmptyExpression { pos: close_pos });
    }

    let body = &content[body_start..];
    let body_pos = content_pos + body_start;
    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut offset = 0;

    for raw in body.split(',') {
        let spec_start = body_pos + offset;
        if raw.is_empty() {
            return Err(Error::EmptyVariable { pos: spec_start });
        }

        let modifier_index = raw.find(['*', ':']).unwrap_or(raw.len());
        let name = &raw[..modifier_index];
        if name.is_empty() {
            return Err(Error::InvalidName { pos: spec_start });
        }

        let mut chars = name.char_indices();
        let (first_index, first) = chars.next().unwrap();
        if !(first.is_ascii_alphabetic() || first == '_') {
            return Err(Error::InvalidName { pos: spec_start + first_index });
        }
        for (i, c) in chars {
            if !name_char(c) {
                return Err(Error::InvalidName { pos: spec_start + i });
            }
        }

        let suffix = &raw[modifier_index..];
        let modifier = if suffix.is_empty() {
            Modifier::None
        } else if suffix == "*" {
            Modifier::Explode
        } else if let Some(digits) = suffix.strip_prefix(':') {
            if digits.is_empty()
                || digits.len() > 4
                || !digits.bytes().all(|b| b.is_ascii_digit())
                || digits.as_bytes()[0] == b'0'
            {
                return Err(Error::InvalidModifier { pos: spec_start + modifier_index });
            }
            let n = digits.parse::<usize>().unwrap_or(0);
            if !(1..=9999).contains(&n) {
                return Err(Error::InvalidModifier { pos: spec_start + modifier_index });
            }
            Modifier::Prefix(n)
        } else {
            return Err(Error::InvalidModifier { pos: spec_start + modifier_index });
        };

        if seen.iter().any(|s| s == name) {
            return Err(Error::DuplicateVariable { pos: spec_start });
        }
        seen.push(name.to_owned());
        specs.push(Specifier {
            name: name.to_owned(),
            start: spec_start,
            modifier,
        });
        offset += raw.len() + 1;
    }

    Ok(Expression { operator, specs })
}

fn parse_template(template: &str) -> Result<Vec<(usize, usize, Expression)>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(rel_close) = bytes[i + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + rel_close;
                if let Some(inner) = bytes[i + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen { pos: i + 1 + inner });
                }
                let content = &template[i + 1..close];
                let expression = parse_expression(content, open + 1, close)?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }
    Ok(expressions)
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(b: u8) -> bool {
    matches!(
        b,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' |
        b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'a'..=b'f' => Some(b - b'a' + 10),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(n: u8) -> u8 {
    if n < 10 { b'0' + n } else { b'A' + n - 10 }
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut out = Vec::with_capacity(bytes.len());
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' && i + 2 < bytes.len()
            && hex_value(bytes[i + 1]).is_some()
            && hex_value(bytes[i + 2]).is_some()
        {
            out.push(b'%');
            out.push(hex_digit(hex_value(bytes[i + 1]).unwrap()));
            out.push(hex_digit(hex_value(bytes[i + 2]).unwrap()));
            i += 3;
        } else if is_unreserved(b) || (reserved && is_reserved(b)) {
            out.push(b);
            i += 1;
        } else {
            out.push(b'%');
            out.push(hex_digit(b >> 4));
            out.push(hex_digit(b & 0x0f));
            i += 1;
        }
    }
    String::from_utf8(out).unwrap_or_default()
}

fn selected_value<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter().rev().find(|(key, _)| *key == name).map(|(_, value)| value)
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let op = expression.operator.unwrap_or('\0');
    let reserved = matches!(op, '+' | '#');
    let separator = match op {
        '.' | '/' => op,
        ';' | '?' | '&' => '&',
        _ => ',',
    };

    let mut fields = Vec::<String>::new();

    for spec in &expression.specs {
        let Some(value) = selected_value(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) && !matches!(value, Value::Scalar(_)) {
            return Err(Error::TypeMismatch { pos: spec.start });
        }

        match value {
            Value::Scalar(s) => {
                let limited;
                let value = match spec.modifier {
                    Modifier::Prefix(n) => {
                        let end = s.char_indices().nth(n).map(|(i, _)| i).unwrap_or(s.len());
                        limited = &s[..end];
                        limited
                    }
                    _ => s.as_str(),
                };
                let encoded = encode(value, reserved);
                fields.push(named_field(op, &spec.name, encoded));
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }
                let explode = matches!(spec.modifier, Modifier::Explode);
                if explode {
                    for item in items {
                        fields.push(named_field(op, &spec.name, encode(item, reserved)));
                    }
                } else {
                    let joined = items.iter().map(|s| encode(s, reserved)).collect::<Vec<_>>().join(",");
                    fields.push(named_field(op, &spec.name, joined));
                }
            }
            Value::Map(entries) => {
                let mut keys = Vec::<&str>::new();
                for (key, _) in entries {
                    if keys.iter().any(|seen| *seen == key) {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                    keys.push(key);
                }
                if entries.is_empty() {
                    continue;
                }
                let explode = matches!(spec.modifier, Modifier::Explode);
                if explode {
                    for (key, val) in entries {
                        let k = encode(key, reserved);
                        let v = encode(val, reserved);
                        let field = if op == ';' && v.is_empty() {
                            k
                        } else {
                            format!("{k}={v}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, val) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(val, reserved));
                    }
                    fields.push(named_field(op, &spec.name, components.join(",")));
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }
    let joined = fields.join(&separator.to_string());
    let prefix = match op {
        '#' | '.' | '/' | ';' => op,
        '?' | '&' => op,
        _ => '\0',
    };
    if prefix == '\0' {
        Ok(joined)
    } else {
        Ok(format!("{prefix}{joined}"))
    }
}

fn named_field(op: char, name: &str, value: String) -> String {
    match op {
        ';' if value.is_empty() => name.to_owned(),
        ';' | '?' | '&' => format!("{name}={value}"),
        _ => value,
    }
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let expressions = parse_template(template)?;
    if expressions.is_empty() {
        return Ok(template.to_owned());
    }

    let mut out = String::with_capacity(template.len());
    let mut cursor = 0;
    for (start, end, expression) in expressions {
        out.push_str(&template[cursor..start]);
        out.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }
    out.push_str(&template[cursor..]);
    Ok(out)
}
