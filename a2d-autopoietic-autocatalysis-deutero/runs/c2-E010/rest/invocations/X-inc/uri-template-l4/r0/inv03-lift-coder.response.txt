#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
    InvalidModifier { pos: usize },
    TypeMismatch { pos: usize },
    DuplicateKey { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Value {
    Scalar(String),
    List(Vec<String>),
    Map(Vec<(String, String)>),
}

#[derive(Debug, Clone)]
struct Specifier {
    name: String,
    start: usize,
    modifier: Modifier,
}

#[derive(Debug, Clone, Copy)]
enum Modifier {
    None,
    Explode,
    Prefix(usize),
}

#[derive(Debug, Clone, Copy)]
enum Operator {
    Simple,
    Reserved,
    Fragment,
    Dot,
    Slash,
    Semicolon,
    Query,
    Continuation,
}

impl Operator {
    fn separator(self) -> char {
        match self {
            Operator::Simple | Operator::Reserved | Operator::Fragment => ',',
            Operator::Dot => '.',
            Operator::Slash => '/',
            Operator::Semicolon => ';',
            Operator::Query | Operator::Continuation => '&',
        }
    }

    fn prefix(self) -> Option<char> {
        match self {
            Operator::Simple | Operator::Reserved => None,
            Operator::Fragment => Some('#'),
            Operator::Dot => Some('.'),
            Operator::Slash => Some('/'),
            Operator::Semicolon => Some(';'),
            Operator::Query => Some('?'),
            Operator::Continuation => Some('&'),
        }
    }

    fn reserved(self) -> bool {
        matches!(self, Operator::Reserved | Operator::Fragment)
    }

    fn named(self) -> bool {
        matches!(
            self,
            Operator::Semicolon | Operator::Query | Operator::Continuation
        )
    }
}

#[derive(Debug, Clone)]
struct Expression {
    operator: Operator,
    specs: Vec<Specifier>,
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn parse_expression(template: &str, open: usize, close: usize) -> Result<Expression, Error> {
    let bytes = template.as_bytes();
    let mut content_start = open + 1;
    let operator = if content_start < close {
        let op = match bytes[content_start] {
            b'+' => Some(Operator::Reserved),
            b'#' => Some(Operator::Fragment),
            b'.' => Some(Operator::Dot),
            b'/' => Some(Operator::Slash),
            b';' => Some(Operator::Semicolon),
            b'?' => Some(Operator::Query),
            b'&' => Some(Operator::Continuation),
            _ => None,
        };
        if let Some(op) = op {
            content_start += 1;
            op
        } else {
            Operator::Simple
        }
    } else {
        Operator::Simple
    };

    if content_start == close {
        return Err(Error::EmptyExpression { pos: close });
    }

    let mut specs = Vec::new();
    let mut seen = Vec::<String>::new();
    let mut start = content_start;

    loop {
        let end = bytes[start..close]
            .iter()
            .position(|&b| b == b',')
            .map(|offset| start + offset)
            .unwrap_or(close);

        if start == end {
            return Err(Error::EmptyVariable { pos: start });
        }

        let mut modifier_at = end;
        for i in start..end {
            if bytes[i] == b'*' || bytes[i] == b':' {
                modifier_at = i;
                break;
            }
        }

        if modifier_at == start {
            return Err(Error::InvalidName { pos: start });
        }

        if !is_name_start(bytes[start]) {
            return Err(Error::InvalidName { pos: start });
        }

        for i in start + 1..modifier_at {
            if !is_name_continue(bytes[i]) {
                return Err(Error::InvalidName { pos: i });
            }
        }

        let name = template[start..modifier_at].to_owned();
        let modifier = if modifier_at == end {
            Modifier::None
        } else if bytes[modifier_at] == b'*' && modifier_at + 1 == end {
            Modifier::Explode
        } else if bytes[modifier_at] == b':' {
            let digits = &bytes[modifier_at + 1..end];
            let valid = !digits.is_empty()
                && digits.len() <= 4
                && digits[0].is_ascii_digit()
                && digits[0] != b'0'
                && digits.iter().all(u8::is_ascii_digit);
            if !valid {
                return Err(Error::InvalidModifier { pos: modifier_at });
            }
            let mut value = 0usize;
            for &digit in digits {
                value = value * 10 + (digit - b'0') as usize;
            }
            if !(1..=9999).contains(&value) {
                return Err(Error::InvalidModifier { pos: modifier_at });
            }
            Modifier::Prefix(value)
        } else {
            return Err(Error::InvalidModifier { pos: modifier_at });
        };

        if seen.iter().any(|existing| existing == &name) {
            return Err(Error::DuplicateVariable { pos: start });
        }
        seen.push(name.clone());
        specs.push(Specifier {
            name,
            start,
            modifier,
        });

        if end == close {
            break;
        }
        start = end + 1;
    }

    Ok(Expression { operator, specs })
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn upper_hex(value: u8) -> u8 {
    if value < 10 {
        b'0' + value
    } else {
        b'A' + (value - 10)
    }
}

fn encode(input: &str, reserved: bool) -> String {
    let bytes = input.as_bytes();
    let mut output = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];
        if byte == b'%' && i + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[i + 1]), hex_value(bytes[i + 2])) {
                output.extend_from_slice(&[b'%', upper_hex(high), upper_hex(low)]);
                i += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte);
        } else {
            output.extend_from_slice(&[b'%', upper_hex(byte >> 4), upper_hex(byte & 0x0f)]);
        }
        i += 1;
    }

    String::from_utf8(output).expect("percent encoding emits ASCII")
}

fn select_binding<'a>(name: &str, vars: &'a [(&str, Value)]) -> Option<&'a Value> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| value)
}

fn named_field(name: &str, value: String, operator: Operator) -> String {
    match operator {
        Operator::Semicolon if value.is_empty() => name.to_owned(),
        _ => {
            let mut field = String::with_capacity(name.len() + value.len() + 1);
            field.push_str(name);
            field.push('=');
            field.push_str(&value);
            field
        }
    }
}

fn expand_expression(expression: &Expression, vars: &[(&str, Value)]) -> Result<String, Error> {
    let operator = expression.operator;
    let mut fields = Vec::<String>::new();

    for spec in &expression.specs {
        let Some(value) = select_binding(&spec.name, vars) else {
            continue;
        };

        if matches!(spec.modifier, Modifier::Prefix(_)) {
            match value {
                Value::Scalar(_) => {}
                Value::List(_) | Value::Map(_) => {
                    return Err(Error::TypeMismatch { pos: spec.start });
                }
            }
        }

        let reserved = operator.reserved();

        match value {
            Value::Scalar(value) => {
                let selected = match spec.modifier {
                    Modifier::Prefix(n) => value.chars().take(n).collect::<String>(),
                    Modifier::None | Modifier::Explode => value.clone(),
                };
                fields.push(if operator.named() {
                    named_field(&spec.name, encode(&selected, reserved), operator)
                } else {
                    encode(&selected, reserved)
                });
            }
            Value::List(items) => {
                if items.is_empty() {
                    continue;
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for item in items {
                        let encoded = encode(item, reserved);
                        fields.push(if operator.named() {
                            named_field(&spec.name, encoded, operator)
                        } else {
                            encoded
                        });
                    }
                } else {
                    let joined = items
                        .iter()
                        .map(|item| encode(item, reserved))
                        .collect::<Vec<_>>()
                        .join(",");
                    fields.push(if operator.named() {
                        named_field(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
            Value::Map(entries) => {
                if entries.is_empty() {
                    continue;
                }

                for i in 0..entries.len() {
                    if entries[..i]
                        .iter()
                        .any(|(key, _)| key == &entries[i].0)
                    {
                        return Err(Error::DuplicateKey { pos: spec.start });
                    }
                }

                if matches!(spec.modifier, Modifier::Explode) {
                    for (key, value) in entries {
                        let encoded_key = encode(key, reserved);
                        let encoded_value = encode(value, reserved);
                        let field = if operator == Operator::Semicolon && encoded_value.is_empty() {
                            encoded_key
                        } else {
                            format!("{encoded_key}={encoded_value}")
                        };
                        fields.push(field);
                    }
                } else {
                    let mut components = Vec::with_capacity(entries.len() * 2);
                    for (key, value) in entries {
                        components.push(encode(key, reserved));
                        components.push(encode(value, reserved));
                    }
                    let joined = components.join(",");
                    fields.push(if operator.named() {
                        named_field(&spec.name, joined, operator)
                    } else {
                        joined
                    });
                }
            }
        }
    }

    if fields.is_empty() {
        return Ok(String::new());
    }

    let mut output = String::new();
    if let Some(prefix) = operator.prefix() {
        output.push(prefix);
    }
    output.push_str(&fields.join(&operator.separator().to_string()));
    Ok(output)
}

pub fn expand(template: &str, vars: &[(&str, Value)]) -> Result<String, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::<(usize, usize, Expression)>::new();
    let mut i = 0;

    while i < bytes.len() {
        match bytes[i] {
            b'}' => return Err(Error::UnexpectedClose { pos: i }),
            b'{' => {
                let open = i;
                let Some(offset) = bytes[i + 1..].iter().position(|&byte| byte == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = i + 1 + offset;

                if let Some(inner_offset) = bytes[open + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + inner_offset,
                    });
                }

                let expression = parse_expression(template, open, close)?;
                expressions.push((open, close + 1, expression));
                i = close + 1;
            }
            _ => i += 1,
        }
    }

    let mut output = String::new();
    let mut cursor = 0;
    for (start, end, expression) in expressions {
        output.push_str(&template[cursor..start]);
        output.push_str(&expand_expression(&expression, vars)?);
        cursor = end;
    }
    output.push_str(&template[cursor..]);
    Ok(output)
}
