fn apply_event(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    if account.version == u64::MAX {
        return Err(ApplyError::VersionOverflow);
    }

    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.status = Status::Open;
            account.limit = *limit;
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.status = Status::Closed;
        }
    }

    account.version = account.version + 1;
    Ok(())
}

    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Err(0);
    }
    if !(b'a'..=b'z').contains(&bytes[0]) {

            journal: BTreeMap::new(),
            position: 0,
            accounts: BTreeMap::new(),
        }

    pub fn ingest(&mut self, records: &[Record]) -> Result<usize, ProjectionError> {
        let mut provisional = self.clone();
        let mut accepted = 0;

        for (index, record) in records.iter().enumerate() {
            if let Err(byte) = validate_id(&record.stream) {
                return Err(ProjectionError::InvalidId { index, byte });
            }
            if record.position == 0 {
                return Err(ProjectionError::ZeroPosition { index });
            }

            if record.position <= provisional.position {
                if provisional.journal.get(&record.position) != Some(record) {
                    return Err(ProjectionError::DuplicateMismatch {
                        index,
                        position: record.position,
                    });
                }
                continue;
            }

            let expected_position = provisional.position + 1;
            if record.position != expected_position {
                return Err(ProjectionError::Gap {
                    index,
                    expected: expected_position,
                    got: record.position,
                });
            }

            let mut account = provisional
                .accounts
                .get(&record.stream)
                .cloned()
                .unwrap_or_else(initial_account);
            let expected_version = account.version + 1;
            if record.version != expected_version {
                return Err(ProjectionError::StreamVersion {
                    index,
                    expected: expected_version,
                    got: record.version,
                });
            }

            if let Err(kind) = apply_event(&mut account, &record.event) {
                return Err(ProjectionError::Rejected { index, kind });
            }

            provisional
                .accounts
                .insert(record.stream.clone(), account);
            provisional
                .journal
                .insert(record.position, record.clone());
            provisional.position = record.position;
            accepted += 1;
        }

        *self = provisional;
        Ok(accepted)
    }

    pub fn account(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.accounts.get(id).cloned())
    }

    pub fn position(&self) -> u64 {
        self.position
    }

    pub fn totals(&self) -> Totals {
        let mut totals = Totals {
            streams: self.accounts.len() as u64,
            open: 0,
            closed: 0,
            balance: 0,
        };
        for account in self.accounts.values() {
            match account.status {
                Status::Open => totals.open += 1,
                Status::Closed => totals.closed += 1,
                Status::New => {}
            }
            totals.balance += account.balance as i128;
        }
        totals
    }

fn apply_event(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    let next_version = account
        .version
        .checked_add(1)
        .ok_or(ApplyError::VersionOverflow)?;

    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.status = Status::Open;
            account.limit = *limit;
            account.balance = 0;
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.status = Status::Closed;
        }
    }

    account.version = next_version;
    Ok(())
}

        if snapshot_every == 0 {
            return Err(SnapshotError::ZeroInterval);
        }
        Ok(Self {
            snapshot_every,
            streams: BTreeMap::new(),
            records: Vec::new(),
        })
    }

    pub fn append(
        &mut self,
        id: &str,
        expected: Expected,
        events: &[Event],
    ) -> Result<Account, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;

        let existing = self.streams.get(id);
        let actual = existing.map(|stream| stream.account.version);
        let matches = match &expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(version) => actual == Some(*version),
        };
        if !matches {
            return Err(StoreError::Conflict { expected, actual });
        }

        let mut account = existing
            .map(|stream| stream.account.clone())
            .unwrap_or_else(initial_account);
        let mut applied = Vec::with_capacity(events.len());
        let mut latest_snapshot = existing.and_then(|stream| stream.snapshot.clone());

        for (index, event) in events.iter().enumerate() {
            apply_event(&mut account, event).map_err(|kind| {
                StoreError::Rejected(ReplayError { index, kind })
            })?;
            applied.push((event.clone(), account.clone()));
            if account.version % self.snapshot_every == 0 {
                latest_snapshot = Some(Snapshot {
                    tag: snapshot_tag(&account),
                    account: account.clone(),
                });
            }
        }

        if events.is_empty() {
            return Ok(account);
        }

        let stream = self
            .streams
            .entry(id.to_owned())
            .or_insert_with(|| StreamData {
                account: initial_account(),
                history: Vec::new(),
                snapshot: None,
            });

        for (event, intermediate_account) in applied {
            let position = self.records.len() as u64 + 1;
            let version = intermediate_account.version;
            self.records.push(Record {
                position,
                stream: id.to_owned(),
                version,
                event: event.clone(),
            });
            stream.history.push(event);
        }
        stream.account = account.clone();
        stream.snapshot = latest_snapshot;
        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|stream| stream.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|stream| {
            stream
                .history
                .iter()
                .enumerate()
                .filter(|(index, _)| (*index as u64) + 1 > after)
                .map(|(_, event)| event.clone())
                .collect()
        }))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self
            .streams
            .get(id)
            .and_then(|stream| stream.snapshot.clone()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.streams.get(id).map(|stream| {
            let start = stream
                .snapshot
                .as_ref()
                .map(|snapshot| snapshot.account.version as usize)
                .unwrap_or(0);
            stream.history[start..].to_vec()
        }))
    }

    pub fn records(&self, after: u64) -> Vec<Record> {
        self.records
            .iter()
            .filter(|record| record.position > after)
            .cloned()
            .collect()
    }
}

fn validate_id(id: &str) -> Result<(), usize> {
    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Err(0);
    }
    for (index, byte) in bytes.iter().enumerate() {
        if index == 24 {
            return Err(24);
        }
        let valid = if index == 0 {
            byte.is_ascii_lowercase()
        } else {
            byte.is_ascii_lowercase() || byte.is_ascii_digit() || *byte == b'-'
        };
        if !valid {
            return Err(index);
        }
    }
    if bytes.len() > 24 {
        return Err(24);
    }
    Ok(())
}

pub fn snapshot_tag(account: &Account) -> u64 {
    let status_code = match account.status {
        Status::New => 0,
        Status::Open => 0xA5,
        Status::Closed => 0x5A,
    };
    account.version.rotate_left(7)
        ^ (account.limit as u64).wrapping_mul(31)
        ^ (account.balance as u64).rotate_left(17)
        ^ status_code
}

fn validate_snapshot(snapshot: &Snapshot) -> Result<(), SnapshotError> {
    let account = &snapshot.account;
    if account.version == 0 {
        return Err(SnapshotError::InvalidVersion);
    }
    if account.status == Status::New
        || !(1..=1_000_000_000).contains(&account.limit)
        || account.balance < 0
        || account.balance > account.limit
        || (account.status == Status::Closed && (account.balance != 0 || account.version < 2))
    {
        return Err(SnapshotError::InvalidState);
    }
    if snapshot.tag != snapshot_tag(account) {
        return Err(SnapshotError::TagMismatch);
    }
    Ok(())
}

pub fn rebuild(snapshot: Option<&Snapshot>, tail: &[Event]) -> Result<Account, RestoreError> {
    let mut account = match snapshot {
        None => return replay(tail).map_err(RestoreError::Tail),
        Some(snapshot) => {
            validate_snapshot(snapshot).map_err(RestoreError::Snapshot)?;
            snapshot.account.clone()
        }
    };

    for (index, event) in tail.iter().enumerate() {
        apply_event(&mut account, event)
            .map_err(|kind| RestoreError::Tail(ReplayError { index, kind }))?;
    }
    Ok(account)
}

#[derive(Debug, Clone, PartialEq, Eq)]

    journal: Vec<Record>,
    position: u64,

}

impl Projection {
    pub fn new() -> Self {
        Self {
            journal: Vec::new(),
            position: 0,
            accounts: BTreeMap::new(),
        }
    }

    pub fn ingest(&mut self, records: &[Record]) -> Result<usize, ProjectionError> {
        let mut provisional = self.clone();
        let mut accepted = 0;

        for (index, record) in records.iter().enumerate() {
            validate_id(&record.stream).map_err(|byte| ProjectionError::InvalidId {
                index,
                byte,
            })?;
            if record.position == 0 {
                return Err(ProjectionError::ZeroPosition { index });
            }

            if record.position <= provisional.position {
                let stored = &provisional.journal[(record.position - 1) as usize];
                if stored != record {
                    return Err(ProjectionError::DuplicateMismatch {
                        index,
                        position: record.position,
                    });
                }
                continue;
            }

            let expected_position = provisional.position + 1;
            if record.position != expected_position {
                return Err(ProjectionError::Gap {
                    index,
                    expected: expected_position,
                    got: record.position,
                });
            }

            let account = provisional
                .accounts
                .get(&record.stream)
                .cloned()
                .unwrap_or_else(initial_account);
            let expected_version = account.version + 1;
            if record.version != expected_version {
                return Err(ProjectionError::StreamVersion {
                    index,
                    expected: expected_version,
                    got: record.version,
                });
            }

            let mut updated = account;
            apply_event(&mut updated, &record.event)
                .map_err(|kind| ProjectionError::Rejected { index, kind })?;
            provisional.accounts.insert(record.stream.clone(), updated);
            provisional.journal.push(record.clone());
            provisional.position = record.position;
            accepted += 1;
        }

        *self = provisional;
        Ok(accepted)
    }

    pub fn account(&self, id: &str) -> Result<Option<Account>, StoreError> {
        validate_id(id).map_err(|byte| StoreError::InvalidId { byte })?;
        Ok(self.accounts.get(id).cloned())
    }

    pub fn position(&self) -> u64 {
        self.position
    }

    pub fn totals(&self) -> Totals {
        let mut totals = Totals {
            streams: self.accounts.len() as u64,
            open: 0,
            closed: 0,
            balance: 0,
        };
        for account in self.accounts.values() {
            match account.status {
                Status::Open => totals.open += 1,
                Status::Closed => totals.closed += 1,
                Status::New => {}
            }
            totals.balance += account.balance as i128;
        }
        totals
    }
}

impl Default for Projection {
    fn default() -> Self {
        Self::new()
    }
}

fn initial_account() -> Account {
    Account {
        status: Status::New,
        balance: 0,

fn id_error(id: &str) -> Option<usize> {
    let bytes = id.as_bytes();
    if bytes.is_empty() {
        return Some(0);
    }
    if !(b'a'..=b'z').contains(&bytes[0]) {
        return Some(0);
    }
    for (i, &byte) in bytes.iter().enumerate().skip(1) {
        if !((b'a'..=b'z').contains(&byte)
            || (b'0'..=b'9').contains(&byte)
            || byte == b'-')
        {
            return Some(i);
        }
    }
    if bytes.len() > 24 {
        return Some(24);
    }
    None
}

fn apply_event(account: &mut Account, event: &Event) -> Result<(), ApplyError> {
    let next_version = account
        .version
        .checked_add(1)
        .ok_or(ApplyError::VersionOverflow)?;

    match event {
        Event::Open { limit } => {
            if !(1..=1_000_000_000).contains(limit) {
                return Err(ApplyError::InvalidLimit);
            }
            if account.status != Status::New {
                return Err(ApplyError::AlreadyOpened);
            }
            account.status = Status::Open;
            account.limit = *limit;
            account.balance = 0;
        }
        Event::Deposit { amount } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *amount > account.limit - account.balance {
                return Err(ApplyError::LimitExceeded);
            }
            account.balance += *amount;
        }
        Event::Withdraw { amount, floor } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *amount <= 0 {
                return Err(ApplyError::InvalidAmount);
            }
            if *floor < 0 || *floor > account.limit {
                return Err(ApplyError::InvalidFloor);
            }
            if *amount > account.balance {
                return Err(ApplyError::InsufficientFunds);
            }
            if account.balance - *amount < *floor {
                return Err(ApplyError::FloorViolation);
            }
            account.balance -= *amount;
        }
        Event::Close { expected_version } => {
            match account.status {
                Status::New => return Err(ApplyError::NotOpened),
                Status::Closed => return Err(ApplyError::Closed),
                Status::Open => {}
            }
            if *expected_version != account.version {
                return Err(ApplyError::CloseVersion {
                    expected: *expected_version,
                    actual: account.version,
                });
            }
            if account.balance != 0 {
                return Err(ApplyError::NonZeroBalance);
            }
            account.status = Status::Closed;
        }
    }

    account.version = next_version;
    Ok(())
}

fn apply_events_from(
    initial: &Account,
    events: &[Event],
) -> Result<Account, ReplayError> {
    let mut account = initial.clone();
    for (index, event) in events.iter().enumerate() {
        if let Err(kind) = apply_event(&mut account, event) {
            return Err(ReplayError { index, kind });
        }
    }
    Ok(account)
}

pub fn replay(events: &[Event]) -> Result<Account, ReplayError> {
    apply_events_from(&initial_account(), events)
}

impl Store {
    pub fn new(snapshot_every: u64) -> Result<Self, SnapshotError> {
        if snapshot_every == 0 {
            return Err(SnapshotError::ZeroInterval);
        }
        Ok(Self {
            snapshot_every,
            streams: std::collections::BTreeMap::new(),
            records: Vec::new(),
        })
    }

    pub fn append(
        &mut self,
        id: &str,
        expected: Expected,
        events: &[Event],
    ) -> Result<Account, StoreError> {
        if let Some(byte) = id_error(id) {
            return Err(StoreError::InvalidId { byte });
        }

        let existing = self.streams.get(id);
        let actual = existing.map(|stream| stream.account.version);
        let matches = match expected {
            Expected::Any => true,
            Expected::Missing => actual.is_none(),
            Expected::Exact(version) => actual == Some(version),
        };
        if !matches {
            return Err(StoreError::Conflict { expected, actual });
        }

        let base = existing
            .map(|stream| stream.account.clone())
            .unwrap_or_else(initial_account);

        if events.is_empty() {
            return Ok(base);
        }

        let mut account = base.clone();
        let mut snapshots = Vec::new();

        for (index, event) in events.iter().enumerate() {
            if let Err(kind) = apply_event(&mut account, event) {
                return Err(StoreError::Rejected(ReplayError { index, kind }));
            }
            if account.version % self.snapshot_every == 0 {
                snapshots.push(Snapshot {
                    tag: snapshot_tag(&account),
                    account: account.clone(),
                });
            }
        }

        let stream = self
            .streams
            .entry(id.to_owned())
            .or_insert_with(|| Stream {
                account: initial_account(),
                events: Vec::new(),
                snapshot: None,
            });

        stream.events.extend(events.iter().cloned());
        stream.account = account.clone();
        if let Some(snapshot) = snapshots.pop() {
            stream.snapshot = Some(snapshot);
        }

        for event in events {
            let position = self.records.len() as u64 + 1;
            let version = stream.events.len() as u64 - events.len() as u64
                + (self.records.iter().filter(|r| r.stream == id).count() as u64)
                + 1;
            // The stream's prior version is the first event's predecessor.
            let version = version.min(stream.account.version);
            self.records.push(Record {
                position,
                stream: id.to_owned(),
                version,
                event: event.clone(),
            });
        }

        // Set versions of newly appended records from the existing history length.
        let first_new_index = self.records.len() - events.len();
        let previous_version = base.version;
        for (offset, record) in self.records[first_new_index..].iter_mut().enumerate() {
            record.version = previous_version + offset as u64 + 1;
        }

        Ok(account)
    }

    pub fn state(&self, id: &str) -> Result<Option<Account>, StoreError> {
        if let Some(byte) = id_error(id) {
            return Err(StoreError::InvalidId { byte });
        }
        Ok(self.streams.get(id).map(|stream| stream.account.clone()))
    }

    pub fn history(&self, id: &str, after: u64) -> Result<Option<Vec<Event>>, StoreError> {
        if let Some(byte) = id_error(id) {
            return Err(StoreError::InvalidId { byte });
        }
        Ok(self.streams.get(id).map(|stream| {
            stream
                .events
                .iter()
                .enumerate()
                .filter(|(index, _)| (*index as u64 + 1) > after)
                .map(|(_, event)| event.clone())
                .collect()
        }))
    }

    pub fn snapshot(&self, id: &str) -> Result<Option<Snapshot>, StoreError> {
        if let Some(byte) = id_error(id) {
            return Err(StoreError::InvalidId { byte });
        }
        Ok(self
            .streams
            .get(id)
            .and_then(|stream| stream.snapshot.clone()))
    }

    pub fn tail(&self, id: &str) -> Result<Option<Vec<Event>>, StoreError> {
        if let Some(byte) = id_error(id) {
            return Err(StoreError::InvalidId { byte });
        }
        Ok(self.streams.get(id).map(|stream| {
            let after = stream
                .snapshot
                .as_ref()
                .map(|snapshot| snapshot.account.version)
                .unwrap_or(0);
            stream
                .events
                .iter()
                .enumerate()
                .filter(|(index, _)| (*index as u64 + 1) > after)
                .map(|(_, event)| event.clone())
                .collect()
        }))
    }

    pub fn records(&self, after: u64) -> Vec<Record> {
        self.records
            .iter()
            .filter(|record| record.position > after)
            .cloned()
            .collect()
    }
}

impl Projection {
    pub fn new() -> Self {
        Self {
            journal: Vec::new(),
            position: 0,
            accounts: std::collections::BTreeMap::new(),
        }
    }

    pub fn ingest(&mut self, records: &[Record]) -> Result<usize, ProjectionError> {
        let mut provisional = self.clone();
        let mut accepted = 0usize;

        for (index, record) in records.iter().enumerate() {
            if let Some(byte) = id_error(&record.stream) {
                return Err(ProjectionError::InvalidId { index, byte });
            }
            if record.position == 0 {
                return Err(ProjectionError::ZeroPosition { index });
            }

            if record.position <= provisional.position {
                let stored = &provisional.journal[(record.position - 1) as usize];
                if stored != record {
                    return Err(ProjectionError::DuplicateMismatch {
                        index,
                        position: record.position,
                    });
                }
                continue;
            }

            let expected_position = provisional.position + 1;
            if record.position != expected_position {
                return Err(ProjectionError::Gap {
                    index,
                    expected: expected_position,
                    got: record.position,
                });
            }

            let current = provisional
                .accounts
                .get(&record.stream)
                .cloned()
                .unwrap_or_else(initial_account);
            let expected_version = current.version + 1;
            if record.version != expected_version {
                return Err(ProjectionError::StreamVersion {
                    index,
                    expected: expected_version,
                    got: record.version,
                });
            }

            let mut updated = current;
            if let Err(kind) = apply_event(&mut updated, &record.event) {
                return Err(ProjectionError::Rejected { index, kind });
            }

            provisional.journal.push(record.clone());
            provisional
                .accounts
                .insert(record.stream.clone(), updated);
            provisional.position = record.position;
            accepted += 1;
        }

        *self = provisional;
        Ok(accepted)
    }

    pub fn account(&self, id: &str) -> Result<Option<Account>, StoreError> {
        if let Some(byte) = id_error(id) {
            return Err(StoreError::InvalidId { byte });
        }
        Ok(self.accounts.get(id).cloned())
    }

    pub fn position(&self) -> u64 {
        self.position
    }

        let mut totals = Totals {
            streams: self.accounts.len() as u64,
            open: 0,
            closed: 0,
            balance: 0,
        };

                Status::New => {}
                Status::Open => totals.open += 1,
                Status::Closed => totals.closed += 1,
            }
            totals.balance += account.balance as i128;
