use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    while let Some(b) = byte(s, *pos) {
        if !is_digit(b) {
            break;
        }
        *pos += 1;
    }
    if start == *pos {
        return match byte(s, start) {
            None | Some(b'.' | b'-' | b'+') => Err(VersionError::MissingCore { at: start }),
            _ => Err(VersionError::InvalidChar { at: start }),
        };
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.starts_with('0') {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits.parse::<u64>().map_err(|_| VersionError::Overflow { at: start })
}

fn identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let mut out = Vec::new();
    let mut i = start;
    loop {
        let id_start = i;
        while i < end && s.as_bytes()[i] != b'.' {
            let b = s.as_bytes()[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if id_start == i {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..i];
        if pre && id.len() > 1 && id.starts_with('0') && id.bytes().all(is_digit) {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if i == end {
            break;
        }
        i += 1;
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = core_field(s, &mut pos)?;
    if byte(s, pos) != Some(b'.') {
        return match byte(s, pos) {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let minor = core_field(s, &mut pos)?;
    if byte(s, pos) != Some(b'.') {
        return match byte(s, pos) {
            None | Some(b'-' | b'+') => Err(VersionError::MissingCore { at: pos }),
            _ => Err(VersionError::InvalidChar { at: pos }),
        };
    }
    pos += 1;
    let patch = core_field(s, &mut pos)?;
    if let Some(b) = byte(s, pos) {
        if b != b'-' && b != b'+' {
            return Err(VersionError::InvalidChar { at: pos });
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();
    if byte(s, pos) == Some(b'-') {
        let start = pos + 1;
        let plus = s[start..].find('+').map(|n| start + n);
        let end = plus.unwrap_or(s.len());
        pre = identifiers(s, start, end, true)?;
        pos = end;
    }
    if byte(s, pos) == Some(b'+') {
        let start = pos + 1;
        build = identifiers(s, start, s.len(), false)?;
        pos = s.len();
    }
    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }
    Ok(Version { major, minor, patch, pre, build })
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(is_digit)
}

fn cmp_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => return Ordering::Equal,
        (true, false) => return Ordering::Greater,
        (false, true) => return Ordering::Less,
        _ => {}
    }
    for (x, y) in a.iter().zip(b) {
        let xn = numeric_id(x);
        let yn = numeric_id(y);
        let c = match (xn, yn) {
            (true, true) => x.len().cmp(&y.len()).then_with(|| x.as_bytes().cmp(y.as_bytes())),
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => x.as_bytes().cmp(y.as_bytes()),
        };
        if c != Ordering::Equal {
            return c;
        }
    }
    a.len().cmp(&b.len())
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| cmp_pre(&a.pre, &b.pre))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Bound {
    version: Version,
    inclusive: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    lower: Option<Bound>,
    upper: Option<Bound>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

fn version_tuple(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn version_at(v: &Version, tuple: (u64, u64, u64), pre: Vec<String>) -> Version {
    Version {
        major: tuple.0,
        minor: tuple.1,
        patch: tuple.2,
        pre,
        build: Vec::new(),
    }
}

fn parse_full_at(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

#[derive(Clone)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
}

fn partial_operand(s: &str, start: usize) -> Result<Partial, RangeError> {
    let has_suffix = s.contains('-') || s.contains('+');
    let fields: Vec<&str> = s.split('.').collect();
    if has_suffix || (fields.len() >= 3 && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X"))) {
        let v = parse_full_at(s, start)?;
        return Ok(Partial {
            nums: vec![v.major, v.minor, v.patch],
            full: Some(v),
        });
    }
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at: start });
    }
    let mut nums = Vec::new();
    let mut wildcard_seen = false;
    for f in fields {
        if matches!(f, "*" | "x" | "X") {
            wildcard_seen = true;
        } else {
            if wildcard_seen || !f.bytes().all(is_digit) || (f.len() > 1 && f.starts_with('0')) {
                return Err(RangeError::Syntax { at: start });
            }
            match f.parse::<u64>() {
                Ok(n) => nums.push(n),
                Err(_) => return Err(RangeError::Syntax { at: start }),
            }
        }
    }
    Ok(Partial { nums, full: None })
}

fn padded(nums: &[u64]) -> (u64, u64, u64) {
    (
        nums.first().copied().unwrap_or(0),
        nums.get(1).copied().unwrap_or(0),
        nums.get(2).copied().unwrap_or(0),
    )
}

fn bump_upper(nums: &[u64], mode: char, at: usize) -> Result<Version, RangeError> {
    let k = nums.len();
    let (mut ma, mut mi, mut pa) = padded(nums);
    match mode {
        'b' => {
            if ma > 0 || k == 1 {
                ma = ma.checked_add(1).ok_or(RangeError::Syntax { at })?;
                mi = 0;
                pa = 0;
            } else if mi > 0 || k == 2 {
                mi = mi.checked_add(1).ok_or(RangeError::Syntax { at })?;
                pa = 0;
            } else {
                pa = pa.checked_add(1).ok_or(RangeError::Syntax { at })?;
            }
        }
        't' => {
            if k == 1 {
                ma = ma.checked_add(1).ok_or(RangeError::Syntax { at })?;
                mi = 0;
                pa = 0;
            } else {
                mi = mi.checked_add(1).ok_or(RangeError::Syntax { at })?;
                pa = 0;
            }
        }
        'p' => {
            if k == 1 {
                ma = ma.checked_add(1).ok_or(RangeError::Syntax { at })?;
                mi = 0;
                pa = 0;
            } else if k == 2 {
                mi = mi.checked_add(1).ok_or(RangeError::Syntax { at })?;
                pa = 0;
            } else {
                pa = pa.checked_add(1).ok_or(RangeError::Syntax { at })?;
            }
        }
        _ => unreachable!(),
    }
    Ok(version_at(&Version { major: 0, minor: 0, patch: 0, pre: vec![], build: vec![] }, (ma, mi, pa), vec!["0".into()]))
}

fn add_lower(arm: &mut Arm, v: Version, inclusive: bool) {
    if arm.lower.as_ref().map_or(true, |b| {
        let c = compare_versions(&v, &b.version);
        c == Ordering::Greater || (c == Ordering::Equal && !inclusive && b.inclusive)
    }) {
        arm.lower = Some(Bound { version: v.clone(), inclusive });
    }
    if !v.pre.is_empty() {
        arm.anchors.push(version_tuple(&v));
    }
}

fn add_upper(arm: &mut Arm, v: Version, inclusive: bool) {
    if arm.upper.as_ref().map_or(true, |b| {
        let c = compare_versions(&v, &b.version);
        c == Ordering::Less || (c == Ordering::Equal && !inclusive && b.inclusive)
    }) {
        arm.upper = Some(Bound { version: v.clone(), inclusive });
    }
    if !v.pre.is_empty() {
        arm.anchors.push(version_tuple(&v));
    }
}

fn exact_token_cmp(tok: &str, start: usize, arm: &mut Arm) -> Result<(), RangeError> {
    let (op, off) = if tok.starts_with(">=") { (">=", 2) }
        else if tok.starts_with("<=") { ("<=", 2) }
        else if tok.starts_with('>') { (">", 1) }
        else if tok.starts_with('<') { ("<", 1) }
        else if tok.starts_with('=') { ("=", 1) }
        else { ("=", 0) };
    let operand = &tok[off..];
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: start });
    }
    let v = parse_full_at(operand, start + off)?;
    match op {
        "=" => {
            add_lower(arm, v.clone(), true);
            add_upper(arm, v, true);
        }
        ">" => add_lower(arm, v, false),
        ">=" => add_lower(arm, v, true),
        "<" => add_upper(arm, v, false),
        "<=" => add_upper(arm, v, true),
        _ => unreachable!(),
    }
    Ok(())
}

fn token_positions(s: &str, base: usize) -> Vec<(&str, usize)> {
    let mut out = Vec::new();
    let bytes = s.as_bytes();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) { i += 1; }
        if i == bytes.len() { break; }
        let st = i;
        while i < bytes.len() && !is_ws(bytes[i]) { i += 1; }
        out.push((&s[st..i], base + st));
    }
    out
}

fn parse_arm(tokens: &[(&str, usize)]) -> Result<Arm, RangeError> {
    let mut arm = Arm { lower: None, upper: None, anchors: Vec::new() };
    let hyphens: Vec<usize> = tokens.iter().enumerate().filter_map(|(i, (t, _))| if *t == "-" { Some(i) } else { None }).collect();
    if !hyphens.is_empty() {
        let hi = hyphens[0];
        if hyphens.len() != 1 || tokens.len() != 3 || hi != 1 {
            return Err(RangeError::Syntax { at: tokens[hi].1 });
        }
        let (lt, ls) = tokens[0];
        let (ut, us) = tokens[2];
        let l = partial_operand(lt, ls)?;
        let u = partial_operand(ut, us)?;
        if l.nums.is_empty() { return Err(RangeError::Syntax { at: ls }); }
        if u.nums.is_empty() { return Err(RangeError::Syntax { at: us }); }
        let lv = l.full.clone().unwrap_or_else(|| version_at(&Version { major:0,minor:0,patch:0,pre:vec![],build:vec![] }, padded(&l.nums), vec![]));
        add_lower(&mut arm, lv, true);
        if let Some(uv) = u.full {
            add_upper(&mut arm, uv, true);
        } else if u.nums.len() == 3 {
            let uv = version_at(&Version { major:0,minor:0,patch:0,pre:vec![],build:vec![] }, padded(&u.nums), vec![]);
            add_upper(&mut arm, uv, true);
        } else {
            let uv = bump_upper(&u.nums, 'p', us)?;
            add_upper(&mut arm, uv, false);
        }
        return Ok(arm);
    }

    for &(tok, start) in tokens {
        if tok.starts_with(">=") || tok.starts_with("<=") || tok.starts_with('>') || tok.starts_with('<') || tok.starts_with('=') {
            exact_token_cmp(tok, start, &mut arm)?;
            continue;
        }
        let (mode, operand, off) = if tok.starts_with('^') { ('b', &tok[1..], 1) }
            else if tok.starts_with('~') { ('t', &tok[1..], 1) }
            else { (' ', tok, 0) };
        if mode != ' ' {
            if operand.is_empty() { return Err(RangeError::Syntax { at: start }); }
            let p = partial_operand(operand, start + off)?;
            if p.nums.is_empty() { return Err(RangeError::Syntax { at: start }); }
            let lower = p.full.clone().unwrap_or_else(|| version_at(&Version {major:0,minor:0,patch:0,pre:vec![],build:vec![]}, padded(&p.nums), vec![]));
            add_lower(&mut arm, lower, true);
            let upper = bump_upper(&p.nums, mode, start + off)?;
            add_upper(&mut arm, upper, false);
            continue;
        }
        if !(tok.starts_with(|c: char| c.is_ascii_digit()) || matches!(tok, "*" | "x" | "X")) {
            return Err(RangeError::Syntax { at: start });
        }
        let p = partial_operand(operand, start)?;
        if let Some(v) = p.full {
            add_lower(&mut arm, v.clone(), true);
            add_upper(&mut arm, v, true);
        } else if !p.nums.is_empty() {
            let lower = version_at(&Version {major:0,minor:0,patch:0,pre:vec![],build:vec![]}, padded(&p.nums), vec![]);
            add_lower(&mut arm, lower, true);
            if p.nums.len() < 3 {
                let upper = bump_upper(&p.nums, 'p', start)?;
                add_upper(&mut arm, upper, false);
            } else {
                add_upper(&mut arm, version_at(&Version {major:0,minor:0,patch:0,pre:vec![],build:vec![]}, padded(&p.nums), vec![]), true);
            }
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut arms = Vec::new();
    let mut start = 0;
    let mut i = 0;
    let mut previous_sep = false;
    while i < bytes.len() {
        if is_ws(bytes[i]) { i += 1; continue; }
        if bytes[i..].starts_with(b"||") {
            let before = &s[start..i];
            let ts = token_positions(before, start);
            if ts.is_empty() {
                return Err(RangeError::Empty { at: i });
            }
            arms.push(parse_arm(&ts)?);
            i += 2;
            while i < bytes.len() && is_ws(bytes[i]) { i += 1; }
            start = i;
            previous_sep = true;
            continue;
        }
        i += 1;
    }
    let tail = &s[start..];
    let ts = token_positions(tail, start);
    if ts.is_empty() {
        if arms.is_empty() { return Err(RangeError::Empty { at: 0 }); }
        if previous_sep { return Err(RangeError::Empty { at: s.len() }); }
    } else {
        arms.push(parse_arm(&ts)?);
    }
    Ok(Range { arms })
}

fn arm_admits(v: &Version, arm: &Arm) -> bool {
    arm.anchors.contains(&version_tuple(v))
}

fn within(v: &Version, arm: &Arm) -> bool {
    if let Some(l) = &arm.lower {
        let c = compare_versions(v, &l.version);
        if c == Ordering::Less || (c == Ordering::Equal && !l.inclusive) { return false; }
    }
    if let Some(u) = &arm.upper {
        let c = compare_versions(v, &u.version);
        if c == Ordering::Greater || (c == Ordering::Equal && !u.inclusive) { return false; }
    }
    true
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|arm| {
        within(v, arm) && (v.pre.is_empty() || include_prerelease || arm_admits(v, arm))
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying(candidates: &[String], r: &Range, max: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        let v = parse_version(s).map_err(|error| CandidateError { index, error })?;
        parsed.push(v);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) { continue; }
        if let Some(j) = best {
            let c = compare_versions(&parsed[i], &parsed[j]);
            if (max && c == Ordering::Greater) || (!max && c == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    satisfying(candidates, r, false)
}

pub fn intersects(_a: &Range, _b: &Range, _include_prerelease: bool) -> bool {
    false
}

fn display_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}

fn bound_text(b: &Option<Bound>, lower: bool) -> String {
    match b {
        None => "*".into(),
        Some(x) => format!("{}{}", if lower {
            if x.inclusive { ">=" } else { ">" }
        } else if x.inclusive { "<=" } else { "<" }, display_version(&x.version)),
    }
}

pub fn canonical(r: &Range) -> String {
    let mut out: Vec<String> = r.arms.iter().map(|a| {
        let mut anchors = a.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let anchor_text = if anchors.is_empty() { "-".into() } else {
            anchors.iter().map(|(x,y,z)| format!("{x}.{y}.{z}")).collect::<Vec<_>>().join(",")
        };
        format!("[{};{};pre={}]", bound_text(&a.lower, true), bound_text(&a.upper, false), anchor_text)
    }).collect();
    out.sort();
    out.dedup();
    if out.is_empty() { "EMPTY".into() } else { out.join(" || ") }
}
