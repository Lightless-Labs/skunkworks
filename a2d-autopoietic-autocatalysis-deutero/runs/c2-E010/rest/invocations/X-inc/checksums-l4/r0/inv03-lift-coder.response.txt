#[derive(Debug, Clone)]
pub struct Stream {
    crc_raw: u32,
    adler_a: u32,
    adler_b: u32,
}

impl Stream {
    pub fn new() -> Self {
        Self {
            crc_raw: 0xffff_ffff,
            adler_a: 1,
            adler_b: 0,
        }
    }

    pub fn checksum(&self) -> Hashes {
        Hashes {
            crc32: self.crc_raw ^ 0xffff_ffff,
            adler32: (self.adler_b << 16) | self.adler_a,
        }
    }

    pub fn from_seed(crc32: u32, adler32: u32) -> Result<Self, SnapshotError> {
        let a = adler32 & 0xffff;
        let b = adler32 >> 16;
        if a >= 65_521 {
            return Err(SnapshotError::Adler { component: 0 });
        }
        if b >= 65_521 {
            return Err(SnapshotError::Adler { component: 1 });
        }
        Ok(Self {
            crc_raw: crc32 ^ 0xffff_ffff,
            adler_a: a,
            adler_b: b,
        })
    }

    pub fn update(&mut self, data: &[u8]) {
        update_crc_adler(&mut self.crc_raw, &mut self.adler_a, &mut self.adler_b, data);
    }

    pub fn update_spans(&mut self, data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
        validate_spans(data, spans)?;
        for span in spans {
            self.update(&data[span.start..span.end]);
        }
        Ok(())
    }

    pub fn reset(&mut self) {
        *self = Self::new();
    }

    pub fn snapshot(&self) -> String {
        let h = self.checksum();
        format!("c{:08x}:a{:08x}", h.crc32, h.adler32)
    }

    pub fn restore(token: &str) -> Result<Self, SnapshotError> {
        let bytes = token.as_bytes();
        if bytes.len() != 19 || bytes[0] != b'c' || bytes[9] != b':' || bytes[10] != b'a' {
            return Err(SnapshotError::Shape);
        }
        let crc = parse_hex_u32(bytes, 1, 9).map_err(|offset| SnapshotError::Hex { offset })?;
        let adler = parse_hex_u32(bytes, 11, 19).map_err(|offset| SnapshotError::Hex { offset })?;
        Self::from_seed(crc, adler)
    }
}

pub fn crc32(data: &[u8]) -> u32 {
    let mut raw = 0xffff_ffff;
    for &byte in data {
        raw ^= byte as u32;
        for _ in 0..8 {
            raw = (raw >> 1) ^ if raw & 1 != 0 { 0xedb8_8320 } else { 0 };
        }
    }
    raw ^ 0xffff_ffff
}

pub fn adler32(data: &[u8]) -> u32 {
    let mut a = 1u32;
    let mut b = 0u32;
    for &byte in data {
        a = (a + byte as u32) % 65_521;
        b = (b + a) % 65_521;
    }
    (b << 16) | a
}

pub fn checksums(data: &[u8]) -> Hashes {
    Hashes {
        crc32: crc32(data),
        adler32: adler32(data),
    }
}

fn update_crc_adler(crc_raw: &mut u32, a: &mut u32, b: &mut u32, data: &[u8]) {
    for &byte in data {
        *crc_raw ^= byte as u32;
        for _ in 0..8 {
            *crc_raw = (*crc_raw >> 1)
                ^ if *crc_raw & 1 != 0 { 0xedb8_8320 } else { 0 };
        }
        *a = (*a + byte as u32) % 65_521;
        *b = (*b + *a) % 65_521;
    }
}

fn validate_spans(data: &[u8], spans: &[Span]) -> Result<(), SpanError> {
    let mut previous_end = 0usize;
    for (i, span) in spans.iter().enumerate() {
        if span.start > span.end {
            return Err(SpanError::Reversed { span: i });
        }
        if span.start > data.len() {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.start,
            });
        }
        if span.end > data.len() {
            return Err(SpanError::OutOfBounds {
                span: i,
                offset: span.end,
            });
        }
        if i > 0 && span.start < previous_end {
            return Err(SpanError::Overlap { span: i });
        }
        previous_end = span.end;
    }
    Ok(())
}

pub fn checksums_spans(data: &[u8], spans: &[Span]) -> Result<Hashes, SpanError> {
    validate_spans(data, spans)?;
    let mut stream = Stream::new();
    for span in spans {
        stream.update(&data[span.start..span.end]);
    }
    Ok(stream.checksum())
}

pub fn fletcher16(data: &[u8]) -> u16 {
    let (mut a, mut b) = (0u32, 0u32);
    for &byte in data {
        a = (a + byte as u32) % 255;
        b = (b + a) % 255;
    }
    ((b as u16) << 8) | a as u16
}

pub fn fletcher32(data: &[u8]) -> u32 {
    let (mut a, mut b) = (0u64, 0u64);
    let mut chunks = data.chunks_exact(2);
    for pair in &mut chunks {
        let word = ((pair[0] as u64) << 8) | pair[1] as u64;
        a = (a + word) % 65_535;
        b = (b + a) % 65_535;
    }
    if let Some(&last) = chunks.remainder().first() {
        a = (a + ((last as u64) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }
    ((b as u32) << 16) | a as u32
}

pub fn fletcher32_spans(data: &[u8], spans: &[Span]) -> Result<u32, SpanError> {
    validate_spans(data, spans)?;
    let (mut a, mut b) = (0u64, 0u64);
    let mut pending: Option<u8> = None;
    for span in spans {
        for &byte in &data[span.start..span.end] {
            if let Some(high) = pending.take() {
                let word = ((high as u64) << 8) | byte as u64;
                a = (a + word) % 65_535;
                b = (b + a) % 65_535;
            } else {
                pending = Some(byte);
            }
        }
    }
    if let Some(high) = pending {
        a = (a + ((high as u64) << 8)) % 65_535;
        b = (b + a) % 65_535;
    }
    Ok(((b as u32) << 16) | a as u32)
}

pub fn crc16(data: &[u8], variant: Crc16) -> u16 {
    let mut reg = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };
    for &byte in data {
        match variant {
            Crc16::Xmodem | Crc16::False => {
                reg ^= (byte as u16) << 8;
                for _ in 0..8 {
                    reg = reg.wrapping_shl(1)
                        ^ if reg & 0x8000 != 0 { 0x1021 } else { 0 };
                }
            }
            Crc16::X25 => {
                reg ^= byte as u16;
                for _ in 0..8 {
                    reg = (reg >> 1) ^ if reg & 1 != 0 { 0x8408 } else { 0 };
                }
            }
        }
    }
    if matches!(variant, Crc16::X25) {
        reg ^ 0xffff
    } else {
        reg
    }
}

pub fn multi_checksums(
    data: &[u8],
    spans: &[Span],
    variant: Crc16,
) -> Result<Multi, SpanError> {
    validate_spans(data, spans)?;
    let mut base = Stream::new();
    let (mut f16a, mut f16b) = (0u32, 0u32);
    let (mut f32a, mut f32b) = (0u64, 0u64);
    let mut pending: Option<u8> = None;
    let mut c16 = match variant {
        Crc16::Xmodem => 0u16,
        Crc16::False | Crc16::X25 => 0xffffu16,
    };

    for span in spans {
        for &byte in &data[span.start..span.end] {
            base.update(std::slice::from_ref(&byte));

            f16a = (f16a + byte as u32) % 255;
            f16b = (f16b + f16a) % 255;

            if let Some(high) = pending.take() {
                let word = ((high as u64) << 8) | byte as u64;
                f32a = (f32a + word) % 65_535;
                f32b = (f32b + f32a) % 65_535;
            } else {
                pending = Some(byte);
            }

            match variant {
                Crc16::Xmodem | Crc16::False => {
                    c16 ^= (byte as u16) << 8;
                    for _ in 0..8 {
                        c16 = c16.wrapping_shl(1)
                            ^ if c16 & 0x8000 != 0 { 0x1021 } else { 0 };
                    }
                }
                Crc16::X25 => {
                    c16 ^= byte as u16;
                    for _ in 0..8 {
                        c16 = (c16 >> 1) ^ if c16 & 1 != 0 { 0x8408 } else { 0 };
                    }
                }
            }
        }
    }

    if let Some(high) = pending {
        f32a = (f32a + ((high as u64) << 8)) % 65_535;
        f32b = (f32b + f32a) % 65_535;
    }
    if matches!(variant, Crc16::X25) {
        c16 ^= 0xffff;
    }
    Ok(Multi {
        base: base.checksum(),
        fletcher16: ((f16b as u16) << 8) | f16a as u16,
        fletcher32: ((f32b as u32) << 16) | f32a as u32,
        crc16: c16,
    })
}

fn crc32_matrix_times(matrix: &[u32; 32], mut vector: u32) -> u32 {
    let mut sum = 0;
    let mut index = 0;
    while vector != 0 {
        if vector & 1 != 0 {
            sum ^= matrix[index];
        }
        vector >>= 1;
        index += 1;
    }
    sum
}

fn crc32_matrix_square(square: &mut [u32; 32], matrix: &[u32; 32]) {
    for n in 0..32 {
        square[n] = crc32_matrix_times(matrix, matrix[n]);
    }
}

pub fn crc32_combine(left: u32, right: u32, right_len: u64) -> u32 {
    if right_len == 0 {
        return left;
    }
    let mut odd = [0u32; 32];
    odd[0] = 0xedb8_8320;
    let mut row = 1u32;
    for item in odd.iter_mut().skip(1) {
        *item = row;
        row <<= 1;
    }
    let mut even = [0u32; 32];
    crc32_matrix_square(&mut even, &odd);
    crc32_matrix_square(&mut odd, &even);

    let mut len = right_len;
    let mut crc = left;
    loop {
        crc32_matrix_square(&mut even, &odd);
        if len & 1 != 0 {
            crc = crc32_matrix_times(&even, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
        crc32_matrix_square(&mut odd, &even);
        if len & 1 != 0 {
            crc = crc32_matrix_times(&odd, crc);
        }
        len >>= 1;
        if len == 0 {
            break;
        }
    }
    crc ^ right
}

pub fn combine_parts(parts: &[Part]) -> Result<Part, CombineError> {
    let mut total = 0u64;
    for (i, part) in parts.iter().enumerate() {
        total = total
            .checked_add(part.len)
            .ok_or(CombineError::LengthOverflow { part: i })?;
    }
    for (i, part) in parts.iter().enumerate() {
        if part.len == 0 && part.crc32 != 0 {
            return Err(CombineError::EmptyCrc { part: i });
        }
    }
    let mut crc = 0u32;
    for part in parts {
        crc = crc32_combine(crc, part.crc32, part.len);
    }
    Ok(Part { crc32: crc, len: total })
}

pub fn manifest(data: &[u8], spans: &[Span]) -> Result<String, SpanError> {
    validate_spans(data, spans)?;
    let mut stream = Stream::new();
    let mut len = 0u64;
    for span in spans {
        let selected = &data[span.start..span.end];
        len += selected.len() as u64;
        stream.update(selected);
    }
    let h = stream.checksum();
    Ok(format!(
        "len={:016x};crc={:08x};adler={:08x}",
        len, h.crc32, h.adler32
    ))
}

pub fn verify(data: &[u8], spans: &[Span], token: &str) -> Result<(), VerifyError> {
    validate_spans(data, spans).map_err(VerifyError::Span)?;
    let bytes = token.as_bytes();
    if bytes.len() != 48
        || &bytes[0..4] != b"len="
        || &bytes[20..25] != b";crc="
        || &bytes[33..40] != b";adler="
    {
        return Err(VerifyError::Shape);
    }
    let len = parse_hex_u64(bytes, 4, 20).map_err(|offset| VerifyError::Hex { offset })?;
    let expected_crc =
        parse_hex_u32(bytes, 25, 33).map_err(|offset| VerifyError::Hex { offset })?;
    let expected_adler =
        parse_hex_u32(bytes, 40, 48).map_err(|offset| VerifyError::Hex { offset })?;

    let mut selected_len = 0u64;
    let mut stream = Stream::new();
    for span in spans {
        let selected = &data[span.start..span.end];
        selected_len += selected.len() as u64;
        stream.update(selected);
    }
    if len != selected_len {
        return Err(VerifyError::Length);
    }
    let actual = stream.checksum();
    if expected_crc != actual.crc32 {
        return Err(VerifyError::Mismatch {
            field: 0,
            expected: expected_crc,
            actual: actual.crc32,
        });
    }
    if expected_adler != actual.adler32 {
        return Err(VerifyError::Mismatch {
            field: 1,
            expected: expected_adler,
            actual: actual.adler32,
        });
    }
    Ok(())
}

fn hex_digit(byte: u8, offset: usize) -> Result<u32, usize> {
    match byte {
        b'0'..=b'9' => Ok((byte - b'0') as u32),
        b'a'..=b'f' => Ok((byte - b'a' + 10) as u32),
        _ => Err(offset),
    }
}

fn parse_hex_u32(bytes: &[u8], start: usize, end: usize) -> Result<u32, usize> {
    let mut value = 0u32;
    for (offset, &byte) in bytes.iter().enumerate().take(end).skip(start) {
        value = (value << 4) | hex_digit(byte, offset)?;
    }
    Ok(value)
}

fn parse_hex_u64(bytes: &[u8], start: usize, end: usize) -> Result<u64, usize> {
    let mut value = 0u64;
    for (offset, &byte) in bytes.iter().enumerate().take(end).skip(start) {
        value = (value << 4) | hex_digit(byte, offset)? as u64;
    }
    Ok(value)
}
