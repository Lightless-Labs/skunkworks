```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
    Root,
    Busy(String),
    Cycle(String),
    NotEmpty(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir,
    File(Vec<u8>),
}

pub struct Fs {
    entries: BTreeMap<String, Entry>,
    cwd: String,
}

impl Fs {
    pub fn new() -> Self {
        let mut entries = BTreeMap::new();
        entries.insert("/".to_string(), Entry::Dir);
        Self {
            entries,
            cwd: "/".to_string(),
        }
    }

    fn normalize(&self, path: &str) -> Result<String, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() {
            return Err(Error::InvalidPath(0));
        }

        let mut parts: Vec<&str> = Vec::new();
        if !path.starts_with('/') {
            parts.extend(self.cwd.split('/').filter(|s| !s.is_empty()));
        }

        for part in path.split('/') {
            match part {
                "" | "." => {}
                ".." => {
                    parts.pop();
                }
                other => parts.push(other),
            }
        }

        if parts.is_empty() {
            Ok("/".to_string())
        } else {
            Ok(format!("/{}", parts.join("/")))
        }
    }

    fn parent_and_name(path: &str) -> (String, String) {
        if path == "/" {
            return ("/".to_string(), String::new());
        }
        let idx = path.rfind('/').unwrap_or(0);
        let parent = if idx == 0 {
            "/".to_string()
        } else {
            path[..idx].to_string()
        };
        (parent, path[idx + 1..].to_string())
    }

    fn prefixes(path: &str) -> Vec<String> {
        if path == "/" {
            return vec!["/".to_string()];
        }
        let mut out = vec!["/".to_string()];
        let mut current = String::new();
        for component in path.split('/').filter(|s| !s.is_empty()) {
            current.push('/');
            current.push_str(component);
            out.push(current.clone());
        }
        out
    }

    fn resolve_existing(&self, path: &str) -> Result<&Entry, Error> {
        let prefixes = Self::prefixes(path);
        for (i, prefix) in prefixes.iter().enumerate() {
            let entry = self
                .entries
                .get(prefix)
                .ok_or_else(|| Error::Missing(prefix.clone()))?;
            if i + 1 < prefixes.len() && !matches!(entry, Entry::Dir) {
                return Err(Error::NotDir(prefix.clone()));
            }
        }
        self.entries
            .get(path)
            .ok_or_else(|| Error::Missing(path.to_string()))
    }

    fn resolve_parent(&self, path: &str) -> Result<(String, String), Error> {
        let (parent, name) = Self::parent_and_name(path);
        self.resolve_existing(&parent)?;
        match self.entries.get(&parent) {
            Some(Entry::Dir) => Ok((parent, name)),
            Some(Entry::File(_)) => Err(Error::NotDir(parent)),
            None => Err(Error::Missing(parent)),
        }
    }

    fn is_within(path: &str, dir: &str) -> bool {
        if dir == "/" {
            return path.starts_with('/');
        }
        path == dir
            || (path.starts_with(dir)
                && path.as_bytes().get(dir.len()) == Some(&b'/'))
    }

    fn has_children(&self, path: &str) -> bool {
        self.entries
            .keys()
            .any(|candidate| candidate != path && Self::is_within(candidate, path))
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Exists(path));
        }
        let (parent, _) = self.resolve_parent(&path)?;
        if self.entries.contains_key(&path) {
            return Err(Error::Exists(path));
        }
        let _ = parent;
        self.entries.insert(path, Entry::Dir);
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::IsDir(path));
        }
        self.resolve_parent(&path)?;
        match self.entries.get(&path) {
            Some(Entry::Dir) => Err(Error::IsDir(path)),
            Some(Entry::File(_)) => {
                self.entries.insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
            None => {
                self.entries.insert(path, Entry::File(data.to_vec()));
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let path = self.normalize(path)?;
        match self.resolve_existing(&path)? {
            Entry::Dir => Err(Error::IsDir(path)),
            Entry::File(data) => Ok(data.clone()),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let path = self.normalize(path)?;
        match self.resolve_existing(&path)? {
            Entry::File(_) => return Err(Error::NotDir(path)),
            Entry::Dir => {}
        }

        let mut children = Vec::new();
        for (candidate, entry) in &self.entries {
            if candidate == "/" {
                continue;
            }
            let (parent, name) = Self::parent_and_name(candidate);
            if parent == path {
                let kind = match entry {
                    Entry::Dir => Kind::Dir,
                    Entry::File(_) => Kind::File,
                };
                children.push((name, kind));
            }
        }
        children.sort_by(|a, b| a.0.cmp(&b.0));
        Ok(children)
    }

    pub fn cwd(&self) -> String {
        self.cwd.clone()
    }

    pub fn chdir(&mut self, path: &str) -> Result<(), Error> {
        let path = self.normalize(path)?;
        match self.resolve_existing(&path)? {
            Entry::Dir => {
                self.cwd = path;
                Ok(())
            }
            Entry::File(_) => Err(Error::NotDir(path)),
        }
    }

    pub fn rename(
        &mut self,
        from: &str,
        to: &str,
        overwrite: bool,
    ) -> Result<(), Error> {
        let from = self.normalize(from)?;
        let to = self.normalize(to)?;

        if from == "/" || to == "/" {
            return Err(Error::Root);
        }

        let (source_parent, _) = self.resolve_parent(&from)?;
        let source_entry = self
            .entries
            .get(&from)
            .cloned()
            .ok_or_else(|| Error::Missing(from.clone()))?;

        if matches!(source_entry, Entry::Dir) && Self::is_within(&self.cwd, &from) {
            return Err(Error::Busy(from));
        }

        let _ = source_parent;
        let (destination_parent, _) = self.resolve_parent(&to)?;

        if matches!(self.entries.get(&to), Some(Entry::Dir))
            && Self::is_within(&self.cwd, &to)
        {
            return Err(Error::Busy(to));
        }

        let _ = destination_parent;
        if from == to {
            return Ok(());
        }

        if matches!(source_entry, Entry::Dir) && Self::is_within(&to, &from) {
            return Err(Error::Cycle(to));
        }

        let destination_entry = self.entries.get(&to).cloned();
        if destination_entry.is_some() && !overwrite {
            return Err(Error::Exists(to));
        }

        if overwrite {
            if let Some(dest) = &destination_entry {
                match (&source_entry, dest) {
                    (Entry::Dir, Entry::File(_)) => return Err(Error::NotDir(to)),
                    (Entry::File(_), Entry::Dir) => return Err(Error::IsDir(to)),
                    (Entry::Dir, Entry::Dir) if self.has_children(&to) => {
                        return Err(Error::NotEmpty(to));
                    }
                    _ => {}
                }
            }
        }

        let moving: Vec<(String, Entry)> = self
            .entries
            .iter()
            .filter(|(candidate, _)| {
                *candidate == &from || Self::is_within(candidate, &from)
            })
            .map(|(candidate, entry)| (candidate.clone(), entry.clone()))
            .collect();

        if let Some(dest) = destination_entry {
            if matches!(dest, Entry::Dir) {
                self.entries.remove(&to);
            } else {
                self.entries.remove(&to);
            }
        }

        for (old_path, _) in &moving {
            self.entries.remove(old_path);
        }
        for (old_path, entry) in moving {
            let suffix = &old_path[from.len()..];
            let new_path = format!("{to}{suffix}");
            self.entries.insert(new_path, entry);
        }

        Ok(())
    }

    pub fn remove(&mut self, path: &str, recursive: bool) -> Result<(), Error> {
        let path = self.normalize(path)?;
        if path == "/" {
            return Err(Error::Root);
        }

        self.resolve_parent(&path)?;
        let entry = self
            .entries
            .get(&path)
            .cloned()
            .ok_or_else(|| Error::Missing(path.clone()))?;

        if matches!(entry, Entry::Dir) && Self::is_within(&self.cwd, &path) {
            return Err(Error::Busy(path));
        }

        if matches!(entry, Entry::Dir) && self.has_children(&path) && !recursive {
            return Err(Error::NotEmpty(path));
        }

        let remove_paths: Vec<String> = self
            .entries
            .keys()
            .filter(|candidate| {
                *candidate == &path || (recursive && Self::is_within(candidate, &path))
            })
            .cloned()
            .collect();
        for candidate in remove_paths {
            self.entries.remove(&candidate);
        }
        Ok(())
    }
}```
