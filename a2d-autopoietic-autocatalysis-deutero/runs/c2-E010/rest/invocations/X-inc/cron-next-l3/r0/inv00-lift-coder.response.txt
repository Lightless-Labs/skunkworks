#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_unrestricted: bool,
    day_of_week_unrestricted: bool,
}

#[derive(Clone, Copy)]
struct Domain {
    min: u32,
    max: u32,
}

impl Domain {
    fn width(self) -> u32 {
        self.max - self.min + 1
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn domain(field: Field) -> Domain {
    match field {
        Field::Minute => Domain { min: 0, max: 59 },
        Field::Hour => Domain { min: 0, max: 23 },
        Field::DayOfMonth => Domain { min: 1, max: 31 },
        Field::Month => Domain { min: 1, max: 12 },
        Field::DayOfWeek => Domain { min: 0, max: 6 },
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

#[derive(Clone, Copy)]
enum Endpoint {
    Number(u32, usize),
    Name(u32),
}

impl Endpoint {
    fn value(self) -> u32 {
        match self {
            Endpoint::Number(value, _) | Endpoint::Name(value) => value,
        }
    }

    fn number_offset(self) -> Option<usize> {
        match self {
            Endpoint::Number(_, offset) => Some(offset),
            Endpoint::Name(_) => None,
        }
    }
}

fn month_name(s: &[u8]) -> Option<u32> {
    const NAMES: [&[u8; 3]; 12] = [
        b"JAN", b"FEB", b"MAR", b"APR", b"MAY", b"JUN",
        b"JUL", b"AUG", b"SEP", b"OCT", b"NOV", b"DEC",
    ];
    NAMES.iter().position(|name| s.eq_ignore_ascii_case(*name)).map(|i| i as u32 + 1)
}

fn weekday_name(s: &[u8]) -> Option<u32> {
    const NAMES: [&[u8; 3]; 7] = [
        b"SUN", b"MON", b"TUE", b"WED", b"THU", b"FRI", b"SAT",
    ];
    NAMES.iter().position(|name| s.eq_ignore_ascii_case(*name)).map(|i| i as u32)
}

fn parse_endpoint(
    field: Field,
    bytes: &[u8],
    offset: usize,
) -> Result<Endpoint, ParseError> {
    if bytes.is_empty() {
        return Err(parse_error(field, offset, 0));
    }

    if bytes.iter().all(u8::is_ascii_digit) {
        if bytes.len() > 1 && bytes[0] == b'0' {
            return Err(parse_error(field, offset, 1));
        }
        let mut value = 0u32;
        for &byte in bytes {
            value = value
                .checked_mul(10)
                .and_then(|v| v.checked_add((byte - b'0') as u32))
                .ok_or_else(|| parse_error(field, offset, 2))?;
        }
        let d = domain(field);
        if value < d.min || value > d.max {
            return Err(parse_error(field, offset, 3));
        }
        return Ok(Endpoint::Number(value, offset));
    }

    let named = match field {
        Field::Month => month_name(bytes),
        Field::DayOfWeek => weekday_name(bytes),
        _ => None,
    };
    named
        .map(Endpoint::Name)
        .ok_or_else(|| parse_error(field, offset, 0))
}

fn parse_field(field: Field, text: &str) -> Result<Vec<bool>, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let d = domain(field);
    let mut selected = vec![false; d.max as usize + 1];

    let mut item_start = 0usize;
    loop {
        let item_end = bytes[item_start..]
            .iter()
            .position(|&b| b == b',')
            .map(|n| item_start + n)
            .unwrap_or(bytes.len());

        if item_start == item_end {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item = &bytes[item_start..item_end];
        let slash_positions: Vec<usize> = item
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        let (base, step_bytes, step_offset) = match slash_positions.as_slice() {
            [] => (item, None, 0),
            [slash] => (&item[..*slash], Some(&item[*slash + 1..]), *slash + 1),
            _ => return Err(parse_error(field, item_start, 0)),
        };

        let (is_star, endpoints): (bool, Vec<(usize, usize)>) = if base == b"*" {
            (true, Vec::new())
        } else {
            let hyphens: Vec<usize> = base
                .iter()
                .enumerate()
                .filter_map(|(i, &b)| (b == b'-').then_some(i))
                .collect();
            match hyphens.as_slice() {
                [] => (false, vec![(0, base.len())]),
                [hyphen] if *hyphen > 0 && *hyphen + 1 < base.len() => {
                    (false, vec![(0, *hyphen), (*hyphen + 1, base.len())])
                }
                _ => return Err(parse_error(field, item_start, 0)),
            }
        };

        let mut parsed_endpoints = Vec::new();
        if !is_star {
            for (start, end) in endpoints {
                let endpoint = parse_endpoint(field, &base[start..end], item_start + start)?;
                parsed_endpoints.push(endpoint);
            }
        }

        let (start, end) = if is_star {
            (d.min, d.max)
        } else if parsed_endpoints.len() == 1 {
            let value = parsed_endpoints[0].value();
            (value, if step_bytes.is_some() { d.max } else { value })
        } else {
            let first = parsed_endpoints[0];
            let second = parsed_endpoints[1];
            let first_value = first.value();
            let second_value = second.value();
            if first_value > second_value {
                let offset = second.number_offset().unwrap_or(item_start);
                return Err(ParseError::ReversedRange { field, offset });
            }
            (first_value, second_value)
        };

        let step = if let Some(step_bytes) = step_bytes {
            if step_bytes.is_empty() || !step_bytes.iter().all(u8::is_ascii_digit) {
                return Err(parse_error(field, item_start, 0));
            }
            let offset = item_start + step_offset;
            if step_bytes.len() > 1 && step_bytes[0] == b'0' {
                return Err(ParseError::LeadingZero { field, offset });
            }
            let mut value = 0u32;
            for &byte in step_bytes {
                value = value
                    .checked_mul(10)
                    .and_then(|v| v.checked_add((byte - b'0') as u32))
                    .ok_or(ParseError::NumberOverflow { field, offset })?;
            }
            if value == 0 || value > d.width() {
                return Err(ParseError::BadStep { field, offset });
            }
            value
        } else {
            1
        };

        let mut value = start;
        while value <= end {
            selected[value as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        if item_end == bytes.len() {
            break;
        }
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = Vec::with_capacity(5);

    for (field, text) in fields.into_iter().zip(parts.iter().copied()) {
        selected.push(parse_field(field, text)?);
    }

    Ok(Cron {
        selected: selected.try_into().ok().unwrap(),
        day_of_month_unrestricted: parts[2] == "*",
        day_of_week_unrestricted: parts[4] == "*",
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_selected)| {
                (is_selected && value <= u8::MAX as usize).then_some(value as u8)
            })
            .collect()
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_validated(at))
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;

        let mut current = after;
        loop {
            if !advance_minute(&mut current) {
                return Ok(None);
            }
            if self.matches_validated(current) {
                return Ok(Some(current));
            }
        }
    }

    fn matches_validated(&self, at: CivilDateTime) -> bool {
        if !self.selected[0][at.minute as usize]
            || !self.selected[1][at.hour as usize]
            || !self.selected[3][at.month as usize]
        {
            return false;
        }

        let dom = self.selected[2][at.day as usize];
        let dow = self.selected[4][weekday(at.year, at.month, at.day) as usize];

        match (
            self.day_of_month_unrestricted,
            self.day_of_week_unrestricted,
        ) {
            (true, true) => true,
            (true, false) => dow,
            (false, true) => dom,
            (false, false) => {
                let last_day = days_in_month(at.year, at.month);
                if at.day == last_day {
                    dom && dow
                } else {
                    dom || dow
                }
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap_year(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap_year(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    // 2000-01-01 was Saturday (6). Count elapsed days from that date.
    let mut days: i64 = 0;
    let mut y = 2000u16;
    while y < year {
        days += if is_leap_year(y) { 366 } else { 365 };
        y += 1;
    }
    for m in 1..month {
        days += days_in_month(year, m) as i64;
    }
    days += day as i64 - 1;
    ((6 + days.rem_euclid(7)) % 7) as u8
}

fn advance_minute(at: &mut CivilDateTime) -> bool {
    if at.minute < 59 {
        at.minute += 1;
        return true;
    }
    at.minute = 0;

    if at.hour < 23 {
        at.hour += 1;
        return true;
    }
    at.hour = 0;

    let last_day = days_in_month(at.year, at.month);
    if at.day < last_day {
        at.day += 1;
        return true;
    }
    at.day = 1;

    if at.month < 12 {
        at.month += 1;
        return true;
    }
    at.month = 1;

    if at.year < 2399 {
        at.year += 1;
        true
    } else {
        false
    }
}
