#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    values: [Vec<bool>; 5],
    unrestricted_days: [bool; 2],
}

#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
pub struct CivilDateTime {
    pub year: u16,
    pub month: u8,
    pub day: u8,
    pub hour: u8,
    pub minute: u8,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DatePart {
    Year,
    Month,
    Day,
    Hour,
    Minute,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DateError {
    InvalidDateTime { component: DatePart },
}

const FIELDS: [Field; 5] = [
    Field::Minute,
    Field::Hour,
    Field::DayOfMonth,
    Field::Month,
    Field::DayOfWeek,
];

fn bounds(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn names(field: Field) -> &'static [(&'static str, u32)] {
    match field {
        Field::Month => &[
            ("JAN", 1), ("FEB", 2), ("MAR", 3), ("APR", 4),
            ("MAY", 5), ("JUN", 6), ("JUL", 7), ("AUG", 8),
            ("SEP", 9), ("OCT", 10), ("NOV", 11), ("DEC", 12),
        ],
        Field::DayOfWeek => &[
            ("SUN", 0), ("MON", 1), ("TUE", 2), ("WED", 3),
            ("THU", 4), ("FRI", 5), ("SAT", 6),
        ],
        _ => &[],
    }
}

fn parse_error(field: Field, offset: usize, kind: u8) -> ParseError {
    match kind {
        0 => ParseError::InvalidSyntax { field, offset },
        1 => ParseError::LeadingZero { field, offset },
        2 => ParseError::NumberOverflow { field, offset },
        3 => ParseError::OutOfRange { field, offset },
        4 => ParseError::ReversedRange { field, offset },
        5 => ParseError::BadStep { field, offset },
        _ => ParseError::EmptyItem { field, offset },
    }
}

fn numeric_value(bytes: &[u8], field: Field, offset: usize) -> Result<u32, ParseError> {
    if bytes.is_empty() || !bytes.iter().all(u8::is_ascii_digit) {
        return Err(parse_error(field, offset, 0));
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(parse_error(field, offset, 1));
    }
    let mut value = 0u32;
    for &byte in bytes {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or_else(|| parse_error(field, offset, 2))?;
    }
    Ok(value)
}

fn atom_value(bytes: &[u8], field: Field, offset: usize) -> Result<(u32, bool), ParseError> {
    if bytes.iter().all(u8::is_ascii_digit) && !bytes.is_empty() {
        let value = numeric_value(bytes, field, offset)?;
        let (min, max) = bounds(field);
        if value < min || value > max {
            return Err(parse_error(field, offset, 3));
        }
        return Ok((value, true));
    }

    if bytes.iter().all(u8::is_ascii_alphabetic) {
        if let Ok(word) = std::str::from_utf8(bytes) {
            let upper = word.to_ascii_uppercase();
            if let Some((_, value)) = names(field).iter().find(|(name, _)| *name == upper) {
                return Ok((*value, false));
            }
        }
    }

    Err(parse_error(field, offset, 0))
}

fn parse_field(
    text: &str,
    field: Field,
    selected: &mut [bool],
) -> Result<bool, ParseError> {
    let bytes = text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_digit()
            || byte.is_ascii_alphabetic()
            || matches!(byte, b'*' | b'-' | b'/' | b','))
        {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = bounds(field);
    let width = max - min + 1;
    let mut unrestricted = false;
    let mut item_start = 0usize;

    for item in text.split(',') {
        if item.is_empty() {
            return Err(ParseError::EmptyItem {
                field,
                offset: item_start,
            });
        }

        let item_bytes = item.as_bytes();
        let slash_positions: Vec<usize> = item_bytes
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'/').then_some(i))
            .collect();
        let base_end = slash_positions.first().copied().unwrap_or(item_bytes.len());

        if slash_positions.len() > 1 || base_end == 0 {
            return Err(parse_error(field, item_start, 0));
        }
        if !slash_positions.is_empty() && base_end + 1 == item_bytes.len() {
            return Err(parse_error(field, item_start, 0));
        }

        let base = &item_bytes[..base_end];
        let step_bytes = if slash_positions.is_empty() {
            None
        } else {
            Some(&item_bytes[base_end + 1..])
        };

        let is_star = base == b"*";
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        if hyphens.len() > 1
            || (is_star && !hyphens.is_empty())
            || (!is_star && hyphens.first().is_some_and(|&i| i == 0 || i + 1 == base.len()))
        {
            return Err(parse_error(field, item_start, 0));
        }

        let range = !hyphens.is_empty();
        if !is_star && !range && base.is_empty() {
            return Err(parse_error(field, item_start, 0));
        }

        let (start, end, start_numeric, end_numeric, end_offset) = if is_star {
            (min, max, false, false, item_start)
        } else if let Some(&hyphen) = hyphens.first() {
            let (start, start_numeric) =
                atom_value(&base[..hyphen], field, item_start)?;
            let end_offset = item_start + hyphen + 1;
            let (end, end_numeric) =
                atom_value(&base[hyphen + 1..], field, end_offset)?;

            if start_numeric && end_numeric {
                if start > end {
                    return Err(parse_error(field, end_offset, 4));
                }
            } else if start > end {
                return Err(parse_error(field, end_offset, 4));
            }
            (start, end, start_numeric, end_numeric, end_offset)
        } else {
            let (value, numeric) = atom_value(base, field, item_start)?;
            let end = if step_bytes.is_some() { max } else { value };
            (value, end, numeric, numeric, item_start)
        };

        // Numeric endpoints are validated in start-then-end order. atom_value has
        // already performed those checks, while named endpoints need none.
        let _ = (start_numeric, end_numeric, end_offset);

        let step = if let Some(step_bytes) = step_bytes {
            let value = numeric_value(step_bytes, field, item_start + base_end + 1)
                .map_err(|error| match error {
                    ParseError::LeadingZero { offset, .. } => parse_error(field, offset, 1),
                    ParseError::NumberOverflow { offset, .. } => parse_error(field, offset, 2),
                    _ => parse_error(field, item_start + base_end + 1, 5),
                })?;
            if value == 0 || value > width {
                return Err(parse_error(field, item_start + base_end + 1, 5));
            }
            value
        } else {
            1
        };

        if item_start == 0 && item == "*" && field == Field::DayOfMonth {
            unrestricted = true;
        }
        if item_start == 0 && item == "*" && field == Field::DayOfWeek {
            unrestricted = true;
        }

        let mut value = start;
        while value <= end {
            selected[(value - min) as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }

        item_start += item.len() + 1;
    }

    Ok(unrestricted)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let pieces: Vec<&str> = input.split(' ').collect();
    if pieces.len() != 5 {
        return Err(ParseError::FieldCount { found: pieces.len() });
    }

    let mut values: [Vec<bool>; 5] = std::array::from_fn(|i| {
        let (min, max) = bounds(FIELDS[i]);
        vec![false; (max - min + 1) as usize]
    });
    let mut unrestricted_days = [false; 2];

    for (i, field) in FIELDS.iter().copied().enumerate() {
        let unrestricted = parse_field(pieces[i], field, &mut values[i])?;
        match field {
            Field::DayOfMonth => unrestricted_days[0] = unrestricted,
            Field::DayOfWeek => unrestricted_days[1] = unrestricted,
            _ => {}
        }
    }

    Ok(Cron {
        values,
        unrestricted_days,
    })
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, _) = bounds(field);
        self.values[index]
            .iter()
            .enumerate()
            .filter_map(|(i, &selected)| selected.then_some((min as usize + i) as u8))
            .collect()
    }

    fn has(&self, field: Field, value: u8) -> bool {
        let index = match field {
            Field::Minute => 0,
            Field::Hour => 1,
            Field::DayOfMonth => 2,
            Field::Month => 3,
            Field::DayOfWeek => 4,
        };
        let (min, max) = bounds(field);
        (value as u32 >= min && value as u32 <= max)
            && self.values[index][(value as u32 - min) as usize]
    }

    pub fn matches(&self, at: CivilDateTime) -> Result<bool, DateError> {
        validate_datetime(at)?;
        Ok(self.matches_valid(at))
    }

    fn matches_valid(&self, at: CivilDateTime) -> bool {
        if !self.has(Field::Minute, at.minute)
            || !self.has(Field::Hour, at.hour)
            || !self.has(Field::Month, at.month)
        {
            return false;
        }

        let dom_restricted = !self.unrestricted_days[0];
        let dow_restricted = !self.unrestricted_days[1];
        if !dom_restricted && !dow_restricted {
            return true;
        }
        if dom_restricted && !dow_restricted {
            return self.has(Field::DayOfMonth, at.day);
        }
        if !dom_restricted && dow_restricted {
            return self.has(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        }

        let dom = self.has(Field::DayOfMonth, at.day);
        let dow = self.has(Field::DayOfWeek, weekday(at.year, at.month, at.day));
        if at.day == days_in_month(at.year, at.month) {
            dom && dow
        } else {
            dom || dow
        }
    }

    pub fn next_fire(&self, after: CivilDateTime) -> Result<Option<CivilDateTime>, DateError> {
        validate_datetime(after)?;
        let mut current = after;
        loop {
            if let Some(next) = advance_minute(current) {
                current = next;
            } else {
                return Ok(None);
            }
            if self.matches_valid(current) {
                return Ok(Some(current));
            }
        }
    }
}

fn validate_datetime(at: CivilDateTime) -> Result<(), DateError> {
    if !(2000..=2399).contains(&at.year) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Year,
        });
    }
    if !(1..=12).contains(&at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Month,
        });
    }
    if at.day == 0 || at.day > days_in_month(at.year, at.month) {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Day,
        });
    }
    if at.hour > 23 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Hour,
        });
    }
    if at.minute > 59 {
        return Err(DateError::InvalidDateTime {
            component: DatePart::Minute,
        });
    }
    Ok(())
}

fn is_leap(year: u16) -> bool {
    year % 4 == 0 && (year % 100 != 0 || year % 400 == 0)
}

fn days_in_month(year: u16, month: u8) -> u8 {
    match month {
        2 => if is_leap(year) { 29 } else { 28 },
        4 | 6 | 9 | 11 => 30,
        _ => 31,
    }
}

fn weekday(year: u16, month: u8, day: u8) -> u8 {
    let mut days = 0u32;
    for y in 2000..year {
        days += if is_leap(y) { 366 } else { 365 };
    }
    for m in 1..month {
        days += days_in_month(year, m) as u32;
    }
    days += (day - 1) as u32;
    ((days + 6) % 7) as u8
}

fn advance_minute(at: CivilDateTime) -> Option<CivilDateTime> {
    if at.year == 2399 && at.month == 12 && at.day == 31 && at.hour == 23 && at.minute == 59 {
        return None;
    }
    let mut next = at;
    if next.minute < 59 {
        next.minute += 1;
    } else {
        next.minute = 0;
        if next.hour < 23 {
            next.hour += 1;
        } else {
            next.hour = 0;
            if next.day < days_in_month(next.year, next.month) {
                next.day += 1;
            } else {
                next.day = 1;
                if next.month < 12 {
                    next.month += 1;
                } else {
                    next.month = 1;
                    next.year += 1;
                }
            }
        }
    }
    Some(next)
}
