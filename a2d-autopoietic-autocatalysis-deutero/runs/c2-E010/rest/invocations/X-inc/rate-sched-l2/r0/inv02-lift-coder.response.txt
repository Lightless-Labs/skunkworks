#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CronError {
    Layout(usize),
    Syntax(usize),
    Number(usize),
    Bounds(usize),
    Descending(usize),
    Step(usize),
}

pub struct Cron {
    minutes: [bool; 60],
    hours: [bool; 24],
}

impl Cron {
    pub fn parse(text: &str) -> Result<Self, CronError> {
        for (i, b) in text.bytes().enumerate() {
            if b.is_ascii_whitespace() && b != b' ' {
                return Err(CronError::Layout(i));
            }
        }

        let spaces: Vec<usize> = text
            .bytes()
            .enumerate()
            .filter_map(|(i, b)| (b == b' ').then_some(i))
            .collect();

        if spaces.is_empty() {
            return Err(CronError::Layout(text.len()));
        }
        if spaces.len() > 1 {
            return Err(CronError::Layout(spaces[1]));
        }

        let split = spaces[0];
        if split == 0 || split + 1 == text.len() {
            return Err(CronError::Layout(split));
        }

        let minute_text = &text[..split];
        let hour_text = &text[split + 1..];
        let mut minutes = [false; 60];
        let mut hours = [false; 24];

        parse_field(minute_text, 0, 59, &mut minutes)?;
        parse_field(hour_text, split + 1, 23, &mut hours)?;

        Ok(Self { minutes, hours })
    }

    pub fn matches(&self, minute: u64) -> bool {
        self.minutes[(minute % 60) as usize]
            && self.hours[((minute / 60) % 24) as usize]
    }
}

fn parse_field(
    field: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let mut start = 0;
    loop {
        let end = field[start..]
            .find(',')
            .map_or(field.len(), |i| start + i);
        parse_item(&field[start..end], offset + start, max, selected)?;
        if end == field.len() {
            break;
        }
        start = end + 1;
    }
    Ok(())
}

fn parse_item(
    item: &str,
    offset: usize,
    max: usize,
    selected: &mut [bool],
) -> Result<(), CronError> {
    let bytes = item.as_bytes();
    let slash_positions: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    if slash_positions.len() > 1 {
        return Err(CronError::Syntax(offset + slash_positions[1]));
    }

    let slash = slash_positions.first().copied();
    let (base_end, step_start) = match slash {
        Some(i) => (i, Some(i + 1)),
        None => (bytes.len(), None),
    };
    let base = &item[..base_end];

    let is_star = base == "*";
    let dash = base.as_bytes().iter().position(|&b| b == b'-');
    if step_start.is_some() && !is_star && dash.is_none() {
        return Err(CronError::Syntax(offset + slash.unwrap()));
    }

    let (first, last, dash_pos) = if is_star {
        (0, max, None)
    } else if let Some(d) = dash {
        let left = parse_number(&base[..d], offset, max)?;
        let right = parse_number(&base[d + 1..], offset + d + 1, max)?;
        if left > right {
            return Err(CronError::Descending(offset + d));
        }
        (left, right, Some(d))
    } else {
        let n = parse_number(base, offset, max)?;
        (n, n, None)
    };

    let step = if let Some(slash_pos) = slash {
        let start = slash_pos + 1;
        let n = parse_number(&item[start..], offset + start, usize::MAX)?;
        let step_max = if max == 59 { 60 } else { 24 };
        if n == 0 || n > step_max {
            return Err(CronError::Step(offset + start));
        }
        n
    } else {
        1
    };

    let _ = dash_pos;
    let mut n = first;
    while n <= last {
        selected[n] = true;
        match n.checked_add(step) {
            Some(next) if next > n => n = next,
            _ => break,
        }
    }
    Ok(())
}

fn parse_number(fragment: &str, offset: usize, max: usize) -> Result<usize, CronError> {
    let bytes = fragment.as_bytes();

    if bytes.is_empty() {
        return Err(CronError::Syntax(offset));
    }
    for (i, &b) in bytes.iter().enumerate() {
        if !b.is_ascii_digit() {
            return Err(CronError::Syntax(offset + i));
        }
    }
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(CronError::Number(offset));
    }

    let mut value = 0usize;
    for &b in bytes {
        let digit = (b - b'0') as usize;
        value = match value.checked_mul(10).and_then(|v| v.checked_add(digit)) {
            Some(v) => v,
            None => return Err(CronError::Bounds(offset)),
        };
        if value > max {
            return Err(CronError::Bounds(offset));
        }
    }
    Ok(value)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Policy {
    Skip,
    Once,
    All,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Occurrence {
    pub id: String,
    pub at: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    TimeReversed,
    TimeRange,
    BadId,
    BadPriority,
    Duplicate,
    Cron(CronError),
}

struct Job {
    id: String,
    cron: Cron,
    priority: u8,
    policy: Policy,
    insertion: usize,
    first_poll: Option<u64>,
}

pub struct Calendar {
    jobs: Vec<Job>,
    last_poll: Option<u64>,
    ids: Vec<String>,
}

impl Calendar {
    pub fn new() -> Self {
        Self {
            jobs: Vec::new(),
            last_poll: None,
            ids: Vec::new(),
        }
    }

    pub fn add(
        &mut self,
        id: &str,
        expression: &str,
        priority: u8,
        policy: Policy,
    ) -> Result<(), Error> {
        if id.is_empty()
            || id.len() > 16
            || !id
                .bytes()
                .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'-')
        {
            return Err(Error::BadId);
        }
        if priority > 9 {
            return Err(Error::BadPriority);
        }
        if self.ids.iter().any(|existing| existing == id) {
            return Err(Error::Duplicate);
        }
        let cron = Cron::parse(expression).map_err(Error::Cron)?;

        self.ids.push(id.to_owned());
        self.jobs.push(Job {
            id: id.to_owned(),
            cron,
            priority,
            policy,
            insertion: self.jobs.len(),
            first_poll: self.last_poll,
        });
        Ok(())
    }

    pub fn poll(&mut self, now: u64) -> Result<Vec<Occurrence>, Error> {
        if let Some(last) = self.last_poll {
            if now < last {
                return Err(Error::TimeReversed);
            }
        }
        if now > 604_800_000 {
            return Err(Error::TimeRange);
        }

        const MINUTE_MS: u64 = 60_000;
        let mut indexed: Vec<(Occurrence, u8, usize)> = Vec::new();

        for job in &self.jobs {
            let interval_start = match (self.last_poll, job.first_poll) {
                (Some(last), Some(added_at)) => last.max(added_at),
                (Some(last), None) => last,
                (None, Some(added_at)) => added_at,
                (None, None) => 0,
            };

            let first_index = if self.last_poll.is_none() && job.first_poll.is_none() {
                0
            } else {
                interval_start / MINUTE_MS + 1
            };
            let last_index = now / MINUTE_MS;

            if first_index > last_index {
                continue;
            }

            let matching = |k: u64| job.cron.matches(k);

            match job.policy {
                Policy::Skip => {
                    if now % MINUTE_MS == 0 {
                        let k = now / MINUTE_MS;
                        if k >= first_index && matching(k) {
                            indexed.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: now,
                                },
                                job.priority,
                                job.insertion,
                            ));
                        }
                    }
                }
                Policy::Once => {
                    if let Some(k) = (first_index..=last_index).rev().find(|&k| matching(k)) {
                        indexed.push((
                            Occurrence {
                                id: job.id.clone(),
                                at: k * MINUTE_MS,
                            },
                            job.priority,
                            job.insertion,
                        ));
                    }
                }
                Policy::All => {
                    for k in first_index..=last_index {
                        if matching(k) {
                            indexed.push((
                                Occurrence {
                                    id: job.id.clone(),
                                    at: k * MINUTE_MS,
                                },
                                job.priority,
                                job.insertion,
                            ));
                        }
                    }
                }
            }
        }

        indexed.sort_by(|a, b| {
            a.0.at
                .cmp(&b.0.at)
                .then_with(|| b.1.cmp(&a.1))
                .then_with(|| a.2.cmp(&b.2))
        });

        self.last_poll = Some(now);
        for job in &mut self.jobs {
            if job.first_poll.is_some() {
                job.first_poll = None;
            }
        }

        Ok(indexed.into_iter().map(|(occurrence, _, _)| occurrence).collect())
    }
}
