```rust
use std::cmp::Ordering;
use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Item {
    pub key: String,
    pub priority: i64,
    pub value: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    EmptyKey,
    InvalidKey { at: usize },
    Duplicate { key: String },
    Missing { key: String },
    WrongDirection { current: i64, requested: i64 },
    Collision { key: String },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum CollisionPolicy {
    KeepSelf,
    TakeOther,
    Reject,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct MergeReport {
    pub moved: usize,
    pub replaced: usize,
    pub discarded: usize,
    pub evicted: Vec<Item>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Admission {
    Inserted,
    Evicted(Item),
    Discarded(Item),
}

#[derive(Clone)]
struct Entry {
    item: Item,
    age: u128,
}

pub struct IndexedHeap {
    entries: Vec<Entry>,
    indices: HashMap<String, usize>,
    capacity: Option<usize>,
    next_age: u128,
}

fn canonicalize(input: &str) -> Result<String, Error> {
    let bytes = input.as_bytes();
    let mut start = 0;
    let mut end = bytes.len();

    while start < end && bytes[start] == b' ' {
        start += 1;
    }
    while end > start && bytes[end - 1] == b' ' {
        end -= 1;
    }
    if start == end {
        return Err(Error::EmptyKey);
    }

    for i in start..end {
        let b = bytes[i];
        let valid = if i == start {
            b.is_ascii_alphabetic()
        } else {
            b.is_ascii_alphabetic() || b.is_ascii_digit() || b == b'-'
        };
        if !valid {
            return Err(Error::InvalidKey { at: i });
        }
    }

    Ok(input[start..end].to_ascii_lowercase())
}

fn compare_entries(a: &Entry, b: &Entry) -> Ordering {
    a.item
        .priority
        .cmp(&b.item.priority)
        .then_with(|| a.item.value.len().cmp(&b.item.value.len()))
        .then_with(|| a.age.cmp(&b.age))
}

impl IndexedHeap {
    pub fn new() -> Self {
        Self {
            entries: Vec::new(),
            indices: HashMap::new(),
            capacity: None,
            next_age: 0,
        }
    }

    pub fn with_capacity(capacity: usize) -> Self {
        Self {
            capacity: Some(capacity),
            ..Self::new()
        }
    }

    pub fn capacity(&self) -> Option<usize> {
        self.capacity
    }

    pub fn len(&self) -> usize {
        self.entries.len()
    }

    pub fn is_empty(&self) -> bool {
        self.len() == 0
    }

    fn rebuild_indices(&mut self) {
        self.indices.clear();
        for (i, entry) in self.entries.iter().enumerate() {
            self.indices.insert(entry.item.key.clone(), i);
        }
    }

    fn ordered_indices(&self) -> Vec<usize> {
        let mut indices: Vec<usize> = (0..self.entries.len()).collect();
        indices.sort_by(|&a, &b| compare_entries(&self.entries[a], &self.entries[b]));
        indices
    }

    fn remove_at(&mut self, index: usize) -> Entry {
        let entry = self.entries.remove(index);
        self.rebuild_indices();
        entry
    }

    fn age_for_new(&mut self) -> u128 {
        if self.next_age == u128::MAX {
            let mut ordered: Vec<usize> = (0..self.entries.len()).collect();
            ordered.sort_by_key(|&i| self.entries[i].age);
            for (age, index) in ordered.into_iter().enumerate() {
                self.entries[index].age = age as u128;
            }
            self.next_age = self.entries.len() as u128;
        }
        let age = self.next_age;
        self.next_age += 1;
        age
    }

    pub fn push(
        &mut self,
        key: &str,
        priority: i64,
        value: &str,
    ) -> Result<Admission, Error> {
        let key = canonicalize(key)?;
        if self.indices.contains_key(&key) {
            return Err(Error::Duplicate { key });
        }

        let proposed = Entry {
            item: Item {
                key,
                priority,
                value: value.to_owned(),
            },
            age: 0,
        };

        if self.capacity == Some(0) {
            return Ok(Admission::Discarded(proposed.item));
        }

        if let Some(limit) = self.capacity {
            if self.entries.len() >= limit {
                let worst_index = self
                    .ordered_indices()
                    .last()
                    .copied()
                    .expect("a full nonzero queue has an item");
                if compare_entries(&proposed, &self.entries[worst_index]) != Ordering::Less {
                    return Ok(Admission::Discarded(proposed.item));
                }
                let age = self.age_for_new();
                let mut proposed = proposed;
                proposed.age = age;
                let evicted = self.remove_at(worst_index).item;
                self.indices.insert(proposed.item.key.clone(), self.entries.len());
                self.entries.push(proposed);
                self.rebuild_indices();
                return Ok(Admission::Evicted(evicted));
            }
        }

        let mut proposed = proposed;
        proposed.age = self.age_for_new();
        self.indices.insert(proposed.item.key.clone(), self.entries.len());
        self.entries.push(proposed);
        Ok(Admission::Inserted)
    }

    pub fn peek(&self) -> Option<Item> {
        self.ordered_indices()
            .first()
            .map(|&i| self.entries[i].item.clone())
    }

    pub fn pop(&mut self) -> Option<Item> {
        let index = self.ordered_indices().first().copied()?;
        Some(self.remove_at(index).item)
    }

    pub fn get(&self, key: &str) -> Result<Option<Item>, Error> {
        let key = canonicalize(key)?;
        Ok(self
            .indices
            .get(&key)
            .map(|&i| self.entries[i].item.clone()))
    }

    pub fn decrease_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let index = *self
            .indices
            .get(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = self.entries[index].item.priority;
        if priority >= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn increase_key(&mut self, key: &str, priority: i64) -> Result<i64, Error> {
        let key = canonicalize(key)?;
        let index = *self
            .indices
            .get(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        let current = self.entries[index].item.priority;
        if priority <= current {
            return Err(Error::WrongDirection {
                current,
                requested: priority,
            });
        }
        self.entries[index].item.priority = priority;
        Ok(current)
    }

    pub fn replace_value(&mut self, key: &str, value: &str) -> Result<String, Error> {
        let key = canonicalize(key)?;
        let index = *self
            .indices
            .get(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        Ok(std::mem::replace(
            &mut self.entries[index].item.value,
            value.to_owned(),
        ))
    }

    pub fn remove(&mut self, key: &str) -> Result<Item, Error> {
        let key = canonicalize(key)?;
        let index = *self
            .indices
            .get(&key)
            .ok_or_else(|| Error::Missing { key: key.clone() })?;
        Ok(self.remove_at(index).item)
    }

    pub fn iter(&self) -> std::vec::IntoIter<Item> {
        self.ordered_indices()
            .into_iter()
            .map(|i| self.entries[i].item.clone())
            .collect::<Vec<_>>()
            .into_iter()
    }

    fn trim_to_capacity(&mut self) -> Vec<Item> {
        let Some(limit) = self.capacity else {
            return Vec::new();
        };
        let excess = self.entries.len().saturating_sub(limit);
        if excess == 0 {
            return Vec::new();
        }

        let mut order = self.ordered_indices();
        order.reverse();
        let mut evicted = Vec::with_capacity(excess);
        for index in order.into_iter().take(excess) {
            evicted.push(self.entries[index].item.clone());
        }
        let removed: std::collections::HashSet<String> =
            evicted.iter().map(|item| item.key.clone()).collect();
        self.entries
            .retain(|entry| !removed.contains(&entry.item.key));
        self.rebuild_indices();
        evicted
    }

    pub fn set_capacity(&mut self, capacity: Option<usize>) -> Vec<Item> {
        self.capacity = capacity;
        self.trim_to_capacity()
    }

    pub fn merge(
        &mut self,
        other: &mut Self,
        policy: CollisionPolicy,
    ) -> Result<MergeReport, Error> {
        let source_order = other.ordered_indices();
        if policy == CollisionPolicy::Reject {
            for index in &source_order {
                let key = &other.entries[*index].item.key;
                if self.indices.contains_key(key) {
                    return Err(Error::Collision { key: key.clone() });
                }
            }
        }

        let source: Vec<Entry> = source_order
            .into_iter()
            .map(|i| other.entries[i].clone())
            .collect();

        let mut report = MergeReport {
            moved: 0,
            replaced: 0,
            discarded: 0,
            evicted: Vec::new(),
        };

        for source_entry in source {
            let key = source_entry.item.key.clone();
            if let Some(&index) = self.indices.get(&key) {
                match policy {
                    CollisionPolicy::KeepSelf => report.discarded += 1,
                    CollisionPolicy::TakeOther => {
                        self.entries[index].item.priority = source_entry.item.priority;
                        self.entries[index].item.value = source_entry.item.value;
                        report.replaced += 1;
                    }
                    CollisionPolicy::Reject => unreachable!(),
                }
            } else {
                let mut entry = source_entry;
                entry.age = self.age_for_new();
                self.indices.insert(key, self.entries.len());
                self.entries.push(entry);
                report.moved += 1;
            }
        }

        other.entries.clear();
        other.indices.clear();
        report.evicted = self.trim_to_capacity();
        Ok(report)
    }
}

impl Default for IndexedHeap {
    fn default() -> Self {
        Self::new()
    }
}
```
