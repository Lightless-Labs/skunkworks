#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    if bytes.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut core = [0u64; 3];

    for (field, value) in core.iter_mut().enumerate() {
        let start = pos;
        while pos < bytes.len() && bytes[pos].is_ascii_digit() {
            pos += 1;
        }

        if start == pos {
            if pos == bytes.len()
                || matches!(bytes[pos], b'.' | b'-' | b'+')
            {
                return Err(VersionError::MissingCore { at: start });
            }
            return Err(VersionError::InvalidChar { at: start });
        }

        if pos - start > 1 && bytes[start] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }

        let mut number = 0u64;
        for &digit in &bytes[start..pos] {
            number = number
                .checked_mul(10)
                .and_then(|n| n.checked_add((digit - b'0') as u64))
                .ok_or(VersionError::Overflow { at: start })?;
        }
        *value = number;

        if field < 2 {
            if pos < bytes.len() && bytes[pos] == b'.' {
                pos += 1;
            } else if pos == bytes.len()
                || matches!(bytes[pos], b'-' | b'+')
            {
                return Err(VersionError::MissingCore { at: pos });
            } else {
                return Err(VersionError::InvalidChar { at: pos });
            }
        }
    }

    let (pre, build) = if pos == bytes.len() {
        (Vec::new(), Vec::new())
    } else {
        match bytes[pos] {
            b'-' => {
                let pre_start = pos + 1;
                let plus = bytes[pre_start..]
                    .iter()
                    .position(|&b| b == b'+')
                    .map(|i| pre_start + i);
                let pre_end = plus.unwrap_or(bytes.len());
                let pre = parse_identifiers(bytes, pre_start, pre_end, true)?;
                let build = if let Some(plus_pos) = plus {
                    parse_identifiers(bytes, plus_pos + 1, bytes.len(), false)?
                } else {
                    Vec::new()
                };
                (pre, build)
            }
            b'+' => {
                let build = parse_identifiers(bytes, pos + 1, bytes.len(), false)?;
                (Vec::new(), build)
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    };

    Ok(Version {
        major: core[0],
        minor: core[1],
        patch: core[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    bytes: &[u8],
    start: usize,
    end: usize,
    check_numeric_leading_zero: bool,
) -> Result<Vec<String>, VersionError> {
    let mut identifiers = Vec::new();
    let mut id_start = start;

    loop {
        let id_end = bytes[id_start..end]
            .iter()
            .position(|&b| b == b'.')
            .map(|i| id_start + i)
            .unwrap_or(end);

        if id_start == id_end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        for i in id_start..id_end {
            let b = bytes[i];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: i });
            }
        }

        if check_numeric_leading_zero
            && bytes[id_start..id_end].iter().all(u8::is_ascii_digit)
            && id_end - id_start > 1
            && bytes[id_start] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }

        identifiers.push(
            std::str::from_utf8(&bytes[id_start..id_end])
                .expect("validated ASCII identifier")
                .to_owned(),
        );

        if id_end == end {
            break;
        }
        id_start = id_end + 1;
    }

    Ok(identifiers)
}

pub fn compare_versions(a: &Version, b: &Version) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    let core_order = (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch));
    if core_order != Ordering::Equal {
        return core_order;
    }

    match (a.pre.is_empty(), b.pre.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (left, right) in a.pre.iter().zip(&b.pre) {
                let left_numeric = left.bytes().all(|b| b.is_ascii_digit());
                let right_numeric = right.bytes().all(|b| b.is_ascii_digit());

                let order = match (left_numeric, right_numeric) {
                    (true, true) => left
                        .len()
                        .cmp(&right.len())
                        .then_with(|| left.as_bytes().cmp(right.as_bytes())),
                    (true, false) => Ordering::Less,
                    (false, true) => Ordering::Greater,
                    (false, false) => left.as_bytes().cmp(right.as_bytes()),
                };

                if order != Ordering::Equal {
                    return order;
                }
            }
            a.pre.len().cmp(&b.pre.len())
        }
    }
}

#[derive(Clone, Debug)]
pub struct Range {
    comparators: Vec<Comparator>,
    prerelease_cores: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0usize;
    let mut comparators = Vec::new();
    let mut prerelease_cores = Vec::new();

    while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
        pos += 1;
    }

    if pos == bytes.len() {
        return Err(RangeError::Empty { at: 0 });
    }

    while pos < bytes.len() {
        let token_start = pos;
        while pos < bytes.len() && !is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;

        let (op, op_len) = match bytes[token_start] {
            b'=' => (Op::Equal, 1),
            b'<' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                (Op::LessEqual, 2)
            }
            b'<' => (Op::Less, 1),
            b'>' if token_start + 1 < token_end && bytes[token_start + 1] == b'=' => {
                (Op::GreaterEqual, 2)
            }
            b'>' => (Op::Greater, 1),
            _ => (Op::Equal, 0),
        };

        let operand_start = token_start + op_len;
        if operand_start >= token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        if !version.pre.is_empty() {
            prerelease_cores.push((version.major, version.minor, version.patch));
        }

        comparators.push(Comparator { op, version });

        while pos < bytes.len() && is_range_whitespace(bytes[pos]) {
            pos += 1;
        }
    }

    Ok(Range {
        comparators,
        prerelease_cores,
    })
}

fn is_range_whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn matches(v: &Version, r: &Range) -> bool {
    let admitted_prerelease = v.pre.is_empty()
        || r.prerelease_cores
            .iter()
            .any(|&core| core == (v.major, v.minor, v.patch));

    if !admitted_prerelease {
        return false;
    }

    r.comparators.iter().all(|comparator| {
        let ordering = compare_versions(v, &comparator.version);
        match comparator.op {
            Op::Equal => ordering == std::cmp::Ordering::Equal,
            Op::Less => ordering == std::cmp::Ordering::Less,
            Op::LessEqual => ordering != std::cmp::Ordering::Greater,
            Op::Greater => ordering == std::cmp::Ordering::Greater,
            Op::GreaterEqual => ordering != std::cmp::Ordering::Less,
        }
    })
}
