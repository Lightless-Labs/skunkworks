use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;

    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    if *pos == start {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    let digits = &bytes[start..*pos];
    if digits.len() > 1 && digits[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &digit in digits {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((digit - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }

    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut pos = ident_start;
        while pos < end && bytes[pos] != b'.' {
            let c = bytes[pos];
            if !(c.is_ascii_alphanumeric() || c == b'-') {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

        if pos == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        if is_pre {
            let ident = &bytes[ident_start..pos];
            if ident.iter().all(u8::is_ascii_digit)
                && ident.len() > 1
                && ident[0] == b'0'
            {
                return Err(VersionError::LeadingZero { at: ident_start });
            }
        }

        result.push(s[ident_start..pos].to_owned());

        if pos == end {
            break;
        }
        ident_start = pos + 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let bytes = s.as_bytes();
    let mut pos = 0;

    let major = parse_core_field(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= bytes.len() || bytes[pos] != b'.' {
            if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;

        let value = parse_core_field(s, &mut pos)?;
        if _ == 0 {
            // The second core value is minor; parse the third below.
            let minor = value;
            if pos >= bytes.len() || bytes[pos] != b'.' {
                if pos >= bytes.len() || matches!(bytes[pos], b'-' | b'+') {
                    return Err(VersionError::MissingCore { at: pos });
                }
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
            let patch = parse_core_field(s, &mut pos)?;

            let (pre, build) = parse_suffixes(s, pos)?;
            return Ok(Version {
                major,
                minor,
                patch,
                pre,
                build,
            });
        }
    }

    unreachable!()
}

fn parse_suffixes(s: &str, mut pos: usize) -> Result<(Vec<String>, Vec<String>), VersionError> {
    let bytes = s.as_bytes();
    let len = bytes.len();

    if pos == len {
        return Ok((Vec::new(), Vec::new()));
    }

    if bytes[pos] == b'-' {
        let pre_start = pos + 1;
        let build_marker = bytes[pre_start..]
            .iter()
            .position(|&b| b == b'+')
            .map(|offset| pre_start + offset);

        let pre_end = build_marker.unwrap_or(len);
        let pre = parse_identifiers(s, pre_start, pre_end, true)?;

        if let Some(plus) = build_marker {
            let build_start = plus + 1;
            let build = parse_identifiers(s, build_start, len, false)?;
            return Ok((pre, build));
        }
        return Ok((pre, Vec::new()));
    }

    if bytes[pos] == b'+' {
        pos += 1;
        let build = parse_identifiers(s, pos, len, false)?;
        return Ok((Vec::new(), build));
    }

    Err(VersionError::InvalidChar { at: pos })
}

fn is_numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(u8::is_ascii_digit)
}

fn compare_pre_identifier(a: &str, b: &str) -> Ordering {
    let a_num = is_numeric_identifier(a);
    let b_num = is_numeric_identifier(b);

    match (a_num, b_num) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a
            .len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (left, right) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_pre_identifier(left, right);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: CompareOp,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum CompareOp {
    Equal,
    Less,
    LessEqual,
    Greater,
    GreaterEqual,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Comparator>>,
}

fn is_range_space(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let bytes = s.as_bytes();
    let mut pos = 0;
    let mut comparators = Vec::new();

    while pos < bytes.len() {
        while pos < bytes.len() && is_range_space(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }

        let token_start = pos;
        while pos < bytes.len() && !is_range_space(bytes[pos]) {
            pos += 1;
        }
        let token_end = pos;

        let mut operand_start = token_start;
        let op = match bytes[token_start] {
            b'=' => {
                operand_start += 1;
                CompareOp::Equal
            }
            b'<' => {
                operand_start += 1;
                if operand_start < token_end && bytes[operand_start] == b'=' {
                    operand_start += 1;
                    CompareOp::LessEqual
                } else {
                    CompareOp::Less
                }
            }
            b'>' => {
                operand_start += 1;
                if operand_start < token_end && bytes[operand_start] == b'=' {
                    operand_start += 1;
                    CompareOp::GreaterEqual
                } else {
                    CompareOp::Greater
                }
            }
            _ => CompareOp::Equal,
        };

        if operand_start == token_end || !bytes[operand_start].is_ascii_digit() {
            return Err(RangeError::Syntax { at: token_start });
        }

        let operand = &s[operand_start..token_end];
        let version = parse_version(operand).map_err(|error| RangeError::Version {
            at: operand_start,
            error,
        })?;

        comparators.push(Comparator { op, version });
    }

    if comparators.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    Ok(Range {
        arms: vec![comparators],
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let prerelease_admitted = v.pre.is_empty()
            || arm.iter().any(|comparator| {
                !comparator.version.pre.is_empty()
                    && (v.major, v.minor, v.patch)
                        == (
                            comparator.version.major,
                            comparator.version.minor,
                            comparator.version.patch,
                        )
            });

        prerelease_admitted
            && arm.iter().all(|comparator| {
                let ordering = compare_versions(v, &comparator.version);
                match comparator.op {
                    CompareOp::Equal => ordering == Ordering::Equal,
                    CompareOp::Less => ordering == Ordering::Less,
                    CompareOp::LessEqual => ordering != Ordering::Greater,
                    CompareOp::Greater => ordering == Ordering::Greater,
                    CompareOp::GreaterEqual => ordering != Ordering::Less,
                }
            })
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    fn pv(s: &str) -> Version {
        parse_version(s).unwrap()
    }

    fn pr(s: &str) -> Range {
        parse_range(s).unwrap()
    }

    #[test]
    fn public_version() {
        let v = pv("2.1.3-rc.1+build.7");
        assert_eq!((v.major, v.minor, v.patch), (2, 1, 3));
        assert_eq!(v.pre, vec!["rc", "1"]);
        assert_eq!(v.build, vec!["build", "7"]);
        assert_eq!(
            compare_versions(&pv("2.1.3-alpha"), &pv("2.1.3-beta")),
            Ordering::Less
        );
    }

    #[test]
    fn public_precedence() {
        assert_eq!(
            compare_versions(&pv("2.1.3-rc"), &pv("2.1.3")),
            Ordering::Less
        );
    }

    #[test]
    fn public_comparators() {
        let r = pr(">=2.1.0 <3.0.0");
        assert!(matches(&pv("2.5.0"), &r));
        assert!(!matches(&pv("3.0.0"), &r));
        assert!(matches(
            &pv("2.1.3-rc"),
            &pr(">=2.1.3-beta <2.1.3")
        ));
    }

    #[test]
    fn public_exact() {
        assert!(matches(&pv("2.1.3"), &pr("=2.1.3")));
        assert!(!matches(&pv("2.1.4"), &pr("=2.1.3")));
    }
}
