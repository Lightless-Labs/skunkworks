#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Job {
    pub id: String,
    pub priority: u8,
    pub dependencies: Vec<String>,
    pub max_attempts: u8,
    pub release_at: u64,
    pub duration: u64,
    pub tag: Option<String>,
    pub failures: u8,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum State {
    Waiting { ready_at: u64, attempt: u8 },
    Running { attempt: u8, finish_at: u64 },
    Succeeded,
    Failed,
    Blocked,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    Config,
    BadId(usize),
    BadPriority(usize),
    BadAttempts(usize),
    BadTime(usize),
    BadTag(usize),
    Duplicate(usize),
    DuplicateDependency(usize, usize),
    MissingDependency(usize, usize),
    Cycle,
    TimeReversed,
    TimeRange,
    Unknown,
}

#[derive(Clone)]
struct Record {
    job: Job,
    dependencies: Vec<usize>,
    state: State,
    waiting_since: u64,
    remaining: Option<u64>,
    segment_order: u64,
}

pub struct Pool {
    records: Vec<Record>,
    capacity: usize,
    aging_ms: u64,
    retry_base: u64,
    retry_cap: u64,
    horizon: u64,
    first_run: bool,
    sequence: u64,
}

impl Pool {
    pub fn new(
        jobs: Vec<Job>,
        capacity: usize,
        aging_ms: u64,
        retry_base: u64,
        retry_cap: u64,
    ) -> Result<Self, Error> {
        if !(1..=16).contains(&capacity)
            || retry_base == 0
            || retry_base > retry_cap
            || retry_cap > 1_000_000_000
        {
            return Err(Error::Config);
        }

        let mut indices = std::collections::HashMap::new();
        for (i, job) in jobs.iter().enumerate() {
            if !valid_token(&job.id, 1, 16) {
                return Err(Error::BadId(i));
            }
            if job.priority > 9 {
                return Err(Error::BadPriority(i));
            }
            if !(1..=32).contains(&job.max_attempts) {
                return Err(Error::BadAttempts(i));
            }
            if job.release_at > 1_000_000_000
                || job.duration == 0
                || job.duration > 1_000_000_000
            {
                return Err(Error::BadTime(i));
            }
            if job
                .tag
                .as_ref()
                .is_some_and(|tag| !valid_token(tag, 1, 8))
            {
                return Err(Error::BadTag(i));
            }
            if indices.insert(job.id.clone(), i).is_some() {
                return Err(Error::Duplicate(i));
            }
        }

        let mut dependencies = Vec::with_capacity(jobs.len());
        for (i, job) in jobs.iter().enumerate() {
            let mut seen = std::collections::HashSet::new();
            let mut resolved = Vec::with_capacity(job.dependencies.len());
            for (j, spelling) in job.dependencies.iter().enumerate() {
                if !seen.insert(spelling) {
                    return Err(Error::DuplicateDependency(i, j));
                }
                match indices.get(spelling) {
                    Some(&index) => resolved.push(index),
                    None => return Err(Error::MissingDependency(i, j)),
                }
            }
            dependencies.push(resolved);
        }

        fn visit(
            node: usize,
            deps: &[Vec<usize>],
            marks: &mut [u8],
        ) -> bool {
            if marks[node] == 1 {
                return false;
            }
            if marks[node] == 2 {
                return true;
            }
            marks[node] = 1;
            for &dep in &deps[node] {
                if !visit(dep, deps, marks) {
                    return false;
                }
            }
            marks[node] = 2;
            true
        }

        let mut marks = vec![0; jobs.len()];
        for node in 0..jobs.len() {
            if !visit(node, &dependencies, &mut marks) {
                return Err(Error::Cycle);
            }
        }

        let records = jobs
            .into_iter()
            .enumerate()
            .map(|(i, job)| Record {
                state: State::Waiting {
                    ready_at: job.release_at,
                    attempt: 0,
                },
                waiting_since: job.release_at,
                remaining: None,
                segment_order: 0,
                job,
                dependencies: dependencies[i].clone(),
            })
            .collect();

        Ok(Self {
            records,
            capacity,
            aging_ms,
            retry_base,
            retry_cap,
            horizon: 0,
            first_run: true,
            sequence: 0,
        })
    }

    pub fn status(&self, id: &str) -> Result<State, Error> {
        self.records
            .iter()
            .find(|record| record.job.id == id)
            .map(|record| record.state.clone())
            .ok_or(Error::Unknown)
    }

    pub fn run(&mut self, until: u64) -> Result<Vec<String>, Error> {
        if !self.first_run && until < self.horizon {
            return Err(Error::TimeReversed);
        }
        if until > 1_000_000_000 {
            return Err(Error::TimeRange);
        }
        if !self.first_run && until == self.horizon {
            return Ok(Vec::new());
        }

        let mut lines = Vec::new();
        let mut time = if self.first_run { 0 } else { self.horizon };

        loop {
            let mut completions: Vec<usize> = self
                .records
                .iter()
                .enumerate()
                .filter_map(|(i, record)| match record.state {
                    State::Running { finish_at, .. } if finish_at == time => Some(i),
                    _ => None,
                })
                .collect();
            completions.sort_by(|&a, &b| self.records[a].job.id.cmp(&self.records[b].job.id));

            for i in completions {
                let attempt = match self.records[i].state {
                    State::Running { attempt, .. } => attempt,
                    _ => continue,
                };
                if attempt > self.records[i].job.failures {
                    self.records[i].state = State::Succeeded;
                    lines.push(format!("{} finish {} {} ok", time, self.records[i].job.id, attempt));
                } else {
                    lines.push(format!("{} finish {} {} fail", time, self.records[i].job.id, attempt));
                    self.records[i].remaining = None;
                    if attempt >= self.records[i].job.max_attempts {
                        self.records[i].state = State::Failed;
                    } else {
                        let shift = u32::from(attempt - 1);
                        let delay = self.retry_base.checked_shl(shift).unwrap_or(u64::MAX).min(self.retry_cap);
                        let ready_at = time.saturating_add(delay);
                        self.records[i].state = State::Waiting { ready_at, attempt };
                        self.records[i].waiting_since = ready_at;
                        lines.push(format!("{} retry {} {} {}", time, self.records[i].job.id, attempt, ready_at));
                    }
                }
            }

            self.propagate_blocked(time, &mut lines);

            self.schedule(time, &mut lines);

            if time == until {
                break;
            }

            let next = self.records.iter().filter_map(|record| match record.state {
                State::Running { finish_at, .. } if finish_at > time => Some(finish_at),
                State::Waiting { ready_at, .. } if ready_at > time => Some(ready_at),
                _ => None,
            }).min();

            match next {
                Some(next_time) if next_time <= until => time = next_time,
                _ => break,
            }
        }

        self.horizon = until;
        self.first_run = false;
        Ok(lines)
    }

    fn propagate_blocked(&mut self, time: u64, lines: &mut Vec<String>) {
        loop {
            let mut newly = Vec::new();
            for (i, record) in self.records.iter().enumerate() {
                if !matches!(record.state, State::Waiting { .. }) {
                    continue;
                }
                if let Some(&failed) = record.dependencies.iter().find(|&&dep| {
                    matches!(self.records[dep].state, State::Failed | State::Blocked)
                }) {
                    newly.push((i, failed));
                }
            }
            if newly.is_empty() {
                break;
            }
            newly.sort_by(|(a, _), (b, _)| self.records[*a].job.id.cmp(&self.records[*b].job.id));
            for (i, failed) in newly {
                self.records[i].state = State::Blocked;
                lines.push(format!(
                    "{} skip {} {}",
                    time, self.records[i].job.id, self.records[failed].job.id
                ));
            }
        }
    }

    fn schedule(&mut self, time: u64, lines: &mut Vec<String>) {
        loop {
            let mut candidates: Vec<usize> = self.records.iter().enumerate().filter_map(|(i, r)| {
                if let State::Waiting { ready_at, .. } = r.state {
                    if ready_at <= time && r.dependencies.iter().all(|&d| matches!(self.records[d].state, State::Succeeded)) {
                        return Some(i);
                    }
                }
                None
            }).collect();

            candidates.sort_by(|&a, &b| {
                let pa = self.effective_priority(a, time);
                let pb = self.effective_priority(b, time);
                pb.cmp(&pa).then_with(|| a.cmp(&b))
            });

            let mut chosen = None;
            for candidate in candidates {
                let tag = self.records[candidate].job.tag.as_ref();
                let running: Vec<usize> = self.records.iter().enumerate().filter_map(|(i, r)| {
                    if matches!(r.state, State::Running { .. }) { Some(i) } else { None }
                }).collect();

                let slot_free = running.len() < self.capacity;
                let tag_free = tag.map_or(true, |tag| {
                    !running.iter().any(|&i| self.records[i].job.tag.as_ref() == Some(tag))
                });

                if slot_free && tag_free {
                    chosen = Some((candidate, None));
                    break;
                }

                let mut victims: Vec<usize> = running.iter().copied().filter(|&victim| {
                    let victim_tag = self.records[victim].job.tag.as_ref();
                    let after_slot = running.len() - 1 < self.capacity;
                    let after_tag = tag.map_or(true, |tag| {
                        victim_tag == Some(tag)
                            || !running.iter().any(|&i| i != victim && self.records[i].job.tag.as_ref() == Some(tag))
                    });
                    after_slot
                        && after_tag
                        && self.records[candidate].job.priority >= self.records[victim].job.priority.saturating_add(2)
                }).collect();

                victims.sort_by(|&a, &b| {
                    self.records[a].job.priority.cmp(&self.records[b].job.priority)
                        .then_with(|| self.records[b].segment_order.cmp(&self.records[a].segment_order))
                        .then_with(|| self.records[b].job.id.cmp(&self.records[a].job.id))
                });

                if let Some(victim) = victims.first().copied() {
                    chosen = Some((candidate, Some(victim)));
                    break;
                }
            }

            let Some((candidate, victim)) = chosen else {
                break;
            };

            if let Some(victim) = victim {
                let remaining = match self.records[victim].state {
                    State::Running { finish_at, .. } => finish_at - time,
                    _ => 0,
                };
                let attempt = match self.records[victim].state {
                    State::Running { attempt, .. } => attempt,
                    _ => 0,
                };
                self.records[victim].state = State::Waiting { ready_at: time, attempt };
                self.records[victim].waiting_since = time;
                self.records[victim].remaining = Some(remaining);
                lines.push(format!(
                    "{} preempt {} by {}",
                    time, self.records[victim].job.id, self.records[candidate].job.id
                ));
            }

            let (attempt, remaining) = match self.records[candidate].state {
                State::Waiting { attempt, .. } => {
                    if let Some(remaining) = self.records[candidate].remaining.take() {
                        (attempt, remaining)
                    } else {
                        (attempt.saturating_add(1), self.records[candidate].job.duration)
                    }
                }
                _ => continue,
            };
            self.sequence = self.sequence.saturating_add(1);
            self.records[candidate].segment_order = self.sequence;
            self.records[candidate].state = State::Running {
                attempt,
                finish_at: time.saturating_add(remaining),
            };
            lines.push(format!(
                "{} start {} {} {}",
                time, self.records[candidate].job.id, attempt, remaining
            ));
        }
    }

    fn effective_priority(&self, index: usize, time: u64) -> u8 {
        let priority = self.records[index].job.priority;
        if self.aging_ms == 0 {
            return priority;
        }
        let waited = time.saturating_sub(self.records[index].waiting_since);
        let bonus = (waited / self.aging_ms).min(9) as u8;
        priority.saturating_add(bonus).min(9)
    }
}

fn valid_token(value: &str, min: usize, max: usize) -> bool {
    let bytes = value.as_bytes();
    (min..=max).contains(&bytes.len())
        && bytes.iter().all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}
