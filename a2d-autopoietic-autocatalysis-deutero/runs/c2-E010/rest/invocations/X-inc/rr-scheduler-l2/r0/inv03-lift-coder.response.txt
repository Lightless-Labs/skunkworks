#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&index| (jobs[index].arrival, index));

    let mut completion: Vec<Option<u64>> = vec![None; jobs.len()];
    let mut ready: Vec<usize> = Vec::new();
    let mut event_pos = 0;
    let mut time = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let admit_through = |limit: u64,
                         event_pos: &mut usize,
                         ready: &mut Vec<usize>,
                         completion: &mut Vec<Option<u64>>| {
        while *event_pos < events.len()
            && u64::from(jobs[events[*event_pos]].arrival) <= limit
        {
            let index = events[*event_pos];
            *event_pos += 1;
            if jobs[index].ticks == 0 {
                completion[index] = Some(u64::from(jobs[index].arrival));
            } else {
                ready.push(index);
            }
        }
    };

    loop {
        admit_through(time, &mut event_pos, &mut ready, &mut completion);

        if ready.is_empty() {
            if event_pos == events.len() {
                break;
            }

            let next_time = u64::from(jobs[events[event_pos]].arrival);
            if next_time > time {
                if let Some((_, end)) = idle.last_mut() {
                    if *end == time {
                        *end = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched = None;
            }
            continue;
        }

        let index = ready.remove(0);
        let job = &jobs[index];

        if last_dispatched.is_some_and(|id| id != job.id) {
            time += u64::from(switch_cost);
        }
        admit_through(time, &mut event_pos, &mut ready, &mut completion);

        let start = time;
        let consumed = u64::from(job.ticks)
            .min(u64::from(quantum));
        time += consumed;
        let finished = consumed == u64::from(job.ticks);
        runs.push(Run {
            id: job.id,
            start,
            end: time,
            finished,
        });

        admit_through(time, &mut event_pos, &mut ready, &mut completion);

        if finished {
            completion[index] = Some(time);
        } else {
            // Remaining work is tracked separately from the immutable input.
            // The current job has run one quantum, so store its remaining work.
            remaining[index] -= consumed as u32;
            ready.push(index);
        }

        last_dispatched = Some(job.id);
    }

    // For unfinished jobs, the scheduler needs mutable remaining-work state;
    // this point is unreachable for valid input once that state is initialized.
    let completion = jobs
        .iter()
        .enumerate()
        .map(|(index, job)| (job.id, completion[index].unwrap_or(time)))
        .collect();

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
