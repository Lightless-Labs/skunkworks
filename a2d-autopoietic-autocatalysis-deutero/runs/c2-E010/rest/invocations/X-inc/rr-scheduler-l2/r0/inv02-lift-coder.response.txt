#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub ticks: u32,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub completion: Vec<(u32, u64)>,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Quantum,
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    Ticks { index: usize },
}

pub fn schedule(jobs: &[Job], quantum: u32, switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=1000).contains(&quantum) {
        return Err(Error::Quantum);
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.ticks > 1000 {
            return Err(Error::Ticks { index });
        }
    }

    let mut events: Vec<usize> = (0..jobs.len()).collect();
    events.sort_by_key(|&i| (jobs[i].arrival, i));

    let mut completion = vec![(0u32, 0u64); jobs.len()];
    let mut completed = vec![false; jobs.len()];
    let mut remaining: Vec<u32> = jobs.iter().map(|job| job.ticks).collect();
    let mut ready: std::collections::VecDeque<usize> = std::collections::VecDeque::new();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut time = 0u64;
    let mut event_pos = 0usize;
    let mut last_dispatched: Option<u32> = None;
    let mut completed_count = 0usize;

    while completed_count < jobs.len() {
        while event_pos < events.len()
            && u64::from(jobs[events[event_pos]].arrival) <= time
        {
            let i = events[event_pos];
            event_pos += 1;

            if jobs[i].ticks == 0 {
                completion[i] = (jobs[i].id, u64::from(jobs[i].arrival));
                completed[i] = true;
                completed_count += 1;
            } else {
                ready.push_back(i);
            }
        }

        if let Some(i) = ready.pop_front() {
            let id = jobs[i].id;
            if last_dispatched.is_some_and(|last| last != id) {
                time += u64::from(switch_cost);
            }

            while event_pos < events.len()
                && u64::from(jobs[events[event_pos]].arrival) <= time
            {
                let arrived = events[event_pos];
                event_pos += 1;

                if jobs[arrived].ticks == 0 {
                    completion[arrived] = (jobs[arrived].id, u64::from(jobs[arrived].arrival));
                    completed[arrived] = true;
                    completed_count += 1;
                } else {
                    ready.push_back(arrived);
                }
            }

            let start = time;
            let consumed = remaining[i].min(quantum);
            time += u64::from(consumed);
            remaining[i] -= consumed;
            let finished = remaining[i] == 0;

            runs.push(Run {
                id,
                start,
                end: time,
                finished,
            });

            while event_pos < events.len()
                && u64::from(jobs[events[event_pos]].arrival) <= time
            {
                let arrived = events[event_pos];
                event_pos += 1;

                if jobs[arrived].ticks == 0 {
                    completion[arrived] = (jobs[arrived].id, u64::from(jobs[arrived].arrival));
                    completed[arrived] = true;
                    completed_count += 1;
                } else {
                    ready.push_back(arrived);
                }
            }

            if finished {
                completion[i] = (id, time);
                completed[i] = true;
                completed_count += 1;
            } else {
                ready.push_back(i);
            }
            last_dispatched = Some(id);
        } else if event_pos < events.len() {
            let next_time = u64::from(jobs[events[event_pos]].arrival);
            if next_time > time {
                if let Some(last_idle) = idle.last_mut() {
                    if last_idle.1 == time {
                        last_idle.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatched = None;
            }
        }
    }

    Ok(Schedule {
        runs,
        idle,
        completion,
        end: time,
    })
}
