#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WordsError {
    Empty,
    InvalidByte { index: usize },
    Separator { index: usize },
    InvalidWord { index: usize },
    Range,
    NonCanonical { index: usize },
}

const LIMIT: u128 = 1_000_000_000_000_000_000;

const SMALL: [&str; 20] = [
    "zero", "one", "two", "three", "four", "five", "six", "seven", "eight", "nine",
    "ten", "eleven", "twelve", "thirteen", "fourteen", "fifteen", "sixteen",
    "seventeen", "eighteen", "nineteen",
];

const TENS: [&str; 8] = [
    "twenty", "thirty", "forty", "fifty", "sixty", "seventy", "eighty", "ninety",
];

const SCALES: [(&str, u128); 6] = [
    ("thousand", 1_000),
    ("million", 1_000_000),
    ("billion", 1_000_000_000),
    ("trillion", 1_000_000_000_000),
    ("quadrillion", 1_000_000_000_000_000),
    ("quintillion", 1_000_000_000_000_000_000),
];

fn encode_group(n: u16) -> String {
    let mut result = String::new();
    if n >= 100 {
        result.push_str(SMALL[(n / 100) as usize]);
        result.push_str(" hundred");
        let remainder = n % 100;
        if remainder != 0 {
            result.push_str(" and ");
            result.push_str(&encode_group(remainder));
        }
        return result;
    }

    if n >= 20 {
        result.push_str(TENS[(n / 10 - 2) as usize]);
        if n % 10 != 0 {
            result.push('-');
            result.push_str(SMALL[(n % 10) as usize]);
        }
        return result;
    }

    result.push_str(SMALL[n as usize]);
    result
}

pub fn words_encode(value: i128) -> Result<String, WordsError> {
    let magnitude = value.unsigned_abs();
    if magnitude > LIMIT {
        return Err(WordsError::Range);
    }
    if magnitude == 0 {
        return Ok("zero".to_string());
    }

    let mut remaining = magnitude;
    let mut groups: Vec<(usize, u16)> = Vec::new();
    for index in (0..=6).rev() {
        let divisor = 1000_u128.pow(index as u32);
        let group = (remaining / divisor) % 1000;
        if group != 0 {
            groups.push((index, group as u16));
        }
    }

    let mut result = String::new();
    if value < 0 {
        result.push_str("minus ");
    }

    for (position, &(scale_index, group)) in groups.iter().enumerate() {
        if position > 0 {
            result.push(' ');
            if scale_index == 0
                && group < 100
                && groups[..position].iter().any(|(index, _)| *index > 0)
            {
                result.push_str("and ");
            }
        }

        result.push_str(&encode_group(group));
        if scale_index > 0 {
            result.push(' ');
            result.push_str(SCALES[scale_index - 1].0);
        }
    }

    Ok(result)
}

fn dictionary_value(word: &str) -> Option<Word> {
    if word == "and" {
        return Some(Word::And);
    }
    if word == "hundred" {
        return Some(Word::Hundred);
    }
    for (i, spelling) in SMALL.iter().enumerate() {
        if word == *spelling {
            return Some(Word::Number(i as u128));
        }
    }
    for (i, spelling) in TENS.iter().enumerate() {
        if word == *spelling {
            return Some(Word::Number((i as u128 + 2) * 10));
        }
    }
    for (spelling, value) in SCALES {
        if word == spelling {
            return Some(Word::Scale(value));
        }
    }
    None
}

#[derive(Clone, Copy)]
enum Word {
    Number(u128),
    And,
    Hundred,
    Scale(u128),
}

pub fn words_decode(input: &str) -> Result<i128, WordsError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(WordsError::Empty);
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if !(byte.is_ascii_lowercase() || byte == b' ' || byte == b'-') {
            return Err(WordsError::InvalidByte { index });
        }
    }

    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            if index == 0
                || index + 1 == bytes.len()
                || !bytes[index - 1].is_ascii_lowercase()
                || !bytes[index + 1].is_ascii_lowercase()
            {
                return Err(WordsError::Separator { index });
            }
        }
    }

    let mut tokens: Vec<(&str, usize)> = Vec::new();
    let mut start = 0;
    for (index, &byte) in bytes.iter().enumerate() {
        if byte == b' ' || byte == b'-' {
            tokens.push((&input[start..index], start));
            start = index + 1;
        }
    }
    tokens.push((&input[start..], start));

    let mut words = Vec::with_capacity(tokens.len());
    for (i, &(token, index)) in tokens.iter().enumerate() {
        if token == "minus" {
            if i != 0 {
                return Err(WordsError::InvalidWord { index });
            }
            words.push(None);
        } else if let Some(word) = dictionary_value(token) {
            words.push(Some(word));
        } else {
            return Err(WordsError::InvalidWord { index });
        }
    }

    let negative = tokens.first().map(|(token, _)| *token == "minus").unwrap_or(false);
    let arithmetic_start = usize::from(negative);
    let mut total: u128 = 0;
    let mut group: u128 = 0;

    for word in words.iter().skip(arithmetic_start) {
        match word {
            Some(Word::Number(value)) => group += value,
            Some(Word::And) => {}
            Some(Word::Hundred) => group *= 100,
            Some(Word::Scale(scale)) => {
                total += group * scale;
                group = 0;
            }
            None => unreachable!(),
        }
        if total + group > LIMIT {
            return Err(WordsError::Range);
        }
    }

    let magnitude = total + group;
    let value = if negative && magnitude != 0 {
        -(magnitude as i128)
    } else {
        magnitude as i128
    };
    let canonical = words_encode(value)?;
    if canonical.as_bytes() != bytes {
        let differing = canonical
            .as_bytes()
            .iter()
            .zip(bytes.iter())
            .position(|(a, b)| a != b)
            .unwrap_or_else(|| canonical.len().min(bytes.len()));
        return Err(WordsError::NonCanonical { index: differing });
    }
    Ok(value)
}
