#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<u8>; 5],
    day_stars: [bool; 2],
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        self.selected[field_index(field)].clone()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn domain(field: Field) -> (u8, u8, u32) {
    match field {
        Field::Minute => (0, 59, 60),
        Field::Hour => (0, 23, 24),
        Field::DayOfMonth => (1, 31, 31),
        Field::Month => (1, 12, 12),
        Field::DayOfWeek => (0, 6, 7),
    }
}

fn error(field: Field, offset: usize, kind: ErrorKind) -> ParseError {
    match kind {
        ErrorKind::InvalidCharacter => ParseError::InvalidCharacter { field, offset },
        ErrorKind::EmptyItem => ParseError::EmptyItem { field, offset },
        ErrorKind::InvalidSyntax => ParseError::InvalidSyntax { field, offset },
        ErrorKind::LeadingZero => ParseError::LeadingZero { field, offset },
        ErrorKind::NumberOverflow => ParseError::NumberOverflow { field, offset },
        ErrorKind::OutOfRange => ParseError::OutOfRange { field, offset },
        ErrorKind::ReversedRange => ParseError::ReversedRange { field, offset },
        ErrorKind::BadStep => ParseError::BadStep { field, offset },
    }
}

#[derive(Clone, Copy)]
enum ErrorKind {
    InvalidCharacter,
    EmptyItem,
    InvalidSyntax,
    LeadingZero,
    NumberOverflow,
    OutOfRange,
    ReversedRange,
    BadStep,
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(error(field, start, ErrorKind::InvalidSyntax));
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(field, start, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or_else(|| error(field, start, ErrorKind::NumberOverflow))?;
    }

    if is_step {
        let (_, _, width) = domain(field);
        if value == 0 || value > width {
            return Err(error(field, start, ErrorKind::BadStep));
        }
    } else {
        let (min, max, _) = domain(field);
        if value < min as u32 || value > max as u32 {
            return Err(error(field, start, ErrorKind::OutOfRange));
        }
    }

    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool; 256],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(error(field, item_start, ErrorKind::EmptyItem));
    }

    let item = &bytes[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    let (base_end, step_range) = match slash_positions.as_slice() {
        [] => (item.len(), None),
        [slash] => (*slash, Some((*slash + 1, item.len()))),
        _ => return Err(error(field, item_start, ErrorKind::InvalidSyntax)),
    };

    if base_end == 0 || step_range.is_some_and(|(s, e)| s == e) {
        return Err(error(field, item_start, ErrorKind::InvalidSyntax));
    }

    let base = &item[..base_end];
    let (base_kind, first, second) = if base == b"*" {
        (BaseKind::Star, None, None)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        match hyphens.as_slice() {
            [] => {
                if base.iter().all(u8::is_ascii_digit) {
                    (BaseKind::Single, Some((0, base.len())), None)
                } else {
                    return Err(error(field, item_start, ErrorKind::InvalidSyntax));
                }
            }
            [hyphen] => {
                let left = &base[..*hyphen];
                let right = &base[*hyphen + 1..];
                if left.is_empty()
                    || right.is_empty()
                    || !left.iter().all(u8::is_ascii_digit)
                    || !right.iter().all(u8::is_ascii_digit)
                {
                    return Err(error(field, item_start, ErrorKind::InvalidSyntax));
                }
                (
                    BaseKind::Range,
                    Some((0, *hyphen)),
                    Some((*hyphen + 1, base.len())),
                )
            }
            _ => return Err(error(field, item_start, ErrorKind::InvalidSyntax)),
        }
    };

    let (min, max, _) = domain(field);
    let (start, end) = match base_kind {
        BaseKind::Star => (min, max),
        BaseKind::Single => {
            let (s, e) = first.unwrap();
            let value = parse_number(base, s, e, field, false)?;
            let end = if step_range.is_some() { max } else { value as u8 };
            (value as u8, end)
        }
        BaseKind::Range => {
            let (s1, e1) = first.unwrap();
            let (s2, e2) = second.unwrap();
            let start = parse_number(base, s1, e1, field, false)?;
            let end = parse_number(base, s2, e2, field, false)?;
            if start > end {
                return Err(error(field, item_start + s2, ErrorKind::ReversedRange));
            }
            (start as u8, end as u8)
        }
    };

    let step = if let Some((s, e)) = step_range {
        parse_number(item, s, e, field, true)?
    } else {
        1
    };

    let mut value = start as u32;
    while value <= end as u32 {
        selected[value as usize] = true;
        value += step;
    }

    Ok(())
}

#[derive(Clone, Copy)]
enum BaseKind {
    Star,
    Single,
    Range,
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected_sets = [
        [false; 256],
        [false; 256],
        [false; 256],
        [false; 256],
        [false; 256],
    ];
    let mut day_stars = [false; 2];

    for (index, (&field_text, &field)) in fields.iter().zip(field_kinds.iter()).enumerate() {
        let bytes = field_text.as_bytes();

        for (offset, &byte) in bytes.iter().enumerate() {
            if !matches!(byte, b'0'..=b'9' | b'*' | b'-' | b'/' | b',') {
                return Err(error(field, offset, ErrorKind::InvalidCharacter));
            }
        }

        if index == 2 && field_text == "*" {
            day_stars[0] = true;
        }
        if index == 4 && field_text == "*" {
            day_stars[1] = true;
        }

        let mut item_start = 0;
        for (offset, &byte) in bytes.iter().enumerate() {
            if byte == b',' {
                parse_item(bytes, item_start, offset, field, &mut selected_sets[index])?;
                item_start = offset + 1;
            }
        }
        parse_item(
            bytes,
            item_start,
            bytes.len(),
            field,
            &mut selected_sets[index],
        )?;
    }

    let selected = std::array::from_fn(|index| {
        selected_sets[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_selected)| is_selected.then_some(value as u8))
            .collect()
    });

    Ok(Cron {
        selected,
        day_stars,
    })
}
