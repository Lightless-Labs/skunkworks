#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Field {
    Minute,
    Hour,
    DayOfMonth,
    Month,
    DayOfWeek,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    FieldCount { found: usize },
    InvalidCharacter { field: Field, offset: usize },
    EmptyItem { field: Field, offset: usize },
    InvalidSyntax { field: Field, offset: usize },
    LeadingZero { field: Field, offset: usize },
    NumberOverflow { field: Field, offset: usize },
    OutOfRange { field: Field, offset: usize },
    ReversedRange { field: Field, offset: usize },
    BadStep { field: Field, offset: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cron {
    selected: [Vec<bool>; 5],
    day_of_month_is_star: bool,
    day_of_week_is_star: bool,
}

impl Cron {
    pub fn allowed(&self, field: Field) -> Vec<u8> {
        let index = field_index(field);
        self.selected[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &selected)| selected.then_some(value as u8))
            .collect()
    }
}

fn field_index(field: Field) -> usize {
    match field {
        Field::Minute => 0,
        Field::Hour => 1,
        Field::DayOfMonth => 2,
        Field::Month => 3,
        Field::DayOfWeek => 4,
    }
}

fn field_spec(field: Field) -> (u32, u32) {
    match field {
        Field::Minute => (0, 59),
        Field::Hour => (0, 23),
        Field::DayOfMonth => (1, 31),
        Field::Month => (1, 12),
        Field::DayOfWeek => (0, 6),
    }
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
) -> Result<u32, ParseError> {
    let offset = start;
    if end - start > 1 && bytes[start] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((byte - b'0') as u32))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    min: u32,
    max: u32,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(ParseError::EmptyItem {
            field,
            offset: item_start,
        });
    }

    let item = &bytes[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    let base_end = slash_positions.first().copied().unwrap_or(item.len());
    let step_start = slash_positions.first().map(|&i| i + 1);

    let base = &item[..base_end];
    let (is_star, start_range, end_range) = if base == b"*" {
        (true, None, None)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        match hyphens.as_slice() {
            [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {
                (false, Some((0, base.len())), None)
            }
            [hyphen]
                if *hyphen > 0
                    && *hyphen + 1 < base.len()
                    && base[..*hyphen].iter().all(u8::is_ascii_digit)
                    && base[*hyphen + 1..].iter().all(u8::is_ascii_digit) =>
            {
                (
                    false,
                    Some((0, *hyphen)),
                    Some((*hyphen + 1, base.len())),
                )
            }
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_start,
                });
            }
        }
    };

    if slash_positions.len() > 1 || step_start.is_some_and(|s| s >= item.len()) {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_start,
        });
    }

    let mut start_value = min;
    let mut end_value = max;

    if let Some((start, end)) = start_range {
        let value = parse_number(item, start, end, field)?;
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_start + start,
            });
        }
        start_value = value;

        if let Some((end_start, end_end)) = end_range {
            let value = parse_number(item, end_start, end_end, field)?;
            if value < min || value > max {
                return Err(ParseError::OutOfRange {
                    field,
                    offset: item_start + end_start,
                });
            }
            end_value = value;
            if start_value > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: item_start + end_start,
                });
            }
        } else if step_start.is_none() {
            end_value = start_value;
        }
    }

    let step = if let Some(step_start) = step_start {
        let step_end = item.len();
        if !item[step_start..step_end].iter().all(u8::is_ascii_digit) {
            return Err(ParseError::InvalidSyntax {
                field,
                offset: item_start,
            });
        }
        let value = parse_number(item, step_start, step_end, field)?;
        if value < 1 || value > max - min + 1 {
            return Err(ParseError::BadStep {
                field,
                offset: item_start + step_start,
            });
        }
        value
    } else {
        1
    };

    if is_star || end_range.is_some() || step_start.is_some() {
        let mut value = start_value;
        while value <= end_value {
            selected[(value - min) as usize] = true;
            match value.checked_add(step) {
                Some(next) => value = next,
                None => break,
            }
        }
    } else {
        selected[(start_value - min) as usize] = true;
    }

    Ok(())
}

fn parse_field(field_text: &str, field: Field) -> Result<Vec<bool>, ParseError> {
    let (min, max) = field_spec(field);
    let mut selected = vec![false; (max - min + 1) as usize];
    let bytes = field_text.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let mut item_start = 0;
    for item_end in (0..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        parse_item(
            bytes,
            item_start,
            item_end,
            field,
            min,
            max,
            &mut selected,
        )?;
        item_start = item_end + 1;
    }

    Ok(selected)
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let fields: Vec<&str> = input.split(' ').collect();
    if fields.len() != 5 {
        return Err(ParseError::FieldCount {
            found: fields.len(),
        });
    }

    let field_kinds = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut selected = Vec::with_capacity(5);
    for (text, field) in fields.iter().zip(field_kinds) {
        selected.push(parse_field(text, field)?);
    }

    Ok(Cron {
        selected: selected.try_into().unwrap(),
        day_of_month_is_star: fields[2] == "*",
        day_of_week_is_star: fields[4] == "*",
    })
}
