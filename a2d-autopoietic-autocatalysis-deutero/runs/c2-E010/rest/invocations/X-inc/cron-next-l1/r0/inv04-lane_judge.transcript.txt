        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,

        vec![false; field_max(fields[i]) - field_min(fields[i]) + 1]

fn parse_item(
    item: &str,
    item_offset: usize,
    field: Field,
    selected: &mut [bool],
) -> Result<(), ParseError> {
    let invalid_syntax = || ParseError::InvalidSyntax {
        field,
        offset: item_offset,
    };

    let slash_count = item.bytes().filter(|&b| b == b'/').count();
    if slash_count > 1 {
        return Err(invalid_syntax());
    }

    let (base, step_text) = match item.split_once('/') {
        Some((base, step)) => (base, Some(step)),
        None => (item, None),
    };

    if base.is_empty() || step_text == Some("") {
        return Err(invalid_syntax());
    }

    let (kind, start_text, end_text) = if base == "*" {
        (0, None, None)
    } else {
        let hyphen_count = base.bytes().filter(|&b| b == b'-').count();
        if hyphen_count > 1 {
            return Err(invalid_syntax());
        }
        if hyphen_count == 1 {
            let (start, end) = base.split_once('-').unwrap();
            if start.is_empty() || end.is_empty() || start == "*" || end == "*" {
                return Err(invalid_syntax());
            }
            (2, Some(start), Some(end))
        } else {
            if base.contains('*') {
                return Err(invalid_syntax());
            }
            (1, Some(base), None)
        }
    };

    let base_offset = item_offset;
    let (start, end) = if let Some(start_text) = start_text {
        let start = parse_domain_number(start_text, base_offset, field)?;
        let end = if let Some(end_text) = end_text {
            let end_offset = base_offset + start_text.len() + 1;
            Some(parse_domain_number(end_text, end_offset, field)?)
        } else {
            None
        };

        if let Some(end_value) = end {
            if start > end_value {
                return Err(ParseError::ReversedRange {
                    field,
                    offset: base_offset + start_text.len() + 1,
                });
            }
        }

        (start, end)
    } else {
        (field_min(field) as u32, Some(field_max(field) as u32))
    };

    let step = if let Some(step_text) = step_text {
        let step_offset = item_offset + base.len() + 1;
        let value = parse_number(step_text, step_offset, field)?;
        if value == 0 || value > field_width(field) {
            return Err(ParseError::BadStep {
                field,
                offset: step_offset,
            });
        }
        Some(value)
    } else {
        None
    };

    let end_value = match (kind, end, step) {
        (1, None, None) => start,
        (1, None, Some(_)) => field_max(field) as u32,
        (_, Some(value), _) => value,
        _ => unreachable!(),
    };

    let increment = step.unwrap_or(1);
    let mut value = start;
    while value <= end_value {
        selected[(value as usize) - field_min(field)] = true;
        match value.checked_add(increment) {
            Some(next) if next > value => value = next,
            _ => break,
        }
    }

    Ok(())
}

fn parse_domain_number(text: &str, offset: usize, field: Field) -> Result<u32, ParseError> {
    let value = parse_number(text, offset, field)?;
    if value < field_min(field) as u32 || value > field_max(field) as u32 {
        return Err(ParseError::OutOfRange { field, offset });
    }
    Ok(value)
}

fn parse_number(text: &str, offset: usize, field: Field) -> Result<u32, ParseError> {
    let bytes = text.as_bytes();
    if bytes.len() > 1 && bytes[0] == b'0' {
        return Err(ParseError::LeadingZero { field, offset });
    }

    let mut value = 0u32;
    for &byte in bytes {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or(ParseError::NumberOverflow { field, offset })?;
    }
    Ok(value)
}

#[derive(Clone, Copy)]

    InvalidCharacter,
    EmptyItem,
    InvalidSyntax,
    LeadingZero,
    NumberOverflow,
    OutOfRange,
    ReversedRange,
    BadStep,
}

fn parse_number(
    bytes: &[u8],
    start: usize,
    end: usize,
    field: Field,
    is_step: bool,
) -> Result<u32, ParseError> {
    if start == end {
        return Err(error(field, start, ErrorKind::InvalidSyntax));
    }
    if end - start > 1 && bytes[start] == b'0' {
        return Err(error(field, start, ErrorKind::LeadingZero));
    }

    let mut value = 0u32;
    for &byte in &bytes[start..end] {
        let digit = (byte - b'0') as u32;
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add(digit))
            .ok_or_else(|| error(field, start, ErrorKind::NumberOverflow))?;
    }

    if is_step {
        let (_, _, width) = domain(field);
        if value == 0 || value > width {
            return Err(error(field, start, ErrorKind::BadStep));
        }
    } else {
        let (min, max, _) = domain(field);
        if value < min as u32 || value > max as u32 {
            return Err(error(field, start, ErrorKind::OutOfRange));
        }
    }

    Ok(value)
}

fn parse_item(
    bytes: &[u8],
    item_start: usize,
    item_end: usize,
    field: Field,
    selected: &mut [bool; 256],
) -> Result<(), ParseError> {
    if item_start == item_end {
        return Err(error(field, item_start, ErrorKind::EmptyItem));
    }

    let item = &bytes[item_start..item_end];
    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    let (base_end, step_range) = match slash_positions.as_slice() {
        [] => (item.len(), None),
        [slash] => (*slash, Some((*slash + 1, item.len()))),
        _ => return Err(error(field, item_start, ErrorKind::InvalidSyntax)),
    };

    if base_end == 0 || step_range.is_some_and(|(s, e)| s == e) {
        return Err(error(field, item_start, ErrorKind::InvalidSyntax));
    }

    let base = &item[..base_end];
    let (base_kind, first, second) = if base == b"*" {
        (BaseKind::Star, None, None)
    } else {
        let hyphens: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        match hyphens.as_slice() {
            [] => {
                if base.iter().all(u8::is_ascii_digit) {
                    (BaseKind::Single, Some((0, base.len())), None)
                } else {
                    return Err(error(field, item_start, ErrorKind::InvalidSyntax));
                }
            }
            [hyphen] => {
                let left = &base[..*hyphen];
                let right = &base[*hyphen + 1..];
                if left.is_empty()
                    || right.is_empty()
                    || !left.iter().all(u8::is_ascii_digit)
                    || !right.iter().all(u8::is_ascii_digit)
                {
                    return Err(error(field, item_start, ErrorKind::InvalidSyntax));
                }
                (
                    BaseKind::Range,
                    Some((0, *hyphen)),
                    Some((*hyphen + 1, base.len())),
                )
            }
            _ => return Err(error(field, item_start, ErrorKind::InvalidSyntax)),
        }
    };

    let (min, max, _) = domain(field);
    let (start, end) = match base_kind {
        BaseKind::Star => (min, max),
        BaseKind::Single => {
            let (s, e) = first.unwrap();
            let value = parse_number(base, s, e, field, false)?;
            let end = if step_range.is_some() { max } else { value as u8 };
            (value as u8, end)
        }
        BaseKind::Range => {
            let (s1, e1) = first.unwrap();
            let (s2, e2) = second.unwrap();
            let start = parse_number(base, s1, e1, field, false)?;
            let end = parse_number(base, s2, e2, field, false)?;
            if start > end {
                return Err(error(field, item_start + s2, ErrorKind::ReversedRange));
            }
            (start as u8, end as u8)
        }
    };

    let step = if let Some((s, e)) = step_range {
        parse_number(item, s, e, field, true)?
    } else {
        1
    };

    let mut value = start as u32;
    while value <= end as u32 {
        selected[value as usize] = true;
        value += step;
    }

    Ok(())
}

        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,

        [false; 256],
        [false; 256],
        [false; 256],
        [false; 256],
        [false; 256],

            item_start,

        selected_sets[index]
            .iter()
            .enumerate()
            .filter_map(|(value, &is_selected)| is_selected.then_some(value as u8))
            .collect()
    });

    Ok(Cron {
        selected,
        day_stars,
    })
}

    }

    let slash_positions: Vec<usize> = item
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| (b == b'/').then_some(i))
        .collect();

    let base_end = if slash_positions.is_empty() {
        item.len()
    } else {
        slash_positions[0]
    };

    let base = &item[..base_end];
    let mut range_dash = None;

    if base == b"*" {
        // Valid wildcard base.
    } else {
        let dashes: Vec<usize> = base
            .iter()
            .enumerate()
            .filter_map(|(i, &b)| (b == b'-').then_some(i))
            .collect();

        match dashes.as_slice() {
            [] if !base.is_empty() && base.iter().all(u8::is_ascii_digit) => {}
            [dash]
                if *dash > 0
                    && *dash + 1 < base.len()
                    && base[..*dash].iter().all(u8::is_ascii_digit)
                    && base[*dash + 1..].iter().all(u8::is_ascii_digit) =>
            {
                range_dash = Some(*dash);
            }
            _ => {
                return Err(ParseError::InvalidSyntax {
                    field,
                    offset: item_offset,
                });
            }
        }
    }

    if !slash_positions.is_empty()
        && (slash_positions.len() != 1 || slash_positions[0] + 1 == item.len())
    {
        return Err(ParseError::InvalidSyntax {
            field,
            offset: item_offset,
        });
    }

    let (min, max) = field_bounds(field);
    let (start, end_number_offset, explicit_range) = if base == b"*" {
        (min, None, false)
    } else if let Some(dash) = range_dash {
        let start_value = parse_number(item, 0, dash, field)?;
        let end_value = parse_number(item, dash + 1, base_end, field)?;
        for (value, offset) in [(start_value, item_offset), (end_value, item_offset + dash + 1)] {
            if value < min || value > max {
                return Err(ParseError::OutOfRange { field, offset });
            }
        }
        if start_value > end_value {
            return Err(ParseError::ReversedRange {
                field,
                offset: item_offset + dash + 1,
            });
        }
        (start_value, Some((end_value, item_offset + dash + 1)), true)
    } else {
        let value = parse_number(item, 0, base_end, field)?;
        if value < min || value > max {
            return Err(ParseError::OutOfRange {
                field,
                offset: item_offset,
            });
        }
        (value, Some((value, item_offset)), false)
    };

    let step = if let Some(&slash) = slash_positions.first() {
        let step_start = slash + 1;
        let step_value = parse_number(item, step_start, item.len(), field)?;
        if step_value == 0 || step_value > field_width(field) {
            return Err(ParseError::BadStep {
                field,
                offset: item_offset + step_start,
            });
        }
        step_value
    } else {
        1
    };

    let end = if slash_positions.is_empty() {
        end_number_offset.map_or(max, |(n, _)| n)
    } else if base == b"*" || !explicit_range {
        max
    } else {
        end_number_offset.map_or(max, |(n, _)| n)
    };

    let mut value = start;
    while value <= end {
        selected[(value - min) as usize] = true;
        match value.checked_add(step) {
            Some(next) => value = next,
            None => break,
        }
    }

    Ok(())
}

fn parse_field(input: &str, field: Field) -> Result<Vec<u8>, ParseError> {
    let bytes = input.as_bytes();

    for (offset, &byte) in bytes.iter().enumerate() {
        if !byte.is_ascii_digit() && !matches!(byte, b'*' | b'-' | b'/' | b',') {
            return Err(ParseError::InvalidCharacter { field, offset });
        }
    }

    let (min, max) = field_bounds(field);
    let mut selected = vec![false; (max - min + 1) as usize];

    let mut item_start = 0usize;
    for item_end in (0..=bytes.len()).filter(|&i| i == bytes.len() || bytes[i] == b',') {
        parse_item(
            &bytes[item_start..item_end],
            item_start,
            field,
            &mut selected,
        )?;
        item_start = item_end + 1;
    }

    Ok(selected
        .iter()
        .enumerate()
        .filter_map(|(i, &is_selected)| {
            is_selected.then_some((min as usize + i) as u8)
        })
        .collect())
}

pub fn parse_cron(input: &str) -> Result<Cron, ParseError> {
    let parts: Vec<&str> = input.split(' ').collect();
    if parts.len() != 5 {
        return Err(ParseError::FieldCount { found: parts.len() });
    }

    let fields = [
        Field::Minute,
        Field::Hour,
        Field::DayOfMonth,
        Field::Month,
        Field::DayOfWeek,
    ];
    let mut allowed = Vec::with_capacity(5);
    for (part, field) in parts.iter().zip(fields) {
        allowed.push(parse_field(part, field)?);
    }

    Ok(Cron {
        allowed: allowed.try_into().unwrap(),
        day_stars: [parts[2] == "*", parts[4] == "*"],
    })
}
