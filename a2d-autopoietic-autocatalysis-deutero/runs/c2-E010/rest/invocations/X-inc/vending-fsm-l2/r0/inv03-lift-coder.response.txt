#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
}

pub struct Machine {
    products: Vec<Product>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
}

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen: Vec<&str> = Vec::new();
        for product in &products {
            let valid_id = (1..=8).contains(&product.id.len())
                && product.id.bytes().all(|b| b.is_ascii_lowercase());
            if !valid_id {
                return Err(Error::BadId {
                    id: product.id.clone(),
                });
            }
            if seen.contains(&product.id.as_str()) {
                return Err(Error::DuplicateId {
                    id: product.id.clone(),
                });
            }
            seen.push(&product.id);
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice {
                    id: product.id.clone(),
                });
            }
        }

        let denoms = [1, 3, 4, 10];
        for i in 0..4 {
            if tubes[i] > 100 {
                return Err(Error::TubeLimit { coin: denoms[i] });
            }
        }

        Ok(Self {
            products,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
        }
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        let index = match coin {
            1 => 0,
            3 => 1,
            4 => 2,
            10 => 3,
            _ => return Err(Error::BadCoin { coin }),
        };

        let _ = index;
        if u32::from(self.credit) + u32::from(coin) > 100 {
            return Err(Error::CreditLimit);
        }

        self.credit += coin;
        self.escrow.push(coin);
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        let Some(product) = self.products.iter().find(|p| p.id == id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };

        if self.selected.as_deref() == Some(product.id.as_str()) {
            return Err(Error::SameSelection);
        }

        self.selected = Some(product.id.clone());
        Ok(())
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        let Some(selected_id) = self.selected.as_ref() else {
            return Err(Error::NoSelection);
        };
        let product = self
            .products
            .iter()
            .find(|p| p.id == *selected_id)
            .expect("selected product must be in catalog");

        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid {
                price,
                credit: self.credit,
            });
        }

        let amount = self.credit - price;
        let denoms = [1u16, 3, 4, 10];
        let mut chosen = [0u16; 4];
        let mut remaining = amount;

        for i in (0..4).rev() {
            let take = self.tubes[i].min(remaining / denoms[i]);
            chosen[i] = take;
            remaining -= take * denoms[i];
        }

        if remaining != 0 {
            if let Some(solution) = best_change(amount, &self.tubes) {
                chosen = solution;
            } else {
                return Err(Error::NoChange { amount });
            }
        }

        let mut proposed = self.tubes;
        for i in 0..4 {
            proposed[i] -= chosen[i];
        }

        for &coin in &self.escrow {
            let index = match coin {
                1 => 0,
                3 => 1,
                4 => 2,
                10 => 3,
                _ => unreachable!(),
            };
            proposed[index] += 1;
        }

        for i in 0..4 {
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: denoms[i] });
            }
        }

        let mut change = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..chosen[i] {
                change.push(denoms[i]);
            }
        }

        let receipt = Receipt {
            id: selected_id.clone(),
            price,
            change,
        };

        self.tubes = proposed;
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();

        Ok(receipt)
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        let returned = self.escrow.iter().rev().copied().collect();
        self.credit = 0;
        self.selected = None;
        self.escrow.clear();
        Ok(returned)
    }
}

fn best_change(amount: u16, tubes: &[u16; 4]) -> Option<[u16; 4]> {
    fn search(
        index: usize,
        remaining: u16,
        tubes: &[u16; 4],
        current: &mut [u16; 4],
        best: &mut Option<[u16; 4]>,
        best_count: &mut u16,
    ) {
        if index == 4 {
            if remaining == 0 {
                let count: u16 = current.iter().copied().sum();
                let better = match best {
                    None => true,
                    Some(existing) => {
                        count < *best_count
                            || (count == *best_count
                                && (3..=0).rev().any(|i| {
                                    let _ = i;
                                    false
                                }))
                    }
                };
                let lex_greater = best.map_or(true, |existing| {
                    for i in (0..4).rev() {
                        if current[i] != existing[i] {
                            return current[i] > existing[i];
                        }
                    }
                    false
                });
                if better || (best.is_some() && count == *best_count && lex_greater) {
                    *best = Some(*current);
                    *best_count = count;
                }
            }
            return;
        }

        let denom = [1u16, 3, 4, 10][index];
        let max_take = tubes[index].min(remaining / denom);
        for take in 0..=max_take {
            current[index] = take;
            search(
                index + 1,
                remaining - take * denom,
                tubes,
                current,
                best,
                best_count,
            );
        }
        current[index] = 0;
    }

    let mut best = None;
    let mut best_count = u16::MAX;
    let mut current = [0u16; 4];
    search(
        0,
        amount,
        tubes,
        &mut current,
        &mut best,
        &mut best_count,
    );
    best
}
