#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
    UnterminatedExpansion { pos: usize },
    InvalidExpansion { pos: usize },
    CommandSubstitution { pos: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Word {
    pub text: String,
    pub glob: Vec<usize>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum JoinError {
    Nul { word: usize, pos: usize },
}

fn char_at(input: &str, pos: usize) -> Option<char> {
    input.get(pos..)?.chars().next()
}

fn next_pos(input: &str, pos: usize) -> usize {
    pos + char_at(input, pos).map(char::len_utf8).unwrap_or(0)
}

fn is_separator(c: char) -> bool {
    matches!(c, ' ' | '\t' | '\r' | '\n')
}

fn is_name_start(c: char) -> bool {
    c == '_' || c.is_ascii_alphabetic()
}

fn is_name_continue(c: char) -> bool {
    is_name_start(c) || c.is_ascii_digit()
}

fn env_value<'a>(env: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    env.iter()
        .find_map(|(key, value)| (*key == name).then_some(*value))
}

fn append(word: &mut Word, s: &str, active_glob: bool) {
    for c in s.chars() {
        if active_glob && matches!(c, '*' | '?' | '[') {
            word.glob.push(word.text.len());
        }
        word.text.push(c);
    }
}

fn decode_default(input: &str) -> String {
    let mut out = String::new();
    let mut pos = 0;
    while pos < input.len() {
        let c = char_at(input, pos).unwrap();
        if c == '\\' {
            pos += 1;
            if pos < input.len() {
                let escaped = char_at(input, pos).unwrap();
                out.push(escaped);
                pos += escaped.len_utf8();
            }
        } else {
            out.push(c);
            pos += c.len_utf8();
        }
    }
    out
}

fn find_brace_close(input: &str, mut pos: usize) -> Option<usize> {
    while pos < input.len() {
        let c = char_at(input, pos)?;
        if c == '\\' {
            pos += 1;
            if pos < input.len() {
                pos = next_pos(input, pos);
            }
        } else if c == '}' {
            return Some(pos);
        } else {
            pos += c.len_utf8();
        }
    }
    None
}

fn expand_braced(
    input: &str,
    dollar: usize,
    env: &[(&str, &str)],
) -> Result<(String, usize), SplitError> {
    let content_start = dollar + 2;
    let close = find_brace_close(input, content_start)
        .ok_or(SplitError::UnterminatedExpansion { pos: dollar })?;
    let content = &input[content_start..close];

    let mut name_end = 0;
    for c in content.chars() {
        if (name_end == 0 && is_name_start(c)) || (name_end > 0 && is_name_continue(c)) {
            name_end += c.len_utf8();
        } else {
            break;
        }
    }

    if name_end == 0 {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    }

    let name = &content[..name_end];
    let rest = &content[name_end..];
    let value = if rest.is_empty() {
        env_value(env, name).unwrap_or("").to_string()
    } else if let Some(default) = rest.strip_prefix(":-") {
        match env_value(env, name) {
            Some(value) if !value.is_empty() => value.to_string(),
            _ => decode_default(default),
        }
    } else {
        return Err(SplitError::InvalidExpansion { pos: dollar });
    };

    Ok((value, close + 1))
}

pub fn split(input: &str, env: &[(&str, &str)]) -> Result<Vec<Word>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut current: Option<Word> = None;
    let mut pos = 0;

    while pos < input.len() {
        let c = char_at(input, pos).unwrap();

        if is_separator(c) {
            if let Some(word) = current.take() {
                words.push(word);
            }
            pos += c.len_utf8();
            continue;
        }

        if current.is_none() {
            current = Some(Word {
                text: String::new(),
                glob: Vec::new(),
            });
        }

        match c {
            '\'' => {
                let opening = pos;
                pos += 1;
                loop {
                    if pos >= input.len() {
                        return Err(SplitError::UnterminatedSingle { pos: opening });
                    }
                    let literal = char_at(input, pos).unwrap();
                    if literal == '\'' {
                        pos += 1;
                        break;
                    }
                    append(current.as_mut().unwrap(), &literal.to_string(), false);
                    pos += literal.len_utf8();
                }
            }
            '"' => {
                let opening = pos;
                pos += 1;
                loop {
                    if pos >= input.len() {
                        return Err(SplitError::UnterminatedDouble { pos: opening });
                    }
                    let inner = char_at(input, pos).unwrap();
                    match inner {
                        '"' => {
                            pos += 1;
                            break;
                        }
                        '\\' => {
                            let slash = pos;
                            pos += 1;
                            if pos >= input.len() {
                                return Err(SplitError::TrailingEscape { pos: slash });
                            }
                            let escaped = char_at(input, pos).unwrap();
                            if matches!(escaped, '"' | '\\' | '$' | '\n') {
                                if escaped != '\n' {
                                    append(
                                        current.as_mut().unwrap(),
                                        &escaped.to_string(),
                                        false,
                                    );
                                }
                                pos += escaped.len_utf8();
                            } else {
                                append(current.as_mut().unwrap(), "\\", false);
                            }
                        }
                        '`' => {
                            return Err(SplitError::CommandSubstitution { pos });
                        }
                        '$' => {
                            let after = next_pos(input, pos);
                            if char_at(input, after) == Some('(') {
                                return Err(SplitError::CommandSubstitution { pos });
                            }
                            if char_at(input, after) == Some('{') {
                                let (value, end) = expand_braced(input, pos, env)?;
                                append(current.as_mut().unwrap(), &value, false);
                                pos = end;
                            } else if char_at(input, after).is_some_and(is_name_start) {
                                let mut end = after;
                                while let Some(ch) = char_at(input, end) {
                                    if !is_name_continue(ch) {
                                        break;
                                    }
                                    end += ch.len_utf8();
                                }
                                let value = env_value(env, &input[after..end]).unwrap_or("");
                                append(current.as_mut().unwrap(), value, false);
                                pos = end;
                            } else {
                                append(current.as_mut().unwrap(), "$", false);
                                pos += 1;
                            }
                        }
                        _ => {
                            append(current.as_mut().unwrap(), &inner.to_string(), false);
                            pos += inner.len_utf8();
                        }
                    }
                }
            }
            '\\' => {
                let slash = pos;
                pos += 1;
                if pos >= input.len() {
                    return Err(SplitError::TrailingEscape { pos: slash });
                }
                let escaped = char_at(input, pos).unwrap();
                if escaped != '\n' {
                    append(current.as_mut().unwrap(), &escaped.to_string(), false);
                }
                pos += escaped.len_utf8();
            }
            '`' => return Err(SplitError::CommandSubstitution { pos }),
            '$' => {
                let after = next_pos(input, pos);
                if char_at(input, after) == Some('(') {
                    return Err(SplitError::CommandSubstitution { pos });
                }
                if char_at(input, after) == Some('{') {
                    let (value, end) = expand_braced(input, pos, env)?;
                    append(current.as_mut().unwrap(), &value, true);
                    pos = end;
                } else if char_at(input, after).is_some_and(is_name_start) {
                    let mut end = after;
                    while let Some(ch) = char_at(input, end) {
                        if !is_name_continue(ch) {
                            break;
                        }
                        end += ch.len_utf8();
                    }
                    let value = env_value(env, &input[after..end]).unwrap_or("");
                    append(current.as_mut().unwrap(), value, true);
                    pos = end;
                } else {
                    append(current.as_mut().unwrap(), "$", true);
                    pos += 1;
                }
            }
            _ => {
                append(current.as_mut().unwrap(), &c.to_string(), true);
                pos += c.len_utf8();
            }
        }
    }

    if let Some(word) = current {
        words.push(word);
    }
    Ok(words)
}

fn bare_candidate(word: &str) -> Option<String> {
    if word.is_empty() || word.contains('\n') {
        return None;
    }
    let mut out = String::new();
    for c in word.chars() {
        if c.is_ascii_alphanumeric()
            || matches!(c, '_' | '-' | '.' | '/' | ':' | '=' | ',' | '+' | '@' | '%')
        {
            out.push(c);
        } else {
            out.push('\\');
            out.push(c);
        }
    }
    Some(out)
}

fn single_candidate(word: &str) -> Option<String> {
    if word.contains('\'') {
        None
    } else {
        Some(format!("'{word}'"))
    }
}

fn double_candidate(word: &str) -> Option<String> {
    if word.contains('`') {
        return None;
    }
    let mut out = String::from("\"");
    for c in word.chars() {
        if matches!(c, '\\' | '"' | '$') {
            out.push('\\');
        }
        out.push(c);
    }
    out.push('"');
    Some(out)
}

pub fn join(words: &[String]) -> Result<String, JoinError> {
    for (word_index, word) in words.iter().enumerate() {
        if let Some(pos) = word.find('\0') {
            return Err(JoinError::Nul {
                word: word_index,
                pos,
            });
        }
    }

    let mut encoded = Vec::with_capacity(words.len());
    for word in words {
        let mut candidates = Vec::new();
        if let Some(candidate) = bare_candidate(word) {
            candidates.push((candidate, 0));
        }
        if let Some(candidate) = single_candidate(word) {
            candidates.push((candidate, 1));
        }
        if let Some(candidate) = double_candidate(word) {
            candidates.push((candidate, 2));
        }

        let candidate = if candidates.is_empty() {
            let mut fallback = String::from("'");
            for c in word.chars() {
                if c == '\'' {
                    fallback.push_str("'\\''");
                } else {
                    fallback.push(c);
                }
            }
            fallback.push('\'');
            fallback
        } else {
            candidates
                .into_iter()
                .min_by_key(|(candidate, priority)| (candidate.len(), *priority))
                .unwrap()
                .0
        };
        encoded.push(candidate);
    }
    Ok(encoded.join(" "))
}
