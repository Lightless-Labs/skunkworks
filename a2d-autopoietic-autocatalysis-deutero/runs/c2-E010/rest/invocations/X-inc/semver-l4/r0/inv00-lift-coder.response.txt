use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn missing_or_invalid(s: &str, at: usize) -> VersionError {
    match s.as_bytes().get(at).copied() {
        None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at },
        _ => VersionError::InvalidChar { at },
    }
}

fn parse_core_number(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return Err(missing_or_invalid(s, start));
    }
    let digits = &s[start..*pos];
    if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    digits
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn valid_identifier_byte(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b == b'-'
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;

    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            if !valid_identifier_byte(bytes[i]) {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }

        let identifier = &s[id_start..i];
        if prerelease
            && identifier.as_bytes().iter().all(u8::is_ascii_digit)
            && identifier.len() > 1
            && identifier.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        result.push(identifier.to_owned());

        if i == end {
            break;
        }
        id_start = i + 1;
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core_number(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return Err(missing_or_invalid(s, pos));
    }
    pos += 1;

    let minor = parse_core_number(s, &mut pos)?;
    if s.as_bytes().get(pos) != Some(&b'.') {
        return Err(missing_or_invalid(s, pos));
    }
    pos += 1;

    let patch = parse_core_number(s, &mut pos)?;
    let bytes = s.as_bytes();

    let mut pre = Vec::new();
    let mut build = Vec::new();

    match bytes.get(pos).copied() {
        None => {}
        Some(b'-') => {
            let pre_start = pos + 1;
            let plus = bytes[pre_start..]
                .iter()
                .position(|&b| b == b'+')
                .map(|i| pre_start + i);
            let pre_end = plus.unwrap_or(bytes.len());
            pre = parse_identifiers(s, pre_start, pre_end, true)?;
            if let Some(plus_pos) = plus {
                let build_start = plus_pos + 1;
                build = parse_identifiers(s, build_start, bytes.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, pos + 1, bytes.len(), false)?;
        }
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core_cmp(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch).cmp(&(b.major, b.minor, b.patch))
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(u8::is_ascii_digit)
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let an = numeric_identifier(a);
    let bn = numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => {
            let aa = a.trim_start_matches('0');
            let bb = b.trim_start_matches('0');
            aa.len().cmp(&bb.len()).then_with(|| aa.as_bytes().cmp(bb.as_bytes()))
        }
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

fn compare_pre(a: &[String], b: &[String]) -> Ordering {
    match (a.is_empty(), b.is_empty()) {
        (true, true) => Ordering::Equal,
        (true, false) => Ordering::Greater,
        (false, true) => Ordering::Less,
        (false, false) => {
            for (x, y) in a.iter().zip(b) {
                let cmp = compare_identifiers(x, y);
                if cmp != Ordering::Equal {
                    return cmp;
                }
            }
            a.len().cmp(&b.len())
        }
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core_cmp(a, b).then_with(|| compare_pre(&a.pre, &b.pre))
}

pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
struct Comparator {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Arm {
    comparisons: Vec<Comparator>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Arm>,
}

#[derive(Clone, Copy, Debug)]
struct Token<'a> {
    text: &'a str,
    start: usize,
}

fn is_range_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn scan_tokens(s: &str) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut tokens = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_range_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !is_range_ws(bytes[i]) {
            i += 1;
        }
        tokens.push(Token {
            text: &s[start..i],
            start,
        });
    }
    tokens
}

fn syntax(at: usize) -> RangeError {
    RangeError::Syntax { at }
}

fn parse_full_operand(token: Token<'_>, operand_start: usize, operand: &str) -> Result<Version, RangeError> {
    if !operand.as_bytes().first().is_some_and(u8::is_ascii_digit) {
        return Err(syntax(token.start));
    }
    parse_version(operand).map_err(|error| RangeError::Version {
        at: operand_start,
        error,
    })
}

fn is_wildcard(field: &str) -> bool {
    matches!(field, "*" | "x" | "X")
}

#[derive(Clone, Debug)]
struct Partial {
    fields: Vec<Option<u64>>,
    numeric_count: usize,
    full: Option<Version>,
}

fn should_parse_full(operand: &str) -> bool {
    if operand.contains('-') || operand.contains('+') {
        return true;
    }
    let fields: Vec<&str> = operand.split('.').collect();
    fields.len() >= 3 && !fields.iter().any(|f| is_wildcard(f))
}

fn parse_partial(
    token: Token<'_>,
    operand_start: usize,
    operand: &str,
) -> Result<Partial, RangeError> {
    if should_parse_full(operand) {
        let full = parse_full_operand(token, operand_start, operand)?;
        return Ok(Partial {
            fields: vec![
                Some(full.major),
                Some(full.minor),
                Some(full.patch),
            ],
            numeric_count: 3,
            full: Some(full),
        });
    }

    let raw: Vec<&str> = operand.split('.').collect();
    if raw.is_empty() || raw.len() > 3 {
        return Err(syntax(token.start));
    }

    let mut fields = Vec::new();
    let mut wildcard_seen = false;
    let mut numeric_count = 0;
    for field in raw {
        if field.is_empty() {
            return Err(syntax(token.start));
        }
        if is_wildcard(field) {
            wildcard_seen = true;
            fields.push(None);
        } else {
            if wildcard_seen
                || !field.as_bytes().iter().all(u8::is_ascii_digit)
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(syntax(token.start));
            }
            let value = field.parse::<u64>().map_err(|_| syntax(token.start))?;
            numeric_count += 1;
            fields.push(Some(value));
        }
    }
    Ok(Partial {
        fields,
        numeric_count,
        full: None,
    })
}

fn partial_lower(p: &Partial) -> Version {
    let major = p.fields.first().and_then(|x| *x).unwrap_or(0);
    let minor = p.fields.get(1).and_then(|x| *x).unwrap_or(0);
    let patch = p.fields.get(2).and_then(|x| *x).unwrap_or(0);
    p.full.clone().unwrap_or(Version {
        major,
        minor,
        patch,
        pre: Vec::new(),
        build: Vec::new(),
    })
}

fn endpoint(major: u64, minor: u64, patch: u64) -> Version {
    Version {
        major,
        minor,
        patch,
        pre: vec!["0".to_owned()],
        build: Vec::new(),
    }
}

fn next_major(v: &Version, at: usize) -> Result<Version, RangeError> {
    let major = v.major.checked_add(1).ok_or_else(|| syntax(at))?;
    Ok(endpoint(major, 0, 0))
}

fn next_minor(v: &Version, at: usize) -> Result<Version, RangeError> {
    let minor = v.minor.checked_add(1).ok_or_else(|| syntax(at))?;
    Ok(endpoint(v.major, minor, 0))
}

fn next_patch(v: &Version, at: usize) -> Result<Version, RangeError> {
    let patch = v.patch.checked_add(1).ok_or_else(|| syntax(at))?;
    Ok(endpoint(v.major, v.minor, patch))
}

fn add_cmp(arm: &mut Arm, op: Op, version: Version) {
    arm.comparisons.push(Comparator { op, version });
}

fn add_anchor(arm: &mut Arm, version: &Version) {
    if !version.pre.is_empty() {
        arm.anchors.push((version.major, version.minor, version.patch));
    }
}

fn add_partial_expansion(
    arm: &mut Arm,
    token: Token<'_>,
    operand_start: usize,
    operand: &str,
    mode: char,
) -> Result<(), RangeError> {
    let p = parse_partial(token, operand_start, operand)?;
    let k = p.numeric_count;
    if mode != 'b' && k == 0 {
        return Err(syntax(token.start));
    }

    let lower = partial_lower(&p);
    if let Some(full) = &p.full {
        if mode == 'b' {
            add_cmp(arm, Op::Eq, full.clone());
            add_anchor(arm, full);
            return Ok(());
        }
    }
    add_cmp(arm, Op::Ge, lower.clone());
    if !lower.pre.is_empty() {
        add_anchor(arm, &lower);
    }

    if k == 0 {
        return Ok(());
    }

    let upper = match mode {
        '^' => {
            if lower.major > 0 || k == 1 {
                next_major(&lower, token.start)?
            } else if lower.minor > 0 || k == 2 {
                next_minor(&lower, token.start)?
            } else {
                next_patch(&lower, token.start)?
            }
        }
        '~' => {
            if k == 1 {
                next_major(&lower, token.start)?
            } else {
                next_minor(&lower, token.start)?
            }
        }
        'b' => match k {
            1 => next_major(&lower, token.start)?,
            2 => next_minor(&lower, token.start)?,
            _ => return Ok(()),
        },
        _ => return Err(syntax(token.start)),
    };
    add_cmp(arm, Op::Lt, upper);
    Ok(())
}

fn parse_comparator_token(token: Token<'_>) -> Result<Comparator, RangeError> {
    let s = token.text;
    let (op, consumed) = if s.starts_with(">=") {
        (Op::Ge, 2)
    } else if s.starts_with("<=") {
        (Op::Le, 2)
    } else if s.starts_with('>') {
        (Op::Gt, 1)
    } else if s.starts_with('<') {
        (Op::Lt, 1)
    } else if s.starts_with('=') {
        (Op::Eq, 1)
    } else {
        (Op::Eq, 0)
    };
    let operand = &s[consumed..];
    let operand_start = token.start + consumed;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(syntax(token.start));
    }
    let version = parse_full_operand(token, operand_start, operand)?;
    Ok(Comparator { op, version })
}

fn split_arms(s: &str, tokens: &[Token<'_>]) -> Result<Vec<Vec<Token<'_>>>, RangeError> {
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms = Vec::new();
    let mut current = Vec::new();
    let mut last_separator: Option<usize> = None;
    for token in tokens {
        if token.text == "||" {
            if current.is_empty() {
                return Err(RangeError::Empty { at: token.start });
            }
            arms.push(std::mem::take(&mut current));
            last_separator = Some(token.start);
        } else {
            current.push(*token);
            last_separator = None;
        }
    }
    if current.is_empty() {
        let at = last_separator.map(|_| s.len()).unwrap_or(s.len());
        return Err(RangeError::Empty { at });
    }
    arms.push(current);
    Ok(arms)
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    let mut arm = Arm {
        comparisons: Vec::new(),
        anchors: Vec::new(),
    };

    let hyphens: Vec<usize> = tokens
        .iter()
        .enumerate()
        .filter_map(|(i, t)| (t.text == "-").then_some(i))
        .collect();
    if !hyphens.is_empty() {
        if tokens.len() != 3 || hyphens != [1] {
            return Err(syntax(tokens[hyphens[0]].start));
        }
        let lower = parse_partial(tokens[0], tokens[0].start, tokens[0].text)?;
        let upper = parse_partial(tokens[2], tokens[2].start, tokens[2].text)?;
        if lower.numeric_count == 0 {
            return Err(syntax(tokens[0].start));
        }
        if upper.numeric_count == 0 {
            return Err(syntax(tokens[2].start));
        }

        let lo = partial_lower(&lower);
        add_cmp(&mut arm, Op::Ge, lo.clone());
        if !lo.pre.is_empty() {
            add_anchor(&mut arm, &lo);
        }

        if let Some(full) = upper.full {
            add_cmp(&mut arm, Op::Le, full.clone());
            add_anchor(&mut arm, &full);
        } else {
            let hi = match upper.numeric_count {
                1 => next_major(&partial_lower(&upper), tokens[2].start)?,
                2 => next_minor(&partial_lower(&upper), tokens[2].start)?,
                _ => {
                    let mut v = partial_lower(&upper);
                    v.pre.clear();
                    add_cmp(&mut arm, Op::Le, v.clone());
                    return Ok(arm);
                }
            };
            add_cmp(&mut arm, Op::Lt, hi);
        }
        return Ok(arm);
    }

    for token in tokens {
        let s = token.text;
        if s.starts_with('^') || s.starts_with('~') {
            let mode = s.as_bytes()[0] as char;
            let operand = &s[1..];
            if operand.is_empty() {
                return Err(syntax(token.start));
            }
            add_partial_expansion(
                &mut arm,
                *token,
                token.start + 1,
                operand,
                mode,
            )?;
        } else if s.starts_with('<') || s.starts_with('>') || s.starts_with('=') {
            let cmp = parse_comparator_token(*token)?;
            add_anchor(&mut arm, &cmp.version);
            arm.comparisons.push(cmp);
        } else if s.starts_with(',') || s.starts_with('(') || s.starts_with(')') || s == "&" {
            return Err(syntax(token.start));
        } else if s.starts_with('*') || s.starts_with('x') || s.starts_with('X')
            || s.as_bytes().first().is_some_and(u8::is_ascii_digit)
        {
            add_partial_expansion(&mut arm, *token, token.start, s, 'b')?;
        } else {
            return Err(syntax(token.start));
        }
    }

    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = scan_tokens(s);
    let token_arms = split_arms(s, &tokens)?;
    let mut arms = Vec::new();
    for tokens in token_arms {
        arms.push(parse_arm(&tokens)?);
    }
    Ok(Range { arms })
}

fn satisfies_comparator(v: &Version, c: &Comparator) -> bool {
    let cmp = compare_versions(v, &c.version);
    match c.op {
        Op::Eq => cmp == Ordering::Equal,
        Op::Lt => cmp == Ordering::Less,
        Op::Le => cmp != Ordering::Greater,
        Op::Gt => cmp == Ordering::Greater,
        Op::Ge => cmp != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include_prerelease: bool) -> bool {
    if !arm.comparisons.iter().all(|c| satisfies_comparator(v, c)) {
        return false;
    }
    if include_prerelease || v.pre.is_empty() {
        return true;
    }
    arm.anchors
        .iter()
        .any(|&(a, b, c)| (v.major, v.minor, v.patch) == (a, b, c))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms
        .iter()
        .any(|arm| arm_matches(v, arm, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<usize> = None;
    for i in 0..candidates.len() {
        if !matches(&parsed[i], r) {
            continue;
        }
        match best {
            None => best = Some(i),
            Some(j) => {
                let cmp = compare_versions(&parsed[i], &parsed[j]);
                if (greatest && cmp == Ordering::Greater) || (!greatest && cmp == Ordering::Less) {
                    best = Some(i);
                }
            }
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

#[derive(Clone, Debug)]
struct Bound {
    version: Version,
    inclusive: bool,
}

fn compare_bound_versions(a: &Bound, b: &Bound) -> Ordering {
    compare_versions(&a.version, &b.version)
}

fn strongest_lower(comparisons: &[Comparator]) -> Option<Bound> {
    let mut best: Option<Bound> = None;
    for c in comparisons {
        let bound = match c.op {
            Op::Eq | Op::Ge => Some(Bound {
                version: c.version.clone(),
                inclusive: true,
            }),
            Op::Gt => Some(Bound {
                version: c.version.clone(),
                inclusive: false,
            }),
            _ => None,
        };
        if let Some(bound) = bound {
            match &best {
                None => best = Some(bound),
                Some(old) => match compare_bound_versions(&bound, old) {
                    Ordering::Greater => best = Some(bound),
                    Ordering::Equal if !bound.inclusive && old.inclusive => best = Some(bound),
                    _ => {}
                },
            }
        }
    }
    best
}

fn strongest_upper(comparisons: &[Comparator]) -> Option<Bound> {
    let mut best: Option<Bound> = None;
    for c in comparisons {
        let bound = match c.op {
            Op::Eq | Op::Le => Some(Bound {
                version: c.version.clone(),
                inclusive: true,
            }),
            Op::Lt => Some(Bound {
                version: c.version.clone(),
                inclusive: false,
            }),
            _ => None,
        };
        if let Some(bound) = bound {
            match &best {
                None => best = Some(bound),
                Some(old) => match compare_bound_versions(&bound, old) {
                    Ordering::Less => best = Some(bound),
                    Ordering::Equal if !bound.inclusive && old.inclusive => best = Some(bound),
                    _ => {}
                },
            }
        }
    }
    best
}

fn strip_build(mut v: Version) -> Version {
    v.build.clear();
    v
}

fn version_string(v: &Version) -> String {
    let mut out = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        out.push('-');
        out.push_str(&v.pre.join("."));
    }
    out
}

fn tuple_string(t: &(u64, u64, u64)) -> String {
    format!("{}.{}.{}", t.0, t.1, t.2)
}

fn arm_canonical(arm: &Arm) -> Option<String> {
    let lower = strongest_lower(&arm.comparisons);
    let upper = strongest_upper(&arm.comparisons);
    let mut anchors = arm.anchors.clone();
    anchors.sort_unstable();
    anchors.dedup();

    let mut parts = Vec::new();
    if let Some(b) = lower {
        let op = if b.inclusive { ">=" } else { ">" };
        parts.push(format!("{op}{}", version_string(&strip_build(b.version))));
    } else {
        parts.push("*".to_owned());
    }
    if let Some(b) = upper {
        let op = if b.inclusive { "<=" } else { "<" };
        parts.push(format!("{op}{}", version_string(&strip_build(b.version))));
    } else {
        parts.push("*".to_owned());
    }
    let anchor_text = if anchors.is_empty() {
        "-".to_owned()
    } else {
        anchors.iter().map(tuple_string).collect::<Vec<_>>().join(",")
    };
    parts.push(format!("pre={anchor_text}"));
    Some(format!("[{}]", parts.join(";")))
}

pub fn canonical(r: &Range) -> String {
    let mut arms: Vec<String> = r.arms.iter().filter_map(arm_canonical).collect();
    arms.sort();
    arms.dedup();
    if arms.is_empty() {
        "EMPTY".to_owned()
    } else {
        arms.join(" || ")
    }
}

#[derive(Clone, Debug)]
pub struct Constraint {
    pub label: String,
    pub expr: SetExpr,
}

#[derive(Clone, Debug)]
pub struct Package {
    pub name: String,
    pub candidates: Vec<String>,
    pub constraints: Vec<Constraint>,
}

#[derive(Clone, Copy, Debug)]
pub enum Preference {
    Highest,
    Lowest,
    StableFirst,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate {
        package: String,
        index: usize,
        error: VersionError,
    },
    InvalidRange {
        package: String,
        label: String,
        path: Vec<usize>,
        error: RangeError,
    },
    NoSolution { package: String, labels: Vec<String> },
}

fn validate_expr_ranges(
    expr: &SetExpr,
    package: &str,
    label: &str,
    path: &mut Vec<usize>,
    output: &mut Vec<(Vec<usize>, Range)>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_owned(),
                label: label.to_owned(),
                path: path.clone(),
                error,
            })?;
            output.push((path.clone(), r));
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for (i, child) in children.iter().enumerate() {
                path.push(i);
                validate_expr_ranges(child, package, label, path, output)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn eval_expr<F>(expr: &SetExpr, v: &Version, mut get_range: F) -> bool
where
    F: FnMut(&str) -> Option<Range>,
{
    match expr {
        SetExpr::Range(s) => get_range(s)
            .as_ref()
            .is_some_and(|r| matches_with(v, r, true)),
        SetExpr::Union(children) => children.iter().any(|x| eval_expr(x, v, &mut get_range)),
        SetExpr::Intersection(children) => {
            children.iter().all(|x| eval_expr(x, v, &mut get_range))
        }
    }
}

fn eval_expr_cached(expr: &SetExpr, v: &Version, ranges: &[(String, Range)], include: bool) -> bool {
    match expr {
        SetExpr::Range(s) => ranges
            .iter()
            .find(|(key, _)| key == s)
            .is_some_and(|(_, r)| matches_with(v, r, include)),
        SetExpr::Union(children) => children
            .iter()
            .any(|x| eval_expr_cached(x, v, ranges, include)),
        SetExpr::Intersection(children) => children
            .iter()
            .all(|x| eval_expr_cached(x, v, ranges, include)),
    }
}

fn combination_is_conflict(
    indexes: &[usize],
    constraints: &[Constraint],
    parsed: &[Version],
    ranges: &[(String, Range)],
    include: bool,
) -> bool {
    parsed.iter().all(|candidate| {
        indexes.iter().any(|&i| {
            !eval_expr_cached(
                &constraints[i].expr,
                candidate,
                ranges,
                include,
            )
        })
    })
}

fn find_conflict_labels(
    constraints: &[Constraint],
    parsed: &[Version],
    ranges: &[(String, Range)],
    include: bool,
) -> Vec<String> {
    if parsed.is_empty() {
        return Vec::new();
    }
    let n = constraints.len();
    for size in 1..=n {
        let mut best: Option<Vec<String>> = None;
        let mut indices: Vec<usize> = (0..size).collect();
        loop {
            if combination_is_conflict(&indices, constraints, parsed, ranges, include) {
                let mut labels: Vec<String> = indices
                    .iter()
                    .map(|&i| constraints[i].label.clone())
                    .collect();
                labels.sort();
                if best.as_ref().is_none_or(|old| labels < *old) {
                    best = Some(labels);
                }
            }

            let mut p = size;
            while p > 0 {
                p -= 1;
                if indices[p] < n - size + p {
                    indices[p] += 1;
                    for j in p + 1..size {
                        indices[j] = indices[j - 1] + 1;
                    }
                    break;
                }
            }
            if p == 0 && indices[0] > n - size {
                break;
            }
            if size == 0 {
                break;
            }
        }
        if let Some(labels) = best {
            return labels;
        }
    }
    Vec::new()
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for package in packages {
        if !names.insert(package.name.clone()) {
            return Err(SelectError::DuplicatePackage {
                name: package.name.clone(),
            });
        }
    }

    for package in packages {
        let mut labels = std::collections::HashSet::new();
        for constraint in &package.constraints {
            if !labels.insert(constraint.label.clone()) {
                return Err(SelectError::DuplicateLabel {
                    package: package.name.clone(),
                    label: constraint.label.clone(),
                });
            }
        }
    }

    let mut parsed_by_package: Vec<Vec<Version>> = Vec::with_capacity(packages.len());
    for package in packages {
        let mut parsed = Vec::with_capacity(package.candidates.len());
        for (index, candidate) in package.candidates.iter().enumerate() {
            parsed.push(parse_version(candidate).map_err(|error| {
                SelectError::InvalidCandidate {
                    package: package.name.clone(),
                    index,
                    error,
                }
            })?);
        }
        parsed_by_package.push(parsed);
    }

    let mut ranges_by_package: Vec<Vec<(String, Range)>> = Vec::with_capacity(packages.len());
    for package in packages {
        let mut ranges = Vec::new();
        for constraint in &package.constraints {
            let mut found = Vec::new();
            validate_expr_ranges(
                &constraint.expr,
                &package.name,
                &constraint.label,
                &mut Vec::new(),
                &mut found,
            )?;
            for (_, range) in found {
                if let SetExpr::Range(_) = constraint.expr {
                    // Root-leaf ranges are also cached below through tree traversal.
                    let _ = range;
                }
            }
            collect_range_strings(&constraint.expr, &mut ranges)?;
        }
        ranges_by_package.push(ranges);
    }

    let mut selections = Vec::with_capacity(packages.len());
    for (package_index, package) in packages.iter().enumerate() {
        let parsed = &parsed_by_package[package_index];
        let ranges = &ranges_by_package[package_index];
        let mut qualifying: Vec<usize> = Vec::new();
        'candidate: for i in 0..parsed.len() {
            for constraint in &package.constraints {
                if !eval_expr_cached(
                    &constraint.expr,
                    &parsed[i],
                    ranges,
                    include_prerelease,
                ) {
                    continue 'candidate;
                }
            }
            qualifying.push(i);
        }

        if qualifying.is_empty() {
            let labels = find_conflict_labels(
                &package.constraints,
                parsed,
                ranges,
                include_prerelease,
            );
            return Err(SelectError::NoSolution {
                package: package.name.clone(),
                labels,
            });
        }

        let mut best = qualifying[0];
        for &i in qualifying.iter().skip(1) {
            let stable_first_better = matches!(preference, Preference::StableFirst)
                && parsed[i].pre.is_empty()
                && !parsed[best].pre.is_empty();
            let stable_first_worse = matches!(preference, Preference::StableFirst)
                && !parsed[i].pre.is_empty()
                && parsed[best].pre.is_empty();
            if stable_first_better {
                best = i;
                continue;
            }
            if stable_first_worse {
                continue;
            }
            let cmp = compare_versions(&parsed[i], &parsed[best]);
            let better = match preference {
                Preference::Highest | Preference::StableFirst => cmp == Ordering::Greater,
                Preference::Lowest => cmp == Ordering::Less,
            };
            if better {
                best = i;
            }
        }
        selections.push(Selection {
            package: package.name.clone(),
            version: package.candidates[best].clone(),
        });
    }
    Ok(selections)
}

fn collect_range_strings(
    expr: &SetExpr,
    ranges: &mut Vec<(String, Range)>,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            if !ranges.iter().any(|(k, _)| k == s) {
                if let Ok(r) = parse_range(s) {
                    ranges.push((s.clone(), r));
                }
            }
        }
        SetExpr::Union(children) | SetExpr::Intersection(children) => {
            for child in children {
                collect_range_strings(child, ranges)?;
            }
        }
    }
    Ok(())
}

fn successor_version(v: &Version) -> Option<Version> {
    if !v.pre.is_empty() {
        let mut next = v.clone();
        next.pre.push("0".to_owned());
        next.build.clear();
        Some(next)
    } else if v.patch < u64::MAX {
        Some(endpoint(v.major, v.minor, v.patch + 1).stable())
    } else if v.minor < u64::MAX {
        Some(endpoint(v.major, v.minor + 1, 0).stable())
    } else if v.major < u64::MAX {
        Some(endpoint(v.major + 1, 0, 0).stable())
    } else {
        None
    }
}

trait StableVersion {
    fn stable(self) -> Version;
}

impl StableVersion for Version {
    fn stable(mut self) -> Version {
        self.pre.clear();
        self
    }
}

fn predecessor_version(v: &Version) -> Option<Version> {
    if !v.pre.is_empty() {
        let mut p = v.clone();
        if p.pre.last().is_some_and(|x| x == "0") {
            p.pre.pop();
            Some(p)
        } else {
            None
        }
    } else {
        None
    }
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    for arm_a in &a.arms {
        for arm_b in &b.arms {
            let mut comps = arm_a.comparisons.clone();
            comps.extend(arm_b.comparisons.clone());
            let anchors_a = &arm_a.anchors;
            let anchors_b = &arm_b.anchors;

            let mut candidates = Vec::new();
            for c in &comps {
                candidates.push(c.version.clone());
                if let Some(s) = successor_version(&c.version) {
                    candidates.push(s);
                }
                if let Some(p) = predecessor_version(&c.version) {
                    candidates.push(p);
                }
            }
            candidates.push(Version {
                major: 0,
                minor: 0,
                patch: 0,
                pre: Vec::new(),
                build: Vec::new(),
            });

            let lo = strongest_lower(&comps);
            if let Some(l) = &lo {
                candidates.push(l.version.clone());
                if !l.inclusive {
                    if let Some(s) = successor_version(&l.version) {
                        candidates.push(s);
                    }
                }
            }
            let hi = strongest_upper(&comps);
            if let Some(h) = &hi {
                candidates.push(h.version.clone());
                if !h.inclusive {
                    if let Some(p) = predecessor_version(&h.version) {
                        candidates.push(p);
                    }
                }
            }

            if !include_prerelease {
                for &(major, minor, patch) in anchors_a.iter().filter(|x| anchors_b.contains(x)) {
                    candidates.push(Version {
                        major,
                        minor,
                        patch,
                        pre: vec!["0".to_owned()],
                        build: Vec::new(),
                    });
                }
            }

            for candidate in candidates {
                if !comps.iter().all(|c| satisfies_comparator(&candidate, c)) {
                    continue;
                }
                if candidate.pre.is_empty() {
                    return true;
                }
                if include_prerelease {
                    return true;
                }
                let tuple = (candidate.major, candidate.minor, candidate.patch);
                if anchors_a.contains(&tuple) && anchors_b.contains(&tuple) {
                    return true;
                }
            }
        }
    }
    false
}

#[cfg(test)]
mod tests {
    use super::*;

    fn pv(s: &str) -> Version {
        parse_version(s).unwrap()
    }

    fn pr(s: &str) -> Range {
        parse_range(s).unwrap()
    }

    #[test]
    fn public_versions_ranges_extrema() {
        let v = pv("2.1.3-rc.1+build.7");
        assert_eq!((v.major, v.minor, v.patch), (2, 1, 3));
        assert_eq!(v.pre, vec!["rc", "1"]);
        assert_eq!(v.build, vec!["build", "7"]);
        assert_eq!(
            compare_versions(&pv("2.1.3-alpha"), &pv("2.1.3-beta")),
            Ordering::Less
        );
        let r = pr(">=2.1.0 <3.0.0");
        assert!(matches(&pv("2.5.0"), &r));
        assert!(!matches(&pv("3.0.0"), &r));
        assert!(matches(
            &pv("2.1.3-rc"),
            &pr(">=2.1.3-beta <2.1.3")
        ));
        for s in ["^2.1.0", "~2.5.0", "2.5.x", "2.1 - 2.8"] {
            assert!(matches(&pv("2.5.1"), &pr(s)));
        }
        assert!(matches(&pv("4.0.0"), &pr("2.x || 4.0.0")));
        let cs = vec![
            "2.2.0".to_string(),
            "2.8.0".to_string(),
            "3.0.0".to_string(),
        ];
        let r = pr("2.x");
        assert_eq!(max_satisfying(&cs, &r), Ok(Some("2.8.0".into())));
        assert_eq!(min_satisfying(&cs, &r), Ok(Some("2.2.0".into())));
    }

    #[test]
    fn public_algebra() {
        let r = pr(">=2.1.0 <2.8.0");
        assert!(intersects(&r, &pr("2.5.x"), false));
        assert_eq!(canonical(&r), "[>=2.1.0;<2.8.0;pre=-]");
        assert!(matches_with(&pv("2.5.0-rc"), &r, true));
    }

    #[test]
    fn public_packages() {
        let ps = vec![Package {
            name: "demo".into(),
            candidates: vec!["2.1.0".into(), "2.7.0".into()],
            constraints: vec![Constraint {
                label: "range".into(),
                expr: SetExpr::Range("2.x".into()),
            }],
        }];
        assert_eq!(
            select_packages(&ps, Preference::Highest, false),
            Ok(vec![Selection {
                package: "demo".into(),
                version: "2.7.0".into()
            }])
        );
        assert_eq!(
            select_packages(&ps, Preference::Lowest, false).unwrap()[0].version,
            "2.1.0"
        );
        assert_eq!(
            select_packages(&ps, Preference::StableFirst, false).unwrap()[0].version,
            "2.7.0"
        );
    }

    #[test]
    fn public_sets_diagnosis() {
        let p = Package {
            name: "demo".into(),
            candidates: vec!["2.1.0".into()],
            constraints: vec![Constraint {
                label: "blocked".into(),
                expr: SetExpr::Intersection(vec![SetExpr::Union(vec![
                    SetExpr::Range("3.x".into()),
                    SetExpr::Range("4.x".into()),
                ])]),
            }],
        };
        assert_eq!(
            select_packages(&[p], Preference::Highest, false),
            Err(SelectError::NoSolution {
                package: "demo".into(),
                labels: vec!["blocked".into()]
            })
        );
    }
}
