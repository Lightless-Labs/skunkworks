use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn is_digit(b: u8) -> bool {
    b.is_ascii_digit()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let bytes = s.as_bytes();
    let start = *pos;
    while *pos < bytes.len() && is_digit(bytes[*pos]) {
        *pos += 1;
    }
    if *pos == start {
        if *pos == bytes.len() || matches!(bytes[*pos], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    let mut value = 0u64;
    for &b in &bytes[start..*pos] {
        value = value
            .checked_mul(10)
            .and_then(|v| v.checked_add((b - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    is_pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    loop {
        let id_start = p;
        while p < end && bytes[p] != b'.' {
            let b = bytes[p];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: p });
            }
            p += 1;
        }
        if p == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        if is_pre {
            let part = &bytes[id_start..p];
            if part.iter().all(|b| is_digit(*b)) && part.len() > 1 && part[0] == b'0' {
                return Err(VersionError::LeadingZero { at: id_start });
            }
        }
        out.push(s[id_start..p].to_string());
        if p == end {
            break;
        }
        p += 1;
        if p == end {
            return Err(VersionError::EmptyIdentifier { at: p });
        }
    }
    Ok(out)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    for _ in 0..2 {
        if pos >= s.len() {
            return Err(VersionError::MissingCore { at: pos });
        }
        if s.as_bytes()[pos] != b'.' {
            if matches!(s.as_bytes()[pos], b'-' | b'+') {
                return Err(VersionError::MissingCore { at: pos });
            }
            return Err(VersionError::InvalidChar { at: pos });
        }
        pos += 1;
        if _ == 0 {
            let minor = parse_core(s, &mut pos)?;
            if pos >= s.len() {
                return Err(VersionError::MissingCore { at: pos });
            }
            if s.as_bytes()[pos] != b'.' {
                if matches!(s.as_bytes()[pos], b'-' | b'+') {
                    return Err(VersionError::MissingCore { at: pos });
                }
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
            let patch = parse_core(s, &mut pos)?;
            return finish_version(s, pos, major, minor, patch);
        }
    }
    unreachable!()
}

fn finish_version(
    s: &str,
    mut pos: usize,
    major: u64,
    minor: u64,
    patch: u64,
) -> Result<Version, VersionError> {
    let bytes = s.as_bytes();
    let mut pre = Vec::new();
    let mut build = Vec::new();
    if pos < bytes.len() {
        match bytes[pos] {
            b'-' => {
                pos += 1;
                let plus = bytes[pos..].iter().position(|b| *b == b'+').map(|x| pos + x);
                let end = plus.unwrap_or(bytes.len());
                pre = parse_identifiers(s, pos, end, true)?;
                pos = end;
                if pos < bytes.len() {
                    pos += 1;
                    build = parse_identifiers(s, pos, bytes.len(), false)?;
                }
            }
            b'+' => {
                pos += 1;
                build = parse_identifiers(s, pos, bytes.len(), false)?;
            }
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }
    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn core(v: &Version) -> (u64, u64, u64) {
    (v.major, v.minor, v.patch)
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.as_bytes().iter().all(|b| is_digit(*b))
}

fn cmp_numeric_id(a: &str, b: &str) -> Ordering {
    a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes()))
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    core(a).cmp(&core(b)).then_with(|| {
        match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let nx = numeric_id(x);
                    let ny = numeric_id(y);
                    let c = match (nx, ny) {
                        (true, true) => cmp_numeric_id(x, y),
                        (true, false) => Ordering::Less,
                        (false, true) => Ordering::Greater,
                        (false, false) => x.as_bytes().cmp(y.as_bytes()),
                    };
                    if c != Ordering::Equal {
                        return c;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        }
    })
}

#[derive(Clone, Debug)]
struct Arm {

    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
struct Test {
    op: Op,
    version: Version,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Copy)]
struct Tok<'a> {
    s: &'a str,
    at: usize,
}

fn ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, start: usize, end: usize) -> Vec<Tok<'_>> {
    let b = s.as_bytes();
    let mut out = Vec::new();
    let mut p = start;
    while p < end {
        while p < end && ws(b[p]) {
            p += 1;
        }
        if p == end {
            break;
        }
        let st = p;
        while p < end && !ws(b[p]) {
            p += 1;
        }
        out.push(Tok { s: &s[st..p], at: st });
    }
    out
}

fn wrap_version(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

fn padded(nums: &[u64]) -> Version {
    Version {
        major: *nums.get(0).unwrap_or(&0),
        minor: *nums.get(1).unwrap_or(&0),
        patch: *nums.get(2).unwrap_or(&0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut n = v.clone();
    n.pre = vec!["0".to_string()];
    n.build.clear();
    let target = match field {
        0 => &mut n.major,
        1 => &mut n.minor,
        _ => &mut n.patch,
    };
    *target = target.checked_add(1).ok_or(RangeError::Syntax { at })?;
    match field {
        0 => {
            n.minor = 0;
            n.patch = 0;
        }
        1 => n.patch = 0,
        _ => {}
    }
    Ok(n)
}

fn partial(tok: Tok<'_>) -> Result<(Vec<u64>, usize, bool), RangeError> {
    let fields: Vec<&str> = tok.s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at: tok.at });
    }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for (i, f) in fields.iter().enumerate() {
        if *f == "*" || *f == "x" || *f == "X" {
            wildcard = true;
            if fields[i + 1..].iter().any(|z| !(*z == "*" || *z == "x" || *z == "X")) {
                return Err(RangeError::Syntax { at: tok.at });
            }
            break;
        }
        if f.is_empty() || !f.as_bytes().iter().all(|b| is_digit(*b)) {
            return Err(RangeError::Syntax { at: tok.at });
        }
        if f.len() > 1 && f.starts_with('0') {
            return Err(RangeError::Syntax { at: tok.at });
        }
        let n = f.parse::<u64>().map_err(|_| RangeError::Syntax { at: tok.at })?;
        nums.push(n);
    }
    let k = nums.len();
    if wildcard {
        Ok((nums, k, true))
    } else {
        Ok((nums, k, fields.len() < 3))
    }
}

fn add_test(arm: &mut Arm, op: Op, version: Version, anchor: bool) {
    if anchor && !version.pre.is_empty() {
        arm.anchors.push(core(&version));
    }
    arm.tests.push(Test { op, version });
}

fn parse_operand(tok: Tok<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let s = tok.s;
    if let Some(opbyte) = s.as_bytes().first().copied() {
        if matches!(opbyte, b'=' | b'<' | b'>') {
            let (op, len) = if s.starts_with("<=") {
                (Op::Le, 2)
            } else if s.starts_with(">=") {
                (Op::Ge, 2)
            } else {
                (match opbyte {
                    b'=' => Op::Eq,
                    b'<' => Op::Lt,
                    _ => Op::Gt,
                }, 1)
            };
            if len == s.len() || !is_digit(s.as_bytes()[len]) {
                return Err(RangeError::Syntax { at: tok.at });
            }
            let v = wrap_version(&s[len..], tok.at + len)?;
            add_test(arm, op, v, true);
            return Ok(());
        }
    }

    if s.starts_with('^') || s.starts_with('~') {
        let sign = s.as_bytes()[0];
        let inner = Tok { s: &s[1..], at: tok.at + 1 };
        let is_full = (inner.s.contains('-') || inner.s.contains('+'))
            || (inner.s.split('.').count() >= 3
                && !inner.s.split('.').any(|f| matches!(f, "*" | "x" | "X")));
        let (lo, k) = if is_full {
            let v = wrap_version(inner.s, inner.at)?;
            (v, 3)
        } else {
            let (nums, k, _) = partial(inner)?;
            if k == 0 {
                return Err(RangeError::Syntax { at: tok.at });
            }
            (padded(&nums), k)
        };
        add_test(arm, Op::Ge, lo.clone(), true);
        let field = if sign == b'^' {
            if lo.major > 0 || k == 1 { 0 }
            else if lo.minor > 0 || k == 2 { 1 }
            else { 2 }
        } else if k == 1 { 0 } else { 1 };
        let hi = bump(&lo, field, tok.at)?;
        add_test(arm, Op::Lt, hi, false);
        return Ok(());
    }

    let is_full = s.contains('-') || s.contains('+')
        || (s.split('.').count() >= 3 && !s.split('.').any(|f| matches!(f, "*" | "x" | "X")));
    if is_full {
        let v = wrap_version(s, tok.at)?;
        add_test(arm, Op::Eq, v, true);
        return Ok(());
    }
    let (nums, k, _) = partial(tok)?;
    let lo = padded(&nums);
    if k == 0 {
        return Ok(());
    }
    if k == 3 {
        add_test(arm, Op::Eq, lo, false);
    } else {
        add_test(arm, Op::Ge, lo.clone(), false);
        let hi = bump(&lo, k - 1, tok.at)?;
        add_test(arm, Op::Lt, hi, false);
    }
    Ok(())
}

fn parse_hyphen_arm(ts: &[Tok<'_>], arm: &mut Arm) -> Result<(), RangeError> {
    let minus = ts.iter().find(|t| t.s == "-").unwrap();
    if ts.len() != 3 || ts[1].s != "-" {
        return Err(RangeError::Syntax { at: minus.at });
    }
    let (ln, lk, _) = partial(ts[0])?;
    let (un, uk, _) = partial(ts[2])?;
    if lk == 0 || uk == 0 {
        return Err(RangeError::Syntax { at: if lk == 0 { ts[0].at } else { ts[2].at } });
    }
    let lower = padded(&ln);
    add_test(arm, Op::Ge, lower, true);
    if uk == 3 && (ts[2].s.contains('-') || ts[2].s.contains('+')) {
        let upper = wrap_version(ts[2].s, ts[2].at)?;
        add_test(arm, Op::Le, upper, true);
    } else if uk == 3 {
        let upper = wrap_version(ts[2].s, ts[2].at)?;
        add_test(arm, Op::Le, upper, true);
    } else {
        let upper = padded(&un);
        let hi = bump(&upper, uk - 1, ts[2].at)?;
        add_test(arm, Op::Lt, hi, false);
    }
    Ok(())
}

fn parse_arm(s: &str, start: usize, end: usize) -> Result<Arm, RangeError> {
    let ts = tokens(s, start, end);
    let mut arm = Arm { tests: Vec::new(), anchors: Vec::new() };
    if ts.iter().any(|t| t.s == "-") {
        parse_hyphen_arm(&ts, &mut arm)?;
    } else {
        for t in ts {
            let first = t.s.as_bytes()[0];
            if !(is_digit(first) || matches!(first, b'*' | b'x' | b'X' | b'^' | b'~' | b'<' | b'>' | b'=')) {
                return Err(RangeError::Syntax { at: t.at });
            }
            parse_operand(t, &mut arm)?;
        }
    }
    Ok(arm)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let b = s.as_bytes();
    let mut arms = Vec::new();
    let mut p = 0;
    let mut saw_non_ws = false;
    while p < b.len() {
        while p < b.len() && ws(b[p]) { p += 1; }
        if p == b.len() { break; }
        saw_non_ws = true;
        let start = p;
        let mut sep = None;
        while p < b.len() {
            if b[p] == b'|' && p + 1 < b.len() && b[p + 1] == b'|' {
                let before = p == start || ws(b[p - 1]);
                let after = p + 2 == b.len() || ws(b[p + 2]);
                if before && after {
                    sep = Some(p);
                    break;
                }
            }
            p += 1;
        }
        let end = sep.unwrap_or(b.len());
        let trimmed_end = {
            let mut x = end;
            while x > start && ws(b[x - 1]) { x -= 1; }
            x
        };
        if start == trimmed_end {
            return Err(RangeError::Empty { at: start });
        }
        arms.push(parse_arm(s, start, trimmed_end)?);
        if let Some(at) = sep {
            p = at + 2;
            while p < b.len() && ws(b[p]) { p += 1; }
            if p == b.len() {
                return Err(RangeError::Empty { at: b.len() });
            }
        } else {
            break;
        }
    }
    if !saw_non_ws {
        return Err(RangeError::Empty { at: 0 });
    }
    Ok(Range { arms })
}

fn test_passes(v: &Version, t: &Test) -> bool {
    let c = compare_versions(v, &t.version);
    match t.op {
        Op::Eq => c == Ordering::Equal,
        Op::Lt => c == Ordering::Less,
        Op::Le => c != Ordering::Greater,
        Op::Gt => c == Ordering::Greater,
        Op::Ge => c != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm, include: bool) -> bool {
    if !arm.tests.iter().all(|t| test_passes(v, t)) {
        return false;
    }
    include || v.pre.is_empty() || arm.anchors.iter().any(|c| *c == core(v))
}

pub fn matches_with(v: &Version, r: &Range, include_prerelease: bool) -> bool {
    r.arms.iter().any(|a| arm_matches(v, a, include_prerelease))
}

pub fn matches(v: &Version, r: &Range) -> bool {
    matches_with(v, r, false)
}

pub fn max_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, true)
}

pub fn min_satisfying(candidates: &[String], r: &Range) -> Result<Option<String>, CandidateError> {
    extrema(candidates, r, false)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn extrema(candidates: &[String], r: &Range, maximum: bool) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        parsed.push(parse_version(s).map_err(|error| CandidateError { index, error })?);
    }
    let mut best: Option<usize> = None;
    for i in 0..parsed.len() {
        if !matches(&parsed[i], r) { continue; }
        best = match best {
            None => Some(i),
            Some(j) => {
                let c = compare_versions(&parsed[i], &parsed[j]);
                if (maximum && c == Ordering::Greater) || (!maximum && c == Ordering::Less) { Some(i) } else { Some(j) }
            }
        };
    }
    Ok(best.map(|i| candidates[i].clone()))
}

#[derive(Clone, Debug)]

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: String,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SelectError {
    DuplicatePackage { name: String },
    DuplicateLabel { package: String, label: String },
    InvalidCandidate { package: String, index: usize, error: VersionError },
    InvalidRange { package: String, label: String, path: Vec<usize>, error: RangeError },
    NoSolution { package: String, labels: Vec<String> },
}

fn expr_ranges(
    expr: &SetExpr,
    path: &mut Vec<usize>,
    out: &mut Vec<(Vec<usize>, Range)>,
    package: &str,
    label: &str,
) -> Result<(), SelectError> {
    match expr {
        SetExpr::Range(s) => {
            let r = parse_range(s).map_err(|error| SelectError::InvalidRange {
                package: package.to_string(),
                label: label.to_string(),
                path: path.clone(),
                error,
            })?;
            out.push((path.clone(), r));
        }
        SetExpr::Union(xs) | SetExpr::Intersection(xs) => {
            for (i, x) in xs.iter().enumerate() {
                path.push(i);
                expr_ranges(x, path, out, package, label)?;
                path.pop();
            }
        }
    }
    Ok(())
}

fn eval_expr(expr: &SetExpr, v: &Version, include: bool, map: &[Range], cursor: &mut usize) -> bool {
    match expr {
        SetExpr::Range(_) => {
            let r = &map[*cursor];
            *cursor += 1;
            matches_with(v, r, include)
        }
        SetExpr::Union(xs) => xs.iter().any(|x| eval_expr(x, v, include, map, cursor)),
        SetExpr::Intersection(xs) => xs.iter().all(|x| eval_expr(x, v, include, map, cursor)),
    }
}

pub fn select_packages(
    packages: &[Package],
    preference: Preference,
    include_prerelease: bool,
) -> Result<Vec<Selection>, SelectError> {
    let mut names = std::collections::HashSet::new();
    for p in packages {
        if !names.insert(p.name.clone()) {
            return Err(SelectError::DuplicatePackage { name: p.name.clone() });
        }
    }
    for p in packages {
        let mut labels = std::collections::HashSet::new();
        for c in &p.constraints {
            if !labels.insert(c.label.clone()) {
                return Err(SelectError::DuplicateLabel { package: p.name.clone(), label: c.label.clone() });
            }
        }
    }
    let mut versions: Vec<Vec<Version>> = Vec::new();
    for p in packages {
        let mut vs = Vec::new();
        for (i, s) in p.candidates.iter().enumerate() {
            vs.push(parse_version(s).map_err(|error| SelectError::InvalidCandidate {
                package: p.name.clone(), index: i, error
            })?);
        }
        versions.push(vs);
    }
    let mut parsed_ranges: Vec<Vec<Vec<Range>>> = Vec::new();
    for p in packages {
        let mut cs = Vec::new();
        for c in &p.constraints {
            let mut leaves = Vec::new();
            expr_ranges(&c.expr, &mut Vec::new(), &mut leaves, &p.name, &c.label)?;
            cs.push(leaves.into_iter().map(|(_, r)| r).collect());
        }
        parsed_ranges.push(cs);
    }

    let mut result = Vec::new();
    for (pi, p) in packages.iter().enumerate() {
        let mut qualifying = Vec::new();
        for i in 0..p.candidates.len() {
            let mut accepted = true;
            for (ci, c) in p.constraints.iter().enumerate() {
                let mut cursor = 0;
                if !eval_expr(&c.expr, &versions[pi][i], include_prerelease, &parsed_ranges[pi][ci], &mut cursor) {
                    accepted = false;
                    break;
                }
            }
            if accepted { qualifying.push(i); }
        }
        if qualifying.is_empty() {
            let mut labels = Vec::new();
            if !p.candidates.is_empty() {
                let n = p.constraints.len();
                'find: for size in 1..=n {
                    let mut choice = Vec::new();
                    if let Some(found) = find_conflict(
                        0, size, &mut choice, p, &versions[pi], &parsed_ranges[pi], include_prerelease
                    ) {
                        labels = found;
                        break 'find;
                    }
                }
            }
            return Err(SelectError::NoSolution { package: p.name.clone(), labels });
        }
        let pick = match preference {
            Preference::Highest => best_index(&qualifying, &versions[pi], true, false),
            Preference::Lowest => best_index(&qualifying, &versions[pi], false, false),
            Preference::StableFirst => {
                let stable: Vec<usize> = qualifying.iter().copied().filter(|i| versions[pi][*i].pre.is_empty()).collect();
                if stable.is_empty() { best_index(&qualifying, &versions[pi], true, false) }
                else { best_index(&stable, &versions[pi], true, false) }
            }
        };
        result.push(Selection { package: p.name.clone(), version: p.candidates[pick].clone() });
    }
    Ok(result)
}

fn best_index(xs: &[usize], vs: &[Version], max: bool, _: bool) -> usize {
    let mut best = xs[0];
    for &i in &xs[1..] {
        let c = compare_versions(&vs[i], &vs[best]);
        if (max && c == Ordering::Greater) || (!max && c == Ordering::Less) {
            best = i;
        }
    }
    best
}

fn find_conflict(
    start: usize,
    need: usize,
    choice: &mut Vec<usize>,
    p: &Package,
    versions: &[Version],
    ranges: &[Vec<Range>],
    include: bool,
) -> Option<Vec<String>> {
    if need == 0 {
        for i in 0..versions.len() {
            let mut ok = true;
            for &ci in choice.iter() {
                let mut cursor = 0;
                if !eval_expr(&p.constraints[ci].expr, &versions[i], include, &ranges[ci], &mut cursor) {
                    ok = false;
                    break;
                }
            }
            if ok { return None; }
        }
        let mut labels: Vec<String> = choice.iter().map(|i| p.constraints[*i].label.clone()).collect();
        labels.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
        return Some(labels);
    }
    if p.constraints.len().saturating_sub(start) < need { return None; }
    for i in start..=p.constraints.len() - need {
        choice.push(i);
        if let Some(result) = find_conflict(i + 1, need - 1, choice, p, versions, ranges, include) {
            return Some(result);
        }
        choice.pop();
    }
    None
}

pub fn intersects(a: &Range, b: &Range, include_prerelease: bool) -> bool {
    let mut boundaries = Vec::new();
    for r in [a, b] {
        for arm in &r.arms {
            for t in &arm.tests {
                boundaries.push(t.version.clone());
            }
        }
    }
    boundaries.sort_by(compare_versions);
    boundaries.dedup_by(|x, y| compare_versions(x, y) == Ordering::Equal);
    for v in &boundaries {
        if matches_with(v, a, include_prerelease) && matches_with(v, b, include_prerelease) {
            return true;
        }
        let stable = Version { major: v.major, minor: v.minor, patch: v.patch, pre: Vec::new(), build: Vec::new() };
        if matches_with(&stable, a, include_prerelease) && matches_with(&stable, b, include_prerelease) {
            return true;
        }
        let low = Version { major: v.major, minor: v.minor, patch: v.patch, pre: vec!["0".into()], build: Vec::new() };
        if matches_with(&low, a, include_prerelease) && matches_with(&low, b, include_prerelease) {
            return true;
        }
    }
    let zero = Version { major: 0, minor: 0, patch: 0, pre: Vec::new(), build: Vec::new() };
    (matches_with(&zero, a, include_prerelease) && matches_with(&zero, b, include_prerelease))
        || (include_prerelease && {
            let z = Version { pre: vec!["0".into()], ..zero.clone() };
            matches_with(&z, a, true) && matches_with(&z, b, true)
        })
}

pub fn canonical(r: &Range) -> String {
    let mut arms = Vec::new();
    for arm in &r.arms {
        let mut lower: Option<(Version, bool)> = None;
        let mut upper: Option<(Version, bool)> = None;
        let mut equality: Option<Version> = None;
        for t in &arm.tests {
            match t.op {
                Op::Eq => equality = Some(t.version.clone()),
                Op::Ge | Op::Gt => {
                    let strict = matches!(t.op, Op::Gt);
                    if lower.as_ref().map_or(true, |(v, s)| {
                        compare_versions(&t.version, v) == Ordering::Greater
                            || (compare_versions(&t.version, v) == Ordering::Equal && strict && !*s)
                    }) {
                        lower = Some((t.version.clone(), strict));
                    }
                }
                Op::Le | Op::Lt => {
                    let strict = matches!(t.op, Op::Lt);
                    if upper.as_ref().map_or(true, |(v, s)| {
                        compare_versions(&t.version, v) == Ordering::Less
                            || (compare_versions(&t.version, v) == Ordering::Equal && strict && !*s)
                    }) {
                        upper = Some((t.version.clone(), strict));
                    }
                }
            }
        }
        if let Some(eq) = equality {
            lower = Some((eq.clone(), false));
            upper = Some((eq, false));
        }
        let lo = lower.map_or("*".to_string(), |(v, strict)| format!("{}{}", if strict { ">" } else { ">=" }, display_version(&v)));
        let hi = upper.map_or("*".to_string(), |(v, strict)| format!("{}{}", if strict { "<" } else { "<=" }, display_version(&v)));
        let mut anchors = arm.anchors.clone();
        anchors.sort();
        anchors.dedup();
        let pre = if anchors.is_empty() { "-".to_string() } else {
            anchors.iter().map(|(a,b,c)| format!("{a}.{b}.{c}")).collect::<Vec<_>>().join(",")
        };
        arms.push(format!("[{lo};{hi};pre={pre}]"));
    }
    arms.sort_by(|a,b| a.as_bytes().cmp(b.as_bytes()));
    arms.dedup();
    if arms.is_empty() { "EMPTY".to_string() } else { arms.join(" || ") }
}

fn display_version(v: &Version) -> String {
    let mut s = format!("{}.{}.{}", v.major, v.minor, v.patch);
    if !v.pre.is_empty() {
        s.push('-');
        s.push_str(&v.pre.join("."));
    }
    s
}
