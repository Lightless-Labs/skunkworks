```rust
use std::collections::HashMap;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Product {
    pub id: String,
    pub price: u16,
    pub stock: u16,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct State {
    pub credit: u16,
    pub selected: Option<String>,
    pub escrow: Vec<u16>,
    pub tubes: [u16; 4],
    pub maintenance: bool,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Receipt {
    pub id: String,
    pub price: u16,
    pub change: Vec<u16>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    EmptyCatalog,
    BadId { id: String },
    DuplicateId { id: String },
    BadPrice { id: String },
    BadCoin { coin: u16 },
    CreditLimit,
    UnknownProduct { id: String },
    SameSelection,
    NoSelection,
    Underpaid { price: u16, credit: u16 },
    TubeLimit { coin: u16 },
    NoChange { amount: u16 },
    StockLimit { id: String },
    SoldOut { id: String },
    Busy,
    ZeroRestock,
    MaintenanceActive,
    MaintenanceRequired,
    AlreadyMode,
    ReconcileMismatch { coin: u16, expected: u16, found: u16 },
    ValueDrift { before: u64, after: u64 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum EventKind {
    Inserted { coin: u16, credit: u16 },
    Selected { id: String },
    Vended { id: String, price: u16, change: Vec<u16> },
    Refunded { coins: Vec<u16> },
    PriceChanged { id: String, old: u16, new: u16 },
    Restocked { id: String, added: u16, stock: u16 },
    Maintenance { enabled: bool },
    Reconciled { before: [u16; 4], after: [u16; 4] },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Event {
    pub seq: usize,
    pub kind: EventKind,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Audit {
    pub inserted: u64,
    pub refunded: u64,
    pub sales: u64,
    pub change: u64,
    pub adjustment: i64,
    pub cash: u64,
}

pub struct Machine {
    products: Vec<Product>,
    index: HashMap<String, usize>,
    credit: u16,
    selected: Option<String>,
    escrow: Vec<u16>,
    tubes: [u16; 4],
    maintenance: bool,
    events: Vec<Event>,
    inserted: u64,
    refunded: u64,
    sales: u64,
    change: u64,
    adjustment: i64,
}

const DENOMS: [u16; 4] = [1, 3, 4, 10];

impl Machine {
    pub fn new(products: Vec<Product>, tubes: [u16; 4]) -> Result<Self, Error> {
        if products.is_empty() {
            return Err(Error::EmptyCatalog);
        }

        let mut seen = HashMap::new();
        for product in &products {
            if product.id.is_empty()
                || product.id.len() > 8
                || !product.id.bytes().all(|b| b.is_ascii_lowercase())
            {
                return Err(Error::BadId { id: product.id.clone() });
            }
            if seen.contains_key(&product.id) {
                return Err(Error::DuplicateId { id: product.id.clone() });
            }
            if !(1..=100).contains(&product.price) {
                return Err(Error::BadPrice { id: product.id.clone() });
            }
            if product.stock > 100 {
                return Err(Error::StockLimit { id: product.id.clone() });
            }
            seen.insert(product.id.clone(), seen.len());
        }

        for (i, count) in tubes.iter().enumerate() {
            if *count > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }

        Ok(Self {
            products,
            index: seen,
            credit: 0,
            selected: None,
            escrow: Vec::new(),
            tubes,
            maintenance: false,
            events: Vec::new(),
            inserted: 0,
            refunded: 0,
            sales: 0,
            change: 0,
            adjustment: 0,
        })
    }

    pub fn state(&self) -> State {
        State {
            credit: self.credit,
            selected: self.selected.clone(),
            escrow: self.escrow.clone(),
            tubes: self.tubes,
            maintenance: self.maintenance,
        }
    }

    fn log(&mut self, kind: EventKind) {
        self.events.push(Event {
            seq: self.events.len(),
            kind,
        });
    }

    pub fn insert(&mut self, coin: u16) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        if !DENOMS.contains(&coin) {
            return Err(Error::BadCoin { coin });
        }
        let new_credit = self.credit as u32 + coin as u32;
        if new_credit > 100 {
            return Err(Error::CreditLimit);
        }
        self.credit = new_credit as u16;
        self.escrow.push(coin);
        self.inserted = self.inserted.saturating_add(coin as u64);
        self.log(EventKind::Inserted { coin, credit: self.credit });
        Ok(())
    }

    pub fn select(&mut self, id: &str) -> Result<(), Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(&idx) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.selected.as_deref() == Some(id) {
            return Err(Error::SameSelection);
        }
        let selected = self.products[idx].id.clone();
        self.selected = Some(selected.clone());
        self.log(EventKind::Selected { id: selected });
        Ok(())
    }

    fn make_change(amount: u16, tubes: [u16; 4]) -> Option<Vec<u16>> {
        if amount == 0 {
            return Some(Vec::new());
        }
        let mut remaining = amount as u32;
        let mut greedy = [0u16; 4];
        for i in (0..4).rev() {
            let take = (tubes[i] as u32).min(remaining / DENOMS[i] as u32);
            greedy[i] = take as u16;
            remaining -= take * DENOMS[i] as u32;
        }
        if remaining == 0 {
            return Some(Self::descending_vector(greedy));
        }

        fn search(
            i: i32,
            remaining: u32,
            tubes: &[u16; 4],
            current: &mut [u16; 4],
            best: &mut Option<[u16; 4]>,
        ) {
            if remaining == 0 {
                let replace = match best {
                    None => true,
                    Some(old) => {
                        let total_new: u32 = current.iter().map(|&x| x as u32).sum();
                        let total_old: u32 = old.iter().map(|&x| x as u32).sum();
                        total_new < total_old || (total_new == total_old && {
                            for j in (0..4).rev() {
                                if current[j] != old[j] {
                                    return_if_lex_greater(current, old, j)
                                }
                            }
                            false
                        })
                    }
                };
                if replace {
                    *best = Some(*current);
                }
                return;
            }
            if i < 0 {
                return;
            }
            let idx = i as usize;
            let denom = DENOMS[idx] as u32;
            let max = (tubes[idx] as u32).min(remaining / denom);
            for count in 0..=max {
                current[idx] = count as u16;
                search(i - 1, remaining - count * denom, tubes, current, best);
            }
            current[idx] = 0;
        }

        fn return_if_lex_greater(current: &[u16; 4], old: &[u16; 4], idx: usize) -> bool {
            current[idx] > old[idx]
        }

        let mut best = None;
        search(3, amount as u32, &tubes, &mut [0; 4], &mut best);
        best.map(Self::descending_vector)
    }

    fn descending_vector(counts: [u16; 4]) -> Vec<u16> {
        let mut result = Vec::new();
        for i in (0..4).rev() {
            for _ in 0..counts[i] {
                result.push(DENOMS[i]);
            }
        }
        result
    }

    pub fn vend(&mut self) -> Result<Receipt, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let Some(id) = self.selected.clone() else {
            return Err(Error::NoSelection);
        };
        let idx = self.index[&id];
        let product = &self.products[idx];
        if product.stock == 0 {
            return Err(Error::SoldOut { id });
        }
        let price = product.price;
        if self.credit < price {
            return Err(Error::Underpaid { price, credit: self.credit });
        }
        let amount = self.credit - price;
        let change = Self::make_change(amount, self.tubes)
            .ok_or(Error::NoChange { amount })?;
        let mut proposed = self.tubes;
        for coin in &change {
            let i = DENOMS.iter().position(|&d| d == *coin).unwrap();
            proposed[i] -= 1;
        }
        for coin in &self.escrow {
            let i = DENOMS.iter().position(|&d| d == *coin).unwrap();
            proposed[i] = proposed[i].saturating_add(1);
            if proposed[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let total_change: u64 = change.iter().map(|&c| c as u64).sum();
        self.tubes = proposed;
        self.products[idx].stock -= 1;
        self.credit = 0;
        self.escrow.clear();
        self.selected = None;
        self.sales = self.sales.saturating_add(price as u64);
        self.change = self.change.saturating_add(total_change);
        self.log(EventKind::Vended {
            id: id.clone(),
            price,
            change: change.clone(),
        });
        Ok(Receipt { id, price, change })
    }

    pub fn refund(&mut self) -> Result<Vec<u16>, Error> {
        if self.maintenance {
            return Err(Error::MaintenanceActive);
        }
        let coins: Vec<u16> = self.escrow.iter().rev().copied().collect();
        let refunded = self.credit as u64;
        self.credit = 0;
        self.escrow.clear();
        self.selected = None;
        self.refunded = self.refunded.saturating_add(refunded);
        self.log(EventKind::Refunded { coins: coins.clone() });
        Ok(coins)
    }

    pub fn products(&self) -> Vec<Product> {
        let mut products = self.products.clone();
        products.sort_by(|a, b| a.id.as_bytes().cmp(b.id.as_bytes()));
        products
    }

    pub fn set_price(&mut self, id: &str, price: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&idx) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        if !(1..=100).contains(&price) {
            return Err(Error::BadPrice { id: id.to_owned() });
        }
        let old = self.products[idx].price;
        self.products[idx].price = price;
        self.log(EventKind::PriceChanged { id: id.to_owned(), old, new: price });
        Ok(())
    }

    pub fn restock(&mut self, id: &str, count: u16) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        let Some(&idx) = self.index.get(id) else {
            return Err(Error::UnknownProduct { id: id.to_owned() });
        };
        if count == 0 {
            return Err(Error::ZeroRestock);
        }
        let sum = self.products[idx].stock as u32 + count as u32;
        if sum > 100 {
            return Err(Error::StockLimit { id: id.to_owned() });
        }
        self.products[idx].stock = sum as u16;
        let stock = self.products[idx].stock;
        self.log(EventKind::Restocked { id: id.to_owned(), added: count, stock });
        Ok(())
    }

    pub fn events(&self) -> &[Event] {
        &self.events
    }

    pub fn set_maintenance(&mut self, enabled: bool) -> Result<(), Error> {
        if enabled == self.maintenance {
            return Err(Error::AlreadyMode);
        }
        if self.credit != 0 {
            return Err(Error::Busy);
        }
        self.maintenance = enabled;
        self.selected = None;
        self.log(EventKind::Maintenance { enabled });
        Ok(())
    }

    pub fn audit(&self) -> Audit {
        let tube_value: u64 = self.tubes.iter().enumerate()
            .map(|(i, &count)| count as u64 * DENOMS[i] as u64)
            .sum();
        Audit {
            inserted: self.inserted,
            refunded: self.refunded,
            sales: self.sales,
            change: self.change,
            adjustment: self.adjustment,
            cash: tube_value + self.credit as u64,
        }
    }

    pub fn reconcile(&mut self, expected: [u16; 4], actual: [u16; 4]) -> Result<(), Error> {
        if !self.maintenance {
            return Err(Error::MaintenanceRequired);
        }
        for i in 0..4 {
            if expected[i] != self.tubes[i] {
                return Err(Error::ReconcileMismatch {
                    coin: DENOMS[i],
                    expected: expected[i],
                    found: self.tubes[i],
                });
            }
        }
        for i in 0..4 {
            if actual[i] > 100 {
                return Err(Error::TubeLimit { coin: DENOMS[i] });
            }
        }
        let value = |counts: [u16; 4]| -> u64 {
            counts.iter().enumerate()
                .map(|(i, &count)| count as u64 * DENOMS[i] as u64)
                .sum()
        };
        let before_value = value(self.tubes);
        let after_value = value(actual);
        if before_value.abs_diff(after_value) > 20 {
            return Err(Error::ValueDrift { before: before_value, after: after_value });
        }
        let delta = after_value as i128 - before_value as i128;
        let signed_delta = delta.clamp(i64::MIN as i128, i64::MAX as i128) as i64;
        self.adjustment = self.adjustment.saturating_add(signed_delta);
        let before = self.tubes;
        self.tubes = actual;
        self.log(EventKind::Reconciled { before, after: actual });
        Ok(())
    }
}
```
