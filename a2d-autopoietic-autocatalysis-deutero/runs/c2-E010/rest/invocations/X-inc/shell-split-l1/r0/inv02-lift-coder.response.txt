#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.find('\0') {
        return Err(SplitError::Nul { pos });
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match ch {
            ' ' | '\t' | '\r' | '\n' => {
                if active {
                    words.push(std::mem::take(&mut word));
                    active = false;
                }
            }
            '\'' => {
                active = true;
                let opening = pos;
                let mut closed = false;

                while let Some((_, inner)) = chars.next() {
                    if inner == '\'' {
                        closed = true;
                        break;
                    }
                    word.push(inner);
                }

                if !closed {
                    return Err(SplitError::UnterminatedSingle { pos: opening });
                }
            }
            '"' => {
                active = true;
                let opening = pos;
                let mut closed = false;

                while let Some((inner_pos, inner)) = chars.next() {
                    match inner {
                        '"' => {
                            closed = true;
                            break;
                        }
                        '\\' => match chars.next() {
                            None => return Err(SplitError::TrailingEscape { pos: inner_pos }),
                            Some((_, escaped @ ('"' | '\\' | '$'))) => word.push(escaped),
                            Some((_, '\n')) => {}
                            Some((_, other)) => {
                                word.push('\\');
                                word.push(other);
                            }
                        },
                        other => word.push(other),
                    }
                }

                if !closed {
                    return Err(SplitError::UnterminatedDouble { pos: opening });
                }
            }
            '\\' => match chars.next() {
                None => return Err(SplitError::TrailingEscape { pos }),
                Some((_, '\n')) => {}
                Some((_, escaped)) => {
                    active = true;
                    word.push(escaped);
                }
            },
            other => {
                active = true;
                word.push(other);
            }
        }
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
