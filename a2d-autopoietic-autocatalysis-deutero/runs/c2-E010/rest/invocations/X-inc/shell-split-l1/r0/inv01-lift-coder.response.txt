#[derive(Debug, Clone, PartialEq, Eq)]
pub enum SplitError {
    Nul { pos: usize },
    TrailingEscape { pos: usize },
    UnterminatedSingle { pos: usize },
    UnterminatedDouble { pos: usize },
}

pub fn split(input: &str) -> Result<Vec<String>, SplitError> {
    if let Some(pos) = input.as_bytes().iter().position(|&byte| byte == 0) {
        return Err(SplitError::Nul { pos });
    }

    #[derive(Clone, Copy)]
    enum Quote {
        Single(usize),
        Double(usize),
    }

    let mut words = Vec::new();
    let mut word = String::new();
    let mut active = false;
    let mut quote = None;
    let mut chars = input.char_indices().peekable();

    while let Some((pos, ch)) = chars.next() {
        match quote {
            Some(Quote::Single(_)) => {
                if ch == '\'' {
                    quote = None;
                } else {
                    word.push(ch);
                }
            }
            Some(Quote::Double(_)) => match ch {
                '"' => quote = None,
                '\\' => match chars.peek().copied() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, next)) if next == '"' || next == '\\' || next == '$' => {
                        chars.next();
                        word.push(next);
                    }
                    Some((_, '\n')) => {
                        chars.next();
                    }
                    Some(_) => word.push('\\'),
                },
                _ => word.push(ch),
            },
            None => match ch {
                ' ' | '\t' | '\r' | '\n' => {
                    if active {
                        words.push(std::mem::take(&mut word));
                        active = false;
                    }
                }
                '\'' => {
                    active = true;
                    quote = Some(Quote::Single(pos));
                }
                '"' => {
                    active = true;
                    quote = Some(Quote::Double(pos));
                }
                '\\' => match chars.peek().copied() {
                    None => return Err(SplitError::TrailingEscape { pos }),
                    Some((_, '\n')) => {
                        chars.next();
                    }
                    Some((_, next)) => {
                        chars.next();
                        active = true;
                        word.push(next);
                    }
                },
                _ => {
                    active = true;
                    word.push(ch);
                }
            },
        }
    }

    match quote {
        Some(Quote::Single(pos)) => return Err(SplitError::UnterminatedSingle { pos }),
        Some(Quote::Double(pos)) => return Err(SplitError::UnterminatedDouble { pos }),
        None => {}
    }

    if active {
        words.push(word);
    }

    Ok(words)
}
