```rust
use std::collections::BTreeMap;

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Kind {
    Dir,
    File,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPath(usize),
    Missing(String),
    NotDir(String),
    IsDir(String),
    Exists(String),
}

#[derive(Debug, Clone)]
enum Entry {
    Dir(BTreeMap<String, Entry>),
    File(Vec<u8>),
}

pub struct Fs {
    root: Entry,
}

impl Fs {
    pub fn new() -> Self {
        Self {
            root: Entry::Dir(BTreeMap::new()),
        }
    }

    fn components(path: &str) -> Result<Vec<&str>, Error> {
        if let Some(i) = path.as_bytes().iter().position(|&b| b == 0) {
            return Err(Error::InvalidPath(i));
        }
        if path.is_empty() || !path.starts_with('/') {
            return Err(Error::InvalidPath(0));
        }
        if path == "/" {
            return Ok(Vec::new());
        }

        let mut parts = Vec::new();
        let bytes = path.as_bytes();
        let mut start = 1;

        for i in 1..=bytes.len() {
            if i == bytes.len() || bytes[i] == b'/' {
                let component = &path[start..i];
                if component.is_empty() || component == "." || component == ".." {
                    let offset = if component.is_empty() {
                        if i == bytes.len() {
                            bytes.len() - 1
                        } else {
                            i
                        }
                    } else {
                        start
                    };
                    return Err(Error::InvalidPath(offset));
                }
                parts.push(component);
                start = i + 1;
            }
        }
        Ok(parts)
    }

    fn prefix(parts: &[&str]) -> String {
        if parts.is_empty() {
            return "/".to_owned();
        }
        let mut result = String::new();
        for part in parts {
            result.push('/');
            result.push_str(part);
        }
        result
    }

    fn lookup<'a>(&'a self, parts: &[&str]) -> Result<&'a Entry, Error> {
        let mut current = &self.root;
        for (i, part) in parts.iter().enumerate() {
            let Entry::Dir(children) = current else {
                return Err(Error::NotDir(Self::prefix(&parts[..i])));
            };
            current = children
                .get(*part)
                .ok_or_else(|| Error::Missing(Self::prefix(&parts[..=i])))?;
        }
        Ok(current)
    }

    fn lookup_parent_mut<'a>(
        &'a mut self,
        parts: &[&str],
    ) -> Result<&'a mut BTreeMap<String, Entry>, Error> {
        let (name, parents) = parts.split_last().expect("non-root path");
        let mut current = &mut self.root;

        for (i, part) in parents.iter().enumerate() {
            let Entry::Dir(children) = current else {
                return Err(Error::NotDir(Self::prefix(&parents[..i])));
            };
            current = children
                .get_mut(*part)
                .ok_or_else(|| Error::Missing(Self::prefix(&parents[..=i])))?;
        }

        match current {
            Entry::Dir(children) => Ok(children),
            Entry::File(_) => Err(Error::NotDir(Self::prefix(parents))),
        }
        .map(|children| {
            let _ = name;
            children
        })
    }

    pub fn mkdir(&mut self, path: &str) -> Result<(), Error> {
        let parts = Self::components(path)?;
        if parts.is_empty() {
            return Err(Error::Exists("/".to_owned()));
        }

        let canonical = Self::prefix(&parts);
        let parent = self.lookup_parent_mut(&parts)?;
        if parent.contains_key(*parts.last().unwrap()) {
            return Err(Error::Exists(canonical));
        }
        parent.insert(
            parts.last().unwrap().to_string(),
            Entry::Dir(BTreeMap::new()),
        );
        Ok(())
    }

    pub fn write(&mut self, path: &str, data: &[u8]) -> Result<(), Error> {
        let parts = Self::components(path)?;
        if parts.is_empty() {
            return Err(Error::IsDir("/".to_owned()));
        }

        let canonical = Self::prefix(&parts);
        let parent = self.lookup_parent_mut(&parts)?;
        match parent.get_mut(*parts.last().unwrap()) {
            Some(Entry::Dir(_)) => Err(Error::IsDir(canonical)),
            Some(Entry::File(contents)) => {
                contents.clear();
                contents.extend_from_slice(data);
                Ok(())
            }
            None => {
                parent.insert(
                    parts.last().unwrap().to_string(),
                    Entry::File(data.to_vec()),
                );
                Ok(())
            }
        }
    }

    pub fn read(&self, path: &str) -> Result<Vec<u8>, Error> {
        let parts = Self::components(path)?;
        let canonical = Self::prefix(&parts);
        match self.lookup(&parts)? {
            Entry::File(contents) => Ok(contents.clone()),
            Entry::Dir(_) => Err(Error::IsDir(canonical)),
        }
    }

    pub fn ls(&self, path: &str) -> Result<Vec<(String, Kind)>, Error> {
        let parts = Self::components(path)?;
        let canonical = Self::prefix(&parts);
        match self.lookup(&parts)? {
            Entry::File(_) => Err(Error::NotDir(canonical)),
            Entry::Dir(children) => Ok(children
                .iter()
                .map(|(name, entry)| {
                    let kind = match entry {
                        Entry::Dir(_) => Kind::Dir,
                        Entry::File(_) => Kind::File,
                    };
                    (name.clone(), kind)
                })
                .collect()),
        }
    }
}

impl Default for Fs {
    fn default() -> Self {
        Self::new()
    }
}
```
