#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Expression<'a> {
    operator: Option<char>,
    names: Vec<&'a str>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);

        let mut fields = Vec::new();
        for name in expression.names {
            if let Some(value) = vars.iter().rev().find(|(key, _)| *key == name) {
                fields.push(encode(value.1, matches!(expression.operator, Some('+' | '#'))));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some('#') {
                output.push('#');
            }
            let separator = ",";
            output.push_str(&fields.join(separator));
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

#[derive(Debug)]
struct ParsedExpression<'a> {
    start: usize,
    end: usize,
    operator: Option<char>,
    names: Vec<&'a str>,
}

fn parse(template: &str) -> Result<Vec<ParsedExpression<'_>>, Error> {
    let mut expressions = Vec::new();
    let mut offset = 0;

    while offset < template.len() {
        let ch = template[offset..].chars().next().unwrap();
        match ch {
            '}' => return Err(Error::UnexpectedClose { pos: offset }),
            '{' => {
                let start = offset;
                let after_open = offset + 1;
                let Some(relative_close) = template[after_open..].find('}') else {
                    return Err(Error::Unclosed { pos: start });
                };
                let end = after_open + relative_close;
                let content = &template[after_open..end];

                if let Some(relative_open) = content.find('{') {
                    return Err(Error::NestedOpen {
                        pos: after_open + relative_open,
                    });
                }

                let (operator, name_content, names_start) = match content.chars().next() {
                    Some(op @ ('+' | '#' | '.' | '/' | ';' | '?' | '&')) => {
                        if op != '+' && op != '#' {
                            return Err(Error::UnsupportedOperator {
                                pos: after_open,
                            });
                        }
                        (Some(op), &content[op.len_utf8()..], after_open + op.len_utf8())
                    }
                    _ => (None, content, after_open),
                };

                if name_content.is_empty() {
                    return Err(Error::EmptyExpression { pos: end });
                }

                let mut names = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut spec_offset = 0;

                for specifier in name_content.split(',') {
                    let spec_start = names_start + spec_offset;
                    if specifier.is_empty() {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    if let Some(invalid) = invalid_name_offset(specifier) {
                        return Err(Error::InvalidName {
                            pos: spec_start + invalid,
                        });
                    }

                    if seen.contains(&specifier) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }

                    seen.push(specifier);
                    names.push(specifier);
                    spec_offset += specifier.len() + 1;
                }

                expressions.push(ParsedExpression {
                    start,
                    end,
                    operator,
                    names,
                });
                offset = end + 1;
            }
            _ => offset += ch.len_utf8(),
        }
    }

    Ok(expressions)
}

fn invalid_name_offset(name: &str) -> Option<usize> {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !is_name_start(bytes[0]) {
        return Some(0);
    }

    bytes
        .iter()
        .enumerate()
        .skip(1)
        .find_map(|(index, byte)| (!is_name_continue(*byte)).then_some(index))
}

fn is_name_start(byte: u8) -> bool {
    byte.is_ascii_alphabetic() || byte == b'_'
}

fn is_name_continue(byte: u8) -> bool {
    is_name_start(byte) || byte.is_ascii_digit()
}

fn encode(value: &str, allow_reserved: bool) -> String {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2])) {
                result.push('%');
                result.push(HEX[high] as char);
                result.push(HEX[low] as char);
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (allow_reserved && is_reserved(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(HEX[(byte >> 4) as usize] as char);
            result.push(HEX[(byte & 0x0f) as usize] as char);
        }
        index += 1;
    }

    result
}

fn hex_value(byte: u8) -> Option<usize> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as usize),
        b'a'..=b'f' => Some((byte - b'a' + 10) as usize),
        b'A'..=b'F' => Some((byte - b'A' + 10) as usize),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}
