#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Clone)]
struct Variable<'a> {
    name: &'a str,
    operator: Option<u8>,
}

struct Expression<'a> {
    variables: Vec<Variable<'a>>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = validate_template(template)?;
    let mut output = String::new();
    let mut cursor = 0;

    for expression in expressions {
        output.push_str(&template[cursor..expression.start]);
        let fields: Vec<&str> = expression
            .variables
            .iter()
            .filter_map(|variable| lookup(vars, variable.name))
            .collect();

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                output.push('#');
            }
            let reserved = matches!(expression.operator, Some(b'+') | Some(b'#'));
            for (index, value) in fields.iter().enumerate() {
                if index > 0 {
                    output.push(',');
                }
                encode_value(value, reserved, &mut output);
            }
        }

        cursor = expression.end + 1;
    }

    output.push_str(&template[cursor..]);
    Ok(output)
}

struct ParsedExpression<'a> {
    start: usize,
    end: usize,
    operator: Option<u8>,
    variables: Vec<Variable<'a>>,
}

fn validate_template(template: &str) -> Result<Vec<ParsedExpression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut pos = 0;

    while pos < bytes.len() {
        match bytes[pos] {
            b'}' => return Err(Error::UnexpectedClose { pos }),
            b'{' => {
                let start = pos;
                let close = match bytes[pos + 1..].iter().position(|&byte| byte == b'}') {
                    Some(offset) => pos + 1 + offset,
                    None => return Err(Error::Unclosed { pos: start }),
                };

                if let Some(offset) = bytes[pos + 1..close]
                    .iter()
                    .position(|&byte| byte == b'{')
                {
                    return Err(Error::NestedOpen { pos: pos + 1 + offset });
                }

                let mut content_start = start + 1;
                let mut operator = None;
                if content_start < close
                    && matches!(
                        bytes[content_start],
                        b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&'
                    )
                {
                    let op = bytes[content_start];
                    if op != b'+' && op != b'#' {
                        return Err(Error::UnsupportedOperator { pos: content_start });
                    }
                    operator = Some(op);
                    content_start += 1;
                }

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut variables = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&byte| byte == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let name_bytes = &bytes[spec_start..spec_end];
                    for (offset, &byte) in name_bytes.iter().enumerate() {
                        let valid = if offset == 0 {
                            byte.is_ascii_alphabetic() || byte == b'_'
                        } else {
                            byte.is_ascii_alphanumeric() || byte == b'_'
                        };
                        if !valid {
                            return Err(Error::InvalidName {
                                pos: spec_start + offset,
                            });
                        }
                    }

                    let name = &template[spec_start..spec_end];
                    if seen.contains(&name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name);
                    variables.push(Variable { name, operator });

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(ParsedExpression {
                    start,
                    end: close,
                    operator,
                    variables,
                });
                pos = close + 1;
            }
            _ => pos += 1,
        }
    }

    Ok(expressions)
}

fn lookup<'a>(vars: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
    vars.iter()
        .rev()
        .find(|(binding_name, _)| *binding_name == name)
        .map(|(_, value)| *value)
}

fn encode_value(value: &str, reserved: bool, output: &mut String) {
    const HEX: &[u8; 16] = b"0123456789ABCDEF";
    let bytes = value.as_bytes();
    let mut pos = 0;

    while pos < bytes.len() {
        let byte = bytes[pos];

        if byte == b'%' && pos + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[pos + 1]), hex_value(bytes[pos + 2]))
            {
                output.push('%');
                output.push(HEX[high] as char);
                output.push(HEX[low] as char);
                pos += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            output.push(byte as char);
        } else {
            output.push('%');
            output.push(HEX[(byte >> 4) as usize] as char);
            output.push(HEX[(byte & 0x0f) as usize] as char);
        }
        pos += 1;
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&' | b'\''
            | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<usize> {
    match byte {
        b'0'..=b'9' => Some((byte - b'0') as usize),
        b'a'..=b'f' => Some((byte - b'a' + 10) as usize),
        b'A'..=b'F' => Some((byte - b'A' + 10) as usize),
        _ => None,
    }
}
