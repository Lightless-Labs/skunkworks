#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    UnexpectedClose { pos: usize },
    Unclosed { pos: usize },
    NestedOpen { pos: usize },
    EmptyExpression { pos: usize },
    UnsupportedOperator { pos: usize },
    EmptyVariable { pos: usize },
    InvalidName { pos: usize },
    DuplicateVariable { pos: usize },
}

#[derive(Debug)]
struct Variable<'a> {
    name: &'a str,
    start: usize,
}

#[derive(Debug)]
struct Expression<'a> {
    operator: Option<u8>,
    variables: Vec<Variable<'a>>,
}

pub fn expand(template: &str, vars: &[(&str, &str)]) -> Result<String, Error> {
    let expressions = parse(template)?;
    let mut output = String::new();

    for expression in expressions {
        let reserved = matches!(expression.operator, Some(b'+' | b'#'));
        let separator = ',';
        let mut fields = Vec::new();

        for variable in expression.variables {
            if let Some(value) = vars
                .iter()
                .rev()
                .find(|(name, _)| *name == variable.name)
                .map(|(_, value)| *value)
            {
                fields.push(encode(value, reserved));
            }
        }

        if !fields.is_empty() {
            if expression.operator == Some(b'#') {
                output.push('#');
            }
            for (index, field) in fields.into_iter().enumerate() {
                if index > 0 {
                    output.push(separator);
                }
                output.push_str(&field);
            }
        }
    }

    // Literal spans are emitted during parsing as special output segments.
    // Reparse the validated template to interleave them with expanded expressions.
    let mut result = String::new();
    let mut cursor = 0;
    let mut expression_index = 0;

    while cursor < template.len() {
        let rest = &template[cursor..];
        if let Some(relative_open) = rest.find('{') {
            let open = cursor + relative_open;
            result.push_str(&template[cursor..open]);
            let close = template[open..]
                .find('}')
                .map(|offset| open + offset)
                .expect("validated expression has a closing brace");
            let expanded = expand_expression(&expressions[expression_index], vars);
            result.push_str(&expanded);
            expression_index += 1;
            cursor = close + 1;
        } else {
            result.push_str(&template[cursor..]);
            break;
        }
    }

    let _ = output;
    Ok(result)
}

fn parse(template: &str) -> Result<Vec<Expression<'_>>, Error> {
    let bytes = template.as_bytes();
    let mut expressions = Vec::new();
    let mut cursor = 0;

    while cursor < bytes.len() {
        match bytes[cursor] {
            b'}' => return Err(Error::UnexpectedClose { pos: cursor }),
            b'{' => {
                let open = cursor;
                let Some(relative_close) = bytes[cursor + 1..].iter().position(|&b| b == b'}') else {
                    return Err(Error::Unclosed { pos: open });
                };
                let close = cursor + 1 + relative_close;

                if let Some(relative_nested) = bytes[open + 1..close].iter().position(|&b| b == b'{') {
                    return Err(Error::NestedOpen {
                        pos: open + 1 + relative_nested,
                    });
                }

                let mut content_start = open + 1;
                let operator = match bytes.get(content_start).copied() {
                    Some(b'+' | b'#' | b'.' | b'/' | b';' | b'?' | b'&') => {
                        let op = bytes[content_start];
                        content_start += 1;
                        if !matches!(op, b'+' | b'#') {
                            return Err(Error::UnsupportedOperator { pos: open + 1 });
                        }
                        Some(op)
                    }
                    _ => None,
                };

                if content_start == close {
                    return Err(Error::EmptyExpression { pos: close });
                }

                let mut variables = Vec::new();
                let mut seen: Vec<&str> = Vec::new();
                let mut spec_start = content_start;

                loop {
                    let spec_end = bytes[spec_start..close]
                        .iter()
                        .position(|&b| b == b',')
                        .map(|offset| spec_start + offset)
                        .unwrap_or(close);

                    if spec_start == spec_end {
                        return Err(Error::EmptyVariable { pos: spec_start });
                    }

                    let name_bytes = &bytes[spec_start..spec_end];
                    for (index, &byte) in name_bytes.iter().enumerate() {
                        let valid = if index == 0 {
                            byte.is_ascii_alphabetic() || byte == b'_'
                        } else {
                            byte.is_ascii_alphanumeric() || byte == b'_'
                        };
                        if !valid {
                            return Err(Error::InvalidName {
                                pos: spec_start + index,
                            });
                        }
                    }

                    let name = &template[spec_start..spec_end];
                    if seen.iter().any(|existing| *existing == name) {
                        return Err(Error::DuplicateVariable { pos: spec_start });
                    }
                    seen.push(name);
                    variables.push(Variable {
                        name,
                        start: spec_start,
                    });

                    if spec_end == close {
                        break;
                    }
                    spec_start = spec_end + 1;
                }

                expressions.push(Expression { operator, variables });
                cursor = close + 1;
            }
            _ => cursor += 1,
        }
    }

    Ok(expressions)
}

fn expand_expression(expression: &Expression<'_>, vars: &[(&str, &str)]) -> String {
    let reserved = matches!(expression.operator, Some(b'+' | b'#'));
    let mut fields = Vec::new();

    for variable in &expression.variables {
        let _ = variable.start;
        if let Some(value) = vars
            .iter()
            .rev()
            .find(|(name, _)| *name == variable.name)
            .map(|(_, value)| *value)
        {
            fields.push(encode(value, reserved));
        }
    }

    if fields.is_empty() {
        return String::new();
    }

    let mut result = String::new();
    if expression.operator == Some(b'#') {
        result.push('#');
    }
    for (index, field) in fields.into_iter().enumerate() {
        if index > 0 {
            result.push(',');
        }
        result.push_str(&field);
    }
    result
}

fn encode(value: &str, reserved: bool) -> String {
    let bytes = value.as_bytes();
    let mut result = String::new();
    let mut index = 0;

    while index < bytes.len() {
        let byte = bytes[index];

        if byte == b'%' && index + 2 < bytes.len() {
            if let (Some(high), Some(low)) = (hex_value(bytes[index + 1]), hex_value(bytes[index + 2])) {
                result.push('%');
                result.push(hex_digit(high));
                result.push(hex_digit(low));
                index += 3;
                continue;
            }
        }

        if is_unreserved(byte) || (reserved && is_reserved(byte)) {
            result.push(byte as char);
        } else {
            result.push('%');
            result.push(hex_digit(byte >> 4));
            result.push(hex_digit(byte & 0x0f));
        }
        index += 1;
    }

    result
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'.' | b'_' | b'~')
}

fn is_reserved(byte: u8) -> bool {
    matches!(
        byte,
        b':' | b'/' | b'?' | b'#' | b'[' | b']' | b'@' | b'!' | b'$' | b'&'
            | b'\'' | b'(' | b')' | b'*' | b'+' | b',' | b';' | b'='
    )
}

fn hex_value(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'a'..=b'f' => Some(byte - b'a' + 10),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn hex_digit(value: u8) -> char {
    match value {
        0..=9 => (b'0' + value) as char,
        _ => (b'A' + value - 10) as char,
    }
}
