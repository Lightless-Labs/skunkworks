#[derive(Debug, Clone, PartialEq, Eq)]
pub struct PatchOptions {
    pub max_offset: usize,
    pub fuzz: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ApplyError {
    InvalidOptions,
    BadHeader { line: usize },
    BadHunkHeader { line: usize },
    BadBody { line: usize },
    CountMismatch { line: usize },
    Misordered { line: usize },
    SourceFormat { byte: usize },
    SourceMissingLf,
    TooLarge,
    NoMatch { hunk: usize },
    Ambiguous { hunk: usize, first: usize, second: usize },
}

#[derive(Clone)]
struct Row {
    kind: u8,
    payload: String,
    line: usize,
}

#[derive(Clone)]
struct Hunk {
    old_start: usize,
    old_count: usize,
    new_start: usize,
    new_count: usize,
    rows: Vec<Row>,
    header_line: usize,
}

fn parse_decimal(bytes: &[u8]) -> Option<usize> {
    if bytes.is_empty() || (bytes.len() > 1 && bytes[0] == b'0') {
        return None;
    }
    let mut value = 0usize;
    for &b in bytes {
        if !b.is_ascii_digit() {
            return None;
        }
        value = value.checked_mul(10)?.checked_add((b - b'0') as usize)?;
        if value > 128 {
            return None;
        }
    }
    Some(value)
}

fn parse_hunk_header(line: &str) -> Option<(usize, usize, usize, usize)> {
    let b = line.as_bytes();
    if b.len() < 13 || !line.starts_with("@@ -") || !line.ends_with(" @@") {
        return None;
    }
    let inner = &line[4..line.len() - 3];
    let (old, new) = inner.split_once(" +")?;
    let (a, bcount) = old.split_once(',')?;
    let (c, d) = new.split_once(',')?;
    if a.contains(',') || bcount.contains(',') || c.contains(',') || d.contains(',') {
        return None;
    }
    Some((
        parse_decimal(a.as_bytes())?,
        parse_decimal(bcount.as_bytes())?,
        parse_decimal(c.as_bytes())?,
        parse_decimal(d.as_bytes())?,
    ))
}

fn physical_lines(text: &str) -> Vec<&str> {
    if text.is_empty() {
        return Vec::new();
    }
    text[..text.len() - 1].split('\n').collect()
}

fn parse_patch(patch: &str) -> Result<Vec<Hunk>, ApplyError> {
    if patch.is_empty() {
        return Ok(Vec::new());
    }
    if !patch.ends_with('\n') {
        return Err(ApplyError::BadHeader {
            line: patch.bytes().filter(|&b| b == b'\n').count() + 1,
        });
    }

    let lines = physical_lines(patch);
    for (i, expected) in ["--- before", "+++ after"].iter().enumerate() {
        if lines.get(i) != Some(expected) {
            return Err(ApplyError::BadHeader { line: i + 1 });
        }
    }
    if lines.len() <= 2 || !lines[2].starts_with("@@") {
        return Err(ApplyError::BadHunkHeader { line: 3 });
    }

    let mut hunks = Vec::new();
    let mut index = 2usize;
    let mut previous_old_end: i128 = 0;
    let mut cumulative_delta: i128 = 0;

    while index < lines.len() {
        let header_line = index + 1;
        let Some((a, old_count, c, new_count)) = parse_hunk_header(lines[index]) else {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        };
        if (old_count > 0 && a == 0) || (new_count > 0 && c == 0) {
            return Err(ApplyError::BadHunkHeader { line: header_line });
        }
        let old_start = if old_count > 0 { a - 1 } else { a };
        let new_start = if new_count > 0 { c - 1 } else { c };

        index += 1;
        let mut rows = Vec::new();
        let mut old_seen = 0usize;
        let mut new_seen = 0usize;
        while index < lines.len() && !lines[index].starts_with("@@") {
            let row_line = index + 1;
            let bytes = lines[index].as_bytes();
            if bytes.is_empty() || !matches!(bytes[0], b' ' | b'-' | b'+') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let payload = &lines[index][1..];
            if payload.contains('\r') {
                return Err(ApplyError::BadBody { line: row_line });
            }
            let kind = bytes[0];
            if kind != b'+' {
                old_seen += 1;
            }
            if kind != b'-' {
                new_seen += 1;
            }
            rows.push(Row {
                kind,
                payload: payload.to_string(),
                line: row_line,
            });
            index += 1;
        }

        if old_seen != old_count || new_seen != new_count {
            return Err(ApplyError::CountMismatch { line: header_line });
        }
        if !rows.iter().any(|r| r.kind == b'+' || r.kind == b'-') {
            return Err(ApplyError::BadBody { line: header_line });
        }

        let old_start_i = old_start as i128;
        if old_start_i < previous_old_end
            || new_start as i128 != old_start_i + cumulative_delta
        {
            return Err(ApplyError::Misordered { line: header_line });
        }

        previous_old_end = old_start_i + old_count as i128;
        cumulative_delta += new_count as i128 - old_count as i128;

        if hunks.len() == 16 {
            return Err(ApplyError::TooLarge);
        }
        hunks.push(Hunk {
            old_start,
            old_count,
            new_start,
            new_count,
            rows,
            header_line,
        });
    }
    Ok(hunks)
}

fn source_lines(source: &str) -> Result<Vec<String>, ApplyError> {
    if let Some(byte) = source.as_bytes().iter().position(|&b| b == b'\r') {
        return Err(ApplyError::SourceFormat { byte });
    }
    if !source.is_empty() && !source.ends_with('\n') {
        return Err(ApplyError::SourceMissingLf);
    }
    let lines: Vec<String> = physical_lines(source).into_iter().map(str::to_string).collect();
    if lines.len() > 128 {
        return Err(ApplyError::TooLarge);
    }
    Ok(lines)
}

fn abs_distance(a: i128, b: i128) -> i128 {
    (a - b).abs()
}

pub fn apply_unified(
    source: &str,
    patch: &str,
    options: PatchOptions,
) -> Result<String, ApplyError> {
    if options.max_offset > 16 || options.fuzz > 2 {
        return Err(ApplyError::InvalidOptions);
    }

    let hunks = parse_patch(patch)?;
    let mut working = source_lines(source)?;
    let mut net: i128 = 0;
    let mut drift: i128 = 0;

    for (hunk_index, hunk) in hunks.iter().enumerate() {
        let mut lead = 0usize;
        while lead < hunk.rows.len() && hunk.rows[lead].kind == b' ' {
            lead += 1;
        }
        let mut trail = 0usize;
        while trail < hunk.rows.len() - lead
            && hunk.rows[hunk.rows.len() - 1 - trail].kind == b' '
        {
            trail += 1;
        }

        let expected = hunk.old_start as i128 + net + drift;
        let mut best_rank: Option<(usize, i128)> = None;
        let mut best_starts = Vec::<usize>::new();
        let mut chosen: Option<(usize, usize, usize)> = None;

        for left_drop in 0..=options.fuzz.min(lead) {
            for right_drop in 0..=options.fuzz.min(trail) {
                if left_drop + right_drop > hunk.old_count {
                    continue;
                }
                let mut old_payloads = Vec::new();
                let mut new_payloads = Vec::new();
                let end = hunk.rows.len().saturating_sub(right_drop);
                for row in &hunk.rows[left_drop..end] {
                    if row.kind != b'+' {
                        old_payloads.push(row.payload.as_str());
                    }
                    if row.kind != b'-' {
                        new_payloads.push(row.payload.as_str());
                    }
                }

                for start in 0..=working.len() {
                    if start + hunk.old_count > working.len() {
                        continue;
                    }
                    let distance = abs_distance(start as i128, expected);
                    if distance > options.max_offset as i128 {
                        continue;
                    }
                    let match_start = start + left_drop;
                    let match_end = start + hunk.old_count - right_drop;
                    if match_end - match_start != old_payloads.len()
                        || working[match_start..match_end]
                            .iter()
                            .map(String::as_str)
                            .ne(old_payloads.iter().copied())
                    {
                        continue;
                    }

                    let rank = (left_drop + right_drop, distance);
                    match best_rank {
                        None => {
                            best_rank = Some(rank);
                            best_starts.push(start);
                            chosen = Some((start, left_drop, right_drop));
                        }
                        Some(prev) if rank < prev => {
                            best_rank = Some(rank);
                            best_starts.clear();
                            best_starts.push(start);
                            chosen = Some((start, left_drop, right_drop));
                        }
                        Some(prev) if rank == prev => {
                            if !best_starts.contains(&start) {
                                best_starts.push(start);
                            }
                            if let Some((chosen_start, chosen_left, _)) = chosen {
                                if start == chosen_start && left_drop < chosen_left {
                                    chosen = Some((start, left_drop, right_drop));
                                }
                            } else {
                                chosen = Some((start, left_drop, right_drop));
                            }
                        }
                        _ => {}
                    }
                }
            }
        }

        let Some((start, left_drop, right_drop)) = chosen else {
            return Err(ApplyError::NoMatch {
                hunk: hunk_index + 1,
            });
        };
        if best_starts.len() > 1 {
            best_starts.sort_unstable();
            return Err(ApplyError::Ambiguous {
                hunk: hunk_index + 1,
                first: best_starts[0],
                second: best_starts[1],
            });
        }

        let body_end = hunk.rows.len() - right_drop;
        let mut replacement = Vec::new();
        for row in &hunk.rows[left_drop..body_end] {
            if row.kind != b'-' {
                replacement.push(row.payload.clone());
            }
        }
        let replace_start = start + left_drop;
        let replace_end = start + hunk.old_count - right_drop;
        working.splice(replace_start..replace_end, replacement);
        if working.len() > 128 {
            return Err(ApplyError::TooLarge);
        }

        let previous_net = net;
        net += hunk.new_count as i128 - hunk.old_count as i128;
        drift = start as i128 - (hunk.old_start as i128 + previous_net);
    }

    let mut output = String::new();
    for line in working {
        output.push_str(&line);
        output.push('\n');
    }
    Ok(output)
}
