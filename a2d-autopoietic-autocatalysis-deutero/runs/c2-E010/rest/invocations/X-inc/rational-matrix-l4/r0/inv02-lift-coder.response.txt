use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128 || n > i64::MAX as i128 || d > i64::MAX as i128 {
            return Err(RationalError::Overflow);
        }
        Ok(Self { num: n as i64, den: d as i64 })
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 { self.num }
    pub fn denominator(&self) -> i64 { self.den }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_i128(
            self.num as i128 * rhs.num as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 { return Err(RationalError::DivideByZero); }
        Self::from_i128(
            self.num as i128 * rhs.den as i128,
            self.den as i128 * rhs.num as i128,
        )
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        Self::from_i128(-(self.num as i128), self.den as i128)
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let b = text.as_bytes();
        let err = |at, kind| RationalError::Parse { at, kind };
        if b.is_empty() { return Err(err(0, ParseKind::Empty)); }

        let mut i = 0;
        if b[i] == b'-' { i += 1; }
        let ns = i;
        while i < b.len() && b[i].is_ascii_digit() { i += 1; }
        if i == ns {
            return Err(err(i, if i == b.len() { ParseKind::InvalidByte } else { ParseKind::InvalidByte }));
        }
        let ne = i;
        let mut ds = None;
        let mut de = None;
        if i < b.len() {
            if b[i] != b'/' { return Err(err(i, ParseKind::InvalidByte)); }
            i += 1;
            ds = Some(i);
            while i < b.len() && b[i].is_ascii_digit() { i += 1; }
            if i == ds.unwrap() {
                return Err(err(i, ParseKind::InvalidByte));
            }
            de = Some(i);
            if i < b.len() { return Err(err(i, ParseKind::InvalidByte)); }
        }

        let neg = b[0] == b'-';
        let digits = &text[ns..ne];
        if digits.len() >= 2 && digits.as_bytes()[0] == b'0' {
            return Err(err(ns + 1, ParseKind::NonCanonical));
        }
        if neg && digits == "0" { return Err(err(0, ParseKind::NonCanonical)); }

        let mut nmag: u128 = 0;
        for ch in digits.bytes() {
            nmag = nmag.saturating_mul(10).saturating_add((ch - b'0') as u128);
        }
        let nlimit = if neg { 1u128 << 63 } else { i64::MAX as u128 };
        if nmag > nlimit { return Err(err(0, ParseKind::OutOfRange)); }

        let mut dmag = 1u128;
        if let (Some(s), Some(e)) = (ds, de) {
            let dd = &text[s..e];
            if dd.len() >= 2 && dd.as_bytes()[0] == b'0' {
                return Err(err(s + 1, ParseKind::NonCanonical));
            }
            dmag = 0;
            for ch in dd.bytes() {
                dmag = dmag.saturating_mul(10).saturating_add((ch - b'0') as u128);
            }
            if dmag > i64::MAX as u128 { return Err(err(s, ParseKind::OutOfRange)); }
            if dd == "0" { return Err(err(s, ParseKind::ZeroDenominator)); }
            if dd == "1" { return Err(err(s - 1, ParseKind::NonCanonical)); }
            if nmag == 0 { return Err(err(s - 1, ParseKind::NonCanonical)); }
            if gcd_u128(nmag, dmag) > 1 { return Err(err(s - 1, ParseKind::NonCanonical)); }
        }
        let n = if neg {
            if nmag == (1u128 << 63) { i64::MIN } else { -(nmag as i64) }
        } else { nmag as i64 };
        Ok(Self { num: n, den: dmag as i64 })
    }
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 { write!(f, "{}", self.num) }
        else { write!(f, "{}/{}", self.num, self.den) }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize { self.pivots.len() }
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Self { rows, cols, data })
    }
    pub fn rows(&self) -> usize { self.rows }
    pub fn cols(&self) -> usize { self.cols }
    pub fn data(&self) -> &[Rational] { &self.data }
    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }
    pub fn transpose(&self) -> Self {
        let len = self.data.len();
        let mut data = Vec::with_capacity(len);
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self { rows: self.cols, cols: self.rows, data }
    }
    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols), right: (rhs.rows, rhs.cols)
            });
        }
        let count = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut out = Vec::with_capacity(count);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row:r,col:c,step:k,source })?;
                }
                out.push(acc);
            }
        }
        Ok(Self { rows:self.rows, cols:rhs.cols, data:out })
    }

    fn eliminate(&self, candidate_cols: usize) -> Result<Reduction, MatrixError> {
        let mut m = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        for col in 0..candidate_cols {
            if p == m.rows { break; }
            let mut best: Option<(i64, u128, usize)> = None;
            for r in p..m.rows {
                let v = m.data[r * m.cols + col];
                if v.num == 0 { continue; }
                let key = (v.den, v.num.unsigned_abs(), r);
                if best.map_or(true, |x| key < x) { best = Some(key); }
            }
            let q = match best { Some((_,_,r)) => r, None => continue };
            if q != p {
                for c in 0..m.cols { m.data.swap(p*m.cols+c, q*m.cols+c); }
                swaps.push((p,q));
            }
            let pivot = m.data[p*m.cols+col];
            for c in 0..m.cols {
                let idx = p*m.cols+c;
                m.data[idx] = m.data[idx].div(&pivot).map_err(|source|
                    MatrixError::Arithmetic { row:p,col:c,step:p,source })?;
            }
            for r in 0..m.rows {
                if r == p { continue; }
                let factor = m.data[r*m.cols+col];
                if factor.num == 0 { continue; }
                for c in 0..m.cols {
                    let prod = factor.mul(&m.data[p*m.cols+c]).map_err(|source|
                        MatrixError::Arithmetic { row:r,col:c,step:p,source })?;
                    let idx = r*m.cols+c;
                    m.data[idx] = m.data[idx].sub(&prod).map_err(|source|
                        MatrixError::Arithmetic { row:r,col:c,step:p,source })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction { matrix:m, pivots, swaps })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> { self.eliminate(self.cols) }
    pub fn rank(&self) -> Result<usize, MatrixError> { Ok(self.rref()?.rank()) }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows:self.rows, cols:self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let count = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = vec![Rational { num:0, den:1 }; count];
        for r in 0..n {
            for c in 0..n {
                data[r*width+c] = self.data[r*n+c];
                if r == c { data[r*width+n+c] = Rational { num:1, den:1 }; }
            }
        }
        let aug = Matrix { rows:n, cols:width, data };
        let red = aug.eliminate(n)?;
        if red.pivots.len() < n { return Err(MatrixError::Singular { rank:red.pivots.len() }); }
        let mut out = Vec::with_capacity(n*n);
        for r in 0..n { for c in 0..n { out.push(red.matrix.data[r*width+n+c]); } }
        Ok(Matrix { rows:n, cols:n, data:out })
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}

impl Matrix {
    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients:self.rows, rhs:rhs.rows });
        }
        let width = self.cols.checked_add(rhs.cols).ok_or(SolveError::AugmentedShapeOverflow)?;
        let count = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(count);
        for r in 0..self.rows {
            for c in 0..self.cols { data.push(self.data[r*self.cols+c]); }
            for c in 0..rhs.cols { data.push(rhs.data[r*rhs.cols+c]); }
        }
        let aug = Matrix { rows:self.rows, cols:width, data };
        let red = aug.eliminate(self.cols).map_err(SolveError::Matrix)?;
        for r in 0..self.rows {
            let coeff_zero = (0..self.cols).all(|c| red.matrix.data[r*width+c].num == 0);
            if coeff_zero {
                for c in 0..rhs.cols {
                    let v = red.matrix.data[r*width+self.cols+c];
                    if v.num != 0 {
                        return Ok(Solution::None {
                            row:r, rhs_col:c,
                            witness:red.matrix.data[r*width..(r+1)*width].to_vec()
                        });
                    }
                }
            }
        }
        let n = self.cols;
        let k = rhs.cols;
        let rank = red.pivots.len();
        let pk = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num:0, den:1 }; pk];
        for (j, &pivot_col) in red.pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pivot_col*k+c] = red.matrix.data[j*width+n+c];
            }
        }
        let particular = Matrix { rows:n, cols:k, data:particular_data };
        if rank == n {
            return Ok(Solution::Unique { values:particular, pivots:red.pivots });
        }
        let mut free = Vec::new();
        let mut piv = vec![false; n];
        for &c in &red.pivots { piv[c] = true; }
        for c in 0..n { if !piv[c] { free.push(c); } }
        let f = free.len();
        let bf = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num:0, den:1 }; bf];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col*f+j] = Rational { num:1, den:1 };
            for (i, &pivot_col) in red.pivots.iter().enumerate() {
                let v = red.matrix.data[i*width+free_col];
                basis_data[pivot_col*f+j] = v.neg().map_err(|source|
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row:pivot_col, col:j, step:rank, source
                    }))?;
            }
        }
        Ok(Solution::Infinite {
            particular, free,
            basis:Matrix { rows:n, cols:f, data:basis_data },
            pivots:red.pivots,
        })
    }
}
