use std::cmp::Ordering;
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n as i64, den: d as i64 })
}

fn rational_from_i128_pair(num: i128, den: i128) -> Result<Rational, RationalError> {
    rational_from_i128(num, den)
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        rational_from_i128_pair(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        rational_from_i128_pair(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128_pair(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128_pair(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational { num: -self.num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        use ParseKind::*;
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: Empty });
        }

        let mut slash = None;
        for (i, &c) in b.iter().enumerate() {
            if c == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse { at: i, kind: InvalidByte });
                }
                slash = Some(i);
                continue;
            }
            if c.is_ascii_digit() || (c == b'-' && i == 0) {
                continue;
            }
            return Err(RationalError::Parse { at: i, kind: InvalidByte });
        }

        let num_end = slash.unwrap_or(b.len());
        let num_start = usize::from(b.first() == Some(&b'-'));
        if num_start == num_end {
            return Err(RationalError::Parse { at: b.len(), kind: InvalidByte });
        }
        if let Some(s) = slash {
            if s + 1 == b.len() {
                return Err(RationalError::Parse { at: b.len(), kind: InvalidByte });
            }
        }

        let num_digits = &b[num_start..num_end];
        let negative = num_start == 1;
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(RationalError::Parse { at: num_start + 1, kind: NonCanonical });
        }
        if negative && num_digits == b"0" {
            return Err(RationalError::Parse { at: 0, kind: NonCanonical });
        }

        let den_digits = slash.map(|s| &b[s + 1..]);
        if let Some(ds) = den_digits {
            if ds.len() >= 2 && ds[0] == b'0' {
                return Err(RationalError::Parse {
                    at: slash.unwrap() + 2,
                    kind: NonCanonical,
                });
            }
        }

        let nmag = parse_magnitude(num_digits);
        let nlimit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let nmag = match nmag {
            Some(v) if v <= nlimit => v,
            _ => return Err(RationalError::Parse { at: 0, kind: OutOfRange }),
        };

        let dmag = if let Some(ds) = den_digits {
            match parse_magnitude(ds) {
                Some(v) if v <= i64::MAX as u128 => v,
                _ => return Err(RationalError::Parse {
                    at: slash.unwrap() + 1,
                    kind: OutOfRange,
                }),
            }
        } else {
            1
        };

        if let Some(ds) = den_digits {
            if dmag == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap() + 1,
                    kind: ZeroDenominator,
                });
            }
            if dmag == 1 {
                return Err(RationalError::Parse { at: slash.unwrap(), kind: NonCanonical });
            }
            if nmag == 0 {
                return Err(RationalError::Parse { at: slash.unwrap(), kind: NonCanonical });
            }
            if gcd_u128(nmag, dmag) > 1 {
                return Err(RationalError::Parse { at: slash.unwrap(), kind: NonCanonical });
            }
            let n = if negative { -(nmag as i128) } else { nmag as i128 };
            return Ok(Rational { num: n as i64, den: dmag as i64 });
        }

        let n = if negative { -(nmag as i128) } else { nmag as i128 };
        Ok(Rational { num: n as i64, den: 1 })
    }
}

fn parse_magnitude(digits: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &c in digits {
        value = value.checked_mul(10)?.checked_add((c - b'0') as u128)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum MatrixError {
    ShapeOverflow,
    DataLength { expected: usize, actual: usize },
    Index { row: usize, col: usize },
    DimensionMismatch { left: (usize, usize), right: (usize, usize) },
    Arithmetic { row: usize, col: usize, step: usize, source: RationalError },
    NotSquare { rows: usize, cols: usize },
    Singular { rank: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Matrix {
    rows: usize,
    cols: usize,
    data: Vec<Rational>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Reduction {
    pub matrix: Matrix,
    pub pivots: Vec<usize>,
    pub swaps: Vec<(usize, usize)>,
}

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()
    }
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Matrix { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Matrix { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Matrix { rows: self.rows, cols: rhs.cols, data })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(Rational { num: if r == c { 1 } else { 0 }, den: 1 });
            }
        }
        let aug = Matrix { rows: n, cols: width, data };
        let red = eliminate(aug, n, width)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            out.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Matrix { rows: n, cols: n, data: out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            data.extend_from_slice(&rhs.data[r * k..(r + 1) * k]);
        }
        let red = eliminate(Matrix { rows: self.rows, cols: width, data }, n, width)
            .map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            let coeff_zero = (0..n).all(|c| red.matrix.data[r * width + c].num == 0);
            if coeff_zero {
                if let Some(c) = (0..k).find(|&c| red.matrix.data[r * width + n + c].num != 0) {
                    return Ok(Solution::None {
                        row: r,
                        rhs_col: c,
                        witness: red.matrix.data[r * width..(r + 1) * width].to_vec(),
                    });
                }
            }
        }

        let rank = red.pivots.len();
        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (j, &pivot_col) in red.pivots.iter().enumerate() {
            let row = j;
            for c in 0..k {
                particular_data[pivot_col * k + c] = red.matrix.data[row * width + n + c];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular_data };

        if rank == n {
            return Ok(Solution::Unique { values: particular, pivots: red.pivots });
        }

        let mut free = Vec::new();
        for c in 0..n {
            if !red.pivots.contains(&c) {
                free.push(c);
            }
        }
        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * f + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in red.pivots.iter().enumerate() {
                let x = red.matrix.data[i * width + free_col];
                let neg = x.neg().map_err(|source| SolveError::Matrix(MatrixError::Arithmetic {
                    row: pivot_col,
                    col: j,
                    step: rank,
                    source,
                }))?;
                basis_data[pivot_col * f + j] = neg;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix { rows: n, cols: f, data: basis_data },
            pivots: red.pivots,
        })
    }
}

fn eliminate(mut matrix: Matrix, pivot_limit: usize, update_width: usize) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let width = matrix.cols;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for col in 0..pivot_limit {
        if p == rows {
            break;
        }
        let mut chosen: Option<usize> = None;
        for r in p..rows {
            let value = matrix.data[r * width + col];
            if value.num == 0 {
                continue;
            }
            if let Some(q) = chosen {
                let old = matrix.data[q * width + col];
                if pivot_key_cmp(value, r, old, q) == Ordering::Less {
                    chosen = Some(r);
                }
            } else {
                chosen = Some(r);
            }
        }
        let Some(q) = chosen else { continue };
        if q != p {
            for c in 0..width {
                matrix.data.swap(p * width + c, q * width + c);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * width + col];
        let mut normalized = Vec::with_capacity(update_width);
        for c in 0..update_width {
            let value = matrix.data[p * width + c];
            normalized.push(value.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                row: p,
                col: c,
                step: p,
                source,
            })?);
        }
        for c in 0..update_width {
            matrix.data[p * width + c] = normalized[c];
        }

        for r in 0..rows {
            if r == p {
                continue;
            }
            let factor = matrix.data[r * width + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..update_width {
                let product = factor.mul(&matrix.data[p * width + c]).map_err(|source| MatrixError::Arithmetic {
                    row: r,
                    col: c,
                    step: p,
                    source,
                })?;
                let result = matrix.data[r * width + c].sub(&product).map_err(|source| MatrixError::Arithmetic {
                    row: r,
                    col: c,
                    step: p,
                    source,
                })?;
                matrix.data[r * width + c] = result;
            }
        }
        pivots.push(col);
        p += 1;
    }

    Ok(Reduction { matrix, pivots, swaps })
}

fn pivot_key_cmp(a: Rational, ai: usize, b: Rational, bi: usize) -> Ordering {
    a.den.cmp(&b.den)
        .then_with(|| a.num.unsigned_abs().cmp(&b.num.unsigned_abs()))
        .then_with(|| ai.cmp(&bi))
}
