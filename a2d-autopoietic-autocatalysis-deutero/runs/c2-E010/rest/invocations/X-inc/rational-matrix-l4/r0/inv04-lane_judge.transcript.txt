    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    fn from_i128(num: i128, den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        let (mut n, mut d) = (num, den);
        if d < 0 {
            n = -n;
            d = -d;
        }
        if n == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
        n /= g;
        d /= g;
        if n < i64::MIN as i128
            || n > i64::MAX as i128
            || d < 1
            || d > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: n as i64,
            den: d as i64,
        })
    }

    fn from_i128_pair(num: i128, den: i128) -> Result<Self, RationalError> {
        Self::from_i128(num, den)
    }

    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_i128(num as i128, den as i128)
    }

        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;
        let a = rhs.den as i128 / g;
        let b = self.den as i128 / g;
        let n = self.num as i128 * a + rhs.num as i128 * b;
        let d = self.den as i128 * a;
        Self::from_i128_pair(n, d)

        let g = gcd_u128(self.den as u128, rhs.den as u128) as i128;

        let b = self.den as i128 / g;
        let n = self.num as i128 * a - rhs.num as i128 * b;
        let d = self.den as i128 * a;
        Self::from_i128_pair(n, d)

        let d = (self.den as i128 / g2) * (rhs.den as i128 / g1);

    }

    DimensionMismatch {
        left: (usize, usize),
        right: (usize, usize),

    Arithmetic {

#[derive(Clone, Debug, PartialEq, Eq)]

impl Reduction {
    pub fn rank(&self) -> usize {
        self.pivots.len()

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum SolveError {
    RowMismatch { coefficients: usize, rhs: usize },
    AugmentedShapeOverflow,
    Matrix(MatrixError),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique {
        values: Matrix,
        pivots: Vec<usize>,
    },
    None {
        row: usize,
        rhs_col: usize,
        witness: Vec<Rational>,
    },
    Infinite {
        particular: Matrix,
        free: Vec<usize>,
        basis: Matrix,
        pivots: Vec<usize>,

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength {
                expected,
                actual: data.len(),
            });
        }
        Ok(Self { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Self {
            rows: self.cols,
            cols: self.rows,
            data,
        }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self
            .rows
            .checked_mul(rhs.cols)
            .ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        let zero = Rational { num: 0, den: 1 };
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = zero;
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic {
                        row: r,
                        col: c,
                        step: k,
                        source,
                    })?;
                }
                data.push(acc);
            }
        }
        Ok(Self {
            rows: self.rows,
            cols: rhs.cols,
            data,
        })
    }

    fn eliminate(
        &self,
        candidate_cols: usize,
    ) -> Result<Reduction, MatrixError> {
        let mut matrix = self.clone();
        let mut pivots = Vec::new();
        let mut swaps = Vec::new();
        let mut p = 0usize;
        let zero = Rational { num: 0, den: 1 };

        for col in 0..candidate_cols {
            if p == matrix.rows {
                break;
            }
            let mut best: Option<(usize, i64, u64)> = None;
            for row in p..matrix.rows {
                let v = matrix.data[row * matrix.cols + col];
                if v.num == 0 {
                    continue;
                }
                let key = (v.den, v.num.unsigned_abs(), row);
                if best.map_or(true, |(br, bd, bn)| {
                    (key.0, key.1, key.2) < (bd, bn, br)
                }) {
                    best = Some((row, key.0, key.1));
                }
            }
            let Some((q, _, _)) = best else {
                continue;
            };
            if q != p {
                for c in 0..matrix.cols {
                    matrix.data.swap(p * matrix.cols + c, q * matrix.cols + c);
                }
                swaps.push((p, q));
            }
            let pivot = matrix.data[p * matrix.cols + col];
            for c in 0..matrix.cols {
                let idx = p * matrix.cols + c;
                matrix.data[idx] = matrix.data[idx]
                    .div(&pivot)
                    .map_err(|source| MatrixError::Arithmetic {
                        row: p,
                        col: c,
                        step: p,
                        source,
                    })?;
            }
            for row in 0..matrix.rows {
                if row == p {
                    continue;
                }
                let factor = matrix.data[row * matrix.cols + col];
                if factor == zero {
                    continue;
                }
                for c in 0..matrix.cols {
                    let idx = row * matrix.cols + c;
                    let product = factor
                        .mul(&matrix.data[p * matrix.cols + c])
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                    matrix.data[idx] = matrix.data[idx]
                        .sub(&product)
                        .map_err(|source| MatrixError::Arithmetic {
                            row,
                            col: c,
                            step: p,
                            source,
                        })?;
                }
            }
            pivots.push(col);
            p += 1;
        }
        Ok(Reduction {
            matrix,
            pivots,
            swaps,
        })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        self.eliminate(self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare {
                rows: self.rows,
                cols: self.cols,
            });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let zero = Rational { num: 0, den: 1 };
        let one = Rational { num: 1, den: 1 };
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(if r == c { one } else { zero });
            }
        }
        let aug = Self {
            rows: n,
            cols: width,
            data,
        };
        let red = aug.eliminate(n)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: red.pivots.len(),
            });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            for c in n..width {
                out.push(red.matrix.data[r * width + c]);
            }
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: out,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..k {
                data.push(rhs.data[r * k + c]);
            }
        }
        let aug = Self {
            rows: self.rows,
            cols: width,
            data,
        };
        let red = aug.eliminate(n).map_err(SolveError::Matrix)?;

        for row in 0..self.rows {
            let coeff_zero = (0..n).all(|c| red.matrix.data[row * width + c].num == 0);
            if coeff_zero {
                if let Some(rhs_col) = (0..k).find(|&c| red.matrix.data[row * width + n + c].num != 0) {
                    let witness = red.matrix.data[row * width..(row + 1) * width].to_vec();
                    return Ok(Solution::None {
                        row,
                        rhs_col,
                        witness,
                    });
                }
            }
        }

        let rank = red.pivots.len();
        let particular_len = n
            .checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let zero = Rational { num: 0, den: 1 };
        let mut particular = vec![zero; particular_len];
        for (j, &pivot_col) in red.pivots.iter().enumerate() {
            for c in 0..k {
                particular[pivot_col * k + c] = red.matrix.data[j * width + n + c];
            }
        }
        let particular = Self {
            rows: n,
            cols: k,
            data: particular,
        };
        let mut present = vec![false; n];
        for &p in &red.pivots {
            present[p] = true;
        }
        let free: Vec<usize> = (0..n).filter(|&c| !present[c]).collect();

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots: red.pivots,
            });
        }

        let f = free.len();
        let basis_len = n
            .checked_mul(f)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis = vec![zero; basis_len];
        let one = Rational { num: 1, den: 1 };
        for (j, &free_col) in free.iter().enumerate() {
            basis[free_col * f + j] = one;
            for (i, &pivot_col) in red.pivots.iter().enumerate() {
                let coefficient = red.matrix.data[i * width + free_col];
                let neg = coefficient.neg().map_err(|source| {
                    SolveError::Matrix(MatrixError::Arithmetic {
                        row: pivot_col,
                        col: j,
                        step: rank,
                        source,
                    })
                })?;
                basis[pivot_col * f + j] = neg;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Self {
                rows: n,
                cols: f,
                data: basis,
            },
            pivots: red.pivots,
        })
    }
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        Self::from_wide(num as i128, den as i128)
    }

    fn from_wide(mut num: i128, mut den: i128) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        if num == 0 {
            return Ok(Self { num: 0, den: 1 });
        }
        if den < 0 {
            num = -num;
            den = -den;
        }
        let g = gcd_u128(num.unsigned_abs(), den as u128) as i128;
        num /= g;
        den /= g;
        if num < i64::MIN as i128
            || num > i64::MAX as i128
            || den < 1
            || den > i64::MAX as i128
        {
            return Err(RationalError::Overflow);
        }
        Ok(Self {
            num: num as i64,
            den: den as i64,
        })
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_wide(
            self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        Self::from_wide(
            self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128,
            self.den as i128 * rhs.den as i128,
        )
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let fail = |at, kind| RationalError::Parse { at, kind };
        if bytes.is_empty() {
            return Err(fail(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }
        let num_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == num_start {
            return Err(fail(i, ParseKind::InvalidByte));
        }

        let mut den_start = None;
        let mut den_end = bytes.len();
        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            i += 1;
            den_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == den_start.unwrap() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            if i < bytes.len() {
                return Err(fail(i, ParseKind::InvalidByte));
            }
            den_end = i;
        }

        let num_end = den_start.map_or(bytes.len(), |s| s - 1);
        let num_digits_start = if negative { 1 } else { 0 };
        let num_digits = num_end - num_digits_start;
        if num_digits >= 2 && bytes[num_digits_start] == b'0' {
            return Err(fail(num_digits_start + 1, ParseKind::NonCanonical));
        }
        if negative && num_digits == 1 && bytes[0] == b'-' && bytes[1] == b'0' {
            return Err(fail(0, ParseKind::NonCanonical));
        }

        if let Some(ds) = den_start {
            if den_end - ds >= 2 && bytes[ds] == b'0' {
                return Err(fail(ds + 1, ParseKind::NonCanonical));
            }
        }

        let parse_magnitude = |start: usize, end: usize, limit: u128| -> Option<u128> {
            let mut value = 0u128;
            for &b in &bytes[start..end] {
                let digit = (b - b'0') as u128;
                value = value.checked_mul(10)?.checked_add(digit)?;
                if value > limit {
                    return None;
                }
            }
            Some(value)
        };

        let num_limit = if negative {
            1u128 << 63
        } else {
            i64::MAX as u128
        };
        let num_mag = parse_magnitude(num_digits_start, num_end, num_limit)
            .ok_or_else(|| fail(0, ParseKind::OutOfRange))?;
        let den_mag = if let Some(ds) = den_start {
            Some(
                parse_magnitude(ds, den_end, i64::MAX as u128)
                    .ok_or_else(|| fail(ds, ParseKind::OutOfRange))?,
            )
        } else {
            None
        };

        if let (Some(ds), Some(dm)) = (den_start, den_mag) {
            if dm == 0 {
                return Err(fail(ds, ParseKind::ZeroDenominator));
            }
            if dm == 1 {
                return Err(fail(ds - 1, ParseKind::NonCanonical));
            }
            if num_mag == 0 {
                return Err(fail(ds - 1, ParseKind::NonCanonical));
            }
            if gcd_u128(num_mag, dm) > 1 {
                return Err(fail(ds - 1, ParseKind::NonCanonical));
            }
        }

        let signed_num = if negative {
            if num_mag == (1u128 << 63) {
                i64::MIN
            } else {
                -(num_mag as i64)
            }
        } else {
            num_mag as i64
        };
        let den = den_mag.unwrap_or(1) as i64;
        Ok(Self {
            num: signed_num,
            den,
        })
    }

            });

                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(Rational {
                    num: if r == c { 1 } else { 0 },
                    den: 1,
                });
            }
        }
        let augmented = Self {
            rows: n,
            cols: width,
            data,
        };
        let reduction = augmented.eliminate(n)?;
        if reduction.pivots.len() < n {
            return Err(MatrixError::Singular {
                rank: reduction.pivots.len(),
            });
        }
        let mut inverse_data = Vec::with_capacity(n * n);
        for r in 0..n {
            inverse_data.extend_from_slice(&reduction.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Self {
            rows: n,
            cols: n,
            data: inverse_data,
        })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch {
                coefficients: self.rows,
                rhs: rhs.rows,
            });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self
            .rows
            .checked_mul(width)
            .ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * n..r * n + n]);
            data.extend_from_slice(&rhs.data[r * k..r * k + k]);
        }
        let augmented = Self {
            rows: self.rows,
            cols: width,
            data,
        };
        let reduction = augmented.eliminate(n).map_err(SolveError::Matrix)?;

        for row in 0..self.rows {
            let coeff_zero = (0..n).all(|c| reduction.matrix.data[row * width + c].num == 0);
            if coeff_zero {
                if let Some(rhs_col) =
                    (0..k).find(|&c| reduction.matrix.data[row * width + n + c].num != 0)
                {
                    return Ok(Solution::None {
                        row,
                        rhs_col,
                        witness: reduction.matrix.data[row * width..(row + 1) * width].to_vec(),
                    });
                }
            }
        }

        let rank = reduction.pivots.len();
        let particular_len = n
            .checked_mul(k)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
            for c in 0..k {
                particular_data[pivot_col * k + c] =
                    reduction.matrix.data[pivot_row * width + n + c];
            }
        }
        let particular = Self {
            rows: n,
            cols: k,
            data: particular_data,
        };

        let mut free = Vec::new();
        let mut pivot_set = vec![false; n];
        for &p in &reduction.pivots {
            pivot_set[p] = true;
        }
        for c in 0..n {
            if !pivot_set[c] {
                free.push(c);
            }
        }

        if rank == n {
            return Ok(Solution::Unique {
                values: particular,
                pivots: reduction.pivots,
            });
        }

        let f = free.len();
        let basis_len = n
            .checked_mul(f)
            .ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * f + j] = Rational { num: 1, den: 1 };
            for (pivot_row, &pivot_col) in reduction.pivots.iter().enumerate() {
                let value = reduction.matrix.data[pivot_row * width + free_col];
                basis_data[pivot_col * f + j] =
                    value.neg().map_err(|source| {
                        SolveError::Matrix(MatrixError::Arithmetic {
                            row: pivot_col,
                            col: j,
                            step: rank,
                            source,
                        })
                    })?;
            }
        }

        Ok(Solution::Infinite {
            particular,
            free,
            basis: Self {
                rows: n,
                cols: f,
                data: basis_data,
            },
            pivots: reduction.pivots,
        })
    }
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn rational_from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let mut n = num;
    let mut d = den;
    if d < 0 {
        n = -n;
        d = -d;
    }
    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }
    let g = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= g;
    d /= g;
    if n < i64::MIN as i128 || n > i64::MAX as i128 || d < 1 || d > i64::MAX as i128 {
        return Err(RationalError::Overflow);
    }
    Ok(Rational { num: n as i64, den: d as i64 })
}

fn rational_from_i128_pair(num: i128, den: i128) -> Result<Rational, RationalError> {
    rational_from_i128(num, den)
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        rational_from_i128(num as i128, den as i128)
    }

        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

        let n = self.num as i128 * rhs.den as i128 + rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        rational_from_i128_pair(n, d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let n = self.num as i128 * rhs.den as i128 - rhs.num as i128 * self.den as i128;
        let d = self.den as i128 * rhs.den as i128;
        rational_from_i128_pair(n, d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        rational_from_i128_pair(self.num as i128 * rhs.num as i128, self.den as i128 * rhs.den as i128)
    }

        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        rational_from_i128_pair(self.num as i128 * rhs.den as i128, self.den as i128 * rhs.num as i128)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        if self.num == i64::MIN {
            return Err(RationalError::Overflow);
        }
        Ok(Rational { num: -self.num, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        use ParseKind::*;
        let b = text.as_bytes();
        if b.is_empty() {
            return Err(RationalError::Parse { at: 0, kind: Empty });
        }

        let mut slash = None;
        for (i, &c) in b.iter().enumerate() {
            if c == b'/' {
                if slash.is_some() {
                    return Err(RationalError::Parse { at: i, kind: InvalidByte });
                }
                slash = Some(i);
                continue;
            }
            if c.is_ascii_digit() || (c == b'-' && i == 0) {
                continue;
            }
            return Err(RationalError::Parse { at: i, kind: InvalidByte });
        }

        let num_end = slash.unwrap_or(b.len());
        let num_start = usize::from(b.first() == Some(&b'-'));
        if num_start == num_end {
            return Err(RationalError::Parse { at: b.len(), kind: InvalidByte });
        }
        if let Some(s) = slash {
            if s + 1 == b.len() {
                return Err(RationalError::Parse { at: b.len(), kind: InvalidByte });
            }
        }

        let num_digits = &b[num_start..num_end];
        let negative = num_start == 1;
        if num_digits.len() >= 2 && num_digits[0] == b'0' {
            return Err(RationalError::Parse { at: num_start + 1, kind: NonCanonical });
        }
        if negative && num_digits == b"0" {
            return Err(RationalError::Parse { at: 0, kind: NonCanonical });
        }

        let den_digits = slash.map(|s| &b[s + 1..]);
        if let Some(ds) = den_digits {
            if ds.len() >= 2 && ds[0] == b'0' {
                return Err(RationalError::Parse {
                    at: slash.unwrap() + 2,
                    kind: NonCanonical,
                });
            }
        }

        let nmag = parse_magnitude(num_digits);
        let nlimit = if negative { 1u128 << 63 } else { i64::MAX as u128 };
        let nmag = match nmag {
            Some(v) if v <= nlimit => v,
            _ => return Err(RationalError::Parse { at: 0, kind: OutOfRange }),
        };

        let dmag = if let Some(ds) = den_digits {
            match parse_magnitude(ds) {
                Some(v) if v <= i64::MAX as u128 => v,
                _ => return Err(RationalError::Parse {
                    at: slash.unwrap() + 1,
                    kind: OutOfRange,
                }),
            }
        } else {
            1
        };

        if let Some(ds) = den_digits {
            if dmag == 0 {
                return Err(RationalError::Parse {
                    at: slash.unwrap() + 1,
                    kind: ZeroDenominator,
                });
            }
            if dmag == 1 {
                return Err(RationalError::Parse { at: slash.unwrap(), kind: NonCanonical });
            }
            if nmag == 0 {
                return Err(RationalError::Parse { at: slash.unwrap(), kind: NonCanonical });
            }
            if gcd_u128(nmag, dmag) > 1 {
                return Err(RationalError::Parse { at: slash.unwrap(), kind: NonCanonical });
            }
            let n = if negative { -(nmag as i128) } else { nmag as i128 };
            return Ok(Rational { num: n as i64, den: dmag as i64 });
        }

        let n = if negative { -(nmag as i128) } else { nmag as i128 };
        Ok(Rational { num: n as i64, den: 1 })
    }
}

    }
    Some(value)
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Solution {
    Unique { values: Matrix, pivots: Vec<usize> },
    None { row: usize, rhs_col: usize, witness: Vec<Rational> },
    Infinite { particular: Matrix, free: Vec<usize>, basis: Matrix, pivots: Vec<usize> },
}

impl Matrix {
    pub fn new(rows: usize, cols: usize, data: Vec<Rational>) -> Result<Self, MatrixError> {
        let expected = rows.checked_mul(cols).ok_or(MatrixError::ShapeOverflow)?;
        if data.len() != expected {
            return Err(MatrixError::DataLength { expected, actual: data.len() });
        }
        Ok(Matrix { rows, cols, data })
    }

    pub fn rows(&self) -> usize {
        self.rows
    }

    pub fn cols(&self) -> usize {
        self.cols
    }

    pub fn data(&self) -> &[Rational] {
        &self.data
    }

    pub fn get(&self, row: usize, col: usize) -> Result<Rational, MatrixError> {
        if row >= self.rows || col >= self.cols {
            return Err(MatrixError::Index { row, col });
        }
        Ok(self.data[row * self.cols + col])
    }

    pub fn transpose(&self) -> Self {
        let mut data = Vec::with_capacity(self.data.len());
        for r in 0..self.cols {
            for c in 0..self.rows {
                data.push(self.data[c * self.cols + r]);
            }
        }
        Matrix { rows: self.cols, cols: self.rows, data }
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, MatrixError> {
        if self.cols != rhs.rows {
            return Err(MatrixError::DimensionMismatch {
                left: (self.rows, self.cols),
                right: (rhs.rows, rhs.cols),
            });
        }
        let len = self.rows.checked_mul(rhs.cols).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            for c in 0..rhs.cols {
                let mut acc = Rational { num: 0, den: 1 };
                for k in 0..self.cols {
                    let a = self.data[r * self.cols + k];
                    let b = rhs.data[k * rhs.cols + c];
                    let term = a.mul(&b).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                    acc = acc.add(&term).map_err(|source| MatrixError::Arithmetic { row: r, col: c, step: k, source })?;
                }
                data.push(acc);
            }
        }
        Ok(Matrix { rows: self.rows, cols: rhs.cols, data })
    }

    pub fn rref(&self) -> Result<Reduction, MatrixError> {
        eliminate(self.clone(), self.cols, self.cols)
    }

    pub fn rank(&self) -> Result<usize, MatrixError> {
        Ok(self.rref()?.rank())
    }

    pub fn inverse(&self) -> Result<Self, MatrixError> {
        if self.rows != self.cols {
            return Err(MatrixError::NotSquare { rows: self.rows, cols: self.cols });
        }
        let n = self.rows;
        let width = n.checked_mul(2).ok_or(MatrixError::ShapeOverflow)?;
        let len = n.checked_mul(width).ok_or(MatrixError::ShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..n {
            for c in 0..n {
                data.push(self.data[r * n + c]);
            }
            for c in 0..n {
                data.push(Rational { num: if r == c { 1 } else { 0 }, den: 1 });
            }
        }
        let aug = Matrix { rows: n, cols: width, data };
        let red = eliminate(aug, n, width)?;
        if red.pivots.len() < n {
            return Err(MatrixError::Singular { rank: red.pivots.len() });
        }
        let mut out = Vec::with_capacity(n * n);
        for r in 0..n {
            out.extend_from_slice(&red.matrix.data[r * width + n..r * width + width]);
        }
        Ok(Matrix { rows: n, cols: n, data: out })
    }

    pub fn solve(&self, rhs: &Self) -> Result<Solution, SolveError> {
        if self.rows != rhs.rows {
            return Err(SolveError::RowMismatch { coefficients: self.rows, rhs: rhs.rows });
        }
        let n = self.cols;
        let k = rhs.cols;
        let width = n.checked_add(k).ok_or(SolveError::AugmentedShapeOverflow)?;
        let len = self.rows.checked_mul(width).ok_or(SolveError::AugmentedShapeOverflow)?;
        let mut data = Vec::with_capacity(len);
        for r in 0..self.rows {
            data.extend_from_slice(&self.data[r * n..(r + 1) * n]);
            data.extend_from_slice(&rhs.data[r * k..(r + 1) * k]);
        }
        let red = eliminate(Matrix { rows: self.rows, cols: width, data }, n, width)
            .map_err(SolveError::Matrix)?;

        for r in 0..self.rows {
            let coeff_zero = (0..n).all(|c| red.matrix.data[r * width + c].num == 0);
            if coeff_zero {
                if let Some(c) = (0..k).find(|&c| red.matrix.data[r * width + n + c].num != 0) {
                    return Ok(Solution::None {
                        row: r,
                        rhs_col: c,
                        witness: red.matrix.data[r * width..(r + 1) * width].to_vec(),
                    });
                }
            }
        }

        let rank = red.pivots.len();
        let particular_len = n.checked_mul(k).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut particular_data = vec![Rational { num: 0, den: 1 }; particular_len];
        for (j, &pivot_col) in red.pivots.iter().enumerate() {
            let row = j;
            for c in 0..k {
                particular_data[pivot_col * k + c] = red.matrix.data[row * width + n + c];
            }
        }
        let particular = Matrix { rows: n, cols: k, data: particular_data };

        if rank == n {
            return Ok(Solution::Unique { values: particular, pivots: red.pivots });
        }

        let mut free = Vec::new();
        for c in 0..n {
            if !red.pivots.contains(&c) {
                free.push(c);
            }
        }
        let f = free.len();
        let basis_len = n.checked_mul(f).ok_or(SolveError::Matrix(MatrixError::ShapeOverflow))?;
        let mut basis_data = vec![Rational { num: 0, den: 1 }; basis_len];
        for (j, &free_col) in free.iter().enumerate() {
            basis_data[free_col * f + j] = Rational { num: 1, den: 1 };
            for (i, &pivot_col) in red.pivots.iter().enumerate() {
                let x = red.matrix.data[i * width + free_col];
                let neg = x.neg().map_err(|source| SolveError::Matrix(MatrixError::Arithmetic {
                    row: pivot_col,
                    col: j,
                    step: rank,
                    source,
                }))?;
                basis_data[pivot_col * f + j] = neg;
            }
        }
        Ok(Solution::Infinite {
            particular,
            free,
            basis: Matrix { rows: n, cols: f, data: basis_data },
            pivots: red.pivots,
        })
    }
}

fn eliminate(mut matrix: Matrix, pivot_limit: usize, update_width: usize) -> Result<Reduction, MatrixError> {
    let rows = matrix.rows;
    let width = matrix.cols;
    let mut pivots = Vec::new();
    let mut swaps = Vec::new();
    let mut p = 0usize;

    for col in 0..pivot_limit {
        if p == rows {
            break;
        }
        let mut chosen: Option<usize> = None;
        for r in p..rows {
            let value = matrix.data[r * width + col];
            if value.num == 0 {
                continue;
            }
            if let Some(q) = chosen {
                let old = matrix.data[q * width + col];
                if pivot_key_cmp(value, r, old, q) == Ordering::Less {
                    chosen = Some(r);
                }
            } else {
                chosen = Some(r);
            }
        }
        let Some(q) = chosen else { continue };
        if q != p {
            for c in 0..width {
                matrix.data.swap(p * width + c, q * width + c);
            }
            swaps.push((p, q));
        }

        let pivot = matrix.data[p * width + col];
        let mut normalized = Vec::with_capacity(update_width);
        for c in 0..update_width {
            let value = matrix.data[p * width + c];
            normalized.push(value.div(&pivot).map_err(|source| MatrixError::Arithmetic {
                row: p,
                col: c,
                step: p,
                source,
            })?);
        }
        for c in 0..update_width {
            matrix.data[p * width + c] = normalized[c];
        }

        for r in 0..rows {
            if r == p {
                continue;
            }
            let factor = matrix.data[r * width + col];
            if factor.num == 0 {
                continue;
            }
            for c in 0..update_width {
                let product = factor.mul(&matrix.data[p * width + c]).map_err(|source| MatrixError::Arithmetic {
                    row: r,
                    col: c,
                    step: p,
                    source,
                })?;
                let result = matrix.data[r * width + c].sub(&product).map_err(|source| MatrixError::Arithmetic {
                    row: r,
                    col: c,
                    step: p,
                    source,
                })?;
                matrix.data[r * width + c] = result;
            }
        }
        pivots.push(col);
        p += 1;
    }

    Ok(Reduction { matrix, pivots, swaps })
}

fn pivot_key_cmp(a: Rational, ai: usize, b: Rational, bi: usize) -> Ordering {
    a.den.cmp(&b.den)
        .then_with(|| a.num.unsigned_abs().cmp(&b.num.unsigned_abs()))
        .then_with(|| ai.cmp(&bi))
}
