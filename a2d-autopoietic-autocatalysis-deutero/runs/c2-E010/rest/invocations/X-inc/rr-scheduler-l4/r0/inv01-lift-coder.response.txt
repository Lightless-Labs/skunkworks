#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy)]
enum EventKind {
    Arrival,
    Wake { job: usize, level: usize },
}

#[derive(Clone, Copy)]
struct Event {
    time: u64,
    index: usize,
    kind: EventKind,
}

fn event_key(event: &Event) -> (u64, u8, usize) {
    let kind = match event.kind {
        EventKind::Arrival => 0,
        EventKind::Wake { .. } => 1,
    };
    (event.time, kind, event.index)
}

fn rounded_mean_x100(total: u64, count: usize) -> u64 {
    if count == 0 {
        0
    } else {
        let numerator = total * 100;
        let denominator = count as u64;
        (numerator + denominator / 2) / denominator
    }
}

pub fn schedule(
    jobs: &[Job],
    quanta: &[u32],
    switch_cost: u32,
) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }

    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }

    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }

    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &burst) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&burst)
            } else {
                burst <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: u64::from(job.arrival),
            index,
            kind: EventKind::Arrival,
        });
    }
    events.sort_by_key(event_key);

    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut burst_position = vec![0usize; jobs.len()];
    let mut remaining_cpu: Vec<u64> = jobs
        .iter()
        .map(|job| u64::from(job.bursts[0]))
        .collect();
    let mut levels: Vec<usize> = jobs.iter().map(|job| job.level).collect();
    let mut completions = vec![0u64; jobs.len()];
    let mut runs = Vec::new();
    let mut idle = Vec::new();

    let mut time = 0u64;
    let mut event_cursor = 0usize;
    let mut completed = 0usize;
    let mut last_dispatch_id: Option<u32> = None;

    while completed < jobs.len() {
        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => queues[levels[event.index]].push_back(event.index),
                EventKind::Wake { job, level } => {
                    levels[job] = level;
                    queues[level].push_back(job);
                }
            }
            event_cursor += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if event_cursor >= events.len() {
                break;
            }
            let next_time = events[event_cursor].time;
            if next_time > time {
                if let Some(last) = idle.last_mut() {
                    if last.1 == time {
                        last.1 = next_time;
                    } else {
                        idle.push((time, next_time));
                    }
                } else {
                    idle.push((time, next_time));
                }
                time = next_time;
                last_dispatch_id = None;
            }
            continue;
        };

        let job_index = queues[level].pop_front().unwrap();
        let job_id = jobs[job_index].id;

        if last_dispatch_id.is_some_and(|last| last != job_id) {
            time += u64::from(switch_cost);
        }

        while event_cursor < events.len() && events[event_cursor].time <= time {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => queues[levels[event.index]].push_back(event.index),
                EventKind::Wake { job, level } => {
                    levels[job] = level;
                    queues[level].push_back(job);
                }
            }
            event_cursor += 1;
        }

        let consumed = remaining_cpu[job_index].min(u64::from(quanta[level]));
        let start = time;
        time += consumed;
        let end = time;
        remaining_cpu[job_index] -= consumed;

        let current_burst_ends = remaining_cpu[job_index] == 0;
        let final_burst = current_burst_ends
            && burst_position[job_index] + 1 == jobs[job_index].bursts.len();

        runs.push(Run {
            id: job_id,
            start,
            end,
            finished: final_burst,
            level,
        });

        if current_burst_ends && !final_burst {
            let io_position = burst_position[job_index] + 1;
            let io_delay = u64::from(jobs[job_index].bursts[io_position]);
            let next_cpu_position = io_position + 1;
            burst_position[job_index] = next_cpu_position;
            remaining_cpu[job_index] =
                u64::from(jobs[job_index].bursts[next_cpu_position]);
            events.push(Event {
                time: end + io_delay,
                index: job_index,
                kind: EventKind::Wake {
                    job: job_index,
                    level: level.saturating_sub(1),
                },
            });
            events[event_cursor..].sort_by_key(event_key);
        } else if !current_burst_ends {
            levels[job_index] = (level + 1).min(quanta.len() - 1);
        } else {
            completions[job_index] = end;
            completed += 1;
        }

        while event_cursor < events.len() && events[event_cursor].time <= end {
            let event = events[event_cursor];
            match event.kind {
                EventKind::Arrival => queues[levels[event.index]].push_back(event.index),
                EventKind::Wake { job, level } => {
                    levels[job] = level;
                    queues[level].push_back(job);
                }
            }
            event_cursor += 1;
        }

        if !final_burst {
            if current_burst_ends {
                // The job is blocked until its wake event.
            } else {
                queues[levels[job_index]].push_back(job_index);
            }
        }

        last_dispatch_id = Some(job_id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = completions[index];
        let turnaround = completion - u64::from(job.arrival);
        let cpu_total: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&burst| u64::from(burst))
            .sum();
        let io_total: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&burst| u64::from(burst))
            .sum();
        let waiting = turnaround - cpu_total - io_total;

        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100: rounded_mean_x100(total_waiting, jobs.len()),
        mean_turnaround_x100: rounded_mean_x100(total_turnaround, jobs.len()),
        end: time,
    })
}
