```rust
#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Job {
    pub id: u32,
    pub arrival: u32,
    pub bursts: Vec<u32>,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Run {
    pub id: u32,
    pub start: u64,
    pub end: u64,
    pub finished: bool,
    pub level: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub idle: Vec<(u64, u64)>,
    pub stats: Vec<Stat>,
    pub mean_waiting_x100: u64,
    pub mean_turnaround_x100: u64,
    pub end: u64,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    LevelCount,
    Quantum { level: usize },
    SwitchCost,
    TooManyJobs,
    DuplicateId { index: usize },
    Arrival { index: usize },
    InitialLevel { index: usize },
    BurstCount { index: usize },
    Burst { index: usize, position: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Stat {
    pub id: u32,
    pub completion: u64,
    pub turnaround: u64,
    pub waiting: u64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
enum EventKind {
    Arrival,
    Wake,
}

#[derive(Clone, Copy, Debug)]
struct Event {
    time: u64,
    kind: EventKind,
    index: usize,
    level: usize,
}

#[derive(Clone, Debug)]
struct State {
    burst_position: usize,
    remaining: u32,
    level: usize,
    completion: Option<u64>,
}

pub fn schedule(jobs: &[Job], quanta: &[u32], switch_cost: u32) -> Result<Schedule, Error> {
    if !(1..=8).contains(&quanta.len()) {
        return Err(Error::LevelCount);
    }
    for (level, &quantum) in quanta.iter().enumerate() {
        if !(1..=1000).contains(&quantum) {
            return Err(Error::Quantum { level });
        }
    }
    if switch_cost > 1000 {
        return Err(Error::SwitchCost);
    }
    if jobs.len() > 64 {
        return Err(Error::TooManyJobs);
    }

    for (index, job) in jobs.iter().enumerate() {
        if jobs[..index].iter().any(|earlier| earlier.id == job.id) {
            return Err(Error::DuplicateId { index });
        }
        if job.arrival > 1000 {
            return Err(Error::Arrival { index });
        }
        if job.level >= quanta.len() {
            return Err(Error::InitialLevel { index });
        }
        if job.bursts.is_empty() || job.bursts.len() > 31 || job.bursts.len() % 2 == 0 {
            return Err(Error::BurstCount { index });
        }
        for (position, &length) in job.bursts.iter().enumerate() {
            let valid = if position % 2 == 0 {
                (1..=1000).contains(&length)
            } else {
                length <= 1000
            };
            if !valid {
                return Err(Error::Burst { index, position });
            }
        }
    }

    if jobs.is_empty() {
        return Ok(Schedule {
            runs: Vec::new(),
            idle: Vec::new(),
            stats: Vec::new(),
            mean_waiting_x100: 0,
            mean_turnaround_x100: 0,
            end: 0,
        });
    }

    let mut states: Vec<State> = jobs
        .iter()
        .map(|job| State {
            burst_position: 0,
            remaining: job.bursts[0],
            level: job.level,
            completion: None,
        })
        .collect();

    let mut events = Vec::with_capacity(jobs.len() * 16);
    for (index, job) in jobs.iter().enumerate() {
        events.push(Event {
            time: u64::from(job.arrival),
            kind: EventKind::Arrival,
            index,
            level: job.level,
        });
    }
    events.sort_by_key(|event| {
        (
            event.time,
            match event.kind {
                EventKind::Arrival => 0u8,
                EventKind::Wake => 1u8,
            },
            event.index,
        )
    });

    let mut next_event = 0usize;
    let mut queues: Vec<std::collections::VecDeque<usize>> =
        (0..quanta.len()).map(|_| std::collections::VecDeque::new()).collect();
    let mut runs = Vec::new();
    let mut idle = Vec::new();
    let mut now = 0u64;
    let mut last_dispatch: Option<u32> = None;
    let mut completed = 0usize;

    while completed < jobs.len() {
        while next_event < events.len() && events[next_event].time <= now {
            let event = events[next_event];
            states[event.index].level = event.level;
            queues[event.level].push_back(event.index);
            next_event += 1;
        }

        let selected_level = queues.iter().position(|queue| !queue.is_empty());
        let Some(level) = selected_level else {
            if next_event >= events.len() {
                break;
            }
            let next_time = events[next_event].time;
            if next_time > now {
                if let Some(last) = idle.last_mut() {
                    if last.1 == now {
                        last.1 = next_time;
                    } else {
                        idle.push((now, next_time));
                    }
                } else {
                    idle.push((now, next_time));
                }
                now = next_time;
                last_dispatch = None;
            }
            continue;
        };

        let index = queues[level].pop_front().expect("selected queue is nonempty");
        let job = &jobs[index];

        if last_dispatch.is_some_and(|last_id| last_id != job.id) {
            now += u64::from(switch_cost);
        }

        while next_event < events.len() && events[next_event].time <= now {
            let event = events[next_event];
            states[event.index].level = event.level;
            queues[event.level].push_back(event.index);
            next_event += 1;
        }

        let quantum = quanta[level];
        let consumed = states[index].remaining.min(quantum);
        let start = now;
        now += u64::from(consumed);
        let end = now;

        states[index].remaining -= consumed;
        let burst_finished = states[index].remaining == 0;
        let job_finished = burst_finished && states[index].burst_position + 1 == job.bursts.len();

        runs.push(Run {
            id: job.id,
            start,
            end,
            finished: job_finished,
            level,
        });

        while next_event < events.len() && events[next_event].time <= end {
            let event = events[next_event];
            states[event.index].level = event.level;
            queues[event.level].push_back(event.index);
            next_event += 1;
        }

        if job_finished {
            states[index].completion = Some(end);
            completed += 1;
        } else if burst_finished {
            let wake_time = end + u64::from(job.bursts[states[index].burst_position + 1]);
            let wake_level = level.saturating_sub(1);
            states[index].burst_position += 2;
            states[index].remaining = job.bursts[states[index].burst_position];

            events.push(Event {
                time: wake_time,
                kind: EventKind::Wake,
                index,
                level: wake_level,
            });
            events[next_event..].sort_by_key(|event| {
                (
                    event.time,
                    match event.kind {
                        EventKind::Arrival => 0u8,
                        EventKind::Wake => 1u8,
                    },
                    event.index,
                )
            });

            while next_event < events.len() && events[next_event].time <= end {
                let event = events[next_event];
                states[event.index].level = event.level;
                queues[event.level].push_back(event.index);
                next_event += 1;
            }
        } else {
            let new_level = (level + 1).min(quanta.len() - 1);
            states[index].level = new_level;
            queues[new_level].push_back(index);
        }

        last_dispatch = Some(job.id);
    }

    let mut stats = Vec::with_capacity(jobs.len());
    let mut total_waiting = 0u64;
    let mut total_turnaround = 0u64;

    for (index, job) in jobs.iter().enumerate() {
        let completion = states[index].completion.unwrap_or(now);
        let turnaround = completion - u64::from(job.arrival);
        let cpu: u64 = job
            .bursts
            .iter()
            .step_by(2)
            .map(|&length| u64::from(length))
            .sum();
        let io: u64 = job
            .bursts
            .iter()
            .skip(1)
            .step_by(2)
            .map(|&length| u64::from(length))
            .sum();
        let waiting = turnaround - cpu - io;
        total_waiting += waiting;
        total_turnaround += turnaround;
        stats.push(Stat {
            id: job.id,
            completion,
            turnaround,
            waiting,
        });
    }

    let count = jobs.len() as u64;
    let mean_waiting_x100 = (total_waiting * 100 + count / 2) / count;
    let mean_turnaround_x100 = (total_turnaround * 100 + count / 2) / count;
    let end = stats.iter().map(|stat| stat.completion).max().unwrap_or(0);

    Ok(Schedule {
        runs,
        idle,
        stats,
        mean_waiting_x100,
        mean_turnaround_x100,
        end,
    })
}
```
