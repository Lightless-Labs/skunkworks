```rust
use std::fmt;

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub struct Rational {
    num: i64,
    den: i64,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum ParseKind {
    Empty,
    InvalidByte,
    NonCanonical,
    OutOfRange,
    ZeroDenominator,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq)]
pub enum RationalError {
    ZeroDenominator,
    DivideByZero,
    Overflow,
    Parse { at: usize, kind: ParseKind },
}

fn gcd_u128(mut a: u128, mut b: u128) -> u128 {
    while b != 0 {
        let r = a % b;
        a = b;
        b = r;
    }
    a
}

fn from_i128(num: i128, den: i128) -> Result<Rational, RationalError> {
    if den == 0 {
        return Err(RationalError::ZeroDenominator);
    }

    let (mut n, mut d) = (num, den);
    if d < 0 {
        n = -n;
        d = -d;
    }

    if n == 0 {
        return Ok(Rational { num: 0, den: 1 });
    }

    let divisor = gcd_u128(n.unsigned_abs(), d as u128) as i128;
    n /= divisor;
    d /= divisor;

    let num = i64::try_from(n).map_err(|_| RationalError::Overflow)?;
    let den = i64::try_from(d).map_err(|_| RationalError::Overflow)?;
    Ok(Rational { num, den })
}

fn from_u128_signed(
    negative: bool,
    magnitude: u128,
    denominator: u128,
) -> Result<Rational, RationalError> {
    if denominator == 0 {
        return Err(RationalError::ZeroDenominator);
    }
    let divisor = gcd_u128(magnitude, denominator);
    let n = magnitude / divisor;
    let d = denominator / divisor;

    if d > i64::MAX as u128 {
        return Err(RationalError::Overflow);
    }

    let num = if negative {
        if n > (1u128 << 63) {
            return Err(RationalError::Overflow);
        }
        if n == (1u128 << 63) {
            i64::MIN
        } else {
            -(n as i64)
        }
    } else {
        i64::try_from(n).map_err(|_| RationalError::Overflow)?
    };

    Ok(Rational {
        num,
        den: d as i64,
    })
}

impl Rational {
    pub fn new(num: i64, den: i64) -> Result<Self, RationalError> {
        if den == 0 {
            return Err(RationalError::ZeroDenominator);
        }
        from_i128(num as i128, den as i128)
    }

    pub fn numerator(&self) -> i64 {
        self.num
    }

    pub fn denominator(&self) -> i64 {
        self.den
    }

    pub fn add(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        from_i128(a * d + c * b, b * d)
    }

    pub fn sub(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        from_i128(a * d - c * b, b * d)
    }

    pub fn mul(&self, rhs: &Self) -> Result<Self, RationalError> {
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        from_i128(a * c, b * d)
    }

    pub fn div(&self, rhs: &Self) -> Result<Self, RationalError> {
        if rhs.num == 0 {
            return Err(RationalError::DivideByZero);
        }
        let a = self.num as i128;
        let b = self.den as i128;
        let c = rhs.num as i128;
        let d = rhs.den as i128;
        from_i128(a * d, b * c)
    }

    pub fn neg(&self) -> Result<Self, RationalError> {
        let n = self.num.checked_neg().ok_or(RationalError::Overflow)?;
        Ok(Self { num: n, den: self.den })
    }

    pub fn parse(text: &str) -> Result<Self, RationalError> {
        let bytes = text.as_bytes();
        let parse_error = |at, kind| RationalError::Parse { at, kind };

        if bytes.is_empty() {
            return Err(parse_error(0, ParseKind::Empty));
        }

        let mut i = 0;
        let negative = bytes[0] == b'-';
        if negative {
            i = 1;
        }

        let numerator_start = i;
        while i < bytes.len() && bytes[i].is_ascii_digit() {
            i += 1;
        }
        if i == numerator_start {
            return Err(parse_error(i, ParseKind::InvalidByte));
        }

        let numerator_end = i;
        let mut slash = None;
        let mut denominator_start = None;
        let mut denominator_end = None;

        if i < bytes.len() {
            if bytes[i] != b'/' {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            slash = Some(i);
            i += 1;
            denominator_start = Some(i);
            while i < bytes.len() && bytes[i].is_ascii_digit() {
                i += 1;
            }
            if i == denominator_start.unwrap() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
            denominator_end = Some(i);
            if i < bytes.len() {
                return Err(parse_error(i, ParseKind::InvalidByte));
            }
        }

        let numerator_digits = &bytes[numerator_start..numerator_end];
        if numerator_digits.len() >= 2 && numerator_digits[0] == b'0' {
            return Err(parse_error(numerator_start + 1, ParseKind::NonCanonical));
        }
        if negative && numerator_digits == b"0" {
            return Err(parse_error(0, ParseKind::NonCanonical));
        }

        if let (Some(start), Some(end)) = (denominator_start, denominator_end) {
            let digits = &bytes[start..end];
            if digits.len() >= 2 && digits[0] == b'0' {
                return Err(parse_error(start + 1, ParseKind::NonCanonical));
            }
        }

        let numerator_magnitude = parse_magnitude(numerator_digits)
            .filter(|&n| n <= if negative { 1u128 << 63 } else { i64::MAX as u128 })
            .ok_or_else(|| parse_error(0, ParseKind::OutOfRange))?;

        let denominator_magnitude = if let (Some(start), Some(end)) =
            (denominator_start, denominator_end)
        {
            let digits = &bytes[start..end];
            Some(
                parse_magnitude(digits)
                    .filter(|&n| n <= i64::MAX as u128)
                    .ok_or_else(|| parse_error(start, ParseKind::OutOfRange))?,
            )
        } else {
            None
        };

        if let (Some(start), Some(value)) = (denominator_start, denominator_magnitude) {
            if value == 0 {
                return Err(parse_error(start, ParseKind::ZeroDenominator));
            }
        }

        if let (Some(slash_at), Some(value)) = (slash, denominator_magnitude) {
            if value == 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if numerator_magnitude == 0 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
            if gcd_u128(numerator_magnitude, value) > 1 {
                return Err(parse_error(slash_at, ParseKind::NonCanonical));
            }
        }

        let den = denominator_magnitude.unwrap_or(1);
        from_u128_signed(negative, numerator_magnitude, den)
            .map_err(|_| parse_error(0, ParseKind::OutOfRange))
    }
}

fn parse_magnitude(digits: &[u8]) -> Option<u128> {
    let mut value = 0u128;
    for &byte in digits {
        value = value.checked_mul(10)?.checked_add((byte - b'0') as u128)?;
    }
    Some(value)
}

impl fmt::Display for Rational {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        if self.den == 1 {
            write!(f, "{}", self.num)
        } else {
            write!(f, "{}/{}", self.num, self.den)
        }
    }
}
```
