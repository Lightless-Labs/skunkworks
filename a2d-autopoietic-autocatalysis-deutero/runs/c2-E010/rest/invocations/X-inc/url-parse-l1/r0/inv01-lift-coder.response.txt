#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Url {
    pub scheme: String,
    pub userinfo: Option<String>,
    pub host: String,
    pub port: Option<u16>,
    pub path: String,
    pub query: Option<String>,
    pub fragment: Option<String>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum ParseError {
    Empty,
    InvalidByte { at: usize },
    InvalidScheme { at: usize },
    MissingAuthority { at: usize },
    InvalidUserInfo { at: usize },
    InvalidHost { at: usize },
    InvalidPort { at: usize },
}

pub fn parse_url(input: &str) -> Result<Url, ParseError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(ParseError::Empty);
    }

    for (at, &byte) in bytes.iter().enumerate() {
        if !(0x21..=0x7e).contains(&byte) {
            return Err(ParseError::InvalidByte { at });
        }
    }

    let colon = match bytes.iter().position(|&byte| byte == b':') {
        Some(at) => at,
        None => return Err(ParseError::InvalidScheme { at: bytes.len() }),
    };

    if colon == 0 || !bytes[0].is_ascii_alphabetic() {
        return Err(ParseError::InvalidScheme { at: 0 });
    }

    for (at, &byte) in bytes[..colon].iter().enumerate().skip(1) {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'+' | b'-' | b'.')) {
            return Err(ParseError::InvalidScheme { at });
        }
    }

    let authority_start = colon + 1;
    if bytes.get(authority_start) != Some(&b'/')
        || bytes.get(authority_start + 1) != Some(&b'/')
    {
        return Err(ParseError::MissingAuthority { at: authority_start });
    }

    let authority_start = authority_start + 2;
    let authority_end = bytes[authority_start..]
        .iter()
        .position(|&byte| matches!(byte, b'/' | b'?' | b'#'))
        .map_or(bytes.len(), |relative| authority_start + relative);
    let authority = &input[authority_start..authority_end];

    let mut at_signs = authority.match_indices('@');
    let first_at = at_signs.next().map(|(at, _)| at);
    let second_at = at_signs.next().map(|(at, _)| at);

    if let Some(at) = second_at {
        return Err(ParseError::InvalidUserInfo {
            at: authority_start + at,
        });
    }

    let (userinfo, host_port_start) = match first_at {
        Some(at) if at == 0 => {
            return Err(ParseError::InvalidUserInfo {
                at: authority_start + at,
            });
        }
        Some(at) => (
            Some(authority[..at].to_owned()),
            authority_start + at + 1,
        ),
        None => (None, authority_start),
    };

    let host_port = &input[host_port_start..authority_end];
    let port_colon = host_port.find(':');
    let host_end = port_colon.unwrap_or(host_port.len());
    let host = &host_port[..host_end];

    if host.is_empty() {
        return Err(ParseError::InvalidHost { at: host_port_start });
    }

    for (offset, byte) in host.bytes().enumerate() {
        if !(byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'-' | b'_')) {
            return Err(ParseError::InvalidHost {
                at: host_port_start + offset,
            });
        }
    }

    let port = if let Some(colon_offset) = port_colon {
        let port_start = host_port_start + colon_offset + 1;
        let port_text = &host_port[colon_offset + 1..];

        if port_text.is_empty() {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        for (offset, byte) in port_text.bytes().enumerate() {
            if !byte.is_ascii_digit() {
                return Err(ParseError::InvalidPort {
                    at: port_start + offset,
                });
            }
        }

        if port_text.len() >= 2 && port_text.as_bytes()[0] == b'0' {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        let mut value: u32 = 0;
        for byte in port_text.bytes() {
            value = match value
                .checked_mul(10)
                .and_then(|value| value.checked_add((byte - b'0') as u32))
            {
                Some(value) => value,
                None => return Err(ParseError::InvalidPort { at: port_start }),
            };
        }

        if !(1..=65535).contains(&value) {
            return Err(ParseError::InvalidPort { at: port_start });
        }

        Some(value as u16)
    } else {
        None
    };

    let fragment_delimiter = bytes[authority_end..]
        .iter()
        .position(|&byte| byte == b'#')
        .map(|relative| authority_end + relative);
    let before_fragment_end = fragment_delimiter.unwrap_or(bytes.len());

    let query_delimiter = bytes[authority_end..before_fragment_end]
        .iter()
        .position(|&byte| byte == b'?')
        .map(|relative| authority_end + relative);

    let component_end = query_delimiter.unwrap_or(before_fragment_end);
    let path = if component_end == authority_end {
        "/".to_owned()
    } else {
        input[authority_end..component_end].to_owned()
    };

    let query = query_delimiter.map(|at| {
        input[at + 1..before_fragment_end].to_owned()
    });
    let fragment = fragment_delimiter.map(|at| input[at + 1..].to_owned());

    Ok(Url {
        scheme: input[..colon].to_ascii_lowercase(),
        userinfo,
        host: host.to_ascii_lowercase(),
        port,
        path,
        query,
        fragment,
    })
}
