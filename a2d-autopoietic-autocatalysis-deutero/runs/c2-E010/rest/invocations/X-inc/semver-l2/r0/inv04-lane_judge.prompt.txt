use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn parse_core_field(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }

    if *pos == start {
        if start == bytes.len() || matches!(bytes[start], b'.' | b'-' | b'+') {
            return Err(VersionError::MissingCore { at: start });
        }
        return Err(VersionError::InvalidChar { at: start });
    }

    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }

    let mut value = 0u64;
    for &digit in &bytes[start..*pos] {
        value = value
            .checked_mul(10)
            .and_then(|n| n.checked_add((digit - b'0') as u64))
            .ok_or(VersionError::Overflow { at: start })?;
    }
    Ok(value)
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut ident_start = start;

    loop {
        let mut pos = ident_start;
        while pos < end && bytes[pos] != b'.' {
            let b = bytes[pos];
            if !(b.is_ascii_alphanumeric() || b == b'-') {
                return Err(VersionError::InvalidChar { at: pos });
            }
            pos += 1;
        }

        if pos == ident_start {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }

        let ident = &s[ident_start..pos];
        if prerelease
            && ident.len() > 1
            && ident.as_bytes()[0] == b'0'
            && ident.bytes().all(|b| b.is_ascii_digit())
        {
            return Err(VersionError::LeadingZero { at: ident_start });
        }

        result.push(ident.to_string());
        if pos == end {
            break;
        }

        ident_start = pos + 1;
        if ident_start == end {
            return Err(VersionError::EmptyIdentifier { at: ident_start });
        }
    }

    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0;
    let major = parse_core_field(s, &mut pos)?;
    if pos >= s.len() {
        return Err(VersionError::MissingCore { at: pos });
    }
    if s.as_bytes()[pos] != b'.' {
        return if matches!(s.as_bytes()[pos], b'-' | b'+') {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let minor = parse_core_field(s, &mut pos)?;
    if pos >= s.len() {
        return Err(VersionError::MissingCore { at: pos });
    }
    if s.as_bytes()[pos] != b'.' {
        return if matches!(s.as_bytes()[pos], b'-' | b'+') {
            Err(VersionError::MissingCore { at: pos })
        } else {
            Err(VersionError::InvalidChar { at: pos })
        };
    }
    pos += 1;

    let patch = parse_core_field(s, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if pos < s.len() {
        match s.as_bytes()[pos] {
            b'-' => {
                let pre_start = pos + 1;
                let build_pos = s.as_bytes()[pre_start..]
                    .iter()
                    .position(|&b| b == b'+')
                    .map(|i| pre_start + i)
                    .unwrap_or(s.len());
                pre = parse_identifiers(s, pre_start, build_pos, true)?;
                pos = build_pos;
            }
            b'+' => {}
            _ => return Err(VersionError::InvalidChar { at: pos }),
        }
    }

    if pos < s.len() {
        if s.as_bytes()[pos] != b'+' {
            return Err(VersionError::InvalidChar { at: pos });
        }
        let build_start = pos + 1;
        build = parse_identifiers(s, build_start, s.len(), false)?;
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn numeric_identifier(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_identifiers(a: &str, b: &str) -> Ordering {
    let an = numeric_identifier(a);
    let bn = numeric_identifier(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a
            .len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let order = compare_identifiers(x, y);
                    if order != Ordering::Equal {
                        return order;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

fn is_range_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<Token<'_>> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut pos = 0;
    while pos < bytes.len() {
        while pos < bytes.len() && is_range_ws(bytes[pos]) {
            pos += 1;
        }
        if pos == bytes.len() {
            break;
        }
        let start = pos;
        while pos < bytes.len() && !is_range_ws(bytes[pos]) {
            pos += 1;
        }
        result.push(Token {
            text: &s[start..pos],
            start,
        });
    }
    result
}

fn partial_operand(text: &str, at: usize) -> Result<Partial, RangeError> {
    let has_suffix = text.contains('-') || text.contains('+');
    let fields: Vec<&str> = text.split('.').collect();
    let has_three_nonwild = fields.len() >= 3
        && !fields.iter().any(|f| matches!(*f, "*" | "x" | "X"));

    if has_suffix || has_three_nonwild {
        let v = version_at(text, at)?;
        return Ok(Partial {
            values: [v.major, v.minor, v.patch],
            numeric_count: 3,
            full: Some(v),
        });
    }

    if fields.is_empty() || fields.len() > 3 {
        return Err(RangeError::Syntax { at });
    }

    let mut values = [0u64; 3];
    let mut numeric_count = 0usize;
    let mut wildcard_seen = false;

    for (i, field) in fields.iter().enumerate() {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if matches!(*field, "*" | "x" | "X") {
            wildcard_seen = true;
            continue;
        }

        if wildcard_seen
            || !field.bytes().all(|b| b.is_ascii_digit())
            || (field.len() > 1 && field.as_bytes()[0] == b'0')
        {
            return Err(RangeError::Syntax { at });
        }

        let mut value = 0u64;
        for b in field.bytes() {
            value = value
                .checked_mul(10)
                .and_then(|n| n.checked_add((b - b'0') as u64))
                .ok_or(RangeError::Syntax { at })?;
        }
        values[i] = value;
        numeric_count += 1;
    }

    Ok(Partial {
        values,
        numeric_count,
        full: None,
    })
}

fn increment(value: u64, at: usize) -> Result<u64, RangeError> {
    value.checked_add(1).ok_or(RangeError::Syntax { at })
}

fn partial_lower(p: &Partial) -> Version {
    let mut v = endpoint(p.values[0], p.values[1], p.values[2]);
    if let Some(full) = &p.full {
        v = full.clone();
    }
    v
}

fn add_partial_expansion(
    arm: &mut Arm,
    p: Partial,
    style: Expansion,
    at: usize,
) -> Result<(), RangeError> {
    let k = p.numeric_count;
    if matches!(style, Expansion::Caret | Expansion::Tilde) && k == 0 {
        return Err(RangeError::Syntax { at });
    }

    if let Some(full) = p.full {
        arm.anchors.extend(anchor_for(&full));
        match style {
            Expansion::Bare => arm.predicates.push(Predicate::Compare {
                op: CompareOp::Eq,
                version: full,
            }),
            Expansion::Caret | Expansion::Tilde => {
                let lower = full.clone();
                let upper = upper_for(&p.values, k, style, at)?;
                arm.predicates.push(Predicate::Interval {
                    lower: Some((lower, true)),
                    upper: Some((upper, false)),
                });
            }
        }
        return Ok(());
    }

    if k == 0 {
        return Ok(());
    }
    let lower = partial_lower(&p);
    let upper = match style {
        Expansion::Bare if k == 3 => {
            arm.predicates.push(Predicate::Compare {
                op: CompareOp::Eq,
                version: lower,
            });
            return Ok(());
        }
        Expansion::Bare => upper_for(&p.values, k, Expansion::Bare, at)?,
        Expansion::Caret | Expansion::Tilde => upper_for(&p.values, k, style, at)?,
    };

    arm.predicates.push(Predicate::Interval {
        lower: Some((lower, true)),
        upper: Some((upper, false)),
    });
    Ok(())
}

#[derive(Clone, Copy)]
enum Expansion {
    Bare,
    Caret,
    Tilde,
}

fn upper_for(
    values: &[u64; 3],
    k: usize,
    style: Expansion,
    at: usize,
) -> Result<Version, RangeError> {
    let [major, minor, patch] = *values;
    let (m, n, p) = match style {
        Expansion::Bare => match k {
            1 => (increment(major, at)?, 0, 0),
            2 => (major, increment(minor, at)?, 0),
            _ => (major, minor, increment(patch, at)?),
        },
        Expansion::Caret => {
            if major > 0 || k == 1 {
                (increment(major, at)?, 0, 0)
            } else if minor > 0 || k == 2 {
                (major, increment(minor, at)?, 0)
            } else {
                (major, minor, increment(patch, at)?)
            }
        }
        Expansion::Tilde => {
            if k == 1 {
                (increment(major, at)?, 0, 0)
            } else {
                (major, increment(minor, at)?, 0)
            }
        }
    };
    Ok(endpoint(m, n, p))
}

fn anchor_for(v: &Version) -> Vec<(u64, u64, u64)> {
    if v.pre.is_empty() {
        Vec::new()
    } else {
        vec![tuple(v)]
    }
}

fn parse_comparator(token: Token<'_>, arm: &mut Arm) -> Result<(), RangeError> {
    let bytes = token.text.as_bytes();
    let (op, op_len) = match bytes.first().copied() {
        Some(b'=') => (Some(CompareOp::Eq), 1),
        Some(b'<') if bytes.get(1) == Some(&b'=') => (Some(CompareOp::Le), 2),
        Some(b'<') => (Some(CompareOp::Lt), 1),
        Some(b'>') if bytes.get(1) == Some(&b'=') => (Some(CompareOp::Ge), 2),
        Some(b'>') => (Some(CompareOp::Gt), 1),
        _ => (None, 0),
    };
    let operand = &token.text[op_len..];
    let operand_at = token.start + op_len;
    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
        return Err(RangeError::Syntax { at: token.start });
    }
    let version = version_at(operand, operand_at)?;
    arm.anchors.extend(anchor_for(&version));
    arm.predicates.push(Predicate::Compare {
        op: op.unwrap_or(CompareOp::Eq),
        version,
    });
    Ok(())
}

fn parse_nonhyphen_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {

    };

    for token in tokens {
        let bytes = token.text.as_bytes();
        match bytes.first().copied() {
            Some(b'<') | Some(b'>') | Some(b'=') => parse_comparator(*token, &mut arm)?,
            Some(b'^') | Some(b'~') => {
                if token.text.len() == 1 {
                    return Err(RangeError::Syntax { at: token.start });
                }
                let style = if bytes[0] == b'^' {
                    Expansion::Caret
                } else {
                    Expansion::Tilde
                };
                let operand = &token.text[1..];
                let operand_at = token.start + 1;
                let p = partial_operand(operand, operand_at)?;
                add_partial_expansion(&mut arm, p, style, operand_at)?;
            }
            Some(b'-') => return Err(RangeError::Syntax { at: token.start }),
            Some(b',') | Some(b'(') | Some(b')') | Some(b'&') => {
                return Err(RangeError::Syntax { at: token.start })
            }
            Some(b) if b.is_ascii_digit() || matches!(b, b'*' | b'x' | b'X') => {
                let p = partial_operand(token.text, token.start)?;
                add_partial_expansion(&mut arm, p, Expansion::Bare, token.start)?;
            }
            _ => return Err(RangeError::Syntax { at: token.start }),
        }
    }

    Ok(arm)
}

fn parse_hyphen_arm(tokens: &[Token<'_>], minus_index: usize) -> Result<Arm, RangeError> {
    if tokens.len() != 3 || minus_index != 1 || tokens[1].text != "-" {
        return Err(RangeError::Syntax {
            at: tokens[minus_index].start,
        });
    }

    let lower = partial_operand(tokens[0].text, tokens[0].start)?;
    let upper = partial_operand(tokens[2].text, tokens[2].start)?;

    if lower.numeric_count == 0 {
        return Err(RangeError::Syntax { at: tokens[0].start });
    }
    if upper.numeric_count == 0 {
        return Err(RangeError::Syntax { at: tokens[2].start });
    }

    let lower_version = partial_lower(&lower);
    let upper_bound = if let Some(full) = upper.full.as_ref() {
        full.clone()
    } else if upper.numeric_count == 3 {
        partial_lower(&upper)
    } else {
        upper_for(
            &upper.values,
            upper.numeric_count,
            Expansion::Bare,
            tokens[2].start,
        )?
    };

    let mut arm = Arm {
        predicates: vec![Predicate::Interval {
            lower: Some((lower_version, true)),
            upper: Some((
                upper_bound,
                upper.full.is_some() || upper.numeric_count == 3,
            )),
        }],
        anchors: Vec::new(),
    };
    if let Some(full) = lower.full {
        arm.anchors.extend(anchor_for(&full));
    }
    if let Some(full) = upper.full {
        arm.anchors.extend(anchor_for(&full));
    }
    Ok(arm)
}

fn parse_arm(tokens: &[Token<'_>]) -> Result<Arm, RangeError> {
    if let Some(i) = tokens.iter().position(|t| t.text == "-") {
        parse_hyphen_arm(tokens, i)
    } else {
        parse_nonhyphen_arm(tokens)
    }
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut arms = Vec::new();
    let mut arm_tokens: Vec<Token<'_>> = Vec::new();
    let mut i = 0usize;

    while i < tokens.len() {
        let token = tokens[i];
        if token.text == "||" {
            if arm_tokens.is_empty() {
                return Err(RangeError::Empty { at: token.start });
            }
            arms.push(parse_arm(&arm_tokens)?);
            arm_tokens.clear();
            i += 1;
            if i == tokens.len() {
                return Err(RangeError::Empty { at: s.len() });
            }
            continue;
        }
        arm_tokens.push(token);
        i += 1;
    }

    if arm_tokens.is_empty() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(parse_arm(&arm_tokens)?);
    Ok(Range { arms })
}

fn comparator_matches(v: &Version, op: CompareOp, target: &Version) -> bool {
    let order = compare_versions(v, target);
    match op {
        CompareOp::Eq => order == Ordering::Equal,
        CompareOp::Lt => order == Ordering::Less,
        CompareOp::Le => order != Ordering::Greater,
        CompareOp::Gt => order == Ordering::Greater,
        CompareOp::Ge => order != Ordering::Less,
    }
}

fn arm_matches(v: &Version, arm: &Arm) -> bool {
    if !v.pre.is_empty() && !arm.anchors.contains(&tuple(v)) {
        return false;
    }

    arm.predicates.iter().all(|predicate| match predicate {
        Predicate::Compare { op, version } => comparator_matches(v, *op, version),
        Predicate::Interval { lower, upper } => {
            let lower_ok = lower.as_ref().is_none_or(|(bound, inclusive)| {
                let cmp = compare_versions(v, bound);
                cmp == Ordering::Greater || (*inclusive && cmp == Ordering::Equal)
            });
            let upper_ok = upper.as_ref().is_none_or(|(bound, inclusive)| {
                let cmp = compare_versions(v, bound);
                cmp == Ordering::Less || (*inclusive && cmp == Ordering::Equal)
            });
            lower_ok && upper_ok
        }
    })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| arm_matches(v, arm))
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    want_max: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, candidate) in candidates.iter().enumerate() {
        let version = parse_version(candidate).map_err(|error| CandidateError { index, error })?;
        parsed.push(version);
    }

    let mut best: Option<(usize, usize)> = None;
    for (index, version) in parsed.iter().enumerate() {
        if !matches(version, r) {
            continue;
        }
        match best {
            None => best = Some((index, index)),
            Some((best_index, _)) => {
                let order = compare_versions(version, &parsed[best_index]);
                if (want_max && order == Ordering::Greater)
                    || (!want_max && order == Ordering::Less)
                {
                    best = Some((index, index));
                }
            }
        }
    }

    Ok(best.map(|(index, _)| candidates[index].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte_at(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn parse_core(s: &str, pos: &mut usize) -> Result<u64, VersionError> {
    let start = *pos;
    let bytes = s.as_bytes();
    while *pos < bytes.len() && bytes[*pos].is_ascii_digit() {
        *pos += 1;
    }
    if *pos == start {
        return Err(match byte_at(s, start) {
            None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at: start },
            _ => VersionError::InvalidChar { at: start },
        });
    }
    if *pos - start > 1 && bytes[start] == b'0' {
        return Err(VersionError::LeadingZero { at: start });
    }
    s[start..*pos]
        .parse::<u64>()
        .map_err(|_| VersionError::Overflow { at: start })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    pre: bool,
) -> Result<Vec<String>, VersionError> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut id_start = start;
    loop {
        let mut i = id_start;
        while i < end && bytes[i] != b'.' {
            if !bytes[i].is_ascii_alphanumeric() && bytes[i] != b'-' {
                return Err(VersionError::InvalidChar { at: i });
            }
            i += 1;
        }
        if i == id_start {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        if pre {
            let numeric = bytes[id_start..i].iter().all(u8::is_ascii_digit);
            if numeric && i - id_start > 1 && bytes[id_start] == b'0' {
                return Err(VersionError::LeadingZero { at: id_start });
            }
        }
        result.push(s[id_start..i].to_owned());
        if i == end {
            break;
        }
        id_start = i + 1;
    }
    Ok(result)
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }
    let mut pos = 0;
    let major = parse_core(s, &mut pos)?;
    if byte_at(s, pos) != Some(b'.') {
        return Err(match byte_at(s, pos) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;
    let minor = parse_core(s, &mut pos)?;
    if byte_at(s, pos) != Some(b'.') {
        return Err(match byte_at(s, pos) {
            None | Some(b'-' | b'+') => VersionError::MissingCore { at: pos },
            _ => VersionError::InvalidChar { at: pos },
        });
    }
    pos += 1;
    let patch = parse_core(s, &mut pos)?;

    let mut pre = Vec::new();
    let mut build = Vec::new();
    match byte_at(s, pos) {
        None => {}
        Some(b'-') => {
            let pre_start = pos + 1;
            let build_mark = s.as_bytes()[pre_start..]
                .iter()
                .position(|&b| b == b'+')
                .map(|n| pre_start + n);
            let pre_end = build_mark.unwrap_or(s.len());
            pre = parse_identifiers(s, pre_start, pre_end, true)?;
            if let Some(mark) = build_mark {
                build = parse_identifiers(s, mark + 1, s.len(), false)?;
            }
        }
        Some(b'+') => {
            build = parse_identifiers(s, pos + 1, s.len(), false)?;
        }
        _ => return Err(VersionError::InvalidChar { at: pos }),
    }

    Ok(Version {
        major,
        minor,
        patch,
        pre,
        build,
    })
}

fn is_numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_id(a: &str, b: &str) -> Ordering {
    match (is_numeric_id(a), is_numeric_id(b)) {
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_id(x, y);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Constraint>>,
}

#[derive(Clone, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

#[derive(Clone, Debug)]
struct Constraint {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Debug)]
struct Arm {
    constraints: Vec<Constraint>,
    anchors: Vec<(u64, u64, u64)>,
}

#[derive(Clone, Debug)]
enum ParsedOperand {
    Full(Version),
    Partial {
        fields: Vec<Option<u64>>,
        numeric_count: usize,
    },
}

fn is_ws(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokenize(s: &str) -> Vec<(usize, &str)> {
    let bytes = s.as_bytes();
    let mut result = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && is_ws(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !is_ws(bytes[i]) {
            i += 1;
        }
        result.push((start, &s[start..i]));
    }
    result
}

fn version_err(at: usize, error: VersionError) -> RangeError {
    RangeError::Version { at, error }
}

fn is_wildcard(s: &str) -> bool {
    matches!(s, "*" | "x" | "X")
}

fn parse_operand(
    text: &str,
    at: usize,
    forced_full: bool,
) -> Result<ParsedOperand, RangeError> {
    let fields: Vec<&str> = text.split('.').collect();
    let has_suffix = text.contains('-') || text.contains('+');
    let all_wild = fields.iter().all(|f| is_wildcard(f));
    if forced_full || has_suffix || (fields.len() >= 3 && !fields.iter().any(|f| is_wildcard(f))) {
        return parse_version(text)
            .map(ParsedOperand::Full)
            .map_err(|e| version_err(at, e));
    }
    if fields.is_empty() || fields.len() > 3 || text.is_empty() {
        return Err(RangeError::Syntax { at });
    }
    let mut parsed = Vec::new();
    let mut wildcard_seen = false;
    let mut numeric_count = 0;
    for field in fields {
        if field.is_empty() {
            return Err(RangeError::Syntax { at });
        }
        if is_wildcard(field) {
            wildcard_seen = true;
            parsed.push(None);
        } else {
            if wildcard_seen
                || !field.bytes().all(|b| b.is_ascii_digit())
                || (field.len() > 1 && field.starts_with('0'))
            {
                return Err(RangeError::Syntax { at });
            }
            let value = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            numeric_count += 1;
            parsed.push(Some(value));
        }
    }
    if all_wild {
        numeric_count = 0;
    }
    Ok(ParsedOperand::Partial {
        fields: parsed,
        numeric_count,
    })
}

fn padded(fields: &[Option<u64>]) -> Version {
    Version {
        major: fields.first().copied().flatten().unwrap_or(0),
        minor: fields.get(1).copied().flatten().unwrap_or(0),
        patch: fields.get(2).copied().flatten().unwrap_or(0),
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn upper_version(base: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut v = Version {
        major: base.major,
        minor: base.minor,
        patch: base.patch,
        pre: vec!["0".to_owned()],
        build: Vec::new(),
    };
    match field {
        0 => {
            v.major = v.major.checked_add(1).ok_or(RangeError::Syntax { at })?;
            v.minor = 0;
            v.patch = 0;
        }
        1 => {
            v.minor = v.minor.checked_add(1).ok_or(RangeError::Syntax { at })?;
            v.patch = 0;
        }
        _ => {
            v.patch = v.patch.checked_add(1).ok_or(RangeError::Syntax { at })?;
        }
    }
    Ok(v)
}

fn push_constraint(out: &mut Vec<Constraint>, op: Op, version: Version, anchor: bool) {
    out.push(Constraint { op, version, anchor });
}

fn partial_lower(p: &ParsedOperand) -> (Version, usize) {
    match p {
        ParsedOperand::Full(v) => (v.clone(), 3),
        ParsedOperand::Partial { fields, numeric_count } => (padded(fields), *numeric_count),
    }
}

fn add_bare(
    out: &mut Vec<Constraint>,
    text: &str,
    at: usize,
) -> Result<(), RangeError> {
    match parse_operand(text, at, false)? {
        ParsedOperand::Full(v) => {
            let anchor = !v.pre.is_empty();
            push_constraint(out, Op::Eq, v, anchor);
        }
        ParsedOperand::Partial { fields, numeric_count } => {
            if numeric_count == 0 {
                return Ok(());
            }
            let lo = padded(&fields);
            push_constraint(out, Op::Ge, lo, false);
            if numeric_count < 3 {
                let field = if numeric_count == 1 { 0 } else { 1 };
                let hi = upper_version(&padded(&fields), field, at)?;
                push_constraint(out, Op::Lt, hi, false);
            }
        }
    }
    Ok(())
}

fn add_caret_tilde(
    out: &mut Vec<Constraint>,
    text: &str,
    at: usize,
    caret: bool,
) -> Result<(), RangeError> {
    let operand_at = at + 1;
    let operand = &text[1..];
    let parsed = parse_operand(operand, operand_at, false)?;
    let (lo, k) = partial_lower(&parsed);
    if k == 0 {
        return Err(RangeError::Syntax { at });
    }
    let anchor = !lo.pre.is_empty();
    push_constraint(out, Op::Ge, lo.clone(), anchor);

    let field = if caret {
        if lo.major > 0 || k == 1 {
            0
        } else if lo.minor > 0 || k == 2 {
            1
        } else {
            2
        }
    } else if k == 1 {
        0
    } else {
        1
    };
    let hi = upper_version(&lo, field, operand_at)?;
    push_constraint(out, Op::Lt, hi, false);
    Ok(())
}

fn parse_hyphen_arm(tokens: &[(usize, &str)]) -> Result<Vec<Constraint>, RangeError> {
    let minus = tokens.iter().find(|(_, t)| *t == "-").unwrap().0;
    if tokens.len() != 3 || tokens[1].1 != "-" {
        return Err(RangeError::Syntax { at: minus });
    }
    let (lower_at, lower_text) = tokens[0];
    let (upper_at, upper_text) = tokens[2];
    let lower = parse_operand(lower_text, lower_at, false)?;
    let upper = parse_operand(upper_text, upper_at, false)?;
    let (lo, lk) = partial_lower(&lower);
    let (up, uk) = partial_lower(&upper);
    if lk == 0 {
        return Err(RangeError::Syntax { at: lower_at });
    }
    if uk == 0 {
        return Err(RangeError::Syntax { at: upper_at });
    }
    let mut out = Vec::new();
    push_constraint(&mut out, Op::Ge, lo.clone(), !lo.pre.is_empty());
    match upper {
        ParsedOperand::Full(v) => push_constraint(&mut out, Op::Le, v.clone(), !v.pre.is_empty()),
        ParsedOperand::Partial { .. } => {
            let hi = if uk == 1 {
                upper_version(&up, 0, upper_at)?
            } else {
                upper_version(&up, 1, upper_at)?
            };
            push_constraint(&mut out, Op::Lt, hi, false);
        }
    }
    Ok(out)
}

fn op_match(v: &Version, c: &Constraint) -> bool {
    match c.op {
        Op::Eq => compare_versions(v, &c.version) == Ordering::Equal,
        Op::Lt => compare_versions(v, &c.version) == Ordering::Less,
        Op::Le => compare_versions(v, &c.version) != Ordering::Greater,
        Op::Gt => compare_versions(v, &c.version) == Ordering::Greater,
        Op::Ge => compare_versions(v, &c.version) != Ordering::Less,
    }
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let tokens = tokenize(s);
    if tokens.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }

    let mut split_arms: Vec<Vec<(usize, &str)>> = vec![Vec::new()];
    let mut separators = Vec::new();
    for tok in tokens {
        if tok.1 == "||" {
            separators.push(tok.0);
            split_arms.push(Vec::new());
        } else {
            split_arms.last_mut().unwrap().push(tok);
        }
    }
    for i in 0..split_arms.len() {
        if split_arms[i].is_empty() {
            let at = if i + 1 < split_arms.len() {
                separators[i]
            } else {
                s.len()
            };
            return Err(RangeError::Empty { at });
        }
    }

    let mut arms = Vec::new();
    for tokens in split_arms {
        let has_minus = tokens.iter().any(|(_, t)| *t == "-");
        let constraints = if has_minus {
            parse_hyphen_arm(&tokens)?
        } else {
            let mut out = Vec::new();
            for &(at, token) in &tokens {
                if token.is_empty() {
                    return Err(RangeError::Syntax { at });
                }
                if token.starts_with('^') || token.starts_with('~') {
                    add_caret_tilde(&mut out, token, at, token.starts_with('^'))?;
                    continue;
                }
                let (op, operand_start, operand, forced_full) = if token.starts_with(">=") {
                    (Some(Op::Ge), at + 2, &token[2..], true)
                } else if token.starts_with("<=") {
                    (Some(Op::Le), at + 2, &token[2..], true)
                } else if token.starts_with('>') {
                    (Some(Op::Gt), at + 1, &token[1..], true)
                } else if token.starts_with('<') {
                    (Some(Op::Lt), at + 1, &token[1..], true)
                } else if token.starts_with('=') {
                    (Some(Op::Eq), at + 1, &token[1..], true)
                } else {
                    (None, at, token, false)
                };
                if let Some(op) = op {
                    if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                        return Err(RangeError::Syntax { at });
                    }
                    let v = match parse_operand(operand, operand_start, forced_full)? {
                        ParsedOperand::Full(v) => v,
                        _ => unreachable!(),
                    };
                    let anchor = !v.pre.is_empty();
                    push_constraint(&mut out, op, v, anchor);
                    continue;
                }

                let first = token.as_bytes()[0];
                if !(first.is_ascii_digit()
                    || is_wildcard(token)
                    || first == b'*'
                    || first == b'x'
                    || first == b'X')
                {
                    return Err(RangeError::Syntax { at });
                }
                add_bare(&mut out, token, at)?;
            }
            out
        };
        arms.push(constraints);
    }
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        let admission = !v.pre.is_empty()
            && arm.iter().any(|c| {
                c.anchor
                    && !c.version.pre.is_empty()
                    && (c.version.major, c.version.minor, c.version.patch)
                        == (v.major, v.minor, v.patch)
            });
        if !v.pre.is_empty() && !admission {
            return false;
        }
        arm.iter().all(|c| op_match(v, c))
    })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct CandidateError {
    pub index: usize,
    pub error: VersionError,
}

fn satisfying_extreme(
    candidates: &[String],
    r: &Range,
    greatest: bool,
) -> Result<Option<String>, CandidateError> {
    let mut parsed = Vec::with_capacity(candidates.len());
    for (index, s) in candidates.iter().enumerate() {
        match parse_version(s) {
            Ok(v) => parsed.push(v),
            Err(error) => return Err(CandidateError { index, error }),
        }
    }
    let mut best: Option<usize> = None;
    for (i, v) in parsed.iter().enumerate() {
        if !matches(v, r) {
            continue;
        }
        if let Some(j) = best {
            let ord = compare_versions(v, &parsed[j]);
            if (greatest && ord == Ordering::Greater) || (!greatest && ord == Ordering::Less) {
                best = Some(i);
            }
        } else {
            best = Some(i);
        }
    }
    Ok(best.map(|i| candidates[i].clone()))
}

pub fn max_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, true)
}

pub fn min_satisfying(
    candidates: &[String],
    r: &Range,
) -> Result<Option<String>, CandidateError> {
    satisfying_extreme(candidates, r, false)
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Version {
    pub major: u64,
    pub minor: u64,
    pub patch: u64,
    pub pre: Vec<String>,
    pub build: Vec<String>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum VersionError {
    Empty,
    MissingCore { at: usize },
    LeadingZero { at: usize },
    Overflow { at: usize },
    EmptyIdentifier { at: usize },
    InvalidChar { at: usize },
}

fn byte(s: &str, i: usize) -> Option<u8> {
    s.as_bytes().get(i).copied()
}

fn missing_or_invalid(s: &str, at: usize) -> VersionError {
    match byte(s, at) {
        None | Some(b'.' | b'-' | b'+') => VersionError::MissingCore { at },
        _ => VersionError::InvalidChar { at },
    }
}

pub fn parse_version(s: &str) -> Result<Version, VersionError> {
    if s.is_empty() {
        return Err(VersionError::Empty);
    }

    let mut pos = 0usize;
    let mut nums = [0u64; 3];

    for field in 0..3 {
        let start = pos;
        while matches!(byte(s, pos), Some(b'0'..=b'9')) {
            pos += 1;
        }
        if pos == start {
            return Err(missing_or_invalid(s, start));
        }

        let digits = &s[start..pos];
        if digits.len() > 1 && digits.as_bytes()[0] == b'0' {
            return Err(VersionError::LeadingZero { at: start });
        }
        nums[field] = digits
            .parse::<u64>()
            .map_err(|_| VersionError::Overflow { at: start })?;

        if field < 2 {
            match byte(s, pos) {
                Some(b'.') => pos += 1,
                _ => return Err(missing_or_invalid(s, pos)),
            }
        }
    }

    let mut pre = Vec::new();
    let mut build = Vec::new();

    if let Some(b'-') = byte(s, pos) {
        pos += 1;
        let pre_start = pos;
        let plus = s[pos..].find('+').map(|n| pos + n).unwrap_or(s.len());
        parse_identifiers(s, pre_start, plus, true, &mut pre)?;
        pos = plus;
    }

    if let Some(b'+') = byte(s, pos) {
        pos += 1;
        parse_identifiers(s, pos, s.len(), false, &mut build)?;
        pos = s.len();
    }

    if pos != s.len() {
        return Err(VersionError::InvalidChar { at: pos });
    }

    Ok(Version {
        major: nums[0],
        minor: nums[1],
        patch: nums[2],
        pre,
        build,
    })
}

fn parse_identifiers(
    s: &str,
    start: usize,
    end: usize,
    prerelease: bool,
    out: &mut Vec<String>,
) -> Result<(), VersionError> {
    if start == end {
        return Err(VersionError::EmptyIdentifier { at: start });
    }
    let mut id_start = start;
    loop {
        let id_end = s[id_start..end]
            .find('.')
            .map(|n| id_start + n)
            .unwrap_or(end);
        if id_start == id_end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
        let id = &s[id_start..id_end];
        for (offset, b) in id.bytes().enumerate() {
            if !matches!(b, b'A'..=b'Z' | b'a'..=b'z' | b'0'..=b'9' | b'-') {
                return Err(VersionError::InvalidChar {
                    at: id_start + offset,
                });
            }
        }
        if prerelease
            && id.bytes().all(|b| b.is_ascii_digit())
            && id.len() > 1
            && id.as_bytes()[0] == b'0'
        {
            return Err(VersionError::LeadingZero { at: id_start });
        }
        out.push(id.to_string());
        if id_end == end {
            break;
        }
        id_start = id_end + 1;
        if id_start == end {
            return Err(VersionError::EmptyIdentifier { at: id_start });
        }
    }
    Ok(())
}

fn numeric_id(s: &str) -> bool {
    !s.is_empty() && s.bytes().all(|b| b.is_ascii_digit())
}

fn compare_id(a: &str, b: &str) -> Ordering {
    let an = numeric_id(a);
    let bn = numeric_id(b);
    match (an, bn) {
        (true, false) => Ordering::Less,
        (false, true) => Ordering::Greater,
        (true, true) => a.len().cmp(&b.len()).then_with(|| a.as_bytes().cmp(b.as_bytes())),
        (false, false) => a.as_bytes().cmp(b.as_bytes()),
    }
}

pub fn compare_versions(a: &Version, b: &Version) -> Ordering {
    (a.major, a.minor, a.patch)
        .cmp(&(b.major, b.minor, b.patch))
        .then_with(|| match (a.pre.is_empty(), b.pre.is_empty()) {
            (true, true) => Ordering::Equal,
            (true, false) => Ordering::Greater,
            (false, true) => Ordering::Less,
            (false, false) => {
                for (x, y) in a.pre.iter().zip(&b.pre) {
                    let ord = compare_id(x, y);
                    if ord != Ordering::Equal {
                        return ord;
                    }
                }
                a.pre.len().cmp(&b.pre.len())
            }
        })
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum RangeError {
    Empty { at: usize },
    Syntax { at: usize },
    Version { at: usize, error: VersionError },
}

#[derive(Clone, Debug)]
pub struct Range {
    arms: Vec<Vec<Predicate>>,
}

#[derive(Clone, Debug)]
struct Predicate {
    op: Op,
    version: Version,
    anchor: bool,
}

#[derive(Clone, Copy, Debug)]
enum Op {
    Eq,
    Lt,
    Le,
    Gt,
    Ge,
}

fn whitespace(b: u8) -> bool {
    matches!(b, b' ' | b'\t' | b'\r' | b'\n')
}

fn tokens(s: &str, base: usize) -> Vec<(String, usize)> {
    let bytes = s.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        while i < bytes.len() && whitespace(bytes[i]) {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && !whitespace(bytes[i]) {
            i += 1;
        }
        out.push((s[start..i].to_string(), base + start));
    }
    out
}

fn cmp_pred(op: Op, candidate: &Version, bound: &Version) -> bool {
    match op {
        Op::Eq => compare_versions(candidate, bound) == Ordering::Equal,
        Op::Lt => compare_versions(candidate, bound) == Ordering::Less,
        Op::Le => compare_versions(candidate, bound) != Ordering::Greater,
        Op::Gt => compare_versions(candidate, bound) == Ordering::Greater,
        Op::Ge => compare_versions(candidate, bound) != Ordering::Less,
    }
}

fn add_pred(out: &mut Vec<Predicate>, op: Op, version: Version, anchor: bool) {
    out.push(Predicate { op, version, anchor });
}

fn full_at(s: &str, at: usize) -> Result<Version, RangeError> {
    parse_version(s).map_err(|error| RangeError::Version { at, error })
}

#[derive(Clone)]
struct Partial {
    nums: Vec<u64>,
    full: Option<Version>,
    k: usize,
}

fn parse_partial(s: &str, at: usize) -> Result<Partial, RangeError> {
    let is_full = s.contains(['-', '+'])
        || (s.split('.').count() >= 3
            && !s.split('.').any(|f| matches!(f, "*" | "x" | "X")));
    if is_full {
        return Ok(Partial {
            nums: Vec::new(),
            full: Some(full_at(s, at)?),
            k: 3,
        });
    }
    let fields: Vec<&str> = s.split('.').collect();
    if fields.is_empty() || fields.len() > 3 || fields.iter().any(|f| f.is_empty()) {
        return Err(RangeError::Syntax { at });
    }
    let mut nums = Vec::new();
    let mut wildcard = false;
    for field in fields {
        if matches!(field, "*" | "x" | "X") {
            wildcard = true;
        } else {
            if wildcard
                || !field.bytes().all(|b| b.is_ascii_digit())
                || (field.len() > 1 && field.as_bytes()[0] == b'0')
            {
                return Err(RangeError::Syntax { at });
            }
            let n = field.parse::<u64>().map_err(|_| RangeError::Syntax { at })?;
            nums.push(n);
        }
    }
    Ok(Partial {
        k: nums.len(),
        nums,
        full: None,
    })
}

fn padded(p: &Partial) -> Version {
    if let Some(v) = &p.full {
        return v.clone();
    }
    let mut n = [0u64; 3];
    for (i, x) in p.nums.iter().enumerate() {
        n[i] = *x;
    }
    Version {
        major: n[0],
        minor: n[1],
        patch: n[2],
        pre: Vec::new(),
        build: Vec::new(),
    }
}

fn bump(v: &Version, field: usize, at: usize) -> Result<Version, RangeError> {
    let mut out = v.clone();
    let slot = match field {
        0 => &mut out.major,
        1 => &mut out.minor,
        _ => &mut out.patch,
    };
    *slot = slot.checked_add(1).ok_or(RangeError::Syntax { at })?;
    if field <= 1 {
        out.minor = if field == 0 { 0 } else { out.minor };
    }
    if field <= 2 {
        out.patch = if field <= 1 { 0 } else { out.patch };
    }
    out.pre = vec!["0".to_string()];
    out.build.clear();
    Ok(out)
}

fn add_partial_bare(out: &mut Vec<Predicate>, p: Partial, at: usize) -> Result<(), RangeError> {
    if let Some(v) = p.full {
        add_pred(out, Op::Eq, v.clone(), !v.pre.is_empty());
    } else if p.k == 1 || p.k == 2 {
        let low = padded(&p);
        let high = bump(&low, p.k - 1, at)?;
        add_pred(out, Op::Ge, low, false);
        add_pred(out, Op::Lt, high, false);
    }
    Ok(())
}

fn process_arm(ts: &[(String, usize)]) -> Result<Vec<Predicate>, RangeError> {
    let mut out = Vec::new();
    let hyphens: Vec<usize> = ts.iter().enumerate().filter(|(_, t)| t.0 == "-").map(|(i, _)| i).collect();
    if !hyphens.is_empty() {
        let h = hyphens[0];
        if h != 1 || ts.len() != 3 {
            return Err(RangeError::Syntax { at: ts[h].1 });
        }
        let (ls, la) = (&ts[0].0, ts[0].1);
        let (us, ua) = (&ts[2].0, ts[2].1);
        let lp = parse_partial(ls, la)?;
        let up = parse_partial(us, ua)?;
        if lp.k == 0 {
            return Err(RangeError::Syntax { at: la });
        }
        if up.k == 0 {
            return Err(RangeError::Syntax { at: ua });
        }
        let low = padded(&lp);
        add_pred(&mut out, Op::Ge, low, lp.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
        if let Some(v) = up.full {
            add_pred(&mut out, Op::Le, v.clone(), !v.pre.is_empty());
        } else {
            let upper = if up.k == 1 || up.k == 2 {
                bump(&padded(&up), up.k - 1, ua)?
            } else {
                padded(&up)
            };
            if up.k < 3 {
                add_pred(&mut out, Op::Lt, upper, false);
            } else {
                add_pred(&mut out, Op::Le, upper, false);
            }
        }
        return Ok(out);
    }

    for (token, start) in ts {
        let b = token.as_bytes();
        if b.is_empty() {
            return Err(RangeError::Syntax { at: *start });
        }
        if token == "||" {
            return Err(RangeError::Syntax { at: *start });
        }
        if matches!(b[0], b'<' | b'>' | b'=') {
            let (op, n) = if token.starts_with("<=") {
                (Op::Le, 2)
            } else if token.starts_with(">=") {
                (Op::Ge, 2)
            } else {
                (match b[0] {
                    b'<' => Op::Lt,
                    b'>' => Op::Gt,
                    _ => Op::Eq,
                }, 1)
            };
            let operand = &token[n..];
            if operand.is_empty() || !operand.as_bytes()[0].is_ascii_digit() {
                return Err(RangeError::Syntax { at: *start });
            }
            let v = full_at(operand, *start + n)?;
            add_pred(&mut out, op, v.clone(), !v.pre.is_empty());
            continue;
        }
        if matches!(b[0], b'^' | b'~') {
            let caret = b[0] == b'^';
            let operand = &token[1..];
            let at = *start + 1;
            let p = parse_partial(operand, at)?;
            if p.k == 0 {
                return Err(RangeError::Syntax { at: *start });
            }
            let low = padded(&p);
            let upper_field = if caret {
                if low.major > 0 || p.k == 1 { 0 }
                else if low.minor > 0 || p.k == 2 { 1 }
                else { 2 }
            } else if p.k == 1 { 0 } else { 1 };
            let high = bump(&low, upper_field, at)?;
            add_pred(&mut out, Op::Ge, low.clone(), p.full.as_ref().is_some_and(|v| !v.pre.is_empty()));
            add_pred(&mut out, Op::Lt, high, false);
            continue;
        }
        if !b[0].is_ascii_digit() && !matches!(b[0], b'*' | b'x' | b'X') {
            return Err(RangeError::Syntax { at: *start });
        }
        let p = parse_partial(token, *start)?;
        add_partial_bare(&mut out, p, *start)?;
    }
    Ok(out)
}

pub fn parse_range(s: &str) -> Result<Range, RangeError> {
    let ts = tokens(s, 0);
    if ts.is_empty() {
        return Err(RangeError::Empty { at: 0 });
    }
    let mut arms: Vec<Vec<Predicate>> = Vec::new();
    let mut start = 0;
    let mut i = 0;
    while i < ts.len() {
        if ts[i].0 == "||" {
            if i == start {
                return Err(RangeError::Empty { at: ts[i].1 });
            }
            arms.push(process_arm(&ts[start..i])?);
            start = i + 1;
        }
        i += 1;
    }
    if start == ts.len() {
        return Err(RangeError::Empty { at: s.len() });
    }
    arms.push(process_arm(&ts[start..])?);
    Ok(Range { arms })
}

pub fn matches(v: &Version, r: &Range) -> bool {
    r.arms.iter().any(|arm| {
        if !arm.iter().all(|p| cmp_pred(p.op, v, &p.version)) {
            return false;
        }
        if v.pre.is_empty() {
            return true;
        }
        arm.iter().any(|p| {
            p.anchor
                && (v.major, v.minor, v.patch)
                    == (p.version.major, p.version.minor, p.version.patch)
        })
    })
}
