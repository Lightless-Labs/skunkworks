#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Task {
    pub duration: u64,
    pub deps: Vec<usize>,
    pub release: u64,
    pub priority: i32,
    pub needs: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Run {
    pub start: u64,
    pub finish: u64,
    pub worker: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Schedule {
    pub runs: Vec<Run>,
    pub makespan: u64,
    pub trace: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Stage {
    Tail,
    Finish,
    Turnaround,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    BadDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    SelfDependency {
        task: usize,
        position: usize,
    },
    DuplicateDependency {
        task: usize,
        position: usize,
        value: usize,
    },
    Cycle {
        blocked: Vec<usize>,
    },
    Workers {
        count: usize,
    },
    Overflow {
        task: usize,
        stage: Stage,
    },
    ResourceShape {
        task: usize,
        expected: usize,
        actual: usize,
    },
    ResourceExceeded {
        task: usize,
        resource: usize,
        need: u32,
        capacity: u32,
    },
}

pub fn schedule(
    tasks: &[Task],
    cooldowns: &[u64],
    capacity: &[u32],
) -> Result<Schedule, Error> {
    let n = tasks.len();

    // Dependency validation is performed in the required task/list order.
    for (task_id, task) in tasks.iter().enumerate() {
        for (position, &value) in task.deps.iter().enumerate() {
            if value >= n {
                return Err(Error::BadDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
            if value == task_id {
                return Err(Error::SelfDependency {
                    task: task_id,
                    position,
                });
            }
            if task.deps[..position].contains(&value) {
                return Err(Error::DuplicateDependency {
                    task: task_id,
                    position,
                    value,
                });
            }
        }
    }

    if !(1..=64).contains(&cooldowns.len()) {
        return Err(Error::Workers {
            count: cooldowns.len(),
        });
    }

    for (task_id, task) in tasks.iter().enumerate() {
        if task.needs.len() != capacity.len() {
            return Err(Error::ResourceShape {
                task: task_id,
                expected: capacity.len(),
                actual: task.needs.len(),
            });
        }
        for (resource, (&need, &available)) in
            task.needs.iter().zip(capacity.iter()).enumerate()
        {
            if need > available {
                return Err(Error::ResourceExceeded {
                    task: task_id,
                    resource,
                    need,
                    capacity: available,
                });
            }
        }
    }

    let mut children = vec![Vec::<usize>::new(); n];
    let mut indegree = vec![0usize; n];
    for (task_id, task) in tasks.iter().enumerate() {
        indegree[task_id] = task.deps.len();
        for &predecessor in &task.deps {
            children[predecessor].push(task_id);
        }
    }

    // Kahn traversal with the smallest currently available ID first.
    let mut available = Vec::<usize>::new();
    for (id, &degree) in indegree.iter().enumerate() {
        if degree == 0 {
            available.push(id);
        }
    }
    let mut kahn_order = Vec::with_capacity(n);
    while !available.is_empty() {
        let smallest_pos = available
            .iter()
            .enumerate()
            .min_by_key(|(_, id)| **id)
            .map(|(position, _)| position)
            .unwrap();
        let id = available.swap_remove(smallest_pos);
        kahn_order.push(id);
        for &child in &children[id] {
            indegree[child] -= 1;
            if indegree[child] == 0 {
                available.push(child);
            }
        }
    }

    if kahn_order.len() != n {
        let mut removed = vec![false; n];
        for &id in &kahn_order {
            removed[id] = true;
        }
        return Err(Error::Cycle {
            blocked: (0..n).filter(|&id| !removed[id]).collect(),
        });
    }

    let mut tails = vec![0u64; n];
    for &id in kahn_order.iter().rev() {
        let max_child_tail = children[id]
            .iter()
            .map(|&child| tails[child])
            .max()
            .unwrap_or(0);
        tails[id] = tasks[id]
            .duration
            .checked_add(max_child_tail)
            .ok_or(Error::Overflow {
                task: id,
                stage: Stage::Tail,
            })?;
    }

    #[derive(Clone, Copy)]
    struct Running {
        task: usize,
        finish: u64,
        worker: usize,
    }

    let worker_count = cooldowns.len();
    let mut runs: Vec<Option<Run>> = vec![None; n];
    let mut started = vec![false; n];
    let mut completed = vec![false; n];
    let mut used = vec![0u64; capacity.len()];
    let mut worker_available = vec![0u64; worker_count];
    let mut running = Vec::<Running>::new();
    let mut trace = String::new();
    use std::fmt::Write as _;
    writeln!(
        &mut trace,
        "workers={};resources={}",
        worker_count,
        capacity.len()
    )
    .unwrap();

    let mut now = 0u64;
    let mut completion_count = 0usize;
    let mut makespan = 0u64;

    while completion_count < n {
        // Complete all tasks finishing now, ordered by task ID.
        let mut finishing: Vec<Running> = running
            .iter()
            .copied()
            .filter(|r| r.finish == now)
            .collect();
        finishing.sort_by_key(|r| r.task);
        if !finishing.is_empty() {
            running.retain(|r| r.finish != now);
            for r in finishing {
                for (resource, &need) in tasks[r.task].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[r.task] = true;
                completion_count += 1;
                makespan = makespan.max(r.finish);
                write_trace_event(&mut trace, now, "end", r.task, r.worker, &used);
            }
        }

        loop {
            let mut best: Option<usize> = None;
            for id in 0..n {
                if started[id]
                    || tasks[id].release > now
                    || !tasks[id].deps.iter().all(|&dep| completed[dep])
                {
                    continue;
                }
                let fits = tasks[id]
                    .needs
                    .iter()
                    .enumerate()
                    .all(|(r, &need)| {
                        u64::from(need) <= u64::from(capacity[r]) - used[r]
                    });
                if !fits {
                    continue;
                }
                let Some(current) = best else {
                    best = Some(id);
                    continue;
                };
                let a = (tails[id], tasks[id].priority, std::cmp::Reverse(id));
                let b = (
                    tails[current],
                    tasks[current].priority,
                    std::cmp::Reverse(current),
                );
                if a > b {
                    best = Some(id);
                }
            }

            let Some(id) = best else {
                break;
            };

            // Select the task first, then its cyclically preferred available worker.
            let worker = (0..worker_count)
                .map(|offset| (id % worker_count + offset) % worker_count)
                .find(|&w| worker_available[w] <= now);
            let Some(worker) = worker else {
                break;
            };

            let finish = now.checked_add(tasks[id].duration).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Finish,
            })?;
            let available_at = finish.checked_add(cooldowns[worker]).ok_or(Error::Overflow {
                task: id,
                stage: Stage::Turnaround,
            })?;

            started[id] = true;
            worker_available[worker] = available_at;
            for (resource, &need) in tasks[id].needs.iter().enumerate() {
                used[resource] += u64::from(need);
            }
            runs[id] = Some(Run {
                start: now,
                finish,
                worker,
            });
            write_trace_event(&mut trace, now, "start", id, worker, &used);

            if finish == now {
                // A zero-duration task completes before the next admission.
                for (resource, &need) in tasks[id].needs.iter().enumerate() {
                    used[resource] -= u64::from(need);
                }
                completed[id] = true;
                completion_count += 1;
                makespan = makespan.max(finish);
                write_trace_event(&mut trace, now, "end", id, worker, &used);
            } else {
                running.push(Running {
                    task: id,
                    finish,
                    worker,
                });
            }
        }

        if completion_count == n {
            break;
        }

        let mut next_time: Option<u64> = None;
        for r in &running {
            if r.finish > now {
                next_time = Some(next_time.map_or(r.finish, |t| t.min(r.finish)));
            }
        }
        for id in 0..n {
            if !started[id] && tasks[id].release > now {
                next_time = Some(next_time.map_or(tasks[id].release, |t| t.min(tasks[id].release)));
            }
        }
        for &available_at in &worker_available {
            if available_at > now {
                next_time = Some(next_time.map_or(available_at, |t| t.min(available_at)));
            }
        }

        // If an unfinished task is dependency/resource blocked, some running task
        // or future release/cooldown event must eventually permit progress.
        let Some(next) = next_time else {
            // Acyclic dependencies and valid individual demands guarantee progress
            // once all relevant events have occurred.
            unreachable!();
        };
        now = next;
    }

    writeln!(&mut trace, "makespan={}", makespan).unwrap();

    Ok(Schedule {
        runs: runs.into_iter().map(Option::unwrap).collect(),
        makespan,
        trace,
    })
}

fn write_trace_event(
    trace: &mut String,
    time: u64,
    kind: &str,
    task: usize,
    worker: usize,
    used: &[u64],
) {
    use std::fmt::Write as _;
    write!(trace, "{}|{}|{}|{}|", time, kind, task, worker).unwrap();
    if used.is_empty() {
        trace.push('-');
    } else {
        for (index, amount) in used.iter().enumerate() {
            if index != 0 {
                trace.push(',');
            }
            write!(trace, "{}", amount).unwrap();
        }
    }
    trace.push('\n');
}
