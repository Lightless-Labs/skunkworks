#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub struct Version {
    pub major: u32,
    pub minor: u32,
    pub patch: u32,
}

#[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord)]
pub enum Constraint {
    Exact(Version),
    AtLeast(Version),
    Before(Version),
    Caret(Version),
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Selection {
    pub package: String,
    pub version: Version,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Package {
    pub name: String,
    pub releases: Vec<Release>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Requirement {
    pub package: String,
    pub constraint: Constraint,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Release {
    pub version: Version,
    pub deps: Vec<Requirement>,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum ResolveError {
    InvalidName {
        site: String,
        name: String,
    },
    DuplicatePackage {
        package: String,
    },
    DuplicateVersion {
        package: String,
        version: Version,
    },
    Missing {
        package: String,
    },
    NoMatch {
        package: String,
        requirements: Vec<Constraint>,
    },
    Conflict {
        package: String,
        selected: Version,
        requirements: Vec<Constraint>,
    },
    Limit {
        attempts: usize,
    },
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || bytes.len() > 24 || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    if bytes.last() == Some(&b'-') {
        return false;
    }
    bytes
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'-')
}

fn validate_names(catalog: &[Package], roots: &[Requirement]) -> Result<(), ResolveError> {
    for (i, root) in roots.iter().enumerate() {
        if !valid_name(&root.package) {
            return Err(ResolveError::InvalidName {
                site: format!("root[{i}]"),
                name: root.package.clone(),
            });
        }
    }
    for (i, package) in catalog.iter().enumerate() {
        if !valid_name(&package.name) {
            return Err(ResolveError::InvalidName {
                site: format!("catalog[{i}]"),
                name: package.name.clone(),
            });
        }
        for (j, release) in package.releases.iter().enumerate() {
            for (k, dep) in release.deps.iter().enumerate() {
                if !valid_name(&dep.package) {
                    return Err(ResolveError::InvalidName {
                        site: format!("catalog[{i}].release[{j}].dep[{k}]"),
                        name: dep.package.clone(),
                    });
                }
            }
        }
    }
    Ok(())
}

fn version_text(version: Version) -> String {
    format!("{}.{}.{}", version.major, version.minor, version.patch)
}

fn constraint_text(constraint: Constraint) -> String {
    match constraint {
        Constraint::Exact(v) => format!("={}", version_text(v)),
        Constraint::AtLeast(v) => format!(">={}", version_text(v)),
        Constraint::Before(v) => format!("<{}", version_text(v)),
        Constraint::Caret(v) => format!("^{}", version_text(v)),
    }
}

fn canonical_constraints(constraints: &mut Vec<Constraint>) {
    constraints.sort();
    constraints.dedup();
}

fn constraints_text(constraints: &[Constraint]) -> String {
    let mut constraints = constraints.to_vec();
    canonical_constraints(&mut constraints);
    constraints
        .iter()
        .map(|constraint| constraint_text(*constraint))
        .collect::<Vec<_>>()
        .join(",")
}

fn matches(version: Version, constraint: Constraint) -> bool {
    match constraint {
        Constraint::Exact(v) => version == v,
        Constraint::AtLeast(v) => version >= v,
        Constraint::Before(v) => version < v,
        Constraint::Caret(v) => {
            version >= v
                && if v.major > 0 {
                    version.major == v.major
                } else if v.minor > 0 {
                    version.major == 0 && version.minor == v.minor
                } else {
                    version == v
                }
        }
    }
}

fn satisfies(version: Version, constraints: &[Constraint]) -> bool {
    constraints
        .iter()
        .all(|constraint| matches(version, *constraint))
}

fn percent_encode(value: &str) -> String {
    let mut out = String::new();
    for byte in value.as_bytes() {
        if byte.is_ascii_alphabetic()
            || byte.is_ascii_digit()
            || matches!(*byte, b'.' | b'_' | b'-')
        {
            out.push(*byte as char);
        } else {
            out.push_str(&format!("%{byte:02X}"));
        }
    }
    out
}

pub fn error_text(error: &ResolveError) -> String {
    match error {
        ResolveError::InvalidName { site, name } => format!(
            "invalid-name|site={}|name={}",
            percent_encode(site),
            percent_encode(name)
        ),
        ResolveError::DuplicatePackage { package } => {
            format!("duplicate-package|package={}", percent_encode(package))
        }
        ResolveError::DuplicateVersion { package, version } => format!(
            "duplicate-version|package={}|version={}",
            percent_encode(package),
            version_text(*version)
        ),
        ResolveError::Missing { package } => {
            format!("missing|package={}", percent_encode(package))
        }
        ResolveError::NoMatch {
            package,
            requirements,
        } => format!(
            "no-match|package={}|constraints={}",
            percent_encode(package),
            constraints_text(requirements)
        ),
        ResolveError::Conflict {
            package,
            selected,
            requirements,
        } => format!(
            "conflict|package={}|selected={}|constraints={}",
            percent_encode(package),
            version_text(*selected),
            constraints_text(requirements)
        ),
        ResolveError::Limit { attempts } => format!("limit|attempts={attempts}"),
    }
}

fn inspect_node(
    catalog: &[Package],
    roots: &[Requirement],
    assignments: &[(String, Version, usize)],
) -> Result<
    (Vec<String>, Vec<String>, Vec<Vec<Constraint>>),
    ResolveError,
> {
    let mut active: Vec<(String, Vec<Constraint>)> = Vec::new();

    fn add_constraint(active: &mut Vec<(String, Vec<Constraint>)>, name: &str, constraint: Constraint) {
        if let Some((_, constraints)) = active.iter_mut().find(|(n, _)| n == name) {
            constraints.push(constraint);
        } else {
            active.push((name.to_string(), vec![constraint]));
        }
    }

    for root in roots {
        add_constraint(&mut active, &root.package, root.constraint);
    }
    for (name, _, release_index) in assignments {
        let package_index = catalog
            .iter()
            .position(|package| package.name == *name)
            .expect("assigned package must exist");
        for dep in &catalog[package_index].releases[*release_index].deps {
            add_constraint(&mut active, &dep.package, dep.constraint);
        }
    }

    for (_, constraints) in &mut active {
        canonical_constraints(constraints);
    }
    active.sort_by(|a, b| a.0.cmp(&b.0));

    if let Some((name, _)) = active
        .iter()
        .find(|(name, _)| !catalog.iter().any(|package| package.name == *name))
    {
        return Err(ResolveError::Missing {
            package: name.clone(),
        });
    }

    for (name, constraints) in &active {
        if let Some((_, version, _)) = assignments.iter().find(|(assigned, _, _)| assigned == name) {
            if !satisfies(*version, constraints) {
                return Err(ResolveError::Conflict {
                    package: name.clone(),
                    selected: *version,
                    requirements: constraints.clone(),
                });
            }
        }
    }

    for (name, constraints) in &active {
        if assignments.iter().any(|(assigned, _, _)| assigned == name) {
            continue;
        }
        let package = catalog.iter().find(|package| package.name == *name).unwrap();
        if !package
            .releases
            .iter()
            .any(|release| satisfies(release.version, constraints))
        {
            return Err(ResolveError::NoMatch {
                package: name.clone(),
                requirements: constraints.clone(),
            });
        }
    }

    let unselected = active
        .iter()
        .filter(|(name, _)| !assignments.iter().any(|(assigned, _, _)| assigned == name))
        .map(|(name, _)| name.clone())
        .collect::<Vec<_>>();

    let candidates = if let Some(name) = unselected.first() {
        let package = catalog.iter().find(|package| package.name == *name).unwrap();
        let constraints = &active.iter().find(|(n, _)| n == name).unwrap().1;
        let mut releases = package
            .releases
            .iter()
            .enumerate()
            .filter(|(_, release)| satisfies(release.version, constraints))
            .map(|(index, release)| (release.version, index))
            .collect::<Vec<_>>();
        releases.sort_by(|a, b| b.0.cmp(&a.0));
        releases.into_iter().map(|(_, index)| index).collect()
    } else {
        Vec::new()
    };

    Ok((unselected, active.iter().map(|(n, _)| n.clone()).collect(), active.into_iter().map(|(_, c)| c).collect()))
}

fn search(
    catalog: &[Package],
    roots: &[Requirement],
    assignments: &mut Vec<(String, Version, usize)>,
    attempts: &mut usize,
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    let (unselected, _, _) = match inspect_node(catalog, roots, assignments) {
        Ok(state) => state,
        Err(error) => return Err(error),
    };

    if unselected.is_empty() {
        let mut result = assignments
            .iter()
            .map(|(package, version, _)| Selection {
                package: package.clone(),
                version: *version,
            })
            .collect::<Vec<_>>();
        result.sort_by(|a, b| a.package.cmp(&b.package));
        return Ok(result);
    }

    let package_name = unselected[0].clone();
    let package = catalog.iter().find(|package| package.name == package_name).unwrap();

    let active_constraints = {
        let mut constraints = Vec::new();
        for root in roots.iter().filter(|root| root.package == package_name) {
            constraints.push(root.constraint);
        }
        for (name, _, release_index) in assignments.iter() {
            let assigned_package = catalog.iter().find(|p| p.name == *name).unwrap();
            for dep in &assigned_package.releases[*release_index].deps {
                if dep.package == package_name {
                    constraints.push(dep.constraint);
                }
            }
        }
        canonical_constraints(&mut constraints);
        constraints
    };

    let mut candidates = package
        .releases
        .iter()
        .enumerate()
        .filter(|(_, release)| satisfies(release.version, &active_constraints))
        .map(|(index, release)| (release.version, index))
        .collect::<Vec<_>>();
    candidates.sort_by(|a, b| b.0.cmp(&a.0));

    let mut first_error = None;
    for (version, release_index) in candidates {
        if *attempts == max_attempts {
            return Err(ResolveError::Limit {
                attempts: max_attempts,
            });
        }
        *attempts += 1;
        assignments.push((package_name.clone(), version, release_index));
        match search(catalog, roots, assignments, attempts, max_attempts) {
            Ok(solution) => return Ok(solution),
            Err(error @ ResolveError::Limit { .. }) => {
                assignments.pop();
                return Err(error);
            }
            Err(error) => {
                assignments.pop();
                if first_error.is_none() {
                    first_error = Some(error);
                }
            }
        }
    }

    Err(first_error.unwrap_or_else(|| ResolveError::NoMatch {
        package: package_name,
        requirements: active_constraints,
    }))
}

pub fn resolve(
    catalog: &[Package],
    roots: &[Requirement],
    max_attempts: usize,
) -> Result<Vec<Selection>, ResolveError> {
    validate_names(catalog, roots)?;

    let mut seen_packages = std::collections::BTreeSet::new();
    for package in catalog {
        if !seen_packages.insert(package.name.clone()) {
            return Err(ResolveError::DuplicatePackage {
                package: package.name.clone(),
            });
        }
    }

    for package in catalog {
        let mut seen_versions = std::collections::BTreeSet::new();
        for release in &package.releases {
            if !seen_versions.insert(release.version) {
                return Err(ResolveError::DuplicateVersion {
                    package: package.name.clone(),
                    version: release.version,
                });
            }
        }
    }

    let mut assignments = Vec::new();
    let mut attempts = 0;
    search(
        catalog,
        roots,
        &mut assignments,
        &mut attempts,
        max_attempts,
    )
}
