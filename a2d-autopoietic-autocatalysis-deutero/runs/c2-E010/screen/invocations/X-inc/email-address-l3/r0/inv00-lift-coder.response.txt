#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
}

#[derive(Clone)]
struct Retained {
    byte: u8,
    pos: usize,
}

#[derive(Clone)]
struct CfwsRun {
    start: usize,
    end: usize,
    first_pos: usize,
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, runs) = cleanup(bytes)?;

    for run in &runs {
        let left_at = run.start > 0 && clean[run.start - 1].byte == b'@';
        let right_at = run.end < clean.len() && clean[run.end].byte == b'@';
        let at_boundary = run.start == 0 || run.end == clean.len();
        if !at_boundary && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary {
                at: run.first_pos,
            });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quoted_end = None;
    if clean[0].byte == b'"' {
        let mut i = 1;
        while i < clean.len() {
            match clean[i].byte {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    quoted_end = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: clean[0].pos });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < clean.len() {
        if let Some(end) = quoted_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if clean[i].byte == b'@' {
            ats.push(i);
        }
        i += 1;
    }

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: clean[ats[1]].pos,
        });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: clean[sep].pos,
        });
    }

    let local_bytes = &clean[..sep];
    let domain_bytes = &clean[sep + 1..];

    let (local, quoted) = if local_bytes[0].byte == b'"' {
        let end = quoted_end.expect("quoted local has a closing quote");
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: clean[end + 1].pos,
            });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            let b = clean[j].byte;
            if b == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: clean[j].pos,
                    });
                }
                let escaped = clean[j + 1].byte;
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: clean[j].pos,
                    });
                }
                decoded.push(escaped);
                origins.push(clean[j].pos);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: clean[j].pos });
                }
                decoded.push(b);
                origins.push(clean[j].pos);
                j += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: clean[end].pos,
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
        (
            String::from_utf8(decoded).expect("validated ASCII"),
            true,
        )
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_bytes[64].pos,
            });
        }

        let mut previous_dot = false;
        for (j, item) in local_bytes.iter().enumerate() {
            let b = item.byte;
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || previous_dot {
                    return Err(EmailError::LocalDot { at: item.pos });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                if !is_ascii_alphanumeric(b)
                    && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
                {
                    return Err(EmailError::LocalChar { at: item.pos });
                }
            }
        }

        (
            String::from_utf8(local_bytes.iter().map(|x| x.byte).collect())
                .expect("validated ASCII"),
            false,
        )
    };

    let domain_start = if domain_bytes.is_empty() {
        input.len()
    } else {
        domain_bytes[0].pos
    };
    if domain_bytes.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let mut domain_end = domain_bytes.len();
    let root_dot_pos = if domain_bytes.last().map(|x| x.byte) == Some(b'.') {
        domain_end -= 1;
        Some(domain_bytes[domain_end].pos)
    } else {
        None
    };

    if domain_end == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_end > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_bytes[253].pos,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0;
    for j in 0..=domain_end {
        if j == domain_end || domain_bytes[j].byte == b'.' {
            if label_start == j {
                let at = if j == domain_end {
                    root_dot_pos.unwrap_or(input.len())
                } else {
                    domain_bytes[j].pos
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    for &(start, end) in &labels {
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_bytes[start + 63].pos,
            });
        }
        if domain_bytes[start].byte == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_bytes[start].pos,
            });
        }
        if domain_bytes[end - 1].byte == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_bytes[end - 1].pos,
            });
        }
        for item in &domain_bytes[start..end] {
            if !is_ascii_alphanumeric(item.byte) && item.byte != b'-' {
                return Err(EmailError::DomainChar { at: item.pos });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *labels.last().expect("at least one label");
    if domain_bytes[last_start..last_end]
        .iter()
        .all(|item| item.byte.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: domain_bytes[last_start].pos,
        });
    }

    let domain = String::from_utf8(
        domain_bytes[..domain_end]
            .iter()
            .map(|x| x.byte.to_ascii_lowercase())
            .collect(),
    )
    .expect("validated ASCII");

    Ok(Address {
        local,
        domain,
        quoted,
    })
}

fn is_ascii_alphanumeric(b: u8) -> bool {
    b.is_ascii_alphanumeric()
}

fn cleanup(bytes: &[u8]) -> Result<(Vec<Retained>, Vec<CfwsRun>), EmailError> {
    let mut clean = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected_quote = false;

    while i < bytes.len() {
        if protected_quote {
            let b = bytes[i];
            clean.push(Retained { byte: b, pos: i });
            i += 1;
            if b == b'\\' && i < bytes.len() {
                clean.push(Retained {
                    byte: bytes[i],
                    pos: i,
                });
                i += 1;
            } else if b == b'"' {
                protected_quote = false;
            }
            continue;
        }

        let is_cfws_start = matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n' | b'(');
        if !is_cfws_start && bytes[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if !is_cfws_start {
            let first_retained = clean.is_empty();
            clean.push(Retained {
                byte: bytes[i],
                pos: i,
            });
            if first_retained && bytes[i] == b'"' {
                protected_quote = true;
            }
            i += 1;
            continue;
        }

        let run_start = clean.len();
        let first_pos = i;
        loop {
            if i >= bytes.len() {
                break;
            }
            match bytes[i] {
                b' ' | b'\t' => i += 1,
                b'\r' | b'\n' => {
                    let at = i;
                    if bytes[i] == b'\r'
                        && i + 2 < bytes.len()
                        && bytes[i + 1] == b'\n'
                        && matches!(bytes[i + 2], b' ' | b'\t')
                    {
                        i += 2;
                        while i < bytes.len() && matches!(bytes[i], b' ' | b'\t') {
                            i += 1;
                        }
                    } else {
                        return Err(EmailError::BadFold { at });
                    }
                }
                b'(' => {
                    let outer = i;
                    let mut stack = Vec::new();
                    stack.push(i);
                    i += 1;
                    while i < bytes.len() && !stack.is_empty() {
                        match bytes[i] {
                            b'\\' => {
                                let slash = i;
                                if i + 1 >= bytes.len()
                                    || !(0x20..=0x7e).contains(&bytes[i + 1])
                                {
                                    return Err(EmailError::CommentEscape { at: slash });
                                }
                                i += 2;
                            }
                            b'(' => {
                                if stack.len() == 3 {
                                    return Err(EmailError::CommentDepth { at: i });
                                }
                                stack.push(i);
                                i += 1;
                            }
                            b')' => {
                                stack.pop();
                                i += 1;
                            }
                            b'\r' | b'\n' => {
                                let at = i;
                                if bytes[i] == b'\r'
                                    && i + 2 < bytes.len()
                                    && bytes[i + 1] == b'\n'
                                    && matches!(bytes[i + 2], b' ' | b'\t')
                                {
                                    i += 2;
                                    while i < bytes.len()
                                        && matches!(bytes[i], b' ' | b'\t')
                                    {
                                        i += 1;
                                    }
                                } else {
                                    return Err(EmailError::BadFold { at });
                                }
                            }
                            b' ' | b'\t' => i += 1,
                            b if (0x20..=0x7e).contains(&b) => i += 1,
                            _ => return Err(EmailError::CommentChar { at: i }),
                        }
                    }
                    if !stack.is_empty() {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                }
                _ => break,
            }
        }
        runs.push(CfwsRun {
            start: run_start,
            end: clean.len(),
            first_pos,
        });
    }

    Ok((clean, runs))
}
