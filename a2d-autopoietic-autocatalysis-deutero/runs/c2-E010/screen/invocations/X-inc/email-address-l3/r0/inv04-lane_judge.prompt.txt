            return Err(LocalTooLong { at: decoded[64].1 });

            let b = clean[k].value;

                return Err(LocalChar { at: clean[k].pos });

            if label_start == k {
                let at = if k < end {
                    clean[k].pos
                } else if end < clean.len() {
                    clean[end].pos
                } else {
                    input.len()
                };
                return Err(EmptyLabel { at });

            labels.push((label_start, k));
            label_start = k + 1;
        }
        k += 1;
    }

        if b - a > 63 {
            return Err(LabelTooLong { at: clean[a + 63].pos });
        }
        if clean[a].value == b'-' {
            return Err(LabelHyphen { at: clean[a].pos });
        }
        if clean[b - 1].value == b'-' {
            return Err(LabelHyphen { at: clean[b - 1].pos });
        }
        for p in a..b {
            let c = clean[p].value;
            if !c.is_ascii_alphanumeric() && c != b'-' {
                return Err(DomainChar { at: clean[p].pos });
            }
        }
    }
    if labels.len() < 2 {
        return Err(DomainNeedsDot { at: domain_start });
    }
    let (last_a, last_b) = *labels.last().unwrap();
    if clean[last_a..last_b].iter().all(|b| b.value.is_ascii_digit()) {
        return Err(NumericTld { at: clean[last_a].pos });
    }

        .iter()
        .map(|b| (b.value as char).to_ascii_lowercase())
        .collect();

    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let (clean, runs) = cleanup(bytes)?;

    for run in &runs {
        let left_at = run.start > 0 && clean[run.start - 1].byte == b'@';
        let right_at = run.end < clean.len() && clean[run.end].byte == b'@';
        let at_boundary = run.start == 0 || run.end == clean.len();
        if !at_boundary && !left_at && !right_at {
            return Err(EmailError::CfwsBoundary {
                at: run.first_pos,
            });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quoted_end = None;
    if clean[0].byte == b'"' {
        let mut i = 1;
        while i < clean.len() {
            match clean[i].byte {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    quoted_end = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote { at: clean[0].pos });
        }
    }

    let mut ats = Vec::new();
    let mut i = 0;
    while i < clean.len() {
        if let Some(end) = quoted_end {
            if i <= end {
                i = end + 1;
                continue;
            }
        }
        if clean[i].byte == b'@' {
            ats.push(i);
        }
        i += 1;
    }

    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: clean[ats[1]].pos,
        });
    }

    let sep = ats[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: clean[sep].pos,
        });
    }

    let local_bytes = &clean[..sep];
    let domain_bytes = &clean[sep + 1..];

    let (local, quoted) = if local_bytes[0].byte == b'"' {
        let end = quoted_end.expect("quoted local has a closing quote");
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: clean[end + 1].pos,
            });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1;
        while j < end {
            let b = clean[j].byte;
            if b == b'\\' {
                if j + 1 >= end {
                    return Err(EmailError::BadEscape {
                        at: clean[j].pos,
                    });
                }
                let escaped = clean[j + 1].byte;
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: clean[j].pos,
                    });
                }
                decoded.push(escaped);
                origins.push(clean[j].pos);
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: clean[j].pos });
                }
                decoded.push(b);
                origins.push(clean[j].pos);
                j += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: clean[end].pos,
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
        (
            String::from_utf8(decoded).expect("validated ASCII"),
            true,
        )
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_bytes[64].pos,
            });
        }

        let mut previous_dot = false;
        for (j, item) in local_bytes.iter().enumerate() {
            let b = item.byte;
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || previous_dot {
                    return Err(EmailError::LocalDot { at: item.pos });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                if !is_ascii_alphanumeric(b)
                    && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
                {
                    return Err(EmailError::LocalChar { at: item.pos });
                }
            }
        }

        (
            String::from_utf8(local_bytes.iter().map(|x| x.byte).collect())
                .expect("validated ASCII"),
            false,
        )
    };

    let domain_start = if domain_bytes.is_empty() {
        input.len()
    } else {
        domain_bytes[0].pos
    };
    if domain_bytes.is_empty() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let mut domain_end = domain_bytes.len();
    let root_dot_pos = if domain_bytes.last().map(|x| x.byte) == Some(b'.') {
        domain_end -= 1;
        Some(domain_bytes[domain_end].pos)
    } else {
        None
    };

    if domain_end == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_end > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_bytes[253].pos,
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = 0;
    for j in 0..=domain_end {
        if j == domain_end || domain_bytes[j].byte == b'.' {
            if label_start == j {
                let at = if j == domain_end {
                    root_dot_pos.unwrap_or(input.len())
                } else {
                    domain_bytes[j].pos
                };
                return Err(EmailError::EmptyLabel { at });
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }

    for &(start, end) in &labels {
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_bytes[start + 63].pos,
            });
        }
        if domain_bytes[start].byte == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_bytes[start].pos,
            });
        }
        if domain_bytes[end - 1].byte == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_bytes[end - 1].pos,
            });
        }
        for item in &domain_bytes[start..end] {
            if !is_ascii_alphanumeric(item.byte) && item.byte != b'-' {
                return Err(EmailError::DomainChar { at: item.pos });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *labels.last().expect("at least one label");
    if domain_bytes[last_start..last_end]
        .iter()
        .all(|item| item.byte.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: domain_bytes[last_start].pos,
        });
    }

    let domain = String::from_utf8(
        domain_bytes[..domain_end]
            .iter()
            .map(|x| x.byte.to_ascii_lowercase())
            .collect(),
    )
    .expect("validated ASCII");

    Ok(Address {
        local,
        domain,
        quoted,
    })
}

fn is_ascii_alphanumeric(b: u8) -> bool {
    b.is_ascii_alphanumeric()
}

fn cleanup(bytes: &[u8]) -> Result<(Vec<Retained>, Vec<CfwsRun>), EmailError> {
    let mut clean = Vec::new();
    let mut runs = Vec::new();
    let mut i = 0;
    let mut protected_quote = false;

    while i < bytes.len() {
        if protected_quote {
            let b = bytes[i];
            clean.push(Retained { byte: b, pos: i });
            i += 1;
            if b == b'\\' && i < bytes.len() {
                clean.push(Retained {
                    byte: bytes[i],
                    pos: i,
                });
                i += 1;
            } else if b == b'"' {
                protected_quote = false;
            }
            continue;
        }

        let is_cfws_start = matches!(bytes[i], b' ' | b'\t' | b'\r' | b'\n' | b'(');
        if !is_cfws_start && bytes[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        if !is_cfws_start {
            let first_retained = clean.is_empty();
            clean.push(Retained {
                byte: bytes[i],
                pos: i,
            });
            if first_retained && bytes[i] == b'"' {
                protected_quote = true;
            }
            i += 1;
            continue;
        }

        let run_start = clean.len();
        let first_pos = i;
        loop {
            if i >= bytes.len() {
                break;
            }
            match bytes[i] {
                b' ' | b'\t' => i += 1,
                b'\r' | b'\n' => {
                    let at = i;
                    if bytes[i] == b'\r'
                        && i + 2 < bytes.len()
                        && bytes[i + 1] == b'\n'
                        && matches!(bytes[i + 2], b' ' | b'\t')
                    {
                        i += 2;
                        while i < bytes.len() && matches!(bytes[i], b' ' | b'\t') {
                            i += 1;
                        }
                    } else {
                        return Err(EmailError::BadFold { at });
                    }
                }
                b'(' => {
                    let outer = i;
                    let mut stack = Vec::new();
                    stack.push(i);
                    i += 1;
                    while i < bytes.len() && !stack.is_empty() {
                        match bytes[i] {
                            b'\\' => {
                                let slash = i;
                                if i + 1 >= bytes.len()
                                    || !(0x20..=0x7e).contains(&bytes[i + 1])
                                {
                                    return Err(EmailError::CommentEscape { at: slash });
                                }
                                i += 2;
                            }
                            b'(' => {
                                if stack.len() == 3 {
                                    return Err(EmailError::CommentDepth { at: i });
                                }
                                stack.push(i);
                                i += 1;
                            }
                            b')' => {
                                stack.pop();
                                i += 1;
                            }
                            b'\r' | b'\n' => {
                                let at = i;
                                if bytes[i] == b'\r'
                                    && i + 2 < bytes.len()
                                    && bytes[i + 1] == b'\n'
                                    && matches!(bytes[i + 2], b' ' | b'\t')
                                {
                                    i += 2;
                                    while i < bytes.len()
                                        && matches!(bytes[i], b' ' | b'\t')
                                    {
                                        i += 1;
                                    }
                                } else {
                                    return Err(EmailError::BadFold { at });
                                }
                            }
                            b' ' | b'\t' => i += 1,
                            b if (0x20..=0x7e).contains(&b) => i += 1,
                            _ => return Err(EmailError::CommentChar { at: i }),
                        }
                    }
                    if !stack.is_empty() {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                }
                _ => break,
            }
        }
        runs.push(CfwsRun {
            start: run_start,
            end: clean.len(),
            first_pos,
        });
    }

    Ok((clean, runs))
}

    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },

#[derive(Clone)]
struct Run {

    end: usize,

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut clean = Vec::<u8>::with_capacity(bytes.len());
    let mut positions = Vec::<usize>::with_capacity(bytes.len());
    let mut runs = Vec::<Run>::new();
    let mut i = 0usize;
    let mut protected = false;
    let mut quote_closed = false;
    let mut quote_escape = false;

    while i < bytes.len() {
        if protected {
            let b = bytes[i];
            clean.push(b);
            positions.push(i);
            i += 1;
            if quote_escape {
                quote_escape = false;
            } else if b == b'\\' {
                quote_escape = true;
            } else if b == b'"' {
                protected = false;
                quote_closed = true;
            }
            continue;
        }

        if !quote_closed && clean.is_empty() && bytes[i] == b'"' {
            protected = true;
            clean.push(bytes[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        let b = bytes[i];
        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            let run_start = i;
            let mut depth = 0usize;
            let mut outer = i;
            let mut consumed_any = false;

            loop {
                if i >= bytes.len() {
                    break;
                }
                let c = bytes[i];
                if depth == 0 {
                    if is_hws(c) {
                        consumed_any = true;
                        i += 1;
                        continue;
                    }
                    if c == b'\r' || c == b'\n' {
                        if c == b'\r'
                            && i + 2 < bytes.len()
                            && bytes[i + 1] == b'\n'
                            && is_hws(bytes[i + 2])
                        {
                            consumed_any = true;
                            i += 2;
                            while i < bytes.len() && is_hws(bytes[i]) {
                                i += 1;
                            }
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }
                    if c == b'(' {
                        if depth == 0 {
                            outer = i;
                        }
                        depth += 1;
                        if depth > 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        consumed_any = true;
                        i += 1;
                        continue;
                    }
                    break;
                } else {
                    if c == b'\\' {
                        if i + 1 >= bytes.len()
                            || !(0x20..=0x7e).contains(&bytes[i + 1])
                        {
                            return Err(EmailError::CommentEscape { at: i });
                        }
                        i += 2;
                        continue;
                    }
                    if c == b'\r' || c == b'\n' {
                        if c == b'\r'
                            && i + 2 < bytes.len()
                            && bytes[i + 1] == b'\n'
                            && is_hws(bytes[i + 2])
                        {
                            i += 2;
                            while i < bytes.len() && is_hws(bytes[i]) {
                                i += 1;
                            }
                            continue;
                        }
                        return Err(EmailError::BadFold { at: i });
                    }
                    if c == b'(' {
                        depth += 1;
                        if depth > 3 {
                            return Err(EmailError::CommentDepth { at: i });
                        }
                        i += 1;
                        continue;
                    }
                    if c == b')' {
                        depth -= 1;
                        i += 1;
                        continue;
                    }
                    if !(0x20..=0x7e).contains(&c) && c != b'\t' {
                        return Err(EmailError::CommentChar { at: i });
                    }
                    i += 1;
                    continue;
                }
            }

            if depth != 0 {
                return Err(EmailError::UnclosedComment { at: outer });
            }
            if consumed_any {
                runs.push(Run {
                    start: run_start,
                    end: i,
                });
            }
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        clean.push(b);
        positions.push(i);
        i += 1;
    }

    // A pending quoted prefix is intentionally handled by the quoted-prefix stage.
    let mut separators = Vec::new();
    let mut quote_end = None;
    if clean.first() == Some(&b'"') {
        let mut j = 1usize;
        let mut escaped = false;
        while j < clean.len() {
            let b = clean[j];
            if escaped {
                escaped = false;
            } else if b == b'\\' {
                escaped = true;
            } else if b == b'"' {
                quote_end = Some(j);
                break;
            }
            j += 1;
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: positions[0],
            });
        }
    }

    for j in 0..clean.len() {
        if clean[j] == b'@' && !matches!(quote_end, Some(end) if j <= end) {
            separators.push(j);
        }
    }

    // CFWS runs are checked after lexical cleanup, against retained bytes and @ signs.
    for run in &runs {
        let mut left = None;
        let mut right = None;
        for (idx, &pos) in positions.iter().enumerate() {
            if pos < run.start {
                left = Some(idx);
            } else if pos >= run.end {
                right = Some(idx);
                break;
            }
        }
        let allowed = left.is_none()
            || right.is_none()
            || left.is_some_and(|idx| clean[idx] == b'@')
            || right.is_some_and(|idx| clean[idx] == b'@');
        if !allowed {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: positions[separators[1]],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: positions[sep],
        });
    }

    let local_bytes = &clean[..sep];
    let local_positions = &positions[..sep];
    let quoted = local_bytes.first() == Some(&b'"');
    let local;

    if !quoted {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: local_positions[64],
            });
        }
        for i in 0..local_bytes.len() {
            let b = local_bytes[i];
            if b == b'.' {
                if i == 0
                    || i + 1 == local_bytes.len()
                    || local_bytes[i - 1] == b'.'
                {
                    return Err(EmailError::LocalDot {
                        at: local_positions[i],
                    });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar {
                    at: local_positions[i],
                });
            }
        }
        local = String::from_utf8(local_bytes.to_vec()).unwrap();
    } else {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: local_positions[end + 1],
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1usize;
        while i < end {
            let b = local_bytes[i];
            if b == b'\\' {
                if i + 1 >= end
                    || !matches!(local_bytes[i + 1], b'"' | b'\\' | b'@')
                {
                    return Err(EmailError::BadEscape {
                        at: local_positions[i],
                    });
                }
                decoded.push(local_bytes[i + 1]);
                origins.push(local_positions[i]);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: local_positions[i],
                    });
                }
                decoded.push(b);
                origins.push(local_positions[i]);
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: local_positions[end],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        local = String::from_utf8(decoded).unwrap();
    }

    let domain_start = sep + 1;
    let mut domain_end = clean.len();
    let mut root_dot = None;
    if domain_end > domain_start && clean[domain_end - 1] == b'.' {
        root_dot = Some(domain_end - 1);
        domain_end -= 1;
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: if domain_start < positions.len() {
                positions[domain_start]
            } else {
                input.len()
            },
        });
    }
    if domain_end - domain_start > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start + 253],
        });
    }

    let mut labels = Vec::<(usize, usize)>::new();
    let mut label_start = domain_start;
    for j in domain_start..domain_end {
        if clean[j] == b'.' {
            if label_start == j {
                return Err(EmailError::EmptyLabel {
                    at: positions[j],
                });
            }
            labels.push((label_start, j));
            label_start = j + 1;
        }
    }
    if label_start == domain_end {
        return Err(EmailError::EmptyLabel {
            at: root_dot.map(|j| positions[j]).unwrap_or(input.len()),
        });
    }
    labels.push((label_start, domain_end));

    for &(start, end) in &labels {
        if end - start > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[start + 63],
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[start],
            });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[end - 1],
            });
        }
        for j in start..end {
            let b = clean[j];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: positions[j],
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot {
            at: positions[domain_start],
        });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: positions[last_start],
        });
    }

    let domain = clean[domain_start..domain_end]
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}

}

#[derive(Clone, Copy)]
struct CfwsRun {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_printable(b: u8) -> bool {
    (0x20..=0x7e).contains(&b)
}

fn is_fold_at(bytes: &[u8], i: usize) -> bool {
    i + 2 < bytes.len()
        && bytes[i] == b'\r'
        && bytes[i + 1] == b'\n'
        && is_hws(bytes[i + 2])
}

fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut cleaned = Vec::<u8>::with_capacity(bytes.len());
    let mut positions = Vec::<usize>::with_capacity(bytes.len());
    let mut runs = Vec::<CfwsRun>::new();
    let mut i = 0usize;
    let mut quoted_prefix = false;
    let mut in_quote = false;

    while i < bytes.len() {
        if in_quote {
            let b = bytes[i];
            cleaned.push(b);
            positions.push(i);
            if b == b'\\' {
                i += 1;
                if i < bytes.len() {
                    cleaned.push(bytes[i]);
                    positions.push(i);
                    i += 1;
                }
            } else {
                if b == b'"' {
                    in_quote = false;
                }
                i += 1;
            }
            continue;
        }

        if !quoted_prefix && cleaned.is_empty() && bytes[i] == b'"' {
            quoted_prefix = true;
            in_quote = true;
            cleaned.push(bytes[i]);
            positions.push(i);
            i += 1;
            continue;
        }

        let begins_run = is_hws(bytes[i]) || bytes[i] == b'(' || is_fold_at(bytes, i);
        if begins_run {
            let start = i;
            loop {
                if i >= bytes.len() {
                    break;
                }
                if is_hws(bytes[i]) {
                    i += 1;
                    continue;
                }
                if is_fold_at(bytes, i) {
                    i += 2;
                    while i < bytes.len() && is_hws(bytes[i]) {
                        i += 1;
                    }
                    continue;
                }
                if bytes[i] == b'\r' || bytes[i] == b'\n' {
                    return Err(EmailError::BadFold { at: i });
                }
                if bytes[i] == b'(' {
                    let outer = i;
                    let mut depth = 0usize;
                    while i < bytes.len() {
                        let b = bytes[i];
                        if b == b'\\' {
                            let slash = i;
                            if i + 1 >= bytes.len() || !is_printable(bytes[i + 1]) {
                                return Err(EmailError::CommentEscape { at: slash });
                            }
                            i += 2;
                            continue;
                        }
                        if b == b'\r' || b == b'\n' {
                            if !is_fold_at(bytes, i) {
                                return Err(EmailError::BadFold { at: i });
                            }
                            i += 2;
                            while i < bytes.len() && is_hws(bytes[i]) {
                                i += 1;
                            }
                            continue;
                        }
                        if b == b'(' {
                            depth += 1;
                            if depth > 3 {
                                return Err(EmailError::CommentDepth { at: i });
                            }
                            i += 1;
                            continue;
                        }
                        if b == b')' {
                            depth -= 1;
                            i += 1;
                            if depth == 0 {
                                break;
                            }
                            continue;
                        }
                        if !is_printable(b) && b != b'\t' {
                            return Err(EmailError::CommentChar { at: i });
                        }
                        i += 1;
                    }
                    if depth != 0 {
                        return Err(EmailError::UnclosedComment { at: outer });
                    }
                    continue;
                }
                break;
            }
            runs.push(CfwsRun { start, end: i });
            continue;
        }

        if bytes[i] == b'\r' || bytes[i] == b'\n' {
            return Err(EmailError::BadFold { at: i });
        }
        if bytes[i] == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }

        cleaned.push(bytes[i]);
        positions.push(i);
        i += 1;
    }

    let mut at_positions = Vec::new();
    for (idx, &b) in cleaned.iter().enumerate() {
        if b == b'@' {
            at_positions.push(idx);
        }
    }
    for run in runs {
        let left_at = run.start > 0
            && positions.iter().any(|&p| p + 1 == run.start)
            && cleaned
                .get(positions.iter().position(|&p| p + 1 == run.start).unwrap_or(usize::MAX))
                == Some(&b'@');
        let right_at = run.end < bytes.len()
            && positions.iter().any(|&p| p == run.end)
            && cleaned
                .get(positions.iter().position(|&p| p == run.end).unwrap_or(usize::MAX))
                == Some(&b'@');
        let at_begin = !positions.iter().any(|&p| p < run.start);
        let at_end = !positions.iter().any(|&p| p >= run.end);
        if !(at_begin || at_end || left_at || right_at) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if cleaned.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quoted_end = None;
    if cleaned.first() == Some(&b'"') {
        let mut j = 1usize;
        while j < cleaned.len() {
            if cleaned[j] == b'\\' {
                j += 2;
            } else if cleaned[j] == b'"' {
                quoted_end = Some(j);
                break;
            } else {
                j += 1;
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: positions[0],
            });
        }
    }

    let mut separators = Vec::new();
    let mut idx = 0usize;
    while idx < cleaned.len() {
        if let Some(end) = quoted_end {
            if idx <= end {
                idx = end + 1;
                continue;
            }
        }
        if cleaned[idx] == b'@' {
            separators.push(idx);
        }
        idx += 1;
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: positions[separators[1]],
        });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: positions[sep],
        });
    }

    let mut local = String::new();
    let quoted = cleaned[0] == b'"';
    if !quoted {
        let local_bytes = &cleaned[..sep];
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: positions[64],
            });
        }
        for j in 0..local_bytes.len() {
            let b = local_bytes[j];
            if b == b'.' {
                if j == 0 || j + 1 == local_bytes.len() || local_bytes[j - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: positions[j] });
                }
            } else if !(b.is_ascii_alphanumeric()
                || b"!#$%&'*+-/=?^_`{|}~".contains(&b))
            {
                return Err(EmailError::LocalChar { at: positions[j] });
            }
            local.push(b as char);
        }
    } else {
        let end = quoted_end.unwrap();
        if end + 1 != sep {
            let bad = end + 1;
            return Err(EmailError::TrailingLocal {
                at: positions.get(bad).copied().unwrap_or(input.len()),
            });
        }
        let mut decoded = Vec::<(u8, usize)>::new();
        let mut j = 1usize;
        while j < end {
            let b = cleaned[j];
            if b == b'\\' {
                let origin = positions[j];
                if j + 1 >= end {
                    return Err(EmailError::BadEscape { at: origin });
                }
                let escaped = cleaned[j + 1];
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape { at: origin });
                }
                decoded.push((escaped, origin));
                j += 2;
            } else {
                if !is_printable(b) {
                    return Err(EmailError::QuotedChar { at: positions[j] });
                }
                decoded.push((b, positions[j]));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: positions[end],
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: decoded[64].1,
            });
        }
        for (b, _) in decoded {
            local.push(b as char);
        }
    }

    let domain_start_idx = sep + 1;
    let domain_start = positions.get(domain_start_idx).copied().unwrap_or(input.len());
    if domain_start_idx >= cleaned.len() {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let mut domain_end = cleaned.len();
    let mut root_dot = None;
    if domain_end > domain_start_idx && cleaned[domain_end - 1] == b'.' {
        root_dot = Some(positions[domain_end - 1]);
        domain_end -= 1;
    }
    if domain_end == domain_start_idx {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_len = domain_end - domain_start_idx;
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: positions[domain_start_idx + 253],
        });
    }

    let domain_bytes = &cleaned[domain_start_idx..domain_end];
    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0usize;
    for j in 0..=domain_bytes.len() {
        if j == domain_bytes.len() || domain_bytes[j] == b'.' {
            labels.push((start, j));
            start = j + 1;
        }
    }

    for (n, &(a, b)) in labels.iter().enumerate() {
        if a == b {
            let at = if a < domain_bytes.len() {
                positions[domain_start_idx + a]
            } else if n > 0 && b > 0 {
                positions[domain_start_idx + b - 1]
            } else {
                root_dot.unwrap_or(domain_start)
            };
            return Err(EmailError::EmptyLabel { at });
        }
        if b - a > 63 {
            return Err(EmailError::LabelTooLong {
                at: positions[domain_start_idx + a + 63],
            });
        }
        if domain_bytes[a] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[domain_start_idx + a],
            });
        }
        if domain_bytes[b - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: positions[domain_start_idx + b - 1],
            });
        }
        for j in a..b {
            let ch = domain_bytes[j];
            if !(ch.is_ascii_alphanumeric() || ch == b'-') {
                return Err(EmailError::DomainChar {
                    at: positions[domain_start_idx + j],
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *labels.last().unwrap();
    if domain_bytes[last_start..last_end]
        .iter()
        .all(|b| b.is_ascii_digit())
    {
        return Err(EmailError::NumericTld {
            at: positions[domain_start_idx + last_start],
        });
    }

    let domain = domain_bytes
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
