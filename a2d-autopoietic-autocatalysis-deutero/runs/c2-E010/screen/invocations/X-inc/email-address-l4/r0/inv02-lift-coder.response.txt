#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct CfwsRun {
    start: usize,
    end: usize,
}

struct Cleaned {
    bytes: Vec<u8>,
    positions: Vec<usize>,
    runs: Vec<CfwsRun>,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn valid_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().is_some_and(is_hws) {
        return None;
    }
    while bytes.get(j).copied().is_some_and(is_hws) {
        j += 1;
    }
    Some(j)
}

fn cleanup(input: &[u8]) -> Result<Cleaned, EmailError> {
    let mut bytes = Vec::with_capacity(input.len());
    let mut positions = Vec::with_capacity(input.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut depth = 0usize;
    let mut outer_open = 0usize;
    let mut in_quote = false;
    let mut pending_run: Option<usize> = None;

    while i < input.len() {
        let b = input[i];

        if in_quote {
            bytes.push(b);
            positions.push(i);
            if b == b'\\' {
                if i + 1 < input.len() {
                    bytes.push(input[i + 1]);
                    positions.push(i + 1);
                    i += 2;
                } else {
                    i += 1;
                }
            } else {
                if b == b'"' {
                    in_quote = false;
                }
                i += 1;
            }
            continue;
        }

        if depth > 0 {
            if let Some(start) = pending_run {
                if !is_hws(b) && valid_fold(input, i).is_none() && b != b'(' && b != b')' && b != b'\\' {
                    runs.push(CfwsRun { start, end: i });
                    pending_run = None;
                }
            }
            if b == b'\\' {
                let Some(&next) = input.get(i + 1) else {
                    return Err(EmailError::CommentEscape { at: i });
                };
                if !(0x20..=0x7e).contains(&next) {
                    return Err(EmailError::CommentEscape { at: i });
                }
                i += 2;
            } else if b == b'(' {
                if depth == 3 {
                    return Err(EmailError::CommentDepth { at: i });
                }
                depth += 1;
                i += 1;
            } else if b == b')' {
                depth -= 1;
                i += 1;
                if depth == 0 {
                    if pending_run.is_none() {
                        pending_run = Some(outer_open);
                    }
                }
            } else if let Some(end) = valid_fold(input, i) {
                if pending_run.is_none() {
                    pending_run = Some(i);
                }
                i = end;
            } else if b == b'\r' || b == b'\n' {
                return Err(EmailError::BadFold { at: i });
            } else if is_hws(b) {
                if pending_run.is_none() {
                    pending_run = Some(i);
                }
                i += 1;
            } else if (0x20..=0x7e).contains(&b) {
                i += 1;
            } else {
                return Err(EmailError::CommentChar { at: i });
            }
            continue;
        }

        if let Some(start) = pending_run {
            let end = if i > start { i } else { start + 1 };
            runs.push(CfwsRun { start, end });
            pending_run = None;
        }

        if bytes.is_empty() && b == b'"' {
            bytes.push(b);
            positions.push(i);
            in_quote = true;
            i += 1;
            continue;
        }
        if b == b'(' {
            outer_open = i;
            depth = 1;
            pending_run = Some(i);
            i += 1;
            continue;
        }
        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        if let Some(end) = valid_fold(input, i) {
            pending_run = Some(i);
            i = end;
            continue;
        }
        if b == b'\r' || b == b'\n' {
            return Err(EmailError::BadFold { at: i });
        }
        if is_hws(b) {
            pending_run = Some(i);
            i += 1;
            continue;
        }

        bytes.push(b);
        positions.push(i);
        i += 1;
    }

    if depth > 0 {
        return Err(EmailError::UnclosedComment { at: outer_open });
    }
    if let Some(start) = pending_run {
        runs.push(CfwsRun {
            start,
            end: input.len(),
        });
    }

    Ok(Cleaned {
        bytes,
        positions,
        runs,
    })
}

fn original_pos(clean: &Cleaned, index: usize, input_len: usize) -> usize {
    clean.positions.get(index).copied().unwrap_or(input_len)
}

fn is_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn dot_atom(local: &[u8]) -> bool {
    if local.is_empty() || local.len() > 64 {
        return false;
    }
    let mut previous_dot = true;
    for (i, &b) in local.iter().enumerate() {
        if b == b'.' {
            if previous_dot || i + 1 == local.len() {
                return false;
            }
            previous_dot = true;
        } else if is_atom_char(b) {
            previous_dot = false;
        } else {
            return false;
        }
    }
    true
}

fn parse_core(
    input: &str,
    allow_single_label: bool,
) -> Result<(Address, usize), EmailError> {
    let raw = input.as_bytes();
    if raw.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(at) = raw.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let clean = cleanup(raw)?;
    for run in &clean.runs {
        let left = clean.positions.iter().position(|&p| p >= run.end);
        let right = clean.positions.iter().rposition(|&p| p < run.start);
        let left_is_boundary = left.is_none();
        let right_is_boundary = right.is_none();
        let touches_at = right.is_some_and(|idx| clean.bytes[idx] == b'@')
            || left.is_some_and(|idx| clean.bytes[idx] == b'@');
        if !left_is_boundary && !right_is_boundary && !touches_at {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let data = &clean.bytes;
    let input_len = raw.len();
    let mut quote_end = None;
    if data.first() == Some(&b'"') {
        let mut i = 1usize;
        while i < data.len() {
            if data[i] == b'\\' {
                i = i.saturating_add(2);
            } else if data[i] == b'"' {
                quote_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quote_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: original_pos(&clean, 0, input_len),
            });
        }
    }

    let mut separators = Vec::new();
    let mut i = 0usize;
    while i < data.len() {
        if data[i] == b'@' {
            if let Some(end) = quote_end {
                if i <= end {
                    i += 1;
                    continue;
                }
            }
            separators.push(i);
        }
        i += 1;
    }
    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt {
            at: original_pos(&clean, separators[1], input_len),
        });
    }
    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal {
            at: original_pos(&clean, sep, input_len),
        });
    }

    let local_raw = &data[..sep];
    let (local, quoted, local_origins) = if local_raw[0] == b'"' {
        let end = quote_end.unwrap();
        if end + 1 != sep {
            return Err(EmailError::TrailingLocal {
                at: original_pos(&clean, end + 1, input_len),
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut j = 1usize;
        while j < end {
            let b = local_raw[j];
            if b == b'\\' {
                let Some(&escaped) = local_raw.get(j + 1) else {
                    return Err(EmailError::BadEscape {
                        at: original_pos(&clean, j, input_len),
                    });
                };
                if escaped != b'"' && escaped != b'\\' && escaped != b'@' {
                    return Err(EmailError::BadEscape {
                        at: original_pos(&clean, j, input_len),
                    });
                }
                decoded.push(escaped);
                origins.push(original_pos(&clean, j, input_len));
                j += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: original_pos(&clean, j, input_len),
                    });
                }
                decoded.push(b);
                origins.push(original_pos(&clean, j, input_len));
                j += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: original_pos(&clean, end, input_len),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
        (decoded, true, origins)
    } else {
        if local_raw.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: original_pos(&clean, 64, input_len),
            });
        }
        let mut previous_dot = false;
        for (j, &b) in local_raw.iter().enumerate() {
            if b == b'.' {
                if j == 0 || j + 1 == local_raw.len() || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: original_pos(&clean, j, input_len),
                    });
                }
                previous_dot = true;
            } else {
                if !is_atom_char(b) {
                    return Err(EmailError::LocalChar {
                        at: original_pos(&clean, j, input_len),
                    });
                }
                previous_dot = false;
            }
        }
        (
            local_raw.to_vec(),
            false,
            (0..local_raw.len())
                .map(|j| original_pos(&clean, j, input_len))
                .collect(),
        )
    };

    let domain_start = sep + 1;
    let mut domain_end = data.len();
    if domain_start < domain_end && data[domain_end - 1] == b'.' {
        domain_end -= 1;
    }
    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain {
            at: original_pos(&clean, domain_start, input_len),
        });
    }
    let domain_raw = &data[domain_start..domain_end];
    if domain_raw.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: original_pos(&clean, domain_start + 253, input_len),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut start = 0usize;
    for j in 0..=domain_raw.len() {
        if j == domain_raw.len() || domain_raw[j] == b'.' {
            labels.push((start, j));
            start = j + 1;
        }
    }
    for &(a, b) in &labels {
        if a == b {
            let index = if a == domain_raw.len() && domain_end < data.len() {
                domain_end
            } else {
                domain_start + a
            };
            return Err(EmailError::EmptyLabel {
                at: original_pos(&clean, index, input_len),
            });
        }
        if b - a > 63 {
            return Err(EmailError::LabelTooLong {
                at: original_pos(&clean, domain_start + a + 63, input_len),
            });
        }
        if domain_raw[a] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(&clean, domain_start + a, input_len),
            });
        }
        if domain_raw[b - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: original_pos(&clean, domain_start + b - 1, input_len),
            });
        }
        for j in a..b {
            let c = domain_raw[j];
            if !c.is_ascii_alphanumeric() && c != b'-' {
                return Err(EmailError::DomainChar {
                    at: original_pos(&clean, domain_start + j, input_len),
                });
            }
        }
    }
    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: original_pos(&clean, domain_start, input_len),
        });
    }
    let (last_start, last_end) = labels[labels.len() - 1];
    if domain_raw[last_start..last_end]
        .iter()
        .all(u8::is_ascii_digit)
    {
        return Err(EmailError::NumericTld {
            at: original_pos(&clean, domain_start + last_start, input_len),
        });
    }

    let domain = domain_raw
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();
    let decoded_local = String::from_utf8(local).unwrap();
    let plus_position = decoded_local.as_bytes().iter().position(|&b| b == b'+');
    Ok((
        Address {
            local: decoded_local,
            domain,
            quoted,
        },
        plus_position.map(|p| local_origins[p]),
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, plus_origin) = parse_core(input, policy.allow_single_label)?;
    if policy.local_case == LocalCase::Lower {
        address.local.make_ascii_lowercase();
    }
    if policy.strip_subaddress {
        if let Some(plus) = address.local.find('+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: plus_origin.unwrap_or(input.len()),
                });
            }
            address.local.truncate(plus);
        }
    }
    let safe = dot_atom(address.local.as_bytes());
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut out = String::new();
    if address.quoted {
        out.push('"');
        for ch in address.local.chars() {
            if ch == '"' || ch == '\\' {
                out.push('\\');
            }
            out.push(ch);
        }
        out.push('"');
    } else {
        out.push_str(&address.local);
    }
    out.push('@');
    out.push_str(&address.domain);
    out
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut keys: Vec<(String, String, usize)> = Vec::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                if let Some((_, _, first)) = keys
                    .iter()
                    .find(|(local, domain, _)| local == &address.local && domain == &address.domain)
                {
                    duplicate_of[index] = Some(*first);
                } else {
                    keys.push((address.local.clone(), address.domain.clone(), index));
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
