use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
    BadFold { at: usize },
    UnclosedComment { at: usize },
    CommentDepth { at: usize },
    CommentChar { at: usize },
    CommentEscape { at: usize },
    UnexpectedClose { at: usize },
    CfwsBoundary { at: usize },
    EmptyNormalizedLocal { at: usize },
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum LocalCase {
    Preserve,
    Lower,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Policy {
    pub local_case: LocalCase,
    pub strip_subaddress: bool,
    pub unquote_safe: bool,
    pub allow_single_label: bool,
}

impl Default for Policy {
    fn default() -> Self {
        Self {
            local_case: LocalCase::Preserve,
            strip_subaddress: false,
            unquote_safe: false,
            allow_single_label: false,
        }
    }
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct DedupResult {
    pub addresses: Vec<Address>,
    pub source_indices: Vec<usize>,
    pub duplicate_of: Vec<Option<usize>>,
    pub errors: Vec<(usize, EmailError)>,
}

#[derive(Clone, Copy)]
struct CfwsRun {
    start: usize,
    end: usize,
}

fn is_hws(b: u8) -> bool {
    b == b' ' || b == b'\t'
}

fn is_fold(bytes: &[u8], i: usize) -> Option<usize> {
    if bytes.get(i) != Some(&b'\r') || bytes.get(i + 1) != Some(&b'\n') {
        return None;
    }
    let mut j = i + 2;
    if !bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        return None;
    }
    while bytes.get(j).copied().map(is_hws).unwrap_or(false) {
        j += 1;
    }
    Some(j)
}

fn lexical_cleanup(input: &str) -> Result<(Vec<u8>, Vec<usize>, Vec<CfwsRun>), EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }
    if let Some(i) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at: i });
    }

    let mut clean = Vec::with_capacity(bytes.len());
    let mut map = Vec::with_capacity(bytes.len());
    let mut runs = Vec::new();
    let mut i = 0usize;
    let mut first_retained = true;
    let mut in_protected_quote = false;

    while i < bytes.len() {
        if in_protected_quote {
            let b = bytes[i];
            clean.push(b);
            map.push(i);
            i += 1;
            if b == b'\\' {
                if i < bytes.len() {
                    clean.push(bytes[i]);
                    map.push(i);
                    i += 1;
                }
            } else if b == b'"' {
                in_protected_quote = false;
            }
            continue;
        }

        let b = bytes[i];
        if b == b'"' && first_retained {
            clean.push(b);
            map.push(i);
            i += 1;
            first_retained = false;
            in_protected_quote = true;
            continue;
        }

        if b == b' ' || b == b'\t' || b == b'\r' || b == b'\n' || b == b'(' {
            let start = i;
            loop {
                if i >= bytes.len() {
                    break;
                }
                match bytes[i] {
                    b' ' | b'\t' => i += 1,
                    b'\r' | b'\n' => {
                        if let Some(next) = is_fold(bytes, i) {
                            i = next;
                        } else {
                            return Err(EmailError::BadFold { at: i });
                        }
                    }
                    b'(' => {
                        let outer = i;
                        let mut depth = 1usize;
                        i += 1;
                        while i < bytes.len() && depth > 0 {
                            match bytes[i] {
                                b'\\' => {
                                    let slash = i;
                                    if i + 1 >= bytes.len()
                                        || !(0x20..=0x7e).contains(&bytes[i + 1])
                                    {
                                        return Err(EmailError::CommentEscape { at: slash });
                                    }
                                    i += 2;
                                }
                                b'(' => {
                                    if depth == 3 {
                                        return Err(EmailError::CommentDepth { at: i });
                                    }
                                    depth += 1;
                                    i += 1;
                                }
                                b')' => {
                                    depth -= 1;
                                    i += 1;
                                }
                                b'\r' | b'\n' => {
                                    if let Some(next) = is_fold(bytes, i) {
                                        i = next;
                                    } else {
                                        return Err(EmailError::BadFold { at: i });
                                    }
                                }
                                c if c == b'\t' || (0x20..=0x7e).contains(&c) => i += 1,
                                _ => return Err(EmailError::CommentChar { at: i }),
                            }
                        }
                        if depth != 0 {
                            return Err(EmailError::UnclosedComment { at: outer });
                        }
                    }
                    _ => break,
                }
            }
            runs.push(CfwsRun { start, end: i });
            continue;
        }

        if b == b')' {
            return Err(EmailError::UnexpectedClose { at: i });
        }
        clean.push(b);
        map.push(i);
        first_retained = false;
        i += 1;
    }

    Ok((clean, map, runs))
}

fn origin(map: &[usize], clean_pos: usize, input_len: usize) -> usize {
    map.get(clean_pos).copied().unwrap_or(input_len)
}

fn is_local_atom_char(b: u8) -> bool {
    b.is_ascii_alphanumeric() || b"!#$%&'*+-/=?^_`{|}~".contains(&b)
}

fn valid_dot_atom(bytes: &[u8]) -> bool {
    if bytes.is_empty() {
        return false;
    }
    for (i, &b) in bytes.iter().enumerate() {
        if b == b'.' {
            if i == 0 || i + 1 == bytes.len() || bytes[i - 1] == b'.' {
                return false;
            }
        } else if !is_local_atom_char(b) {
            return false;
        }
    }
    true
}

fn parse_core(input: &str, allow_single_label: bool) -> Result<(Address, Vec<usize>), EmailError> {
    let input_len = input.len();
    let (clean, map, runs) = lexical_cleanup(input)?;

    for run in runs {
        let left = clean.iter().position(|_| false);
        let _ = left;
        let left_retained = map.iter().rposition(|&p| p < run.start);
        let right_retained = map.iter().position(|&p| p >= run.end);
        let left_is_boundary = left_retained.is_none();
        let right_is_boundary = right_retained.is_none();
        let left_is_at = left_retained.map(|k| clean[k] == b'@').unwrap_or(false);
        let right_is_at = right_retained.map(|k| clean[k] == b'@').unwrap_or(false);
        if !(left_is_boundary || right_is_boundary || left_is_at || right_is_at) {
            return Err(EmailError::CfwsBoundary { at: run.start });
        }
    }

    if clean.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    let mut quoted_end = None;
    if clean[0] == b'"' {
        let mut i = 1usize;
        while i < clean.len() {
            if clean[i] == b'\\' {
                i += 2;
            } else if clean[i] == b'"' {
                quoted_end = Some(i);
                break;
            } else {
                i += 1;
            }
        }
        if quoted_end.is_none() {
            return Err(EmailError::UnclosedQuote {
                at: origin(&map, 0, input_len),
            });
        }
    }

    let mut ats = Vec::new();
    for i in 0..clean.len() {
        if clean[i] == b'@' && !quoted_end.map(|q| i <= q).unwrap_or(false) {
            ats.push(i);
        }
    }
    if ats.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if ats.len() >= 2 {
        return Err(EmailError::MultipleAt {
            at: origin(&map, ats[1], input_len),
        });
    }

    let at = ats[0];
    if at == 0 {
        return Err(EmailError::EmptyLocal {
            at: origin(&map, at, input_len),
        });
    }

    let local_bytes = &clean[..at];
    let (local, quoted, local_origins) = if let Some(q) = quoted_end {
        if q + 1 != at {
            return Err(EmailError::TrailingLocal {
                at: origin(&map, q + 1, input_len),
            });
        }
        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1usize;
        while i < q {
            let b = clean[i];
            if b == b'\\' {
                if i + 1 >= q {
                    return Err(EmailError::BadEscape {
                        at: origin(&map, i, input_len),
                    });
                }
                let escaped = clean[i + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape {
                        at: origin(&map, i, input_len),
                    });
                }
                decoded.push(escaped);
                origins.push(origin(&map, i, input_len));
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar {
                        at: origin(&map, i, input_len),
                    });
                }
                decoded.push(b);
                origins.push(origin(&map, i, input_len));
                i += 1;
            }
        }
        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal {
                at: origin(&map, q, input_len),
            });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origins[64],
            });
        }
        (
            String::from_utf8(decoded).expect("validated ASCII"),
            true,
            origins,
        )
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong {
                at: origin(&map, 64, input_len),
            });
        }
        let mut previous_dot = false;
        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || previous_dot {
                    return Err(EmailError::LocalDot {
                        at: origin(&map, i, input_len),
                    });
                }
                previous_dot = true;
            } else {
                previous_dot = false;
                if !is_local_atom_char(b) {
                    return Err(EmailError::LocalChar {
                        at: origin(&map, i, input_len),
                    });
                }
            }
        }
        (
            String::from_utf8(local_bytes.to_vec()).expect("validated ASCII"),
            false,
            (0..local_bytes.len())
                .map(|i| origin(&map, i, input_len))
                .collect(),
        )
    };

    let domain_start = at + 1;
    let raw_domain = &clean[domain_start..];
    if raw_domain.is_empty() {
        return Err(EmailError::EmptyDomain {
            at: origin(&map, domain_start, input_len),
        });
    }
    let has_root = raw_domain.last() == Some(&b'.');
    let domain_end = if has_root {
        domain_start + raw_domain.len() - 1
    } else {
        clean.len()
    };
    let domain = &clean[domain_start..domain_end];
    if domain.is_empty() {
        return Err(EmailError::EmptyDomain {
            at: origin(&map, domain_start, input_len),
        });
    }
    if domain.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: origin(&map, domain_start + 253, input_len),
        });
    }

    let mut labels: Vec<(usize, usize)> = Vec::new();
    let mut label_start = domain_start;
    for i in domain_start..=domain_end {
        if i == domain_end || clean[i] == b'.' {
            labels.push((label_start, i));
            label_start = i + 1;
        }
    }

    for &(start, end) in &labels {
        if start == end {
            let pos = if start == domain_start {
                start
            } else if start == domain_end && has_root {
                domain_end
            } else {
                start.saturating_sub(1)
            };
            return Err(EmailError::EmptyLabel {
                at: origin(&map, pos, input_len),
            });
        }
        let len = end - start;
        if len > 63 {
            return Err(EmailError::LabelTooLong {
                at: origin(&map, start + 63, input_len),
            });
        }
        if clean[start] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: origin(&map, start, input_len),
            });
        }
        if clean[end - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: origin(&map, end - 1, input_len),
            });
        }
        for i in start..end {
            let b = clean[i];
            if !b.is_ascii_alphanumeric() && b != b'-' {
                return Err(EmailError::DomainChar {
                    at: origin(&map, i, input_len),
                });
            }
        }
    }

    if labels.len() < 2 && !allow_single_label {
        return Err(EmailError::DomainNeedsDot {
            at: origin(&map, domain_start, input_len),
        });
    }

    let (last_start, last_end) = *labels.last().expect("nonempty domain labels");
    if clean[last_start..last_end].iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: origin(&map, last_start, input_len),
        });
    }

    let lowered_domain = domain
        .iter()
        .map(|b| b.to_ascii_lowercase() as char)
        .collect::<String>();

    Ok((
        Address {
            local,
            domain: lowered_domain,
            quoted,
        },
        local_origins,
    ))
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    parse_core(input, false).map(|(address, _)| address)
}

pub fn normalize_email(input: &str, policy: &Policy) -> Result<Address, EmailError> {
    let (mut address, origins) = parse_core(input, policy.allow_single_label)?;
    let mut local = address.local.into_bytes();

    if policy.local_case == LocalCase::Lower {
        for b in &mut local {
            *b = b.to_ascii_lowercase();
        }
    }

    let mut local_origins = origins;
    if policy.strip_subaddress {
        if let Some(plus) = local.iter().position(|&b| b == b'+') {
            if plus == 0 {
                return Err(EmailError::EmptyNormalizedLocal {
                    at: local_origins.get(plus).copied().unwrap_or(input.len()),
                });
            }
            local.truncate(plus);
            local_origins.truncate(plus);
        }
    }

    let safe = valid_dot_atom(&local);
    if local.is_empty() {
        return Err(EmailError::EmptyNormalizedLocal { at: input.len() });
    }
    address.local = String::from_utf8(local).expect("validated ASCII local");
    if !safe {
        address.quoted = true;
    } else if policy.unquote_safe {
        address.quoted = false;
    }
    Ok(address)
}

pub fn format_email(address: &Address) -> String {
    let mut output = String::new();
    if address.quoted {
        output.push('"');
        for c in address.local.chars() {
            if c == '"' || c == '\\' {
                output.push('\\');
            }
            output.push(c);
        }
        output.push('"');
    } else {
        output.push_str(&address.local);
    }
    output.push('@');
    output.push_str(&address.domain);
    output
}

pub fn deduplicate(inputs: &[&str], policy: &Policy) -> DedupResult {
    let mut addresses = Vec::new();
    let mut source_indices = Vec::new();
    let mut duplicate_of = vec![None; inputs.len()];
    let mut errors = Vec::new();
    let mut seen: HashMap<(String, String), usize> = HashMap::new();

    for (index, input) in inputs.iter().enumerate() {
        match normalize_email(input, policy) {
            Err(error) => errors.push((index, error)),
            Ok(address) => {
                let key = (address.local.clone(), address.domain.clone());
                if let Some(&first_index) = seen.get(&key) {
                    duplicate_of[index] = Some(first_index);
                } else {
                    seen.insert(key, index);
                    source_indices.push(index);
                    addresses.push(address);
                }
            }
        }
    }

    DedupResult {
        addresses,
        source_indices,
        duplicate_of,
        errors,
    }
}
