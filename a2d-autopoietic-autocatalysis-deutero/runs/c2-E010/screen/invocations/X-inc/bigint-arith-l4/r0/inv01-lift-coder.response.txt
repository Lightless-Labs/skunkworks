```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1,000,000,000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;
    const POW_LIMIT: u32 = 1024;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: Vec::new(),
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.limbs.is_empty() {
            self.negative = false;
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs
            .len()
            .cmp(&rhs.limbs.len())
            .then_with(|| self.limbs.iter().rev().cmp(rhs.limbs.iter().rev()))
    }

    fn add_abs(&self, rhs: &Self) -> Self {
        let n = self.limbs.len().max(rhs.limbs.len());
        let mut limbs = Vec::with_capacity(n + 1);
        let mut carry = 0u64;

        for i in 0..n {
            let a = self.limbs.get(i).copied().unwrap_or(0) as u64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = a + b + carry;
            limbs.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            limbs.push(carry as u32);
        }

        Self {
            negative: false,
            limbs,
        }
    }

    // Requires |self| >= |rhs|.
    fn sub_abs(&self, rhs: &Self) -> Self {
        let mut limbs = Vec::with_capacity(self.limbs.len());
        let mut borrow = 0i64;

        for i in 0..self.limbs.len() {
            let a = self.limbs[i] as i64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut value = a - b - borrow;
            if value < 0 {
                value += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            limbs.push(value as u32);
        }

        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    fn mul_small_abs(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }

        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * factor as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_abs(&mut self, addend: u32) {
        let mut carry = addend as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let value = self.limbs[i] as u64 + carry;
            self.limbs[i] = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
            i += 1;
        }
    }

    fn div_rem_small_abs(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = vec![0u32; self.limbs.len()];
        let mut remainder = 0u64;

        for i in (0..self.limbs.len()).rev() {
            let value = remainder * Self::BASE + self.limbs[i] as u64;
            quotient[i] = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }

        let mut q = Self {
            negative: false,
            limbs: quotient,
        };
        q.normalize();
        (q, remainder as u32)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut first_invalid = None;
        let mut digit_count = 0usize;
        let mut first_digit_index = None;
        let mut first_digit_value = None;
        let mut previous_is_digit = false;
        let mut first_bad_separator = None;
        let mut first_excess = None;
        let mut digits_seen = 0usize;

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                if !previous_is_digit {
                    first_bad_separator.get_or_insert(index);
                }
                previous_is_digit = false;
                continue;
            }

            match Self::digit_value(byte) {
                Some(value) if value < radix => {
                    digit_count += 1;
                    if first_digit_index.is_none() {
                        first_digit_index = Some(index);
                        first_digit_value = Some(value);
                    }
                    digits_seen += 1;
                    if digits_seen == max_digits.saturating_add(1) {
                        first_excess = Some(index);
                    }
                    previous_is_digit = true;
                }
                _ => {
                    first_invalid.get_or_insert(index);
                    previous_is_digit = false;
                }
            }
        }

        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                let left_ok = index > start
                    && Self::digit_value(bytes[index - 1])
                        .is_some_and(|v| v < radix);
                let right_ok = index + 1 < bytes.len()
                    && Self::digit_value(bytes[index + 1])
                        .is_some_and(|v| v < radix);
                if !left_ok || !right_ok {
                    first_bad_separator.get_or_insert(index);
                }
            }
        }

        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        if digit_count > 1 && first_digit_value == Some(0) {
            return Err(Error::LeadingZero {
                index: first_digit_index.unwrap(),
            });
        }

        if digit_count > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: first_excess.unwrap(),
            });
        }

        let mut result = Self::zero();
        for &byte in bytes.iter().skip(start) {
            if byte == b'_' {
                continue;
            }
            let digit = Self::digit_value(byte).unwrap();
            result.mul_small_abs(radix);
            result.add_small_abs(digit);
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let alphabet = b"0123456789abcdefghijklmnopqrstuvwxyz";
        let mut value = self.abs();
        let mut digits = Vec::new();

        if value.is_zero() {
            digits.push(b'0');
        } else {
            while !value.is_zero() {
                let (quotient, remainder) = value.div_rem_small_abs(radix);
                digits.push(alphabet[remainder as usize]);
                value = quotient;
            }
            digits.reverse();
        }

        while digits.len() < min_digits {
            digits.insert(0, b'0');
        }

        let separator_count = if group == 0 {
            0
        } else {
            (digits.len() - 1) / group
        };
        let mut result = String::with_capacity(
            digits.len() + separator_count + usize::from(self.negative),
        );
        if self.negative {
            result.push('-');
        }

        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                result.push('_');
            }
            result.push(digit as char);
        }
        Ok(result)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = self.add_abs(rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Greater => {
                    let mut result = self.sub_abs(rhs);
                    result.negative = self.negative && !result.is_zero();
                    result
                }
                Ordering::Less => {
                    let mut result = rhs.sub_abs(self);
                    result.negative = rhs.negative && !result.is_zero();
                    result
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut limbs = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value = limbs[index] as u64 + a as u64 * b as u64 + carry;
                limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }

            let mut index = i + rhs.limbs.len();
            while carry != 0 {
                if index == limbs.len() {
                    limbs.push(0);
                }
                let value = limbs[index] as u64 + carry;
                limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                index += 1;
            }
        }

        let mut result = Self {
            negative: self.negative != rhs.negative,
            limbs,
        };
        result.normalize();
        result
    }

    fn div_rem_abs(&self, rhs: &Self) -> (Self, Self) {
        if self.cmp_abs(rhs) == Ordering::Less {
            return (Self::zero(), self.abs());
        }

        let mut quotient_be = Vec::with_capacity(self.limbs.len());
        let mut remainder = Self::zero();

        for &limb in self.limbs.iter().rev() {
            if !remainder.is_zero() {
                remainder.limbs.insert(0, limb);
            } else if limb != 0 {
                remainder.limbs.push(limb);
            }

            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let product = rhs.mul_small_value(mid as u32);
                match product.cmp_abs(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }

            quotient_be.push(best as u32);
            if best != 0 {
                let product = rhs.mul_small_value(best as u32);
                remainder = remainder.sub_abs(&product);
            }
        }

        quotient_be.reverse();
        let mut quotient = Self {
            negative: false,
            limbs: quotient_be,
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }

    fn mul_small_value(&self, factor: u32) -> Self {
        let mut result = self.abs();
        result.mul_small_abs(factor);
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = self.div_rem_abs(rhs);
        quotient.negative = (self.negative != rhs.negative) && !quotient.is_zero();
        remainder.negative = self.negative && !remainder.is_zero();
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }

        let mut exp = exponent;
        let mut base = self.clone();
        let mut result = Self::one();

        while exp != 0 {
            if exp & 1 == 1 {
                result = result.mul(&base);
            }
            exp >>= 1;
            if exp != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();

        while !b.is_zero() {
            let (_, remainder) = a.div_rem_abs(&b);
            a = b;
            b = remainder;
        }

        a
    }
}
```
