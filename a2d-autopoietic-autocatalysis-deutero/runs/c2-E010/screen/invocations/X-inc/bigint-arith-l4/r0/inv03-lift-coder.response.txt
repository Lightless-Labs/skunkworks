```rust
use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>,
}

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            digits: vec![1],
        }
    }

    fn from_digits(negative: bool, mut digits: Vec<u8>) -> Self {
        let first = digits.iter().position(|&d| d != 0);
        match first {
            Some(i) => {
                if i > 0 {
                    digits.drain(..i);
                }
                Self { negative, digits }
            }
            None => Self::zero(),
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        self.digits
            .len()
            .cmp(&rhs.digits.len())
            .then_with(|| self.digits.cmp(&rhs.digits))
    }

    fn abs_add(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let (mut i, mut j, mut carry) = (a.len(), b.len(), 0u16);
        while i > 0 || j > 0 || carry > 0 {
            let av = if i > 0 { i -= 1; a[i] as u16 } else { 0 };
            let bv = if j > 0 { j -= 1; b[j] as u16 } else { 0 };
            let sum = av + bv + carry;
            out.push((sum % 10) as u8);
            carry = sum / 10;
        }
        out.reverse();
        out
    }

    // Requires a >= b.
    fn abs_sub(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());
        let (mut i, mut j, mut borrow) = (a.len(), b.len(), 0i16);
        while i > 0 {
            i -= 1;
            let av = a[i] as i16 - borrow;
            let bv = if j > 0 { j -= 1; b[j] as i16 } else { 0 };
            if av < bv {
                out.push((av + 10 - bv) as u8);
                borrow = 1;
            } else {
                out.push((av - bv) as u8);
                borrow = 0;
            }
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out.reverse();
        out
    }

    fn abs_mul(a: &[u8], b: &[u8]) -> Vec<u8> {
        if a.len() == 1 && a[0] == 0 || b.len() == 1 && b[0] == 0 {
            return vec![0];
        }
        let mut accum = vec![0u32; a.len() + b.len()];
        for (i, &av) in a.iter().enumerate() {
            for (j, &bv) in b.iter().enumerate() {
                accum[i + j + 1] += av as u32 * bv as u32;
            }
        }
        for k in (1..accum.len()).rev() {
            let carry = accum[k] / 10;
            accum[k] %= 10;
            accum[k - 1] += carry;
        }
        let digits: Vec<u8> = accum.into_iter().map(|x| x as u8).collect();
        let first = digits.iter().position(|&d| d != 0).unwrap_or(digits.len() - 1);
        digits[first..].to_vec()
    }

    fn abs_mul_small(a: &[u8], m: u8) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u16;
        for &d in a.iter().rev() {
            let x = d as u16 * m as u16 + carry;
            out.push((x % 10) as u8);
            carry = x / 10;
        }
        while carry > 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        out.reverse();
        Self::from_digits(false, out).digits
    }

    fn abs_div_rem(a: &[u8], b: &[u8]) -> (Vec<u8>, Vec<u8>) {
        let mut quotient = Vec::with_capacity(a.len());
        let mut remainder = vec![0];
        for &digit in a {
            if remainder.len() == 1 && remainder[0] == 0 {
                remainder[0] = digit;
            } else {
                remainder.push(digit);
            }
            let first = remainder.iter().position(|&d| d != 0);
            if let Some(i) = first {
                if i > 0 {
                    remainder.drain(..i);
                }
            } else {
                remainder = vec![0];
            }
            let mut q = 0;
            while Self::cmp_digits(&remainder, b) != Ordering::Less {
                remainder = Self::abs_sub(&remainder, b);
                q += 1;
            }
            quotient.push(q);
        }
        let quotient = Self::from_digits(false, quotient).digits;
        let remainder = Self::from_digits(false, remainder).digits;
        (quotient, remainder)
    }

    fn cmp_digits(a: &[u8], b: &[u8]) -> Ordering {
        a.len().cmp(&b.len()).then_with(|| a.cmp(b))
    }

    fn mul_small_add(a: &[u8], m: u8, add: u8) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = add as u16;
        for &d in a.iter().rev() {
            let x = d as u16 * m as u16 + carry;
            out.push((x % 10) as u8);
            carry = x / 10;
        }
        while carry > 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        out.reverse();
        Self::from_digits(false, out).digits
    }

    fn div_small(a: &[u8], divisor: u8) -> (Vec<u8>, u8) {
        let mut quotient = Vec::with_capacity(a.len());
        let mut remainder = 0u16;
        for &d in a {
            let x = remainder * 10 + d as u16;
            quotient.push((x / divisor as u16) as u8);
            remainder = x % divisor as u16;
        }
        (Self::from_digits(false, quotient).digits, remainder as u8)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    fn digit_char(digit: u32) -> char {
        match digit {
            0..=9 => (b'0' + digit as u8) as char,
            _ => (b'a' + (digit - 10) as u8) as char,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        let mut digit_positions = Vec::new();
        for index in start..bytes.len() {
            if bytes[index] == b'_' {
                continue;
            }
            match Self::digit_value(bytes[index]) {
                Some(value) if value < radix => digit_positions.push(index),
                _ => {
                    if invalid.is_none() {
                        invalid = Some(index);
                    }
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || bytes[index - 1] == b'_'
                    || bytes[index + 1] == b'_')
            {
                return Err(Error::Separator { index });
            }
        }

        if digit_positions.is_empty() {
            // With no invalid bytes, the magnitude can only consist of underscores.
            return Err(Error::Separator { index: start });
        }

        if digit_positions.len() > 1 && bytes[digit_positions[0]] == b'0' {
            return Err(Error::LeadingZero {
                index: digit_positions[0],
            });
        }

        if digit_positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_positions[max_digits],
            });
        }

        let mut value = vec![0u8];
        for &index in &digit_positions {
            let digit = Self::digit_value(bytes[index]).unwrap() as u8;
            value = Self::mul_small_add(&value, radix as u8, digit);
        }
        Ok(Self::from_digits(negative, value))
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut digits = if self.is_zero() {
            vec![0]
        } else {
            let mut magnitude = self.digits.clone();
            let mut out = Vec::new();
            while !(magnitude.len() == 1 && magnitude[0] == 0) {
                let (quotient, remainder) = Self::div_small(&magnitude, radix as u8);
                out.push(remainder as u32);
                magnitude = quotient;
            }
            out.reverse();
            out
        };
        if digits.len() < min_digits {
            let mut padded = vec![0; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut result = String::new();
        if self.negative && !self.is_zero() {
            result.push('-');
        }
        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                result.push('_');
            }
            result.push(Self::digit_char(digit));
        }
        Ok(result)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            return Self::from_digits(
                self.negative,
                Self::abs_add(&self.digits, &rhs.digits),
            );
        }
        match self.abs_cmp(rhs) {
            Ordering::Greater => Self::from_digits(
                self.negative,
                Self::abs_sub(&self.digits, &rhs.digits),
            ),
            Ordering::Less => Self::from_digits(
                rhs.negative,
                Self::abs_sub(&rhs.digits, &self.digits),
            ),
            Ordering::Equal => Self::zero(),
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        Self::from_digits(
            self.negative ^ rhs.negative,
            Self::abs_mul(&self.digits, &rhs.digits),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (q, r) = Self::abs_div_rem(&self.digits, &rhs.digits);
        Ok((
            Self::from_digits(self.negative ^ rhs.negative, q),
            Self::from_digits(self.negative, r),
        ))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: LIMIT,
            });
        }
        let mut n = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while n > 0 {
            if n & 1 == 1 {
                result = result.mul(&base);
            }
            n >>= 1;
            if n > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = Self::from_digits(false, self.digits.clone());
        let mut b = Self::from_digits(false, rhs.digits.clone());
        while !b.is_zero() {
            let (_, remainder) = a.div_rem(&b).unwrap();
            a = b;
            b = Self::from_digits(false, remainder.digits);
        }
        a
    }
}
```
