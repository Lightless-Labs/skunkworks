    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs. Zero is represented by [].
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    fn zero() -> Self {
        Self { negative: false, limbs: Vec::new() }
    }

    fn one() -> Self {
        Self { negative: false, limbs: vec![1] }
    }

    fn from_u32(n: u32) -> Self {
        if n == 0 { Self::zero() } else { Self { negative: false, limbs: vec![n] } }
    }

    fn is_zero(&self) -> bool {
        self.limbs.is_empty()
    }

    fn normalize(&mut self) {
        while self.limbs.last() == Some(&0) {
            self.limbs.pop();
        }
        if self.is_zero() {
            self.negative = false;
        }
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs.len().cmp(&rhs.limbs.len()).then_with(|| {
            self.limbs.iter().rev().cmp(rhs.limbs.iter().rev())
        })
    }

    fn add_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let mut carry = 0u64;
        for i in 0..a.len().max(b.len()) {
            let sum = a.get(i).copied().unwrap_or(0) as u64
                + b.get(i).copied().unwrap_or(0) as u64 + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 { out.push(carry as u32); }
        out
    }

    // Requires a >= b.
    fn sub_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        let mut out = Vec::with_capacity(a.len());
        let mut borrow = 0i64;
        for i in 0..a.len() {
            let mut v = a[i] as i64 - b.get(i).copied().unwrap_or(0) as i64 - borrow;
            if v < 0 {
                v += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(v as u32);
        }
        while out.last() == Some(&0) { out.pop(); }
        out
    }

    fn mul_abs_small(a: &[u32], m: u32) -> Vec<u32> {
        if m == 0 || a.is_empty() { return Vec::new(); }
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u64;
        for &x in a {
            let v = x as u64 * m as u64 + carry;
            out.push((v % Self::BASE) as u32);
            carry = v / Self::BASE;
        }
        if carry != 0 { out.push(carry as u32); }
        out
    }

    fn add_small_abs(a: &mut Vec<u32>, n: u32) {
        let mut carry = n as u64;
        let mut i = 0;
        while carry != 0 {
            if i == a.len() { a.push(0); }
            let v = a[i] as u64 + carry;
            a[i] = (v % Self::BASE) as u32;
            carry = v / Self::BASE;
            i += 1;
        }
    }

    fn mul_abs(a: &[u32], b: &[u32]) -> Vec<u32> {
        if a.is_empty() || b.is_empty() { return Vec::new(); }
        let mut out = vec![0u32; a.len() + b.len()];
        for (i, &x) in a.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &y) in b.iter().enumerate() {
                let k = i + j;
                let v = out[k] as u64 + x as u64 * y as u64 + carry;
                out[k] = (v % Self::BASE) as u32;
                carry = v / Self::BASE;
            }
            let mut k = i + b.len();
            while carry != 0 {
                if k == out.len() { out.push(0); }
                let v = out[k] as u64 + carry;
                out[k] = (v % Self::BASE) as u32;
                carry = v / Self::BASE;
                k += 1;
            }
        }
        while out.last() == Some(&0) { out.pop(); }
        out
    }

    fn div_rem_abs(a: &[u32], b: &[u32]) -> (Vec<u32>, Vec<u32>) {
        debug_assert!(!b.is_empty());
        if a.is_empty() { return (Vec::new(), Vec::new()); }
        let mut quotient_be = Vec::with_capacity(a.len());
        let mut rem = Vec::<u32>::new();

        for &digit in a.iter().rev() {
            rem.insert(0, digit);
            while rem.last() == Some(&0) { rem.pop(); }

            let mut lo = 0u64;
            let mut hi = Self::BASE - 1;
            let mut best = 0u64;
            while lo <= hi {
                let mid = lo + (hi - lo) / 2;
                let product = Self::mul_abs_small(b, mid as u32);
                match Self::cmp_slices(&product, &rem) {
                    Ordering::Greater => {
                        if mid == 0 { break; }
                        hi = mid - 1;
                    }
                    _ => {
                        best = mid;
                        lo = mid + 1;
                    }
                }
            }
            quotient_be.push(best as u32);
            if best != 0 {
                let product = Self::mul_abs_small(b, best as u32);
                rem = Self::sub_abs(&rem, &product);
            }
        }
        quotient_be.reverse();
        while quotient_be.last() == Some(&0) { quotient_be.pop(); }
        (quotient_be, rem)
    }

    fn cmp_slices(a: &[u32], b: &[u32]) -> Ordering {
        a.len().cmp(&b.len()).then_with(|| a.iter().rev().cmp(b.iter().rev()))
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }
        let (start, negative) = match bytes[0] {
            b'+' => (1, false),
            b'-' => (1, true),
            _ => (0, false),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (i, &byte) in bytes.iter().enumerate().skip(start) {
            if byte != b'_' && !matches!(Self::digit_value(byte), Some(v) if v < radix) {
                invalid = Some(i);
                break;
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        let mut malformed_separator = None;
        for i in start..bytes.len() {
            if bytes[i] == b'_' {
                let prev_ok = i > start
                    && matches!(Self::digit_value(bytes[i - 1]), Some(v) if v < radix);
                let next_ok = i + 1 < bytes.len()
                    && matches!(Self::digit_value(bytes[i + 1]), Some(v) if v < radix);
                if !(prev_ok && next_ok) {
                    malformed_separator = Some(i);
                    break;
                }
            }
        }
        if let Some(index) = malformed_separator {
            return Err(Error::Separator { index });
        }

        let digits: Vec<(usize, u32)> = (start..bytes.len())
            .filter(|&i| bytes[i] != b'_')
            .map(|i| (i, Self::digit_value(bytes[i]).unwrap()))
            .collect();

        if digits.len() > 1 && digits[0].1 == 0 {
            return Err(Error::LeadingZero { index: digits[0].0 });
        }
        if digits.len() > max_digits {
            return Err(Error::TooManyDigits { limit: max_digits, index: digits[max_digits].0 });
        }

        let mut out = Self::zero();
        for &(_, d) in &digits {
            out.limbs = Self::mul_abs_small(&out.limbs, radix);
            Self::add_small_abs(&mut out.limbs, d);
        }
        out.negative = negative && !out.is_zero();
        Ok(out)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut n = self.limbs.clone();
        let mut digits = Vec::<u8>::new();
        if n.is_empty() {
            digits.push(b'0');
        } else {
            while !n.is_empty() {
                let mut rem = 0u64;
                for limb in n.iter_mut().rev() {
                    let cur = rem * Self::BASE + *limb as u64;
                    *limb = (cur / radix as u64) as u32;
                    rem = cur % radix as u64;
                }
                while n.last() == Some(&0) { n.pop(); }
                digits.push(if rem < 10 { b'0' + rem as u8 } else { b'a' + (rem as u8 - 10) });
            }
            digits.reverse();
        }
        if digits.len() < min_digits {
            let mut padded = vec![b'0'; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut out = String::with_capacity(digits.len() + digits.len() / group.max(1) + 1);
        if self.negative && !self.is_zero() { out.push('-'); }
        for (i, &d) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(d as char);
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => self.cmp_abs(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut out = Self { negative: self.negative, limbs: Self::add_abs(&self.limbs, &rhs.limbs) };
            out.normalize();
            out
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut out = Self { negative: self.negative, limbs: Self::sub_abs(&self.limbs, &rhs.limbs) };
                    out.normalize();
                    out
                }
                Ordering::Less => {
                    let mut out = Self { negative: rhs.negative, limbs: Self::sub_abs(&rhs.limbs, &self.limbs) };
                    out.normalize();
                    out
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut out = self.clone();
        if !out.is_zero() { out.negative = !out.negative; }
        out
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() { 0 } else if self.negative { -1 } else { 1 }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        let mut out = Self {
            negative: self.negative ^ rhs.negative,
            limbs: Self::mul_abs(&self.limbs, &rhs.limbs),
        };
        out.normalize();
        out
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (q, r) = Self::div_rem_abs(&self.limbs, &rhs.limbs);
        let mut quotient = Self { negative: self.negative ^ rhs.negative, limbs: q };
        let mut remainder = Self { negative: self.negative, limbs: r };
        quotient.normalize();
        remainder.normalize();
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge { exponent, limit: LIMIT });
        }
        let mut e = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while e != 0 {
            if e & 1 == 1 { result = result.mul(&base); }
            e >>= 1;
            if e != 0 { base = base.mul(&base); }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = Self { negative: false, limbs: self.limbs.clone() };
        let mut b = Self { negative: false, limbs: rhs.limbs.clone() };
        while !b.is_zero() {
            let (_, r) = a.div_rem(&b).expect("nonzero gcd divisor");
            a = b;
            b = r;
        }
        a
    }

    fn abs(&self) -> Self {
        let mut result = self.clone();
        result.negative = false;
        result
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        self.limbs
            .len()
            .cmp(&rhs.limbs.len())
            .then_with(|| self.limbs.iter().rev().cmp(rhs.limbs.iter().rev()))
    }

    fn add_abs(&self, rhs: &Self) -> Self {
        let n = self.limbs.len().max(rhs.limbs.len());
        let mut limbs = Vec::with_capacity(n + 1);
        let mut carry = 0u64;

        for i in 0..n {
            let a = self.limbs.get(i).copied().unwrap_or(0) as u64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = a + b + carry;
            limbs.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            limbs.push(carry as u32);
        }

        Self {
            negative: false,
            limbs,
        }
    }

    // Requires |self| >= |rhs|.
    fn sub_abs(&self, rhs: &Self) -> Self {
        let mut limbs = Vec::with_capacity(self.limbs.len());
        let mut borrow = 0i64;

        for i in 0..self.limbs.len() {
            let a = self.limbs[i] as i64;
            let b = rhs.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut value = a - b - borrow;
            if value < 0 {
                value += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            limbs.push(value as u32);
        }

        let mut result = Self {
            negative: false,
            limbs,
        };
        result.normalize();
        result
    }

    fn mul_small_abs(&mut self, factor: u32) {
        if factor == 0 || self.is_zero() {
            self.limbs.clear();
            self.negative = false;
            return;
        }

        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * factor as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_abs(&mut self, addend: u32) {
        let mut carry = addend as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let value = self.limbs[i] as u64 + carry;
            self.limbs[i] = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
            i += 1;
        }
    }

    fn div_rem_small_abs(&self, divisor: u32) -> (Self, u32) {
        let mut quotient = vec![0u32; self.limbs.len()];
        let mut remainder = 0u64;

        for i in (0..self.limbs.len()).rev() {
            let value = remainder * Self::BASE + self.limbs[i] as u64;
            quotient[i] = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }

        let mut q = Self {
            negative: false,
            limbs: quotient,
        };
        q.normalize();
        (q, remainder as u32)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut first_invalid = None;
        let mut digit_count = 0usize;
        let mut first_digit_index = None;
        let mut first_digit_value = None;
        let mut previous_is_digit = false;
        let mut first_bad_separator = None;
        let mut first_excess = None;
        let mut digits_seen = 0usize;

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                if !previous_is_digit {
                    first_bad_separator.get_or_insert(index);
                }
                previous_is_digit = false;
                continue;
            }

            match Self::digit_value(byte) {
                Some(value) if value < radix => {
                    digit_count += 1;
                    if first_digit_index.is_none() {
                        first_digit_index = Some(index);
                        first_digit_value = Some(value);
                    }
                    digits_seen += 1;
                    if digits_seen == max_digits.saturating_add(1) {
                        first_excess = Some(index);
                    }
                    previous_is_digit = true;
                }
                _ => {
                    first_invalid.get_or_insert(index);
                    previous_is_digit = false;
                }
            }
        }

        if let Some(index) = first_invalid {
            return Err(Error::InvalidDigit { index });
        }

        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                let left_ok = index > start
                    && Self::digit_value(bytes[index - 1])
                        .is_some_and(|v| v < radix);
                let right_ok = index + 1 < bytes.len()
                    && Self::digit_value(bytes[index + 1])
                        .is_some_and(|v| v < radix);
                if !left_ok || !right_ok {
                    first_bad_separator.get_or_insert(index);
                }
            }
        }

        if let Some(index) = first_bad_separator {
            return Err(Error::Separator { index });
        }

        if digit_count > 1 && first_digit_value == Some(0) {
            return Err(Error::LeadingZero {
                index: first_digit_index.unwrap(),
            });
        }

        if digit_count > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: first_excess.unwrap(),
            });
        }

        let mut result = Self::zero();
        for &byte in bytes.iter().skip(start) {
            if byte == b'_' {
                continue;
            }
            let digit = Self::digit_value(byte).unwrap();
            result.mul_small_abs(radix);
            result.add_small_abs(digit);
        }
        result.negative = negative && !result.is_zero();
        Ok(result)
    }

    pub fn format(
        &self,
        radix: u32,
        min_digits: usize,
        group: usize,
    ) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let alphabet = b"0123456789abcdefghijklmnopqrstuvwxyz";
        let mut value = self.abs();
        let mut digits = Vec::new();

        if value.is_zero() {
            digits.push(b'0');
        } else {
            while !value.is_zero() {
                let (quotient, remainder) = value.div_rem_small_abs(radix);
                digits.push(alphabet[remainder as usize]);
                value = quotient;
            }
            digits.reverse();
        }

        while digits.len() < min_digits {
            digits.insert(0, b'0');
        }

        let separator_count = if group == 0 {
            0
        } else {
            (digits.len() - 1) / group
        };
        let mut result = String::with_capacity(
            digits.len() + separator_count + usize::from(self.negative),
        );
        if self.negative {
            result.push('-');
        }

        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                result.push('_');
            }
            result.push(digit as char);
        }
        Ok(result)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = self.add_abs(rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Greater => {
                    let mut result = self.sub_abs(rhs);
                    result.negative = self.negative && !result.is_zero();
                    result
                }
                Ordering::Less => {
                    let mut result = rhs.sub_abs(self);
                    result.negative = rhs.negative && !result.is_zero();
                    result
                }
                Ordering::Equal => Self::zero(),
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }

        let mut limbs = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value = limbs[index] as u64 + a as u64 * b as u64 + carry;
                limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }

            let mut index = i + rhs.limbs.len();
            while carry != 0 {
                if index == limbs.len() {
                    limbs.push(0);
                }
                let value = limbs[index] as u64 + carry;
                limbs[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                index += 1;
            }
        }

        let mut result = Self {
            negative: self.negative != rhs.negative,
            limbs,
        };
        result.normalize();
        result
    }

    fn div_rem_abs(&self, rhs: &Self) -> (Self, Self) {
        if self.cmp_abs(rhs) == Ordering::Less {
            return (Self::zero(), self.abs());
        }

        let mut quotient_be = Vec::with_capacity(self.limbs.len());
        let mut remainder = Self::zero();

        for &limb in self.limbs.iter().rev() {
            if !remainder.is_zero() {
                remainder.limbs.insert(0, limb);
            } else if limb != 0 {
                remainder.limbs.push(limb);
            }

            let mut low = 0u64;
            let mut high = Self::BASE - 1;
            let mut best = 0u64;
            while low <= high {
                let mid = (low + high) / 2;
                let product = rhs.mul_small_value(mid as u32);
                match product.cmp_abs(&remainder) {
                    Ordering::Greater => {
                        if mid == 0 {
                            break;
                        }
                        high = mid - 1;
                    }
                    _ => {
                        best = mid;
                        low = mid + 1;
                    }
                }
            }

            quotient_be.push(best as u32);
            if best != 0 {
                let product = rhs.mul_small_value(best as u32);
                remainder = remainder.sub_abs(&product);
            }
        }

        quotient_be.reverse();
        let mut quotient = Self {
            negative: false,
            limbs: quotient_be,
        };
        quotient.normalize();
        remainder.normalize();
        (quotient, remainder)
    }

    fn mul_small_value(&self, factor: u32) -> Self {
        let mut result = self.abs();
        result.mul_small_abs(factor);
        result
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (mut quotient, mut remainder) = self.div_rem_abs(rhs);
        quotient.negative = (self.negative != rhs.negative) && !quotient.is_zero();
        remainder.negative = self.negative && !remainder.is_zero();
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        if exponent > Self::POW_LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: Self::POW_LIMIT,
            });
        }

        let mut exp = exponent;
        let mut base = self.clone();
        let mut result = Self::one();

        while exp != 0 {
            if exp & 1 == 1 {
                result = result.mul(&base);
            }
            exp >>= 1;
            if exp != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();

        while !b.is_zero() {
            let (_, remainder) = a.div_rem_abs(&b);
            a = b;
            b = remainder;
        }

        a
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,

}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    digits: Vec<u8>,
}

impl Big {
    fn zero() -> Self {
        Self {
            negative: false,
            digits: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            digits: vec![1],
        }
    }

    fn from_digits(negative: bool, mut digits: Vec<u8>) -> Self {
        let first = digits.iter().position(|&d| d != 0);
        match first {
            Some(i) => {
                if i > 0 {
                    digits.drain(..i);
                }
                Self { negative, digits }
            }
            None => Self::zero(),
        }
    }

    fn is_zero(&self) -> bool {
        self.digits.len() == 1 && self.digits[0] == 0
    }

    fn abs_cmp(&self, rhs: &Self) -> Ordering {
        self.digits
            .len()
            .cmp(&rhs.digits.len())
            .then_with(|| self.digits.cmp(&rhs.digits))
    }

    fn abs_add(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len().max(b.len()) + 1);
        let (mut i, mut j, mut carry) = (a.len(), b.len(), 0u16);
        while i > 0 || j > 0 || carry > 0 {
            let av = if i > 0 { i -= 1; a[i] as u16 } else { 0 };
            let bv = if j > 0 { j -= 1; b[j] as u16 } else { 0 };
            let sum = av + bv + carry;
            out.push((sum % 10) as u8);
            carry = sum / 10;
        }
        out.reverse();
        out
    }

    // Requires a >= b.
    fn abs_sub(a: &[u8], b: &[u8]) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len());
        let (mut i, mut j, mut borrow) = (a.len(), b.len(), 0i16);
        while i > 0 {
            i -= 1;
            let av = a[i] as i16 - borrow;
            let bv = if j > 0 { j -= 1; b[j] as i16 } else { 0 };
            if av < bv {
                out.push((av + 10 - bv) as u8);
                borrow = 1;
            } else {
                out.push((av - bv) as u8);
                borrow = 0;
            }
        }
        while out.len() > 1 && out.last() == Some(&0) {
            out.pop();
        }
        out.reverse();
        out
    }

    fn abs_mul(a: &[u8], b: &[u8]) -> Vec<u8> {
        if a.len() == 1 && a[0] == 0 || b.len() == 1 && b[0] == 0 {
            return vec![0];
        }
        let mut accum = vec![0u32; a.len() + b.len()];
        for (i, &av) in a.iter().enumerate() {
            for (j, &bv) in b.iter().enumerate() {
                accum[i + j + 1] += av as u32 * bv as u32;
            }
        }
        for k in (1..accum.len()).rev() {
            let carry = accum[k] / 10;
            accum[k] %= 10;
            accum[k - 1] += carry;
        }
        let digits: Vec<u8> = accum.into_iter().map(|x| x as u8).collect();
        let first = digits.iter().position(|&d| d != 0).unwrap_or(digits.len() - 1);
        digits[first..].to_vec()
    }

    fn abs_mul_small(a: &[u8], m: u8) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = 0u16;
        for &d in a.iter().rev() {
            let x = d as u16 * m as u16 + carry;
            out.push((x % 10) as u8);
            carry = x / 10;
        }
        while carry > 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        out.reverse();
        Self::from_digits(false, out).digits
    }

    fn abs_div_rem(a: &[u8], b: &[u8]) -> (Vec<u8>, Vec<u8>) {
        let mut quotient = Vec::with_capacity(a.len());
        let mut remainder = vec![0];
        for &digit in a {
            if remainder.len() == 1 && remainder[0] == 0 {
                remainder[0] = digit;
            } else {
                remainder.push(digit);
            }
            let first = remainder.iter().position(|&d| d != 0);
            if let Some(i) = first {
                if i > 0 {
                    remainder.drain(..i);
                }
            } else {
                remainder = vec![0];
            }
            let mut q = 0;
            while Self::cmp_digits(&remainder, b) != Ordering::Less {
                remainder = Self::abs_sub(&remainder, b);
                q += 1;
            }
            quotient.push(q);
        }
        let quotient = Self::from_digits(false, quotient).digits;
        let remainder = Self::from_digits(false, remainder).digits;
        (quotient, remainder)
    }

    fn cmp_digits(a: &[u8], b: &[u8]) -> Ordering {
        a.len().cmp(&b.len()).then_with(|| a.cmp(b))
    }

    fn mul_small_add(a: &[u8], m: u8, add: u8) -> Vec<u8> {
        let mut out = Vec::with_capacity(a.len() + 1);
        let mut carry = add as u16;
        for &d in a.iter().rev() {
            let x = d as u16 * m as u16 + carry;
            out.push((x % 10) as u8);
            carry = x / 10;
        }
        while carry > 0 {
            out.push((carry % 10) as u8);
            carry /= 10;
        }
        out.reverse();
        Self::from_digits(false, out).digits
    }

    fn div_small(a: &[u8], divisor: u8) -> (Vec<u8>, u8) {
        let mut quotient = Vec::with_capacity(a.len());
        let mut remainder = 0u16;
        for &d in a {
            let x = remainder * 10 + d as u16;
            quotient.push((x / divisor as u16) as u8);
            remainder = x % divisor as u16;
        }
        (Self::from_digits(false, quotient).digits, remainder as u8)
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    fn digit_char(digit: u32) -> char {
        match digit {
            0..=9 => (b'0' + digit as u8) as char,
            _ => (b'a' + (digit - 10) as u8) as char,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        let mut digit_positions = Vec::new();
        for index in start..bytes.len() {
            if bytes[index] == b'_' {
                continue;
            }
            match Self::digit_value(bytes[index]) {
                Some(value) if value < radix => digit_positions.push(index),
                _ => {
                    if invalid.is_none() {
                        invalid = Some(index);
                    }
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || bytes[index - 1] == b'_'
                    || bytes[index + 1] == b'_')
            {
                return Err(Error::Separator { index });
            }
        }

        if digit_positions.is_empty() {
            // With no invalid bytes, the magnitude can only consist of underscores.
            return Err(Error::Separator { index: start });
        }

        if digit_positions.len() > 1 && bytes[digit_positions[0]] == b'0' {
            return Err(Error::LeadingZero {
                index: digit_positions[0],
            });
        }

        if digit_positions.len() > max_digits {
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: digit_positions[max_digits],
            });
        }

        let mut value = vec![0u8];
        for &index in &digit_positions {
            let digit = Self::digit_value(bytes[index]).unwrap() as u8;
            value = Self::mul_small_add(&value, radix as u8, digit);
        }
        Ok(Self::from_digits(negative, value))
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let mut digits = if self.is_zero() {
            vec![0]
        } else {
            let mut magnitude = self.digits.clone();
            let mut out = Vec::new();
            while !(magnitude.len() == 1 && magnitude[0] == 0) {
                let (quotient, remainder) = Self::div_small(&magnitude, radix as u8);
                out.push(remainder as u32);
                magnitude = quotient;
            }
            out.reverse();
            out
        };
        if digits.len() < min_digits {
            let mut padded = vec![0; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut result = String::new();
        if self.negative && !self.is_zero() {
            result.push('-');
        }
        for (i, &digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                result.push('_');
            }
            result.push(Self::digit_char(digit));
        }
        Ok(result)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.abs_cmp(rhs),
            (true, true) => self.abs_cmp(rhs).reverse(),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            return Self::from_digits(
                self.negative,
                Self::abs_add(&self.digits, &rhs.digits),
            );
        }
        match self.abs_cmp(rhs) {
            Ordering::Greater => Self::from_digits(
                self.negative,
                Self::abs_sub(&self.digits, &rhs.digits),
            ),
            Ordering::Less => Self::from_digits(
                rhs.negative,
                Self::abs_sub(&rhs.digits, &self.digits),
            ),
            Ordering::Equal => Self::zero(),
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        if self.is_zero() {
            Self::zero()
        } else {
            Self {
                negative: !self.negative,
                digits: self.digits.clone(),
            }
        }
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        Self::from_digits(
            self.negative ^ rhs.negative,
            Self::abs_mul(&self.digits, &rhs.digits),
        )
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let (q, r) = Self::abs_div_rem(&self.digits, &rhs.digits);
        Ok((
            Self::from_digits(self.negative ^ rhs.negative, q),
            Self::from_digits(self.negative, r),
        ))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: LIMIT,
            });
        }
        let mut n = exponent;
        let mut base = self.clone();
        let mut result = Self::one();
        while n > 0 {
            if n & 1 == 1 {
                result = result.mul(&base);
            }
            n >>= 1;
            if n > 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = Self::from_digits(false, self.digits.clone());
        let mut b = Self::from_digits(false, rhs.digits.clone());
        while !b.is_zero() {
            let (_, remainder) = a.div_rem(&b).unwrap();
            a = b;
            b = Self::from_digits(false, remainder.digits);
        }
        a
    }
}

use std::cmp::Ordering;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum Error {
    Empty,
    SignOnly,
    InvalidDigit { index: usize },
    Separator { index: usize },
    LeadingZero { index: usize },
    TooManyDigits { limit: usize, index: usize },
    DivisionByZero,
    InvalidRadix { radix: u32 },
    ExponentTooLarge { exponent: u32, limit: u32 },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Big {
    negative: bool,
    // Little-endian base-1_000_000_000 limbs.
    limbs: Vec<u32>,
}

impl Big {
    const BASE: u64 = 1_000_000_000;

    fn zero() -> Self {
        Self {
            negative: false,
            limbs: vec![0],
        }
    }

    fn one() -> Self {
        Self {
            negative: false,
            limbs: vec![1],
        }
    }

    fn from_limbs(negative: bool, mut limbs: Vec<u32>) -> Self {
        while limbs.len() > 1 && limbs.last() == Some(&0) {
            limbs.pop();
        }
        if limbs.is_empty() {
            limbs.push(0);
        }
        let is_zero = limbs.len() == 1 && limbs[0] == 0;
        Self {
            negative: negative && !is_zero,
            limbs,
        }
    }

    fn is_zero(&self) -> bool {
        self.limbs.len() == 1 && self.limbs[0] == 0
    }

    fn abs(&self) -> Self {
        Self::from_limbs(false, self.limbs.clone())
    }

    fn cmp_abs(&self, rhs: &Self) -> Ordering {
        match self.limbs.len().cmp(&rhs.limbs.len()) {
            Ordering::Equal => {
                for (a, b) in self.limbs.iter().rev().zip(rhs.limbs.iter().rev()) {
                    match a.cmp(b) {
                        Ordering::Equal => {}
                        other => return other,
                    }
                }
                Ordering::Equal
            }
            other => other,
        }
    }

    fn add_abs(a: &Self, b: &Self) -> Self {
        let len = a.limbs.len().max(b.limbs.len());
        let mut out = Vec::with_capacity(len + 1);
        let mut carry = 0u64;
        for i in 0..len {
            let av = a.limbs.get(i).copied().unwrap_or(0) as u64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as u64;
            let sum = av + bv + carry;
            out.push((sum % Self::BASE) as u32);
            carry = sum / Self::BASE;
        }
        if carry != 0 {
            out.push(carry as u32);
        }
        Self::from_limbs(false, out)
    }

    // Requires |a| >= |b|.
    fn sub_abs(a: &Self, b: &Self) -> Self {
        let mut out = Vec::with_capacity(a.limbs.len());
        let mut borrow = 0i64;
        for i in 0..a.limbs.len() {
            let av = a.limbs[i] as i64;
            let bv = b.limbs.get(i).copied().unwrap_or(0) as i64;
            let mut value = av - bv - borrow;
            if value < 0 {
                value += Self::BASE as i64;
                borrow = 1;
            } else {
                borrow = 0;
            }
            out.push(value as u32);
        }
        Self::from_limbs(false, out)
    }

    fn mul_small_abs(&mut self, multiplier: u32) {
        if multiplier == 0 || self.is_zero() {
            *self = Self::zero();
            return;
        }
        let mut carry = 0u64;
        for limb in &mut self.limbs {
            let value = *limb as u64 * multiplier as u64 + carry;
            *limb = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
        }
        while carry != 0 {
            self.limbs.push((carry % Self::BASE) as u32);
            carry /= Self::BASE;
        }
    }

    fn add_small_abs(&mut self, addend: u32) {
        let mut carry = addend as u64;
        let mut i = 0;
        while carry != 0 {
            if i == self.limbs.len() {
                self.limbs.push(0);
            }
            let value = self.limbs[i] as u64 + carry;
            self.limbs[i] = (value % Self::BASE) as u32;
            carry = value / Self::BASE;
            i += 1;
        }
    }

    fn div_rem_small_abs(&self, divisor: u32) -> (Self, u32) {
        let mut out = vec![0; self.limbs.len()];
        let mut remainder = 0u64;
        for i in (0..self.limbs.len()).rev() {
            let value = remainder * Self::BASE + self.limbs[i] as u64;
            out[i] = (value / divisor as u64) as u32;
            remainder = value % divisor as u64;
        }
        (Self::from_limbs(false, out), remainder as u32)
    }

    fn push_base_limb(&mut self, limb: u32) {
        if self.is_zero() {
            self.limbs[0] = limb;
        } else {
            self.limbs.insert(0, limb);
        }
    }

    fn digit_value(byte: u8) -> Option<u32> {
        match byte {
            b'0'..=b'9' => Some((byte - b'0') as u32),
            b'a'..=b'z' => Some((byte - b'a' + 10) as u32),
            b'A'..=b'Z' => Some((byte - b'A' + 10) as u32),
            _ => None,
        }
    }

    pub fn parse(text: &str, radix: u32, max_digits: usize) -> Result<Self, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }
        let bytes = text.as_bytes();
        if bytes.is_empty() {
            return Err(Error::Empty);
        }

        let (negative, start) = match bytes[0] {
            b'+' => (false, 1),
            b'-' => (true, 1),
            _ => (false, 0),
        };
        if start == bytes.len() {
            return Err(Error::SignOnly);
        }

        let mut invalid = None;
        for (index, &byte) in bytes.iter().enumerate().skip(start) {
            if byte == b'_' {
                continue;
            }
            match Self::digit_value(byte) {
                Some(value) if value < radix => {}
                _ => {
                    invalid = Some(index);
                    break;
                }
            }
        }
        if let Some(index) = invalid {
            return Err(Error::InvalidDigit { index });
        }

        for index in start..bytes.len() {
            if bytes[index] == b'_'
                && (index == start
                    || index + 1 == bytes.len()
                    || bytes[index - 1] == b'_'
                    || bytes[index + 1] == b'_')
            {
                return Err(Error::Separator { index });
            }
        }

        let digit_count = bytes[start..].iter().filter(|&&b| b != b'_').count();
        let first_digit_index = (start..bytes.len()).find(|&i| bytes[i] != b'_').unwrap();
        if digit_count > 1 && Self::digit_value(bytes[first_digit_index]) == Some(0) {
            return Err(Error::LeadingZero {
                index: first_digit_index,
            });
        }
        if digit_count > max_digits {
            let excess_ordinal = max_digits;
            let mut seen = 0usize;
            let mut excess_index = first_digit_index;
            for (i, &byte) in bytes.iter().enumerate().skip(start) {
                if byte != b'_' {
                    if seen == excess_ordinal {
                        excess_index = i;
                        break;
                    }
                    seen += 1;
                }
            }
            return Err(Error::TooManyDigits {
                limit: max_digits,
                index: excess_index,
            });
        }

        let mut value = Self::zero();
        for &byte in &bytes[start..] {
            if byte == b'_' {
                continue;
            }
            value.mul_small_abs(radix);
            value.add_small_abs(Self::digit_value(byte).unwrap());
        }
        value.negative = negative && !value.is_zero();
        Ok(value)
    }

    pub fn format(&self, radix: u32, min_digits: usize, group: usize) -> Result<String, Error> {
        if !(2..=36).contains(&radix) {
            return Err(Error::InvalidRadix { radix });
        }

        let mut value = self.abs();
        let mut digits = Vec::new();
        if value.is_zero() {
            digits.push(0u32);
        } else {
            while !value.is_zero() {
                let (quotient, remainder) = value.div_rem_small_abs(radix);
                digits.push(remainder);
                value = quotient;
            }
            digits.reverse();
        }

        if digits.len() < min_digits {
            let mut padded = vec![0; min_digits - digits.len()];
            padded.extend(digits);
            digits = padded;
        }

        let mut out = String::new();
        if self.negative && !self.is_zero() {
            out.push('-');
        }
        for (i, digit) in digits.iter().enumerate() {
            if group > 0 && i > 0 && (digits.len() - i) % group == 0 {
                out.push('_');
            }
            out.push(match *digit {
                0..=9 => (b'0' + *digit as u8) as char,
                _ => (b'a' + (*digit as u8 - 10)) as char,
            });
        }
        Ok(out)
    }

    pub fn cmp_value(&self, rhs: &Self) -> Ordering {
        match (self.negative, rhs.negative) {
            (true, false) => Ordering::Less,
            (false, true) => Ordering::Greater,
            (false, false) => self.cmp_abs(rhs),
            (true, true) => rhs.cmp_abs(self),
        }
    }

    pub fn add(&self, rhs: &Self) -> Self {
        if self.negative == rhs.negative {
            let mut result = Self::add_abs(self, rhs);
            result.negative = self.negative && !result.is_zero();
            result
        } else {
            match self.cmp_abs(rhs) {
                Ordering::Equal => Self::zero(),
                Ordering::Greater => {
                    let mut result = Self::sub_abs(self, rhs);
                    result.negative = self.negative && !result.is_zero();
                    result
                }
                Ordering::Less => {
                    let mut result = Self::sub_abs(rhs, self);
                    result.negative = rhs.negative && !result.is_zero();
                    result
                }
            }
        }
    }

    pub fn sub(&self, rhs: &Self) -> Self {
        self.add(&rhs.neg())
    }

    pub fn neg(&self) -> Self {
        let mut result = self.clone();
        if !result.is_zero() {
            result.negative = !result.negative;
        }
        result
    }

    pub fn signum(&self) -> i8 {
        if self.is_zero() {
            0
        } else if self.negative {
            -1
        } else {
            1
        }
    }

    pub fn mul(&self, rhs: &Self) -> Self {
        if self.is_zero() || rhs.is_zero() {
            return Self::zero();
        }
        let mut out = vec![0u32; self.limbs.len() + rhs.limbs.len()];
        for (i, &a) in self.limbs.iter().enumerate() {
            let mut carry = 0u64;
            for (j, &b) in rhs.limbs.iter().enumerate() {
                let index = i + j;
                let value = out[index] as u64 + a as u64 * b as u64 + carry;
                out[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
            }
            let mut index = i + rhs.limbs.len();
            while carry != 0 {
                if index == out.len() {
                    out.push(0);
                }
                let value = out[index] as u64 + carry;
                out[index] = (value % Self::BASE) as u32;
                carry = value / Self::BASE;
                index += 1;
            }
        }
        Self::from_limbs(self.negative ^ rhs.negative, out)
    }

    pub fn div_rem(&self, rhs: &Self) -> Result<(Self, Self), Error> {
        if rhs.is_zero() {
            return Err(Error::DivisionByZero);
        }
        let dividend = self.abs();
        let divisor = rhs.abs();
        if dividend.cmp_abs(&divisor) == Ordering::Less {
            return Ok((Self::zero(), self.clone()));
        }

        let mut quotient_be = Vec::with_capacity(dividend.limbs.len());
        let mut remainder = Self::zero();
        for &limb in dividend.limbs.iter().rev() {
            remainder.push_base_limb(limb);
            let mut low = 0u32;
            let mut high = (Self::BASE - 1) as u32;
            while low < high {
                let mid = low + (high - low + 1) / 2;
                let mut product = divisor.clone();
                product.mul_small_abs(mid);
                if product.cmp_abs(&remainder) != Ordering::Greater {
                    low = mid;
                } else {
                    high = mid - 1;
                }
            }
            quotient_be.push(low);
            if low != 0 {
                let mut product = divisor.clone();
                product.mul_small_abs(low);
                remainder = Self::sub_abs(&remainder, &product);
            }
        }
        quotient_be.reverse();
        let quotient = Self::from_limbs(self.negative ^ rhs.negative, quotient_be);
        remainder.negative = self.negative && !remainder.is_zero();
        Ok((quotient, remainder))
    }

    pub fn pow(&self, exponent: u32) -> Result<Self, Error> {
        const LIMIT: u32 = 1024;
        if exponent > LIMIT {
            return Err(Error::ExponentTooLarge {
                exponent,
                limit: LIMIT,
            });
        }
        let mut result = Self::one();
        let mut base = self.clone();
        let mut power = exponent;
        while power != 0 {
            if power & 1 != 0 {
                result = result.mul(&base);
            }
            power >>= 1;
            if power != 0 {
                base = base.mul(&base);
            }
        }
        Ok(result)
    }

    pub fn gcd(&self, rhs: &Self) -> Self {
        let mut a = self.abs();
        let mut b = rhs.abs();
        while !b.is_zero() {
            let (_, remainder) = a.div_rem(&b).expect("nonzero divisor");
            a = b;
            b = remainder.abs();
        }
        a
    }
