```rust
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

struct ServerState {
    id: u32,
    opens: u64,
    overhead: u64,
    opened: bool,
    idle_since: u64,
    busy_until: Option<u64>,
    customer: Option<u32>,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut customer_order: Vec<usize> = (0..customers.len()).collect();
    customer_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut states: Vec<ServerState> = servers
        .iter()
        .map(|server| ServerState {
            id: server.id,
            opens: server.opens,
            overhead: server.overhead,
            opened: false,
            idle_since: server.opens,
            busy_until: None,
            customer: None,
        })
        .collect();

    let mut events = Vec::new();
    let mut next_customer = 0;
    let mut waiting: Vec<usize> = Vec::new();
    let mut finished_count = 0;
    let mut now = 0u64;

    loop {
        let mut completions: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (state.busy_until == Some(now)).then_some(i)
            })
            .collect();
        completions.sort_by_key(|&i| states[i].id);

        for i in completions {
            let customer = states[i].customer.take().unwrap();
            events.push(Event::Finish {
                time: now,
                customer,
                server: states[i].id,
            });
            states[i].busy_until = None;
            states[i].idle_since = now;
            finished_count += 1;
        }

        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.opens == now).then_some(i)
            })
            .collect();
        opening.sort_by(|&a, &b| states[b].id.cmp(&states[a].id));

        for i in opening {
            states[i].opened = true;
            events.push(Event::Open {
                time: now,
                server: states[i].id,
            });
        }

        while next_customer < customer_order.len()
            && customers[customer_order[next_customer]].arrival == now
        {
            let index = customer_order[next_customer];
            events.push(Event::Arrive {
                time: now,
                customer: customers[index].id,
            });
            waiting.push(index);
            next_customer += 1;
        }

        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.busy_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.overhead.cmp(&b.overhead))
                        .then_with(|| b.id.cmp(&a.id))
                })
                .map(|(i, _)| i);

            let Some(server_index) = selected else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &states[server_index];

            let Some(after_service) = now.checked_add(customer.service) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                });
            };
            let Some(finish) = after_service.checked_add(server.overhead) else {
                return Err(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                });
            };

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                states[server_index].idle_since = now;
                finished_count += 1;
            } else {
                states[server_index].busy_until = Some(finish);
                states[server_index].customer = Some(customer.id);
            }
        }

        if finished_count == customers.len() {
            break;
        }

        let next_arrival = customer_order
            .get(next_customer)
            .map(|&i| customers[i].arrival);
        let next_open = states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.opens)
            .min();
        let next_finish = states.iter().filter_map(|state| state.busy_until).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&time| time > now)
            .min();

        match next_time {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}
```
