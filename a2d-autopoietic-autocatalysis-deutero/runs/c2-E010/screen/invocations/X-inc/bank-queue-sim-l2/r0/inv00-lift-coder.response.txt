#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open { time: u64, server: u32 },
    Arrive { time: u64, customer: u32 },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish { time: u64, customer: u32, server: u32 },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers { count: usize },
    ZeroServerId { index: usize },
    DuplicateServerId { index: usize, id: u32 },
    ZeroCustomerId { index: usize },
    DuplicateCustomerId { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    occupied: bool,
    idle_since: u64,
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    finish: u64,
    server_index: usize,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|prior| prior.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|prior| prior.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut customer_order: Vec<usize> = (0..customers.len()).collect();
    customer_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut server_states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            occupied: false,
        })
        .collect();

    let mut events = Vec::new();
    let mut waiting: Vec<usize> = Vec::new();
    let mut next_customer = 0;
    let mut jobs: Vec<Job> = Vec::new();
    let mut now = 0u64;

    loop {
        let mut finishing: Vec<usize> = jobs
            .iter()
            .enumerate()
            .filter_map(|(i, job)| (job.finish == now).then_some(i))
            .collect();
        finishing.sort_by_key(|&i| server_states[jobs[i].server_index].server.id);

        for i in finishing.into_iter().rev() {
            let job = jobs.remove(i);
            let server = &mut server_states[job.server_index];
            server.occupied = false;
            server.idle_since = now;
            events.push(Event::Finish {
                time: now,
                customer: customers[job.customer_index()].id,
                server: server.server.id,
            });
        }

        let mut opening: Vec<usize> = server_states
            .iter()
            .enumerate()
            .filter_map(|(i, state)| {
                (!state.opened && state.server.opens == now).then_some(i)
            })
            .collect();
        opening.sort_by_key(|&i| std::cmp::Reverse(server_states[i].server.id));
        for i in opening {
            let state = &mut server_states[i];
            state.opened = true;
            events.push(Event::Open {
                time: now,
                server: state.server.id,
            });
        }

        let mut arriving = Vec::new();
        while next_customer < customer_order.len()
            && customers[customer_order[next_customer]].arrival == now
        {
            arriving.push(customer_order[next_customer]);
            next_customer += 1;
        }
        for &i in &arriving {
            events.push(Event::Arrive {
                time: now,
                customer: customers[i].id,
            });
        }
        waiting.extend(arriving);

        loop {
            if waiting.is_empty() {
                break;
            }

            let Some(server_index) = server_states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && !state.occupied)
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(i, _)| i)
            else {
                break;
            };

            let customer_index = waiting.remove(0);
            let customer = &customers[customer_index];
            let server = &server_states[server_index].server;
            let finish = customer
                .service
                .checked_add(now)
                .and_then(|value| value.checked_add(server.overhead))
                .ok_or(Error::TimeOverflow {
                    customer: customer.id,
                    server: server.id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: customer.id,
                server: server.id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: customer.id,
                    server: server.id,
                });
                server_states[server_index].idle_since = now;
            } else {
                server_states[server_index].occupied = true;
                jobs.push(Job {
                    customer_index,
                    finish,
                    server_index,
                });
            }
        }

        if next_customer == customer_order.len()
            && server_states.iter().all(|state| state.opened)
            && jobs.is_empty()
            && waiting.is_empty()
        {
            break;
        }

        let next_arrival = customer_order
            .get(next_customer)
            .map(|&i| customers[i].arrival);
        let next_open = server_states
            .iter()
            .filter(|state| !state.opened)
            .map(|state| state.server.opens)
            .min();
        let next_finish = jobs.iter().map(|job| job.finish).min();

        let next_time = [next_arrival, next_open, next_finish]
            .into_iter()
            .flatten()
            .filter(|&time| time > now)
            .min();

        match next_time {
            Some(time) => now = time,
            None => break,
        }
    }

    Ok(events)
}

#[derive(Clone)]
struct Job {
    customer_index: usize,
    finish: u64,
    server_index: usize,
}
