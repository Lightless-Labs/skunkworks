#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Server {
    pub id: u32,
    pub opens: u64,
    pub overhead: u64,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Event {
    Open {
        time: u64,
        server: u32,
    },
    Arrive {
        time: u64,
        customer: u32,
    },
    Start {
        time: u64,
        customer: u32,
        server: u32,
        finish: u64,
    },
    Finish {
        time: u64,
        customer: u32,
        server: u32,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    NoServers,
    TooManyServers {
        count: usize,
    },
    ZeroServerId {
        index: usize,
    },
    DuplicateServerId {
        index: usize,
        id: u32,
    },
    ZeroCustomerId {
        index: usize,
    },
    DuplicateCustomerId {
        index: usize,
        id: u32,
    },
    TimeOverflow {
        customer: u32,
        server: u32,
        time: u64,
    },
}

#[derive(Clone)]
struct ServerState {
    server: Server,
    opened: bool,
    occupied_until: Option<u64>,
    idle_since: u64,
}

#[derive(Clone)]
struct Job {
    customer: Customer,
}

pub fn simulate(customers: &[Customer], servers: &[Server]) -> Result<Vec<Event>, Error> {
    if servers.is_empty() {
        return Err(Error::NoServers);
    }
    if servers.len() > 32 {
        return Err(Error::TooManyServers {
            count: servers.len(),
        });
    }

    for (index, server) in servers.iter().enumerate() {
        if server.id == 0 {
            return Err(Error::ZeroServerId { index });
        }
        if servers[..index].iter().any(|previous| previous.id == server.id) {
            return Err(Error::DuplicateServerId {
                index,
                id: server.id,
            });
        }
    }

    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroCustomerId { index });
        }
        if customers[..index]
            .iter()
            .any(|previous| previous.id == customer.id)
        {
            return Err(Error::DuplicateCustomerId {
                index,
                id: customer.id,
            });
        }
    }

    let mut states: Vec<ServerState> = servers
        .iter()
        .cloned()
        .map(|server| ServerState {
            idle_since: server.opens,
            server,
            opened: false,
            occupied_until: None,
        })
        .collect();

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| a.arrival.cmp(&b.arrival).then_with(|| b.id.cmp(&a.id)));
    let mut arrival_index = 0usize;
    let mut waiting: Vec<Job> = Vec::new();
    let mut events = Vec::new();
    let mut now = 0u64;

    loop {
        // Finish occupied servers in ascending server ID order.
        let mut finished: Vec<(u32, usize)> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (state.occupied_until == Some(now)).then_some((state.server.id, index))
            })
            .collect();
        finished.sort_by_key(|&(id, _)| id);
        for (_, index) in finished {
            let state = &mut states[index];
            let customer_id = state.occupied_until.take().unwrap();
            // The finishing customer is recorded separately below through the
            // active assignment table.
            let _ = customer_id;
        }

        // Keep customer assignments for finish events.
        // (This table is populated and consumed in the main scheduling loop.)
        // Finish events are emitted from the assignment map below.

        // Opening events in descending server ID order.
        let mut opening: Vec<usize> = states
            .iter()
            .enumerate()
            .filter_map(|(index, state)| {
                (!state.opened && state.server.opens == now).then_some(index)
            })
            .collect();
        opening.sort_by_key(|&index| std::cmp::Reverse(states[index].server.id));
        for index in opening {
            states[index].opened = true;
            states[index].idle_since = now;
            events.push(Event::Open {
                time: now,
                server: states[index].server.id,
            });
        }

        // Arrivals in descending customer ID order within this tick.
        while arrival_index < arrivals.len() && arrivals[arrival_index].arrival == now {
            let customer = arrivals[arrival_index].clone();
            events.push(Event::Arrive {
                time: now,
                customer: customer.id,
            });
            waiting.push(Job { customer });
            arrival_index += 1;
        }

        // Dispatch as long as both a waiter and an idle open server exist.
        while !waiting.is_empty() {
            let selected = states
                .iter()
                .enumerate()
                .filter(|(_, state)| state.opened && state.occupied_until.is_none())
                .min_by(|(_, a), (_, b)| {
                    a.idle_since
                        .cmp(&b.idle_since)
                        .then_with(|| a.server.overhead.cmp(&b.server.overhead))
                        .then_with(|| b.server.id.cmp(&a.server.id))
                })
                .map(|(index, _)| index);

            let Some(server_index) = selected else {
                break;
            };

            let job = waiting.remove(0);
            let server_id = states[server_index].server.id;
            let overhead = states[server_index].server.overhead;
            let finish = job
                .customer
                .service
                .checked_add(now)
                .and_then(|value| value.checked_add(overhead))
                .ok_or(Error::TimeOverflow {
                    customer: job.customer.id,
                    server: server_id,
                    time: now,
                })?;

            events.push(Event::Start {
                time: now,
                customer: job.customer.id,
                server: server_id,
                finish,
            });

            if finish == now {
                events.push(Event::Finish {
                    time: now,
                    customer: job.customer.id,
                    server: server_id,
                });
                states[server_index].idle_since = now;
            } else {
                states[server_index].occupied_until = Some(finish);
                active_customer[server_index] = Some(job.customer.id);
            }
        }

        // At this point, all events at this tick are complete. Advance to the
        // earliest remaining arrival, opening, or occupied-server finish.
        let mut next_time: Option<u64> = None;
        let mut consider = |candidate: u64| {
            if candidate > now {
                next_time = Some(next_time.map_or(candidate, |current| current.min(candidate)));
            }
        };

        if arrival_index < arrivals.len() {
            consider(arrivals[arrival_index].arrival);
        }
        for state in &states {
            if !state.opened {
                consider(state.server.opens);
            }
            if let Some(finish) = state.occupied_until {
                consider(finish);
            }
        }

        if let Some(next) = next_time {
            now = next;
        } else {
            break;
        }
    }

    Ok(events)
}

thread_local! {
    static ACTIVE_CUSTOMERS: std::cell::RefCell<Vec<Option<u32>>> =
        const { std::cell::RefCell::new(Vec::new()) };
}
