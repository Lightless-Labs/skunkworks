#[derive(Debug, Clone, PartialEq, Eq)]
pub enum CsvError {
    UnexpectedQuote { record: usize, byte: usize },
    TrailingByte { record: usize, byte: usize },
    BareCr { record: usize, byte: usize },
    UnclosedQuote { record: usize, byte: usize },
    Nul { record: usize, byte: usize },
    InvalidConfig { component: ConfigPart },
    DuplicateHeader { name: String, first: usize, second: usize },
    HeaderWidth { record: usize, expected: usize, actual: usize },
    ReaderFinished,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ConfigPart {
    Delimiter,
    Quote,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum DuplicatePolicy {
    Reject,
    KeepFirst,
    KeepLast,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub delimiter: u8,
    pub quote: u8,
    pub headers: bool,
    pub duplicates: DuplicatePolicy,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Table {
    pub header: Option<Vec<String>>,
    pub rows: Vec<Vec<String>>,
    pub duplicates: DuplicatePolicy,
}

impl Table {
    pub fn column_index(&self, name: &str) -> Option<usize> {
        let header = self.header.as_ref()?;
        match self.duplicates {
            DuplicatePolicy::KeepLast => header.iter().rposition(|n| n == name),
            DuplicatePolicy::KeepFirst | DuplicatePolicy::Reject => {
                header.iter().position(|n| n == name)
            }
        }
    }
}

fn validate_dialect(delimiter: u8, quote: u8) -> Result<(), ConfigPart> {
    if !(0x21..=0x7e).contains(&delimiter) {
        return Err(ConfigPart::Delimiter);
    }
    if !(0x21..=0x7e).contains(&quote) {
        return Err(ConfigPart::Quote);
    }
    if delimiter == quote {
        return Err(ConfigPart::Quote);
    }
    Ok(())
}

fn parse_dialect(
    input: &str,
    delimiter: u8,
    quote: u8,
) -> Result<Vec<Vec<String>>, CsvError> {
    let bytes = input.as_bytes();
    let mut i = if input.starts_with('\u{feff}') { 3 } else { 0 };
    let mut records = Vec::new();
    if i == bytes.len() {
        return Ok(records);
    }

    let mut row = Vec::new();
    let mut field = String::new();
    let mut quoted = false;
    let mut after_quote = false;
    let mut quote_open = 0usize;
    let mut record_index = 0usize;
    let mut field_started = false;

    while i < bytes.len() {
        let b = bytes[i];

        if b == 0 {
            return Err(CsvError::Nul {
                record: record_index,
                byte: i,
            });
        }

        if !quoted && b == b'\r' && bytes.get(i + 1) != Some(&b'\n') {
            return Err(CsvError::BareCr {
                record: record_index,
                byte: i,
            });
        }

        if quoted {
            if b == quote {
                if bytes.get(i + 1) == Some(&quote) {
                    field.push(quote as char);
                    i += 2;
                    continue;
                }
                quoted = false;
                after_quote = true;
                i += 1;
                continue;
            }
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
            continue;
        }

        if after_quote {
            if b == delimiter {
                row.push(std::mem::take(&mut field));
                after_quote = false;
                field_started = false;
                i += 1;
                continue;
            }
            if b == b'\n' || b == b'\r' {
                row.push(std::mem::take(&mut field));
                records.push(std::mem::take(&mut row));
                record_index += 1;
                after_quote = false;
                field_started = false;
                if b == b'\r' {
                    i += 2;
                } else {
                    i += 1;
                }
                continue;
            }
            return Err(CsvError::TrailingByte {
                record: record_index,
                byte: i,
            });
        }

        if b == delimiter {
            row.push(std::mem::take(&mut field));
            field_started = false;
            i += 1;
        } else if b == b'\n' || b == b'\r' {
            row.push(std::mem::take(&mut field));
            records.push(std::mem::take(&mut row));
            record_index += 1;
            field_started = false;
            if b == b'\r' {
                i += 2;
            } else {
                i += 1;
            }
        } else if b == quote {
            if field_started || !field.is_empty() {
                return Err(CsvError::UnexpectedQuote {
                    record: record_index,
                    byte: i,
                });
            }
            quoted = true;
            quote_open = i;
            i += 1;
        } else {
            field_started = true;
            let ch = input[i..].chars().next().unwrap();
            field.push(ch);
            i += ch.len_utf8();
        }
    }

    if quoted {
        return Err(CsvError::UnclosedQuote {
            record: record_index,
            byte: quote_open,
        });
    }

    if !row.is_empty() || !field.is_empty() || after_quote || field_started {
        row.push(field);
        records.push(row);
    }

    Ok(records)
}

pub fn parse_csv(input: &str) -> Result<Vec<Vec<String>>, CsvError> {
    parse_dialect(input, b',', b'"')
}

fn duplicate_header(header: &[String]) -> Option<CsvError> {
    for second in 0..header.len() {
        if let Some(first) = header[..second].iter().position(|n| n == &header[second]) {
            return Some(CsvError::DuplicateHeader {
                name: header[second].clone(),
                first,
                second,
            });
        }
    }
    None
}

fn build_table(records: Vec<Vec<String>>, config: Config) -> Result<Table, CsvError> {
    let (header, rows) = if config.headers {
        if records.is_empty() {
            (None, Vec::new())
        } else {
            let mut records = records.into_iter();
            (records.next(), records.collect::<Vec<_>>())
        }
    } else {
        (None, records)
    };

    if let Some(ref h) = header {
        if config.duplicates == DuplicatePolicy::Reject {
            if let Some(e) = duplicate_header(h) {
                return Err(e);
            }
        }
        for (idx, row) in rows.iter().enumerate() {
            if row.len() != h.len() {
                return Err(CsvError::HeaderWidth {
                    record: idx + 1,
                    expected: h.len(),
                    actual: row.len(),
                });
            }
        }
    }

    Ok(Table {
        header,
        rows,
        duplicates: config.duplicates,
    })
}

pub struct StreamReader {
    config: Config,
    input: String,
    cached: Option<Result<Table, CsvError>>,
}

impl StreamReader {
    pub fn new(config: Config) -> Result<Self, CsvError> {
        validate_dialect(config.delimiter, config.quote)
            .map_err(|component| CsvError::InvalidConfig { component })?;
        Ok(Self {
            config,
            input: String::new(),
            cached: None,
        })
    }

    pub fn feed(&mut self, chunk: &str) -> Result<(), CsvError> {
        if self.cached.is_some() {
            return Err(CsvError::ReaderFinished);
        }
        self.input.push_str(chunk);
        Ok(())
    }

    pub fn finish(&mut self) -> Result<Table, CsvError> {
        if let Some(result) = &self.cached {
            return result.clone();
        }
        let result = parse_dialect(&self.input, self.config.delimiter, self.config.quote)
            .and_then(|records| build_table(records, self.config));
        self.cached = Some(result.clone());
        result
    }

    pub fn reset(&mut self) {
        self.input.clear();
        self.cached = None;
    }
}

pub fn read_csv(input: &str, config: Config) -> Result<Table, CsvError> {
    let mut reader = StreamReader::new(config)?;
    reader.feed(input)?;
    reader.finish()
}

    pub errors: Vec<CellError>,

fn integer_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    if b.is_empty() {
        return false;
    }
    let start = usize::from(b[0] == b'-');
    if start == b.len() {
        return false;
    }
    if b[start] == b'0' {
        return start + 1 == b.len();
    }
    if !(b'1'..=b'9').contains(&b[start]) {
        return false;
    }
    b[start + 1..].iter().all(u8::is_ascii_digit)
}

fn decimal_syntax(s: &str) -> bool {
    let b = s.as_bytes();
    let start = usize::from(b.first() == Some(&b'-'));
    if start >= b.len() {
        return false;
    }
    let dot = match b[start..].iter().position(|&x| x == b'.') {
        Some(p) => start + p,
        None => return false,
    };
    if dot + 3 != b.len() || !b[dot + 1..].iter().all(u8::is_ascii_digit) {
        return false;
    }
    if dot == start {
        return false;
    }
    if b[start] == b'0' {
        dot == start + 1
    } else {
        (b'1'..=b'9').contains(&b[start])
            && b[start + 1..dot].iter().all(u8::is_ascii_digit)
    }
}

fn parse_integer(s: &str) -> Result<i64, CellErrorKind> {
    if !integer_syntax(s) {
        return Err(CellErrorKind::InvalidInteger);
    }
    s.parse::<i64>()
        .map_err(|_| CellErrorKind::IntegerOverflow)
}

fn parse_decimal(s: &str) -> Result<i64, CellErrorKind> {
    if !decimal_syntax(s) || s == "-0.00" {
        return Err(CellErrorKind::InvalidDecimal);
    }
    let negative = s.starts_with('-');
    let digits = if negative { &s[1..] } else { s };
    let dot = digits.find('.').unwrap();
    let whole = digits[..dot]
        .parse::<i128>()
        .map_err(|_| CellErrorKind::DecimalOverflow)?;
    let frac = digits[dot + 1..].parse::<i128>().unwrap();
    let magnitude = whole * 100 + frac;
    let signed = if negative { -magnitude } else { magnitude };
    i64::try_from(signed).map_err(|_| CellErrorKind::DecimalOverflow)
}

fn convert_value(s: &str, spec: &ColumnSpec) -> Result<Value, CellErrorKind> {
    if spec.kind == ColumnType::Text {
        return Ok(Value::Text(s.to_owned()));
    }
    if s.is_empty() {
        return Err(CellErrorKind::Empty);
    }
    match spec.kind {
        ColumnType::Text => Ok(Value::Text(s.to_owned())),
        ColumnType::Integer => parse_integer(s).map(Value::Integer),
        ColumnType::Decimal => parse_decimal(s).map(Value::Decimal),
        ColumnType::Boolean => match s {
            "yes" | "1" => Ok(Value::Boolean(true)),
            "no" | "0" => Ok(Value::Boolean(false)),
            _ => Err(CellErrorKind::InvalidBoolean),
        },
    }
}

pub fn extract_columns(
    table: &Table,
    columns: &[ColumnSpec],
    policy: ErrorPolicy,
) -> TypedResult {
    let mut result = TypedResult {
        rows: Vec::new(),
        errors: Vec::new(),
    };

    for (row_index, row) in table.rows.iter().enumerate() {
        let mut output = Vec::with_capacity(columns.len());
        let mut row_failed = false;
        for (spec_index, spec) in columns.iter().enumerate() {
            if policy == ErrorPolicy::FirstPerRow && row_failed
                || policy == ErrorPolicy::Stop && !result.errors.is_empty()
            {
                output.push(None);
                continue;
            }

            let column = match &spec.column {
                ColumnRef::Index(i) => Some(*i),
                ColumnRef::Name(name) => table.column_index(name),
            };
            let Some(column) = column else {
                output.push(None);
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: None,
                    kind: CellErrorKind::UnknownColumn,
                });
                row_failed = true;
                continue;
            };

            let Some(cell) = row.get(column) else {
                output.push(None);
                result.errors.push(CellError {
                    row: row_index,
                    spec: spec_index,
                    column: Some(column),
                    kind: CellErrorKind::MissingCell,
                });
                row_failed = true;
                continue;
            };

            let converted = if cell.is_empty() && spec.nullable {
                Ok(Value::Null)
            } else {
                convert_value(cell, spec)
            };
            match converted {
                Ok(v) => output.push(Some(v)),
                Err(kind) => {
                    output.push(None);
                    result.errors.push(CellError {
                        row: row_index,
                        spec: spec_index,
                        column: Some(column),
                        kind,
                    });
                    row_failed = true;
                }
            }
            if policy == ErrorPolicy::Stop && row_failed {
                break;
            }
        }
        result.rows.push(output);
        if policy == ErrorPolicy::Stop && !result.errors.is_empty() {
            break;
        }
    }
    result
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct WriteConfig {

    pub quote: u8,
    pub line_ending: LineEnding,
    pub final_terminator: bool,

}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum WriteError {
    InvalidConfig { component: ConfigPart },
    Nul {
        record: usize,
        column: usize,
        byte: usize,
    },
    Width {
        record: usize,
        expected: usize,
        actual: usize,
    },
    ZeroColumns {
        record: usize,
    },
    DuplicateHeader {
        name: String,
        first: usize,
        second: usize,
    },
}

pub fn write_csv(table: &Table, config: WriteConfig) -> Result<String, WriteError> {
    validate_dialect(config.delimiter, config.quote)
        .map_err(|component| WriteError::InvalidConfig { component })?;

    let mut original_records: Vec<&Vec<String>> = Vec::new();
    if let Some(header) = &table.header {
        original_records.push(header);
    }
    original_records.extend(table.rows.iter());

    for (record, row) in original_records.iter().enumerate() {
        for (column, field) in row.iter().enumerate() {
            if let Some(byte) = field.as_bytes().iter().position(|&b| b == 0) {
                return Err(WriteError::Nul {
                    record,
                    column,
                    byte,
                });
            }
        }
    }

    if let Some(header) = &table.header {
        for (idx, row) in table.rows.iter().enumerate() {
            if row.len() != header.len() {
                return Err(WriteError::Width {
                    record: idx + 1,
                    expected: header.len(),
                    actual: row.len(),
                });
            }
        }
    }

    for (record, row) in original_records.iter().enumerate() {
        if row.is_empty() {
            return Err(WriteError::ZeroColumns { record });
        }
    }

    if config.mode == WriteMode::Canonical {
        if let Some(header) = &table.header {
            if table.duplicates == DuplicatePolicy::Reject {
                if let Some(CsvError::DuplicateHeader { name, first, second }) =
                    duplicate_header(header)
                {
                    return Err(WriteError::DuplicateHeader {
                        name,
                        first,
                        second,
                    });
                }
            }
        }
    }

    let mut records: Vec<Vec<String>> = Vec::new();
    match config.mode {
        WriteMode::Preserve => {
            if let Some(h) = &table.header {
                records.push(h.clone());
            }
            records.extend(table.rows.clone());
        }
        WriteMode::Canonical => {
            if let Some(header) = &table.header {
                let mut chosen: Vec<usize> = Vec::new();
                for (i, name) in header.iter().enumerate() {
                    let last = table.duplicates == DuplicatePolicy::KeepLast;
                    let occurrence = if last {
                        header[i + 1..].iter().any(|n| n == name)
                    } else {
                        header[..i].iter().any(|n| n == name)
                    };
                    if !occurrence {
                        chosen.push(i);
                    }
                }
                chosen.sort_by(|&a, &b| header[a].as_bytes().cmp(header[b].as_bytes()));
                records.push(chosen.iter().map(|&i| header[i].clone()).collect());
                let mut projected: Vec<Vec<String>> = table
                    .rows
                    .iter()
                    .map(|row| chosen.iter().map(|&i| row[i].clone()).collect())
                    .collect();
                projected.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b.iter()) {
                        match x.as_bytes().cmp(y.as_bytes()) {
                            std::cmp::Ordering::Equal => {}
                            order => return order,
                        }
                    }
                    a.len().cmp(&b.len())
                });
                records.extend(projected);
            } else {
                let mut rows = table.rows.clone();
                rows.sort_by(|a, b| {
                    for (x, y) in a.iter().zip(b.iter()) {
                        match x.as_bytes().cmp(y.as_bytes()) {
                            std::cmp::Ordering::Equal => {}
                            order => return order,
                        }
                    }
                    a.len().cmp(&b.len())
                });
                records = rows;
            }
        }
    }

    if records.is_empty() {
        return Ok(String::new());
    }

    let canonical = config.mode == WriteMode::Canonical;
    let ending: &[u8] = if canonical || config.line_ending == LineEnding::Lf {
        b"\n"
    } else {
        b"\r\n"
    };
    let final_terminator = canonical || config.final_terminator;
    let mut output = Vec::new();

    for (record_index, row) in records.iter().enumerate() {
        for (column_index, field) in row.iter().enumerate() {
            if column_index > 0 {
                output.push(config.delimiter);
            }
            let bytes = field.as_bytes();
            let starts_bom = field.starts_with('\u{feff}');
            let first_output_field = record_index == 0 && column_index == 0;
            let quote_field = bytes.contains(&config.delimiter)
                || bytes.contains(&config.quote)
                || bytes.contains(&b'\r')
                || bytes.contains(&b'\n')
                || bytes.first().is_some_and(|b| *b == b' ' || *b == b'\t')
                || bytes.last().is_some_and(|b| *b == b' ' || *b == b'\t')
                || (first_output_field && starts_bom)
                || (row.len() == 1 && field.is_empty());
            if quote_field {
                output.push(config.quote);
                for &b in bytes {
                    output.push(b);
                    if b == config.quote {
                        output.push(b);
                    }
                }
                output.push(config.quote);
            } else {
                output.extend_from_slice(bytes);
            }
        }
        if record_index + 1 < records.len() || final_terminator {
            output.extend_from_slice(ending);
        }
    }

    Ok(String::from_utf8(output).unwrap())
}
