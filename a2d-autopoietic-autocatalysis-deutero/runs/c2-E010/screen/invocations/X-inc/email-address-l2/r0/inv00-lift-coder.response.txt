#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let quoted_prefix = if bytes.first() == Some(&b'"') {
        let mut closing = None;
        let mut i = 1;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => i += 2,
                b'"' => {
                    closing = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        match closing {
            Some(pos) => Some(pos),
            None => return Err(EmailError::UnclosedQuote { at: 0 }),
        }
    } else {
        None
    };

    let mut separators = Vec::new();
    for i in 0..bytes.len() {
        if bytes[i] == b'@' && !quoted_prefix.is_some_and(|close| i <= close) {
            separators.push(i);
        }
    }

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let separator = separators[0];
    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];
    let (local, quoted) = if local_bytes[0] == b'"' {
        let close = quoted_prefix.expect("quoted local has a closing quote");
        if close + 1 != separator {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            let b = bytes[i];
            if b == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: i });
                }
                let escaped = bytes[i + 1];
                match escaped {
                    b'"' | b'\\' | b'@' => {
                        decoded.push(escaped);
                        origins.push(i);
                        i += 2;
                    }
                    _ => return Err(EmailError::BadEscape { at: i }),
                }
            } else {
                if !(0x20..=0x7e).contains(&b) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                decoded.push(b);
                origins.push(i);
                i += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (
            String::from_utf8(decoded).expect("validated ASCII bytes"),
            true,
        )
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for i in 0..local_bytes.len() {
            let b = local_bytes[i];
            if b == b'.' {
                if i == 0
                    || i + 1 == local_bytes.len()
                    || local_bytes[i - 1] == b'.'
                {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: i });
            }
        }
        (
            String::from_utf8(local_bytes.to_vec()).expect("validated ASCII bytes"),
            false,
        )
    };

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let root_dot = raw_domain.last() == Some(&b'.');
    let domain_len = raw_domain.len() - usize::from(root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }
    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain_bytes = &raw_domain[..domain_len];
    let mut label_start = 0;
    let mut labels = Vec::new();

    for i in 0..=domain_bytes.len() {
        if i == domain_bytes.len() || domain_bytes[i] == b'.' {
            if i == label_start {
                let at = if i == domain_bytes.len() && root_dot {
                    input.len() - 1
                } else {
                    domain_start + i
                };
                return Err(EmailError::EmptyLabel { at });
            }

            let label = &domain_bytes[label_start..i];
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: domain_start + label_start + 63,
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + label_start,
                });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: domain_start + i - 1,
                });
            }
            for (offset, &b) in label.iter().enumerate() {
                if !b.is_ascii_alphanumeric() && b != b'-' {
                    return Err(EmailError::DomainChar {
                        at: domain_start + label_start + offset,
                    });
                }
            }
            labels.push(label);
            label_start = i + 1;
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let last = labels[labels.len() - 1];
    if last.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + domain_len - last.len(),
        });
    }

    let domain = String::from_utf8(domain_bytes.to_vec())
        .expect("validated ASCII bytes")
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted,
    })
}
