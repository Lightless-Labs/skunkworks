#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
    UnclosedQuote { at: usize },
    TrailingLocal { at: usize },
    BadEscape { at: usize },
    QuotedChar { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();
    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&b| b >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut quote_close = None;
    if bytes[0] == b'"' {
        let mut i = 1;
        while i < bytes.len() {
            match bytes[i] {
                b'\\' => {
                    i += 2;
                }
                b'"' => {
                    quote_close = Some(i);
                    break;
                }
                _ => i += 1,
            }
        }
        if quote_close.is_none() {
            return Err(EmailError::UnclosedQuote { at: 0 });
        }
    }

    let protected_end = quote_close.map(|i| i + 1).unwrap_or(0);
    let separators: Vec<usize> = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &b)| {
            if b == b'@' && (protected_end == 0 || i >= protected_end) {
                Some(i)
            } else {
                None
            }
        })
        .collect();

    if separators.is_empty() {
        return Err(EmailError::MissingAt);
    }
    if separators.len() > 1 {
        return Err(EmailError::MultipleAt { at: separators[1] });
    }

    let sep = separators[0];
    if sep == 0 {
        return Err(EmailError::EmptyLocal { at: sep });
    }

    let local_bytes = &bytes[..sep];
    let (local, quoted) = if let Some(close) = quote_close {
        if close + 1 != sep {
            return Err(EmailError::TrailingLocal { at: close + 1 });
        }

        let mut decoded = Vec::new();
        let mut origins = Vec::new();
        let mut i = 1;
        while i < close {
            if bytes[i] == b'\\' {
                if i + 1 >= close {
                    return Err(EmailError::BadEscape { at: i });
                }
                let escaped = bytes[i + 1];
                if !matches!(escaped, b'"' | b'\\' | b'@') {
                    return Err(EmailError::BadEscape { at: i });
                }
                decoded.push(escaped);
                origins.push(i);
                i += 2;
            } else {
                if !(0x20..=0x7e).contains(&bytes[i]) {
                    return Err(EmailError::QuotedChar { at: i });
                }
                decoded.push(bytes[i]);
                origins.push(i);
                i += 1;
            }
        }

        if decoded.is_empty() {
            return Err(EmailError::EmptyLocal { at: close });
        }
        if decoded.len() > 64 {
            return Err(EmailError::LocalTooLong { at: origins[64] });
        }
        (String::from_utf8(decoded).unwrap(), true)
    } else {
        if local_bytes.len() > 64 {
            return Err(EmailError::LocalTooLong { at: 64 });
        }

        for (i, &b) in local_bytes.iter().enumerate() {
            if b == b'.' {
                if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                    return Err(EmailError::LocalDot { at: i });
                }
            } else if !b.is_ascii_alphanumeric()
                && !b"!#$%&'*+-/=?^_`{|}~".contains(&b)
            {
                return Err(EmailError::LocalChar { at: i });
            }
        }
        (String::from_utf8(local_bytes.to_vec()).unwrap(), false)
    };

    let domain_start = sep + 1;
    let raw_domain = &bytes[domain_start..];
    let root_dot = if raw_domain.last() == Some(&b'.') {
        Some(input.len() - 1)
    } else {
        None
    };
    let domain_end = if root_dot.is_some() {
        input.len() - 1
    } else {
        input.len()
    };

    if domain_start == domain_end {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    let domain_bytes = &bytes[domain_start..domain_end];
    if domain_bytes.len() > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let mut label_starts = Vec::new();
    let mut label_ranges = Vec::new();
    let mut start = 0;
    for i in 0..=domain_bytes.len() {
        if i == domain_bytes.len() || domain_bytes[i] == b'.' {
            let label_start = domain_start + start;
            let label_end = i;
            if start == i {
                let at = if i == domain_bytes.len() {
                    root_dot.unwrap_or(domain_start + i)
                } else {
                    domain_start + i
                };
                return Err(EmailError::EmptyLabel { at });
            }
            let label = &domain_bytes[start..label_end];
            if label.len() > 63 {
                return Err(EmailError::LabelTooLong {
                    at: label_start + 63,
                });
            }
            if label[0] == b'-' {
                return Err(EmailError::LabelHyphen { at: label_start });
            }
            if label[label.len() - 1] == b'-' {
                return Err(EmailError::LabelHyphen {
                    at: label_start + label.len() - 1,
                });
            }
            if let Some(offset) = label
                .iter()
                .position(|&b| !b.is_ascii_alphanumeric() && b != b'-')
            {
                return Err(EmailError::DomainChar {
                    at: label_start + offset,
                });
            }
            label_starts.push(label_start);
            label_ranges.push((start, label_end));
            start = i + 1;
        }
    }

    if label_ranges.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_start, last_end) = *label_ranges.last().unwrap();
    let last_label = &domain_bytes[last_start..last_end];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_start,
        });
    }

    let domain = String::from_utf8(domain_bytes.to_vec())
        .unwrap()
        .to_ascii_lowercase();
    Ok(Address {
        local,
        domain,
        quoted,
    })
}
