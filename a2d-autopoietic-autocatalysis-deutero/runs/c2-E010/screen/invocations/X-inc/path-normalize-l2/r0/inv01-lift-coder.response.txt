#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/').filter(|component| !component.is_empty()).collect()
}

fn reduce<'a>(
    stack: &mut Vec<&'a str>,
    parts: impl IntoIterator<Item = (&'a str, usize)>,
    absolute: bool,
) -> Result<(), PathError> {
    for (part, at) in parts {
        match part {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|last| *last != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..");
                }
            }
            _ => stack.push(part),
        }
    }
    Ok(())
}

fn render(stack: &[&str], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    let parts = components(path)
        .into_iter()
        .scan(0usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((part, start))
        });

    reduce(&mut stack, parts, absolute)?;
    Ok(render(&stack, absolute, intent))
}

fn normalize_context(path: &str) -> Result<Vec<String>, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    if !absolute {
        return Err(PathError::InvalidCwd);
    }

    let mut stack: Vec<String> = Vec::new();
    for part in components(path) {
        match part {
            "." => {}
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: path.find("..").unwrap_or(0) });
                }
            }
            _ => stack.push(part.to_owned()),
        }
    }
    Ok(stack)
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let mut cwd_stack = normalize_context(cwd).map_err(|_| PathError::InvalidCwd)?;
    let mut home_stack = match home {
        Some(home_path) => Some(
            normalize_context(home_path).map_err(|_| PathError::InvalidHome)?,
        ),
        None => None,
    };

    validate_raw(path)?;

    let bytes = path.as_bytes();
    let tilde = bytes.first() == Some(&b'~');
    let (absolute, mut stack, remainder, prefix_len) = if tilde {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home_stack.take().ok_or(PathError::MissingHome)?;
        (true, home, &path[if bytes.len() > 1 { 2 } else { 1 }..], if bytes.len() > 1 { 2 } else { 1 })
    } else if bytes.first() == Some(&b'/') {
        (true, Vec::new(), path, 0)
    } else {
        (true, std::mem::take(&mut cwd_stack), path, 0)
    };

    let intent = directory_intent(path);
    let mut offset = prefix_len;
    let mut parts = Vec::new();
    for part in remainder.split('/') {
        if !part.is_empty() {
            parts.push((part, offset));
        }
        offset += part.len() + 1;
    }

    let mut owned: Vec<String> = stack.drain(..).collect();
    for (part, at) in parts {
        match part {
            "." => {}
            ".." => {
                if owned.pop().is_none() {
                    return Err(PathError::Escape { at });
                }
            }
            _ => owned.push(part.to_owned()),
        }
    }

    let refs: Vec<&str> = owned.iter().map(String::as_str).collect();
    Ok(render(&refs, absolute, intent))
}
