pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.rsplit('/')
        .find(|component| !component.is_empty())
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str, skip_prefix: usize) -> Vec<(usize, &str)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = skip_prefix;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((start, &path[start..i]));
    }
    result
}

fn reduce(
    stack: &mut Vec<String>,
    path: &str,
    skip_prefix: usize,
    absolute: bool,
    offset_base: usize,
) -> Result<(), PathError> {
    for (at, component) in components(path, skip_prefix) {
        match component {
            "." => {}
            ".." => {
                if stack.last().is_some_and(|item| item != "..") {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape {
                        at: offset_base + at,
                    });
                } else {
                    stack.push("..".to_string());
                }
            }
            _ => stack.push(component.to_string()),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, dir_intent: bool) -> String {
    let mut output = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_string()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        output = "/".to_string();
    }

    if dir_intent && output != "/" && output != "." {
        output.push('/');
    }
    output
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let mut stack = Vec::new();
    reduce(&mut stack, path, 0, absolute, 0)?;
    Ok(render(&stack, absolute, directory_intent(path)))
}

}

    } else {
        (cwd_components, 0)

fn reduce_components<I>(
    components: I,
    absolute: bool,
) -> Result<Vec<String>, PathError>
where
    I: IntoIterator<Item = (String, usize)>,
{
    let mut result: Vec<String> = Vec::new();

    for (component, at) in components {
        match component.as_str() {
            "" | "." => {}
            ".." => {
                if result.last().is_some_and(|last| last != "..") {
                    result.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    result.push(component);
                }
            }
            _ => result.push(component),
        }
    }

    Ok(result)
}

fn render(components: &[String], absolute: bool, dir_intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", components.join("/"))
    } else if components.is_empty() {
        ".".to_string()
    } else {
        components.join("/")
    };

    if dir_intent && result != "/" && result != "." {
        result.push('/');
    }

    result
}

fn normalize_components(path: &str) -> Result<(bool, Vec<String>), PathError> {
    let absolute = path.as_bytes().first() == Some(&b'/');
    let components = path
        .split('/')
        .scan(0usize, |offset, component| {
            let at = *offset;
            *offset += component.len() + 1;
            Some((component.to_string(), at))
        });

    Ok((absolute, reduce_components(components, absolute)?))
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let dir_intent = directory_intent(path);
    let (absolute, components) = normalize_components(path)?;
    Ok(render(&components, absolute, dir_intent))
}

fn validated_context(path: &str) -> Result<Vec<String>, PathError> {
    if path.as_bytes().first() != Some(&b'/') {
        return Err(PathError::InvalidCwd);
    }

    match normalize(path) {
        Ok(normalized) => {
            let components = normalized
                .split('/')
                .filter(|component| !component.is_empty())
                .map(str::to_string)
                .collect();
            Ok(components)
        }
        Err(_) => Err(PathError::InvalidCwd),
    }
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_components = validated_context(cwd)?;

    let home_components = match home {
        Some(home_path) => {
            if home_path.as_bytes().first() != Some(&b'/') {
                return Err(PathError::InvalidHome);
            }

            match normalize(home_path) {
                Ok(normalized) => Some(
                    normalized
                        .split('/')
                        .filter(|component| !component.is_empty())
                        .map(str::to_string)
                        .collect::<Vec<_>>(),
                ),
                Err(_) => return Err(PathError::InvalidHome),
            }
        }
        None => None,
    };

    validate_raw(path)?;
    let dir_intent = directory_intent(path);
    let bytes = path.as_bytes();

    let (mut result, remainder, remainder_offset) = if bytes[0] == b'/' {

    } else if bytes[0] == b'~' {

        let consumed = if bytes.len() > 1 { 2 } else { 1 };
        let remainder = &path[consumed..];

    } else {
        (cwd_components, path, 0usize)
    };

    let components = remainder.split('/').scan(
        remainder_offset,

            let at = *offset;

            Some((component.to_string(), at))
        },
    );

        result
            .into_iter()
            .map(|component| (component, 0))
            .chain(components),
        true,
    )?;

    Ok(render(&result, true, dir_intent))
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }

    Ok(())
}

fn components(path: &str) -> Vec<(usize, &str)> {
    let mut result = Vec::new();
    let mut start = None;

    for (i, b) in path.bytes().enumerate() {
        if b == b'/' {
            if let Some(s) = start.take() {
                result.push((s, &path[s..i]));
            }
        } else if start.is_none() {
            start = Some(i);
        }
    }

    if let Some(s) = start {
        result.push((s, &path[s..]));
    }

    result
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components(path)
            .last()
            .is_some_and(|(_, component)| *component == "." || *component == "..")
}

fn reduce_components<'a>(
    initial: &mut Vec<&'a str>,
    path: &'a str,
    parts: &[(usize, &'a str)],
    offset: usize,
    absolute: bool,
) -> Result<(), PathError> {
    for &(at, part) in parts {
        match part {
            "." => {}
            ".." => {
                if initial.last().is_some_and(|last| *last != "..") {
                    initial.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: offset + at });
                } else {
                    initial.push("..");
                }
            }
            _ => initial.push(part),
        }
    }
    let _ = path;
    Ok(())
}

fn format_path(parts: &[&str], absolute: bool, directory: bool) -> String {
    let mut result = if absolute {
        format!("/{}", parts.join("/"))
    } else if parts.is_empty() {
        ".".to_string()
    } else {
        parts.join("/")
    };

    if absolute && parts.is_empty() {
        result = "/".to_string();
    }

    if directory && result != "/" && result != "." {
        result.push('/');
    }

    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;

    let absolute = path.as_bytes().first() == Some(&b'/');
    let directory = directory_intent(path);
    let parts = components(path);
    let mut reduced = Vec::new();

    reduce_components(&mut reduced, path, &parts, 0, absolute)?;
    Ok(format_path(&reduced, absolute, directory))
}

fn normalized_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized)
        .into_iter()
        .map(|(_, part)| part.to_string())
        .collect())
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = normalized_context(cwd).map_err(|_| PathError::InvalidCwd)?;

    let home_parts = match home {
        Some(home) => Some(normalized_context(home).map_err(|_| PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;

    let directory = directory_intent(path);
    let tilde = path.starts_with('~');
    let (mut reduced, remainder, remainder_offset) = if tilde {
        if path == "~" {
            let parts = home_parts.ok_or(PathError::MissingHome)?;
            (parts, "", path.len())
        } else if path.as_bytes().get(1) == Some(&b'/') {
            let parts = home_parts.ok_or(PathError::MissingHome)?;
            (parts, &path[2..], 2)
        } else {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
    } else if path.starts_with('/') {
        (Vec::new(), path, 0)
    } else {
        (cwd_parts, path, 0)
    };

    let remainder_parts = components(remainder);
    let borrowed: Vec<(usize, &str)> = remainder_parts;
    let mut owned_reduced: Vec<String> = reduced.drain(..).map(|s| s).collect();
    let mut string_parts: Vec<&str> = owned_reduced.iter().map(String::as_str).collect();

    reduce_components(
        &mut string_parts,
        remainder,
        &borrowed,
        remainder_offset,
        true,
    )?;

    owned_reduced = string_parts.into_iter().map(str::to_string).collect();
    let refs: Vec<&str> = owned_reduced.iter().map(String::as_str).collect();
    Ok(format_path(&refs, true, directory))
}
