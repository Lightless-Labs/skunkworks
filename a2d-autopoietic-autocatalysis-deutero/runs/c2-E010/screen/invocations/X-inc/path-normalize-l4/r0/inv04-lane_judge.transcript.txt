use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    let bytes = path.as_bytes();

    if bytes.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = bytes.iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = bytes.iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = bytes.iter().position(|&b| (1..=31).contains(&b) || b == 127) {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components_with_offsets(path: &str) -> Vec<(&str, usize)> {
    path.split('/')
        .scan(0usize, |offset, part| {
            let start = *offset;
            *offset += part.len() + 1;
            Some((part, start))
        })
        .filter(|(part, _)| !part.is_empty())
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || components_with_offsets(path)
            .last()
            .map(|(part, _)| *part == "." || *part == "..")
            .unwrap_or(false)
}

fn reduce_components(
    stack: &mut Vec<String>,
    components: &[(&str, usize)],
    absolute: bool,
) -> Result<(), PathError> {
    for &(component, at) in components {
        match component {
            "." => {}
            ".." => {
                if stack.last().map(|s| s.as_str()) != Some("..")
                    && !stack.is_empty()
                {
                    stack.pop();
                } else if absolute {
                    return Err(PathError::Escape { at });
                } else {
                    stack.push("..".to_owned());
                }
            }
            name => stack.push(name.to_owned()),
        }
    }
    Ok(())
}

fn render(stack: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        format!("/{}", stack.join("/"))
    } else if stack.is_empty() {
        ".".to_owned()
    } else {
        stack.join("/")
    };

    if absolute && stack.is_empty() {
        return "/".to_owned();
    }
    if !absolute && stack.is_empty() {
        return ".".to_owned();
    }
    if intent {
        result.push('/');
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let components = components_with_offsets(path);
    let mut stack = Vec::new();
    reduce_components(&mut stack, &components, absolute)?;
    Ok(render(&stack, absolute, intent))
}

fn validate_context(value: &str, cwd: bool) -> Result<Vec<String>, PathError> {
    let normalized = normalize(value).map_err(|_| {
        if cwd {
            PathError::InvalidCwd
        } else {
            PathError::InvalidHome
        }
    })?;
    if !value.starts_with('/') {
        return Err(if cwd {
            PathError::InvalidCwd
        } else {
            PathError::InvalidHome
        });
    }
    Ok(components_with_offsets(&normalized)
        .into_iter()
        .map(|(part, _)| part.to_owned())
        .collect())
}

fn request_parts<'a>(path: &'a str) -> Result<(bool, bool, Vec<(&'a str, usize)>, Option<Vec<String>>), PathError> {
    validate_raw(path)?;

    let bytes = path.as_bytes();
    let intent = directory_intent(path);
    if bytes[0] == b'~' {
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let remainder_start = if bytes.len() > 1 { 2 } else { 1 };
        let remainder = &path[remainder_start..];
        let parts = components_with_offsets(remainder)
            .into_iter()
            .map(|(part, at)| (part, at + remainder_start))
            .collect();
        return Ok((false, intent, parts, Some(Vec::new())));
    }

    let absolute = bytes[0] == b'/';
    Ok((absolute, intent, components_with_offsets(path), None))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd, true)?;
    let home_components = match home {
        Some(value) => Some(validate_context(value, false)?),
        None => None,
    };

    let (absolute, intent, parts, tilde) = request_parts(path)?;
    let mut stack = if absolute {
        Vec::new()
    } else if tilde.is_some() {
        home_components.ok_or(PathError::MissingHome)?
    } else {
        cwd_components
    };
    reduce_components(&mut stack, &parts, true)?;
    Ok(render(&stack, true, intent))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut indices = HashMap::new();

    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .map(|normalized| normalized == link.path)
                .unwrap_or(false);
        let target_valid = validate_raw(&link.target).is_ok();

        if !path_valid || !target_valid {
            return Err(PathError::InvalidLink { index });
        }

        if let Some(&first) = indices.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        indices.insert(link.path.clone(), index);
    }

    Ok(indices)
}

struct Traversal<'a> {
    links: &'a [Link],
    indices: &'a HashMap<String, usize>,
    active: Vec<usize>,
    expansions: usize,
    max_expansions: usize,
}

impl Traversal<'_> {
    fn process(
        &mut self,
        stack: &mut Vec<String>,
        components: &[(&str, usize)],
        absolute: bool,
    ) -> Result<(), PathError> {
        for &(component, at) in components {
            match component {
                "." => continue,
                ".." => {
                    if stack.is_empty() {
                        if absolute {
                            return Err(PathError::Escape { at });
                        }
                        stack.push("..".to_owned());
                    } else if stack.last().map(|s| s.as_str()) == Some("..") {
                        if absolute {
                            return Err(PathError::Escape { at });
                        }
                        stack.push("..".to_owned());
                    } else {
                        stack.pop();
                    }
                }
                name => {
                    stack.push(name.to_owned());
                    let key = format!("/{}", stack.join("/"));
                    if let Some(&index) = self.indices.get(&key) {
                        if self.active.contains(&index) {
                            return Err(PathError::Loop { index });
                        }
                        if self.expansions >= self.max_expansions {
                            return Err(PathError::Limit { index });
                        }
                        self.expansions += 1;
                        self.active.push(index);

                        stack.pop();
                        let target = &self.links[index].target;
                        let target_absolute = target.starts_with('/');
                        if target_absolute {
                            stack.clear();
                        }
                        let target_parts = components_with_offsets(target);
                        self.process(stack, &target_parts, true)?;

                        self.active.pop();
                    }
                }
            }
        }
        Ok(())
    }
}

fn resolve_links_internal(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_components = validate_context(cwd, true)?;
    let home_components = match home {
        Some(value) => Some(validate_context(value, false)?),
        None => None,
    };
    let indices = validate_links(links)?;
    let (absolute, intent, parts, tilde) = request_parts(path)?;

    let mut stack = Vec::new();
    let mut traversal = Traversal {
        links,
        indices: &indices,
        active: Vec::new(),
        expansions: 0,
        max_expansions,
    };

    if absolute {
        traversal.process(&mut stack, &[], true)?;
    } else if tilde.is_some() {
        stack = home_components.ok_or(PathError::MissingHome)?;
        let context_parts: Vec<(&str, usize)> =
            stack.iter().map(|part| (part.as_str(), 0)).collect();
        traversal.process(&mut Vec::new(), &[], true)?;
        let mut rebuilt = Vec::new();
        traversal.process(&mut rebuilt, &context_parts, true)?;
        stack = rebuilt;
    } else {
        let context_parts: Vec<(&str, usize)> =
            cwd_components.iter().map(|part| (part.as_str(), 0)).collect();
        traversal.process(&mut stack, &context_parts, true)?;
    }

    traversal.process(&mut stack, &parts, true)?;
    Ok(render(&stack, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links_internal(path, cwd, home, links, max_expansions)
}

fn absolute_components(path: &str) -> Vec<String> {
    components_with_offsets(path)
        .into_iter()
        .map(|(part, _)| part.to_owned())
        .collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();

    let mut parts = Vec::new();
    parts.extend((common..base.len()).map(|_| "..".to_owned()));
    parts.extend(target[common..].iter().cloned());

    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base = resolve_links(base, cwd, home, links, max_expansions)?;
    let resolved_target = resolve_links(target, cwd, home, links, max_expansions)?;
    let base_components = absolute_components(&resolved_base);
    let target_components = absolute_components(&resolved_target);
    let target_is_directory = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_components, &target_components)];

    for link in links {
        let probe = resolve_links_internal(&link.path, "/", None, links, max_expansions);
        let Ok(probe) = probe else {
            continue;
        };
        let probe_components = absolute_components(&probe);
        if probe_components.len() > target_components.len()
            || probe_components
                .iter()
                .zip(target_components.iter())
                .any(|(a, b)| a != b)
        {
            continue;
        }

        let mut candidate = route(&base_components, &absolute_components(&link.path));
        let suffix = &target_components[probe_components.len()..];
        if !suffix.is_empty() {
            if candidate == "." {
                candidate = suffix.join("/");
            } else {
                candidate.push('/');
                candidate.push_str(&suffix.join("/"));
            }
        }
        candidates.push(candidate);
    }

    let mut valid = Vec::new();

    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }

        if target_is_directory && candidate != "." {
            candidate.push('/');
        }

        if let Ok(resolved) =
            resolve_links_internal(&candidate, &resolved_base, home, links, max_expansions)
        {
            if absolute_components(&resolved) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| {
        a.len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes()))
    });
    valid
        .into_iter()
        .next()
        .ok_or(PathError::Escape { at: 0 })
}

    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    path.split('/')
        .filter(|component| !component.is_empty())
        .last()
        .is_some_and(|component| component == "." || component == "..")
}

fn components(path: &str) -> Vec<&str> {
    path.split('/')
        .filter(|component| !component.is_empty())
        .collect()
}

fn render_absolute(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return "/".to_string();
    }
    let mut result = String::from("/");
    result.push_str(&parts.join("/"));
    if intent {
        result.push('/');
    }
    result
}

fn render_normalized(parts: &[String], absolute: bool, intent: bool) -> String {
    if absolute {
        return render_absolute(parts, intent);
    }
    if parts.is_empty() {
        return ".".to_string();
    }
    let mut result = parts.join("/");
    if intent {
        result.push('/');
    }
    result
}

fn reduce_component(
    stack: &mut Vec<String>,
    component: &str,
    absolute: bool,
    at: usize,
) -> Result<(), PathError> {
    match component {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|part| part != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_string());
            }
        }
        _ => stack.push(component.to_string()),
    }
    Ok(())
}

fn normalize_validated(path: &str) -> Result<String, PathError> {
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();

    for (at, component) in indexed_components(path, 0) {
        reduce_component(&mut stack, component, absolute, at)?;
    }

    Ok(render_normalized(&stack, absolute, intent))
}

fn indexed_components(path: &str, base: usize) -> Vec<(usize, &str)> {
    let mut result = Vec::new();
    let bytes = path.as_bytes();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        result.push((base + start, &path[start..i]));
    }
    result
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    normalize_validated(path)
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }
    match normalize(path) {
        Ok(normalized) => Ok(components(&normalized)
            .into_iter()
            .map(str::to_string)
            .collect()),
        Err(_) => Err(error),
    }
}

fn request_tilde(path: &str, home: Option<&str>) -> Result<(bool, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((false, 0));
    }
    if path.len() > 1 && !path[1..].starts_with('/') {
        return Err(PathError::UnsupportedTilde { at: 1 });
    }
    if home.is_none() {
        return Err(PathError::MissingHome);
    }
    Ok((true, if path.len() > 1 { 2 } else { 1 }))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let (tilde, consumed) = request_tilde(path, home)?;
    let absolute = !tilde && path.starts_with('/');
    let mut stack = if absolute {
        Vec::new()
    } else if tilde {
        home_parts.unwrap_or_default()
    } else {
        cwd_parts
    };

    let intent = directory_intent(path);
    let remainder = &path[consumed..];
    let offset_base = consumed;

    for (at, component) in indexed_components(remainder, offset_base) {
        reduce_component(&mut stack, component, true, at)?;
    }

    Ok(render_absolute(&stack, intent))
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut lookup = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = lookup.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        lookup.insert(link.path.clone(), index);
    }
    Ok(lookup)
}

struct LinkWalk<'a> {
    links: &'a [Link],
    lookup: &'a HashMap<String, usize>,
    max_expansions: usize,
    expansions: usize,
    active: Vec<usize>,
}

impl LinkWalk<'_> {
    fn process(
        &mut self,
        path: &str,
        base_offset: usize,
        stack: &mut Vec<String>,
        absolute: bool,
    ) -> Result<(), PathError> {
        for (at, component) in indexed_components(path, base_offset) {
            match component {
                "." => {}
                ".." => {
                    if stack.last().is_some_and(|part| part != "..") {
                        stack.pop();
                    } else if absolute {
                        return Err(PathError::Escape { at });
                    } else {
                        stack.push("..".to_string());
                    }
                }
                _ => {
                    stack.push(component.to_string());
                    let key = render_absolute(stack, false);
                    if let Some(&index) = self.lookup.get(&key) {
                        self.expand(index, stack)?;
                    }
                }
            }
        }
        Ok(())
    }

    fn expand(&mut self, index: usize, stack: &mut Vec<String>) -> Result<(), PathError> {
        if self.active.contains(&index) {
            return Err(PathError::Loop { index });
        }
        if self.expansions >= self.max_expansions {
            return Err(PathError::Limit { index });
        }
        self.expansions += 1;

        let link = &self.links[index];
        let target = link.target.as_str();
        let absolute = target.starts_with('/');
        if absolute {
            stack.clear();
        } else {
            stack.pop();
        }

        self.active.push(index);
        let result = self.process(target, 0, stack, true);
        self.active.pop();
        result
    }
}

fn resolve_links_validated(
    path: &str,
    cwd_parts: &[String],
    home_parts: Option<&[String]>,
    links: &[Link],
    lookup: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let (tilde, consumed) = request_tilde(path, home_parts.map(|_| ""))?;

    let absolute = !tilde && path.starts_with('/');
    let mut stack = if absolute {
        Vec::new()
    } else if tilde {
        home_parts.unwrap_or(&[]).to_vec()
    } else {
        cwd_parts.to_vec()
    };

    let mut walker = LinkWalk {
        links,
        lookup,
        max_expansions,
        expansions: 0,
        active: Vec::new(),
    };

    let result = walker.process(&path[consumed..], consumed, &mut stack, true);
    result?;
    Ok(render_absolute(&stack, directory_intent(path)))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };
    let lookup = validate_links(links)?;
    resolve_links_validated(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        links,
        &lookup,
        max_expansions,
    )
}

fn resolve_links_with_context(
    path: &str,
    cwd_parts: &[String],
    home_parts: Option<&[String]>,
    links: &[Link],
    lookup: &HashMap<String, usize>,
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links_validated(
        path,
        cwd_parts,
        home_parts,
        links,
        lookup,
        max_expansions,
    )
}

fn component_strings(path: &str) -> Vec<String> {
    components(path).into_iter().map(str::to_string).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let mut common = 0;
    while common < base.len()
        && common < target.len()
        && base[common] == target[common]
    {
        common += 1;
    }
    let mut result = Vec::new();
    result.extend((common..base.len()).map(|_| "..".to_string()));
    result.extend(target[common..].iter().cloned());
    if result.is_empty() {
        ".".to_string()
    } else {
        result.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(value) => Some(validate_context(value, PathError::InvalidHome)?),
        None => None,
    };
    let lookup = validate_links(links)?;

    let resolved_base = resolve_links_with_context(
        base,
        &cwd_parts,
        home_parts.as_deref(),
        links,
        &lookup,
        max_expansions,
    )?;
    let resolved_target = resolve_links_with_context(
        target,
        &cwd_parts,
        home_parts.as_deref(),
        links,
        &lookup,
        max_expansions,
    )?;

    let base_parts = component_strings(&resolved_base);
    let target_parts = component_strings(&resolved_target);
    let target_is_directory = resolved_target.ends_with('/');
    let mut candidates = vec![route(&base_parts, &target_parts)];

    for link in links {
        let probe = resolve_links_with_context(
            &link.path,
            &[],
            None,
            links,
            &lookup,
            max_expansions,
        );
        let Ok(probe) = probe else {
            continue;
        };
        let ancestor = component_strings(&probe);
        if ancestor.len() <= target_parts.len()
            && ancestor
                .iter()
                .zip(target_parts.iter())
                .all(|(left, right)| left == right)
        {
            let key = component_strings(&link.path);
            let mut candidate = route(&base_parts, &key);
            let suffix = &target_parts[ancestor.len()..];
            if !suffix.is_empty() {
                if candidate == "." {
                    candidate = suffix.join("/");
                } else {
                    candidate.push('/');
                    candidate.push_str(&suffix.join("/"));
                }
            }
            candidates.push(candidate);
        }
    }

    let mut valid = Vec::new();
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate.insert_str(0, "./");
        }
        if target_is_directory && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        let Ok(resolved) = resolve_links_with_context(
            &candidate,
            &base_parts,
            home_parts.as_deref(),
            links,
            &lookup,
            max_expansions,
        ) else {
            continue;
        };

        if component_strings(&resolved) == target_parts {
            valid.push(candidate);
        }
    }

    valid.sort_by(|left, right| {
        left.len()
            .cmp(&right.len())
            .then_with(|| left.as_bytes().cmp(right.as_bytes()))
    });
    valid.into_iter().next().ok_or(PathError::Escape { at: 0 })
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]

    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn build_path(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce(
    stack: &mut Vec<String>,
    raw: &str,
    offset: usize,
    absolute: bool,
    allow_relative_parents: bool,
) -> Result<(), PathError> {
    let mut cursor = 0;
    for part in raw.split('/') {
        let start = cursor;
        cursor += part.len() + 1;

        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at: offset + start });
            } else if allow_relative_parents {
                stack.push("..".to_owned());
            } else {
                return Err(PathError::Escape { at: offset + start });
            }
        } else {
            stack.push(part.to_owned());
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    reduce(&mut stack, path, 0, absolute, !absolute)?;
    Ok(build_path(&stack, absolute, intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized))
}

fn validated_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            let normalized = normalize(path).map_err(|_| PathError::InvalidHome)?;
            Ok(Some(components(&normalized)))
        }
    }
}

fn tilde_remainder(path: &str) -> Result<Option<&str>, PathError> {
    if !path.starts_with('~') {
        return Ok(None);
    }
    if path == "~" {
        return Ok(Some(""));
    }
    if path.as_bytes().get(1) == Some(&b'/') {
        return Ok(Some(&path[2..]));
    }
    Err(PathError::UnsupportedTilde { at: 1 })
}

fn resolve_core(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validated_home(home)?;
    validate_raw(path)?;

    let tilde = tilde_remainder(path)?;
    let absolute = path.starts_with('/');
    let (mut stack, remainder, offset) = if absolute {
        (Vec::new(), path, 0)
    } else if let Some(rest) = tilde {
        let parts = home_parts.ok_or(PathError::MissingHome)?;
        let offset = if path == "~" { path.len() } else { 2 };
        (parts, rest, offset)
    } else {
        (cwd_parts, path, 0)
    };

    reduce(&mut stack, remainder, offset, true, false)?;
    let intent = if tilde.is_some() {
        path != "~" && path.as_bytes().get(1) == Some(&b'/')
            && directory_intent(path)
    } else {
        directory_intent(path)
    };
    Ok(build_path(&stack, true, intent))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    resolve_core(path, cwd, home)
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut map = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = map.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        map.insert(link.path.clone(), index);
    }
    Ok(map)
}

fn absolute_string(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn walk_stream(
    stack: &mut Vec<String>,
    raw: &str,
    offset: usize,
    absolute: bool,
    links: &[Link],
    link_map: &HashMap<String, usize>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    let mut cursor = 0;
    let parts: Vec<(String, usize)> = raw
        .split('/')
        .map(|part| {
            let start = cursor;
            cursor += part.len() + 1;
            (part.to_owned(), start)
        })
        .collect();

    walk_parts(
        stack,
        &parts,
        offset,
        absolute,
        links,
        link_map,
        active,
        expansions,
        max_expansions,
    )
}

fn walk_parts(
    stack: &mut Vec<String>,
    parts: &[(String, usize)],
    offset: usize,
    absolute: bool,
    links: &[Link],
    link_map: &HashMap<String, usize>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    for (position, (part, local_at)) in parts.iter().enumerate() {
        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape {
                    at: offset + *local_at,
                });
            }
            continue;
        }

        stack.push(part.clone());
        let key = absolute_string(stack);
        if let Some(&index) = link_map.get(&key) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if *expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            *expansions += 1;
            active.push(index);

            let target = &links[index].target;
            let target_absolute = target.starts_with('/');
            if target_absolute {
                stack.clear();
            } else {
                stack.pop();
            }

            let result = walk_stream(
                stack,
                target,
                0,
                target_absolute,
                links,
                link_map,
                active,
                expansions,
                max_expansions,
            );
            active.pop();
            result?;

            // Remaining components continue from the expanded target.
            if position + 1 < parts.len() {
                return walk_parts(
                    stack,
                    &parts[position + 1..],
                    offset,
                    absolute,
                    links,
                    link_map,
                    active,
                    expansions,
                    max_expansions,
                );
            }
            return Ok(());
        }
    }
    let _ = absolute;
    Ok(())
}

fn resolve_links_core(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validated_home(home)?;
    let link_map = validate_links(links)?;
    validate_raw(path)?;

    let tilde = tilde_remainder(path)?;
    let (mut stack, remainder, offset, absolute) = if path.starts_with('/') {
        (Vec::new(), path, 0, true)
    } else if let Some(rest) = tilde {
        (home_parts.ok_or(PathError::MissingHome)?, rest, if path == "~" { path.len() } else { 2 }, false)
    } else {
        (cwd_parts, path, 0, false)
    };

    let mut active = Vec::new();
    let mut expansions = 0;
    walk_stream(
        &mut stack,
        remainder,
        offset,
        true,
        links,
        &link_map,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;

    let intent = if tilde.is_some() {
        path != "~" && path.as_bytes().get(1) == Some(&b'/') && directory_intent(path)
    } else {
        directory_intent(path)
    };
    let _ = absolute;
    Ok(build_path(&stack, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links_core(path, cwd, home, links, max_expansions)
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base =
        resolve_links_core(base, cwd, home, links, max_expansions)?;
    let resolved_target =
        resolve_links_core(target, cwd, home, links, max_expansions)?;
    let base_parts = components(&resolved_base);
    let target_parts = components(&resolved_target);
    let target_dir = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];
    for link in links {
        if let Ok(probe) =
            resolve_links_core(&link.path, "/", None, links, max_expansions)
        {
            let probe_parts = components(&probe);
            if target_parts.starts_with(&probe_parts) {
                let mut candidate = route(&base_parts, &components(&link.path));
                let suffix = &target_parts[probe_parts.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let mut best: Option<String> = None;
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_dir && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        if let Ok(resolved) =
            resolve_links_core(&candidate, &resolved_base, home, links, max_expansions)
        {
            if components(&resolved) == target_parts {
                match &best {
                    None => best = Some(candidate),
                    Some(current)
                        if candidate.len() < current.len()
                            || (candidate.len() == current.len()
                                && candidate.as_bytes() < current.as_bytes()) =>
                    {
                        best = Some(candidate);
                    }
                    _ => {}
                }
            }
        }
    }

    Ok(best.unwrap_or_else(|| route(&base_parts, &target_parts)))
}

use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn directory_intent(path: &str) -> bool {
    if path.ends_with('/') {
        return true;
    }
    path.split('/')
        .filter(|part| !part.is_empty())
        .last()
        .is_some_and(|part| part == "." || part == "..")
}

fn parts(path: &str) -> Vec<&str> {
    path.split('/').filter(|part| !part.is_empty()).collect()
}

fn reduce_component(stack: &mut Vec<String>, part: &str, absolute: bool, at: usize) -> Result<(), PathError> {
    match part {
        "." => {}
        ".." => {
            if stack.last().is_some_and(|v| v != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at });
            } else {
                stack.push("..".to_string());
            }
        }
        _ => stack.push(part.to_string()),
    }
    Ok(())
}

fn format_path(stack: &[String], absolute: bool, intent: bool) -> String {
    let joined = stack.join("/");
    if absolute {
        if joined.is_empty() {
            "/".to_string()
        } else if intent {
            format!("/{joined}/")
        } else {
            format!("/{joined}")
        }
    } else if joined.is_empty() {
        ".".to_string()
    } else if intent {
        format!("{joined}/")
    } else {
        joined
    }
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.starts_with('/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    let mut offset = 0;
    for part in path.split('/') {
        if !part.is_empty() {
            reduce_component(&mut stack, part, absolute, offset)?;
        }
        offset += part.len() + 1;
    }
    Ok(format_path(&stack, absolute, intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(parts(&normalized).into_iter().map(str::to_string).collect())
}

fn validate_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            let normalized = normalize(path).map_err(|_| PathError::InvalidHome)?;
            Ok(Some(parts(&normalized).into_iter().map(str::to_string).collect()))
        }
    }
}

fn tilde_request(path: &str) -> Result<(bool, usize), PathError> {
    if !path.starts_with('~') {
        return Ok((false, 0));
    }
    if path == "~" {
        return Ok((true, 1));
    }
    if path.as_bytes().get(1) == Some(&b'/') {
        return Ok((true, 2));
    }
    Err(PathError::UnsupportedTilde { at: 1 })
}

fn traversal_offset(path: &str, base: usize) -> Vec<(String, usize)> {
    let mut out = Vec::new();
    let mut start = base;
    for component in path[base..].split('/') {
        if !component.is_empty() {
            out.push((component.to_string(), start));
        }
        start += component.len() + 1;
    }
    out
}

fn traverse_plain(
    path: &str,
    start: Vec<String>,
    absolute: bool,
    prefix: usize,
    intent: bool,
) -> Result<String, PathError> {
    let mut stack = start;
    for (part, at) in traversal_offset(path, prefix) {
        reduce_component(&mut stack, &part, true, at)?;
    }
    Ok(format_path(&stack, true, intent))
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (tilde, consumed) = tilde_request(path)?;

    if tilde {
        let start = home_parts.ok_or(PathError::MissingHome)?;
        return traverse_plain(path, start, true, consumed, intent);
    }
    if path.starts_with('/') {
        traverse_plain(path, Vec::new(), true, 0, intent)
    } else {
        traverse_plain(path, cwd_parts, false, 0, intent)
    }
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, (usize, String)>, PathError> {
    let mut map = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some((first, _)) = map.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        map.insert(link.path.clone(), (index, link.target.clone()));
    }
    Ok(map)
}

fn stack_path(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn push_component(
    stack: &mut Vec<String>,
    part: &str,
    at: usize,
    links: &HashMap<String, (usize, String)>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    match part {
        "." => return Ok(()),
        ".." => {
            if stack.pop().is_none() {
                return Err(PathError::Escape { at });
            }
            return Ok(());
        }
        _ => stack.push(part.to_string()),
    }

    let key = stack_path(stack);
    if let Some((index, target)) = links.get(&key) {
        if active.contains(index) {
            return Err(PathError::Loop { index: *index });
        }
        if *expansions >= max_expansions {
            return Err(PathError::Limit { index: *index });
        }
        *expansions += 1;
        active.push(*index);
        if !target.starts_with('/') {
            stack.pop();
        } else {
            stack.clear();
        }
        let target_parts = traversal_offset(target, 0);
        for (target_part, target_at) in target_parts {
            process_component(
                stack,
                &target_part,
                target_at,
                links,
                active,
                expansions,
                max_expansions,
            )?;
        }
        active.pop();
    }
    Ok(())
}

fn process_component(
    stack: &mut Vec<String>,
    part: &str,
    at: usize,
    links: &HashMap<String, (usize, String)>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    if part == "." {
        return Ok(());
    }
    if part == ".." {
        if stack.pop().is_none() {
            return Err(PathError::Escape { at });
        }
        return Ok(());
    }
    push_component(
        stack,
        part,
        at,
        links,
        active,
        expansions,
        max_expansions,
    )
}

fn resolve_links_internal(
    path: &str,
    cwd_parts: &[String],
    home_parts: Option<&[String]>,
    links: &HashMap<String, (usize, String)>,
    max_expansions: usize,
) -> Result<String, PathError> {
    validate_raw(path)?;
    let intent = directory_intent(path);
    let (tilde, consumed) = tilde_request(path)?;

    let mut stack = if tilde {
        home_parts.ok_or(PathError::MissingHome)?.to_vec()
    } else if path.starts_with('/') {
        Vec::new()
    } else {
        cwd_parts.to_vec()
    };

    let mut active = Vec::new();
    let mut expansions = 0usize;
    for (part, at) in traversal_offset(path, consumed) {
        process_component(
            &mut stack,
            &part,
            at,
            links,
            &mut active,
            &mut expansions,
            max_expansions,
        )?;
    }
    Ok(format_path(&stack, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    let table = validate_links(links)?;
    resolve_links_internal(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        &table,
        max_expansions,
    )
}

fn component_vec(path: &str) -> Vec<String> {
    parts(path).into_iter().map(str::to_string).collect()
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target)
        .take_while(|(a, b)| a == b)
        .count();
    let mut result = vec!["..".to_string(); base.len() - common];
    result.extend(target[common..].iter().cloned());
    if result.is_empty() {
        ".".to_string()
    } else {
        result.join("/")
    }
}

fn candidate_with_suffix(mut prefix: String, suffix: &[String]) -> String {
    if suffix.is_empty() {
        return prefix;
    }
    if prefix == "." {
        return suffix.join("/");
    }
    if !prefix.is_empty() {
        prefix.push('/');
    }
    prefix.push_str(&suffix.join("/"));
    prefix
}

fn protect_tilde(candidate: String) -> String {
    if candidate.starts_with('~') {
        format!("./{candidate}")
    } else {
        candidate
    }
}

fn add_intent(candidate: String, target: &str) -> String {
    if candidate != "." && target.ends_with('/') && !candidate.ends_with('/') {
        format!("{candidate}/")
    } else {
        candidate
    }
}

fn resolve_with_existing_table(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    table: &HashMap<String, (usize, String)>,
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    resolve_links_internal(
        path,
        &cwd_parts,
        home_parts.as_deref(),
        table,
        max_expansions,
    )
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validate_home(home)?;
    let table = validate_links(links)?;

    let resolved_base = resolve_links_internal(
        base,
        &cwd_parts,
        home_parts.as_deref(),
        &table,
        max_expansions,
    )?;
    let resolved_target = resolve_links_internal(
        target,
        &cwd_parts,
        home_parts.as_deref(),
        &table,
        max_expansions,
    )?;

    let base_components = component_vec(&resolved_base);
    let target_components = component_vec(&resolved_target);
    let mut candidates = vec![route(&base_components, &target_components)];

    for link in links {
        let probe = resolve_with_existing_table(&link.path, "/", None, &table, max_expansions);
        if let Ok(probe) = probe {
            let anchor = component_vec(&probe);
            if anchor.len() <= target_components.len()
                && anchor
                    .iter()
                    .zip(&target_components)
                    .all(|(a, b)| a == b)
            {
                let key_components = component_vec(&link.path);
                let mut candidate = route(&base_components, &key_components);
                candidate = candidate_with_suffix(candidate, &target_components[anchor.len()..]);
                candidates.push(candidate);
            }
        }
    }

    let target_directory = resolved_target.ends_with('/');
    let mut valid = Vec::new();
    for candidate in candidates {
        let candidate = protect_tilde(candidate);
        let candidate = add_intent(candidate, &resolved_target);
        if let Ok(result) = resolve_links_internal(
            &candidate,
            &base_components,
            home_parts.as_deref(),
            &table,
            max_expansions,
        ) {
            if component_vec(&result) == target_components {
                valid.push(candidate);
            }
        }
    }

    valid.sort_by(|a, b| {
        a.len()
            .cmp(&b.len())
            .then_with(|| a.as_bytes().cmp(b.as_bytes()))
    });
    let result = valid.into_iter().next().expect("direct candidate must validate");

    if target_directory && result == "." {
        return Ok(".".to_string());
    }
    Ok(result)
}
