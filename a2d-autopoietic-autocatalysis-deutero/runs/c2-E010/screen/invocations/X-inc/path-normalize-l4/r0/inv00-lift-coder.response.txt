use std::collections::HashMap;

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str) -> Vec<String> {
    path.split('/')
        .filter(|part| !part.is_empty())
        .map(str::to_owned)
        .collect()
}

fn directory_intent(path: &str) -> bool {
    path.ends_with('/')
        || path
            .split('/')
            .filter(|part| !part.is_empty())
            .last()
            .is_some_and(|part| part == "." || part == "..")
}

fn build_path(parts: &[String], absolute: bool, intent: bool) -> String {
    let mut result = if absolute {
        if parts.is_empty() {
            "/".to_owned()
        } else {
            format!("/{}", parts.join("/"))
        }
    } else if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    };

    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn reduce(
    stack: &mut Vec<String>,
    raw: &str,
    offset: usize,
    absolute: bool,
    allow_relative_parents: bool,
) -> Result<(), PathError> {
    let mut cursor = 0;
    for part in raw.split('/') {
        let start = cursor;
        cursor += part.len() + 1;

        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            if stack.last().is_some_and(|last| last != "..") {
                stack.pop();
            } else if absolute {
                return Err(PathError::Escape { at: offset + start });
            } else if allow_relative_parents {
                stack.push("..".to_owned());
            } else {
                return Err(PathError::Escape { at: offset + start });
            }
        } else {
            stack.push(part.to_owned());
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut stack = Vec::new();
    reduce(&mut stack, path, 0, absolute, !absolute)?;
    Ok(build_path(&stack, absolute, intent))
}

fn validate_context(path: &str) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(PathError::InvalidCwd);
    }
    let normalized = normalize(path).map_err(|_| PathError::InvalidCwd)?;
    Ok(components(&normalized))
}

fn validated_home(home: Option<&str>) -> Result<Option<Vec<String>>, PathError> {
    match home {
        None => Ok(None),
        Some(path) => {
            if !path.starts_with('/') {
                return Err(PathError::InvalidHome);
            }
            let normalized = normalize(path).map_err(|_| PathError::InvalidHome)?;
            Ok(Some(components(&normalized)))
        }
    }
}

fn tilde_remainder(path: &str) -> Result<Option<&str>, PathError> {
    if !path.starts_with('~') {
        return Ok(None);
    }
    if path == "~" {
        return Ok(Some(""));
    }
    if path.as_bytes().get(1) == Some(&b'/') {
        return Ok(Some(&path[2..]));
    }
    Err(PathError::UnsupportedTilde { at: 1 })
}

fn resolve_core(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validated_home(home)?;
    validate_raw(path)?;

    let tilde = tilde_remainder(path)?;
    let absolute = path.starts_with('/');
    let (mut stack, remainder, offset) = if absolute {
        (Vec::new(), path, 0)
    } else if let Some(rest) = tilde {
        let parts = home_parts.ok_or(PathError::MissingHome)?;
        let offset = if path == "~" { path.len() } else { 2 };
        (parts, rest, offset)
    } else {
        (cwd_parts, path, 0)
    };

    reduce(&mut stack, remainder, offset, true, false)?;
    let intent = if tilde.is_some() {
        path != "~" && path.as_bytes().get(1) == Some(&b'/')
            && directory_intent(path)
    } else {
        directory_intent(path)
    };
    Ok(build_path(&stack, true, intent))
}

pub fn resolve(
    path: &str,
    cwd: &str,
    home: Option<&str>,
) -> Result<String, PathError> {
    resolve_core(path, cwd, home)
}

fn validate_links(links: &[Link]) -> Result<HashMap<String, usize>, PathError> {
    let mut map = HashMap::new();
    for (index, link) in links.iter().enumerate() {
        let valid_path = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !valid_path || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }
        if let Some(&first) = map.get(&link.path) {
            return Err(PathError::DuplicateLink {
                first,
                second: index,
            });
        }
        map.insert(link.path.clone(), index);
    }
    Ok(map)
}

fn absolute_string(stack: &[String]) -> String {
    if stack.is_empty() {
        "/".to_owned()
    } else {
        format!("/{}", stack.join("/"))
    }
}

fn walk_stream(
    stack: &mut Vec<String>,
    raw: &str,
    offset: usize,
    absolute: bool,
    links: &[Link],
    link_map: &HashMap<String, usize>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    let mut cursor = 0;
    let parts: Vec<(String, usize)> = raw
        .split('/')
        .map(|part| {
            let start = cursor;
            cursor += part.len() + 1;
            (part.to_owned(), start)
        })
        .collect();

    walk_parts(
        stack,
        &parts,
        offset,
        absolute,
        links,
        link_map,
        active,
        expansions,
        max_expansions,
    )
}

fn walk_parts(
    stack: &mut Vec<String>,
    parts: &[(String, usize)],
    offset: usize,
    absolute: bool,
    links: &[Link],
    link_map: &HashMap<String, usize>,
    active: &mut Vec<usize>,
    expansions: &mut usize,
    max_expansions: usize,
) -> Result<(), PathError> {
    for (position, (part, local_at)) in parts.iter().enumerate() {
        if part.is_empty() || part == "." {
            continue;
        }
        if part == ".." {
            if stack.pop().is_none() {
                return Err(PathError::Escape {
                    at: offset + *local_at,
                });
            }
            continue;
        }

        stack.push(part.clone());
        let key = absolute_string(stack);
        if let Some(&index) = link_map.get(&key) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if *expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            *expansions += 1;
            active.push(index);

            let target = &links[index].target;
            let target_absolute = target.starts_with('/');
            if target_absolute {
                stack.clear();
            } else {
                stack.pop();
            }

            let result = walk_stream(
                stack,
                target,
                0,
                target_absolute,
                links,
                link_map,
                active,
                expansions,
                max_expansions,
            );
            active.pop();
            result?;

            // Remaining components continue from the expanded target.
            if position + 1 < parts.len() {
                return walk_parts(
                    stack,
                    &parts[position + 1..],
                    offset,
                    absolute,
                    links,
                    link_map,
                    active,
                    expansions,
                    max_expansions,
                );
            }
            return Ok(());
        }
    }
    let _ = absolute;
    Ok(())
}

fn resolve_links_core(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd)?;
    let home_parts = validated_home(home)?;
    let link_map = validate_links(links)?;
    validate_raw(path)?;

    let tilde = tilde_remainder(path)?;
    let (mut stack, remainder, offset, absolute) = if path.starts_with('/') {
        (Vec::new(), path, 0, true)
    } else if let Some(rest) = tilde {
        (home_parts.ok_or(PathError::MissingHome)?, rest, if path == "~" { path.len() } else { 2 }, false)
    } else {
        (cwd_parts, path, 0, false)
    };

    let mut active = Vec::new();
    let mut expansions = 0;
    walk_stream(
        &mut stack,
        remainder,
        offset,
        true,
        links,
        &link_map,
        &mut active,
        &mut expansions,
        max_expansions,
    )?;

    let intent = if tilde.is_some() {
        path != "~" && path.as_bytes().get(1) == Some(&b'/') && directory_intent(path)
    } else {
        directory_intent(path)
    };
    let _ = absolute;
    Ok(build_path(&stack, true, intent))
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    resolve_links_core(path, cwd, home, links, max_expansions)
}

fn route(base: &[String], target: &[String]) -> String {
    let common = base
        .iter()
        .zip(target.iter())
        .take_while(|(a, b)| a == b)
        .count();
    let mut parts = vec!["..".to_owned(); base.len() - common];
    parts.extend(target[common..].iter().cloned());
    if parts.is_empty() {
        ".".to_owned()
    } else {
        parts.join("/")
    }
}

pub fn relative_to(
    base: &str,
    target: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let resolved_base =
        resolve_links_core(base, cwd, home, links, max_expansions)?;
    let resolved_target =
        resolve_links_core(target, cwd, home, links, max_expansions)?;
    let base_parts = components(&resolved_base);
    let target_parts = components(&resolved_target);
    let target_dir = resolved_target.ends_with('/');

    let mut candidates = vec![route(&base_parts, &target_parts)];
    for link in links {
        if let Ok(probe) =
            resolve_links_core(&link.path, "/", None, links, max_expansions)
        {
            let probe_parts = components(&probe);
            if target_parts.starts_with(&probe_parts) {
                let mut candidate = route(&base_parts, &components(&link.path));
                let suffix = &target_parts[probe_parts.len()..];
                if !suffix.is_empty() {
                    if candidate == "." {
                        candidate = suffix.join("/");
                    } else {
                        candidate.push('/');
                        candidate.push_str(&suffix.join("/"));
                    }
                }
                candidates.push(candidate);
            }
        }
    }

    let mut best: Option<String> = None;
    for mut candidate in candidates {
        if candidate.starts_with('~') {
            candidate = format!("./{candidate}");
        }
        if target_dir && candidate != "." && !candidate.ends_with('/') {
            candidate.push('/');
        }

        if let Ok(resolved) =
            resolve_links_core(&candidate, &resolved_base, home, links, max_expansions)
        {
            if components(&resolved) == target_parts {
                match &best {
                    None => best = Some(candidate),
                    Some(current)
                        if candidate.len() < current.len()
                            || (candidate.len() == current.len()
                                && candidate.as_bytes() < current.as_bytes()) =>
                    {
                        best = Some(candidate);
                    }
                    _ => {}
                }
            }
        }
    }

    Ok(best.unwrap_or_else(|| route(&base_parts, &target_parts)))
}
