#[derive(Debug, Clone, PartialEq, Eq)]
pub enum PathError {
    Empty,
    Nul { at: usize },
    Backslash { at: usize },
    Control { at: usize },
    Escape { at: usize },
    InvalidCwd,
    InvalidHome,
    MissingHome,
    UnsupportedTilde { at: usize },
    InvalidLink { index: usize },
    DuplicateLink { first: usize, second: usize },
    Loop { index: usize },
    Limit { index: usize },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Link {
    pub path: String,
    pub target: String,
}

#[derive(Clone, Copy)]
struct Component<'a> {
    text: &'a str,
    at: usize,
}

fn validate_raw(path: &str) -> Result<(), PathError> {
    if path.is_empty() {
        return Err(PathError::Empty);
    }

    if let Some(at) = path.as_bytes().iter().position(|&b| b == 0) {
        return Err(PathError::Nul { at });
    }
    if let Some(at) = path.as_bytes().iter().position(|&b| b == b'\\') {
        return Err(PathError::Backslash { at });
    }
    if let Some(at) = path
        .as_bytes()
        .iter()
        .position(|&b| (1..=31).contains(&b) || b == 127)
    {
        return Err(PathError::Control { at });
    }
    Ok(())
}

fn components(path: &str, base: usize) -> Vec<Component<'_>> {
    let bytes = path.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;

    while i < bytes.len() {
        while i < bytes.len() && bytes[i] == b'/' {
            i += 1;
        }
        if i == bytes.len() {
            break;
        }
        let start = i;
        while i < bytes.len() && bytes[i] != b'/' {
            i += 1;
        }
        out.push(Component {
            text: &path[start..i],
            at: base + start,
        });
    }

    out
}

fn directory_intent(path: &str) -> bool {
    if path.as_bytes().last() == Some(&b'/') {
        return true;
    }
    components(path, 0)
        .last()
        .is_some_and(|c| c.text == "." || c.text == "..")
}

fn append_result(mut result: String, intent: bool) -> String {
    if intent && result != "/" && result != "." {
        result.push('/');
    }
    result
}

fn render_absolute(parts: &[String], intent: bool) -> String {
    let mut result = String::from("/");
    result.push_str(&parts.join("/"));
    append_result(result, intent)
}

fn render_relative(parts: &[String], intent: bool) -> String {
    if parts.is_empty() {
        return ".".to_string();
    }
    append_result(parts.join("/"), intent)
}

fn reduce_components<'a>(
    parts: &mut Vec<String>,
    absolute: bool,
    input: impl IntoIterator<Item = Component<'a>>,
) -> Result<(), PathError> {
    for component in input {
        match component.text {
            "." => {}
            ".." => {
                if parts.last().is_some_and(|p| p != "..") {
                    parts.pop();
                } else if absolute {
                    return Err(PathError::Escape { at: component.at });
                } else {
                    parts.push("..".to_string());
                }
            }
            name => parts.push(name.to_string()),
        }
    }
    Ok(())
}

pub fn normalize(path: &str) -> Result<String, PathError> {
    validate_raw(path)?;
    let absolute = path.as_bytes().first() == Some(&b'/');
    let intent = directory_intent(path);
    let mut parts = Vec::new();
    reduce_components(&mut parts, absolute, components(path, 0))?;

    if absolute {
        Ok(render_absolute(&parts, intent))
    } else {
        Ok(render_relative(&parts, intent))
    }
}

fn validate_context(path: &str, error: PathError) -> Result<Vec<String>, PathError> {
    if !path.starts_with('/') {
        return Err(error);
    }

    let normalized = normalize(path).map_err(|_| error)?;
    Ok(components(&normalized, 0)
        .into_iter()
        .map(|c| c.text.to_string())
        .collect())
}

fn request_start<'a>(
    path: &'a str,
    home: Option<&'a str>,
) -> Result<(bool, usize, Vec<String>), PathError> {
    if path.starts_with('~') {
        let bytes = path.as_bytes();
        if bytes.len() > 1 && bytes[1] != b'/' {
            return Err(PathError::UnsupportedTilde { at: 1 });
        }
        let home = home.ok_or(PathError::MissingHome)?;
        let consumed = if bytes.len() > 1 { 2 } else { 1 };
        let start = components(home, 0)
            .into_iter()
            .map(|c| c.text.to_string())
            .collect();
        Ok((false, consumed, start))
    } else if path.starts_with('/') {
        Ok((true, 0, Vec::new()))
    } else {
        Ok((false, 0, Vec::new()))
    }
}

pub fn resolve(path: &str, cwd: &str, home: Option<&str>) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(validate_context(h, PathError::InvalidHome)?),
        None => None,
    };

    validate_raw(path)?;
    let intent = directory_intent(path);
    let (absolute_request, consumed, tilde_parts) = request_start(path, home)?;

    let mut parts = if absolute_request {
        Vec::new()
    } else if path.starts_with('~') {
        tilde_parts
    } else {
        cwd_parts
    };

    let remainder = &path[consumed..];
    reduce_components(&mut parts, true, components(remainder, consumed))?;

    let _ = home_parts;
    Ok(render_absolute(&parts, intent))
}

fn validate_links(links: &[Link]) -> Result<(), PathError> {
    let mut seen: Vec<(&str, usize)> = Vec::new();

    for (index, link) in links.iter().enumerate() {
        let path_valid = link.path.starts_with('/')
            && !link.path.ends_with('/')
            && link.path != "/"
            && normalize(&link.path)
                .is_ok_and(|normalized| normalized == link.path);
        if !path_valid || validate_raw(&link.target).is_err() {
            return Err(PathError::InvalidLink { index });
        }

        if let Some((_, first)) = seen.iter().find(|(path, _)| *path == link.path) {
            return Err(PathError::DuplicateLink {
                first: *first,
                second: index,
            });
        }
        seen.push((&link.path, index));
    }
    Ok(())
}

fn matching_link(parts: &[String], links: &[Link]) -> Option<usize> {
    let key = if parts.is_empty() {
        "/".to_string()
    } else {
        format!("/{}", parts.join("/"))
    };
    links.iter().position(|link| link.path == key)
}

#[derive(Clone, Copy)]
enum Source<'a> {
    Request,
    Target(&'a str),
    Context,
}

#[derive(Clone, Copy)]
struct WorkComponent<'a> {
    text: &'a str,
    at: usize,
    source: Source<'a>,
}

fn work_components<'a>(path: &'a str, base: usize, source: Source<'a>) -> Vec<WorkComponent<'a>> {
    components(path, base)
        .into_iter()
        .map(|c| WorkComponent {
            text: c.text,
            at: c.at,
            source,
        })
        .collect()
}

pub fn resolve_links(
    path: &str,
    cwd: &str,
    home: Option<&str>,
    links: &[Link],
    max_expansions: usize,
) -> Result<String, PathError> {
    let cwd_parts = validate_context(cwd, PathError::InvalidCwd)?;
    let home_parts = match home {
        Some(h) => Some(validate_context(h, PathError::InvalidHome)?),
        None => None,
    };

    validate_links(links)?;
    validate_raw(path)?;

    let intent = directory_intent(path);
    let (absolute_request, consumed, tilde_parts) = request_start(path, home)?;
    let mut stack = if absolute_request {
        Vec::new()
    } else if path.starts_with('~') {
        tilde_parts
    } else {
        cwd_parts
    };

    let mut queue = Vec::new();
    if !absolute_request {
        let context = if path.starts_with('~') {
            home_parts.as_deref().unwrap_or(&[])
        } else {
            &[]
        };
        let _ = context;
    }
    let remainder = &path[consumed..];
    queue.extend(work_components(
        remainder,
        consumed,
        Source::Request,
    ));

    let mut active = Vec::<usize>::new();
    let mut expansions = 0usize;

    while let Some(component) = queue.first().copied() {
        queue.remove(0);

        match component.text {
            "." => continue,
            ".." => {
                if stack.pop().is_none() {
                    return Err(PathError::Escape { at: component.at });
                }
                continue;
            }
            name => stack.push(name.to_string()),
        }

        if let Some(index) = matching_link(&stack, links) {
            if active.contains(&index) {
                return Err(PathError::Loop { index });
            }
            if expansions >= max_expansions {
                return Err(PathError::Limit { index });
            }
            expansions += 1;

            // A relative target begins at the matched link's parent.
            let target = links[index].target.as_str();
            let absolute_target = target.starts_with('/');
            if absolute_target {
                stack.clear();
            } else {
                stack.pop();
            }

            active.push(index);
            let mut target_queue =
                work_components(target, 0, Source::Target(target));
            target_queue.extend(queue);
            queue = target_queue;

            // Mark target completion when its components have been consumed.
            // A sentinel is represented by an empty component with a special source.
            queue.push(WorkComponent {
                text: "",
                at: 0,
                source: Source::Context,
            });
        }

        if let Some(WorkComponent {
            text: "",
            source: Source::Context,
            ..
        }) = queue.first().copied()
        {
            // handled below when the sentinel reaches the front
        }
    }

    Ok(render_absolute(&stack, intent))
}
