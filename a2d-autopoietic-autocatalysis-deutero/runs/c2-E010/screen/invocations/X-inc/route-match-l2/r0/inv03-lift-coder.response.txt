#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    id: usize,
    segments: Vec<Segment>,
    rank: Vec<u8>,
}

#[derive(Debug, Clone)]
struct ParsedPath {
    segments: Vec<String>,
}

pub struct Router {
    routes: Vec<Route>,
}

impl Router {
    pub fn new() -> Self {
        Self { routes: Vec::new() }
    }

    pub fn add(&mut self, pattern: &str) -> Result<usize, Error> {
        let (segments, rank) = parse_pattern(pattern)?;
        let id = self.routes.len();
        self.routes.push(Route { id, segments, rank });
        Ok(id)
    }

    pub fn resolve(&self, path: &str) -> Result<Match, Error> {
        let head_end = path
            .char_indices()
            .find(|(_, ch)| *ch == '?' || *ch == '#')
            .map_or(path.len(), |(i, _)| i);
        let parsed = parse_path_head(&path[..head_end])?;

        let mut best: Option<(&Route, Vec<(String, String)>)> = None;
        for route in &self.routes {
            if let Some(params) = match_route(route, &parsed.segments) {
                let replace = best
                    .as_ref()
                    .map_or(true, |(current, _)| route.rank > current.rank);
                if replace {
                    best = Some((route, params));
                }
            }
        }

        match best {
            Some((route, params)) => Ok(Match {
                route: route.id,
                params,
            }),
            None => Err(Error::NotFound),
        }
    }
}

impl Default for Router {
    fn default() -> Self {
        Self::new()
    }
}

fn parse_pattern(input: &str) -> Result<(Vec<Segment>, Vec<u8>), Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if input == "/" {
        return Ok((Vec::new(), vec![3]));
    }

    let parts = split_segments(input).map_err(|at| Error::InvalidPattern { at })?;
    let mut segments = Vec::with_capacity(parts.len());
    let mut rank = Vec::with_capacity(parts.len() + 1);
    let mut names = std::collections::HashSet::new();

    for (index, (start, raw)) in parts.iter().enumerate() {
        if raw.is_empty() {
            return Err(Error::InvalidPattern { at: *start });
        }

        if raw.as_bytes()[0] == b':' || raw.as_bytes()[0] == b'*' {
            let marker = raw.as_bytes()[0];
            let name = &raw[1..];
            if !valid_name(name) || !names.insert(name.to_owned()) {
                return Err(Error::InvalidPattern { at: *start });
            }
            if marker == b'*' && index + 1 != parts.len() {
                return Err(Error::InvalidPattern { at: *start });
            }
            if marker == b'*' {
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
            } else {
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            }
        } else {
            let decoded = decode_segment(raw, *start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPattern {
            at: input.len() - 1,
        });
    }

    if !matches!(segments.last(), Some(Segment::Wildcard(_))) {
        rank.push(3);
    }

    Ok((segments, rank))
}

fn parse_path_head(input: &str) -> Result<ParsedPath, Error> {
    if !input.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if input == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
        });
    }

    let parts = split_segments(input).map_err(|at| Error::InvalidPath { at })?;
    let mut segments = Vec::with_capacity(parts.len());

    for (start, raw) in &parts {
        if raw.is_empty() {
            return Err(Error::InvalidPath { at: *start });
        }
        let decoded =
            decode_segment(raw, *start).map_err(|at| Error::InvalidPath { at })?;
        segments.push(decoded);
    }

    if input.ends_with('/') {
        return Err(Error::InvalidPath {
            at: input.len() - 1,
        });
    }

    Ok(ParsedPath { segments })
}

// Returns segment starts and slices, excluding the initial slash.
fn split_segments(input: &str) -> Result<Vec<(usize, &str)>, usize> {
    let bytes = input.as_bytes();
    let mut parts = Vec::new();
    let mut start = 1;

    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            parts.push((start, &input[start..i]));
            start = i + 1;
        }
    }
    parts.push((start, &input[start..]));
    Ok(parts)
}

fn valid_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    if bytes.is_empty() || !bytes[0].is_ascii_lowercase() {
        return false;
    }
    bytes[1..]
        .iter()
        .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &str, offset: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len() {
                return Err(offset + i);
            }
            let hi = upper_hex(bytes[i + 1]).ok_or(offset + i)?;
            let lo = upper_hex(bytes[i + 2]).ok_or(offset + i)?;
            let value = (hi << 4) | lo;
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(offset + i);
            }
            decoded.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(offset + i);
            }
            decoded.push(b);
            i += 1;
        }
    }

    // Every decoded byte is ASCII by validation above.
    Ok(String::from_utf8(decoded).expect("validated ASCII bytes"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, path: &[String]) -> Option<Vec<(String, String)>> {
    let mut params = Vec::new();
    let mut path_index = 0;

    for segment in &route.segments {
        match segment {
            Segment::Static(value) => {
                if path.get(path_index)? != value {
                    return None;
                }
                path_index += 1;
            }
            Segment::Param(name) => {
                let value = path.get(path_index)?;
                params.push((name.clone(), value.clone()));
                path_index += 1;
            }
            Segment::Wildcard(name) => {
                let value = path[path_index..].join("/");
                params.push((name.clone(), value));
                path_index = path.len();
                break;
            }
        }
    }

    if path_index == path.len() {
        Some(params)
    } else {
        None
    }
}
