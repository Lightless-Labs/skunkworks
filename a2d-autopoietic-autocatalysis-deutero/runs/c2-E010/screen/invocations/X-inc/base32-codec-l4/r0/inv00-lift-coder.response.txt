#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

pub struct Encoder {
    config: Config,
    buffered: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            buffered: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = self.buffered.clone();
        combined.extend_from_slice(input);

        let groups = combined.len() / 5;
        let mut output = String::with_capacity(groups * 8);
        for chunk in combined[..groups * 5].chunks_exact(5) {
            encode_bytes(chunk, self.config.alphabet, &mut output);
        }

        self.buffered = combined[groups * 5..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if !self.buffered.is_empty() {
            let symbol_count = (self.buffered.len() * 8 + 4) / 5;
            encode_bytes(&self.buffered, self.config.alphabet, &mut output);
            output.truncate(symbol_count);

            if self.config.padding == Padding::Required {
                while output.len() % 8 != 0 {
                    output.push('=');
                }
            }
        }

        self.buffered.clear();
        self.closed = true;
        Ok(output)
    }
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn encode_bytes(bytes: &[u8], alphabet: Alphabet, output: &mut String) {
    let symbols = alphabet_bytes(alphabet);
    let mut accumulator: u64 = 0;
    for &byte in bytes {
        accumulator = (accumulator << 8) | u64::from(byte);
    }

    let total_bits = bytes.len() * 8;
    let symbol_count = (total_bits + 4) / 5;
    for i in 0..symbol_count {
        let shift = total_bits as isize - 5 * (i as isize + 1);
        let value = if shift >= 0 {
            ((accumulator >> shift as usize) & 31) as usize
        } else {
            ((accumulator << (-shift) as usize) & 31) as usize
        };
        output.push(symbols[value] as char);
    }
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    bit_buffer: u8,
    bit_count: u8,
    last_value: u8,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            bit_buffer: 0,
            bit_count: 0,
            last_value: 0,
            last_data_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: None,
            closed: false,
        }
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            if byte == b'=' {
                if next.config.padding == Padding::Omitted {
                    return Err(StreamError::Padding { offset });
                }

                let residue = next.data_count % 8;
                let expected = match residue {
                    2 => 6,
                    4 => 4,
                    5 => 3,
                    7 => 1,
                    _ => return Err(StreamError::Padding { offset }),
                };

                if let Some(expected_count) = next.expected_padding {
                    if next.padding_count >= expected_count {
                        return Err(StreamError::Padding { offset });
                    }
                } else {
                    next.expected_padding = Some(expected);
                    next.first_padding_offset = Some(offset);
                }

                next.padding_count += 1;
                next.position += 1;
                continue;
            }

            let (value, canonical) = match recognize(byte, next.config.alphabet) {
                Some(found) => found,
                None => return Err(StreamError::InvalidByte { offset, byte }),
            };

            if next.config.canonical && byte != canonical {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: canonical,
                });
            }

            if next.first_padding_offset.is_some() {
                return Err(StreamError::Padding { offset });
            }

            next.data_count += 1;
            next.last_value = value;
            next.last_data_offset = Some(offset);
            next.bit_buffer = (next.bit_buffer << 5) | value;
            next.bit_count += 5;

            while next.bit_count >= 8 {
                next.bit_count -= 8;
                output.push((next.bit_buffer >> next.bit_count) & 0xff);
                if next.bit_count == 0 {
                    next.bit_buffer = 0;
                } else {
                    next.bit_buffer &= (1u16.wrapping_shl(next.bit_count.into()) - 1) as u8;
                }
            }

            next.position += 1;
        }

        self.copy_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        }

        if self.config.padding == Padding::Required {
            let expected = match residue {
                0 => 0,
                2 => 6,
                4 => 4,
                5 => 3,
                7 => 1,
                _ => unreachable!(),
            };
            if self.padding_count != expected {
                return Err(StreamError::Padding {
                    offset: self.first_padding_offset.unwrap_or(self.position),
                });
            }
        }

        let unused_bits = match residue {
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused_bits != 0 && (self.last_value & ((1u8 << unused_bits) - 1)) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }

    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            last_value: self.last_value,
            last_data_offset: self.last_data_offset,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        }
    }

    fn copy_from(&mut self, other: Self) {
        *self = other;
    }
}

fn recognize(byte: u8, alphabet: Alphabet) -> Option<(u8, u8)> {
    let uppercase = byte.to_ascii_uppercase();
    let symbols = alphabet_bytes(alphabet);
    symbols
        .iter()
        .position(|&symbol| symbol == uppercase)
        .map(|value| (value as u8, uppercase))
}
