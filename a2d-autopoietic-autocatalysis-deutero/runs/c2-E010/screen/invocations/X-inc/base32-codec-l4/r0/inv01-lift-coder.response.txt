#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub struct Config {
    pub alphabet: Alphabet,
    pub padding: Padding,
    pub canonical: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum StreamError {
    InvalidByte { offset: usize, byte: u8 },
    NonCanonical { offset: usize, expected: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
    Closed,
}

const STANDARD: &[u8; 32] = b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567";
const HEX: &[u8; 32] = b"0123456789ABCDEFGHIJKLMNOPQRSTUV";

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => STANDARD,
        Alphabet::Hex => HEX,
    }
}

pub struct Encoder {
    config: Config,
    pending: Vec<u8>,
    closed: bool,
}

impl Encoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            pending: Vec::new(),
            closed: false,
        }
    }

    pub fn push(&mut self, input: &[u8]) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut combined = self.pending.clone();
        combined.extend_from_slice(input);

        let complete_len = combined.len() / 5 * 5;
        let mut output = String::new();
        let alphabet = alphabet_bytes(self.config.alphabet);

        for chunk in combined[..complete_len].chunks_exact(5) {
            let bits = ((chunk[0] as u64) << 32)
                | ((chunk[1] as u64) << 24)
                | ((chunk[2] as u64) << 16)
                | ((chunk[3] as u64) << 8)
                | chunk[4] as u64;
            for shift in (0..8).rev() {
                output.push(alphabet[((bits >> (shift * 5)) & 31) as usize] as char);
            }
        }

        self.pending = combined[complete_len..].to_vec();
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<String, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut output = String::new();
        if !self.pending.is_empty() {
            let alphabet = alphabet_bytes(self.config.alphabet);
            let count = self.pending.len() * 8;
            let symbols = (count + 4) / 5;
            let mut bits = 0u64;

            for &byte in &self.pending {
                bits = (bits << 8) | byte as u64;
            }
            bits <<= 40 - count;

            for i in 0..symbols {
                let shift = 35 - i * 5;
                output.push(alphabet[((bits >> shift) & 31) as usize] as char);
            }

            if self.config.padding == Padding::Required {
                while output.len() % 8 != 0 {
                    output.push('=');
                }
            }
        }

        self.pending.clear();
        self.closed = true;
        Ok(output)
    }
}

pub struct Decoder {
    config: Config,
    position: usize,
    data_count: usize,
    bit_buffer: u8,
    bit_count: u8,
    last_data_offset: Option<usize>,
    first_padding_offset: Option<usize>,
    padding_count: usize,
    expected_padding: Option<usize>,
    closed: bool,
}

impl Decoder {
    pub fn new(config: Config) -> Self {
        Self {
            config,
            position: 0,
            data_count: 0,
            bit_buffer: 0,
            bit_count: 0,
            last_data_offset: None,
            first_padding_offset: None,
            padding_count: 0,
            expected_padding: None,
            closed: false,
        }
    }

    fn expected_padding(residue: usize) -> Option<usize> {
        match residue {
            0 => Some(0),
            2 => Some(6),
            4 => Some(4),
            5 => Some(3),
            7 => Some(1),
            _ => None,
        }
    }

    fn classify(&self, byte: u8, offset: usize) -> Result<Option<u8>, StreamError> {
        if byte == b'=' {
            return Ok(None);
        }

        let upper = byte.to_ascii_uppercase();
        let alphabet = alphabet_bytes(self.config.alphabet);
        if let Some(index) = alphabet.iter().position(|&b| b == upper) {
            if self.config.canonical && byte != upper {
                return Err(StreamError::NonCanonical {
                    offset,
                    expected: upper,
                });
            }
            return Ok(Some(index as u8));
        }

        Err(StreamError::InvalidByte { offset, byte })
    }

    pub fn push(&mut self, input: &str) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let mut next = self.clone_state();
        let mut output = Vec::new();

        for &byte in input.as_bytes() {
            let offset = next.position;
            let symbol = next.classify(byte, offset)?;

            match symbol {
                None => {
                    if next.config.padding == Padding::Omitted {
                        return Err(StreamError::Padding { offset });
                    }

                    if next.first_padding_offset.is_none() {
                        let expected = Self::expected_padding(next.data_count % 8);
                        match expected {
                            None | Some(0) => {
                                return Err(StreamError::Padding { offset });
                            }
                            Some(count) => {
                                next.first_padding_offset = Some(offset);
                                next.expected_padding = Some(count);
                            }
                        }
                    }

                    next.padding_count += 1;
                    if next.padding_count > next.expected_padding.unwrap_or(0) {
                        return Err(StreamError::Padding { offset });
                    }
                }
                Some(value) => {
                    if next.first_padding_offset.is_some() {
                        return Err(StreamError::Padding { offset });
                    }

                    next.data_count += 1;
                    next.last_data_offset = Some(offset);
                    next.bit_buffer = (next.bit_buffer << 5) | value;
                    next.bit_count += 5;

                    while next.bit_count >= 8 {
                        next.bit_count -= 8;
                        output.push((next.bit_buffer >> next.bit_count) as u8);
                        next.bit_buffer &= (1u16.checked_shl(next.bit_count as u32).unwrap_or(0) - 1) as u8;
                    }
                }
            }
            next.position += 1;
        }

        self.copy_from(next);
        Ok(output)
    }

    pub fn finish(&mut self) -> Result<Vec<u8>, StreamError> {
        if self.closed {
            return Err(StreamError::Closed);
        }

        let residue = self.data_count % 8;
        let Some(expected) = Self::expected_padding(residue) else {
            return Err(StreamError::InvalidLength {
                offset: self.position,
            });
        };

        if self.config.padding == Padding::Required && self.padding_count != expected {
            return Err(StreamError::Padding {
                offset: self.first_padding_offset.unwrap_or(self.position),
            });
        }

        let unused = match residue {
            0 => 0,
            2 => 2,
            4 => 4,
            5 => 1,
            7 => 3,
            _ => 0,
        };
        if unused != 0 && self.bit_buffer & ((1u8 << unused) - 1) != 0 {
            return Err(StreamError::TrailingBits {
                offset: self.last_data_offset.unwrap_or(self.position),
            });
        }

        self.closed = true;
        Ok(Vec::new())
    }
}

impl Decoder {
    fn clone_state(&self) -> Self {
        Self {
            config: self.config,
            position: self.position,
            data_count: self.data_count,
            bit_buffer: self.bit_buffer,
            bit_count: self.bit_count,
            last_data_offset: self.last_data_offset,
            first_padding_offset: self.first_padding_offset,
            padding_count: self.padding_count,
            expected_padding: self.expected_padding,
            closed: self.closed,
        }
    }

    fn copy_from(&mut self, other: Self) {
        self.position = other.position;
        self.data_count = other.data_count;
        self.bit_buffer = other.bit_buffer;
        self.bit_count = other.bit_count;
        self.last_data_offset = other.last_data_offset;
        self.first_padding_offset = other.first_padding_offset;
        self.padding_count = other.padding_count;
        self.expected_padding = other.expected_padding;
        self.closed = other.closed;
    }
}
