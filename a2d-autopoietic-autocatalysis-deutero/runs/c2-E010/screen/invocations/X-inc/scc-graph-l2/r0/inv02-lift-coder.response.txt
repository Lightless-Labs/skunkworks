pub const MAX_VERTICES: usize = 4096;
pub const MAX_EDGES: usize = 32768;

#[derive(Clone, Debug, PartialEq, Eq)]
pub enum GraphError {
    TooManyVertices { count: usize, limit: usize },
    TooManyEdges { limit: usize },
    BadEndpoint { edge: usize, side: u8, vertex: usize },
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Arc {
    pub from: usize,
    pub to: usize,
    pub multiplicity: usize,
    pub first_edge: usize,
}

#[derive(Clone, Debug, PartialEq, Eq)]
pub struct Graph {
    pub components: Vec<Vec<usize>>,
    pub component_of: Vec<usize>,
    pub cyclic: Vec<bool>,
    pub internal_edges: Vec<usize>,
    pub arcs: Vec<Arc>,
}

pub fn analyze(n: usize, edges: &[(usize, usize)]) -> Result<Graph, GraphError> {
    if n > MAX_VERTICES {
        return Err(GraphError::TooManyVertices {
            count: n,
            limit: MAX_VERTICES,
        });
    }

    if edges.len() > MAX_EDGES {
        return Err(GraphError::TooManyEdges {
            limit: MAX_EDGES,
        });
    }

    for (edge, &(from, to)) in edges.iter().enumerate() {
        if from >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 0,
                vertex: from,
            });
        }
        if to >= n {
            return Err(GraphError::BadEndpoint {
                edge,
                side: 1,
                vertex: to,
            });
        }
    }

    let mut adjacency = vec![Vec::new(); n];
    let mut reverse = vec![Vec::new(); n];
    for &(from, to) in edges {
        adjacency[from].push(to);
        reverse[to].push(from);
    }

    // Iterative DFS computes finishing order without risking call-stack overflow.
    let mut seen = vec![false; n];
    let mut order = Vec::with_capacity(n);
    for start in 0..n {
        if seen[start] {
            continue;
        }

        seen[start] = true;
        let mut stack = vec![(start, 0usize)];
        while let Some((vertex, next_index)) = stack.pop() {
            if next_index < adjacency[vertex].len() {
                stack.push((vertex, next_index + 1));
                let next = adjacency[vertex][next_index];
                if !seen[next] {
                    seen[next] = true;
                    stack.push((next, 0));
                }
            } else {
                order.push(vertex);
            }
        }
    }

    let mut raw_components: Vec<Vec<usize>> = Vec::new();
    let mut assigned = vec![false; n];
    for &start in order.iter().rev() {
        if assigned[start] {
            continue;
        }

        assigned[start] = true;
        let mut members = Vec::new();
        let mut stack = vec![start];
        while let Some(vertex) = stack.pop() {
            members.push(vertex);
            for &next in &reverse[vertex] {
                if !assigned[next] {
                    assigned[next] = true;
                    stack.push(next);
                }
            }
        }
        members.sort_unstable();
        raw_components.push(members);
    }

    raw_components.sort_by(|a, b| {
        b.len()
            .cmp(&a.len())
            .then_with(|| a[0].cmp(&b[0]))
    });

    let components = raw_components;
    let mut component_of = vec![0usize; n];
    for (component_id, members) in components.iter().enumerate() {
        for &vertex in members {
            component_of[vertex] = component_id;
        }
    }

    let mut cyclic: Vec<bool> = components
        .iter()
        .map(|members| members.len() >= 2)
        .collect();
    let mut internal_edges = vec![0usize; components.len()];
    let mut arc_counts: Vec<(usize, usize, usize, usize)> = Vec::new();

    for (edge_index, &(from, to)) in edges.iter().enumerate() {
        let source_component = component_of[from];
        let target_component = component_of[to];

        if source_component == target_component {
            internal_edges[source_component] += 1;
            if from == to {
                cyclic[source_component] = true;
            }
        } else {
            arc_counts.push((source_component, target_component, 1, edge_index));
        }
    }

    arc_counts.sort_unstable_by(|a, b| {
        a.0.cmp(&b.0).then_with(|| b.1.cmp(&a.1))
    });

    let mut arcs = Vec::new();
    for (from, to, _, edge_index) in arc_counts {
        if let Some(last) = arcs.last_mut() {
            if last.from == from && last.to == to {
                last.multiplicity += 1;
                continue;
            }
        }
        arcs.push(Arc {
            from,
            to,
            multiplicity: 1,
            first_edge: edge_index,
        });
    }

    Ok(Graph {
        components,
        component_of,
        cyclic,
        internal_edges,
        arcs,
    })
}
