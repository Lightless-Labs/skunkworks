```rust
use std::collections::{BTreeMap, BTreeSet, HashMap};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Clone)]
enum Entry {
    Value { value: String, deadline: Option<u64> },
    Tombstone,
}

#[derive(Clone)]
struct Version {
    revision: u64,
    entry: Entry,
}

#[derive(Clone)]
struct Write {
    entry: Entry,
}

struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Write>,
}

pub struct Store {
    clock: u64,
    revision: u64,
    versions: BTreeMap<String, Vec<Version>>,
    transactions: HashMap<u64, Transaction>,
    snapshots: BTreeMap<String, BTreeMap<String, (String, Option<u64>)>>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            versions: BTreeMap::new(),
            transactions: HashMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn check_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn selected_entry(&self, key: &str, tx: Option<u64>) -> Option<Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(write) = transaction.writes.get(key) {
                return Some(write.entry.clone());
            }
        }

        let versions = self.versions.get(key)?;
        match tx {
            Some(id) => {
                let begin_revision = self.transactions.get(&id)?.begin_revision;
                versions
                    .iter()
                    .rev()
                    .find(|version| version.revision <= begin_revision)
                    .map(|version| version.entry.clone())
            }
            None => versions.last().map(|version| version.entry.clone()),
        }
    }

    fn live_value(entry: &Entry, now: u64) -> Option<&str> {
        match entry {
            Entry::Value { value, deadline } if deadline.is_none_or(|d| d > now) => {
                Some(value)
            }
            _ => None,
        }
    }

    fn visible(&self, key: &str, tx: Option<u64>, now: u64) -> Option<String> {
        let entry = self.selected_entry(key, tx)?;
        Self::live_value(&entry, now).map(str::to_owned)
    }

    fn visible_map(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, String> {
        let mut keys: BTreeSet<String> = self.versions.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                keys.extend(transaction.writes.keys().cloned());
            }
        }
        keys.into_iter()
            .filter_map(|key| {
                self.visible(&key, tx, now).map(|value| (key, value))
            })
            .collect()
    }

    fn append_version(&mut self, key: String, entry: Entry, revision: u64) {
        self.versions
            .entry(key)
            .or_default()
            .push(Version { revision, entry });
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit(now)?;
        let transaction = self
            .transactions
            .get(&id)
            .ok_or(Error::UnknownTransaction(id))?;

        if !transaction.writes.is_empty() {
            let begin_revision = transaction.begin_revision;
            for key in transaction.writes.keys() {
                if self
                    .versions
                    .get(key)
                    .and_then(|items| items.last())
                    .is_some_and(|version| version.revision > begin_revision)
                {
                    return Err(Error::Conflict {
                        tx: id,
                        key: key.clone(),
                    });
                }
            }

            self.revision += 1;
            let revision = self.revision;
            let writes = transaction
                .writes
                .iter()
                .map(|(key, write)| (key.clone(), write.entry.clone()))
                .collect::<Vec<_>>();
            for (key, entry) in writes {
                self.append_version(key, entry, revision);
            }
            self.transactions.remove(&id);
            Ok(revision)
        } else {
            self.transactions.remove(&id);
            Ok(self.revision)
        }
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit(now)?;
        if self.transactions.remove(&id).is_none() {
            return Err(Error::UnknownTransaction(id));
        }
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }

        let entry = match ttl {
            Some(0) => Entry::Tombstone,
            Some(duration) => Entry::Value {
                value: value.to_owned(),
                deadline: Some(now.checked_add(duration).ok_or(Error::DeadlineOverflow)?),
            },
            None => Entry::Value {
                value: value.to_owned(),
                deadline: None,
            },
        };

        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_owned(), Write { entry });
        } else {
            self.revision += 1;
            self.append_version(key.to_owned(), entry, self.revision);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let was_live = self.visible(key, tx, now).is_some();
        if let Some(id) = tx {
            self.transactions
                .get_mut(&id)
                .unwrap()
                .writes
                .insert(key.to_owned(), Write { entry: Entry::Tombstone });
        } else {
            self.revision += 1;
            self.append_version(key.to_owned(), Entry::Tombstone, self.revision);
        }
        Ok(was_live)
    }

    pub fn get(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
    ) -> Result<Option<String>, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        if key.is_empty() {
            return Err(Error::KeyEmpty);
        }
        Ok(self.visible(key, tx, now))
    }

    pub fn count(
        &mut self,
        now: u64,
        tx: Option<u64>,
        value: &str,
    ) -> Result<usize, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Ok(self
            .visible_map(tx, now)
            .values()
            .filter(|current| current.as_str() == value)
            .count())
    }

    fn validate_range(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|end| end <= start) {
            return Err(Error::InvalidRange);
        }
        Ok(())
    }

    fn matches(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.is_none_or(|end| key < end)
            && prefix.is_none_or(|prefix| key.starts_with(prefix))
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_range(start, end)?;

        if let Some(cursor) = cursor {
            if cursor.start != start
                || cursor.end.as_deref() != end
                || cursor.prefix.as_deref() != prefix
                || cursor.offset != offset
                || cursor.last.is_empty()
                || !Self::matches(&cursor.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
        }

        if limit == 0 {
            return Ok(Page {
                items: Vec::new(),
                next: None,
            });
        }

        let mut keys = self
            .visible_map(tx, now)
            .into_iter()
            .filter(|(key, _)| Self::matches(key, start, end, prefix))
            .filter(|(key, _)| cursor.is_none_or(|cursor| key > &cursor.last))
            .collect::<Vec<_>>();

        if cursor.is_none() {
            keys = keys.into_iter().skip(offset).collect();
        }

        let has_more = keys.len() > limit;
        let items = keys.into_iter().take(limit).collect::<Vec<_>>();
        let next = if has_more {
            items.last().map(|(last, _)| Cursor {
                start: start.to_owned(),
                end: end.map(str::to_owned),
                prefix: prefix.map(str::to_owned),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };
        Ok(Page { items, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit(now)?;
        self.check_tx(tx)?;
        Self::validate_range(start, end)?;

        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in self.visible_map(tx, now) {
            if !Self::matches(&key, start, end, prefix) {
                continue;
            }
            let parsed = Self::parse_canonical_i64(&value).ok_or_else(|| {
                Error::InvalidNumber {
                    key: key.clone(),
                    value: value.clone(),
                }
            })?;
            sum = sum.checked_add(parsed).ok_or_else(|| Error::SumOverflow {
                key: key.clone(),
            })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    fn parse_canonical_i64(value: &str) -> Option<i64> {
        if value == "0" {
            return Some(0);
        }

        let digits = value.strip_prefix('-').unwrap_or(value);
        if digits.is_empty()
            || digits.starts_with('0')
            || !digits.bytes().all(|b| b.is_ascii_digit())
        {
            return None;
        }
        value.parse::<i64>().ok()
    }

    fn valid_snapshot_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, byte) in name.bytes().enumerate() {
            let valid = if at == 0 {
                byte.is_ascii_lowercase()
            } else {
                byte.is_ascii_lowercase() || byte.is_ascii_digit() || byte == b'_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        Self::valid_snapshot_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }

        let saved = self
            .versions
            .iter()
            .filter_map(|(key, versions)| {
                let entry = &versions.last()?.entry;
                match entry {
                    Entry::Value { value, deadline } if deadline.is_none_or(|d| d > now) => {
                        Some((key.clone(), (value.clone(), *deadline)))
                    }
                    _ => None,
                }
            })
            .collect();
        self.snapshots.insert(name.to_owned(), saved);
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let saved = self
            .snapshots
            .get(name)
            .ok_or_else(|| Error::SnapshotMissing(name.to_owned()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let copied = saved
            .iter()
            .filter(|(_, (_, deadline))| deadline.is_none_or(|d| *d > now))
            .map(|(key, (value, deadline))| {
                (
                    key.clone(),
                    Entry::Value {
                        value: value.clone(),
                        deadline: *deadline,
                    },
                )
            })
            .collect::<BTreeMap<_, _>>();

        let mut union = BTreeSet::new();
        for (key, versions) in &self.versions {
            if let Some(version) = versions.last() {
                if Self::live_value(&version.entry, now).is_some() {
                    union.insert(key.clone());
                }
            }
        }
        union.extend(copied.keys().cloned());

        if union.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in union {
            let entry = copied.get(&key).cloned().unwrap_or(Entry::Tombstone);
            self.append_version(key, entry, revision);
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_owned()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit(now)?;
        let mut removed = 0;
        for versions in self.versions.values_mut() {
            if versions.is_empty() {
                continue;
            }
            let mut keep = BTreeSet::new();
            keep.insert(versions.len() - 1);
            for transaction in self.transactions.values() {
                if let Some(index) = versions
                    .iter()
                    .rposition(|version| version.revision <= transaction.begin_revision)
                {
                    keep.insert(index);
                }
            }
            let mut index = 0;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit(now)?;
        let versions = self.versions.values().map(Vec::len).sum();
        let tombstones = self
            .versions
            .values()
            .flat_map(|items| items.iter())
            .filter(|version| matches!(version.entry, Entry::Tombstone))
            .count();
        let live_keys = self
            .versions
            .iter()
            .filter(|(_, items)| {
                items.last().is_some_and(|version| {
                    Self::live_value(&version.entry, now).is_some()
                })
            })
            .count();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.versions.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions: self
                .transactions
                .values()
                .filter(|transaction| transaction.read_only)
                .count(),
            staged_writes: self
                .transactions
                .values()
                .map(|transaction| transaction.writes.len())
                .sum(),
            snapshots: self.snapshots.len(),
        })
    }
}
```
