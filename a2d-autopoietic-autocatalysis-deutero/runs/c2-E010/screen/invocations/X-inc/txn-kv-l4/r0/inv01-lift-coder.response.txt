use std::collections::{BTreeMap, BTreeSet, HashMap};

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    KeyEmpty,
    NameEmpty,
    NameInvalid { at: usize },
    SnapshotLimit,
    TimeBackwards { previous: u64, now: u64 },
    DeadlineOverflow,
    UnknownTransaction(u64),
    TransactionExists(u64),
    ReadOnly(u64),
    SnapshotMissing(String),
    ActiveTransactions,
    Conflict { tx: u64, key: String },
    InvalidRange,
    CursorMismatch,
    InvalidNumber { key: String, value: String },
    SumOverflow { key: String },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub clock: u64,
    pub revision: u64,
    pub keys: usize,
    pub live_keys: usize,
    pub versions: usize,
    pub tombstones: usize,
    pub active_transactions: usize,
    pub read_only_transactions: usize,
    pub staged_writes: usize,
    pub snapshots: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Cursor {
    pub start: String,
    pub end: Option<String>,
    pub prefix: Option<String>,
    pub offset: usize,
    pub last: String,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Page {
    pub items: Vec<(String, String)>,
    pub next: Option<Cursor>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Aggregate {
    pub count: usize,
    pub sum: i64,
}

#[derive(Debug, Clone)]
struct Entry {
    value: Option<String>,
    deadline: Option<u64>,
}

#[derive(Debug, Clone)]
struct Version {
    revision: u64,
    entry: Entry,
}

#[derive(Debug)]
struct Transaction {
    begin_revision: u64,
    read_only: bool,
    writes: BTreeMap<String, Entry>,
}

#[derive(Debug, Clone)]
struct Snapshot {
    entries: BTreeMap<String, Entry>,
}

#[derive(Debug)]
pub struct Store {
    clock: u64,
    revision: u64,
    data: BTreeMap<String, Vec<Version>>,
    transactions: HashMap<u64, Transaction>,
    snapshots: BTreeMap<String, Snapshot>,
}

impl Store {
    pub fn new() -> Self {
        Self {
            clock: 0,
            revision: 0,
            data: BTreeMap::new(),
            transactions: HashMap::new(),
            snapshots: BTreeMap::new(),
        }
    }

    fn admit_time(&mut self, now: u64) -> Result<(), Error> {
        if now < self.clock {
            return Err(Error::TimeBackwards {
                previous: self.clock,
                now,
            });
        }
        self.clock = now;
        Ok(())
    }

    fn check_tx(&self, tx: Option<u64>) -> Result<(), Error> {
        if let Some(id) = tx {
            if !self.transactions.contains_key(&id) {
                return Err(Error::UnknownTransaction(id));
            }
        }
        Ok(())
    }

    fn selected_entry(&self, key: &str, tx: Option<u64>) -> Option<Entry> {
        if let Some(id) = tx {
            let transaction = self.transactions.get(&id)?;
            if let Some(entry) = transaction.writes.get(key) {
                return Some(entry.clone());
            }
            self.data.get(key)?.iter()
                .rev()
                .find(|version| version.revision <= transaction.begin_revision)
                .map(|version| version.entry.clone())
        } else {
            self.data.get(key)?.last().map(|version| version.entry.clone())
        }
    }

    fn live_entry(entry: Option<Entry>, now: u64) -> Option<Entry> {
        entry.filter(|e| e.value.is_some() && e.deadline.map_or(true, |d| d > now))
    }

    fn visible(&self, tx: Option<u64>, now: u64) -> BTreeMap<String, String> {
        let mut keys: BTreeSet<String> = self.data.keys().cloned().collect();
        if let Some(id) = tx {
            if let Some(transaction) = self.transactions.get(&id) {
                keys.extend(transaction.writes.keys().cloned());
            }
        }
        let mut result = BTreeMap::new();
        for key in keys {
            if let Some(entry) = Self::live_entry(self.selected_entry(&key, tx), now) {
                if let Some(value) = entry.value {
                    result.insert(key, value);
                }
            }
        }
        result
    }

    fn append(&mut self, key: String, entry: Entry, revision: u64) {
        self.data.entry(key).or_default().push(Version { revision, entry });
    }

    fn key_valid(key: &str) -> Result<(), Error> {
        if key.is_empty() {
            Err(Error::KeyEmpty)
        } else {
            Ok(())
        }
    }

    fn validate_name(name: &str) -> Result<(), Error> {
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        for (at, ch) in name.char_indices() {
            let valid = if at == 0 {
                ch.is_ascii_lowercase()
            } else {
                ch.is_ascii_lowercase() || ch.is_ascii_digit() || ch == '_'
            };
            if !valid {
                return Err(Error::NameInvalid { at });
            }
        }
        Ok(())
    }

    fn validate_range(start: &str, end: Option<&str>) -> Result<(), Error> {
        if end.is_some_and(|e| e <= start) {
            Err(Error::InvalidRange)
        } else {
            Ok(())
        }
    }

    fn matches_range(key: &str, start: &str, end: Option<&str>, prefix: Option<&str>) -> bool {
        key >= start
            && end.map_or(true, |e| key < e)
            && prefix.map_or(true, |p| key.starts_with(p))
    }

    pub fn begin(&mut self, now: u64, id: u64, read_only: bool) -> Result<(), Error> {
        self.admit_time(now)?;
        if self.transactions.contains_key(&id) {
            return Err(Error::TransactionExists(id));
        }
        self.transactions.insert(
            id,
            Transaction {
                begin_revision: self.revision,
                read_only,
                writes: BTreeMap::new(),
            },
        );
        Ok(())
    }

    pub fn commit(&mut self, now: u64, id: u64) -> Result<u64, Error> {
        self.admit_time(now)?;
        self.check_tx(Some(id))?;

        let transaction = self.transactions.get(&id).unwrap();
        let begin_revision = transaction.begin_revision;
        if !transaction.writes.is_empty() {
            for key in transaction.writes.keys() {
                if self.data.get(key)
                    .and_then(|versions| versions.last())
                    .is_some_and(|version| version.revision > begin_revision)
                {
                    return Err(Error::Conflict { tx: id, key: key.clone() });
                }
            }
        }

        let transaction = self.transactions.remove(&id).unwrap();
        if transaction.writes.is_empty() {
            return Ok(self.revision);
        }
        self.revision += 1;
        let revision = self.revision;
        for (key, entry) in transaction.writes {
            self.append(key, entry, revision);
        }
        Ok(revision)
    }

    pub fn rollback(&mut self, now: u64, id: u64) -> Result<(), Error> {
        self.admit_time(now)?;
        self.check_tx(Some(id))?;
        self.transactions.remove(&id);
        Ok(())
    }

    pub fn set(
        &mut self,
        now: u64,
        tx: Option<u64>,
        key: &str,
        value: &str,
        ttl: Option<u64>,
    ) -> Result<(), Error> {
        self.admit_time(now)?;
        self.check_tx(tx)?;
        Self::key_valid(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let deadline = match ttl {
            Some(d) => Some(now.checked_add(d).ok_or(Error::DeadlineOverflow)?),
            None => None,
        };
        let entry = if ttl == Some(0) {
            Entry { value: None, deadline: None }
        } else {
            Entry { value: Some(value.to_owned()), deadline }
        };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_owned(), entry);
        } else {
            self.revision += 1;
            self.append(key.to_owned(), entry, self.revision);
        }
        Ok(())
    }

    pub fn delete(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<bool, Error> {
        self.admit_time(now)?;
        self.check_tx(tx)?;
        Self::key_valid(key)?;
        if let Some(id) = tx {
            if self.transactions.get(&id).unwrap().read_only {
                return Err(Error::ReadOnly(id));
            }
        }
        let exists = Self::live_entry(self.selected_entry(key, tx), now).is_some();
        let tombstone = Entry { value: None, deadline: None };
        if let Some(id) = tx {
            self.transactions.get_mut(&id).unwrap().writes.insert(key.to_owned(), tombstone);
        } else {
            self.revision += 1;
            self.append(key.to_owned(), tombstone, self.revision);
        }
        Ok(exists)
    }

    pub fn get(&mut self, now: u64, tx: Option<u64>, key: &str) -> Result<Option<String>, Error> {
        self.admit_time(now)?;
        self.check_tx(tx)?;
        Self::key_valid(key)?;
        Ok(Self::live_entry(self.selected_entry(key, tx), now).and_then(|e| e.value))
    }

    pub fn count(&mut self, now: u64, tx: Option<u64>, value: &str) -> Result<usize, Error> {
        self.admit_time(now)?;
        self.check_tx(tx)?;
        Ok(self.visible(tx, now).values().filter(|v| v.as_str() == value).count())
    }

    pub fn scan(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
        offset: usize,
        limit: usize,
        cursor: Option<&Cursor>,
    ) -> Result<Page, Error> {
        self.admit_time(now)?;
        self.check_tx(tx)?;
        Self::validate_range(start, end)?;

        let after = if let Some(c) = cursor {
            if c.start != start
                || c.end.as_deref() != end
                || c.prefix.as_deref() != prefix
                || c.offset != offset
                || c.last.is_empty()
                || !Self::matches_range(&c.last, start, end, prefix)
            {
                return Err(Error::CursorMismatch);
            }
            Some(c.last.as_str())
        } else {
            None
        };

        if limit == 0 {
            return Ok(Page { items: Vec::new(), next: None });
        }

        let visible = self.visible(tx, now);
        let mut matching: Vec<(String, String)> = visible
            .into_iter()
            .filter(|(key, _)| Self::matches_range(key, start, end, prefix))
            .filter(|(key, _)| after.map_or(true, |last| key.as_str() > last))
            .collect();

        if after.is_none() {
            matching = matching.into_iter().skip(offset).collect();
        }
        let has_more = matching.len() > limit;
        matching.truncate(limit);
        let next = if has_more {
            matching.last().map(|(last, _)| Cursor {
                start: start.to_owned(),
                end: end.map(str::to_owned),
                prefix: prefix.map(str::to_owned),
                offset,
                last: last.clone(),
            })
        } else {
            None
        };
        Ok(Page { items: matching, next })
    }

    pub fn aggregate(
        &mut self,
        now: u64,
        tx: Option<u64>,
        start: &str,
        end: Option<&str>,
        prefix: Option<&str>,
    ) -> Result<Aggregate, Error> {
        self.admit_time(now)?;
        self.check_tx(tx)?;
        Self::validate_range(start, end)?;

        let visible = self.visible(tx, now);
        let mut count = 0usize;
        let mut sum = 0i64;
        for (key, value) in visible {
            if !Self::matches_range(&key, start, end, prefix) {
                continue;
            }
            let parsed = parse_canonical_i64(&value).ok_or_else(|| Error::InvalidNumber {
                key: key.clone(),
                value: value.clone(),
            })?;
            sum = sum.checked_add(parsed).ok_or_else(|| Error::SumOverflow {
                key: key.clone(),
            })?;
            count += 1;
        }
        Ok(Aggregate { count, sum })
    }

    pub fn snapshot(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit_time(now)?;
        Self::validate_name(name)?;
        if !self.snapshots.contains_key(name) && self.snapshots.len() >= 3 {
            return Err(Error::SnapshotLimit);
        }
        let mut entries = BTreeMap::new();
        for (key, versions) in &self.data {
            if let Some(version) = versions.last() {
                if let Some(entry) = Self::live_entry(Some(version.entry.clone()), now) {
                    entries.insert(key.clone(), entry);
                }
            }
        }
        self.snapshots.insert(name.to_owned(), Snapshot { entries });
        Ok(())
    }

    pub fn restore(&mut self, now: u64, name: &str) -> Result<u64, Error> {
        self.admit_time(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        let snapshot = self.snapshots.get(name)
            .ok_or_else(|| Error::SnapshotMissing(name.to_owned()))?;
        if !self.transactions.is_empty() {
            return Err(Error::ActiveTransactions);
        }

        let copied: BTreeMap<String, Entry> = snapshot.entries.iter()
            .filter_map(|(key, entry)| {
                Self::live_entry(Some(entry.clone()), now).map(|e| (key.clone(), e))
            })
            .collect();
        let current = self.visible(None, now);
        let keys: BTreeSet<String> = current.keys().chain(copied.keys()).cloned().collect();
        if keys.is_empty() {
            return Ok(self.revision);
        }

        self.revision += 1;
        let revision = self.revision;
        for key in keys {
            let entry = copied.get(&key).cloned()
                .unwrap_or(Entry { value: None, deadline: None });
            self.append(key, entry, revision);
        }
        Ok(revision)
    }

    pub fn forget(&mut self, now: u64, name: &str) -> Result<(), Error> {
        self.admit_time(now)?;
        if name.is_empty() {
            return Err(Error::NameEmpty);
        }
        if self.snapshots.remove(name).is_none() {
            return Err(Error::SnapshotMissing(name.to_owned()));
        }
        Ok(())
    }

    pub fn gc(&mut self, now: u64) -> Result<usize, Error> {
        self.admit_time(now)?;
        let mut removed = 0usize;
        for versions in self.data.values_mut() {
            let mut keep = BTreeSet::new();
            if !versions.is_empty() {
                keep.insert(versions.len() - 1);
            }
            for transaction in self.transactions.values() {
                if let Some(index) = versions.iter().rposition(|v| v.revision <= transaction.begin_revision) {
                    keep.insert(index);
                }
            }
            let mut index = 0usize;
            versions.retain(|_| {
                let retain = keep.contains(&index);
                index += 1;
                if !retain {
                    removed += 1;
                }
                retain
            });
        }
        Ok(removed)
    }

    pub fn stats(&mut self, now: u64) -> Result<Stats, Error> {
        self.admit_time(now)?;
        let versions = self.data.values().map(Vec::len).sum();
        let tombstones = self.data.values()
            .flat_map(|v| v.iter())
            .filter(|v| v.entry.value.is_none())
            .count();
        let live_keys = self.visible(None, now).len();
        let read_only_transactions = self.transactions.values().filter(|t| t.read_only).count();
        let staged_writes = self.transactions.values().map(|t| t.writes.len()).sum();
        Ok(Stats {
            clock: self.clock,
            revision: self.revision,
            keys: self.data.len(),
            live_keys,
            versions,
            tombstones,
            active_transactions: self.transactions.len(),
            read_only_transactions,
            staged_writes,
            snapshots: self.snapshots.len(),
        })
    }
}

fn parse_canonical_i64(value: &str) -> Option<i64> {
    if value.is_empty() {
        return None;
    }
    let digits = value.strip_prefix('-').unwrap_or(value);
    if digits.is_empty()
        || !digits.bytes().all(|b| b.is_ascii_digit())
        || (digits.len() > 1 && digits.starts_with('0'))
        || value == "-0"
    {
        return None;
    }
    value.parse::<i64>().ok()
}
