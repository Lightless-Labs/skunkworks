use std::cmp::Ordering;

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow {
        customer: u32,
        time: u64,
        stage: OverflowStage,
    },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };

    Some(if round_up { quotient + 1 } else { quotient })
}

#[derive(Clone)]
struct Job {
    customer: Customer,
    deadline: Option<u64>,
    entry: u64,
    waited: u64,
    remaining: u64,
    paused_at: Option<u64>,
    ever_started: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    start: u64,
    work_at: u64,
    finish: u64,
    protected: bool,
}

fn waiter_order(a: &Job, b: &Job) -> Ordering {
    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.paused_at, b.paused_at) {
            (Some(x), Some(y)) => x
                .cmp(&y)
                .then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

fn best_waiter(queue: &[usize], jobs: &[Job]) -> Option<usize> {
    queue
        .iter()
        .copied()
        .min_by(|&a, &b| waiter_order(&jobs[a], &jobs[b]))
}

fn expiration_order(a: &Job, b: &Job) -> Ordering {
    a.deadline
        .cmp(&b.deadline)
        .then_with(|| a.customer.id.cmp(&b.customer.id))
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut seen_ids = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if seen_ids.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        seen_ids.push(customer.id);

        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        if let Some(patience) = customer.patience {
            if customer.arrival.checked_add(patience).is_none() {
                return Err(Error::DeadlineOverflow {
                    index,
                    id: customer.id,
                });
            }
        }
    }

    let mut arrival_order: Vec<usize> = (0..customers.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut jobs: Vec<Job> = customers
        .iter()
        .cloned()
        .map(|customer| Job {
            deadline: customer
                .patience
                .map(|patience| customer.arrival + patience),
            entry: customer.arrival,
            waited: 0,
            remaining: customer.service,
            paused_at: None,
            ever_started: false,
            customer,
        })
        .collect();

    let mut events = Vec::new();
    let mut queue: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0;
    let mut previous_dispatch = None;

    let mut accepted = 0usize;
    let mut balked = 0usize;
    let mut reneged = 0usize;
    let mut completed = 0usize;
    let mut started = 0usize;
    let mut preemptions = 0usize;

    let mut busy_ticks = 0u128;
    let mut wait_ticks = 0u128;
    let mut turnaround_ticks = 0u128;
    let mut completion_order = Vec::new();

    if customers.is_empty() {
        return Ok(Report {
            events,
            stats: Stats {
                accepted: 0,
                balked: 0,
                reneged: 0,
                completed: 0,
                started: 0,
                preemptions: 0,
                elapsed: 0,
                busy_ticks: 0,
                wait_ticks: 0,
                turnaround_ticks: 0,
                mean_wait_milli: 0,
                mean_turnaround_milli: 0,
                utilization_bps: 0,
                completion_order,
            },
        });
    }

    let mut now = customers[arrival_order[0]].arrival;

    while completed + reneged + balked < customers.len() {
        // Phase A: finish the active service before this tick's other work.
        if active.as_ref().is_some_and(|a| a.finish == now) {
            let finished = active.take().unwrap();
            let job = &jobs[finished.index];
            events.push(Event {
                time: now,
                customer: job.customer.id,
                kind: Kind::Finish,
            });
            completed += 1;
            turnaround_ticks += (now - job.customer.arrival) as u128;
            completion_order.push(job.customer.id);
        }

        // Phase B: expire customers that were already waiting.
        let mut expired: Vec<usize> = queue
            .iter()
            .copied()
            .filter(|&index| jobs[index].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| expiration_order(&jobs[a], &jobs[b]));
        for index in expired {
            queue.retain(|&item| item != index);
            let segment = now - jobs[index].entry;
            jobs[index].waited += segment;
            wait_ticks += segment as u128;
            events.push(Event {
                time: now,
                customer: jobs[index].customer.id,
                kind: Kind::Renege {
                    waited: jobs[index].waited,
                },
            });
            reneged += 1;
        }

        // Phase C: process every arrival at this tick without dispatching.
        while next_arrival < arrival_order.len()
            && customers[arrival_order[next_arrival]].arrival == now
        {
            let index = arrival_order[next_arrival];
            next_arrival += 1;

            events.push(Event {
                time: now,
                customer: jobs[index].customer.id,
                kind: Kind::Arrive,
            });

            let limit = config.waiting_capacity + usize::from(active.is_none());
            if queue.len() >= limit {
                events.push(Event {
                    time: now,
                    customer: jobs[index].customer.id,
                    kind: Kind::Balk,
                });
                balked += 1;
            } else {
                accepted += 1;
                queue.push(index);
            }
        }

        // Phase D: reconsider the running customer once protection has ended.
        if let Some(current) = active.clone() {
            if now >= current.work_at {
                if let Some(ref mut running) = active {
                    running.protected = false;
                }
            }

            if !current.protected && now >= current.work_at {
                if let Some(waiter) = best_waiter(&queue, &jobs) {
                    if jobs[waiter].customer.priority < jobs[current.index].customer.priority {
                        let index = current.index;
                        let remaining = current.finish - now;
                        let paused_id = jobs[index].customer.id;

                        events.push(Event {
                            time: now,
                            customer: paused_id,
                            kind: Kind::Preempt {
                                remaining,
                                by: jobs[waiter].customer.id,
                            },
                        });
                        preemptions += 1;
                        busy_ticks += (now - current.start) as u128;
                        active = None;

                        jobs[index].remaining = remaining;
                        jobs[index].entry = now;
                        jobs[index].paused_at = Some(now);

                        jobs[index].deadline = match jobs[index].customer.patience {
                            Some(budget) => {
                                let remaining_budget = budget - jobs[index].waited;
                                Some(now.checked_add(remaining_budget).ok_or(
                                    Error::TimeOverflow {
                                        customer: paused_id,
                                        time: now,
                                        stage: OverflowStage::Patience,
                                    },
                                )?)
                            }
                            None => None,
                        };
                        queue.push(index);
                    }
                }
            }
        }

        // Phase E: dispatch until one customer is active or the queue is empty.
        loop {
            if active.is_some() {
                break;
            }
            let Some(index) = best_waiter(&queue, &jobs) else {
                break;
            };
            queue.retain(|&item| item != index);

            let segment = now - jobs[index].entry;
            jobs[index].waited += segment;
            wait_ticks += segment as u128;

            let switch_cost = if previous_dispatch
                .is_some_and(|id| id != jobs[index].customer.id)
            {
                config.switch_cost
            } else {
                0
            };

            let work_at = now.checked_add(switch_cost).ok_or(Error::TimeOverflow {
                customer: jobs[index].customer.id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(jobs[index].remaining).ok_or(
                Error::TimeOverflow {
                    customer: jobs[index].customer.id,
                    time: now,
                    stage: OverflowStage::Finish,
                },
            )?;

            let kind = if jobs[index].ever_started {
                Kind::Resume {
                    remaining: jobs[index].remaining,
                    work_at,
                    finish,
                }
            } else {
                started += 1;
                Kind::Start {
                    remaining: jobs[index].remaining,
                    work_at,
                    finish,
                }
            };

            jobs[index].ever_started = true;
            previous_dispatch = Some(jobs[index].customer.id);
            events.push(Event {
                time: now,
                customer: jobs[index].customer.id,
                kind,
            });

            if finish == now {
                events.push(Event {
                    time: now,
                    customer: jobs[index].customer.id,
                    kind: Kind::Finish,
                });
                completed += 1;
                turnaround_ticks += (now - jobs[index].customer.arrival) as u128;
                completion_order.push(jobs[index].customer.id);
                continue;
            }

            active = Some(Active {
                index,
                start: now,
                work_at,
                finish,
                protected: now < work_at,
            });
        }

        // Phase F: expire newly queued customers after dispatches.
        let mut expired: Vec<usize> = queue
            .iter()
            .copied()
            .filter(|&index| jobs[index].deadline.is_some_and(|d| d <= now))
            .collect();
        expired.sort_by(|&a, &b| expiration_order(&jobs[a], &jobs[b]));
        for index in expired {
            queue.retain(|&item| item != index);
            let segment = now - jobs[index].entry;
            jobs[index].waited += segment;
            wait_ticks += segment as u128;
            events.push(Event {
                time: now,
                customer: jobs[index].customer.id,
                kind: Kind::Renege {
                    waited: jobs[index].waited,
                },
            });
            reneged += 1;
        }

        if completed + reneged + balked == customers.len() {
            break;
        }

        let mut next_time: Option<u64> = None;
        let mut consider = |time: u64| {
            if time > now {
                next_time = Some(next_time.map_or(time, |old| old.min(time)));
            }
        };

        if next_arrival < arrival_order.len() {
            consider(customers[arrival_order[next_arrival]].arrival);
        }
        if let Some(running) = &active {
            consider(running.finish);
            if running.work_at > now {
                consider(running.work_at);
            }
        }
        for &index in &queue {
            if let Some(deadline) = jobs[index].deadline {
                consider(deadline);
            }
        }

        now = next_time.expect("pending work must have a future event");
    }

    let elapsed = events.last().map_or(0, |event| event.time);
    let mean_wait_milli = if accepted == 0 {
        0
    } else {
        rounded_ratio(wait_ticks * 1000, accepted as u128).unwrap()
    };
    let mean_turnaround_milli = if completed == 0 {
        0
    } else {
        rounded_ratio(turnaround_ticks * 1000, completed as u128).unwrap()
    };
    let utilization_bps = if elapsed == 0 {
        0
    } else {
        rounded_ratio(busy_ticks * 10_000, elapsed as u128).unwrap() as u32
    };

    Ok(Report {
        events,
        stats: Stats {
            accepted,
            balked,
            reneged,
            completed,
            started,
            preemptions,
            elapsed,
            busy_ticks,
            wait_ticks,
            turnaround_ticks,
            mean_wait_milli,
            mean_turnaround_milli,
            utilization_bps,
            completion_order,
        },
    })
}
