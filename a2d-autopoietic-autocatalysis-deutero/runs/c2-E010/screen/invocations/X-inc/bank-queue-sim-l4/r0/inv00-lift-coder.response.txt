#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };
    Some(quotient + u128::from(round_up))
}

#[derive(Clone)]
struct State {
    customer: Customer,
    accepted: bool,
    started: bool,
    terminal: bool,
    remaining: u64,
    queue_entry: u64,
    deadline: Option<u64>,
    waited: u64,
    pause_time: Option<u64>,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    protected: bool,
}

fn push(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event { time, customer: id, kind });
}

fn best_waiter(queue: &[usize], states: &[State]) -> Option<usize> {
    queue.iter().copied().min_by(|&a, &b| {
        let x = &states[a];
        let y = &states[b];
        x.customer
            .priority
            .cmp(&y.customer.priority)
            .then_with(|| match (x.pause_time, y.pause_time) {
                (Some(px), Some(py)) => px.cmp(&py).then(x.customer.id.cmp(&y.customer.id)),
                (Some(_), None) => std::cmp::Ordering::Less,
                (None, Some(_)) => std::cmp::Ordering::Greater,
                (None, None) => x
                    .customer
                    .arrival
                    .cmp(&y.customer.arrival)
                    .then(y.customer.id.cmp(&x.customer.id)),
            })
    })
}

fn consume_wait(state: &mut State, now: u64) -> u64 {
    let segment = now - state.queue_entry;
    state.waited += segment;
    segment
}

fn expire_waiters(
    queue: &mut Vec<usize>,
    states: &mut [State],
    events: &mut Vec<Event>,
    now: u64,
    stats: &mut Stats,
) {
    let mut expired: Vec<(u64, u32, usize)> = queue
        .iter()
        .filter_map(|&i| {
            states[i]
                .deadline
                .filter(|&d| d <= now)
                .map(|d| (d, states[i].customer.id, i))
        })
        .collect();
    expired.sort_by_key(|&(deadline, id, _)| (deadline, id));

    for (_, _, index) in expired {
        if let Some(pos) = queue.iter().position(|&i| i == index) {
            queue.remove(pos);
            let state = &mut states[index];
            consume_wait(state, now);
            state.terminal = true;
            stats.reneged += 1;
            push(
                events,
                now,
                state.customer.id,
                Kind::Renege { waited: state.waited },
            );
        }
    }
}

fn checked_time(customer: u32, time: u64, value: u64, stage: OverflowStage) -> Result<u64, Error> {
    time.checked_add(value).ok_or(Error::TimeOverflow { customer, time, stage })
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge { capacity: config.waiting_capacity });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers { count: customers.len() });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = Vec::with_capacity(customers.len());
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&customer.id) {
            return Err(Error::DuplicateId { index, id: customer.id });
        }
        ids.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority { index, value: customer.priority });
        }
        let deadline = match customer.patience {
            Some(patience) => Some(
                customer
                    .arrival
                    .checked_add(patience)
                    .ok_or(Error::DeadlineOverflow { index, id: customer.id })?,
            ),
            None => None,
        };
        initial_deadlines.push(deadline);
    }

    let mut states: Vec<State> = customers
        .iter()
        .enumerate()
        .map(|(i, c)| State {
            customer: c.clone(),
            accepted: false,
            started: false,
            terminal: false,
            remaining: c.service,
            queue_entry: c.arrival,
            deadline: initial_deadlines[i],
            waited: 0,
            pause_time: None,
        })
        .collect();

    let mut arrival_order: Vec<usize> = (0..customers.len()).collect();
    arrival_order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then(customers[a].priority.cmp(&customers[b].priority))
            .then(customers[b].id.cmp(&customers[a].id))
    });

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut events = Vec::new();
    let mut queue: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut next_arrival = 0usize;
    let mut now = customers
        .iter()
        .map(|c| c.arrival)
        .min()
        .unwrap_or(0);
    let mut last_dispatched: Option<u32> = None;

    loop {
        if let Some(a) = active.as_ref() {
            if a.finish == now {
                let index = a.index;
                active = None;
                let state = &mut states[index];
                state.terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - state.customer.arrival) as u128;
                stats.completion_order.push(state.customer.id);
                push(&mut events, now, state.customer.id, Kind::Finish);
            } else if a.protected && a.work_at == now {
                if let Some(a) = active.as_mut() {
                    a.protected = false;
                }
            }
        }

        expire_waiters(&mut queue, &mut states, &mut events, now, &mut stats);

        while next_arrival < arrival_order.len()
            && customers[arrival_order[next_arrival]].arrival == now
        {
            let index = arrival_order[next_arrival];
            next_arrival += 1;
            let id = states[index].customer.id;
            push(&mut events, now, id, Kind::Arrive);

            let limit = config.waiting_capacity + usize::from(active.is_none());
            if queue.len() < limit {
                states[index].accepted = true;
                states[index].queue_entry = now;
                queue.push(index);
                stats.accepted += 1;
            } else {
                stats.balked += 1;
                states[index].terminal = true;
                push(&mut events, now, id, Kind::Balk);
            }
        }

        if let Some(a) = active.as_ref() {
            if !a.protected {
                if let Some(waiter) = best_waiter(&queue, &states) {
                    if states[waiter].customer.priority < states[a.index].customer.priority {
                        let paused_index = a.index;
                        let remaining = a.finish - now;
                        let by = states[waiter].customer.id;
                        push(
                            &mut events,
                            now,
                            states[paused_index].customer.id,
                            Kind::Preempt { remaining, by },
                        );
                        stats.preemptions += 1;
                        stats.busy_ticks += (now - states[paused_index].queue_entry.min(now)) as u128;
                        active = None;

                        let paused = &mut states[paused_index];
                        paused.remaining = remaining;
                        paused.queue_entry = now;
                        paused.pause_time = Some(now);
                        paused.deadline = match paused.customer.patience {
                            Some(_) => Some(checked_time(
                                paused.customer.id,
                                now,
                                paused.remaining.min(0),
                                OverflowStage::Patience,
                            )?),
                            None => None,
                        };
                        queue.push(paused_index);
                    }
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let Some(index) = best_waiter(&queue, &states) else {
                break;
            };
            let pos = queue.iter().position(|&i| i == index).unwrap();
            queue.remove(pos);

            let state = &mut states[index];
            let waited_segment = consume_wait(state, now);
            stats.wait_ticks += waited_segment as u128;
            state.deadline = match state.customer.patience {
                Some(budget) => Some(now.saturating_add(budget.saturating_sub(state.waited))),
                None => None,
            };

            let switch = if last_dispatched.is_some_and(|id| id != state.customer.id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = checked_time(state.customer.id, now, switch, OverflowStage::WorkAt)?;
            let finish = checked_time(state.customer.id, work_at, state.remaining, OverflowStage::Finish)?;
            let was_started = state.started;
            let remaining = state.remaining;
            let id = state.customer.id;
            let kind = if was_started {
                Kind::Resume { remaining, work_at, finish }
            } else {
                state.started = true;
                stats.started += 1;
                Kind::Start { remaining, work_at, finish }
            };
            push(&mut events, now, id, kind);
            last_dispatched = Some(id);
            active = Some(Active {
                index,
                work_at,
                finish,
                protected: work_at > now,
            });

            if finish == now {
                active = None;
                let state = &mut states[index];
                state.terminal = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - state.customer.arrival) as u128;
                stats.completion_order.push(id);
                push(&mut events, now, id, Kind::Finish);
                continue;
            }
        }

        expire_waiters(&mut queue, &mut states, &mut events, now, &mut stats);

        if next_arrival == arrival_order.len()
            && active.is_none()
            && queue.is_empty()
        {
            break;
        }

        let mut next_time: Option<u64> = None;
        if next_arrival < arrival_order.len() {
            next_time = Some(customers[arrival_order[next_arrival]].arrival);
        }
        if let Some(a) = &active {
            next_time = Some(next_time.map_or(a.finish, |t| t.min(a.finish)));
            if a.protected && a.work_at > now {
                next_time = Some(next_time.map_or(a.work_at, |t| t.min(a.work_at)));
            }
        }
        for &index in &queue {
            if let Some(deadline) = states[index].deadline {
                if deadline > now {
                    next_time = Some(next_time.map_or(deadline, |t| t.min(deadline)));
                }
            }
        }
        let Some(next) = next_time else {
            break;
        };
        now = next;
    }

    stats.elapsed = events.last().map(|e| e.time).unwrap_or(0);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
