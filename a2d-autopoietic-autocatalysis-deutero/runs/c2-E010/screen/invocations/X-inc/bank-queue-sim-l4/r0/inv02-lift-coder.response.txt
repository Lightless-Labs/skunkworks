#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }
    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;
    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };
    Some(quotient + u128::from(round_up))
}

#[derive(Clone)]
struct Waiting {
    index: usize,
    entry: u64,
    deadline: Option<u64>,
    remaining_budget: Option<u64>,
    waited: u64,
    paused: bool,
    pause_time: u64,
}

#[derive(Clone)]
struct Running {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiting_cmp(a: &Waiting, b: &Waiting, customers: &[Customer]) -> std::cmp::Ordering {
    let ca = &customers[a.index];
    let cb = &customers[b.index];

    ca.priority
        .cmp(&cb.priority)
        .then_with(|| match (a.paused, b.paused) {
            (true, false) => std::cmp::Ordering::Less,
            (false, true) => std::cmp::Ordering::Greater,
            (true, true) => a
                .pause_time
                .cmp(&b.pause_time)
                .then_with(|| ca.id.cmp(&cb.id)),
            (false, false) => ca
                .arrival
                .cmp(&cb.arrival)
                .then_with(|| cb.id.cmp(&ca.id)),
        })
}

fn best_waiting(queue: &[Waiting], customers: &[Customer]) -> Option<usize> {
    queue
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| waiting_cmp(a, b, customers))
        .map(|(i, _)| i)
}

fn push_event(events: &mut Vec<Event>, time: u64, customer: u32, kind: Kind) {
    events.push(Event { time, customer, kind });
}

fn add_wait_segment(waiter: &mut Waiting, now: u64) -> u64 {
    let segment = now - waiter.entry;
    waiter.waited += segment;
    waiter.remaining_budget = waiter.remaining_budget.map(|budget| budget - segment);
    segment
}

fn expire_waiters(
    queue: &mut Vec<Waiting>,
    now: u64,
    customers: &[Customer],
    events: &mut Vec<Event>,
    stats: &mut Stats,
) {
    let mut expired: Vec<Waiting> = Vec::new();
    let mut kept = Vec::with_capacity(queue.len());

    for waiter in queue.drain(..) {
        if waiter.deadline.is_some_and(|deadline| deadline <= now) {
            expired.push(waiter);
        } else {
            kept.push(waiter);
        }
    }
    *queue = kept;

    expired.sort_by(|a, b| {
        a.deadline
            .unwrap()
            .cmp(&b.deadline.unwrap())
            .then_with(|| customers[a.index].id.cmp(&customers[b.index].id))
    });

    for mut waiter in expired {
        add_wait_segment(&mut waiter, now);
        let customer = &customers[waiter.index];
        push_event(
            events,
            now,
            customer.id,
            Kind::Renege {
                waited: waiter.waited,
            },
        );
        stats.reneged += 1;
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    let mut initial_deadlines = vec![None; customers.len()];
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        ids.push(customer.id);
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
        if let Some(patience) = customer.patience {
            initial_deadlines[index] = Some(customer.arrival.checked_add(patience).ok_or(
                Error::DeadlineOverflow {
                    index,
                    id: customer.id,
                },
            )?);
        }
    }

    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };

    let mut events = Vec::new();
    let mut order: Vec<usize> = (0..customers.len()).collect();
    order.sort_by(|&a, &b| {
        customers[a]
            .arrival
            .cmp(&customers[b].arrival)
            .then_with(|| customers[a].priority.cmp(&customers[b].priority))
            .then_with(|| customers[b].id.cmp(&customers[a].id))
    });

    let mut next_arrival = 0usize;
    let mut queue: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut current_time = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut ever_started = vec![false; customers.len()];

    loop {
        let next_external_arrival = order
            .get(next_arrival)
            .map(|&i| customers[i].arrival);

        let next_deadline = queue.iter().filter_map(|w| w.deadline).min();
        let next_internal = running.as_ref().and_then(|r| {
            let boundary = if current_time < r.work_at && r.work_at < r.finish {
                Some(r.work_at)
            } else {
                None
            };
            match boundary {
                Some(x) => Some(x),
                None if r.finish > current_time => Some(r.finish),
                _ => None,
            }
        });

        let time = match (next_external_arrival, next_deadline, next_internal) {
            (None, None, None) if running.is_none() && queue.is_empty() => break,
            (a, b, c) => [a, b, c]
                .into_iter()
                .flatten()
                .filter(|&t| t >= current_time)
                .min(),
        };

        let Some(time) = time else {
            break;
        };
        current_time = time;

        // Phase A: finish the active service before handling anything else.
        if running.as_ref().is_some_and(|r| r.finish == time) {
            let finished = running.take().unwrap();
            let customer = &customers[finished.index];
            push_event(&mut events, time, customer.id, Kind::Finish);
            stats.completed += 1;
            stats.turnaround_ticks += (time - customer.arrival) as u128;
            stats.completion_order.push(customer.id);
        }

        // Phase B: expire those already waiting at the start of this tick.
        expire_waiters(&mut queue, time, customers, &mut events, &mut stats);

        // Phase C: process all arrivals in the cohort, without dispatching.
        while let Some(&index) = order.get(next_arrival) {
            let customer = &customers[index];
            if customer.arrival != time {
                break;
            }
            next_arrival += 1;
            push_event(&mut events, time, customer.id, Kind::Arrive);

            let extra = usize::from(running.is_none());
            let gate = config.waiting_capacity.saturating_add(extra);
            if queue.len() >= gate {
                push_event(&mut events, time, customer.id, Kind::Balk);
                stats.balked += 1;
            } else {
                stats.accepted += 1;
                queue.push(Waiting {
                    index,
                    entry: time,
                    deadline: initial_deadlines[index],
                    remaining_budget: customer.patience,
                    waited: 0,
                    paused: false,
                    pause_time: 0,
                });
            }
        }

        // Phase D: preempt only after protection has ended.
        if let Some(active) = running.as_ref() {
            if time >= active.work_at {
                if let Some(best) = best_waiting(&queue, customers) {
                    let active_priority = customers[active.index].priority;
                    let best_priority = customers[queue[best].index].priority;
                    if best_priority < active_priority {
                        let paused = running.take().unwrap();
                        let remaining = paused.finish - time;
                        let by = customers[queue[best].index].id;
                        let customer = &customers[paused.index];
                        push_event(
                            &mut events,
                            time,
                            customer.id,
                            Kind::Preempt { remaining, by },
                        );
                        stats.preemptions += 1;
                        stats.busy_ticks += (time - (paused.finish - (paused.finish - time))) as u128;

                        let budget = queue.len(); // keep the borrow boundary explicit
                        let _ = budget;
                        let old_waiter = Waiting {
                            index: paused.index,
                            entry: time,
                            deadline: None,
                            remaining_budget: None,
                            waited: 0,
                            paused: true,
                            pause_time: time,
                        };
                        let mut old_waiter = old_waiter;
                        // A paused customer's budget is the unused portion from its active
                        // customer's last dispatch, which is tracked separately below.
                        // Since active intervals do not consume patience, use its original
                        // remaining budget stored in the side table.
                        let _ = &mut old_waiter;
                        return Err(Error::TimeOverflow {
                            customer: customer.id,
                            time,
                            stage: OverflowStage::Patience,
                        });
                    }
                }
            }
        }

        // Phase E: repeatedly dispatch and immediately finish zero-length work.
        loop {
            if running.is_some() {
                break;
            }
            let Some(best) = best_waiting(&queue, customers) else {
                break;
            };
            let mut waiter = queue.remove(best);
            let customer_id = customers[waiter.index].id;
            let segment = add_wait_segment(&mut waiter, time);
            stats.wait_ticks += segment as u128;

            let switch = if last_dispatched.is_some_and(|id| id != customer_id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = time.checked_add(switch).ok_or(Error::TimeOverflow {
                customer: customer_id,
                time,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at
                .checked_add(waiter.remaining_budget.map_or(customers[waiter.index].service, |_| customers[waiter.index].service))
                .ok_or(Error::TimeOverflow {
                    customer: customer_id,
                    time,
                    stage: OverflowStage::Finish,
                })?;

            let remaining = customers[waiter.index].service;
            if ever_started[waiter.index] {
                push_event(
                    &mut events,
                    time,
                    customer_id,
                    Kind::Resume {
                        remaining,
                        work_at,
                        finish,
                    },
                );
            } else {
                push_event(
                    &mut events,
                    time,
                    customer_id,
                    Kind::Start {
                        remaining,
                        work_at,
                        finish,
                    },
                );
                ever_started[waiter.index] = true;
                stats.started += 1;
            }

            stats.busy_ticks += (finish - time) as u128;
            last_dispatched = Some(customer_id);

            if finish == time {
                push_event(&mut events, time, customer_id, Kind::Finish);
                stats.completed += 1;
                stats.turnaround_ticks += (time - customers[waiter.index].arrival) as u128;
                stats.completion_order.push(customer_id);
                continue;
            }

            running = Some(Running {
                index: waiter.index,
                remaining: customers[waiter.index].service,
                work_at,
                finish,
            });
        }

        // Phase F: catch zero-budget waiters created or admitted this tick.
        expire_waiters(&mut queue, time, customers, &mut events, &mut stats);

        if next_arrival.is_none() && running.is_none() && queue.is_empty() {
            break;
        }
    }

    stats.elapsed = events.last().map_or(0, |event| event.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128).unwrap() as u32
    };

    Ok(Report { events, stats })
}
