#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Customer {
    pub id: u32,
    pub arrival: u64,
    pub service: u64,
    pub priority: u8,
    pub patience: Option<u64>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Config {
    pub switch_cost: u64,
    pub waiting_capacity: usize,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Kind {
    Arrive,
    Balk,
    Renege { waited: u64 },
    Start { remaining: u64, work_at: u64, finish: u64 },
    Resume { remaining: u64, work_at: u64, finish: u64 },
    Preempt { remaining: u64, by: u32 },
    Finish,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum OverflowStage {
    WorkAt,
    Finish,
    Patience,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    CapacityTooLarge { capacity: usize },
    TooManyCustomers { count: usize },
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    DeadlineOverflow { index: usize, id: u32 },
    TimeOverflow { customer: u32, time: u64, stage: OverflowStage },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Stats {
    pub accepted: usize,
    pub balked: usize,
    pub reneged: usize,
    pub completed: usize,
    pub started: usize,
    pub preemptions: usize,
    pub elapsed: u64,
    pub busy_ticks: u128,
    pub wait_ticks: u128,
    pub turnaround_ticks: u128,
    pub mean_wait_milli: u128,
    pub mean_turnaround_milli: u128,
    pub utilization_bps: u32,
    pub completion_order: Vec<u32>,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Report {
    pub events: Vec<Event>,
    pub stats: Stats,
}

pub fn rounded_ratio(numerator: u128, denominator: u128) -> Option<u128> {
    if denominator == 0 {
        return None;
    }

    let quotient = numerator / denominator;
    let remainder = numerator % denominator;
    let half = denominator / 2;

    let round_up = if denominator % 2 == 0 {
        remainder > half || (remainder == half && quotient % 2 == 1)
    } else {
        remainder > half
    };

    Some(quotient + u128::from(round_up))
}

#[derive(Clone)]
struct CustomerState {
    customer: Customer,
    deadline: Option<u64>,
    remaining: u64,
    waiting_since: u64,
    waited: u64,
    pause_time: Option<u64>,
    started: bool,
    accepted: bool,
    done: bool,
}

#[derive(Clone)]
struct Active {
    index: usize,
    work_at: u64,
    finish: u64,
    interval_start: u64,
}

fn event(events: &mut Vec<Event>, time: u64, id: u32, kind: Kind) {
    events.push(Event {
        time,
        customer: id,
        kind,
    });
}

fn best_waiter(waiting: &[usize], states: &[CustomerState]) -> Option<usize> {
    waiting.iter().copied().min_by(|&a, &b| {
        let sa = &states[a];
        let sb = &states[b];
        let ca = &sa.customer;
        let cb = &sb.customer;

        ca.priority
            .cmp(&cb.priority)
            .then_with(|| match (sa.pause_time, sb.pause_time) {
                (Some(pa), Some(pb)) => pa.cmp(&pb).then_with(|| ca.id.cmp(&cb.id)),
                (Some(_), None) => std::cmp::Ordering::Less,
                (None, Some(_)) => std::cmp::Ordering::Greater,
                (None, None) => ca
                    .arrival
                    .cmp(&cb.arrival)
                    .then_with(|| cb.id.cmp(&ca.id)),
            })
    })
}

fn consume_wait(state: &mut CustomerState, now: u64, wait_ticks: &mut u128) {
    let segment = now - state.waiting_since;
    state.waited = state.waited.saturating_add(segment);
    *wait_ticks += segment as u128;
    if let Some(deadline) = state.deadline {
        state.deadline = Some(deadline.saturating_sub(segment));
    }
}

fn expire_waiters(
    now: u64,
    waiting: &mut Vec<usize>,
    states: &mut [CustomerState],
    events: &mut Vec<Event>,
    stats: &mut Stats,
    wait_ticks: &mut u128,
) {
    let mut expired: Vec<usize> = waiting
        .iter()
        .copied()
        .filter(|&i| states[i].deadline.is_some_and(|d| d <= now))
        .collect();
    expired.sort_by_key(|&i| (states[i].deadline.unwrap(), states[i].customer.id));

    for i in expired {
        if let Some(pos) = waiting.iter().position(|&x| x == i) {
            waiting.remove(pos);
            let state = &mut states[i];
            consume_wait(state, now, wait_ticks);
            state.done = true;
            stats.reneged += 1;
            event(
                events,
                now,
                state.customer.id,
                Kind::Renege {
                    waited: state.waited,
                },
            );
        }
    }
}

pub fn simulate(customers: &[Customer], config: &Config) -> Result<Report, Error> {
    if config.waiting_capacity > 4096 {
        return Err(Error::CapacityTooLarge {
            capacity: config.waiting_capacity,
        });
    }
    if customers.len() > 4096 {
        return Err(Error::TooManyCustomers {
            count: customers.len(),
        });
    }

    let mut ids = Vec::with_capacity(customers.len());
    for (index, c) in customers.iter().enumerate() {
        if c.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if ids.contains(&c.id) {
            return Err(Error::DuplicateId { index, id: c.id });
        }
        ids.push(c.id);
        if c.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: c.priority,
            });
        }
        if let Some(p) = c.patience {
            if c.arrival.checked_add(p).is_none() {
                return Err(Error::DeadlineOverflow {
                    index,
                    id: c.id,
                });
            }
        }
    }

    let mut states: Vec<CustomerState> = customers
        .iter()
        .cloned()
        .map(|customer| CustomerState {
            remaining: customer.service,
            deadline: customer
                .patience
                .and_then(|p| customer.arrival.checked_add(p)),
            waiting_since: customer.arrival,
            waited: 0,
            pause_time: None,
            started: false,
            accepted: false,
            done: false,
            customer,
        })
        .collect();

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        let ca = &states[a].customer;
        let cb = &states[b].customer;
        ca.arrival
            .cmp(&cb.arrival)
            .then_with(|| ca.priority.cmp(&cb.priority))
            .then_with(|| cb.id.cmp(&ca.id))
    });

    let mut events = Vec::new();
    let mut stats = Stats {
        accepted: 0,
        balked: 0,
        reneged: 0,
        completed: 0,
        started: 0,
        preemptions: 0,
        elapsed: 0,
        busy_ticks: 0,
        wait_ticks: 0,
        turnaround_ticks: 0,
        mean_wait_milli: 0,
        mean_turnaround_milli: 0,
        utilization_bps: 0,
        completion_order: Vec::new(),
    };
    let mut waiting: Vec<usize> = Vec::new();
    let mut active: Option<Active> = None;
    let mut arrival_pos = 0;
    let mut now = 0u64;
    let mut last_dispatched: Option<u32> = None;

    loop {
        if arrival_pos == arrivals.len()
            && active.is_none()
            && waiting.is_empty()
        {
            break;
        }

        let next_arrival = arrivals
            .get(arrival_pos)
            .map(|&i| states[i].customer.arrival);
        let mut next_time = next_arrival;

        if let Some(a) = &active {
            for t in [Some(a.work_at), Some(a.finish)].into_iter().flatten() {
                if t >= now && next_time.map_or(true, |n| t < n) {
                    next_time = Some(t);
                }
            }
        }
        for &i in &waiting {
            if let Some(d) = states[i].deadline {
                if d >= now && next_time.map_or(true, |n| d < n) {
                    next_time = Some(d);
                }
            }
        }

        let Some(t) = next_time else { break };
        now = t;

        if let Some(a) = active.clone() {
            if a.finish == now {
                let s = &mut states[a.index];
                stats.busy_ticks += (now - a.interval_start) as u128;
                s.done = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - s.customer.arrival) as u128;
                stats.completion_order.push(s.customer.id);
                event(&mut events, now, s.customer.id, Kind::Finish);
                active = None;
            }
        }

        expire_waiters(
            now,
            &mut waiting,
            &mut states,
            &mut events,
            &mut stats,
            &mut stats.wait_ticks,
        );

        while arrival_pos < arrivals.len()
            && states[arrivals[arrival_pos]].customer.arrival == now
        {
            let i = arrivals[arrival_pos];
            arrival_pos += 1;
            let id = states[i].customer.id;
            event(&mut events, now, id, Kind::Arrive);

            let limit = config.waiting_capacity + usize::from(active.is_none());
            if waiting.len() >= limit {
                stats.balked += 1;
                states[i].done = true;
                event(&mut events, now, id, Kind::Balk);
            } else {
                stats.accepted += 1;
                states[i].accepted = true;
                states[i].waiting_since = now;
                waiting.push(i);
            }
        }

        if let Some(a) = active.clone() {
            if now >= a.work_at {
                if let Some(w) = best_waiter(&waiting, &states) {
                    if states[w].customer.priority < states[a.index].customer.priority {
                        let remaining = a.finish - now;
                        let by = states[w].customer.id;
                        let paused_id = states[a.index].customer.id;
                        event(
                            &mut events,
                            now,
                            paused_id,
                            Kind::Preempt { remaining, by },
                        );
                        stats.busy_ticks += (now - a.interval_start) as u128;
                        stats.preemptions += 1;
                        active = None;

                        let state = &mut states[a.index];
                        state.remaining = remaining;
                        state.waiting_since = now;
                        state.pause_time = Some(now);
                        if let Some(budget) = state.customer.patience {
                            let spent = state.waited;
                            let remaining_budget = budget.saturating_sub(spent);
                            state.deadline = Some(now.checked_add(remaining_budget).ok_or(
                                Error::TimeOverflow {
                                    customer: paused_id,
                                    time: now,
                                    stage: OverflowStage::Patience,
                                },
                            )?);
                        }
                        waiting.push(a.index);
                    }
                }
            }
        }

        loop {
            if active.is_some() {
                break;
            }
            let Some(i) = best_waiter(&waiting, &states) else {
                break;
            };
            waiting.retain(|&x| x != i);

            let id = states[i].customer.id;
            let was_started = states[i].started;
            let remaining = states[i].remaining;
            let cost = if last_dispatched.is_some_and(|previous| previous != id) {
                config.switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::WorkAt,
            })?;
            let finish = work_at.checked_add(remaining).ok_or(Error::TimeOverflow {
                customer: id,
                time: now,
                stage: OverflowStage::Finish,
            })?;

            consume_wait(&mut states[i], now, &mut stats.wait_ticks);
            states[i].deadline = states[i]
                .customer
                .patience
                .map(|p| {
                    let remaining_budget = p.saturating_sub(states[i].waited);
                    now.saturating_add(remaining_budget)
                });
            if !was_started {
                stats.started += 1;
                states[i].started = true;
            }
            last_dispatched = Some(id);
            event(
                &mut events,
                now,
                id,
                if was_started {
                    Kind::Resume {
                        remaining,
                        work_at,
                        finish,
                    }
                } else {
                    Kind::Start {
                        remaining,
                        work_at,
                        finish,
                    }
                },
            );

            if finish == now {
                states[i].done = true;
                stats.completed += 1;
                stats.turnaround_ticks += (now - states[i].customer.arrival) as u128;
                stats.completion_order.push(id);
                event(&mut events, now, id, Kind::Finish);
                continue;
            }

            active = Some(Active {
                index: i,
                work_at,
                finish,
                interval_start: now,
            });
        }

        expire_waiters(
            now,
            &mut waiting,
            &mut states,
            &mut events,
            &mut stats,
            &mut stats.wait_ticks,
        );

        if now == u64::MAX {
            break;
        }
    }

    stats.elapsed = events.last().map_or(0, |e| e.time);
    stats.mean_wait_milli = if stats.accepted == 0 {
        0
    } else {
        rounded_ratio(stats.wait_ticks * 1000, stats.accepted as u128).unwrap()
    };
    stats.mean_turnaround_milli = if stats.completed == 0 {
        0
    } else {
        rounded_ratio(stats.turnaround_ticks * 1000, stats.completed as u128).unwrap()
    };
    stats.utilization_bps = if stats.elapsed == 0 {
        0
    } else {
        rounded_ratio(stats.busy_ticks * 10000, stats.elapsed as u128)
            .unwrap()
            .min(u32::MAX as u128) as u32
    };

    Ok(Report { events, stats })
}
