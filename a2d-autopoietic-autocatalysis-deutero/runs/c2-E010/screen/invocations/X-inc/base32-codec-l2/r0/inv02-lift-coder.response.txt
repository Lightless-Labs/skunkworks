#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Alphabet {
    Standard,
    Hex,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Padding {
    Required,
    Omitted,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum DecodeError {
    InvalidByte { offset: usize, byte: u8 },
    Padding { offset: usize },
    InvalidLength { offset: usize },
    TrailingBits { offset: usize },
}

fn alphabet_bytes(alphabet: Alphabet) -> &'static [u8; 32] {
    match alphabet {
        Alphabet::Standard => b"ABCDEFGHIJKLMNOPQRSTUVWXYZ234567",
        Alphabet::Hex => b"0123456789ABCDEFGHIJKLMNOPQRSTUV",
    }
}

fn value(byte: u8, alphabet: &[u8; 32]) -> Option<u8> {
    alphabet.iter().position(|&b| b == byte).map(|v| v as u8)
}

pub fn encode(input: &[u8], alphabet: Alphabet, padding: Padding) -> String {
    let symbols = alphabet_bytes(alphabet);
    let mut output = String::new();
    let mut accumulator: u32 = 0;
    let mut bits = 0usize;

    for &byte in input {
        accumulator = (accumulator << 8) | u32::from(byte);
        bits += 8;
        while bits >= 5 {
            bits -= 5;
            let index = ((accumulator >> bits) & 31) as usize;
            output.push(symbols[index] as char);
        }
    }

    if bits > 0 {
        let index = ((accumulator << (5 - bits)) & 31) as usize;
        output.push(symbols[index] as char);
    }

    if padding == Padding::Required {
        while output.len() % 8 != 0 {
            output.push('=');
        }
    }

    output
}

pub fn decode(
    input: &str,
    alphabet: Alphabet,
    padding: Padding,
) -> Result<Vec<u8>, DecodeError> {
    let bytes = input.as_bytes();
    let n = bytes.len();
    let symbols = alphabet_bytes(alphabet);

    // Pass 1: validate every byte lexically.
    for (offset, &byte) in bytes.iter().enumerate() {
        if byte != b'=' && value(byte, symbols).is_none() {
            return Err(DecodeError::InvalidByte { offset, byte });
        }
    }

    let p = bytes.iter().position(|&byte| byte == b'=').unwrap_or(n);

    // Pass 2: validate padding mode and placement.
    match padding {
        Padding::Omitted if p < n => return Err(DecodeError::Padding { offset: p }),
        Padding::Required if p < n => {
            if let Some(offset) = bytes[p..]
                .iter()
                .position(|&byte| byte != b'=')
                .map(|relative| p + relative)
            {
                return Err(DecodeError::Padding { offset });
            }
        }
        _ => {}
    }

    let data_len = p;
    let residue = data_len % 8;

    // Pass 3: validate data length.
    if !matches!(residue, 0 | 2 | 4 | 5 | 7) {
        return Err(DecodeError::InvalidLength { offset: n });
    }

    // Pass 4: validate required padding shape.
    if padding == Padding::Required {
        let expected = match residue {
            0 => 0,
            2 => 6,
            4 => 4,
            5 => 3,
            7 => 1,
            _ => unreachable!(),
        };
        if n - p != expected || n % 8 != 0 {
            return Err(DecodeError::Padding { offset: p });
        }
    }

    // Pass 5: validate unused trailing bits.
    let unused_bits = match residue {
        0 => 0,
        2 => 2,
        4 => 4,
        5 => 1,
        7 => 3,
        _ => unreachable!(),
    };
    if unused_bits > 0 {
        let last = value(bytes[data_len - 1], symbols).unwrap();
        if last & ((1u8 << unused_bits) - 1) != 0 {
            return Err(DecodeError::TrailingBits {
                offset: data_len - 1,
            });
        }
    }

    let mut output = Vec::with_capacity(data_len * 5 / 8);
    let mut accumulator: u32 = 0;
    let mut bits = 0usize;

    for &byte in &bytes[..data_len] {
        let val = value(byte, symbols).unwrap();
        accumulator = (accumulator << 5) | u32::from(val);
        bits += 5;
        if bits >= 8 {
            bits -= 8;
            output.push(((accumulator >> bits) & 0xff) as u8);
        }
    }

    Ok(output)
}
