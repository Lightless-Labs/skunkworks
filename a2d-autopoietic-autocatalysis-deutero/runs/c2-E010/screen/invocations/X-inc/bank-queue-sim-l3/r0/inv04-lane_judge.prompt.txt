#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {

    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },

struct Job {
    customer: Customer,
    remaining: u64,
    started: bool,
    pause_time: Option<u64>,
}

struct Running {

    work_at: u64,
    finish: u64,
}

fn waiter_cmp(a: &Job, b: &Job) -> std::cmp::Ordering {
    use std::cmp::Ordering;

    a.customer
        .priority
        .cmp(&b.customer.priority)
        .then_with(|| match (a.pause_time, b.pause_time) {
            (Some(at), Some(bt)) => at.cmp(&bt).then_with(|| a.customer.id.cmp(&b.customer.id)),
            (Some(_), None) => Ordering::Less,
            (None, Some(_)) => Ordering::Greater,
            (None, None) => a
                .customer
                .arrival
                .cmp(&b.customer.arrival)
                .then_with(|| b.customer.id.cmp(&a.customer.id)),
        })
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut ids = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !ids.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<Customer> = customers.to_vec();
    arrivals.sort_by(|a, b| {
        a.arrival
            .cmp(&b.arrival)
            .then_with(|| a.priority.cmp(&b.priority))
            .then_with(|| b.id.cmp(&a.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Job> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0usize;
    let mut last_dispatched: Option<u32> = None;

    loop {
        if running.is_none() && waiting.is_empty() && next_arrival == arrivals.len() {
            break;
        }

        let mut now = u64::MAX;
        let mut has_tick = false;

        if next_arrival < arrivals.len() {
            now = arrivals[next_arrival].arrival;
            has_tick = true;
        }
        if let Some(active) = &running {
            if !has_tick || active.finish < now {
                now = active.finish;
                has_tick = true;
            }
            if active.work_at > now || !has_tick {
                if !has_tick || active.work_at < now {
                    now = active.work_at;
                    has_tick = true;
                }
            }
        }
        if !has_tick {
            break;
        }

        if running
            .as_ref()
            .is_some_and(|active| active.finish == now)
        {
            let active = running.take().unwrap();
            trace.push(Event {
                time: now,
                customer: active.job.customer.id,
                kind: Kind::Finish,
            });
        }

        while next_arrival < arrivals.len() && arrivals[next_arrival].arrival == now {
            let customer = arrivals[next_arrival].clone();
            trace.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Job {
                remaining: customer.service,
                customer,
                started: false,
                pause_time: None,
            });
            next_arrival += 1;
        }

        if let Some(active) = &running {
            if now >= active.work_at {
                let best = waiting
                    .iter()
                    .min_by(|a, b| waiter_cmp(a, b));
                if let Some(best) = best {
                    if best.customer.priority < active.job.customer.priority {
                        let by = best.customer.id;
                        let mut active = running.take().unwrap();
                        let remaining = active.finish - now;
                        trace.push(Event {
                            time: now,
                            customer: active.job.customer.id,
                            kind: Kind::Preempt { remaining, by },
                        });
                        active.job.remaining = remaining;
                        active.job.pause_time = Some(now);
                        waiting.push(active.job);
                    }
                }
            }
        }

        while running.is_none() && !waiting.is_empty() {
            let best_index = (0..waiting.len())
                .min_by(|&a, &b| waiter_cmp(&waiting[a], &waiting[b]))
                .unwrap();
            let job = waiting.remove(best_index);
            let customer_id = job.customer.id;
            let cost = if last_dispatched.is_some_and(|id| id != customer_id) {
                switch_cost
            } else {
                0
            };
            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: customer_id,
                time: now,
            })?;
            let finish = work_at.checked_add(job.remaining).ok_or(Error::TimeOverflow {
                customer: customer_id,
                time: now,
            })?;

            let kind = if job.started {
                Kind::Resume {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: job.remaining,
                    work_at,
                    finish,
                }
            };
            trace.push(Event {
                time: now,
                customer: customer_id,
                kind,
            });
            last_dispatched = Some(customer_id);

            if finish == now {
                trace.push(Event {
                    time: now,
                    customer: customer_id,
                    kind: Kind::Finish,
                });
            } else {
                let mut job = job;
                job.started = true;
                job.pause_time = None;
                running = Some(Running {
                    job,
                    work_at,
                    finish,
                });
            }
        }
    }

    Ok(trace)
}

    Finish,

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Event {
    pub time: u64,
    pub customer: u32,
    pub kind: Kind,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    ZeroId { index: usize },
    DuplicateId { index: usize, id: u32 },
    BadPriority { index: usize, value: u8 },
    TimeOverflow { customer: u32, time: u64 },
}

#[derive(Clone)]
struct Waiting {
    index: usize,
    paused_at: Option<u64>,
    remaining: u64,
}

#[derive(Clone)]
struct Running {
    index: usize,
    remaining: u64,
    work_at: u64,
    finish: u64,
}

fn waiter_before(a: &Waiting, b: &Waiting, customers: &[Customer]) -> bool {
    let ca = &customers[a.index];
    let cb = &customers[b.index];

    if ca.priority != cb.priority {
        return ca.priority < cb.priority;
    }

    match (a.paused_at, b.paused_at) {
        (Some(ta), Some(tb)) => (ta, ca.id) < (tb, cb.id),
        (Some(_), None) => true,
        (None, Some(_)) => false,
        (None, None) => {
            ca.arrival < cb.arrival || (ca.arrival == cb.arrival && ca.id > cb.id)
        }
    }
}

fn pop_best(waiting: &mut Vec<Waiting>, customers: &[Customer]) -> Option<Waiting> {
    let best = waiting
        .iter()
        .enumerate()
        .min_by(|(_, a), (_, b)| {
            if waiter_before(a, b, customers) {
                std::cmp::Ordering::Less
            } else if waiter_before(b, a, customers) {
                std::cmp::Ordering::Greater
            } else {
                std::cmp::Ordering::Equal
            }
        })
        .map(|(i, _)| i)?;
    Some(waiting.swap_remove(best))
}

pub fn simulate(customers: &[Customer], switch_cost: u64) -> Result<Vec<Event>, Error> {
    use std::collections::HashSet;

    let mut seen = HashSet::new();
    for (index, customer) in customers.iter().enumerate() {
        if customer.id == 0 {
            return Err(Error::ZeroId { index });
        }
        if !seen.insert(customer.id) {
            return Err(Error::DuplicateId {
                index,
                id: customer.id,
            });
        }
        if customer.priority > 3 {
            return Err(Error::BadPriority {
                index,
                value: customer.priority,
            });
        }
    }

    if customers.is_empty() {
        return Ok(Vec::new());
    }

    let mut arrivals: Vec<usize> = (0..customers.len()).collect();
    arrivals.sort_by(|&a, &b| {
        let ca = &customers[a];
        let cb = &customers[b];
        ca.arrival
            .cmp(&cb.arrival)
            .then_with(|| ca.priority.cmp(&cb.priority))
            .then_with(|| cb.id.cmp(&ca.id))
    });

    let mut trace = Vec::new();
    let mut waiting: Vec<Waiting> = Vec::new();
    let mut running: Option<Running> = None;
    let mut next_arrival = 0;
    let mut now = 0u64;
    let mut last_dispatched: Option<u32> = None;
    let mut started = vec![false; customers.len()];
    let mut finished_count = 0;

    while finished_count < customers.len() {
        let next_time = [
            arrivals.get(next_arrival).map(|&i| customers[i].arrival),
            running.as_ref().map(|r| r.finish),
            running.as_ref().map(|r| r.work_at).filter(|&t| t > now),
        ]
        .into_iter()
        .flatten()
        .filter(|&t| t >= now)
        .min();

        let Some(tick) = next_time else {
            break;
        };
        now = tick;

        if running.as_ref().is_some_and(|r| r.finish == now) {
            let r = running.take().unwrap();
            trace.push(Event {
                time: now,
                customer: customers[r.index].id,
                kind: Kind::Finish,
            });
            finished_count += 1;
        }

        while next_arrival < arrivals.len()
            && customers[arrivals[next_arrival]].arrival == now
        {
            let index = arrivals[next_arrival];
            let customer = &customers[index];
            trace.push(Event {
                time: now,
                customer: customer.id,
                kind: Kind::Arrive,
            });
            waiting.push(Waiting {
                index,
                paused_at: None,
                remaining: customer.service,
            });
            next_arrival += 1;
        }

        if let Some(r) = running.as_ref() {
            if now >= r.work_at {
                let best_priority = waiting
                    .iter()
                    .map(|w| customers[w.index].priority)
                    .min();
                if best_priority.is_some_and(|p| p < customers[r.index].priority) {
                    let best = pop_best(&mut waiting, customers).unwrap();
                    let by = customers[best.index].id;
                    let r = running.take().unwrap();
                    let remaining = r.finish - now;
                    trace.push(Event {
                        time: now,
                        customer: customers[r.index].id,
                        kind: Kind::Preempt { remaining, by },
                    });
                    waiting.push(Waiting {
                        index: r.index,
                        paused_at: Some(now),
                        remaining,
                    });
                    waiting.push(best);
                }
            }
        }

        loop {
            if running.is_some() {
                break;
            }
            let Some(waiter) = pop_best(&mut waiting, customers) else {
                break;
            };
            let customer = &customers[waiter.index];
            let cost = if last_dispatched == Some(customer.id) {
                0
            } else if last_dispatched.is_none() {
                0
            } else {
                switch_cost
            };

            let work_at = now.checked_add(cost).ok_or(Error::TimeOverflow {
                customer: customer.id,
                time: now,
            })?;
            let finish = work_at.checked_add(waiter.remaining).ok_or(Error::TimeOverflow {
                customer: customer.id,
                time: now,
            })?;

                Kind::Resume {
                    remaining: waiter.remaining,
                    work_at,
                    finish,
                }
            } else {
                Kind::Start {
                    remaining: waiter.remaining,
                    work_at,
                    finish,
                }

                time: now,
                customer: customer.id,
                kind,

                    time: now,
                    customer: customer.id,
                    kind: Kind::Finish,
                });

            running = Some(Running {
                index: waiter.index,
                remaining: waiter.remaining,
                work_at,
                finish,
            });
        }
    }

    Ok(trace)
}
