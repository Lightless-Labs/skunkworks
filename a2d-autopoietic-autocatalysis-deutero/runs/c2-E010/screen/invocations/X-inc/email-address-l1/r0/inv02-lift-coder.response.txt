#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Address {
    pub local: String,
    pub domain: String,
    pub quoted: bool,
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EmailError {
    EmptyInput,
    NonAscii { at: usize },
    MissingAt,
    MultipleAt { at: usize },
    EmptyLocal { at: usize },
    LocalTooLong { at: usize },
    LocalDot { at: usize },
    LocalChar { at: usize },
    EmptyDomain { at: usize },
    DomainTooLong { at: usize },
    EmptyLabel { at: usize },
    LabelTooLong { at: usize },
    LabelHyphen { at: usize },
    DomainChar { at: usize },
    DomainNeedsDot { at: usize },
    NumericTld { at: usize },
}

pub fn parse_email(input: &str) -> Result<Address, EmailError> {
    let bytes = input.as_bytes();

    if bytes.is_empty() {
        return Err(EmailError::EmptyInput);
    }

    if let Some(at) = bytes.iter().position(|&byte| byte >= 128) {
        return Err(EmailError::NonAscii { at });
    }

    let mut separators = bytes
        .iter()
        .enumerate()
        .filter_map(|(i, &byte)| (byte == b'@').then_some(i));

    let separator = match separators.next() {
        Some(at) => at,
        None => return Err(EmailError::MissingAt),
    };

    if let Some(at) = separators.next() {
        return Err(EmailError::MultipleAt { at });
    }

    if separator == 0 {
        return Err(EmailError::EmptyLocal { at: separator });
    }

    let local_bytes = &bytes[..separator];
    if local_bytes.len() > 64 {
        return Err(EmailError::LocalTooLong { at: 64 });
    }

    for (i, &byte) in local_bytes.iter().enumerate() {
        let at = i;
        if byte == b'.' {
            if i == 0 || i + 1 == local_bytes.len() || local_bytes[i - 1] == b'.' {
                return Err(EmailError::LocalDot { at });
            }
        } else if !byte.is_ascii_alphanumeric()
            && !b"!#$%&'*+-/=?^_`{|}~".contains(&byte)
        {
            return Err(EmailError::LocalChar { at });
        }
    }

    let domain_start = separator + 1;
    let raw_domain = &bytes[domain_start..];
    let root_dot = raw_domain.last().copied() == Some(b'.');
    let domain_len = raw_domain.len() - usize::from(root_dot);

    if domain_len == 0 {
        return Err(EmailError::EmptyDomain { at: domain_start });
    }

    if domain_len > 253 {
        return Err(EmailError::DomainTooLong {
            at: domain_start + 253,
        });
    }

    let domain_bytes = &raw_domain[..domain_len];
    let mut labels: Vec<(usize, &[u8])> = Vec::new();
    let mut label_start = 0;

    for (i, &byte) in domain_bytes.iter().enumerate() {
        if byte == b'.' {
            if i == label_start {
                return Err(EmailError::EmptyLabel {
                    at: domain_start + i,
                });
            }
            labels.push((label_start, &domain_bytes[label_start..i]));
            label_start = i + 1;
        }
    }

    if label_start == domain_len {
        let at = if root_dot {
            input.len() - 1
        } else {
            input.len()
        };
        return Err(EmailError::EmptyLabel { at });
    }

    labels.push((label_start, &domain_bytes[label_start..]));

    for &(offset, label) in &labels {
        if label.len() > 63 {
            return Err(EmailError::LabelTooLong {
                at: domain_start + offset + 63,
            });
        }

        if label[0] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + offset,
            });
        }
        if label[label.len() - 1] == b'-' {
            return Err(EmailError::LabelHyphen {
                at: domain_start + offset + label.len() - 1,
            });
        }

        for (i, &byte) in label.iter().enumerate() {
            if !byte.is_ascii_alphanumeric() && byte != b'-' {
                return Err(EmailError::DomainChar {
                    at: domain_start + offset + i,
                });
            }
        }
    }

    if labels.len() < 2 {
        return Err(EmailError::DomainNeedsDot { at: domain_start });
    }

    let (last_offset, last_label) = labels[labels.len() - 1];
    if last_label.iter().all(u8::is_ascii_digit) {
        return Err(EmailError::NumericTld {
            at: domain_start + last_offset,
        });
    }

    let local = String::from_utf8(local_bytes.to_vec()).expect("ASCII input");
    let domain = String::from_utf8(domain_bytes.to_vec())
        .expect("ASCII input")
        .to_ascii_lowercase();

    Ok(Address {
        local,
        domain,
        quoted: false,
    })
}
