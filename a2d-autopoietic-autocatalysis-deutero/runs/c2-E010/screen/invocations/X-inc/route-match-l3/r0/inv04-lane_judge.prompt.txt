    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

struct Route {
    id: usize,
    method: String,
    segments: Vec<Segment>,
    trailing_slash: bool,
    rank: Vec<u8>,
    has_wildcard: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;

        let (segments, trailing_slash) =
            parse_pattern(pattern).map_err(|at| Error::InvalidPattern { at })?;

        let has_wildcard = segments
            .iter()
            .any(|segment| matches!(segment, Segment::Wildcard(_)));

        let mut rank: Vec<u8> = segments
            .iter()
            .map(|segment| match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            })
            .collect();

        if !has_wildcard {
            rank.push(3);
        }

        let id = self.routes.len();
        self.routes.push(Route {
            id,
            method: method.to_string(),
            segments,
            trailing_slash,
            rank,
            has_wildcard,
        });

        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;

        let head_end = path
            .char_indices()
            .find_map(|(i, c)| if c == '?' || c == '#' { Some(i) } else { None })
            .unwrap_or(path.len());
        let head = &path[..head_end];

        let (request_segments, request_trailing_slash) =
            parse_request_path(head).map_err(|at| Error::InvalidPath { at })?;

        let mut candidates: Vec<(&Route, Vec<(String, String)>)> = Vec::new();

        for route in &self.routes {
            if self.policy == SlashPolicy::Strict
                && route.trailing_slash != request_trailing_slash
            {
                continue;
            }

            if let Some(params) = match_route(&route.segments, &request_segments) {
                candidates.push((route, params));
            }
        }

        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible: Vec<(&Route, Vec<(String, String)>)> = if method == "HEAD" {
            if candidates.iter().any(|(route, _)| route.method == "HEAD") {
                candidates
                    .into_iter()
                    .filter(|(route, _)| route.method == "HEAD")
                    .collect()
            } else {
                candidates
                    .into_iter()
                    .filter(|(route, _)| route.method == "GET")
                    .collect()
            }
        } else {
            candidates
                .into_iter()
                .filter(|(route, _)| route.method == method)
                .collect()
        };

        if eligible.is_empty() {
            let mut allowed: Vec<String> = self
                .routes
                .iter()
                .filter(|route| {
                    self.path_candidate(route, &request_segments, request_trailing_slash)
                })
                .map(|route| route.method.clone())
                .collect();

            if allowed.iter().any(|m| m == "GET") {
                allowed.push("HEAD".to_string());
            }
            allowed.sort();
            allowed.dedup();

            return Err(Error::MethodNotAllowed { allowed });
        }

        let (winner, params) = eligible
            .into_iter()
            .max_by(|(a, _), (b, _)| {
                a.rank
                    .cmp(&b.rank)
                    // Earlier registration wins ties.
                    .then_with(|| b.id.cmp(&a.id))
            })
            .expect("eligible candidates are nonempty");

        let redirect = if self.policy == SlashPolicy::Redirect
            && winner.trailing_slash != request_trailing_slash
        {
            if head == "/" {
                None
            } else if request_trailing_slash {
                Some(head[..head.len() - 1].to_string())
            } else {
                Some(format!("{head}/"))
            }
        } else {
            None
        };

        Ok(Match {
            route: winner.id,
            params,
            redirect,
        })
    }

    fn path_candidate(
        &self,
        route: &Route,
        request_segments: &[String],
        request_trailing_slash: bool,
    ) -> bool {
        if self.policy == SlashPolicy::Strict
            && route.trailing_slash != request_trailing_slash
        {
            return false;
        }

        match_route(&route.segments, request_segments).is_some()
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if !method.is_empty() && method.bytes().all(|b| b.is_ascii_uppercase()) {
        Ok(())
    } else {
        Err(Error::InvalidMethod)
    }

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), usize> {
    if !pattern.starts_with('/') {
        return Err(0);
    }

    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing_slash = pattern.ends_with('/');
    let end = if trailing_slash {
        pattern.len() - 1
    } else {
        pattern.len()
    };

    let mut segments = Vec::new();
    let mut seen_names = Vec::<String>::new();
    let mut start = 1;

    while start < end {
        let next = pattern[start..end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(end);

        if next == start {
            return Err(start);
        }

        let raw = &pattern[start..next];
        let first = raw.as_bytes()[0];

        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !valid_capture_name(name) || seen_names.iter().any(|existing| existing == name) {
                return Err(start);
            }
            if first == b'*' && next != end {
                return Err(start);
            }

            seen_names.push(name.to_string());
            if first == b':' {
                segments.push(Segment::Param(name.to_string()));
            } else {
                segments.push(Segment::Wildcard(name.to_string()));
            }
        } else {
            let decoded = decode_static_segment(raw, start)?;
            segments.push(Segment::Static(decoded));
        }

        if next == end {
            break;
        }
        start = next + 1;
    }

    if segments.is_empty() {
        return Err(1);
    }

    Ok((segments, trailing_slash))
}

fn parse_request_path(path: &str) -> Result<(Vec<String>, bool), usize> {
    if !path.starts_with('/') {
        return Err(0);
    }

    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing_slash = path.ends_with('/');
    let end = if trailing_slash {
        path.len() - 1
    } else {
        path.len()
    };

    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let next = path[start..end]
            .find('/')
            .map(|offset| start + offset)
            .unwrap_or(end);

        if next == start {
            return Err(start);
        }

        segments.push(decode_static_segment(&path[start..next], start)?);

        if next == end {
            break;
        }
        start = next + 1;
    }

    if segments.is_empty() {
        return Err(1);
    }

    Ok((segments, trailing_slash))
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_static_segment(raw: &str, base: usize) -> Result<String, usize> {
    let bytes = raw.as_bytes();
    let mut decoded = Vec::with_capacity(bytes.len());
    let mut i = 0;

    while i < bytes.len() {
        let byte = bytes[i];

        if byte == b'%' {
            if i + 2 >= bytes.len() {
                return Err(base + i);
            }

            let hi = uppercase_hex(bytes[i + 1]).ok_or(base + i)?;
            let lo = uppercase_hex(bytes[i + 2]).ok_or(base + i)?;
            let value = (hi << 4) | lo;

            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(base + i);
            }

            decoded.push(value);
            i += 3;
        } else if is_unreserved(byte) {
            decoded.push(byte);
            i += 1;
        } else {
            return Err(base + i);
        }
    }

    Ok(String::from_utf8(decoded).expect("decoded segments contain ASCII only"))
}

fn uppercase_hex(byte: u8) -> Option<u8> {
    match byte {
        b'0'..=b'9' => Some(byte - b'0'),
        b'A'..=b'F' => Some(byte - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(byte: u8) -> bool {
    byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'~' | b'-')
}

fn match_route(pattern: &[Segment], request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_index = pattern
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));

    let fixed_len = wildcard_index.unwrap_or(pattern.len());
    if wildcard_index.is_none() && request.len() != pattern.len() {
        return None;
    }
    if wildcard_index.is_some() && request.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();

    for i in 0..fixed_len {
        match &pattern[i] {
            Segment::Static(value) if value != &request[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), request[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }

    if let Some(index) = wildcard_index {
        if let Segment::Wildcard(name) = &pattern[index] {
            let value = request[index..].join("/");
            params.push((name.clone(), value));
        }
    }

    Some(params)
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool, Vec<u8>), Error> {
    let bytes = pattern.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false, vec![3]));
    }

    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut names = Vec::<String>::new();
    let mut rank = Vec::new();
    let mut start = 1;

    while start < end {
        let stop = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|i| start + i)
            .unwrap_or(end);
        if stop == start {
            return Err(Error::InvalidPattern { at: start });
        }

        if raw.starts_with(':') || raw.starts_with('*') {
            let wildcard = raw.starts_with('*');
            let name = &raw[1..];
            if !valid_name(name) || names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: start });
            }
            if wildcard && stop != end {
                return Err(Error::InvalidPattern { at: start });
            }
            names.push(name.to_owned());
            if wildcard {
                segments.push(Segment::Wildcard(name.to_owned()));
                rank.push(0);
            } else {
                segments.push(Segment::Param(name.to_owned()));
                rank.push(1);
            }
        } else {
            let decoded = decode_segment(raw, start, true)?;
            segments.push(Segment::Static(decoded));
            rank.push(2);
        }
        if stop == end {
            break;
        }
        start = stop + 1;
    }

    if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
        rank.push(3);
    }
    Ok((segments, trailing, rank))
}

fn parse_path(path: &str) -> Result<(Vec<String>, bool), Error> {
    let bytes = path.as_bytes();
    if bytes.first() != Some(&b'/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok((Vec::new(), false));
    }

    let trailing = bytes.last() == Some(&b'/');
    let end = if trailing { bytes.len() - 1 } else { bytes.len() };
    let mut segments = Vec::new();
    let mut start = 1;

    while start < end {
        let stop = bytes[start..end]
            .iter()
            .position(|&b| b == b'/')
            .map(|i| start + i)
            .unwrap_or(end);
        if stop == start {
            return Err(Error::InvalidPath { at: start });
        }
        segments.push(
            decode_segment(&path[start..stop], start, false)
                .map_err(|e| match e {
                    Error::InvalidPattern { at } => Error::InvalidPath { at },
                    other => other,
                })?,
        );
        if stop == end {
            break;
        }
        start = stop + 1;
    }
    Ok((segments, trailing))
}

fn valid_name(name: &str) -> bool {
    let b = name.as_bytes();
    !b.is_empty()
        && b[0].is_ascii_lowercase()
        && b[1..]
            .iter()
            .all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || *c == b'_')
}

fn decode_segment(raw: &str, offset: usize, pattern: bool) -> Result<String, Error> {
    let invalid = |at| {
        if pattern {
            Error::InvalidPattern { at }
        } else {
            Error::InvalidPath { at }
        }
    };
    let bytes = raw.as_bytes();
    let mut out = Vec::new();
    let mut i = 0;
    while i < bytes.len() {
        let b = bytes[i];
        if b == b'%' {
            if i + 2 >= bytes.len()
                || !bytes[i + 1].is_ascii_hexdigit()
                || !bytes[i + 2].is_ascii_hexdigit()
                || bytes[i + 1].is_ascii_lowercase()
                || bytes[i + 2].is_ascii_lowercase()
            {
                return Err(invalid(offset + i));
            }
            let value = (hex(bytes[i + 1]) << 4) | hex(bytes[i + 2]);
            if !(0x20..=0x7e).contains(&value)
                || value == b'/'
                || is_unreserved(value)
            {
                return Err(invalid(offset + i));
            }
            out.push(value);
            i += 3;
        } else {
            if !is_unreserved(b) {
                return Err(invalid(offset + i));
            }
            out.push(b);
            i += 1;
        }
    }
    Ok(String::from_utf8(out).unwrap())
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn hex(b: u8) -> u8 {
    match b {
        b'0'..=b'9' => b - b'0',
        b'A'..=b'F' => b - b'A' + 10,
        _ => 0,
    }
}

fn match_segments(pattern: &[Segment], path: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_at = pattern
        .iter()
        .position(|s| matches!(s, Segment::Wildcard(_)));
    let fixed_len = wildcard_at.unwrap_or(pattern.len());
    if wildcard_at.is_none() && path.len() != fixed_len {
        return None;
    }
    if wildcard_at.is_some() && path.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        match &pattern[i] {
            Segment::Static(value) if value != &path[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), path[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }
    if let Some(i) = wildcard_at {
        if let Segment::Wildcard(name) = &pattern[i] {
            params.push((name.clone(), path[i..].join("/")));
        }
    }
    Some(params)
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub enum Error {
    InvalidPattern { at: usize },
    InvalidPath { at: usize },
    NotFound,
    InvalidMethod,
    MethodNotAllowed { allowed: Vec<String> },
}

#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Match {
    pub route: usize,
    pub params: Vec<(String, String)>,
    pub redirect: Option<String>,
}

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum SlashPolicy {
    Strict,
    Ignore,
    Redirect,
}

#[derive(Debug, Clone)]
enum Segment {
    Static(String),
    Param(String),
    Wildcard(String),
}

#[derive(Debug, Clone)]
struct Route {
    method: String,
    segments: Vec<Segment>,
    trailing: bool,
    rank: Vec<u8>,
}

#[derive(Debug, Clone)]
struct ParsedPath {
    segments: Vec<String>,
    trailing: bool,
}

pub struct Router {
    policy: SlashPolicy,
    routes: Vec<Route>,
}

impl Router {
    pub fn new(policy: SlashPolicy) -> Self {
        Self {
            policy,
            routes: Vec::new(),
        }
    }

    pub fn add(&mut self, method: &str, pattern: &str) -> Result<usize, Error> {
        validate_method(method)?;
        let (segments, trailing) = parse_pattern(pattern)?;
        let mut rank = Vec::with_capacity(segments.len() + 1);
        for segment in &segments {
            rank.push(match segment {
                Segment::Static(_) => 2,
                Segment::Param(_) => 1,
                Segment::Wildcard(_) => 0,
            });
        }
        if !segments.iter().any(|s| matches!(s, Segment::Wildcard(_))) {
            rank.push(3);
        }
        let id = self.routes.len();
        self.routes.push(Route {
            method: method.to_owned(),
            segments,
            trailing,
            rank,
        });
        Ok(id)
    }

    pub fn resolve(&self, method: &str, path: &str) -> Result<Match, Error> {
        validate_method(method)?;
        let head_end = path
            .char_indices()
            .find_map(|(i, c)| matches!(c, '?' | '#').then_some(i))
            .unwrap_or(path.len());
        let head = &path[..head_end];
        let parsed = parse_path(head)?;

        let mut candidates: Vec<(usize, Vec<(String, String)>)> = Vec::new();
        for (idx, route) in self.routes.iter().enumerate() {
            if self.policy == SlashPolicy::Strict && route.trailing != parsed.trailing {
                continue;
            }
            if let Some(params) = match_route(route, &parsed.segments) {
                candidates.push((idx, params));
            }
        }
        if candidates.is_empty() {
            return Err(Error::NotFound);
        }

        let eligible_methods: Vec<&str> = if method == "HEAD" {
            if candidates
                .iter()
                .any(|(idx, _)| self.routes[*idx].method == "HEAD")
            {
                vec!["HEAD"]
            } else {
                vec!["GET"]
            }
        } else {
            vec![method]
        };

        let eligible: Vec<(usize, Vec<(String, String)>)> = candidates
            .iter()
            .filter(|(idx, _)| eligible_methods.contains(&self.routes[*idx].method.as_str()))
            .cloned()
            .collect();

        if eligible.is_empty() {
            let mut allowed = Vec::<String>::new();
            for (idx, _) in &candidates {
                let m = &self.routes[*idx].method;
                if !allowed.contains(m) {
                    allowed.push(m.clone());
                }
            }
            if allowed.iter().any(|m| m == "GET") && !allowed.iter().any(|m| m == "HEAD") {
                allowed.push("HEAD".to_owned());
            }
            allowed.sort();
            return Err(Error::MethodNotAllowed { allowed });
        }

        let mut winner = eligible[0].clone();
        for candidate in eligible.iter().skip(1) {
            if self.routes[candidate.0].rank > self.routes[winner.0].rank {
                winner = candidate.clone();
            }
        }

        let route = &self.routes[winner.0];
        let redirect = if self.policy == SlashPolicy::Redirect
            && !head.is_empty()
            && route.trailing != parsed.trailing
        {
            let mut target = head.to_owned();
            if route.trailing {
                target.push('/');
            } else if target.ends_with('/') {
                target.pop();
            }
            Some(target)
        } else {
            None
        };

        Ok(Match {
            route: winner.0,
            params: winner.1,
            redirect,
        })
    }
}

fn validate_method(method: &str) -> Result<(), Error> {
    if method.is_empty() || !method.bytes().all(|b| b.is_ascii_uppercase()) {
        Err(Error::InvalidMethod)
    } else {
        Ok(())
    }
}

fn parse_pattern(pattern: &str) -> Result<(Vec<Segment>, bool), Error> {
    if !pattern.starts_with('/') {
        return Err(Error::InvalidPattern { at: 0 });
    }
    if pattern == "/" {
        return Ok((Vec::new(), false));
    }

    let bytes = pattern.as_bytes();
    let mut raw_segments = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == start {
                return Err(Error::InvalidPattern { at: start });
            }
            raw_segments.push((start, i));
            start = i + 1;
        }
    }
    let trailing = bytes.last() == Some(&b'/');
    if trailing {
        if start == bytes.len() {
            // The slash terminates the preceding segment.
        } else {
            raw_segments.push((start, bytes.len()));
        }
    } else {
        if start == bytes.len() {
            return Err(Error::InvalidPattern { at: start });
        }
        raw_segments.push((start, bytes.len()));
    }

    let mut segments = Vec::with_capacity(raw_segments.len());
    let mut names = Vec::<String>::new();
    for (seg_start, end) in raw_segments {
        let raw = &pattern[seg_start..end];
        let first = raw.as_bytes()[0];
        if first == b':' || first == b'*' {
            let name = &raw[1..];
            if !valid_capture_name(name) {
                return Err(Error::InvalidPattern { at: seg_start });
            }
            if names.iter().any(|n| n == name) {
                return Err(Error::InvalidPattern { at: seg_start });
            }
            names.push(name.to_owned());
            if first == b'*' {
                // A wildcard must be final; check this before later segments are parsed.
                let is_final = seg_start + raw.len() == end
                    && !pattern[end..].contains('/');
                // Segment boundaries are represented by the next raw segment's start.
                let final_segment = end == pattern.len() || (trailing && end + 1 == pattern.len());
                let _ = is_final;
                if !final_segment {
                    return Err(Error::InvalidPattern { at: seg_start });
                }
                segments.push(Segment::Wildcard(name.to_owned()));
            } else {
                segments.push(Segment::Param(name.to_owned()));
            }
        } else {
            let decoded = decode_segment(raw.as_bytes(), seg_start)
                .map_err(|at| Error::InvalidPattern { at })?;
            segments.push(Segment::Static(decoded));
        }
    }
    Ok((segments, trailing))
}

fn parse_path(path: &str) -> Result<ParsedPath, Error> {
    if !path.starts_with('/') {
        return Err(Error::InvalidPath { at: 0 });
    }
    if path == "/" {
        return Ok(ParsedPath {
            segments: Vec::new(),
            trailing: false,
        });
    }

    let bytes = path.as_bytes();
    let trailing = bytes.last() == Some(&b'/');
    let mut decoded = Vec::new();
    let mut start = 1;
    for i in 1..bytes.len() {
        if bytes[i] == b'/' {
            if i == start {
                return Err(Error::InvalidPath { at: start });
            }
            decoded.push(
                decode_segment(&bytes[start..i], start)
                    .map_err(|at| Error::InvalidPath { at })?,
            );
            start = i + 1;
        }
    }
    if !trailing {
        if start == bytes.len() {
            return Err(Error::InvalidPath { at: start });
        }
        decoded.push(
            decode_segment(&bytes[start..], start).map_err(|at| Error::InvalidPath { at })?,
        );
    } else if start < bytes.len() {
        decoded.push(
            decode_segment(&bytes[start..bytes.len() - 1], start)
                .map_err(|at| Error::InvalidPath { at })?,
        );
    }
    Ok(ParsedPath {
        segments: decoded,
        trailing,
    })
}

fn valid_capture_name(name: &str) -> bool {
    let bytes = name.as_bytes();
    !bytes.is_empty()
        && bytes[0].is_ascii_lowercase()
        && bytes[1..]
            .iter()
            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || *b == b'_')
}

fn decode_segment(raw: &[u8], offset: usize) -> Result<String, usize> {
    let mut out = Vec::with_capacity(raw.len());
    let mut i = 0;
    while i < raw.len() {
        let b = raw[i];
        if b == b'%' {
            if i + 2 >= raw.len() {
                return Err(offset + i);
            }
            let hi = upper_hex(raw[i + 1]).ok_or(offset + i)?;
            let lo = upper_hex(raw[i + 2]).ok_or(offset + i)?;
            let decoded = hi * 16 + lo;
            if !(0x20..=0x7e).contains(&decoded)
                || decoded == b'/'
                || is_unreserved(decoded)
            {
                return Err(offset + i);
            }
            out.push(decoded);
            i += 3;
        } else if is_unreserved(b) {
            out.push(b);
            i += 1;
        } else {
            return Err(offset + i);
        }
    }
    // All output bytes are ASCII by construction.
    Ok(String::from_utf8(out).expect("decoded bytes are ASCII"))
}

fn upper_hex(b: u8) -> Option<u8> {
    match b {
        b'0'..=b'9' => Some(b - b'0'),
        b'A'..=b'F' => Some(b - b'A' + 10),
        _ => None,
    }
}

fn is_unreserved(b: u8) -> bool {
    b.is_ascii_alphanumeric() || matches!(b, b'.' | b'_' | b'~' | b'-')
}

fn match_route(route: &Route, request: &[String]) -> Option<Vec<(String, String)>> {
    let wildcard_at = route
        .segments
        .iter()
        .position(|segment| matches!(segment, Segment::Wildcard(_)));
    let fixed_len = wildcard_at.unwrap_or(route.segments.len());
    if wildcard_at.is_none() && request.len() != fixed_len {
        return None;
    }
    if wildcard_at.is_some() && request.len() < fixed_len {
        return None;
    }

    let mut params = Vec::new();
    for i in 0..fixed_len {
        match &route.segments[i] {
            Segment::Static(value) if value != &request[i] => return None,
            Segment::Static(_) => {}
            Segment::Param(name) => params.push((name.clone(), request[i].clone())),
            Segment::Wildcard(_) => unreachable!(),
        }
    }
    if let Some(i) = wildcard_at {
        let Segment::Wildcard(name) = &route.segments[i] else {
            unreachable!()
        };
        params.push((name.clone(), request[i..].join("/")));
    }
    Some(params)
}
